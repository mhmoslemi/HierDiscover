"""
Hybrid optimizer for the self‑convolution L∞ problem (improved).

The algorithm works on the probability simplex (∑a_i = 1).  It follows the
same overall structure as the reference implementation, but incorporates
several enhancements that make the search deeper in the same time budget:

* **No mandatory symmetrisation** – the optimum is not known to be symmetric,
  so we keep the full search space.
* **Longer Adam phase** – a larger base step and a slower λ‑decay give a
  more accurate soft‑max surrogate optimisation.
* **Line‑search sub‑gradient descent** – the step size is reduced until the
  exact L∞ objective improves, which makes the descent much more stable.
* **More aggressive annealing** – a higher start temperature, a slower
  exponential cooling schedule and more steps (controlled by the remaining
  time) provide a thorough polishing stage.
* **Dynamic time allocation** – the total budget (≈1000 s) is split among
  increasing sequence lengths.  Most of the time is spent on the largest
  length that still fits comfortably into memory (here 32768), allowing
  many optimisation iterations.

The routine returns a plain Python list of non‑negative numbers.  The
`evaluate_sequence` function will normalise/clamp the values automatically,
so we keep the sum exactly 1 throughout the optimisation.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for all lengths)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)   # remove tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing temperature λ > 0

    Returns
    -------
    grad: dense gradient of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()

    # cross‑correlation of the weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (active‑set sub‑gradient)
# ----------------------------------------------------------------------


def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a * a)[k] at the current a.
    Returns (grad, max_value).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)               # length 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max surrogate
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate.
    Returns the best vector seen (on the simplex) and its exact objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule
    lam_factor = 1.0          # start with λ ≈ max(conv)
    lam_decay = 0.99
    lam_min = 1e-12

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # sum(a) == 1 throughout
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam it={it} n={n} score={cur_score:.6f}")

        # update λ (smoothness)
        lam = max(max_b * lam_factor, lam_min)
        lam_factor = max(lam_factor * lam_decay, 1e-9)

        # gradient of the smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam parameter updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size – a bit more aggressive than before
        base_step = 1.2 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with back‑tracking line search on the true L∞‑objective.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    step0 = 0.8
    it = 0

    while time.time() < deadline:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        # initial step estimate
        step = step0 / math.sqrt(it + 1.0)

        # back‑tracking line search – ensure the max‑conv does not increase
        while step > 1e-12:
            a_trial = a - step * grad
            a_trial = _project_simplex(a_trial)
            conv_trial = _conv_self(a_trial)
            new_max = conv_trial.max()
            if new_max <= max_val + 1e-14:
                break
            step *= 0.5

        a = a_trial
        cur_score = 2.0 * n * new_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad it={it} n={n} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polishing)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.1,
                   end_temp: float = 1e-5,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Move a small ε from a source index to a destination index.
    More aggressive cooling and larger step count than the reference version.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # up to 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal it={it} n={n} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """A handful of plausible starting vectors derived from a base distribution."""
    variants = []
    n = base_seq.shape[0]

    # 1. warm‑start copy
    variants.append(base_seq.copy())

    # 2. positive sine shape
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals = np.maximum(sin_vals, 0.0)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 3. triangular shape
    left = np.linspace(1, n // 2, n // 2, endpoint=False)
    right = np.linspace(n // 2, 1, n - n // 2)
    tri = np.concatenate([left, right])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 4. arcsine‑like shape (1/√(x(1‑x)))
    t = (np.arange(n) + 0.5) / n
    arc = 1.0 / np.sqrt(t * (1.0 - t) + 1e-12)
    arc /= arc.sum()
    variants.append(arc)

    # 5. random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Re‑use a previously found construction if it exists
    # ------------------------------------------------------------------
    global_a = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                global_a = starter / s
    except Exception:
        global_a = None

    # Length schedule – start modest and grow geometrically.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths)   # avoid systematic bias if we stop early

    # For dynamic time split we keep a simple weighting based on log2(n)
    remaining_weight = sum(math.log2(L) for L in lengths)

    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a time slice proportional to log2(n)
        weight = math.log2(n)
        slice_time = (deadline - time.time()) * (weight / remaining_weight)
        # ensure a minimal budget for each length
        slice_time = max(slice_time, 8.0)
        slice_deadline = time.time() + slice_time
        remaining_weight -= weight

        # ------------------------------------------------------------------
        # initialise a0 (warm start / several variants)
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_candidates = _initial_variants(a0)

        # ------------------------------------------------------------------
        # Adam optimisation phase (several restarts)
        # ------------------------------------------------------------------
        best_local_a = None
        best_local_score = float('inf')

        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            # give each restart a fair share of the remaining slice time
            sub_deadline = time.time() + (slice_deadline - time.time()) / max(1, len(init_candidates) - init_candidates.index(start_vec))
            a_opt, score_opt = _run_adam_opt(start_vec, n, sub_deadline, start_time, symmetrize=False)
            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam local n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        # record any global improvement from the Adam phase
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Adam n={n} score={best_local_score:.6f}")

        # ------------------------------------------------------------------
        # Deterministic sub‑gradient refinement
        # ------------------------------------------------------------------
        if time.time() < deadline:
            sub_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.15)
            a_sub, score_sub = _run_subgradient(best_local_a, n, sub_deadline, start_time)
            if score_sub < best_score:
                best_score = score_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after subgrad n={n} score={score_sub:.6f}")

        # ------------------------------------------------------------------
        # Annealing polishing (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.20)
            a_ref, score_ref = _run_annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=0.1,
                end_temp=1e-5,
                steps=250000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # remember the current best distribution for up‑sampling
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list (evaluator will clamp / normalise as needed)
    return [float(x) for x in best_seq]
