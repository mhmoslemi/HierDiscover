def _run_coordinate_descent(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2 * n * _conv_self(a).max()
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        max_val = conv.max()
        k = conv.argmax()
        # compute pair products for indices overlapping at shift k
        # overlapping indices i: max(0, k-(n-1)) .. min(n-1, k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # compute products
        indices = np.arange(i_start, i_end + 1)
        partner = k - indices
        products = a[indices] * a[partner]
        # find i with max product
        max_idx = np.argmax(products)
        i_max = indices[max_idx]
        # pick j as index with smallest a value (excluding i_max)
        j_min = np.argmin(a)
        if a[j_min] > a[i_max]:
            # already minimal, skip
            continue
        # epsilon move: fraction up to 0.5 of a[i_max]
        eps = a[i_max] * 0.2  # 20%
        if eps <= 1e-12:
            continue
        a[i_max] -= eps
        a[j_min] += eps
        # re-normalize exact sum to 1
        s = a.sum()
        if abs(s - 1.0) > 1e-12:

        # compute new score
        score = 2 * n * _conv_self(a).max()
        if score < best_score:
            best_score = score

            print(f"[{time.time() - start_time:6.2f}s] coord improve: iter={iteration}, score={score:.6f}")

"""
Propose a step‑function (non‑negative vector) that minimises ``evaluate_sequence``.

The algorithm is a 4‑phase optimiser:

1. **Adam on a soft‑max approximation** – fast smooth optimisation that
   quickly lowers the log‑sum‑exp of the convolution.
2. **Projected sub‑gradient descent** – exact descent on the true
   max‑convolution, using Nesterov momentum.
3. **Greedy coordinate descent** – each iteration picks the product that
   contributes most to the current maximal convolution and moves a small
   fraction of mass away from it.  This directly attacks the worst term.
4. **Final simulated‑annealing walk** – tiny random mass‑transfer moves
   with a slow cooling schedule to fine‑tune the result.

All phases respect the time budget (default 1000 s) and always keep the
vector on the probability simplex (non‑negative, sum = 1), which is
invariant for the evaluation function.  The routine uses the global
``height_sequence_1`` as a warm‑start if it exists, otherwise random
initialisation.  The implementation is self‑contained and uses only
NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max (log‑sum‑exp) of the convolution."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))          # normalise for stability
    w = exps / exps.sum()
    rev_a = a[::-1]

    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max objective
# ----------------------------------------------------------------------
def _run_softmax_adam(a0: np.ndarray, n: int, deadline: float, start_time: float,
                      lr0: float = 0.02,
                      beta1: float = 0.9, beta2: float = 0.999,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    # temperature schedule for the soft‑max
    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] soft‑max‑adam improve: "
                  f"iter={it}, score={cur_score:.6f}")

        # temperature (smoothness) – decays geometrically
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + 1e-12)

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected, with momentum)
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.1,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: "
                  f"iter={it}, score={cur_score:.6f}")

        k = conv.argmax()                     # index attaining the max

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        if k < n:
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy coordinate descent – directly reduces the worst product
# ----------------------------------------------------------------------
def _run_coordinate_descent(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            max_iter: int = 10 ** 6) -> tuple[np.ndarray, float]:
    """At each step move a fraction of the mass from the pair that
    contributes most to the current maximum convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        k = conv.argmax()

        # indices that overlap at shift k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        partner = k - idx
        products = a[idx] * a[partner]

        # worst contributing pair
        worst = np.argmax(products)
        i_w = idx[worst]
        j_w = partner[worst]

        # pick a destination index with the smallest current mass
        dest = np.argmin(a)

        # move a modest fraction of the larger of the two involved entries
        eps = min(a[i_w], a[j_w]) * 0.15
        if eps <= 0.0:
            continue

        # split the epsilon equally between i_w and j_w, send to dest
        a[i_w] -= eps / 2.0
        a[j_w] -= eps / 2.0
        a[dest] += eps

        # re‑normalise (tiny drift may occur)
        s = a.sum()
        if abs(s - 1.0) > 1e-12:
            a /= s

        score = 2.0 * n * _conv_self(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] coord improve: "
                  f"iter={it}, score={score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the mass at i
        eps = a[i] * np.random.rand() * 0.20
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"iter={it}, score={score:.6f}")

    return best_seq, best_score

# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, performs a smooth Adam phase,
    an exact sub‑gradient phase, a greedy coordinate descent and finishes
    with a simulated‑annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness itself
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine problem size – use the size of a previous construction if

    try:
        starter = np.asarray(height_sequence_1, dtype=float)

                a0 = _upsample(starter, n)

    best_seq = a0.copy()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, time.time() + 0.3 * (deadline - start_time))

    # ------------------------------------------------------------------

    if s > 0.0:

    else:

    print(f"[{time.time() - start_time:6.2f}s] finished – final score {best_score:.6f}")
    return [float(x) for x in best_seq]
