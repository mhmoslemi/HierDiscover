def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math, random
    random.seed(seed); np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    def compute_score(a):
        conv = _conv_self(a)
        return 2.0 * len(a) * conv.max()

    # choose length
    N = 65536
    # initialize arcsine shape
    t = (np.arange(N) + 0.5) / N
    a = 1.0 / np.sqrt(t * (1 - t))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    a = a / a_sum

    best_a = a.copy()
    best_score = compute_score(a)

    iter = 0
    while time.time() < deadline:
        iter += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # compute subgradient
        grad, _ = _gradient_max_conv(a)
        # step size schedule
        eta = 0.5 / math.sqrt(iter)  # maybe 0.5 / sqrt(iter)
        # exponentiated update
        # avoid overflow: cap grad values? Use np.clip
        # compute log update
        log_a = np.log(a + 1e-30) - eta * grad
        # shift for numerical stability
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        # renorm
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0/a.shape[0])
        else:
            a /= a_sum

        # symmetrize: optional (make symmetric)
        if random.random() < 0.1:
            a = (a + a[::-1]) * 0.5

        # compute score
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start:.2f}s] iter={iter} improvement score={cur_score:.6f}")

    # final annealing if time left
    if time.time() < deadline:
        a = best_a.copy()
        # simple annealing
        anneal_iterations = int(2e5)
        start_temp = 5e-2
        end_temp = 1e-5
        for i in range(anneal_iterations):
            if time.time() >= deadline:
                break
            T = start_temp * (end_temp / start_temp) ** (i / anneal_iterations)
            # choose random source and dest
            src, dst = np.random.randint(0, N, size=2)
            if src == dst or a[src] <= 0:

            eps = a[src] * np.random.rand() * 0.1

def _exp_gradient_step(a, grad, eta):
    log_a = np.log(a + 1e-30) - eta * grad
    log_a -= np.max(log_a)  # for stability
    a_new = np.exp(log_a)
    a_new_sum = a_new.sum()
    if a_new_sum == 0.0:
        return np.full_like(a_new, 1.0 / a_new.size)

"""
Hybrid optimizer for the self‑convolution L∞ problem.

The algorithm works on the probability simplex (∑ a_i = 1) because the
objective returned by `evaluate_sequence` is scale‑invariant.
It proceeds in four stages:

1. **Multi‑scale warm‑starts** – start from a small distribution,
   then repeatedly up‑sample to a larger length.
2. **Exact sub‑gradient descent** – at each iteration we compute a
   sub‑gradient of the max‑convolution (active‑set) and perform an
   *exponentiated‑gradient* (mirror‑descent) update, which keeps the
   vector non‑negative and automatically projects onto the simplex
   after renormalisation.
3. **Occasional symmetrisation** – the optimal solution is believed to be
   symmetric, so every few iterations we replace a by (a+a[::-1])/2.
4. **Final annealing polish** – cheap mass‑transfer moves accepted
   according to a temperature schedule provide a last improvement.

The routine respects the given time budget and returns the best sequence
found (as a plain Python list).  All helper functions are defined at the
module level to avoid closures.
"""

import time
import math
import random
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (fallback, not used in the main loop)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using real FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution (active‑set)
# ----------------------------------------------------------------------
def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a * a)[k] at the current a.
    Returns (grad, max_value).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)          # length 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) step on the simplex
# ----------------------------------------------------------------------
def _exp_gradient_step(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform a mirror‑descent update on the probability simplex using
    the Kullback‑Leibler geometry (i.e. an exponentiated‑gradient step).
    """
    # add a tiny epsilon to avoid log(0)
    log_a = np.log(a + 1e-30) - eta * grad
    # shift for numerical stability
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(a, 1.0 / a.size)
    return a_new / s


# ----------------------------------------------------------------------
# Optimisation for a single length
# ----------------------------------------------------------------------
def _optimize_length(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Run exponentiated‑gradient sub‑gradient descent for a given length.
    Returns the best vector (sum = 1) and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    # initial score
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline:
        it += 1
        # obtain sub‑gradient and current max convolution value
        grad, max_val = _gradient_max_conv(a)
        # diminishing step size
        eta = 0.5 / math.sqrt(it)
        # exponentiated‑gradient update
        a = _exp_gradient_step(a, grad, eta)

        # occasional symmetrisation (helps because optimal solutions are symmetric)
        if it % 10 == 0:
            a = (a + a[::-1]) * 0.5

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] length={n} it={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (final polishing)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_seq: np.ndarray,
                   cur_best_score: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 150000) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: move a small ε from a source index to a destination index.
    Returns the best sequence found during the run and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12   # up to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve length={n} it={it} score={score:.6f}")

    return best_seq, best_score

    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    global_a = None  # keeps the best distribution for up‑sampling

    # Length schedule – start small, grow geometrically.
    lengths = [2048, 8192, 32768, 65536, 131072]

        # --------------------------------------------------------------
        if global_a is not None:

            a0 = 1.0 / np.sqrt(t * (1.0 - t) + 1e-12)
            a0 = np.maximum(a0, 0.0)

            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after descent n={n} score={score_opt:.6f}")

        # --------------------------------------------------------------
        # Final annealing polishing (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(3.0, (deadline - time.time()) * 0.15)
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_seq,
                best_score,
                start_temp=5e-2,
                end_temp=1e-5,

            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # remember best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should rarely be needed
    # ------------------------------------------------------------------

    except Exception:

    # Return a plain Python list – the evaluator will clamp / normalise.
    return [float(x) for x in best_seq]
