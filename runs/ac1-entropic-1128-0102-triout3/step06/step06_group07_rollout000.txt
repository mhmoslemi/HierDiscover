"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  The search consists of three phases:

1.  **Parametric “U‑shaped” families** –  simple formulas
    a_i = (d_i + c)^p, a_i = exp(-α d_i) (where d_i = min(i,n‑1‑i))
    are tried for a grid of parameters.  The best candidate from this
    cheap search is kept.

2.  **Projected Adam optimisation** – starting from the best
    candidate we run a few thousand Adam updates with a projection onto
    the probability simplex.  The sub‑gradient of the max‑convolution
    is computed analytically and averaged over all (almost) maximal
    entries.

3.  **Optional up‑sampling + fine‑tuning** – if time permits we double
    the length of the vector, repeat the Adam optimisation, and finally
    perform a short simulated‑annealing mass‑transfer refinement.

All helpers are top‑level, no closures are used, and the code respects
the supplied time budget (default 1000 s).  The final sequence is
normalised to sum = 1 (the scaling does not affect `evaluate_sequence`).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over active set)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length old_n to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, c: float = 0.0) -> np.ndarray:
    """U‑shape a_i = (d_i + c)^p, d_i = min(i,n‑1‑i)."""
    i = np.arange(n)
    d = np.minimum(i, n - 1 - i).astype(float)
    a = (d + c) ** p
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def _make_exp_shape(n: int, alpha: float) -> np.ndarray:
    """U‑shape a_i = exp(-α·d_i)."""
    i = np.arange(n)
    d = np.minimum(i, n - 1 - i).astype(float)
    a = np.exp(-alpha * d)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr: float = 0.05,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with Euclidean projection onto the probability simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: iter={t:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        grad = _subgradient_max(a, conv)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing: transfer a random fraction of mass from a
    source index to a destination index.  The temperature follows a
    geometric schedule.
    """
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30  # move up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that
    minimises `evaluate_sequence`.  The routine combines a cheap
    parametric sweep, Adam optimisation, optional up‑sampling and a
    short annealing phase, all respecting the overall time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine a starting length and an initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                base_seq = tmp / tmp.sum()
                base_n = tmp.size
            else:
                raise ValueError
        except Exception:
            base_seq = None
            base_n = 4096
    else:
        base_seq = None
        base_n = 4096

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric sweep (U‑shapes)
    # ------------------------------------------------------------------
    n = base_n
    best_seq = None
    best_score = float("inf")

    # Grid for power‑shape (p, c)
    p_vals = np.linspace(0.0, 5.0, 21)          # 0 … 5
    c_vals = np.array([0.0, 0.001, 0.01, 0.1, 1.0])  # offset to avoid exact zeros
    # Grid for exponential shape (α)
    alpha_vals = np.linspace(0.0, 5.0, 21)       # 0 … 5

    # Helper to evaluate candidate and keep the best
    def try_candidate(vec):
        nonlocal best_seq, best_score
        if vec is None or vec.size == 0:
            return
        sc = _objective(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.copy()
            print(f"[{time.time() - start_time:6.2f}s] Param sweep improve: score={sc:.6f}")

    # (a) power family
    for p in p_vals:
        for c in c_vals:
            if time.time() >= deadline:
                break
            vec = _make_power_shape(n, p, c)
            try_candidate(vec)

    # (b) exponential family
    for alpha in alpha_vals:
        if time.time() >= deadline:
            break
        vec = _make_exp_shape(n, alpha)
        try_candidate(vec)

    # If no candidate was kept (should not happen), fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the current length
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # a few dozen Adam iterations are usually enough for a noticeable drop
        adam_deadline = min(deadline, time.time() + 0.35 * (deadline - time.time()))
        a_opt, sc_opt = _run_adam(best_seq, n, adam_deadline, start_time,
                                  lr=0.05, max_iter=25000)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

    # ------------------------------------------------------------------
    # Phase 3 – optional up‑sampling + further Adam (if time permits)
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:                 # need a few seconds at least
        n2 = n * 2                                    # double the resolution
        up = _upsample(best_seq, n2)
        print(f"[{time.time() - start_time:6.2f}s] Upsampling to n={n2}")
        # short Adam on the larger vector
        up_deadline = min(deadline, time.time() + 0.25 * (deadline - time.time()))
        a_opt2, sc_opt2 = _run_adam(up, n2, up_deadline, start_time,
                                    lr=0.04, max_iter=20000)
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.copy()
            n = n2

    # ------------------------------------------------------------------
    # Phase 4 – light simulated annealing (final fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _run_annealing(
            best_seq, n, deadline, start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final anneal improve: score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (for safety) and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Convert to Python list of floats (the harness expects this)
    return [float(x) for x in best_seq]
