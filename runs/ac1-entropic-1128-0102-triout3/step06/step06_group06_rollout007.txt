"""
Non‑negative step‑function optimiser.

The algorithm builds a good candidate in three stages:

1. **Parametric warm‑start** – a short random search over several analytic
   families (product‑power, Gaussian, Beta, arcsine) yields a strong initial
   shape for each length.

2. **Entropic mirror‑descent on the simplex** – we minimise a smooth
   surrogate τ·log∑exp(conv/τ) of the ∞‑norm of the self‑convolution.
   The gradient is obtained analytically via a single FFT‑based correlation.
   The previous implementation had an off‑by‑one error in the gradient
   extraction; this version fixes it and uses a slightly slower temperature
   annealing that keeps the surrogate smooth for longer, improving convergence.

3. **Simple mass‑transfer simulated annealing** – a cheap polishing step that
   transfers a small random amount of mass between two positions and accepts
   moves according to a geometric temperature schedule.

The search proceeds on a schedule of increasing lengths
(256 → 512 → 1024 → 2048 → 4096 → 8192).  Early lengths receive a tiny
budget for rapid exploration, while the bulk of the time is spent on the
largest length, where the bound can be driven closest to the theoretical
limit (≈ 1).  All operations respect the non‑negativity and simplex constraints
required by `evaluate_sequence`.
"""

# ----------------------------------------------------------------------
# FFT‑based convolutions (self and cross)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (scale‑invariant)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
    2·n·max(a*a).
    """
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Parametric families for quick warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

def entropic_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,

    Minimise  max(a*a) via an entropic (soft‑max) surrogate.
    Uses exponentiated‑gradient updates (KL‑projection) and a slow temperature
    annealing schedule.  Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # Temperature schedule: start fairly warm, end very cold.
    tau = 5.0 / n            # initial τ  (≈ 0.001 for n≈5000)
    tau_end = 5e-3 / n       # final τ  (≈ 1e-6)
    decay = 0.9997

    # Step‑size schedule (mirror‑descent step η)
    eta0 = 0.8
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- exact objective ----------
        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] entropic improve n={n} it={it} score={cur_sc:.6f}"
            )

        # ---------- soft‑max weights ----------
        shift = conv.max()
        w = np.exp((conv - shift) / tau)               # shape 2n‑1
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the smooth surrogate ----------
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)                      # length = 3n‑2
        idx = (n - 1) + np.arange(n)                   # correct index extraction
        grad = 2.0 * corr[idx]

        # ---------- exponentiated‑gradient update ----------
        step = eta0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)            # avoid overflow
        a = a * np.exp(expo)

        # ---------- re‑normalise onto the simplex ----------
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ---------- anneal temperature ----------
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Transfer a random ε from one index to another.
    Accept moves that improve the score or according to a geometric temperature schedule.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
    `evaluate_sequence`.  Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None
    global_a = None

    # Try to reuse a previous construction if it exists.
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s > 0.0:
            global_a = tmp / s
    except Exception:
        global_a = None

    # ------------------------------------------------------------------
    # Stage 1 – rapid exploration on small lengths
    # ------------------------------------------------------------------
    small_lengths = [256, 512, 1024, 2048]
    for n in small_lengths:
        if time.time() >= deadline:
            break

        # Initialise (upsample previous best if available)
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # Parametric warm‑starts (a few seconds)
        # ------------------------------------------------------------------
        shape_deadline = time.time() + min(2.0, (deadline - time.time()) * 0.1)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.choice([0, 1, 2, 3])
            if kind == 0:          # product‑power
                p = np.random.uniform(-2.0, 3.0)
                cand = shape_product_power(n, p)
            elif kind == 1:        # Gaussian
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 2:        # Beta
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            else:                  # arcsine
                cand = shape_arcsine(n)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(
                    f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}"
                )

        # ------------------------------------------------------------------
        # Entropic mirror‑descent (short run)
        # ------------------------------------------------------------------
        gd_deadline = time.time() + min(6.0, (deadline - time.time()) * 0.15)
        a_opt, sc_opt = entropic_gradient_descent(
            local_best_a, n, gd_deadline, start_time
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after entropic n={n} score={sc_opt:.6f}"
            )

        # ------------------------------------------------------------------
        # Polishing anneal (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + min(3.0, (deadline - time.time()) * 0.1)
            a_ref, sc_ref = annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=60000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}"
                )

        # Preserve the best vector for the next up‑sample.
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Stage 2 – heavy optimisation on the largest feasible length
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Choose a large length that still fits comfortably in memory/time.
        # 8192 is a good trade‑off; if we have very little time left we fall back to 4096.
        target_n = 8192 if (deadline - time.time()) > 150 else 4096

        if global_a is not None:
            a0 = upsample(global_a, target_n)
        else:
            a0 = np.random.rand(target_n)
            a0 /= a0.sum()

        # Main entropic optimisation – consume the majority of the budget.
        gd_deadline = deadline - 8.0  # leave a few seconds for final polish
        a_opt, sc_opt = entropic_gradient_descent(
            a0, target_n, gd_deadline, start_time
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] final entropic n={target_n} score={sc_opt:.6f}"
            )

        # Final cheap annealing polish (uses the remaining few seconds)
        if time.time() < deadline:
            anneal_deadline = deadline
            a_ref, sc_ref = annealing(
                best_seq,
                target_n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=80000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] final anneal n={target_n} score={sc_ref:.6f}"
                )

    # ------------------------------------------------------------------
    # Safety fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list; the external evaluator will clamp / normalise as needed.
    return [float(x) for x in best_seq]
