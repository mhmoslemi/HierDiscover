"""
Propose a non‑negative step function that beats the current best bound.

Algorithm
---------
1. **Initialisation** – generate diverse starting vectors:
   * uniform Dirichlet draws,
   * a “spectral” start obtained from a random‑phase unit‑magnitude FFT
     (the resulting time‑domain signal is shifted to be non‑negative).

2. **Optimization** – minimise  F(a)=2·n·max(conv(a))  under a≥0,
   Σa=1.
   * Instead of Euclidean projection we work on the *logits* y with
     a = softmax(y).  Positivity and the simplex constraint are then
     automatic.
   * The sub‑gradient of `max(conv)` is computed analytically
     (averaged over all almost‑maximal indices) and multiplied by 2·n.
   * Using the chain rule we obtain a sub‑gradient w.r.t. y and update
     y with the Adam optimiser (adaptive learning rates, bias correction).

3. **Multi‑stage schedule** – the optimisation is carried out on three
   increasing lengths (≈2 k, 8 k, 16 k).  After each stage the best
   vector is up‑sampled (linear interpolation) to the next size, which
   provides a good warm‑start.

4. **Fine‑tuning** – a short simulated‑annealing phase moves a small
   amount of mass from a high‑probability index to a random one,
   accepting downhill or Metropolis steps.  This helps escape shallow
   plateaus.

All stages respect the global time budget (default 1000 s).  The best
vector found is normalised to sum = 1 and returned as a plain Python
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (non‑circular)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – averaged over the (approximately) active set
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Return an (averaged) sub‑gradient of max(conv) w.r.t. a."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in active:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if active.size > 1:
        grad /= active.size
    return grad


# ----------------------------------------------------------------------
# Softmax (stable) – maps logits y -> probability vector a on the simplex
# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the logits y (works on the probability simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iters: int = 10_000_000,
                   lr: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Optimise F(a)=2·n·max(conv(a)) using Adam on logits y.
    Returns the best a seen and its score.
    """
    # initialise logits from a0 (avoid log(0))
    y = np.log(np.maximum(a0, 1e-12))

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = _softmax(y)
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        a = _softmax(y)                       # always on the simplex
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] n={n:5d}  "
                  f"adam it={it:,}  score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        g_a = 2.0 * n * _subgradient_max(a, conv)

        # chain rule for softmax: grad_y = a * (g_a - <g_a,a>)
        dot = np.dot(g_a, a)
        g_y = a * (g_a - dot)

        # Adam update on y
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional learning‑rate decay (helps later stages)
        if it % 1000 == 0:
            lr = lr / math.sqrt(it)          # very gentle decay

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple linear up‑sampling (preserves the simplex)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Spectral random‑phase initialiser
# ----------------------------------------------------------------------
def _spectral_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative simplex vector by inverse FFT of a unit‑magnitude
    random‑phase spectrum.  The resulting signal is shifted to be non‑negative
    and normalised.
    """
    # length for a linear convolution of size n
    N = 1 << ((2 * n - 1).bit_length())   # next power of two
    # Build a Hermitian spectrum with unit magnitude (except DC)
    X = np.empty(N, dtype=complex)

    # DC component (real, set to 1)
    X[0] = 1.0 + 0j

    if N % 2 == 0:
        half = N // 2
        # positive frequencies 1 .. half‑1
        phases = rng.uniform(0.0, 2.0 * np.pi, half - 1)
        X[1:half] = np.exp(1j * phases)
        # Nyquist frequency (real)
        X[half] = 1.0 + 0j
        # mirror
        X[half + 1:] = np.conj(X[1:half][::-1])
    else:
        half = (N - 1) // 2
        phases = rng.uniform(0.0, 2.0 * np.pi, half)
        X[1:half + 1] = np.exp(1j * phases)
        X[half + 1:] = np.conj(X[1:half + 1][::-1])

    # time‑domain real signal
    b = np.real(np.fft.ifft(X))
    a = b[:n]                      # truncate to required length
    a -= a.min()                  # shift to make all entries ≥ 0
    a += 1e-12                    # avoid exact zeros for the log step
    a /= a.sum()                  # normalise to the simplex
    return a


# ----------------------------------------------------------------------
# Light mass‑transfer simulated annealing (fine‑tuning)
# ----------------------------------------------------------------------
def _anneal_refine(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source index proportionally to its current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30   # move up to 30 %
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n} "
                  f"it={it:,}  score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that
    minimises `evaluate_sequence`.  The routine uses a multi‑stage
    Adam optimiser, spectral initialisation and a final annealing pass.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction is present, use it as a possible starter
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 : many short Adam runs on a modest size (n≈2048)
    # ------------------------------------------------------------------
    n0 = 2048
    slice_time = max(30.0, 0.12 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    restarts = 10
    for r in range(restarts):
        if time.time() >= deadline:
            break

        # choose an initial vector
        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        elif r % 3 == 0:
            a0 = _spectral_init(n0, rng)
        else:
            a0 = rng.random(n0)
            a0 /= a0.sum()

        # run a short Adam optimisation
        a_opt, score = _adam_optimize(
            a0, n0, slice_deadline, start_time,
            max_iters=20_000,          # cheap runs
            lr=0.8,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in case nothing was produced (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * (_conv_self(best_seq).max())

    # ------------------------------------------------------------------
    # Stage 1 – upscale to a medium size (≈8192) and run a longer Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    a1 = _upsample(best_seq, n1)
    remaining = deadline - time.time()
    grad_time1 = max(25.0, 0.35 * remaining)
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _adam_optimize(
        a1, n1, grad_deadline1, start_time,
        max_iters=200_000,
        lr=0.6,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale to a large size (≈16384) and continue optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 16384
    a2 = _upsample(best_seq, n2)
    remaining = deadline - time.time()
    grad_time2 = max(20.0, 0.30 * remaining)
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _adam_optimize(
        a2, n2, grad_deadline2, start_time,
        max_iters=400_000,
        lr=0.5,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _anneal_refine(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
