"""
Search for a non‑negative step sequence that minimises `evaluate_sequence`.

The algorithm consists of three phases.

1. **Random shape sampling** – generate many cheap candidates from a few
   parametric families (truncated exponential, Beta‑pdf, piecewise‑linear)
   and keep the best one.

2. **Projected Adam on a smooth log‑sum‑exp surrogate** – start from the
   best sampled shape and run a long Adam optimisation on the surrogate
   𝓛_λ(a)=λ·log Σ exp(conv(a)/λ).  The surrogate tightly approximates the
   true max‑norm for small λ and provides a usable gradient.
   All updates are projected onto the probability simplex (∑aᵢ=1,
   aᵢ≥0).

3. **Short simulated‑annealing polish** – finally apply a cheap mass‑transfer
   annealing that explores the neighbourhood of the Adam result and may
   further lower the exact objective.

The routine respects the time budget (`budget_s`) and always returns a
valid list (fallback: uniform distribution) even if the search is cut
short.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution helpers (real FFT)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant; sum(a) is assumed ≈1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of L_λ(a)=λ·log Σ exp(conv/λ).

    The derivation uses the identity
    ∂conv/∂a_i = 2·(a ∗ rev(w))[i]   where w are the softmax weights.
    """
    n = a.shape[0]
    max_val = conv.max()
    # softmax over the convolution values
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # reverse a for the convolution
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)          # length 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]   # central slice
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimisation on the surrogate
# ----------------------------------------------------------------------
def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False,
                  max_iters: int = 10**7) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth surrogate; returns the best encountered."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # start with a relatively large λ and decay it gradually
    lam_factor = 0.3          # λ = lam_factor * max(conv)
    lam_decay = 0.85
    lam_decay_interval = 5000

    it = 0
    while time.time() < slice_deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam iter={it} best={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule
        step = 0.6 / math.sqrt(it + 1.0)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing polishing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing; returns the best sequence seen."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # transfer up to 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal it={it} best={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Shape generators for the random‑sampling phase
# ----------------------------------------------------------------------
def _sample_exponential(n: int) -> np.ndarray:
    """Truncated exponential a_i ∝ (1‑r) r^i, r∈(0,1)."""
    # choose r = 1‑c/n with c∈[0.2,3.0] (covers a useful range)
    c = 0.2 + 2.8 * random.random()
    r = 1.0 - c / n
    if r <= 0.0:
        r = 0.5
    i = np.arange(n, dtype=float)
    a = r ** i
    a = a / a.sum()
    return a


def _sample_beta(n: int) -> np.ndarray:
    """Beta‑pdf on a regular grid (α,β > 0)."""
    a = 0.1 + 4.9 * random.random()   # α∈[0.1,5.0]
    b = 0.1 + 4.9 * random.random()   # β∈[0.1,5.0]
    x = np.linspace(0.0, 1.0, n)
    y = (x ** (a - 1)) * ((1.0 - x) ** (b - 1))
    if y.sum() == 0.0:
        y = np.full_like(x, 1.0 / n)
    else:
        y = y / y.sum()
    return y


def _sample_piecewise_linear(n: int, knots: int = 8) -> np.ndarray:
    """Random piecewise‑linear shape defined by `knots` random heights."""
    xs = np.linspace(0, n - 1, knots)
    ys = np.random.rand(knots)
    a = np.interp(np.arange(n), xs, ys)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a
