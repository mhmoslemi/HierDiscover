def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0

    # initial length
    n = 8192
    # optional using previous best (height_sequence_1)
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        init = np.maximum(init,0)
        if init.size > 0:
            # upsample or downsample to n
            seq = _upsample(init, n)
        else:
            raise Exception
    except Exception:
        seq = np.random.rand(n)
        seq /= seq.sum()
    best_seq = seq.copy()
    best_score = _objective(best_seq)

    # main loop: at each iteration, do gradient step on log-sum-exp surrogate
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv and gradient
        conv = _conv_self(best_seq)
        # compute log-sum-exp surrogate gradient
        lam = max(conv.max() * 0.3, 1e-12) # lambda parameter
        grad = _gradient_from_conv(best_seq, conv, lam)
        # update by Adam (simple)
        step = 0.5 / np.sqrt(it+1)
        best_seq = best_seq - step * grad
        best_seq = _project_simplex(best_seq)

        # evaluate
        cur_score = _objective(best_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = best_seq.copy()
            print(f"{time.time()-start_time:.2f}s best={best_score:.6f}")

        # periodic random mass transfer (simulated annealing)
        if it % 200 == 0:
            # try random move
            for _ in range(5):
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i==j or best_seq[i]==0: continue
                eps = best_seq[i] * np.random.rand() * 0.2
                prop = best_seq.copy()
                prop[i] -= eps
                prop[j] += eps
                prop = _project_simplex(prop) # ensure simplex
                prop_score = _objective(prop)
                if prop_score < best_score:
                    best_seq = prop
                    best_score = prop_score
                    print(f"{time.time()-start_time:.2f}s anneal improve score={best_score:.6f}")

        # occasional LP improvement (every 1000 iterations)
        if it % 1000 == 0:
            a_cur = _run_lp_improve(best_seq, n, step=0.2, max_iters=2, deadline=deadline, start_time=start_time)

            if cur_score < best_score:
                best_seq = a_cur
                best_score = cur_score
                print(f"{time.time()-start_time:.2f}s LP improve score={best_score:.6f}")

    return [float(x) for x in best_seq]

"""
Hybrid optimisation for the self‑convolution peak.

The search proceeds as follows:

1. Warm‑start from the previously found best sequence (``height_sequence_1``)
   if it exists, otherwise from a uniform vector.
2. A schedule of increasing lengths is used (4096 → 8192 → 16384).
   At each length the current best distribution is up‑sampled (linear
   interpolation) and refined.
3. Refinement consists of:
   * Projected Adam optimisation on the smooth log‑sum‑exp surrogate of
     max(a*a) (the main workhorse, runs until the per‑stage deadline).
   * An LP “swap” step (only for n ≤ 4096, where the dense matrix still fits
     in memory) to increase the total mass while keeping the convolution
     bounded.
   * A few iterations of sub‑gradient descent on the true max‑norm.
   * A mass‑transfer simulated‑annealing polish.
4. The best distribution found at the current length is kept as the warm‑start
   for the next larger length.
5. After the last length a final long Adam run (with symmetry) is performed
   if time permits.

All steps stay on the probability simplex (∑aᵢ = 1), therefore the evaluation
function 2·n·max(conv)/∑a² is scale‑invariant.  The routine respects the
global time budget (default 1000 s) and always returns a valid non‑negative
sequence.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project the vector v onto the probability simplex {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real, fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using FFT."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact evaluation used during the search (∑a=1 ⇒ objective = 2·n·max)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Adam optimisation on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate:
        L_λ(a) = λ·log Σ exp(conv/λ).
    """
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # reverse a for convolution with the weight vector
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)               # length = 3·n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]      # central slice (size n)
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth surrogate.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3          # λ = lam_factor * max(conv)
    lam_decay = 0.9
    lam_decay_interval = 3000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming “swap” step (classic LP attack)
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """Build the (2n‑1)×n matrix for the LP bound a*g ≤ M."""
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _run_lp_improve(a: np.ndarray,
                    n: int,
                    step: float = 0.2,
                    max_iters: int = 2,
                    deadline: float = None,
                    start_time: float = None) -> np.ndarray:
    """
    Repeated LP‑based improvement.
    Solves: maximise Σ b_j  s.t.  b_j≥0  and  (a*b)_k ≤ M for every k,
    where M = max(a*a).
    """
    a_cur = a.copy()
    for _ in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        conv = _conv_self(a_cur)
        M = conv.max()
        if n > 4096:
            # The matrix would be too large – skip LP for very big n.
            break
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σ b => minimise -Σ b

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30.0})

        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:          # no improvement possible
            break

        g = b / sum_b
        a_cur = (1.0 - step) * a_cur + step * g
        a_cur = _project_simplex(a_cur)

        if start_time is not None:
            sc = _objective(a_cur)
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} step={step:.2f} score={sc:.6f}")

    return a_cur


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑norm
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         slice_deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        max_idx = np.flatnonzero(conv == M)

        grad = np.zeros_like(a)
        for k in max_idx:
            # Contribution from pairs (i, k-i)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            grad[i_range] += a[j_range]

            # Mirror contribution (k-i, i) gives the same term,
            # already accounted for because the sum goes over all i.
            # No double counting needed.

        step = 0.3 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap annealing (mass transfer) for final polishing
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # move up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and initial shape generators
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and re‑normalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
