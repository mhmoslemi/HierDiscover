def _run_adam_smoothed(a0, n, deadline, start_time):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.1
    it = 0
    best_a = a.copy()
    best_score = float('inf')
    # schedule: list of (alpha, steps)
    schedule = [(0.5, 20000), (2.0, 20000), (5.0, 20000), (10.0, 20000), (20.0, 20000), (40.0, 20000), (80.0, 20000)]
    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            it += 1
            conv = _conv_self(a)
            # compute smooth max using log-sum-exp
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0:
                continue
            w /= w_sum
            # gradient: 2 * convolution of a with reversed w
            w_rev = w[::-1]
            fft_len = _next_pow2(3 * n - 2)
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
            # Adam updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1**it)
            v_hat = v / (1.0 - beta2**it)
            step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
            a = a - step
            # Project onto simplex
            a = _project_simplex(a)
            # compute current objective score (exact max)
            cur_score = 2.0 * n * conv.max()  # conv from earlier
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam-smoothed improve: iter={it}, alpha={alpha:.2f}, score={cur_score:.6f}")
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative vector) that minimises the bound

    2·n·max(conv(a,a)) / (sum(a))**2

using a three‑phase scheme:

1.  Adam‑optimised smoothed‑max (log‑sum‑exp) with a continuation
    schedule for the smoothing parameter α.  This quickly drives the
    shape away from the uniform distribution.

2.  Adam on the true max‑convolution (α ≫ 1) for the bulk of the time.
    The adaptive learning‑rate of Adam together with a simple
    step‑size decay yields a more powerful descent than the plain
    projected sub‑gradient used previously.

3.  A short simulated‑annealing polish that performs tiny mass‑transfer
    moves to escape shallow plateaux.

All iterates stay on the probability simplex (non‑negative, sum = 1) by
projecting after each update.  The routine respects the supplied time
budget (≈1000 s) and falls back to the best sequence seen so far.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    Uses every index whose convolution value is within rel_tol of the
    global maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _run_adam_smoothed(a0: np.ndarray, n: int,
                       deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max
          f_α(a) = (1/α)·log(∑_k e^{α·conv_k}),
    with a continuation schedule for α.
    Returns the best vector encountered and its exact bound.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.08     # base learning rate (scaled later by 1/√t)

    it = 0
    # continuation schedule (α, number of iterations)
    schedule = [
        (0.5, 20000),
        (2.0, 20000),
        (5.0, 20000),
        (10.0, 20000),
        (20.0, 20000),
        (40.0, 20000),
        (80.0, 20000),
    ]

    fft_len = _next_pow2(3 * n - 2)

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a)

            # smooth‑max weights
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

            # gradient of the smooth objective
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = (base_lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = _project_simplex(a)

            # exact bound for bookkeeping
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam‑smooth improve: "
                      f"it={it:,} α={alpha:.1f} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the true max (α ≫ 1)
# ----------------------------------------------------------------------
def _run_adam_exact(a0: np.ndarray, n: int,
                    deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser where the gradient is taken w.r.t. the true max
    (α is set to a large constant so that the soft‑max is essentially a
    hard max).  The learning‑rate decays as 1/√t.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.04     # a bit smaller than the smoothed phase

    it = 0
    # Very large α – the soft‑max collapses to the arg‑max
    alpha = 400.0
    fft_len = _next_pow2(3 * n - 2)

    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam‑exact improve: "
                  f"it={it:,} score={cur_score:.6f}")

        # smoothed‑max weights (α large)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient (same formula as the smoothed case)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (base_lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # tiny random jitter every few thousand steps to avoid stagnation
        if it % 5000 == 0:
            i_src = rng.integers(n)
            i_dst = rng.integers(n)
            if a[i_src] > 1e-12:
                eps_move = a[i_src] * 0.02 * rng.random()
                a[i_src] -= eps_move
                a[i_dst] += eps_move

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polish (identical to the previous implementation)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealing that accepts uphill moves with a
    temperature schedule.  Operates on the exact bound (no smoothing).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"it={it:,} score={score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.
    The routine works in three phases described above.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if present.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter /= starter.sum()
    except Exception:
        starter = None

    final_n = 4800
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – Adam on smoothed‑max (≈15 % of the budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(5.0, remaining * 0.15)

    a_smooth, score_smooth = _run_adam_smoothed(
        a0, final_n, phase1_deadline, start_time
    )
    if score_smooth < best_score:
        best_score = score_smooth
        best_seq = a_smooth.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_smooth:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on the true max (≈75 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(10.0, remaining * 0.75)
        a_exact, score_exact = _run_adam_exact(
            best_seq, final_n, phase2_deadline, start_time
        )
        if score_exact < best_score:
            best_score = score_exact
            best_seq = a_exact.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_exact:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final annealing polish (any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=150_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: score={score_final:.6f}")

    # --------------------------------------------------------------
    # Normalise (protect against drift) and return a plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
