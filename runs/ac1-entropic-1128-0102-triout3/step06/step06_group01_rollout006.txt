The optimiser works in four stages  

1. **Initialisation** – generate a few promising shapes (uniform,
   the previous best construction, and a family of symmetric Beta‑densities).  
   The best of these is taken as a start vector `a`.

2. **Smoothed Adam optimisation** – the vector is parametrised by
   logits `y` (`a = softmax(y)`).  
   A smooth approximation of the max‑convolution (`log‑sum‑exp`) is
   minimised with Adam while the smoothing parameter `α` is increased
   from 0.5 up to 200.  The whole stage uses a fixed fraction of the
   remaining time budget.

3. **Exact sub‑gradient polishing** – a projected (simplex) gradient
   descent on the true non‑smooth objective
   `f(a)=2·n·max(a*a)/(∑a)²` is run for the next time slice.

4. **Simulated‑annealing fine‑tuning** – random mass‑transfer moves
   (`src → dst`) are accepted according to a temperature schedule.
   The move is evaluated with a fast FFT‑based convolution; the best
   vector found in this stage is kept.

All heavy work is performed with FFTs, which makes a single convolution
for `n = 4800` cheap (≈ 0.1 ms).  The algorithm respects the 1000 s budget
by checking the deadline after every major loop.

The resulting sequence is normalised, clipped to the allowed range
(`≤ 1000`) and returned as a plain list of floats.

```python
"""
Non‑negative step function optimiser
------------------------------------
* Initialise with a few hand‑crafted shapes (uniform, previous best,
  symmetric Beta densities).
* Smooth Adam optimisation on the log‑sum‑exp surrogate of the max
  self‑convolution, with an increasing smoothing factor α.
* Exact projected sub‑gradient descent on the true objective.
* Simulated‑annealing mass‑transfer refinement.
* All stages share the same 1000 s time budget and use FFT‑based
  convolutions for speed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  g(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂g/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


# ----------------------------------------------------------------------
# Helper for the original (non‑smoothed) objective
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of evaluate_sequence (without the safety‑clamp)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Projected gradient descent on the exact objective
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.94,
                         step0: float = 0.25,
                         max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (simple momentum) for the true objective.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a, n)
    it = 0
    eps_proj = 1e-12

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self_fft(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the probability simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

        # occasional numerical clean‑up
        if it % 1000 == 0:
            a = np.maximum(a, eps_proj)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.5,
                   end_temp: float = 1e-6,
                   steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated annealing.  A fraction ε of the mass at a
    source index is moved to a destination index; acceptance follows
    the Metropolis rule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self_fft(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric cooling)
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source proportionally to current mass (more likely to move)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # move a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self_fft(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Candidate generation (Beta family)
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, n: int) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on a regular grid, normalised to sum = 1.
    The grid points are placed at (i+0.5)/n to avoid the singular end‑points.
    """
    xs = (np.arange(n) + 0.5) / n          # (0,1) open interval
    # log‑pdf to stay stable for very small/large α
    log_pdf = (alpha - 1.0) * (np.log(xs) + np.log(1.0 - xs))
    # log of the beta normalising constant B(α,α)
    log_beta = math.lgamma(alpha) + math.lgamma(alpha) - math.lgamma(2.0 * alpha)
    log_pdf -= log_beta
    seq = np.exp(log_pdf)
    # safety: replace possible NaNs (should not appear) with zero
    seq = np.nan_to_num(seq, nan=0.0, posinf=0.0, neginf=0.0)
    s = seq.sum()
    if s == 0:
        # fall back to uniform if something went wrong
        return np.full(n, 1.0 / n)
    return seq / s


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method combines:
        * a small set of hand‑crafted starting shapes (including a Beta family),
        * a smooth Adam phase with increasing smoothing α,
        * exact projected sub‑gradient polishing,
        * a long simulated‑annealing fine‑tuning stage.
    The whole routine respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Problem size (use the size of the previously known construction if possible)
    # ------------------------------------------------------------------
    n = 4800
    # try to import the previous best construction
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Build a modest collection of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # previous best (if any)
    if starter is not None:
        if starter.shape[0] != n:
            # linear interpolation to the required length
            old_n = starter.shape[0]
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, starter)
            up = up / up.sum()
            candidates.append(up)
        else:
            candidates.append(starter.copy())

    # a sweep over symmetric Beta(α,α) – α ∈ (0.2, 2.0)
    alphas = np.concatenate([
        np.linspace(0.2, 0.8, 13),   # finer near the U‑shaped region
        np.linspace(0.9, 2.0, 12)
    ])
    for a in alphas:
        candidates.append(_beta_sequence(a, n))

    # ------------------------------------------------------------------
    # Pick the best among the cheap candidates (fast FFT based objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for seq in candidates:
        sc = _objective(seq, n)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[{time.time() - start_time:6.2f}s] initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth Adam optimisation (log‑softmax parametrisation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        adam_deadline = time.time() + 0.38 * remaining  # ≈ 38 % of the remaining budget
        # initialise logits
        eps = 1e-16
        y = np.log(np.maximum(best_seq, eps))

        # pre‑compute FFT length for the smooth gradient
        fft_len = _next_pow2(3 * n - 2)

        # Adam hyper‑parameters
        beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
        lr0 = 0.12
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        t = 0

        # smoothing schedule – we go further than the reference code
        alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]

        for alpha in alpha_schedule:
            # iterate until we hit the stage deadline
            while time.time() < adam_deadline:
                # ---------- compute a = softmax(y) ----------
                maxy = np.max(y)
                exp_y = np.exp(y - maxy)
                a = exp_y / exp_y.sum()

                # ---------- smoothed objective ----------
                conv = _conv_self_fft(a)
                scaled = alpha * conv
                max_scaled = scaled.max()
                # stable log‑sum‑exp
                lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
                f_val = lse / alpha
                obj = 2.0 * n * f_val

                # ---------- gradient w.r.t. a ----------
                grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

                # ---------- chain rule to logits ----------
                grad_y = a * (grad_a - np.dot(grad_a, a))

                # ---------- Adam update ----------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_y
                v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)

                lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
                y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

                # ---------- occasional exact evaluation ----------
                if t % 400 == 0:
                    exact_score = _objective(a, n)
                    if exact_score < best_score:
                        best_score = exact_score
                        best_seq = a.copy()
                        print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={t:,}, score={exact_score:.6f}")

            # stop early if we ran out of time
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient projected gradient polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        polish_deadline = time.time() + 0.30 * remaining   # about 30 % of what is left
        a_polish, score_polish = _run_gradient_descent(
            best_seq, n, polish_deadline, start_time,
            beta=0.94, step0=0.25
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – long simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        # allocate the rest of the budget to annealing
        a_final, score_final = _run_annealing(
            best_seq, n,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2,   # start a bit warmer than the reference
            end_temp=1e-6,
            steps=int(1.2e6)   # upper bound – the loop will stop by deadline anyway
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise to a proper probability vector, clip to the allowed range,
    # and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final sanity – truncate any accidental overflow (>1000) which would be
    # ignored by evaluate_sequence anyway
    best_seq = np.minimum(best_seq, 1000.0)

    return [float(v) for v in best_seq.tolist()]
```
