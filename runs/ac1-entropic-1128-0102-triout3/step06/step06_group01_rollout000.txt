"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – start from the supplied construction (if any) or a
   random simplex point.  The vector is symmetrised (a = (a+a[::-1])/2) because
   the optimum is expected to be symmetric.

2. **Smooth Adam phase** – minimise a log‑sum‑exp smooth approximation of the
   max‑convolution using a soft‑max parametrisation.
   Several smoothing levels (α) are traversed; the optimiser works on the
   logits (no explicit projection is needed).

3. **Heavy‑weight sub‑gradient descent** – switch to the exact objective
   (2·n·max(conv)) and run a projected (simplex) sub‑gradient method with
   momentum for the *remaining* time budget.  After each update the vector
   is re‑symmetrised and projected onto the probability simplex.

4. **Long simulated‑annealing refinement** – a mass‑transfer annealer that
   repeatedly moves a random fraction of mass from a “high‑gradient” index
   to a “low‑gradient” one.  Temperature decays geometrically.

5. **Normalisation & return** – clip tiny negatives, renormalise to sum‑1,
   and return a plain ``list`` of floats.

All phases respect the global deadline (≈1000 s).  The routine prints
progress messages and the best bound seen so far.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _simplex_proj(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i}."""
    return 0.5 * (a + a[::-1])


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**9):
    """
    Projected (simplex) sub‑gradient descent with momentum on
    f(a) = 2·n·max(conv(a)).
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # current convolution and max
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: iter={it:,}, score={cur_score:.6f}")

        # active set of indices achieving (almost) the max
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient (as in the reference code)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # gradient step + projection
        v = a - step * momentum
        a = _simplex_proj(v)

        # enforce symmetry (helps the search)
        a = _symmetrize(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-7,
                   steps: int = 500_000):
    """
    Mass‑transfer simulated annealing.
    Returns (best_seq, best_score).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportionally to its weight (more mass = more moves)
        src = rng.choice(n, p=a / a.sum())
        # destination uniformly
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # propose moving a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # enforce symmetry after the move (keeps search within the symmetric subspace)
        a_prop = _symmetrize(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines a smooth Adam stage, a long sub‑gradient descent,
    and a final annealing phase, all respecting the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Load a starter construction if one is provided in the global scope.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # problem size – larger n gives more flexibility but costs more time.
    n = 4800

    if starter is not None:
        if starter.shape[0] != n:
            a = _upsample(starter, n)
        else:
            a = starter.copy()
    else:
        a = np.random.rand(n)
        a /= a.sum()

    # enforce symmetry immediately
    a = _symmetrize(a)

    # ------------------------------------------------------------------
    # Adam optimisation on a smooth approximation of max‑convolution.
    # ------------------------------------------------------------------
    # Soft‑max parametrisation
    eps = 1e-16
    y = np.log(np.maximum(a, eps))

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – start mild, finish with a sharp approximation
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    iters_per_alpha = 8000   # more iterations than the baseline

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # keep track of the best exact score seen so far
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # ---- soft‑max to obtain a on the simplex ----
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()
            a = _symmetrize(a)          # maintain symmetry

            # ---- smooth objective ----
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val

            # ---- gradient w.r.t a (smooth) ----
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # ---- chain rule to logits y ----
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # ---- Adam update ----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 250 == 0:
                seq = a.tolist()
                sc = evaluate_sequence(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Sub‑gradient descent with momentum on the true objective.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate roughly 55 % of the remaining budget to hard descent
        remaining = deadline - time.time()
        gd_deadline = time.time() + 0.55 * remaining
        a_gd, score_gd = _run_gradient_descent(
            best_seq,
            n,
            gd_deadline,
            start_time,
            beta=0.95,
            step0=0.30,
        )
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD final improve: score={score_gd:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-7,
            steps=500_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal final improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise and safeguard before returning.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # Final sanity check (optional)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] Finished – final score={final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
