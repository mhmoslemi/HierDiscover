def _run_subgradient_descent(a0, n, deadline, start_time, step0=0.1):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_conv_self_fft(a))
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        max_val = conv.max()
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad improve: n={n} iter={it} score={score:.6f}")
        k_max = int(np.argmax(conv))
        # gradient: for each i, gradient_i = 2 * a[k_max - i] if valid else 0
        # Use vectorized:
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # step size diminishing
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * grad)
    return best_a, best_score

def _run_local_search(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_conv_self_fft(a))
    it = 0
    while time.time() < deadline:
        it += 1
        # propose move: pick donor i weighted by a_i (or by contribution)
        # We'll also compute contributions for current max to target region.
        # Use simple scheme: pick two random indices i,j, transfer small delta.
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # ensure donor has enough mass
        donor = i if a[i] > a[j] else j
        rec = j if donor == i else i
        # Two random deltas: up to half of donor mass
        delta = a[donor] * np.random.rand() * 0.5
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[rec] += delta
        # ensure nonneg
        if a_new[donor] < 0:
            continue
        # compute score
        score_new = 2.0 * n * np.max(_conv_self_fft(a_new))
        # accept if improvement or with small probability (simulated annealing)
        if score_new < best_score or np.random.rand() < 0.001:
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] local improve: n={n} iter={it} score={score_new:.6f}")
    return best_a, best_score

"""
Optimisation of a non‑negative step‑function for the objective
    score = 2·n·max(conv(a,a))   (with sum(a)=1)

The problem is convex (max of convex quadratics) but a direct solver is
expensive for thousands of variables.  The algorithm therefore combines
three cheap, scalable phases:

1. **Warm‑start / up‑sampling** – use the previous best construction
   (global variable ``height_sequence_1``) if it exists, otherwise a random
   simplex point.  The vector is up‑sampled (linear interpolation) to a
   fairly large final size (default 9600).

2. **Projected sub‑gradient descent on the true non‑smooth objective**.
   For the current maximal convolution entry `k*` the sub‑gradient has a
   simple closed form:
       g_i = 2·a_{k*‑i}  (if 0≤k*‑i<n, else 0)
   An Euclidean projection onto the probability simplex keeps the iterate
   feasible.  A diminishing step size (∝1/√t) together with occasional
   printing of improvements drives the score down quickly.

3. **Targeted stochastic local search** – a Metropolis‑style walk that
   transfers a small fraction of mass from a “donor’’ index to a
   “receiver’’ index.  Acceptance follows a slow exponential cooling schedule,
   allowing occasional uphill moves to escape shallow minima.

The three phases are allocated roughly 60 % / 30 % / 10 % of the total time
budget (≈1000 s).  The best vector seen over the whole run is returned as a
plain Python list, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x≥0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is all equal or negative; return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: linear up‑sampling that preserves total mass
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate seq from its length to new_n, then renormalise to sum 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        out = seq.copy()
    else:
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, new_n)
        out = np.interp(new_x, old_x, seq)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.full_like(out, 1.0 / new_n)
    return out


# ----------------------------------------------------------------------
# Sub‑gradient descent on 2·n·max(conv)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method for the exact max‑convolution objective.
    Returns the best vector seen and its final score.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_score = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline:
        it += 1
        # evaluate current objective
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # index attaining the maximum (any if multiple)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k‑i} when the index exists
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        step = step0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = _project_simplex(a - step * grad)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted stochastic local search (peak‑shaving)
# ----------------------------------------------------------------------
def _run_local_search(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑shifting walk with a slowly cooling Metropolis acceptance.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    conv = _conv_self_fft(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    it = 0

    # temperature schedule
    T_start = 5e-2
    T_end = 1e-6

    while time.time() < deadline:
        it += 1

        # pick donor proportional to its contribution to the current max entry
        conv = _conv_self_fft(a)
        k_max = int(np.argmax(conv))

        # contributions of each index to that entry
        i_arr = np.arange(n)
        j_arr = k_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor: weighted by contribution (focus on hot spots)
        if contrib.sum() > 0:
            donor = np.random.choice(n, p=contrib / contrib.sum())
        else:
            donor = np.random.randint(n)

        # receiver: pick from low‑contribution region (bottom 20 %)
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            receiver = np.random.randint(n)
        else:
            receiver = np.random.choice(low_idx)

        if donor == receiver or a[donor] <= 1e-15:
            continue

        # move a random fraction (up to 30 %) of donor mass
        eps = a[donor] * np.random.rand() * 0.3
        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps

        # evaluate proposal
        new_score = 2.0 * n * _conv_self_fft(a_new).max()
        delta = new_score - cur_score

        # cooling factor
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed

        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score
