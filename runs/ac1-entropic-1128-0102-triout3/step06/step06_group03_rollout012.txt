"""
Improved step‑function search.

The algorithm works in three stages:

1. **Warm‑start** – load the previously discovered best sequence
   (global variable ``height_sequence_1``) and up‑sample it to a
   target length (≈4800).  If no starter exists, a random uniform
   vector is used.

2. **Projected soft‑max gradient descent** – a smooth approximation of
   the max‑convolution is minimised with Nesterov momentum.  The whole
   allocated gradient budget (≈60 % of the total time) is spent on this
   stage, using a relatively high initial temperature to explore the
   landscape.

3. **Accelerated peak‑shaving local search** – the current maximal
   convolution entry is identified and a small amount of mass is moved
   from the indices that contribute most to that peak to indices that
   contribute little.  An incremental convolution update (O(n) per move)
   makes the walk fast, and a simple simulated‑annealing schedule
   allows occasional uphill moves.  The remaining time (≈40 %) is used
   here.

The final sequence is normalised to sum = 1 and returned as a plain
Python list for the evaluation harness.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, Σ x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: gradient of a soft‑max of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of Σ_k w_k(conv_k) * conv_k with w_k = soft‑max(conv/temperature).
    The factor 2 comes from the derivative of aᵀJ_k a (Toeplitz matrix).
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))          # extra stability
    w = exps / exps.sum()

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)          # length = n + (2n‑1) - 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]         # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Optimiser: projected gradient descent with soft‑max gradient
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 1.0,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected Nesterov gradient descent on the smoothed max‑convolution.
    Returns the best point seen and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (exponential decay)
    T_start = 2.0e-1          # a bit hotter than the original code
    T_end = 5.0e-5

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: n={n}, "
                  f"iter={it}, score={cur_score:.6f}")

        # anneal temperature
        T = T_start * (T_end / T_start) ** (it / 30000.0)

        # gradient of the smooth max
        grad = _softmax_gradient(a, conv, T)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method consists of a warm‑start, a long soft‑max gradient descent and
    a fast peak‑shaving local-search phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Load optional starter sequence (previous best construction)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        a = starter / starter.sum()
    except Exception:
        # fallback: a moderate‑size random simplex point
        init_n = 2000
        a = np.random.rand(init_n)
        a /= a.sum()

    # ------------------------------------------------------------------
    # Target length – keep the length of the starter (≈4800) or enlarge a bit
    # ------------------------------------------------------------------
    target_n = max(4800, a.shape[0])
    if a.shape[0] != target_n:
        a = _upsample(a, target_n)
    a = a / a.sum()

    # ------------------------------------------------------------------
    # Gradient‑descent phase (≈60 % of total time)
    # ------------------------------------------------------------------
    grad_deadline = start_time + (deadline - start_time) * 0.60
    a, gd_score = _run_softmax_gd(
        a, target_n, grad_deadline, start_time,
        step0=1.0, beta=0.9, max_iter=10 ** 8
    )

    # ------------------------------------------------------------------
    # Accelerated local search (peak‑shaving)
    # ------------------------------------------------------------------
    n = target_n
    conv = _conv_self(a)               # full linear convolution (2n‑1)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    it = 0
    recompute_interval = 4000          # occasional exact recomputation
    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------------
        # Determine the current maximal convolution entry and contributions
        # ----------------------------------------------------------------
        k_max = int(np.argmax(conv))
        i_arr = np.arange(n)
        j_arr = k_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor – sample proportionally to contribution to the current peak
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            donor = np.random.choice(n, p=prob)
        else:
            donor = np.random.randint(n)

        # recipient – pick from a low‑contribution region
        low_thr = np.percentile(contrib, 20)          # bottom 20 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            recipient = np.random.randint(n)
        else:
            recipient = np.random.choice(low_idx)

        if donor == recipient or a[donor] <= 1e-15:
            continue

        # Transfer a random fraction (up to 30 %) of the donor mass
        eps = a[donor] * np.random.rand() * 0.3

        # ----------------------------------------------------------------
        # Incrementally update the convolution (exact, O(n))
        # ----------------------------------------------------------------
        conv_len = conv.shape[0]          # = 2*n-1
        inc = np.zeros(conv_len, dtype=float)

        # term 2 * a * δ  (δ has -eps at donor, +eps at recipient)
        inc[donor:donor + n] -= eps * a
        inc[recipient:recipient + n] += eps * a
        inc *= 2.0

        # term δ * δ (three non‑zero entries)
        inc[2 * donor] += eps * eps
        inc[2 * recipient] += eps * eps
        inc[donor + recipient] -= 2.0 * eps * eps

        conv_new = conv + inc
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # ----------------------------------------------------------------
        # Metropolis acceptance with a cooling schedule
        # ----------------------------------------------------------------
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T_start = 5.0e-2
        T_end = 1.0e-6
        T = T_start * (T_end / T_start) ** elapsed

        accept = (new_score < cur_score) or (
            np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-12))
        )

        if accept:
            a[donor] -= eps
            a[recipient] += eps
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"score={new_score:.6f}")

        # ----------------------------------------------------------------
        # Periodically recompute the full convolution to avoid drift
        # ----------------------------------------------------------------
        if it % recompute_interval == 0:
            conv = _conv_self(a)

    # ------------------------------------------------------------------
    # Final normalisation (tiny drift may happen because of rounding)
    # ------------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(x) for x in best_a]
