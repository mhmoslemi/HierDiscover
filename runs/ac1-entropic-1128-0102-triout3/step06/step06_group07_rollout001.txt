def _run_exchange(a0, n, deadline, start_time,
                  max_iters=30, glueness=0.5, tolerance=1e-9):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    t = 0.5
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        M = conv.max()
        # build constraint matrix A ub
        # Approach: for small n (<= 2048) build dense matrix; else use active constraints
        if n <= 2048:
            # Build dense
            # Use np.zeros((2*n-1, n))
            # Fill using loops
            rows = 2*n-1
            A = np.zeros((rows, n), dtype=float)
            for i in range(n):
                A[i:i+n, i] = a[i]
            b = np.full(rows, M)
            # solve LP
            # bounds: (0,None)
            res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs')
        else:
            # use active constraints
            # find rows within epsilon of M
            eps = 1e-12 * M
            active_rows = np.where(conv >= M - eps)[0]
            if active_rows.size == 0:
                active_rows = np.array([int(np.argmax(conv))])
            rows = active_rows.size
            # Build sparse matrix with shape rows x n
            data = []
            row_idx = []
            col_idx = []
            for k_idx, k in enumerate(active_rows):
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                # i runs from i_start..i_end, column i
                for i in range(i_start, i_end+1):
                    data.append(a[i])
                    row_idx.append(k_idx)
                    col_idx.append(i)
            A_sparse = csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))
            b = np.full(rows, M)
            res = linprog(-np.ones(n), A_ub=A_sparse, b_ub=b, bounds=[(0, None)] * n,
                         method='highs')
        if not res.success:
            print("[exchange] LP failed at iter", it, "status", res.message)
            break
        g = res.x
        sumg = g.sum()
        if sumg == 0:
            break
        g_scaled = g / sumg  # to sum =1
        # direction d = g_scaled - a
        a_new = (1 - t) * a + t * g_scaled
        # project onto simplex to ensure sum=1
        a_new = _project_simplex(a_new)

            a = a_new

"""
Propose a non‑negative step function minimizing `evaluate_sequence`.

Approach
========
1. **Stage 0 – Small‑scale exploration**  
   Multiple random restarts of projected gradient descent on `n≈1024`
   (simplex projection via Euclidean projection).  A Nesterov‑style
   momentum and a 1/√t step schedule are used.

2. **Stage 0‑exchange – Linear‑programming “exchange” step**  
   Starting from the best small vector, solve a linear program that
   maximises the total mass of a new vector `g` while keeping the
   convolution `f*g` ≤ current peak `M`.  The LP gives a direction that
   flattens the auto‑convolution.  The new candidate is a convex
   combination `f←(1‑t) f + t ĝ` (with `ĝ` scaled to sum = 1) and is
   projected back onto the simplex.  The process is repeated a few
   times (time‑bounded).  This step often yields a noticeable drop in
   `max(conv(f))`.

3. **Stage 1 – Upsample to medium size (`n≈4096`)**  
   Linear interpolation, renormalisation and a longer gradient descent.

4. **Stage 2 – Upsample to large size (`n≈12288`)**  
   Same as Stage 1 but with more iterations.

5. **Stage 3 – Light simulated‑annealing refinement**  
   Mass‑transfer moves with a geometric temperature schedule.

All stages respect the overall time budget (default 1000 s).  The final
sequence is normalised to sum = 1 and returned as a plain Python list.

The core optimisation target is `F(a)=2·n·max(conv(a))` with the
constraint `a_i≥0, Σa_i=1`.  Reducing `max(conv(a))` directly improves
`evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σ x = 1} (O(n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – correct diagonal handling.
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Return an (averaged) sub‑gradient of max(conv)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent (Nesterov style) with the corrected sub‑gradient
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Minimise F(a) = 2·n·max(conv(a)) under a ≥ 0, Σ a = 1.
    Returns the best vector encountered and its score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Light simulated annealing: move ε from a high‑mass index to a random one.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Exchange step (LP) – reduce max(conv) while preserving total mass
# ----------------------------------------------------------------------
def _run_exchange(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  max_iters: int = 12,
                  init_t: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform the linear‑programming exchange iteration described in the
    problem statement.  Each iteration solves:

        max   Σ g_i
        s.t.  (a * g)_k ≤ M   for all k,
              g_i ≥ 0

    where M = max(conv(a)).  The solution `g` is scaled to sum = 1 and
    blended with the current vector using a step size `t`.  The routine
    stops after `max_iters` iterations or when the time budget expires.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    t = init_t

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()

        # --- build active‑set constraints (rows where conv is near the peak) ---
        eps = max(1e-12 * M, 0.0)
        active_rows = np.where(conv >= M - eps)[0]
        # ensure at least one constraint
        if active_rows.size == 0:
            active_rows = np.array([int(np.argmax(conv))])

        row_cnt = active_rows.size
        data = []
        row_idx = []
        col_idx = []

        for r, k in enumerate(active_rows):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # column indices i_start..i_end contribute a[i] to this row
            cols = np.arange(i_start, i_end + 1, dtype=int)
            vals = a[cols]
            data.append(vals)
            row_idx.append(np.full_like(cols, r, dtype=int))
            col_idx.append(cols)

        data = np.concatenate(data)
        row_idx = np.concatenate(row_idx)
        col_idx = np.concatenate(col_idx)

        A_ub = csr_matrix((data, (row_idx, col_idx)), shape=(row_cnt, n))
        b_ub = np.full(row_cnt, M)

        # solve LP: maximise Σ g_i  ⇔ minimise -Σ g_i
        res = linprog(-np.ones(n),
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            print(f"[{time.time() - start_time:6.2f}s] LP failed (iter {it}): {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        g_scaled = g / sum_g  # enforce Σ g = 1

        # convex blend and re‑project onto simplex
        a_candidate = (1.0 - t) * a + t * g_scaled
        a_candidate = _project_simplex(a_candidate)

        cand_score = 2.0 * n * _conv_self(a_candidate).max()

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:6.2f}s] exchange improve: n={n}, iter={it}, score={cand_score:.6f}")
            # keep step size, maybe increase a little
            t = min(0.9, t * 1.05)
        else:
            # no improvement → shrink step size
            t *= 0.5
            if t < 1e-3:
                break

    return best_a, best_score

    """
    Search for a step‑function (list of non‑negative floats) that minimises

    gradient descent → final simulated‑annealing.
    """

    start_time = time.time()
    # safety margin for harness bookkeeping
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short runs on a modest size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    slice_time = max(30.0, 0.10 * (deadline - time.time()))   # ≈10 % of budget
    slice_deadline = time.time() + slice_time

    restarts = 8
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        beta = np.random.uniform(0.80, 0.95)
        step0 = np.random.uniform(0.20, 0.45)

        a_opt, score = _run_gradient_descent(
            a0, n0, slice_deadline, start_time,
            beta=beta,
            step0=step0,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # ------------------------------------------------------------------
    # Stage 0‑exchange – LP based refinement (small size only)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        exchange_deadline = time.time() + max(15.0, 0.12 * (deadline - time.time()))
        a_ex, score_ex = _run_exchange(best_seq, n0, exchange_deadline, start_time,
                                       max_iters=12, init_t=0.5)
        if score_ex < best_score:
            best_score = score_ex
            best_seq = a_ex.copy()
            print(f"[{time.time() - start_time:6.2f}s] exchange stage improve: score={score_ex:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – upscale to medium size (n≈4096) and longer gradient descent
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time1 = max(20.0, 0.35 * remaining)          # ≈35 % of what is left
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _run_gradient_descent(
        a1, n1, grad_deadline1, start_time,
        beta=0.92,
        step0=0.28,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale to a large size (n≈12288) and continue optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    grad_time2 = max(15.0, 0.30 * remaining)          # ≈30 % of the rest
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _run_gradient_descent(
        a2, n2, grad_deadline2, start_time,
        beta=0.94,
        step0=0.24,
    )

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline:
