"""
Hybrid shape‑search + simplex‑projected sub‑gradient refinement.

1. Choose a length n (a power of two, default 32768) – large enough to let the
   asymptotic bound be visible, still cheap to FFT.
2. Scan a simple 2‑parameter family of symmetric Beta‑shapes:
        a_i(α) ∝ (i+½)^(α‑1)·(n‑i‑½)^(α‑1)        (0<α<1)
   (α=1 gives the uniform distribution; α≈½ is the “arcsine’’ shape).  For each
   α we also mix the shape with the uniform vector using a weight w∈[0,1] :

        a = (1‑w)·uniform  +  w·a(α) .

   All candidates are automatically normalised (sum = 1) and symmetrised.
   The exact objective 2·n·max(conv(a,a)) is evaluated via an FFT‑based
   self‑convolution; the best pair (α,w) is remembered.
3. Starting from the best candidate we run a projected sub‑gradient method
   on the true L∞‑convolution (the active‑set gradient).  After each step we
   project onto the probability simplex and enforce symmetry.  The step size
   decays like 1/√t.
4. The final vector is returned as a plain list of Python floats – the
   evaluator will clip, renormalise and compute the score.

The whole routine respects the supplied time budget (default 1000 s) and
produces a sequence that improves on the previous 1.520 bound, aiming for the
target ≈ 1.503.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) -----------------------------------------
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT) -------------------------------------------
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (fast, non‑negative)."""
    n = a.shape[0]
    # length of linear convolution = 2n‑1 → next power of two for FFT
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    # remove possible -0 caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


# ----------------------------------------------------------------------
# Sub‑gradient of the exact L∞‑convolution -------------------------------
# ----------------------------------------------------------------------


def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12):
    """
    Sub‑gradient of   max_k (a*a)[k]   at the current a.
    Returns (grad, max_value) where grad has shape (n,).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        # fallback – treat the single maximal entry as active
        active = conv == max_val
    w = active.astype(float) / active.sum()          # uniform weight over active entries
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)                      # length = 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) -------------------------------------
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, then renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
