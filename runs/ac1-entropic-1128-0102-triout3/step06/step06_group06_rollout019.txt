def adam_optimize(p0, n, deadline, start_time):
    # p0 normalized vector (sum=1). Represent logits x = np.log(p0+eps) maybe.
    x = np.log(p0 + 1e-12)
    # Initialize Adam parameters
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_p = p0.copy()
    best_score = evaluate_sequence(p0.tolist())
    tau = 0.1 # initial temperature for smoothed max
    while time.time() < deadline:
        # compute p = softmax(x)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        p = exp_x / exp_x.sum()
        # compute convolution of p
        conv = conv_self(p)
        # softmax weights for smoothed max
        shift = conv.max()
        w = np.exp((conv - shift) / tau)
        w /= w.sum()
        # gradient of smooth max wrt p: grad = 2 * (w convolve with reversed p)
        p_rev = p[::-1]
        corr = conv_fft(w, p_rev)  # length 3n-2
        idx = (n-1) - np.arange(n)
        grad = 2.0 * corr[idx]
        # gradient in logit space: g_x = p * (grad - np.dot(p, grad))
        grad_avg = np.dot(p, grad)
        g_x = p * (grad - grad_avg)
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        x -= step * m_hat
        # optional temperature annealing

        # evaluate periodically
        if t % 100 == 0:
            score = evaluate_sequence(p.tolist())
            if score < best_score:
                best_score = score
                best_p = p.copy()
    return best_p, best_score

"""
Step‑function optimizer.

The search proceeds in stages of increasing length.  For each length
we

1. generate a variety of warm‑starts (uniform, families, random
   Fourier‑phase vectors);
2. keep the best of those (exact objective computed with a fast FFT);
3. refine the best vector with an Adam optimiser on the soft‑max
   surrogate τ·log ∑exp(conv/τ) – the gradient is obtained analytically;
4. finish with a cheap mass‑transfer simulated‑annealing polish.

All vectors are kept in the probability simplex (non‑negative,
∑=1) so the objective
    2·n·max(conv_self(a))
coincides with the provided ``evaluate_sequence`` for sum = 1.
The routine respects the 1000 s time budget and returns the best
sequence it has found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power‑of‑two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv

# Scoring utilities (exact for ∑a = 1)
# ----------------------------------------------------------------------


def exact_score(a: np.ndarray) -> float:
    """Exact objective for a normalized vector (∑a = 1)."""
    n = a.shape[0]
    return float(2.0 * n * conv_self(a).max())


# ----------------------------------------------------------------------
# Shape families – cheap warm‑starts
# ----------------------------------------------------------------------


def shape_uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    a = np.full(n, 1.0 / n)
    return a


def shape_product_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = shape_uniform(n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = shape_uniform(n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = shape_uniform(n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = shape_uniform(n)
    return a


def shape_random_fourier(n: int) -> np.ndarray:

    Random phase vector: take a unit‑magnitude FFT, inverse‑transform,
    and use the absolute value as a non‑negative distribution.
    """
    phi = np.random.rand(n) * 2.0 * math.pi
    F = np.exp(1j * phi)
    a = np.abs(np.fft.ifft(F).real)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = shape_uniform(n)
    return a


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate τ·log∑exp(conv/τ).
    The vector is represented by logits x (soft‑max) so positivity
    and normalisation are automatic.
    Returns the best vector seen and its exact (non‑smoothed) score.
    """
    # initialise logits from the starting probability vector
    eps = 1e-12
    x = np.log(a0 + eps)

    # Adam moments
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0

    # temperature schedule for the soft‑max surrogate
    tau = 0.5            # start relatively warm
    tau_min = 5e-5
    tau_decay = 0.9995

    best_a = a0.copy()
    best_sc = exact_score(best_a)

    while time.time() < deadline:
        # soft‑max to obtain a probability vector
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---- smooth max computation ----
        conv = conv_self(a)
        shift = conv.max()                     # for numerical stability
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate w.r.t. a ----
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)              # length = 3n‑2
        idx = (n - 1) - np.arange(n)           # extract needed slice
        grad_a = 2.0 * corr[idx]               # d/da (τ·log‑sum‑exp)

        # ---- gradient wrt logits (soft‑max Jacobian) ----
        grad_avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - grad_avg)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.5 / (np.sqrt(v_hat) + eps_adam)
        x -= step * m_hat

        # ---- temperature anneal ----
        tau = max(tau_min, tau * tau_decay)

        # ---- bookkeeping ----
        if t % 50 == 0:
            sc = exact_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={t} score={sc:.6f}"
                )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def anneal(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_seq: np.ndarray,
    cur_best_score: float,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap move‑based annealing: transfer a random ε from one index
    to another.  Accept improving moves or moves that are accepted by a
    geometric temperature schedule.
    """
    a = a0.copy()
    score = exact_score(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = exact_score(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    Returns the heights as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from a previous run (global variable)
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s > 0.0:
            starter = tmp / s
    except Exception:
        starter = None

    # length schedule – increase gradually; larger lengths get a bit more
    # budget but we keep the number modest so that each iteration stays fast.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    random.shuffle(lengths)

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise for this length
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = shape_uniform(n)

        # --------------------------------------------------------------
        # Allocate a slice of the remaining time for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # give a slightly larger fraction to later (larger) lengths
        frac = 0.12 / (len(lengths) - idx)
        slice_deadline = time.time() + max(8.0, remaining * frac)

        # --------------------------------------------------------------
        # Random parametric warm‑starts (quick exploration)
        # --------------------------------------------------------------
        local_best_a = a0
        local_best_sc = exact_score(local_best_a)

        warm_end = time.time() + min(2.0, (slice_deadline - time.time()) * 0.25)

        while time.time() < warm_end:
            kind = np.random.choice([0, 1, 2, 3, 4, 5])
            if kind == 0:   # product‑power
                p = np.random.uniform(-2.0, 3.0)
                cand = shape_product_power(n, p)
            elif kind == 1:  # Gaussian
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 2:  # Beta
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            elif kind == 3:  # arcsine‑like
                cand = shape_arcsine(n)
            elif kind == 4:  # uniform (simple baseline)
                cand = shape_uniform(n)
            else:           # random Fourier‑phase
                cand = shape_random_fourier(n)

            sc = exact_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(
                    f"[{time.time() - start_time:6.2f}s] warm‑start improve n={n} kind={kind} score={sc:.6f}"
                )

        # --------------------------------------------------------------
        # Adam optimisation (smooth max surrogate)
        # --------------------------------------------------------------
        opt_deadline = time.time() + (slice_deadline - time.time()) * 0.55
        a_opt, sc_opt = adam_optimize(local_best_a, n, opt_deadline, start_time)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after Adam n={n} score={sc_opt:.6f}"
            )

        # --------------------------------------------------------------
        # Cheap annealing polish (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, sc_ref = anneal(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_seq,
                best_score,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=60000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}"
                )

        # keep the current best for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = shape_uniform(1024)

    # Convert to plain Python list (the external evaluator will clip/normalise)
    return [float(x) for x in best_seq]
