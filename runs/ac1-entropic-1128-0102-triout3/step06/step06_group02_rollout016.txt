def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    # Determine n = 9600 (or maybe 8000) - we can set a high n.
    n = 9600

    # Phase 0: generate candidates from predefined families.
    # Families: uniform, linear ramp (a_i ∝ i+1), reversed ramp, quadratic (i+1)^p normalized,
    # cosine window a_i = sin(pi*(i+0.5)/n), raised cosine a_i = (1 - cos(2π i/(n-1))) maybe.
    # Also exponential families: a_i = exp(- λ i) with λ tuned.

    best_seq = None
    best_score = float('inf')
    families = []
    families.append(lambda n: np.full(n, 1.0 / n))
    families.append(lambda n: np.arange(1, n+1, dtype=float))
    families.append(lambda n: np.arange(n, 0, -1, dtype=float))
    families.append(lambda n: (np.arange(n) + 1) ** 2)
    families.append(lambda n: np.sin(np.pi * (np.arange(n)+0.5) / n) ** 2)
    families.append(lambda n: 1 - np.cos(2*np.pi * np.arange(n) / (n-1)))  # not good near boundaries

    # For each family produce sequence, normalize, evaluate.
    for fam in families:
        a = fam(n).astype(float)
        a = a / a.sum()
        score = 2 * n * _conv_self(a).max()
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Initial family score: {score}")

    # Exponential search
    # We'll search lambda in logspace for lambda from 1e-6 to 1e-2 maybe.
    # Actually lambda = c/n approx. Use c from 0.2 to 5, step 0.2.
    for c in np.arange(0.2, 6.01, 0.2):
        lam = c / n
        a = np.exp(-lam * np.arange(n))
        a = a / a.sum()
        score = 2 * n * _conv_self(a).max()
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Exponential c={c:.2f} score={score}")

    # After initial candidates we now have best_seq.

    # Phase 1: softmax gradient descent with longer schedule.
    # We'll allocate remaining time for gradient.
    grad_deadline = deadline - 30.0  # reserve 30 sec for final annealing.
    a_opt, score_opt = _run_softmax_gd(best_seq, n, grad_deadline, start,
                                       step0=0.6, beta=0.9, max_iter=10**7)
    if score_opt < best_score:
        best_score = score_opt
        best_seq = a_opt.copy()
        print(f"After softmax best score {best_score}")

    # Phase 2: subgradient descent for remainder of grad time.
    a_opt2, score_opt2 = _run_subgradient_gd(best_seq, n, grad_deadline, start,
                                            step0=0.1, beta=0.9, max_iter=10**7)
    if score_opt2 < best_score:
        best_score = score_opt2
        best_seq = a_opt2.copy()
        print(f"After subgrad best score {best_score}")

    # Phase 3: final annealing for remaining time.
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(best_seq, n, deadline, start,
                                          best_score, best_seq,
                                          start_temp=0.2, end_temp=1e-8, steps=5_000_000)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"Final anneal best score {best_score}")

    # Return Python list.
    best_seq = best_seq / best_seq.sum()
    return [float(x) for x in best_seq]

"""
Multi‑restart optimisation for the step‑function minimisation problem.

The algorithm works in three stages:

1. **Exploratory seeding** – generate a variety of simple families
   (uniform, linear ramps, cosine windows, exponential decays) and keep
   the best one on a modest size (default 2400).  Several random restarts
   (uniform/exponential) are also performed while time permits.

2. **Progressive refinement** – the current best candidate is up‑sampled
   to the final length (default 9600) and is then optimised by three
   inexpensive phases:
      * smooth soft‑max gradient descent,
      * exact sub‑gradient descent on the true max‑convolution,
      * a long simulated‑annealing walk.
   All phases respect the remaining time budget.

3. **Return** – the final vector is normalised to sum = 1 and converted
   to a plain Python list for the evaluation harness.

The implementation uses FFT‑based convolutions, projects onto the
probability simplex after each update and prints occasional progress
messages.  It respects a single‑CPU execution model and works under the
1000 s time limit imposed by the harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution).
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max of the convolution (weighted sub‑gradient)."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()                     # soft‑max weights over the convolution entries

    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected (Nesterov) gradient descent on the soft‑max surrogate."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n}, iter={it}, score={cur_score:.6f}")

        # geometric temperature schedule
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true max‑convolution.
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.1,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected (Nesterov) sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        k = conv.argmax()                     # index attaining the max

        # sub‑gradient of max(conv) w.r.t. a (see reference code)
        grad = np.zeros_like(a)
        if k < n:
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves).
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.2,
                   end_temp: float = 1e-8,
                   steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 20 % of the mass at i (smaller moves later automatically via T)
        eps = a[i] * np.random.rand() * 0.20
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# One optimisation run on a given length.
# ----------------------------------------------------------------------
def _optimize_one_run(a0: np.ndarray,
                      n: int,
                      time_budget: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Run soft‑max GD → sub‑gradient GD → annealing on a single start vector.
    The three phases split the allotted time roughly 35 % / 35 % / 30 %.
    """
    # deadlines for the three phases
    soft_deadline = time.time() + time_budget * 0.35
    a_soft, score_soft = _run_softmax_gd(
        a0, n, soft_deadline, start_time,
        step0=0.6, beta=0.9, max_iter=10 ** 7
    )

    # sub‑gradient phase
    sub_deadline = time.time() + time_budget * 0.35
    a_sub, score_sub = _run_subgradient_gd(
        a_soft, n, sub_deadline, start_time,
        step0=0.1, beta=0.9, max_iter=10 ** 7
    )

    # pick the better of the two
    if score_sub < score_soft:
        best_seq, best_score = a_sub, score_sub
    else:
        best_seq, best_score = a_soft, score_soft

    # annealing phase (uses the remaining time up to the overall deadline)
    remaining_deadline = start_time + time_budget
    if time.time() < remaining_deadline:
        a_final, score_final = _run_annealing(
            best_seq, n,
            remaining_deadline, start_time,
            best_score, best_seq,
            start_temp=0.2, end_temp=1e-8,
            steps=2_000_000
        )
        if score_final < best_score:
            best_seq, best_score = a_final, score_final

    return best_seq, best_score

        )
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(f"[{time.time() - start_time:6.2f}s] final subgrad improve: score={score_sub:.6f}")

        # annealing walk
        anneal_deadline = time.time() + anneal_budget
        a_final, score_final = _run_annealing(
            best_seq, final_n,
            anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=0.2, end_temp=1e-8,
            steps=5_000_000
        )
        if score_final < best_score:
            best_score, best_seq = score_final, a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # 4. Final normalisation and return.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
