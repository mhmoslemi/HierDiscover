    e = np.exp(z - maxz)
    return e / e.sum()

"""
Propose a step‑function (non‑negative vector) that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Warm‑start** – if the global variable ``height_sequence_1`` exists it is
   normalised to sum‑1 and used as the initial shape.
2. **Progressive up‑sampling** – the current best vector is linearly interpolated
   to larger lengths (2000 → 4000 → 8000 → 9600).  After each up‑sample a
   fast L‑BFGS optimisation is performed.
3. **L‑BFGS on a smooth “log‑sum‑exp” surrogate** – the objective
   ``T * logsumexp(conv/T)`` (with conv = a*a) is a smooth approximation of the
   max‑convolution.  The variables are the unconstrained logits ``z`` with
   ``a = softmax(z)`` guaranteeing non‑negativity and unit sum.
   For a sequence of decreasing temperatures ``T`` (0.2 → … → 0.001) we
   repeatedly run L‑BFGS, warm‑starting each run from the previous result.
   Analytic gradients are provided:
        ∂f/∂a = 2 * conv( rev(a), softmax(conv/T) )
        ∂f/∂z_i = a_i ( ∂f/∂a_i – Σ_j a_j ∂f/∂a_j )
4. **Final simulated‑annealing polish** – a cheap Metropolis walk with tiny
   mass transfers refines the best solution found.
5. The best sequence (as a plain Python list) is returned.

The whole routine respects the given time budget (default 1000 s).
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:m]
    return conv


def _grad_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  T·logsumexp(conv/T)  w.r.t. a, where
    w_i = exp(conv_i/T) / Σ_j exp(conv_j/T).
    """
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    # central n entries correspond to the gradient for each a_i
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, then renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _objective_and_grad(z: np.ndarray, n: int, temp: float):
    """
    Returns (value, gradient) of  f(z)=temp*logsumexp(conv(z)/temp)
    where a = softmax(z) and conv = a * a.
    """
    a = _softmax(z)

    # convolution
    conv = _conv_self(a)

    # softmax weights over convolution entries
    # using a numerically stable formulation
    scaled = (conv - conv.max()) / temp
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()

    # gradient w.r.t. a
    grad_a = _grad_wrt_a(a, w)

    # chain rule: ∂f/∂z_i = a_i * (grad_a_i - Σ_j a_j grad_a_j)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    # objective value
    val = temp * logsumexp(conv / temp)

    return val, grad_z


def _lbfgs_optimize(a_start: np.ndarray, n: int, temp: float,
                    max_iter: int, deadline: float) -> np.ndarray:
    """
    Runs a bounded L‑BFGS optimisation for a given temperature.
    Returns the probability vector a = softmax(z) after optimisation.
    """
    # initialise logits; add a tiny epsilon to avoid log(0)
    eps = 1e-14
    z0 = np.log(a_start + eps)

    # scipy's minimize works with a callable returning (f, grad)
    def fun(z):
        return _objective_and_grad(z, n, temp)

    # remaining time based bound on the number of iterations
    # (each iteration is cheap, but we honour the outer deadline)
    remaining = deadline - time.time()
    # give at most max_iter iterations, but we also cap by remaining time
    # assume ~0.008 s per L‑BFGS iteration for n≈10 k (conservative)
    it_cap = int(min(max_iter, max(5, remaining / 0.008)))
    options = {"maxiter": it_cap, "ftol": 1e-12, "disp": False}

    res = minimize(fun, z0, method="L-BFGS-B", jac=True, options=options)
    a_opt = _softmax(res.x)
    return a_opt


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _annealing_refine(a0: np.ndarray, n: int, deadline: float,
                     start_time: float, best_score: float,
                     steps: int = 500_000) -> np.ndarray:
    """Metropolis walk moving a small fraction of mass between two positions."""
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best = best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = 0.2 * (1e-7 / 0.2) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # transfer up to 15 % of the mass at i
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best:
            best = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq
