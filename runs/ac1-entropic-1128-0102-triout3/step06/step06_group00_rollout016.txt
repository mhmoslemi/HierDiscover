def _grad_max_conv(a, conv, tol=1e-12):
    max_val = conv.max()
    active = conv >= max_val * (1 - tol)
    w = active.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros_like(a)
    w = w / w_sum
    # compute convolution of a with reversed w.
    # w_rev needed for gradient formula as used before.
    w_rev = w[::-1]
    L = _next_pow2(len(a) + len(w_rev) - 1)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wf, n=L)[: len(a) + len(w_rev) - 1]
    # gradient expression: we need indices that correspond to a gradient size n.
    # In _run_exponential_gradient they compute:
    # grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    # Explanation: because conv of a and w_rev yields length n + len(w_rev) - 1.
    # For each i in 0..n-1, gradient for a_i = 2 sum_{k} a_k * w_{i+k}? Wait.
    # Use same formula.
    n = a.shape[0]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

"""
Fast sub‑gradient optimisation of  max(convolution)  on the simplex.

The code works in three phases:

1.  **Coarse optimisation** – a tiny problem (n≈1024) is solved for many
    iterations.  The sub‑gradient of the exact max‑convolution is obtained
    with two FFTs (O(n log n)) and a Polyak step size
    ``step = (f(x)‑LB)/‖g‖²`` where ``LB = 2 n/(2 n‑1)`` is the provable
    lower bound.  Frequent projections onto the probability simplex keep the
    iterate feasible and allow us to perform hundreds of thousands of cheap
    updates within the budget.

2.  **Scale‑up** – the best coarse solution is linearly up‑sampled to a
    medium size (≈2400) and then to the final size (4800).  After each
    up‑sampling a short gradient run is performed to recover the loss of
    precision introduced by interpolation.

3.  **Annealing polish** – a classic simulated‑annealing mass‑transfer
    routine makes a final sweep for tiny improvements.

All time limits are respected; the function returns a list of non‑negative
heights whose sum is 1 (the normalisation does not affect the objective).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length while preserving the total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Convolution helpers (linear convolution via FFT)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_max_conv(a: np.ndarray, conv: np.ndarray,
                   rel_tol: float = 1e-6) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All entries of conv that are within ``rel_tol`` of the maximum are
    treated as active and combined with equal weight (standard choice for a
    sub‑gradient of the pointwise maximum).
    The gradient is obtained by a single convolution in the Fourier domain.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    # binary mask of active constraints
    active = conv >= max_val * (1.0 - rel_tol)
    w = active.astype(float)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.zeros_like(a)
    w /= w_sum                      # any convex combination works

    # gradient = 2 * (a ⊛ w_rev)   (⊛ = linear convolution)
    w_rev = w[::-1]
    L = _next_pow2(3 * n - 2)       # length for a ⊛ w_rev
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wf, n=L)[: 3 * n - 2]

    # extract the ``n`` relevant entries
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish (same as the reference implementation)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000):
    """
    Tiny metropolis walk that moves a fraction of the mass between two
    positions; an exponential cooling schedule is used.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)          # source proportional to its mass
        dst = rng.integers(n)             # destination uniform
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Core optimisation loop (fast projected sub‑gradient descent)
# ----------------------------------------------------------------------
def _optimise_on_simplex(a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rel_tol: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Repeatedly:
        * evaluate f(a) = 2·n·max(conv(a,a))
        * compute a sub‑gradient of max(conv)
        * perform a Polyak step ``(f‑LB)/‖g‖²`` and project onto the simplex
    The best iterate seen is returned.
    """
    n = a.shape[0]
    LB = 2.0 * n / (2 * n - 1)                     # theoretical lower bound
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        g = _grad_max_conv(a, conv, rel_tol=rel_tol)
        norm_sq = np.dot(g, g)
        if norm_sq == 0.0:
            break

        # Polyak step towards the lower bound; keep it in a safe range
        step = (cur_score - LB) / norm_sq
        if step <= 0 or step > 1.0:
            step = 0.5 / math.sqrt(it)

        a = a - step * g
        a = _project_simplex(a)

    return best_a, best_score
