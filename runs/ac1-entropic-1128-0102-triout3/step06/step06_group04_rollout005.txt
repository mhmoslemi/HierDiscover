def _gradient_from_conv(a, conv, lam):
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # conv of w with reversed a
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)  # need _conv_fft defined
    n = a.shape[0]
    grad = 2.0 * tmp[n-1:n-1+n]
    return grad

def _conv_fft(a,b):
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)

    c = np.fft.irfft(A * B, n=nfft)[:n]

   exhibit the asymptotic improvement while keeping the quadratic LP
   matrix (size n²) tractable.

3. Three optimisation phases are cycled until the time budget expires:

   * **Projected Adam on a log‑sum‑exp surrogate** – smooth approximation
     of the max‑convolution.  The step‑size and temperature (λ) are
     decayed, and the iterate is always projected onto the probability
     simplex (∑a=1, a≥0).

   * **Linear‑programming “swap” step** (Frank‑Wolfe style).  With the
     current vector ``a`` we solve

        maximise   1ᵀg    subject to   g≥0,
                                         ‖a∗g‖_∞ ≤ M ,

     where M = max(a∗a).  The solution ``g`` is mixed into ``a`` with a
     small mixing weight; this strictly increases the total mass
     possible under the same bound and therefore lowers the objective
     after renormalisation.

   * **Simulated‑annealing mass‑transfer polishing** – occasional random
     “move” of a fraction of mass from one coordinate to another,
     accepted with Metropolis probability based on the true objective.

   The cycle repeats, each phase receiving a fraction of the remaining
   time.  The best vector seen throughout the run is kept.

4. Finally the best vector is returned as a plain Python list.  The
   evaluator will automatically clamp/normalise it, but we keep the sum
   equal to 1 to avoid the “sum < 0.01” penalty.

All operations stay on the simplex, making the objective scale‑invariant.
The routine respects the 1000 s budget and prints progress.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact evaluation of the objective (sum(a)≈1)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Adam on the smooth log‑sum‑exp surrogate of max
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of   L_λ(a) = λ·log Σ exp(conv/λ).
    The derivative wrt a_i is 2·∑_k w_k·a_{k-i},
    where w is the softmax over conv/λ.
    """
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # convolution of w with reversed a gives the needed sums
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)                # length = 2n‑1 + n‑1 = 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]      # central slice of length n
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam on the log‑sum‑exp surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3          # λ = lam_factor * max(conv)
    lam_decay = 0.8
    decay_interval = 3000

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        if it % decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-12)

        lam = max(conv.max() * lam_factor, 1e-12)
        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.6 / math.sqrt(it + 1.0)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming swap (Frank‑Wolfe) step
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A where
        A_{k,i} = f_i   if   i ≤ k < i + n,
                = 0      otherwise.
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _run_lp_improve(a: np.ndarray,
                    n: int,
                    step: float = 0.15,
                    max_iters: int = 5,
                    deadline: float = None,
                    start_time: float = None) -> np.ndarray:
    """
    Perform a few Frank‑Wolfe style LP updates:
    maximise total mass of g subject to ‖a∗g‖_∞ ≤ M,
    then mix g into a with weight ``step``.
    """
    a_cur = a.copy()
    for _ in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        conv = _conv_self(a_cur)
        M = conv.max()
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(g)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 10.0})

        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:      # no improvement possible
            break

        g = g / sum_g                 # renormalise to sum = 1
        a_cur = (1.0 - step) * a_cur + step * g
        a_cur = _project_simplex(a_cur)

        if start_time is not None:
            sc = _objective(a_cur)
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} step={step:.2f} score={sc:.6f}")

    return a_cur


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        max_idx = np.flatnonzero(conv == M)

        grad = np.zeros_like(a)
        # For each k attaining the maximum, accumulate the sub‑gradient
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            grad[i_range] += a[j_range]                     # a_{k-i}
            # the symmetric term (i‑k) is already covered when k iterates over all max positions,
            # but we add it explicitly for correctness:
            i_start2 = max(k, 0)          # actually i_start2 = k (since k≥0)
            i_range2 = np.arange(i_start2, n)
            j_range2 = i_range2 - k
            grad[i_range2] += a[j_range2]

        step = 0.3 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing polishing (mass transfer)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # move up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Resampling helpers
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation between old and new length, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the ``budget_s`` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found sequence if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter /= s
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose a fixed working length (4096 is a good trade‑off)
    # ------------------------------------------------------------------
    n = 4096

    if starter is not None:
        a = _upsample(starter, n)
    else:
        a = np.random.rand(n)
        a /= a.sum()

    best_seq = a.copy()
    best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # Main optimisation loop – cycle through phases while time permits
    # ------------------------------------------------------------------
    phase_cycle = ['adam', 'lp', 'subgrad', 'anneal']
    phase_idx = 0

    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:          # not enough time for another full phase
            break

        phase = phase_cycle[phase_idx % len(phase_cycle)]
        phase_idx += 1

        if phase == 'adam':
            # allocate roughly 60 % of remaining time to Adam
            phase_dead = time.time() + remaining * 0.60
            a_new, score = _run_adam_opt(a, n, phase_dead, start_time, symmetrize=True)
            # Adam already returns best encountered inside the phase
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] *** New global best after Adam: {score:.6f}")

        elif phase == 'lp':
            # allocate ~15 % of time to a few LP updates
            phase_dead = time.time() + remaining * 0.15
            a_new = _run_lp_improve(a, n,
                                    step=0.15,
                                    max_iters=5,
                                    deadline=phase_dead,
                                    start_time=start_time)
            score = _objective(a_new)
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] *** New global best after LP: {score:.6f}")

        elif phase == 'subgrad':
            # allocate ~10 % to exact sub‑gradient descent
            phase_dead = time.time() + remaining * 0.10
            a_new, score = _subgradient_descent(a, n, phase_dead, start_time)
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] *** New global best after sub‑grad: {score:.6f}")

        else:  # anneal
            # allocate the rest to annealing (≈15 %)
            phase_dead = time.time() + remaining * 0.15
            a_new, score = _run_annealing(a, n,
                                          deadline=phase_dead,
                                          start_time=start_time,
                                          cur_best_score=best_score,
                                          cur_best_seq=best_seq,
                                          start_temp=2e-2,
                                          end_temp=1e-6,
                                          steps=200000)
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] *** New global best after anneal: {score:.6f}")

        # prepare for next phase
        a = best_seq.copy()

    # ------------------------------------------------------------------
    # Final polishing if any time left (short Adam + anneal passes)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        final_dead = deadline - 0.5
        a_tmp, sc = _run_adam_opt(best_seq, n, final_dead, start_time, symmetrize=True)
        if sc < best_score:
            best_score, best_seq = sc, a_tmp.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final Adam polish score={sc:.6f}")

    if time.time() < deadline - 0.5:
        anneal_dead = deadline - 0.2
        best_seq, best_score = _run_annealing(best_seq, n,
                                              deadline=anneal_dead,
                                              start_time=start_time,
                                              cur_best_score=best_score,
                                              cur_best_seq=best_seq,
                                              start_temp=1e-2,
                                              end_temp=1e-6,
                                              steps=250000)

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return plain Python list
    return [float(x) for x in best_seq]
