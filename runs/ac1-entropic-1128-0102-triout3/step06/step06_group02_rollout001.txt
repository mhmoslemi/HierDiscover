"""
Optimiser for the step‑function attack.

The algorithm consists of three stages that respect a given time budget:

1. **Adam on a soft‑max approximation of the max‑convolution** –  
   a fast first‑order optimiser that flattens the convolution while
   keeping the vector on the probability simplex.

2. **Projected sub‑gradient descent on the true max‑convolution** –  
   fine‑tunes the solution using the exact sub‑gradient
   (g_i = 2·a_{k-i}, where k is the index of the current peak).

3. **Simulated‑annealing mass‑transfer walk** –  
   a long, low‑temperature stochastic walk that can escape
   shallow local minima.

The routine works on a single resolution (default 15000, but it
automatically down‑scales if the time budget is tight) and always
returns a list of non‑negative numbers that sum to one.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max of the convolution (smooth max)."""
    maxc = conv.max()
    scaled = (conv - maxc) / max(temperature, 1e-12)
    exps = np.exp(scaled)
    w = exps / exps.sum()

    # reverse a (for convolution with the weight vector)
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]

    # extract the central n entries (corresponding to the gradient)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Scoring helper (assumes a sums to 1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the evaluation_sequence value for a (sum(a) ≈ 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam optimisation of the soft‑max of the convolution."""
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.02

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)

    # temperature schedule
    T_start, T_end = 0.12, 1e-4

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        # temperature decreases roughly linearly with iteration count
        T = T_start * (T_end / T_start) ** (it / max_iter)

        grad = _softmax_gradient(a, conv, T)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected) on the true max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     max_iter: int = 150_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Sub‑grad improve: n={n}, iter={it}, score={cur_score:.6f}")

        k = conv.argmax()  # location of the current peak

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        if k < n:
            # indices 0 … k contribute
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            # indices k‑n+1 … n‑1 contribute
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        # Nesterov‑style momentum
        momentum = 0.9 * momentum + 0.1 * grad

        # step size – diminishing like 1/√t
        step = 0.15 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   init_temp: float = 0.15,
                   final_temp: float = 1e-7,
                   steps: int = 2_500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a small fraction of mass between two positions."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score(a)
    best_seq = a.copy()
    best_score = cur_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = init_temp * (final_temp / init_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 15 % of the mass at i
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _score(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: n={n}, iter={it}, score={cur_score:.6f}")

    return best_seq, best_score
