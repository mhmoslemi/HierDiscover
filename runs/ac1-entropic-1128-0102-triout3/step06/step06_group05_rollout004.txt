"""
Hybrid search for a low‑L∞ self‑convolution sequence.

The algorithm consists of three stages:

1. **Parametric warm‑start** – we sweep a few one‑parameter families
   (beta‑type, power‑law, and a mixture with the uniform vector) on a
   modest size (n≈2048) and keep the best candidates.

2. **Refinement on the warm‑start size** – a short Adam optimisation on
   the smooth log‑sum‑exp surrogate of the max‑convolution reduces the
   objective further.

3. **Large‑scale optimisation** – the best vector is up‑sampled to a
   much larger length (up to 65536 depending on the remaining time) and
   refined with a long Adam run, a projected sub‑gradient descent and a
   final cheap simulated‑annealing polishing phase.  All iterates are
   projected onto the probability simplex and symmetrised, because optimal
   solutions are believed to be symmetric.

The code respects the time budget (default 1000 s) and returns a Python
list of non‑negative heights.  The objective is evaluated internally as
`2 * n * max(conv)` (identical to `evaluate_sequence` for a sum‑1
normalisation).  The routine works with only a single CPU core and uses
only numpy and the standard library.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for all lengths)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)  # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the smoothed L∞ surrogate (soft‑max)
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing temperature λ > 0

    Returns
    -------
    grad: dense gradient of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)  # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (active‑set)
# ----------------------------------------------------------------------


def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a * a)[k] at the current a.
    Returns (grad, max_value).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)  # length 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate
# ----------------------------------------------------------------------


def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              symmetrize: bool = False,
              max_iter: int = None) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate.
    Returns the best vector seen and its exact objective value
    (2·n·max(conv) because the vector stays on the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule parameters
    lam_factor = 0.3            # initial λ relative to max(conv)
    lam_decay = 0.98
    lam_min_factor = 1e-9

    it = 0
    while time.time() < deadline:
        it += 1
        if max_iter is not None and it > max_iter:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam it={it} n={n} score={cur_score:.6f}")

        # update λ
        lam = max(max_b * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, lam_min_factor)

        # gradient of smoothed surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.8 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic sub‑gradient descent on the exact max
# ----------------------------------------------------------------------


def _run_subgradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    step0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad it={it} n={n} score={cur_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: transfer a small ε from a source index to a destination index.
    Returns the best sequence found during the run and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12  # up to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        a_prop = _project_simplex(a_prop)  # keep on simplex

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal it={it} n={n} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Parametric shape generators
# ----------------------------------------------------------------------


def _beta_shape(n: int, p: float) -> np.ndarray:
    """Symmetric beta‑type shape a_i ∝ (x(1‑x))^p, x∈(0,1)."""
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    a = np.power(x * (1.0 - x) + eps, p)
    a /= a.sum()
    return a


def _power_shape(n: int, gamma: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ (i+½)^‑γ + (n‑i‑½)^‑γ."""
    i = np.arange(n)
    a = (i + 0.5) ** (-gamma) + (n - i - 0.5) ** (-gamma)
    a /= a.sum()
    return a


def _mix_shape(n: int, w: float) -> np.ndarray:
    """Mixture of arcsine (p = –0.5) and uniform, weight w on the arcsine part."""
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    arc = np.power(x * (1.0 - x) + eps, -0.5)
    uni = np.ones_like(arc)
    a = w * arc + (1.0 - w) * uni
    a /= a.sum()
    return a


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective for a sum‑1 vector: 2·n·max(a*a)."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

            best_score, best_a = sc_ref, a_ref

    # ------------------------------------------------------------------
    # 3. Upsample to a large size and devote most time to it
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 400.0:
        n_large = 65536
    elif remaining > 200.0:
        n_large = 32768
    elif remaining > 80.0:
        n_large = 16384
    else:
        n_large = 8192

    a_large = _upsample(best_a, n_large)

    # ---- Adam on large size ------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + max(30.0, remaining * 0.6)
        a_adam, sc_adam = _run_adam(a_large, n_large, adam_deadline,
                                 start_time, symmetrize=True)
        if sc_adam < best_score:
            best_score, best_a = sc_adam, a_adam

    # ---- Sub‑gradient refinement -------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + max(10.0, remaining * 0.25)
        a_sub, sc_sub = _run_subgradient(best_a, n_large, sub_deadline, start_time)
        if sc_sub < best_score:
            best_score, best_a = sc_sub, a_sub

    # ---- Final annealing polishing -----------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        anneal_deadline = deadline
        a_ann, sc_ann = _run_annealing(
            best_a,
            n_large,
            anneal_deadline,
            start_time,
            best_score,
            best_a,
            start_temp=5e-2,
            end_temp=1e-5,
            steps=200000,
        )
        if sc_ann < best_score:
            best_score, best_a = sc_ann, a_ann

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best_a]
