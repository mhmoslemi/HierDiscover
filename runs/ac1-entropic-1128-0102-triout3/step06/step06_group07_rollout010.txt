def _subgradient_softmax(a, conv, alpha):
    n = len(a)
    max_val = conv.max()
    w = np.exp(alpha * (conv - max_val))
    Z = w.sum()
    if Z == 0:
        # fallback to uniform
        w = np.ones_like(w)
        Z = w.size
    # compute conv of w with rev(a) via FFT
    # length conv_len = len(w) + len(a_rev) - 1 = (2*n-1) + n - 1 = 3*n - 2
    a_rev = a[::-1]
    # pad lengths to next power of two
    conv_len = 3*n - 2
    fft_len = 1 << ((conv_len - 1).bit_length())
    # compute FFT
    Wf = np.fft.rfft(w, n=fft_len)
    Af = np.fft.rfft(a_rev, n=fft_len)
    # compute product
    prod = Wf * Af
    # inverse transform to get full conv
    full = np.fft.irfft(prod, n=fft_len)[:conv_len]
    # slice: slice from n-1 to n-1+n
    grad = 2.0 * full[n-1:n-1+n] / Z
    return grad

def _run_gradient_descent_smooth(a0, n, deadline, start_time, beta=0.9, step0=0.5, alpha0=1.0, alpha_max=30.0, alpha_steps=10000, max_iter=10**7):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] smooth grad improve: n={n}, it={it:,}, score={cur_score:.6f}")
        # compute alpha schedule
        if it <= alpha_steps:
            alpha = alpha0 + (alpha_max - alpha0) * (it / alpha_steps)
        else:
            alpha = alpha_max
        grad = _subgradient_softmax(a, conv, alpha)
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(it)
        # update via mirror descent (multiplicative)
        a = a * np.exp(-step * momentum)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)
    return best_a, best_score

"""
Improved search for a non‑negative step‑function that minimises
`evaluate_sequence`.

Main ideas
----------
1.  Multi‑stage pipeline (short random sampling → up‑sampling → long
    smooth gradient descent → final annealing) while respecting the
    global time budget.
2.  Gradient descent works on the simplex by *mirror descent*:
    a ← a * exp(‑step·gradient) and renormalisation, which keeps a ≥ 0
    and Σa = 1 without an Euclidean projection.
3.  The gradient of the “soft‑max’’ of the self‑convolution
    (instead of the pure max) provides a smoother descent direction.
    The soft‑max temperature `α` is increased during the run so that the
    early phase explores a smooth landscape and the late phase focuses
    on the true maximum.
4.  A light simulated‑annealing phase with mass‑transfer moves refines
    the final vector.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (temperature α) sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _subgradient_softmax(a: np.ndarray, conv: np.ndarray,
                         alpha: float = 10.0) -> np.ndarray:
    """
    Compute an (averaged) gradient of the soft‑max of `conv`,
    i.e.  g_i = 2 * Σ_k w_k * a_{k-i}   with   w_k ∝ exp(α·(conv_k−max)).
    """
    n = a.shape[0]
    max_val = conv.max()
    # weights for soft‑max (non‑negative, max weight = 1)
    w = np.exp(alpha * (conv - max_val))
    Z = w.sum()
    if Z == 0.0:             # numerical safety – fall back to uniform
        w = np.ones_like(w)
        Z = w.size

    # cross‑correlation w ⋆ a   (grad_i = 2 * (w ⋆ a)[i])
    a_rev = a[::-1]

    # length of the linear convolution w * a_rev
    conv_len = w.size + a_rev.size - 1          # = 3·n‑2
    fft_len = 1 << ((conv_len - 1).bit_length())

    Wf = np.fft.rfft(w, n=fft_len)
    Af = np.fft.rfft(a_rev, n=fft_len)
    prod = Wf * Af
    full = np.fft.irfft(prod, n=fft_len)[:conv_len]

    # slice the middle part (indices n‑1 … n‑1+n‑1) → length n
    grad = 2.0 * full[n - 1 : n - 1 + n] / Z
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent gradient descent with a gradually increasing α
# ----------------------------------------------------------------------
def _run_gradient_descent_smooth(a0: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 start_time: float,
                                 *,
                                 beta: float = 0.9,
                                 step0: float = 0.5,
                                 alpha0: float = 1.0,
                                 alpha_max: float = 30.0,
                                 alpha_steps: int = 20000,
                                 max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise  F(a)=2·n·max(a*a)  on the simplex using mirror descent.
    The temperature α of the soft‑max increases linearly from alpha0 to
    alpha_max over ``alpha_steps`` iterations.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # current convolution and objective
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth‑grad improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}")

        # temperature schedule for the soft‑max
        if it <= alpha_steps:
            alpha = alpha0 + (alpha_max - alpha0) * (it / alpha_steps)
        else:
            alpha = alpha_max

        # gradient of the smoothed max
        grad = _subgradient_softmax(a, conv, alpha)

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size  (√t‑schedule)
        step = step0 / math.sqrt(it)

        # mirror‑descent update  (multiplicative)
        a = a * np.exp(-step * momentum)

        # re‑normalise to the probability simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing that moves a fraction of the mass
    from a source index to a random destination.  Uses a geometric
    temperature schedule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source proportional to its mass (high‑mass indices move first)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis criterion
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear interpolation up‑sampling and re‑normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` linearly to length `new_n` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up.fill(1.0 / new_n)
    return up
