import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2*len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·conv_k) with respect to a.
    Returns a vector of shape (n,).
    """
    n = a.shape[0]

    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    # pick the slice that corresponds to indices 0..n‑1
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``seq`` to length ``new_n`` while preserving its sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _run_gradient_descent(a0: np.ndarray, n: int,
                          deadline: float, start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum update (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score

def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,

"""
Search for a non‑negative step function that minimizes ``evaluate_sequence``.

The program works in three stages:

1. *Exploration on a short sequence* (≈1500 entries).  Multiple random
   starts are optimised with Adam on a smooth approximation of the
   max‑convolution.  The best short candidate is kept.

2. *Upsampling* to the target length (n = 4800) and a long refinement that
   consists of
   – Adam optimisation with a schedule of increasing smoothing parameters,
   – a projected sub‑gradient polishing phase (exact max‑convolution),
   – a final simulated‑annealing fine‑tuning.

All phases respect the supplied time budget (≈1000 s).  The code uses only
numpy and standard‑library facilities and returns a list of floats ready
for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max:
        f(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                          deadline: float, start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7):
    """
    Projected sub‑gradient descent for min 2·n·max(conv(a,a)).
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max entry
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # projection onto the simplex
        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score

        # initialise either from the global starter (occasionally) or randomly
        if starter is not None and np.random.rand() < 0.2:
            a0 = _upsample(starter, n_small)
        else:
            a0 = np.random.rand(n_small)
            a0 /= a0.sum()
        a_opt, sc = _run_adam_optim(a0, n_small, small_deadline, start_time)
        if sc < best_small_score:
            best_small_score = sc
            best_small_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Small‑phase new best: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to the target length and fine‑tune
    # ------------------------------------------------------------------
    n_large = 4800
    if starter is not None:
        base_seq = starter
    elif best_small_seq is not None:
        base_seq = _upsample(best_small_seq, n_large)
    else:
        base_seq = np.full(n_large, 1.0 / n_large)

    a_final, final_score = _run_adam_optim(base_seq, n_large, deadline, start_time)

    # ------------------------------------------------------------------
    # Normalise and return the best candidate found
    # ------------------------------------------------------------------
    a_final = np.maximum(a_final, 0.0)
    tot = a_final.sum()
    if tot > 0:
        a_final /= tot
    else:
        a_final = np.full_like(a_final, 1.0 / n_large)

    print(f"[{time.time() - start_time:6.2f}s] Finished – best score {final_score:.6f}")
    return a_final.tolist()
