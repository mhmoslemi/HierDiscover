"""
Hybrid random‑restart / sub‑gradient optimizer for the
self‑convolution L∞ problem.

The algorithm works on the probability simplex (∑a_i = 1) because the
objective returned by `evaluate_sequence` is scale‑invariant.
For each chosen length we generate a diversified pool of symmetric
candidate shapes (uniform, sine, Hann, triangular, arcsine, Gaussian,
exponential, Beta‑family, random Dirichlet).  Every candidate is refined
by a simple projected sub‑gradient method on the exact max‑convolution.
The best vector found is up‑sampled and used as a warm‑start for the next
larger length.  The routine respects a 1000 s budget and returns the
best sequence discovered.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for all lengths)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)            # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate L_λ(a) = λ·log∑exp(conv/λ).
    """
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)           # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a*a)[k] at the current a.
    Returns (grad, max_value).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)                 # length 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Simple projected sub‑gradient optimiser (exact max‑conv objective)
# ----------------------------------------------------------------------
def _optimize(a_init: np.ndarray,
              n: int,
              time_budget: float,
              start_time: float) -> tuple[np.ndarray, float]:
    """
    Refine a candidate vector a_init on the simplex using sub‑gradient descent
    on the exact max‑convolution objective.
    Returns the best vector found and its objective value 2·n·max(conv).
    """
    a = a_init.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    deadline = time.time() + time_budget
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5               # enforce symmetry
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] refine n={n} it={it} score={cur_score:.6f}")

        # occasional random kick to escape plateaus
        if it % 500 == 0 and random.random() < 0.15:
            a = a + 0.02 * (np.random.rand(n) - 0.5)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    best_score = float('inf')
    best_seq = None
    global_a = None

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered solution if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                global_a = starter / s
    except Exception:
        pass

    # Length schedule – we start medium and grow.
    lengths = [4096, 8192, 16384, 32768]
    random.shuffle(lengths)  # avoid bias if we stop early

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- initialise a0 (warm‑start / random) --------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ----- generate a diversified set of start candidates ----------
        t = (np.arange(n) + 0.5) / n
        init_candidates = []

        # 1. warm‑start copy
        init_candidates.append(a0.copy())

        # 2. uniform
        init_candidates.append(np.full(n, 1.0 / n))

        # 3. half‑sine
        sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        init_candidates.append(sin_vals / sin_vals.sum())

        # 4. Hann (raised‑cosine) window
        hann = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
        init_candidates.append(hann / hann.sum())

        # 5. triangular
        left = np.arange(1, n // 2 + 1)
        right = np.arange(n // 2, 0, -1) if n % 2 == 0 else np.arange(n // 2 + 1, 0, -1)
        tri = np.concatenate([left, right])
        init_candidates.append(tri / tri.sum())

        # 6. arcsine (Beta(0.5,0.5))
        arcs = 1.0 / np.sqrt(t * (1 - t) + 1e-12)
        init_candidates.append(arcs / arcs.sum())

        # 7. Gaussian‑like (σ adjusted to cover most of the support)
        for sigma in [0.12, 0.18, 0.24]:
            gau = np.exp(-0.5 * ((np.arange(n) - (n - 1) / 2.0) / (sigma * n)) ** 2)
            init_candidates.append(gau / gau.sum())

        # 8. Exponential decay from the centre
        for beta in [0.001, 0.002, 0.005, 0.01]:
            dist = np.abs(np.arange(n) - (n - 1) / 2.0)
            expv = np.exp(-beta * dist)
            init_candidates.append(expv / expv.sum())

        # 9. Beta‑family (symmetric)
        for alpha in [0.2, 0.3, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0]:
            beta_vec = (t ** (alpha - 1)) * ((1 - t) ** (alpha - 1))
            init_candidates.append(beta_vec / beta_vec.sum())

        # 10. Random Dirichlet draws
        for _ in range(6):
            rnd = np.random.rand(n)
            init_candidates.append(rnd / rnd.sum())

        # make every candidate symmetric (most known optima are symmetric)
        init_candidates = [(c + c[::-1]) * 0.5 for c in init_candidates]

        # ------------------------------------------------------------------
        # Evaluate raw candidates quickly and keep the best to seed optimisation
        # ------------------------------------------------------------------
        for cand in init_candidates:
            if time.time() >= deadline:
                break
            conv = _conv_self(cand)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                global_a = best_seq.copy()
                print(f"[{time.time() - start_time:6.2f}s] raw candidate n={n} score={score:.6f}")

        # ------------------------------------------------------------------
        # Refine the current best with sub‑gradient descent (exact objective)
        # ------------------------------------------------------------------
        if best_seq is not None and time.time() < deadline:
            remaining = deadline - time.time()
            # give the optimiser at most a few seconds per length; it will stop earlier
            opt_time = min(remaining * 0.6, 30.0)
            a_opt, sc_opt = _optimize(best_seq, n, opt_time, start_time)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                global_a = best_seq.copy()
                print(f"[{time.time() - start_time:6.2f}s] refined n={n} score={sc_opt:.6f}")

        # ------------------------------------------------------------------
        # Small random restarts to escape possible local minima
        # ------------------------------------------------------------------
        if time.time() < deadline:
            restarts = 4
            for _ in range(restarts):
                if time.time() >= deadline:
                    break
                perturb = best_seq + 0.05 * (np.random.rand(n) - 0.5)
                perturb = np.maximum(perturb, 0.0)
                perturb = _project_simplex(perturb)
                perturb = (perturb + perturb[::-1]) * 0.5
                remaining = deadline - time.time()
                opt_time = min(remaining * 0.4, 20.0)
                a_opt, sc_opt = _optimize(perturb, n, opt_time, start_time)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_seq = a_opt.copy()
                    global_a = best_seq.copy()
                    print(f"[{time.time() - start_time:6.2f}s] restart improve n={n} score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024.0)

    # Return as plain Python list for the external evaluator
    return [float(x) for x in best_seq]
