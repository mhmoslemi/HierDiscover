def conv_fft(a, b):
    L = 1 << ((a.size + b.size - 2).bit_length())
    fa = np.fft.rfft(a, L)

def _run_weighted_adam(a0, n, p, steps, start_time, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    m=zeros; v=zeros
    for it in range(1, steps+1):
        if time.time()>=deadline: break
        conv = _conv_self(a) # length 2n-1
        # compute score
        score = 2*n*conv.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # compute weight
        w = conv ** (p-1)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.full_like(w, 1.0/(w.size))
        # compute gradient
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)
        grad = 2 * grad_full[n-1:n-1+n]  # central slice
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1 - beta1**it)
        v_hat = v/(1 - beta2**it)
        step_size = 0.5 / (np.sqrt(it) + 1e-6)   # tune
        a = a - step_size * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex

"""
Optimisation of a non‑negative step‑function for the self‑convolution peak.

The strategy builds a multi‑scale solution:
* start from the supplied global sequence (or a random Dirichlet vector);
* repeatedly up‑sample to larger lengths (4096 → 8192 → 16384 → 32768);
* on each length run a *p‑norm* Adam optimiser.  For a given p we
  minimise  Σ (a*a)[k]**p  (p large approximates the ∞‑norm) using the
  exact gradient obtained via FFT.
* after the p‑norm phases a short log‑sum‑exp Adam polish is applied;
* finally a light simulated‑annealing “mass‑transfer’’ step refines the
  solution.

All updates stay on the probability simplex (∑a=1, a≥0) via projection,
so the objective 2·n·max(a*a) is scale‑invariant and can be evaluated
with our fast FFT code.  The routine respects the 1000 s budget and
returns the best sequence found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic linear–algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    n = v.size
    if n == 0:
        return v
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # the vector is already on the simplex (or all entries are equal)
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using real FFT (fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear, not circular)."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(a*a) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient utilities for p‑norm (smooth) surrogate
# ----------------------------------------------------------------------


def _grad_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ w_k·(a*a)[k] w.r.t. a.
    The weight vector `w` has length 2·n‑1 and is assumed non‑negative.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # convolution of w with reversed a, length = (2n‑1)+n‑1 = 3n‑2
    tmp = _conv_fft(w, a_rev)
    # central slice corresponds to the gradient entries
    grad = 2.0 * tmp[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser for a given p‑norm surrogate
# ----------------------------------------------------------------------


def _adam_pnorm(a0: np.ndarray,
                p: float,
                slice_deadline: float,
                start_time: float,
                verbose: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the surrogate Σ (a*a)[k]**p .
    The weight vector is w = (a*a)**(p‑1) (normalised to sum = 1) which
    yields the exact gradient of the surrogate.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * a.shape[0] * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if verbose:
                print(f"[{time.time() - start_time:6.2f}s] p={p:.0f} it={it:5d} "
                      f"score={cur_score:.6f}")

        # build weight vector w = conv**(p‑1)  (avoid overflow for large p)
        # small additive epsilon keeps the gradient well‑behaved when conv=0
        w = np.power(conv + 1e-30, p - 1.0)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        grad = _grad_weighted(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size that slowly decays with iterations; the factor 0.5 proved
        # robust in practice but can be tweaked.
        step = 0.5 / math.sqrt(it + 1.0)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # keep a on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Log‑sum‑exp (smooth max) Adam (used for polishing)
# ----------------------------------------------------------------------


def _adam_logsumexp(a0: np.ndarray,
                    slice_deadline: float,
                    start_time: float,
                    verbose: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the λ·log‑sum‑exp surrogate of the max‑norm.
    The schedule for λ (controlled by `lam_factor`) mirrors the reference
    implementation but with a slightly more aggressive decay.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.5          # λ = lam_factor * max(conv)
    lam_decay = 0.85
    lam_decay_interval = 3000
    it = 0

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * a.shape[0] * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if verbose:
                print(f"[{time.time() - start_time:6.2f}s] log‑exp it={it:5d} "
                      f"score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)

        # smooth weights = softmax(conv/lam)
        max_val = conv.max()
        w = np.exp((conv - max_val) / lam)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        grad = _grad_weighted(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (mass transfer)
# ----------------------------------------------------------------------


def _anneal(a0: np.ndarray,
            deadline: float,
            start_time: float,
            cur_best_score: float,
            cur_best_seq: np.ndarray,
            start_temp: float = 3e-2,
            end_temp: float = 1e-5,
            max_steps: int = 200_000,
            verbose: bool = False) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing that respects the simplex."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    n = a.shape[0]
    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling schedule
        T = start_temp * (end_temp / start_temp) ** (step / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move a fraction of the source mass (up to 15 %)
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            if verbose:
                print(f"[{time.time() - start_time:6.2f}s] anneal it={step} "
                      f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling utilities
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
