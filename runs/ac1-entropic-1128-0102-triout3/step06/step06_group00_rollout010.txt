"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  The algorithm mixes three ideas:

1.  A quick exponentiated‑gradient warm‑up on a smoothed max‑convolution.
2.  Adam optimisation of the same smoothed objective (large alpha → true max).
   This yields many fast, high‑quality iterations.
3.  A long projected sub‑gradient polish and a short simulated‑annealing
   finish‑up.

All iterates stay on the probability simplex (sum = 1) and the routine
obeys the supplied time budget.  The code is deliberately self‑contained;
no external files or network access are required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max (quick warm‑up)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    """
    Multiplicative updates based on the log‑sum‑exp smooth approximation
    of max(conv). Returns the best iterate seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # --- smooth‑max gradient ------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # --- multiplicative update ---------------------------------
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Adam on the smoothed max (many fast iterations)
# ----------------------------------------------------------------------
def _run_adam_smooth(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     base_lr: float = 0.05,
                     alpha0: float = 5.0,
                     alpha_max: float = 200.0,
                     alpha_step_every: int = 2000):
    """
    Adam optimiser applied to the log‑sum‑exp smooth approximation of
    max(conv).  The gradient of the smooth objective is
        g = 2·n·2·corr(a, w_rev),
    where w is the soft‑max over the convolution values.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_score = float("inf")
    alpha = alpha0
    L = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: iter={t:,}, "
                  f"score={cur_score:.6f}, α={alpha:.1f}")

        # increase smoothing temperature slowly
        if t % alpha_step_every == 0:
            alpha = min(alpha * 1.3, alpha_max)

        # --- soft‑max weights ---------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # --- gradient of the smooth objective -----------------------
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]          # ∂max/∂a
        grad = grad * (2.0 * n)                                   # ∂f/∂a

        # --- Adam update --------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # --- projection onto simplex --------------------------------
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : exact‑max projected sub‑gradient descent (polish)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    """
    Classic (momentum‑accelerated) projected sub‑gradient descent on
      f(a)=2·n·max(conv(a,a)).
    Returns the best vector encountered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # exact sub‑gradient (all near‑max entries)
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 : tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000):
    """
    Mass‑transfer moves with exponential cooling.  Only a few hundred
    thousand attempts – enough to shake out very shallow local minima.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

            alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            eta0=0.5,
            steps_per_alpha=4000,
        )
        if score_warm < best_score:
            best_score, best_seq = score_warm, a_warm.copy()
            print(f"[{time.time() - start_time:.2f}s] after warm‑up score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on the smoothed max (major chunk of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.80 * remaining)

        a_adam, score_adam = _run_adam_smooth(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            base_lr=0.05,
            alpha0=5.0,
            alpha_max=200.0,
            alpha_step_every=2000,
        )
        if score_adam < best_score:
            best_score, best_seq = score_adam, a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] after Adam score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exact‑max projected sub‑gradient descent (polish)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # keep a few seconds for the final annealing pass
        phase3_deadline = time.time() + max(10.0, 0.90 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            n,
            phase3_deadline,
            start_time,
            step0=0.20,
            beta=0.95,
        )
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] after sub‑grad polish score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – brief annealing polish (any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if score_final < best_score:
            best_score, best_seq = score_final, a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] final best score={best_score:.6f}")
    return [float(x) for x in best_seq]
