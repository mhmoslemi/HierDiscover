The code below searches for a non‑negative step‑function (a probability vector) that minimizes  

``2·n·max(conv(a,a)) / (∑a)²``  

which is exactly the value returned by the supplied `evaluate_sequence`.  
The algorithm is a time‑bounded hybrid optimiser:

1. **Progressive up‑sampling** – optimisation starts on a short vector and the
   solution is repeatedly interpolated to a finer resolution.
2. **Smooth Adam optimisation** – a soft‑max approximation of the maximum
   convolution is minimised with the Adam optimiser on the simplex
   (exponentiated‑gradient + Euclidean projection).  The temperature of the
   soft‑max is annealed from `0.1` to `1e‑4`.
3. **Exact sub‑gradient descent with back‑tracking line‑search** – after the
   smooth phase the true max‑convolution is used.  The sub‑gradient points
   away from the entries that contribute most to the current maximal
   overlap; a simple line‑search guarantees a decrease whenever possible.
4. **Targeted simulated‑annealing** – the current maximal offset `k*` is
   identified, the pair of indices that contribute the most to it is selected,
   a small fraction of mass is moved to a random location and the move is
   accepted according to a slow cooling schedule.  Because the move is *targeted*
   the probability of a successful improvement is much higher than in a plain
   random walk.
5. **Multiple restarts** – each resolution stage is attempted a few times with
   different random seeds; the best intermediate solution is retained.

All phases respect the overall time budget (`budget_s` seconds).  The final
vector is returned as a plain Python list, ready for `evaluate_sequence`.

"""
Hybrid optimiser for the max‑convolution bound.
1. progressive up‑sampling,
2. Adam on a soft‑max approximation,
3. exact sub‑gradient descent with line‑search,
4. targeted simulated‑annealing.
All steps keep the vector on the probability simplex.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT, O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return np.max(_conv_self(a))


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up.fill(1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient – used by Adam
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) of the convolution.
    Returns ∇_a  ( T·log Σ exp(conv_k / T) ).
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()                     # soft weights, sum to 1
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]       # central n entries
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the smooth max
# ----------------------------------------------------------------------
def _run_softmax_adam(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step0: float = 0.15,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth max (log‑sum‑exp) of the self‑convolution.
    Temperature is geometrically annealed from 1e‑1 to 1e‑4.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    T_start = 1e-1
    T_end = 1e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        # smooth objective (the exact bound we finally care about)
        cur_obj = 2.0 * n * (T_start * (np.log(np.exp((conv - conv.max()) / T_start).sum())) + conv.max())
        # keep the *exact* bound for comparison
        exact_score = 2.0 * n * conv.max()
        if exact_score < best_score:
            best_score = exact_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it}, exact={exact_score:.6f}")

        # temperature schedule
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a_new)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with simple back‑tracking line search
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.8,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    A short back‑tracking line‑search guarantees descent whenever possible.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _max_conv(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        k = conv.argmax()

        # sub‑gradient of max(conv) at k
        grad = np.zeros_like(a)
        if k < n:
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * grad

        # back‑tracking line‑search
        step = step0 / math.sqrt(it)
        for _ in range(5):
            a_prop = _project_simplex(a - step * momentum)
            new_score = 2.0 * n * _max_conv(a_prop)
            if new_score < cur_score:
                a = a_prop
                break
            step *= 0.5     # shrink step if no improvement

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 0.2,
                            end_temp: float = 1e-7,
                            steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer walk that always picks a source index involved in the current
    maximal convolution.  The move is accepted with a Metropolis criterion.
    """
    a = a0.copy()
    score = 2.0 * n * _max_conv(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # compute current maximum offset and partner products
        conv = _conv_self(a)
        k = conv.argmax()
        # partner for each i (or -1 if out of range)
        partners = np.full(n, -1, dtype=int)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i
            partners[i] = j

        # contribution of each i to the maximal sum (ordered pairs counted twice)
        contrib = np.zeros(n)
        mask = partners >= 0
        contrib[mask] = a[mask] * a[partners[mask]]
        # double count for i!=partner
        double_mask = mask & (np.arange(n) != partners)
        contrib[double_mask] *= 2.0

        # select a source index proportionally to its contribution
        total = contrib.sum()
        if total == 0.0 or np.all(contrib == 0):
            i_src = np.random.randint(0, n)
        else:
            probs = contrib / total
            i_src = np.random.choice(n, p=probs)

        if a[i_src] <= 0.0:
            continue

        # pick a destination that is *not* the partner (to actually change the max)
        j_dst = np.random.randint(0, n)
        while j_dst == i_src:
            j_dst = np.random.randint(0, n)

        # move a modest fraction of the source mass
        eps = a[i_src] * np.random.rand() * 0.25   # up to 25 % of the source
        a_prop = a.copy()
        a_prop[i_src] -= eps
        a_prop[j_dst] += eps

        new_score = 2.0 * n * _max_conv(a_prop)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score
