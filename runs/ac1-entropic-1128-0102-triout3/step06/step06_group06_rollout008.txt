def conv_self(a):

def entropic_optimize(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = score_of(a)
    tau = 20.0 / n
    tau_end = tau * 1e-4
    decay = 0.9999
    eta0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] entropic improve n={n} it={it} score={cur_score:.6f}")
        shift = conv.max()
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)  # length = 3n-2
        # Extract gradient part: as before: idx = (n-1) - np.arange(n)
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]
        step = eta0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700, 700)
        a = a * np.exp(expo)
        a = normalize(a)
        tau = max(tau_end, tau * decay)
    return best_a, best_score

def mass_transfer_optimize(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = score_of(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_idx = np.argmax(conv)
        # gradient weight g
        j = max_idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        g = np.zeros(n)
        g[mask] = a[j[mask]]
        # choose source and target
        src = np.argmax(g)
        dst = np.argmin(g)
        if src == dst:
            # random pair fallback
            i, jrand = np.random.randint(0, n, size=2)
            src, dst = i, jrand
        # epsilon proportion
        eps = min(a[src] * 0.5, 0.001)  # small step
        if eps <= 0:
            # can't move, break
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = normalize(a_prop)
        new_score = score_of(a_prop)
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] mass_transfer improve n={n} it={it} score={new_score:.6f}")
        else:
            # maybe accept with small probability to escape

    return best_a, best_score

"""
Propose a step‑function that minimises `evaluate_sequence`.

The search consists of three phases:

1. **Broad warm‑start** – generate many analytic shapes (product‑power,
   Gaussian, Beta, arcsine, symmetric exponential) for a moderate size
   (≈ 4 k). Keep the best one according to the exact objective
   `2·n·max(conv)`.

2. **Entropic mirror descent** – minimise a smooth surrogate of the
   ∞‑norm (`τ·log ∑exp(conv/τ)`) on the probability simplex by an
   exponentiated‑gradient update.  The temperature τ starts larger than in
   the reference implementation and decays slowly, allowing a much more
   thorough optimisation on the final large size (≈ 32 k).

3. **Targeted mass‑transfer polishing** – repeatedly move a small amount
   of mass from indices that contribute most to the current maximal
   convolution entry to those that contribute least.  Moves are kept
   only if they improve the exact score.  This simple greedy step often
   squeezes the bound a few hundredths lower.

The algorithm respects the non‑negativity and normalisation constraints
required by `evaluate_sequence`.  It adapts the time budget dynamically
and returns the best vector found before the deadline.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Return the smallest power‑of‑two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Normalisation / scoring helpers
# ----------------------------------------------------------------------


def normalise(a: np.ndarray) -> np.ndarray:
    """Project a onto the probability simplex (non‑negative, sum = 1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s
    return a


def score_of(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    return normalise(up)


# ----------------------------------------------------------------------
# Parametric families for warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = term ** p
    return normalise(a)


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return normalise(a)


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    return normalise(a)


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = 1.0 / np.sqrt(term)
    return normalise(a)


def shape_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the ends: a_i ∝ exp(-λ·dist)."""
    i = np.arange(n)
    dist = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * dist)
    return normalise(a)


# ----------------------------------------------------------------------
# Entropic (soft‑max) mirror descent on the simplex
# ----------------------------------------------------------------------


def entropic_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv) via a smooth surrogate.
    Uses exponentiated‑gradient updates with a slowly decaying temperature.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # temperature schedule (larger start, slower decay)
    tau = 20.0 / n            # start a bit hotter than the reference
    tau_end = tau * 1e-4      # final temperature
    decay = 0.9999

    eta0 = 0.5                # step‑size prefactor
    it = 0

    while time.time() < deadline:
        it += 1

        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] entropic improve n={n} it={it} score={cur_sc:.6f}"
            )

        # soft‑max weights
        shift = conv.max()
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate (cross‑correlation)
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)          # length = 3n‑2
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # exponentiated‑gradient update
        step = eta0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)   # prevent overflow
        a = a * np.exp(expo)
        a = normalise(a)

        # anneal temperature
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Targeted mass‑transfer polishing (greedy hill climbing)
# ----------------------------------------------------------------------


def mass_transfer_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Greedy improvement: move a small ε from an index that contributes most
    to the current maximal convolution entry to one that contributes least.
    Accept only improving moves.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_idx = int(np.argmax(conv))

        # compute contribution weights g_i = a_{max_idx-i} (zero outside support)
        j = max_idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        g = np.zeros(n)
        g[mask] = a[j[mask]]

        # source: largest g where we actually have mass;
        # target: smallest g (could be zero).
        src = np.argmax(g)
        dst = np.argmin(g)

        if src == dst or a[src] <= 0.0:
            # fallback random move
            src, dst = np.random.randint(0, n, size=2)
            if src == dst or a[src] <= 0.0:
                continue

        # step size – a modest fraction of the source mass
        eps = min(a[src] * 0.4, 0.001)
        if eps <= 0.0:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = normalise(a_prop)

        new_sc = score_of(a_prop)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(
                f"[{time.time() - start_time:6.2f}s] mass‑transfer improve n={n} it={it} score={new_sc:.6f}"
            )
        # else reject the move (no simulated‑annealing here – greedy is enough)

    return best_a, best_sc
