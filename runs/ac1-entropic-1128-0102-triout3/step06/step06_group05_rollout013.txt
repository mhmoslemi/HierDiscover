def generate_shape(n, shape_type, param=None):
    t = (np.arange(n) + 0.5) / n
    if shape_type == 'arcsine':
        vals = 1.0 / np.sqrt(t * (1 - t) + 1e-12)
    elif shape_type == 'beta':
        a, b = param
        vals = (t ** (a - 1)) * ((1 - t) ** (b - 1) + 1e-12)
    elif shape_type == 'triangular':
        mid = 0.5
        vals = np.where(t <= mid, t / mid, (1 - t) / (1 - mid))
    elif shape_type == 'sine':
        vals = np.sin(np.pi * t)
    elif shape_type == 'uniform':
        vals = np.ones_like(t)
    else:
        raise ValueError
    vals = np.maximum(vals, 0)
    s = vals.sum()
    if s > 0:
        vals /= s
    return vals

```
def _fast_anneal(a, conv, deadline, start_time, cur_best_score, cur_best_seq, start_temp=0.01, end_temp=1e-6, reeval_interval=50000):
    n = a.shape[0]
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = 2.0 * n * conv.max()
    move = 0
    while time.time() < deadline:
        # compute temperature based on elapsed fraction of remaining time
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        T = start_temp * (1 - frac) + end_temp * frac
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j or a[i] <= 0:
            continue
        # epsilon up to 10% of source mass
        eps = a[i] * np.random.rand() * 0.1
        if eps <= 0:
            continue
        # compute candidate conv
        conv_new = conv.copy()
        # linear updates
        conv_new[i:i+n] -= 2.0 * eps * a
        conv_new[j:j+n] += 2.0 * eps * a
        # quadratic updates
        idx2i = 2*i
        idx2j = 2*j
        idxij = i + j
        L = conv_new.shape[0]
        if idx2i < L:
            conv_new[idx2i] += eps*eps
        if idx2j < L:
            conv_new[idx2j] += eps*eps
        if idxij < L:
            conv_new[idxij] -= 2.0 * eps*eps
        # clip negative
        np.maximum(conv_new, 0.0, out=conv_new)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - cur_score
        accept = delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12))
        if accept:
            a[i] -= eps
            a[j] += eps
            conv = conv_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] fast_anneal improve n={n} move={move} score={cur_score:.6f}")
        move += 1
        # periodic recompute to avoid drift
        if move % reeval_interval == 0:
            s = a.sum()
            if s != 0:
                a /= s
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] fast_anneal recompute n={n} move={move} score={cur_score:.6f}")
    return best_seq, best_score
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

which is *scale‑invariant*.  After normalising the vector to the probability
simplex \(\sum_i a_i=1\) the objective becomes simply  

\[
\text{score}=2\,n\;\max (a*a).
\]

The optimiser works in three stages:

1. **A good starting shape** – several analytically‑motivated shapes
   (arcsine, Beta‑distributions, sine, uniform…) are generated for a
   moderate size ( 16 384 ).  The one with the smallest initial score is
   taken as the seed.

2. **Fast simulated‑annealing on the simplex** – mass is transferred from a
   source index `i` to a destination index `j`.  
   The self‑convolution is updated **incrementally** in  
   \(\mathcal O(n)\) using the exact formula  

\[
(a+\Delta)*(a+\Delta)=a*a
+2\Delta*a+\Delta*\Delta,
\]

   so a single move costs only two slice operations and a `max` scan.
   Acceptance follows a temperature schedule that cools linearly with the
   elapsed time.

3. **Hierarchical up‑sampling** – after a fixed budget on the current length
   the best vector is linearly interpolated to a larger size (doubling the
   length) and the annealing phase is repeated.  The final length is chosen
   as large as the remaining time budget allows (up to 32768, which safely
   fits into memory).

All vectors are kept on the simplex (moving mass preserves the sum) and the
best vector seen so far is returned as a plain Python list.  The whole
procedure respects the 1000 s budget and prints occasional progress messages.



```python
"""
Hybrid optimiser for the self‑convolution L∞ problem.

Algorithm
---------

1. Build a handful of analytic base shapes (arcsine, Beta‑distributions,
   sine, uniform) on a modest size (n≈16384) and pick the one with the
   smallest initial score 2·n·max(a*a).

2. Run a fast simulated‑annealing loop that moves a small ε‑mass from a
   source index i to a destination j.  The convolution is updated
   incrementally (O(n) slice operations) using the exact formula

       conv' = conv - 2·ε·a shifted by i + 2·ε·a shifted by j
               + ε²·δ_{2i} + ε²·δ_{2j} - 2·ε²·δ_{i+j} .

   Occasionally the convolution is recomputed from scratch (FFT) to
   eliminate accumulated numerical drift.

3. When the time budget for the current length is exhausted, up‑sample the
   best vector to the next larger length (doubling each time) and repeat
   the annealing phase.  The final best vector is returned as a list of
   floats, ready for `evaluate_sequence`.

The optimiser stays inside the 1000 s limit, prints intermediate scores,
and works entirely on the probability simplex (no explicit projection
required because ε‑moves preserve the sum).
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities ------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using FFT (fast for all lengths)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)   # eliminate tiny negatives
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape generators ------------------------------------------------------
# ----------------------------------------------------------------------


def _shape_arcsine(n: int) -> np.ndarray:
    t = (np.arange(n) + 0.5) / n
    vals = 1.0 / np.sqrt(t * (1.0 - t) + 1e-12)
    vals /= vals.sum()
    return vals


def _shape_beta(n: int, a: float, b: float) -> np.ndarray:
    t = (np.arange(n) + 0.5) / n
    vals = (t ** (a - 1.0)) * ((1.0 - t) ** (b - 1.0) + 1e-12)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _shape_sine(n: int) -> np.ndarray:
    t = (np.arange(n) + 0.5) / n
    vals = np.sin(np.pi * t)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _shape_uniform(n: int) -> np.ndarray:
    vals = np.ones(n, dtype=float)
    vals /= vals.sum()
    return vals


def _choose_initial_shape(n: int) -> np.ndarray:
    """Return the best analytic shape for length n (lowest initial score)."""
    candidates = [
        (_shape_arcsine, ()),
        (_shape_beta, (0.3, 0.3)),
        (_shape_beta, (0.4, 0.4)),
        (_shape_sine, ()),
        (_shape_uniform, ()),
    ]
    best_seq = None
    best_score = float("inf")
    for fun, args in candidates:
        seq = fun(n, *args) if args else fun(n)
        conv = _conv_self(seq)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = seq
    # try to use a previously found global sequence if available
    try:
        global_seq = np.asarray(height_sequence_1, dtype=float)
        if global_seq.ndim == 1 and global_seq.size > 0:
            gs = np.maximum(global_seq, 0.0)
            s = gs.sum()
            if s > 0.0:
                gs = gs / s
                if gs.shape[0] != n:
                    gs = _upsample(gs, n)
                conv = _conv_self(gs)
                score = 2.0 * n * conv.max()
                if score < best_score:
                    best_score = score
                    best_seq = gs
    except Exception:
        pass
    return best_seq

# Fast annealing with incremental convolution update -------------------
# ----------------------------------------------------------------------


def _fast_anneal(a: np.ndarray,
                 conv: np.ndarray,
                 deadline: float,
                 start_time: float,
                 cur_best_score: float,
                 cur_best_seq: np.ndarray,
                 start_temp: float = 0.02,
                 end_temp: float = 1e-6,
                 reeval_interval: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Greedy+annealing mass‑transfer moves.

    Parameters
    ----------
    a               : current probability vector (sum = 1)
    conv            : a*a (length 2n‑1)
    deadline        : absolute time when the run must stop
    start_time      : global start time (for temperature schedule)
    cur_best_score  : best score seen so far (2·n·max(conv))
    cur_best_seq    : copy of the best sequence
    start_temp      : temperature at the beginning of this phase
    end_temp        : temperature at the end of this phase
    reeval_interval : how many moves before a full FFT recompute

    Returns
    -------
    best_seq, best_score
    """
    n = a.shape[0]
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = 2.0 * n * conv.max()
    move = 0

    while time.time() < deadline:
        # linear cooling based on elapsed fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(elapsed / total, 1.0)
        T = start_temp * (1.0 - frac) + end_temp * frac

        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j or a[i] <= 0.0:
            continue

        # transfer up to 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.10
        if eps <= 0.0:
            continue

        # --------------------------------------------------------------
        # Incremental convolution update (exact)
        # --------------------------------------------------------------
        conv_new = conv.copy()

        # linear part: 2·ε·(a shifted by j – a shifted by i)
        conv_new[i:i + n] -= 2.0 * eps * a
        conv_new[j:j + n] += 2.0 * eps * a

        # quadratic part: ε² at 2i, 2j and -2ε² at i+j
        L = conv_new.shape[0]
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        if idx2i < L:
            conv_new[idx2i] += eps * eps
        if idx2j < L:
            conv_new[idx2j] += eps * eps
        if idxij < L:
            conv_new[idxij] -= 2.0 * eps * eps

        np.maximum(conv_new, 0.0, out=conv_new)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - cur_score

        accept = delta < 0.0 or (np.random.rand() < math.exp(-delta / max(T, 1e-12)))
        if accept:
            a[i] -= eps
            a[j] += eps
            conv = conv_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] fast_anneal improve n={n} move={move} score={cur_score:.6f}")

        move += 1

        # --------------------------------------------------------------
        # Periodic full recomputation (keeps numerical errors in check)
        # --------------------------------------------------------------
        if move % reeval_interval == 0:
            a_sum = a.sum()
            if a_sum != 0.0:
                a /= a_sum
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] fast_anneal recompute n={n} move={move} score={cur_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    The routine respects the `budget_s` time limit, performs a fast
    simulated‑annealing optimisation on several length scales and returns a
    Python list of the best sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a safety margin for the wrapper
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Stage 0 – pick an analytically good start (n ≈ 16384)
    # ------------------------------------------------------------------
    n0 = 16384
    a = _choose_initial_shape(n0)

    # ------------------------------------------------------------------
    # Stage 1 – fast annealing at n0
    # ------------------------------------------------------------------
    conv = _conv_self(a)
    best_seq = a.copy()
    best_score = 2.0 * n0 * conv.max()
    print(f"[{time.time() - start_time:6.2f}s] initial n={n0} score={best_score:.6f}")

    # allocate a fraction of the remaining budget to this size
    remaining = deadline - time.time()
    if remaining > 0.0:
        slice_deadline = time.time() + max(20.0, remaining * 0.30)
        best_seq, best_score = _fast_anneal(
            a,
            conv,
            slice_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=best_score * 0.02,
            end_temp=1e-6,
            reeval_interval=80_000,
        )

    # ------------------------------------------------------------------
    # Hierarchical up‑sampling & further annealing
    # ------------------------------------------------------------------
    lengths = [32768, 65536]          # try larger sizes if time permits
    for n in lengths:
        if time.time() >= deadline:
            break
        # up‑sample the current best vector
        a = _upsample(best_seq, n)
        # keep it on the simplex
        a = a / a.sum()
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] upsample to n={n} score={cur_score:.6f}")

        # allocate time proportionally to the remaining budget
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_deadline = time.time() + max(10.0, remaining * 0.45)

        best_seq, best_score = _fast_anneal(
            a,
            conv,
            slice_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=best_score * 0.02,
            end_temp=1e-6,
            reeval_interval=120_000,
        )

    # ------------------------------------------------------------------
    # Final safety: if nothing was ever found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096)

    # Return as a plain Python list (evaluate_sequence will clamp as needed)
    return [float(v) for v in best_seq]
```
