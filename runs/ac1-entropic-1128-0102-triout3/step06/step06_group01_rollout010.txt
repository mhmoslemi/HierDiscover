def _project_simplex(v):
    # Standard algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step function with a low
evaluate_sequence() value.

Algorithm
---------
1. Choose a moderate problem size (default n≈9600).  If a
   previously‑found construction `height_sequence_1` exists it is
   interpolated to that length, otherwise a random Dirichlet vector
   is used.

2. Run a smooth Adam optimisation on the log‑softmax parametrisation.
   The objective is a log‑sum‑exp smooth approximation of the
   self‑convolution maximum.  The smoothing parameter α is increased
   gradually (continuation) and a few thousand iterations are spent
   at each α.

3. Polishing phase – projected sub‑gradient descent on the *exact*
   max‑convolution.  The sub‑gradient for the active (maximal)
   convolution entries is obtained efficiently with FFTs
   (no O(n²) loops).  Momentum and a diminishing stepsize are used.
   This phase runs for the bulk of the remaining time.

4. A short simulated‑annealing refinement moves a small fraction of
   mass from a source index to a destination index.  The temperature
   schedule is geometric.

5. Multiple restarts are performed (the best solution is kept) by
   perturbing the current best vector before the Adam stage.

The whole routine respects the supplied time budget (≈1000 s) and
returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max:
        f(a)= (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _grad_active_submax(a: np.ndarray, conv: np.ndarray,
                        active: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) for the active set `active`.
    Uses the same FFT trick as _grad_smooth_max but with a 0/1 mask.
    """
    n = a.shape[0]
    w = np.zeros_like(conv)
    w[active] = 1.0
    w_rev = w[::-1]

    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up[:] = 1.0 / new_n
    return up


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Smooth Adam optimisation on the log‑softmax parametrisation.
    Returns the best vector seen and its exact evaluate_sequence score.
    """
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))

    fft_len = _next_pow2(3 * n - 2)

    # continuation schedule for the smoothing parameter α
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 3000

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a0.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # soft‑max
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # smoothed objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val

            # gradient w.r.t a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient w.r.t logits y (soft‑max Jacobian)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         step0: float = 0.25, beta: float = 0.94) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses FFT‑based sub‑gradient computation (fast) and a simple
    momentum scheme.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    it = 0
    fft_len = _next_pow2(3 * n - 2)
    rel_tol = 1e-12

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # active set (indices attaining the maximum)
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # FFT‑based sub‑gradient
        grad = _grad_active_submax(a, conv, active, fft_len)

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score


def _annealing(a0: np.ndarray, n: int,
               deadline: float, start_time: float,
               cur_best_score: float, cur_best_seq: np.ndarray,
               start_temp: float = 5e-2, end_temp: float = 1e-6,
               steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer simulated annealing.
    Moves a random ε‑fraction of mass from a source index to a
    destination index, accepting according to a geometric temperature schedule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The method combines:
        * smooth Adam optimisation with continuation in α,
        * fast projected sub‑gradient polishing (FFT‑based),
        * a short simulated‑annealing fine‑tuning,
    and runs several restarts within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Problem size and initial construction
    # ------------------------------------------------------------------
    default_n = 9600               # larger than the previous 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            s = starter.sum()
            if s > 0:
                starter = starter / s
            else:
                starter = np.full_like(starter, 1.0 / starter.size)
        a0 = _upsample(starter, default_n)
    except Exception:
        a0 = rng.random(default_n)
        a0 /= a0.sum()

    n = a0.shape[0]

    # ------------------------------------------------------------------
    # Keep track of the globally best solution
    # ------------------------------------------------------------------
    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[0.00s] initial score (n={n}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main loop – a few restarts with time budgeting
    # ------------------------------------------------------------------
    remaining_restarts = 4
    while time.time() < deadline and remaining_restarts > 0:
        now = time.time()
        time_left = deadline - now
        if time_left < 30.0:            # not enough time for a full restart
            break

        # allocate fractions of the remaining time for the phases
        adam_frac = 0.35
        subgrad_frac = 0.55
        anneal_frac = max(0.0, 1.0 - adam_frac - subgrad_frac)

        adam_deadline = now + time_left * adam_frac
        subgrad_deadline = adam_deadline + time_left * subgrad_frac
        final_deadline = deadline   # the annealing phase uses whatever is left

        # ------------------------------------------------------------------
        # Starting point for this restart – a perturbed version of the current best
        # ------------------------------------------------------------------
        if remaining_restarts == 4:
            start_vec = best_seq.copy()
        else:
            # random Dirichlet‑like perturbation
            perturb = rng.random(n)
            mix = 0.7 * rng.random() + 0.2   # between 0.2 and 0.9
            start_vec = mix * best_seq + (1.0 - mix) * perturb
            start_vec = np.maximum(start_vec, 0.0)
            start_vec /= start_vec.sum()

        # ------------------------------------------------------------------
        # Smooth Adam optimisation
        # ------------------------------------------------------------------
        a_adam, sc_adam = _adam_optimize(start_vec, n, adam_deadline, start_time)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:6.2f}s] New best after Adam: {best_score:.6f}")

        # ------------------------------------------------------------------
        # Fast sub‑gradient polishing (major part of remaining time)
        # ------------------------------------------------------------------
        a_pg, sc_pg = _subgradient_descent(best_seq, n, subgrad_deadline, start_time)
        if sc_pg < best_score:
            best_score = sc_pg
            best_seq = a_pg.copy()
            print(f"[{time.time() - start_time:6.2f}s] New best after polishing: {best_score:.6f}")

        # ------------------------------------------------------------------
        # Brief annealing refinement (if any time left)
        # ------------------------------------------------------------------
        if time.time() < final_deadline and anneal_frac > 0.0:
            # estimate a reasonable number of annealing steps
            rem = final_deadline - time.time()
            # about 200 k steps per 20 s on a typical machine → scale linearly
            steps = int(min(400_000, max(10_000, rem * 10_000)))
            a_ann, sc_ann = _annealing(best_seq, n,
                                       final_deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=5e-2,
                                       end_temp=1e-7,
                                       steps=steps)
            if sc_ann < best_score:
                best_score = sc_ann
                best_seq = a_ann.copy()
                print(f"[{time.time() - start_time:6.2f}s] New best after annealing: {best_score:.6f}")

        remaining_restarts -= 1

    # ------------------------------------------------------------------
    # Final clean‑up: ensure non‑negativity and normalisation
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"[{time.time() - start_time:6.2f}s] final best score (n={n}): {best_score:.6f}")
    return [float(x) for x in best_seq.tolist()]
