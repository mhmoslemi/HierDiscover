# Helper: Euclidean projection onto the probability simplex

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)

    s = up.sum()
    if s > 0:

    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
