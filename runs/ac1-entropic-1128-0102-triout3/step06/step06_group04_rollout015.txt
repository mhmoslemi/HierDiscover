"""
Improved search for a step‑function minimizing 2·n·max(conv).

The algorithm:
* Starts from the best known sequence (global variable `height_sequence_1`);
  if unavailable it creates a random Dirichlet start.
* Grows the length gradually (256 → … → target length up to 16384).
* For each length it runs:
    – a short projected‑Adam optimisation on a smooth log‑sum‑exp surrogate,
    – a cheap LP “swap’’ step,
    – a sub‑gradient descent on the true objective,
    – a simulated‑annealing polish.
* Finally a *targeted local‑search* phase repeatedly moves a small
  amount of mass from the entry that participates most in the current
  maximal convolution value to a random receiver.  The move is evaluated
  with an exact FFT convolution; only improving moves are kept.
  The phase keeps the best sequence ever seen and runs until the time
  budget expires.

All operations stay on the probability simplex (∑a_i = 1), hence the
objective is scale‑invariant.  The routine respects the 1000 s budget
and always returns a valid non‑negative sequence.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real, fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using FFT."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact evaluation (∑a≈1).  Returns 2·n·max(conv)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of L_λ(a) = λ·log Σ exp(conv/λ).
    """
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)            # length 3n‑2
    grad = 2.0 * tmp[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth surrogate; returns best encountered."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3          # λ = lam_factor * max(conv)
    lam_decay = 0.8
    lam_decay_interval = 3000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)
        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming swap step
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A with entries
        A_{k,i} = f_i   if   i ≤ k < i + n,
                = 0      otherwise.
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _run_lp_improve(a: np.ndarray,
                    n: int,
                    step: float = 0.2,
                    max_iters: int = 3,
                    deadline: float = None,
                    start_time: float = None) -> np.ndarray:
    """Repeated LP‑based improvement of a."""
    a_cur = a.copy()
    for _ in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        conv = _conv_self(a_cur)
        M = conv.max()
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(b)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30.0})

        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b / sum_b
        a_cur = (1.0 - step) * a_cur + step * g
        a_cur = _project_simplex(a_cur)

        if start_time is not None:
            sc = _objective(a_cur)
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} step={step:.2f} score={sc:.6f}")

    return a_cur

# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         slice_deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        max_idx = np.flatnonzero(conv == M)

        grad = np.zeros_like(a)
        for k in max_idx:
            # First half of the convolution (i ≤ k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            grad[i_range] += a[j_range]
            # Second half (symmetry)
            i_start2 = max(k, 0)
            i_range2 = np.arange(i_start2, n)
            j_range2 = i_range2 - k
            grad[i_range2] += a[j_range2]

        # step size
        step = 0.3 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score

def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 150000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _targeted_local_search(a0: np.ndarray,
                           deadline: float,
                           start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small amount of mass from a coordinate that
    participates most in the current maximal convolution value to a
    random receiver.  Exact convolutions are recomputed after every
    accepted move; the best sequence is remembered.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    it = 0

    # Pre‑select indices that currently hold non‑zero mass
    nonzero_mask = a > 1e-12

    while time.time() < deadline:
        it += 1

        # Occasionally recompute the convolution from scratch to
        # eliminate accumulated rounding error.
        if it % 500 == 0:
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] LS recompute n={n} it={it} score={cur_score:.6f}")

        # 1) locate the index k with maximal convolution value
        k_max = int(conv.argmax())

        # 2) among all pairs (i, j) with i + j = k_max pick the one with
        #    the largest product a[i] * a[j].
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        idx = np.argmax(contrib)
        donor = i_range[idx]                     # the larger member of the pair
        partner = j_range[idx]

        # If the chosen donor is (almost) empty, pick a random non‑zero entry.
        if a[donor] < 1e-12:
            nz = np.where(a > 1e-12)[0]
            if nz.size == 0:
                break
            donor = np.random.choice(nz)

        # 3) pick a random receiver different from donor
        receiver = np.random.randint(0, n)
        while receiver == donor:
            receiver = np.random.randint(0, n)

        # 4) move a small fraction of the donor's mass
        eps = a[donor] * np.random.rand() * 0.3 + 1e-12
        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps

        # 5) evaluate the exact objective for the trial vector
        conv_new = _conv_self(a_new)
        new_score = 2.0 * n * conv_new.max()

        # 6) accept if the objective improves
        if new_score < cur_score:
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] LS improve n={n} it={it} score={new_score:.6f}")

        # Occasionally perform a completely random cheap move to help escape
        if np.random.rand() < 0.03:
            nz = np.where(a > 1e-12)[0]
            if nz.size > 1:
                donor = np.random.choice(nz)
                receiver = np.random.randint(0, n)
                while receiver == donor:
                    receiver = np.random.randint(0, n)
                eps = a[donor] * np.random.rand() * 0.1 + 1e-12
                a_rand = a.copy()
                a_rand[donor] -= eps
                a_rand[receiver] += eps
                conv_rand = _conv_self(a_rand)
                rand_score = 2.0 * n * conv_rand.max()
                if rand_score < cur_score:
                    a = a_rand
                    conv = conv_rand
                    cur_score = rand_score
                    if rand_score < best_score:
                        best_score = rand_score
                        best_a = a.copy()
                        print(f"[{time.time() - start_time:6.2f}s] LS random improve n={n} it={it} score={rand_score:.6f}")

    return best_a, best_score

"""
Improved search for a step‑function minimizing 2·n·max(conv).

The algorithm:
* Starts from the best known sequence (global `height_sequence_1`);
  falls back to a random Dirichlet start.
* Grows the length gradually (256 → … → up to 16384) and for each
  length applies a pipeline:
    – Projected Adam optimisation on a log‑sum‑exp surrogate,
    – A cheap LP “swap’’ step,
    – Sub‑gradient descent on the true max‑norm,
    – Simulated‑annealing polish,
    – **Targeted local‑search** (new component) that repeatedly moves a
      small amount of mass from the entry that contributes most to the
      current maximal convolution value to a random receiver.
      Exact convolutions are recomputed after every accepted move;
      the best sequence seen is kept.
* A final polishing stage repeats Adam (with optional symmetry) and the
  targeted local‑search on the largest length attained.
All steps stay on the probability simplex (∑a_i = 1), so the objective
`2·n·max(conv)` is scale‑invariant.  The routine respects the 1000 s
budget and always returns a valid non‑negative list.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact objective (∑a≈1): 2·n·max(conv)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of L_λ(a)=λ·logΣexp(conv/λ)."""
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)                # length 3n‑2
    grad = 2.0 * tmp[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam on the surrogate; returns best seen."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lam_factor, lam_decay, lam_step = 0.3, 0.8, 3000
    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")
        if it % lam_step == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(conv.max() * lam_factor, 1e-12)
        grad = _gradient_from_conv(a, conv, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.6 / math.sqrt(it + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        if symmetrize:
            a = (a + a[::-1]) * 0.5
        a = _project_simplex(a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming swap step
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """Build (2n‑1)×n matrix A such that A·x ≤ M ⇔ (f*x)≤M."""
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _run_lp_improve(a: np.ndarray,
                    n: int,
                    step: float = 0.2,
                    max_iters: int = 3,
                    deadline: float = None,
                    start_time: float = None) -> np.ndarray:
    """Iterated LP improvement of a."""
    a_cur = a.copy()
    for _ in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        conv = _conv_self(a_cur)
        M = conv.max()
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)            # maximise sum(b)
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if not res.success:
            break
        b = res.x
        sm = b.sum()
        if sm <= 1.0 + 1e-12:
            break
        g = b / sm
        a_cur = (1.0 - step) * a_cur + step * g
        a_cur = _project_simplex(a_cur)
        if start_time is not None:
            sc = _objective(a_cur)
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} step={step:.2f} score={sc:.6f}")
    return a_cur


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         slice_deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑norm."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        max_idx = np.flatnonzero(conv == M)
        grad = np.zeros_like(a)
        for k in max_idx:
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)
            i_range = np.arange(i0, i1 + 1)
            j_range = k - i_range
            grad[i_range] += a[j_range]
            i0b = max(k, 0)
            i_range2 = np.arange(i0b, n)
            j_range2 = i_range2 - k
            grad[i_range2] += a[j_range2]
        step = 0.3 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polish (mass‑transfer)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 150000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        new_score = _objective(a_prop)
        delta = new_score - score
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to new length and renormalisation to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Targeted local‑search (new component)
# ----------------------------------------------------------------------


def _targeted_local_search(a0: np.ndarray,
                           deadline: float,
                           start_time: float) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from the entry that contributes most to the
    current maximal convolution value to a random receiver.
    Exact convolutions are recomputed after every accepted move.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    it = 0

    while time.time() < deadline:
        it += 1

        if it % 500 == 0:                     # clean‑up occasional recompute
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] LS recompute n={n} it={it} score={cur_score:.6f}")

        # 1) index of the maximal convolution entry
        k_max = int(conv.argmax())

        # 2) find the pair (i, j) with i+j=k_max that maximises a[i]*a[j]
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        idx = np.argmax(contrib)
        donor = i_range[idx]          # one member of the worst pair
        partner = j_range[idx]

        if a[donor] < 1e-12:          # fallback to any non‑zero entry
            nz = np.where(a > 1e-12)[0]
            if nz.size == 0:
                break
            donor = np.random.choice(nz)

        # 3) pick a random receiver distinct from donor
        receiver = np.random.randint(0, n)
        while receiver == donor:
            receiver = np.random.randint(0, n)

        # 4) move a tiny fraction of donor's mass
        eps = a[donor] * np.random.rand() * 0.3 + 1e-12
        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps

        # 5) evaluate exact objective
        conv_new = _conv_self(a_new)
        new_score = 2.0 * n * conv_new.max()

        # 6) accept only improving moves
        if new_score < cur_score:
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] LS improve n={n} it={it} score={new_score:.6f}")

        # occasional random cheap move to help escape plateaus
        if np.random.rand() < 0.03:
            nz = np.where(a > 1e-12)[0]
            if nz.size > 1:
                donor = np.random.choice(nz)
                receiver = np.random.randint(0, n)
                while receiver == donor:
                    receiver = np.random.randint(0, n)
                eps = a[donor] * np.random.rand() * 0.1 + 1e-12
                a_rand = a.copy()
                a_rand[donor] -= eps
                a_rand[receiver] += eps
                conv_rand = _conv_self(a_rand)
                rand_score = 2.0 * n * conv_rand.max()
                if rand_score < cur_score:
                    a = a_rand
                    conv = conv_rand
                    cur_score = rand_score
                    if rand_score < best_score:
                        best_score = rand_score
                        best_a = a.copy()
                        print(f"[{time.time() - start_time:6.2f}s] LS random improve n={n} it={it} score={rand_score:.6f}")

    return best_a, best_score

# Main entry point required by the platform

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Warm‑start from a known construction if present
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # --------------------------------------------------------------
    # Length schedule – start small, end with the largest length affordable
    # --------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    random.shuffle(lengths)

    global_a = starter
    best_score = float('inf')
    best_seq = None

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # ----- initialise a0 -------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()
        if np.random.rand() < 0.3:            # occasional symmetry boost
            a0 = (a0 + a0[::-1]) * 0.5
            a0 = _project_simplex(a0)

        # ----- allocate a time slice for this length -----------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        fraction = 0.12 / (len(lengths) - idx)   # later lengths get more time
        slice_deadline = time.time() + max(6.0, remaining * fraction)

        # ----- a few Adam runs from diverse starts -------------------------
        best_local_a = None
        best_local_score = float('inf')
        # three simple starting variants
        sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        sine = np.maximum(sine, 0.0)
        sine_sum = sine.sum()
        if sine_sum > 0.0:
            sine /= sine_sum
        init_candidates = [a0, sine, np.random.rand(n)]

        for cand in init_candidates:
            if time.time() >= slice_deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sym = (np.random.rand() < 0.3)
            a_opt, score_opt = _run_adam_opt(cand, n,
                                             slice_deadline, start_time,
                                             symmetrize=sym)
            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] adam local best n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        # ----- cheap LP swap ------------------------------------------------
        lp_deadline = min(slice_deadline, time.time() + 8.0)
        best_local_a = _run_lp_improve(best_local_a, n,
                                       step=0.2, max_iters=2,
                                       deadline=lp_deadline,
                                       start_time=start_time)
        best_local_score = _objective(best_local_a)

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after LP n={n} score={best_local_score:.6f}")

        # ----- sub‑gradient refinement --------------------------------------
        if time.time() < deadline:
            sg_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.15)
            a_sg, score_sg = _subgradient_descent(best_local_a, n,
                                                  slice_deadline=sg_deadline,
                                                  start_time=start_time)
            if score_sg < best_score:
                best_score = score_sg
                best_seq = a_sg.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after sub‑grad n={n} score={score_sg:.6f}")

        # ----- annealing polish --------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(best_local_a, n,
                                              deadline=anneal_deadline,
                                              start_time=start_time,
                                              cur_best_score=best_score,
                                              cur_best_seq=best_seq,
                                              start_temp=5e-2,
                                              end_temp=1e-5,
                                              steps=150000)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # ----- targeted local‑search (new ingredient) ----------------------
        if time.time() < deadline:
            ls_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.25)
            a_ls, score_ls = _targeted_local_search(best_seq, ls_deadline, start_time)
            if score_ls < best_score:
                best_score = score_ls
                best_seq = a_ls.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after targeted LS n={n} score={score_ls:.6f}")

        # keep for next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Final heavy polishing on the largest length attained
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        final_deadline = deadline - 0.5
        a_final, score_final = _run_adam_opt(best_seq, n,
                                             slice_deadline=final_deadline,
                                             start_time=start_time,
                                             symmetrize=True)
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final Adam improve score={score_final:.6f}")

        if time.time() < deadline - 0.5:
            ls_deadline = deadline - 0.2
            best_seq, best_score = _targeted_local_search(best_seq, ls_deadline, start_time)
            print(f"[{time.time() - start_time:6.2f}s] final targeted LS score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – always return a valid sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    return [float(x) for x in best_seq]
