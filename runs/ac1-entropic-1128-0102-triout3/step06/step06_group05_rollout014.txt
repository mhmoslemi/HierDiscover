"""
Hybrid optimizer for the self‑convolution L∞ problem.

The algorithm works on the probability simplex (∑a_i = 1) – the
objective returned by `evaluate_sequence` is homogeneous, so scaling
does not matter.

Process
-------
1. Warm‑start from the globally supplied sequence `height_sequence_1`
   (if present) or from a random Dirichlet draw.
2. Grow the vector geometrically (up‑sample) up to 65536 entries,
   keeping the shape roughly the same.
3. For each size run a cheap *mass‑transfer* simulated‑annealing
   optimiser.  A move transfers a small ε from a source index to a
   destination index; using the exact convolution update formula only
   three entries of the convolution change, so the move costs O(1).
   The annealer accepts improving moves and, with a temperature schedule,
   occasionally worse moves to escape local minima.
4. After annealing a short Adam optimisation on the smooth log‑sum‑exp
   surrogate is performed for a final refinement.
5. The best sequence seen over the whole run is returned as a plain
   Python list.

The whole routine respects the `budget_s` time limit (default 1000 s).
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) -----------------------------------------
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities -------------------------------------------------
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for all lengths)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)   # eliminate tiny negatives
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimisation on the smooth (log‑sum‑exp) surrogate -------------
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Stable soft‑max weights for the surrogate."""
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(w, 1.0 / w.size)
    w /= w_sum
    return w


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of the log‑sum‑exp surrogate L_λ(a)."""
    n = a.shape[0]
    w = _softmax_weights(conv, lam)
    # cross‑correlation of w with reversed a
    a_rev = a[::-1]
    # linear convolution (FFT)
    size = len(w) + len(a_rev) - 1
    nfft = 1 << ((size - 1).bit_length())
    W = np.fft.rfft(w, n=nfft)
    A = np.fft.rfft(a_rev, n=nfft)
    full = np.fft.irfft(W * A, n=nfft)[:size]
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate.
    Returns the best vector seen and its exact objective value (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3            # start λ ≈ lam_factor·max(conv)
    lam_decay = 0.985
    lam_min_factor = 1e-9

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        lam = max(max_b * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, lam_min_factor)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.8 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Incremental mass‑transfer simulated annealing --------------------------
# ----------------------------------------------------------------------


def _run_incremental_annealing(a: np.ndarray,
                               n: int,
                               conv: np.ndarray,
                               cur_best_seq: np.ndarray,
                               cur_best_score: float,
                               deadline: float,
                               start_time: float,
                               start_temp: float = 5e-2,
                               end_temp: float = 1e-6,
                               max_frac: float = 0.20,
                               steps: int = 2_000_000) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Cheap annealing where each move transfers ε from a source index to a
    destination index.  The convolution is updated exactly using the
    analytical formula – only three entries change – thus a move costs
    O(1) apart from recomputing the global maximum.

    Parameters
    ----------
    a               : current probability vector (sum = 1)
    conv            : a * a (full convolution, length 2n‑1)
    cur_best_seq    : best sequence seen so far
    cur_best_score  : its exact objective 2·n·max(conv)
    deadline        : absolute time after which the routine must stop
    start_time      : time when the whole optimisation began (for logging)
    start_temp      : initial temperature (in objective units)
    end_temp        : final temperature
    max_frac        : maximal fraction of the source mass that can be moved
    steps           : hard limit on number of annealing steps

    Returns
    -------
    a, best_seq, best_score
    """
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    # pre‑compute total number of steps possible in the allotted time
    # (the loop also respects the hard `steps` bound)
    step = 0
    while time.time() < deadline and step < steps:
        step += 1
        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (step / steps)

        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * max_frac * np.random.rand()
        if eps <= 0.0:
            continue

        # store old values for a possible rollback
        a_i_old = a[i]
        a_j_old = a[j]
        conv_2i_old = conv[2 * i]
        conv_2j_old = conv[2 * j]
        conv_ij_old = conv[i + j]

        # ----------- update the probability vector -------------
        a[i] = a_i_old - eps
        a[j] = a_j_old + eps

        # ----------- exact convolution update -----------------
        # offsets that change: 2i, 2j, i+j
        conv[2 * i] = conv_2i_old + (-2.0 * eps * a_i_old + eps * eps)
        conv[2 * j] = conv_2j_old + ( 2.0 * eps * a_j_old + eps * eps)
        conv[i + j] = conv_ij_old + (2.0 * eps * (a_i_old - a_j_old) - 2.0 * eps * eps)

        # eliminate possible tiny negatives caused by rounding
        if conv[2 * i] < 0.0:
            conv[2 * i] = 0.0
        if conv[2 * j] < 0.0:
            conv[2 * j] = 0.0
        if conv[i + j] < 0.0:
            conv[i + j] = 0.0

        # ----------- evaluate new objective -------------------
        new_max = conv.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score

        # acceptance test
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            # move accepted
            cur_score = new_score
            cur_max = new_max
        else:
            # rollback
            a[i] = a_i_old
            a[j] = a_j_old
            conv[2 * i] = conv_2i_old
            conv[2 * j] = conv_2j_old
            conv[i + j] = conv_ij_old

        # record global improvement
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} step={step} score={cur_score:.6f}")

    return a, best_seq, best_score

# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # a small safety margin (the caller adds its own 0.5 s guard)
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start from a globally provided construction if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0.0:
                a_global = starter / total
            else:
                a_global = None
        else:
            a_global = None
    except Exception:
        a_global = None

    # ------------------------------------------------------------------
    # Initial vector (size 4096) – either up‑sampled warm start or random
    # ------------------------------------------------------------------
    init_len = 4096
    if a_global is not None:

    else:
        a = np.random.rand(init_len)
        a /= a.sum()

    # maintain the best overall solution
    best_overall_seq = a.copy()
    best_overall_score = float('inf')

    # ------------------------------------------------------------------
    # Length schedule – geometrically growing sizes (up to 65536)
    # ------------------------------------------------------------------
    length_schedule = [8192, 16384, 32768, 65536]

    for n in length_schedule:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------

        a = _upsample(a, n)

        # keep the best candidate for the current length

        best_score_len = cur_score

            best_overall_seq = best_seq_len.copy()

            steps=3_000_000,            # large but the loop also respects the time limit
        )

        if best_score_len < best_overall_score:
            best_overall_score = best_score_len
            best_overall_seq = best_seq_len.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={best_score_len:.6f}")

        # ------------------------------------------------------------------
        # Short Adam refinement (helps to settle on a smoother shape)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            slice_deadline = time.time() + max(2.0, (deadline - time.time()) * 0.15)
            a_opt, score_opt = _run_adam_opt(best_seq_len, n, slice_deadline, start_time, symmetrize=False)
            if score_opt < best_overall_score:
                best_overall_score = score_opt
                best_overall_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Adam n={n} score={score_opt:.6f}")

        # keep the best sequence as warm‑start for the next larger size
        a = best_overall_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – shouldn't be needed but ensures a valid return value
    # ------------------------------------------------------------------
    if best_overall_seq is None or best_overall_seq.size == 0:
        fallback_n = 2048
        best_overall_seq = np.full(fallback_n, 1.0 / fallback_n)

    # Return a plain Python list – the evaluator will clamp / normalise as needed
    return [float(x) for x in best_overall_seq]
