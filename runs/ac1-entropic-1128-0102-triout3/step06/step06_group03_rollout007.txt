    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

        if starter.ndim != 1 or starter.size ==0:
            starter = None
    except Exception:
        starter = None

    # Helper to compute score (2*n*max convolution)
    def compute_score(seq):
        a = np.asarray(seq, dtype=float)
        n = a.shape[0]
        conv = _conv_self(a)
        return float(2.0 * n * conv.max())

    # Choose final length N (maybe 9600)

"""
Optimise a non‑negative step‑function (a probability vector) for
the objective used in ``evaluate_sequence``:

    score = 2 * n * max(conv(a, a))   where conv is the linear convolution.

The optimiser consists of three phases that share the same time budget
(≈ 1000 s):

1. **Smooth max‑gradient descent** – a projected Nesterov GD on a
   soft‑max approximation of the maximum convolution coefficient.
   The temperature is exponentially annealed, the step size decays as 1/√t,
   and every iteration is projected onto the probability simplex.

2. **Simulated annealing** – random mass transfers between two positions.
   Proposals are accepted with a Metropolis rule whose temperature follows
   a simple exponential schedule.  This phase is very good at escaping
   shallow local minima left by the gradient step.

3. **Targeted peak‑shaving local search** – a specialised move that shifts
   mass away from indices that contribute most to the current maximal
   convolution entry.  A slow cooling schedule allows occasional uphill
   moves for fine‑tuning.

The best sequence seen in all phases is returned (normalised to sum‑to‑1).
The code is written to respect the 1000 s wall‑time limit – each phase
checks the deadline and stops early if necessary.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: gradient of a soft‑max of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of Σ_k w_k(conv_k) * conv_k with w_k = soft‑max(conv/temperature).
    The factor 2 comes from the derivative of aᵀJ_k a (Toeplitz matrix).
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))   # extra stability
    w = exps / exps.sum()

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)          # length = n + (2n‑1) - 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]         # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Optimiser: projected (Nesterov) gradient descent with soft‑max gradient
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected Nesterov gradient descent on a smooth approximation of the
    max‑convolution.  Returns the best point seen and its score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (exponential decay)
    T_start = 5.0e-2
    T_end = 1.0e-5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: n={n}, iter={it}, score={cur_score:.6f}")

        # anneal temperature
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient of the smooth max
        grad = _softmax_gradient(a, conv, T)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer) optimiser
# ----------------------------------------------------------------------
def _run_simulated_annealing(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             cur_score: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer SA.
    Proposes a move: move a random fraction of donor mass to a random receiver.
    Accepts with Metropolis rule; temperature decays exponentially.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule
    T_start = 5.0e-2
    T_end = 1.0e-6
    it = 0

    while time.time() < deadline:
        it += 1

        # pick donor (probability proportional to current mass)
        if a.sum() == 0:
            break
        prob = a / a.sum()
        donor = np.random.choice(n, p=prob)
        receiver = np.random.randint(n)

        if donor == receiver or a[donor] < 1e-12:
            continue

        # transfer a small fraction (up to 10 %) of donor mass
        eps = a[donor] * np.random.rand() * 0.10
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[receiver] += eps

        # evaluate
        conv = _conv_self(a_prop)
        new_score = 2.0 * n * conv.max()
        delta = new_score - cur_score

        # temperature according to elapsed fraction
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed

        # Metropolis acceptance
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve: n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving) optimiser
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of probability mass away from
    indices that contribute most to the current maximal convolution entry.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule for Metropolis acceptance
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contributions of each index to this maximal sum
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor: proportional to contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient: pick somewhere with low contribution
        low_thr = np.percentile(contrib, 20)            # bottom 20 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # transfer a tiny fraction (up to 5 % of donor)
        eps = a[i_donor] * np.random.rand() * 0.05
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The pipeline is:
        1. Warm‑start from the previous best (``height_sequence_1``) if present.
        2. Soft‑max projected gradient descent on a moderately large n.
        3. Simulated annealing for fine‑grained stochastic optimisation.
        4. Targeted peak‑shaving local search.
    The best sequence (normalised to sum = 1) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    # ------------------------------------------------------------------
    # Load optional starter (previous best construction)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose a working length (large enough to give a good lower bound)
    # ------------------------------------------------------------------
    N = 9600  # same as the previous best‑performing size

    # ------------------------------------------------------------------
    # Initialise a0
    # ------------------------------------------------------------------
    if starter is not None:
        a0 = _upsample(starter, N)
    else:
        a0 = np.random.rand(N)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Phase 1 – smooth max gradient descent
    # ------------------------------------------------------------------
    a_gd, best_score = _run_softmax_gd(
        a0, N, deadline, start_time,
        step0=0.6, beta=0.9
    )

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing (mass‑transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, score_sa = _run_simulated_annealing(
            a_gd, N, deadline, start_time, best_score
        )
        if score_sa < best_score:
            best_score = score_sa
            a_gd = a_sa
            print(f"[{time.time() - start_time:6.2f}s] SA overall improve: score={score_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted local search (peak shaving)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_loc, score_loc = _run_targeted_local_search(
            a_gd, N, deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            a_gd = a_loc
            print(f"[{time.time() - start_time:6.2f}s] final local improve: score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (tiny drift may happen because of rounding)
    # ------------------------------------------------------------------
    best_seq = np.asarray(a_gd, dtype=float)
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
