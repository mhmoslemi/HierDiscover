def propose_candidate(seed=0,budget_s=1000,**kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # length selection:
    n = 4096
    # attempt to get previous best if defined
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = len(starter)
            a = np.maximum(starter, 0.0)
            if a.sum() > 0:
                a = a / a.sum()
            else:
                a = None
        else:
            a = None
    except Exception:
        a = None

    if a is None or a.size != n:
        # generate initial shape: exponential with tuned parameter c ~2
        c = 2.0
        r = np.exp(-c/n)
        raw = np.power(r, np.arange(n))
        a = raw / raw.sum()
    
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    cur_a = a.copy()
    cur_score = best_score

    # Simulated annealing parameters
    T_start = 5e-2
    T_end = 1e-6
    total_time = deadline - start_time
    iter = 0

    while time.time() < deadline:
        iter += 1
        # update temperature
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T_start * (T_end / T_start) ** frac

        # pick src and dst indices
        i = np.random.randint(0, n)
        if cur_a[i] <= 1e-12:
            continue
        j = np.random.randint(0, n)
        if i == j:
            continue

        # pick epsilon proportional to a[i], up to e.g., 30% of a[i]
        eps = cur_a[i] * (np.random.rand() * 0.3)
        a_new = cur_a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # no need to reproject (since small move preserves positivity and sum)
        # but due to floating errors, ensure sum close to 1
        # compute objective
        new_score = evaluate_sequence(a_new.tolist())
        delta = new_score - cur_score
        if delta < 0 or np.exp(-delta / (T + 1e-12)) > np.random.rand():
            cur_a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal iteration {iter} new best score {best_score:.6f}")

        # occasional heavy perturbation: random reset of a few coordinates?
        if iter % 20000 == 0:
            # random perturbation: add Gaussian noise and reproject onto simplex
            noise = np.random.randn(n) * 0.01
            a_tmp = cur_a + noise
            a_tmp = np.maximum(a_tmp, 0.0)
            if a_tmp.sum() > 0:
                a_tmp = a_tmp / a_tmp.sum()
            # accept if improves or according to temperature
            new_score = evaluate_sequence(a_tmp.tolist())
            delta = new_score - cur_score
            if delta < 0 or np.exp(-delta / (T + 1e-12)) > np.random.rand():
                cur_a = a_tmp
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_tmp.copy()
                    print(f"[{time.time() - start_time:6.2f}s] random perturb best {best_score:.6f}")

    # final local gradient descent (projected Adam) maybe short
    # project onto simplex ensure sum exactly 1
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()
    # optionally polish with few iterations of Adam on LSE surrogate
    a_opt, score_opt = _run_adam_opt(best_a, n,

        best_score = score_opt
        print(f"[{time.time() - start_time:6.2f}s] final Adam polish improved score {best_score:.6f}")

    # final return
    return [float(x) for x in best_a]

"""
Propose a candidate step function using a directed simulated‑annealing
procedure driven by the sub‑gradient of the current max‑convolution.
The algorithm:

1. Warm‑start from the previously discovered best sequence (if present);
   otherwise initialise a smooth exponential shape.
2. Repeatedly move a small amount of mass from indices that contribute
   most to the current peak of the self‑convolution to indices that
   contribute least.  Acceptance follows a classic annealing schedule.
3. Occasionally inject a small random perturbation to escape plateaus.
4. Finish with a short projected‑Adam refinement on the smooth
   log‑sum‑exp surrogate of the max‑convolution.
All operations stay on the probability simplex (∑aᵢ = 1) and the routine
stops before the time‑budget expires, returning the best sequence found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT based convolution)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast)."""
    n = a.shape[0] + b.shape[0] - 1
    # next power of two for speed
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using FFT."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Objective utilities (scale‑invariant, sum is kept =1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(a*a) (sum(a)=1)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------


def _maxconv_gradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Returned gradient satisfies: for any feasible direction d,
    d·grad ≤ max_{k∈argmax(conv)} Σ_i d_i a_{k-i} + Σ_i d_i a_{i-k}.
    """
    conv = _conv_self(a)
    max_val = conv.max()
    # weight vector: 1 on the active shift(s), 0 elsewhere
    w = (conv == max_val).astype(float)
    # convolution of w with reversed a gives exactly the sub‑gradient
    grad = 2.0 * _conv_fft(w, a[::-1])
    n = a.shape[0]
    # central slice corresponding to indices 0 … n‑1
    grad = grad[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Simple projected Adam optimisation on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_lse(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate:
        L_λ(a) = λ·log Σ_k exp(conv_k / λ)
    """
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    grad = 2.0 * _conv_fft(w, a[::-1])
    n = a.shape[0]
    grad = grad[n - 1 : n - 1 + n]
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam on the LSE surrogate; returns best encountered sequence."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3
    lam_decay = 0.8
    lam_decay_interval = 3000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)
        grad = _gradient_lse(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.6 / math.sqrt(it + 1.0)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Warm‑start with the previously discovered construction (if any)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # --------------------------------------------------------------
    # Choose problem size – 4096 gave the best result historically.
    # --------------------------------------------------------------
    n = 4096
    if starter is not None and starter.size == n:
        a_cur = starter.copy()
    else:
        # fallback: a smooth exponential shape a_i ∝ exp(-c·i/n)
        c = 2.0
        r = np.exp(-c / n)
        raw = r ** np.arange(n, dtype=float)
        a_cur = raw / raw.sum()

    best_a = a_cur.copy()
    best_score = _objective(best_a)

    # --------------------------------------------------------------
    # Directed simulated annealing driven by the sub‑gradient of the peak
    # --------------------------------------------------------------
    T_start = 5e-2
    T_end = 1e-6
    total_time = deadline - start_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T_start * (T_end / T_start) ** frac  # exponential cooling

        # Gradient of the current max‑convolution (helps us pick source/target)
        grad = _maxconv_gradient(a_cur)

        # source – favour large gradient entries
        src_weights = grad.copy()
        src_sum = src_weights.sum()
        if src_sum <= 0.0:
            src_probs = np.full(n, 1.0 / n)
        else:
            src_probs = src_weights / src_sum

        # target – favour small gradient entries
        maxg = grad.max()
        ming = grad.min()
        dest_weights = maxg - grad
        dest_sum = dest_weights.sum()
        if dest_sum <= 0.0:
            dest_probs = np.full(n, 1.0 / n)
        else:
            dest_probs = dest_weights / dest_sum

        i = np.random.choice(n, p=src_probs)
        if a_cur[i] <= 1e-14:
            continue
        j = np.random.choice(n, p=dest_probs)
        if i == j:
            continue

        # move a small random fraction of the source mass to the target
        eps_frac = np.random.rand() * 0.15  # up to 15 % of source mass
        eps = a_cur[i] * eps_frac
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # Exact objective (sum is kept =1)
        new_score = _objective(a_new)
        delta = new_score - best_score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a_cur = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal iter={iter_cnt} new best score={best_score:.6f}")

        # occasional random perturbation to broaden search
        if iter_cnt % 20000 == 0:
            noise = np.random.randn(n) * 0.005
            a_tmp = a_cur + noise
            a_tmp = np.maximum(a_tmp, 0.0)
            if a_tmp.sum() > 0.0:
                a_tmp = a_tmp / a_tmp.sum()
                tmp_score = _objective(a_tmp)
                if tmp_score < best_score or math.exp(-(tmp_score - best_score) / T) > np.random.rand():
                    a_cur = a_tmp
                    best_score = tmp_score
                    best_a = a_tmp.copy()
                    print(f"[{time.time() - start_time:6.2f}s] random perturb best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final short Adam polish (smooth surrogate) – respects remaining time
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 2.0:
        slice_deadline = time.time() + time_left
        a_opt, score_opt = _run_adam_opt(best_a, n,
                                         slice_deadline=slice_deadline,
                                         start_time=start_time,
                                         symmetrize=False)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] final Adam polish improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – ensure a valid return
    # ------------------------------------------------------------------
    if best_a is None or best_a.size == 0:
        best_a = np.full(512, 1.0 / 512.0)

    # Return plain Python list (evaluate_sequence will re‑normalise/clamp)
    return [float(x) for x in best_a]
