assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

The problem is non‑convex because the maximum of a quadratic form is involved, but  

* the value is invariant under scaling – we may keep the vector on the probability simplex;
* the objective can be smoothed with a log‑sum‑exp, which gives a differentiable surrogate;
* a true sub‑gradient of the unsmoothed max can be computed exactly from the active convolution entries.

The code below implements a **three‑phase search** that is tuned for the 1000 s time budget:

1. **Parameter sweep** – a few analytically‑motivated families (uniform, power‑law, sinusoidal) are sampled and the best one is kept as a start point.
2. **Exponential‑gradient warm‑up** – mirror‑descent on the smooth surrogate (`log‑sum‑exp`) with a slowly increasing smoothing parameter `α`.  
   This quickly finds a good basin of attraction.
3. **Projected sub‑gradient descent** – classic sub‑gradient steps on the true `max` objective, with momentum and an adaptive step‑size schedule.
   The descent is performed in blocks; after each block a tiny random perturbation is applied to escape possible plateaus.
4. **Annealing polish** – a fast “mass‑transfer’’ simulated‑annealing that updates the convolution incrementally (O(n) per move) and accepts uphill moves with an exponential cooling schedule.
5. **Final mixing & cleanup** – a short line‑search between the best vector and the uniform distribution (and its reverse) is performed, then the vector is projected onto the simplex and returned.

All heavy work is performed with FFT‑based convolutions (`_conv_self`) and the code respects the provided time budget at every stage.

```python

A fast multi‑stage optimiser for the step‑function objective

    evaluate_sequence(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The optimiser works as follows:

1.  Sample a few analytically‑inspired families (uniform, power‑law,
    sinusoidal) and a few random draws – keep the best as a start.

2.  Warm‑up with exponentiated‑gradient on the smooth log‑sum‑exp

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no clipping)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Warm‑up: exponentiated‑gradient on a smooth approximation of max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 12000):
    """Mirror‑descent on log‑sum‑exp(max(conv)).  Returns best (a,score)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    fft_len = _next_pow2(3 * n - 2)          # safe for correlation
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                f"iter={it:,}, score={cur_score:.6f}"
            )

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # ------- smooth‑max gradient (log‑sum‑exp) ----------
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)           # safe scaling
        w /= w.sum()                              # softmax over all shifts
        w_rev = w[::-1]                           # needed for correlation
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        # gradient of smooth max w.r.t. a
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # ------- multiplicative (mirror) update -------------
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a = _project_simplex(a)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray,
                      conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a using all near‑max entries."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.5,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    """Projected (momentum) sub‑gradient descent on the true objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                f"iter={it:,}, score={cur_score:.6f}"
            )

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing polish (incremental convolution update)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000):
    """
    Mass transfer moves with exponential cooling.
    The convolution is updated incrementally in O(n) per move.
    """
    a = a0.copy()
    conv = _conv_self(a)                     # current convolution
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    L = conv.shape[0]
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source according to current mass, destination uniformly
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # --- incremental update of the convolution -----------------
        # Δconv[t] = 2*eps*(a[t-dst] - a[t-src])  (out‑of‑range → 0)
        delta = np.zeros_like(conv)
        # source contribution (subtraction)
        src_start = src
        src_end = src + n
        src_range = slice(src_start, min(src_end, L))
        src_off = np.arange(src_range.stop - src_range.start, dtype=int)
        delta[src_range] -= 2.0 * eps * a[src_off]

        # destination contribution (addition)
        dst_start = dst
        dst_end = dst + n
        dst_range = slice(dst_start, min(dst_end, L))
        dst_off = np.arange(dst_range.stop - dst_range.start, dtype=int)
        delta[dst_range] += 2.0 * eps * a[dst_off]

        conv_prop = conv + delta
        new_score = 2.0 * n * conv_prop.max() / (a_prop.sum() ** 2)

        # ----- Metropolis acceptance ------------------------------
        delta_obj = new_score - score
        if delta_obj < 0 or math.exp(-delta_obj / T) > rng.random():
            a = a_prop
            conv = conv_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] anneal improve: "
                f"iter={it:,}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Small utility – generate a few analytically‑motivated shapes
# ----------------------------------------------------------------------
def _generate_family_shapes(n: int):
    """Yield a handful of deterministic shapes (already normalised)."""
    # uniform
    yield np.full(n, 1.0 / n)

    # power law in distance from centre  (U‑shaped)
    mid = (n - 1) / 2.0
    idx = np.arange(n, dtype=float)
    dist = np.abs(idx - mid) + 1e-12
    for p in [0.3, 0.6, 1.0, 1.5, 2.0]:
        a = dist ** p
        a /= a.sum()
        yield a

    # sinusoidal families (bell‑shaped)
    x = np.linspace(0.0, np.pi, n)
    for p in [1, 2, 4]:
        a = np.sin(x) ** p
        a /= a.sum()
        yield a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that
    approximately minimises ``evaluate_sequence`` under the given
    time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Determine the length of the construction
    # --------------------------------------------------------------
    final_n = 4800                     # a good compromise between quality & speed

    # --------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if it exists.
    # --------------------------------------------------------------
    starter = None
    try:
        h = np.asarray(height_sequence_1, dtype=float)
        if h.ndim == 1 and h.size > 0:
            starter = h / h.sum()
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Build a pool of candidate starting points
    # --------------------------------------------------------------
    pool = []

    if starter is not None:
        # up‑sample the previous best to the required length
        old_n = starter.shape[0]
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, final_n)
        up = np.interp(new_x, old_x, starter)
        up /= up.sum()
        pool.append(up)

    # deterministic families
    pool.extend(_generate_family_shapes(final_n))

    # a few random Dirichlet draws
    for _ in range(4):
        a = np.random.rand(final_n)
        a /= a.sum()
        pool.append(a)

    # --------------------------------------------------------------
    # 2️⃣  Pick the best among the cheap samples
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for a in pool:
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    print(f"[{time.time() - start_time:.2f}s] init best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Warm‑up: exponentiated‑gradient on a smoothed max
    # --------------------------------------------------------------
    if time.time() < deadline:
        warm_dead = time.time() + min(0.15 * (deadline - time.time()), 70.0)
        a_warm, sc_warm = _run_exponential_gradient(
            best_seq,
            final_n,
            warm_dead,
            start_time,
            alpha_schedule=[0.5, 2, 5, 10, 20, 40, 80, 160, 320],
            eta0=0.4,
            steps_per_alpha=12000,
        )
        if sc_warm < best_score:
            best_score = sc_warm
            best_seq = a_warm.copy()
            print(f"[{time.time() - start_time:.2f}s] warm‑up improve: {sc_warm:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Main projected sub‑gradient descent (blockwise with restarts)
    # --------------------------------------------------------------
    while time.time() < deadline:
        # allocate at most half of the remaining budget to this block,
        # but never more than a couple of minutes – we want time for polishing.
        remaining = deadline - time.time()
        block_dead = time.time() + min(0.45 * remaining, 180.0)

        a_sub, sc_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            block_dead,
            start_time,
            step0=0.5,
            beta=0.94,
            max_iter=10**7,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad block improve: {sc_sub:.6f}")

        # ---------------------- tiny random kick --------------------
        if time.time() >= deadline:
            break
        # Add a modest perturbation and continue
        pert = best_seq * (1.0 + 0.02 * np.random.randn(final_n))
        pert = np.maximum(pert, 0.0)
        if pert.sum() == 0.0:
            pert = np.random.rand(final_n)
        pert /= pert.sum()
        best_seq = pert

    # --------------------------------------------------------------
    # 5️⃣  Final annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_fin, sc_fin = _run_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=1e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal final improve: {sc_fin:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Very short line‑search mixing with uniform / reversed vectors
    # --------------------------------------------------------------
    if time.time() < deadline:
        uniform = np.full(final_n, 1.0 / final_n)
        rev = best_seq[::-1]
        for t in np.linspace(0.0, 0.08, 9):
            for mix in (uniform, rev):
                cand = (1.0 - t) * best_seq + t * mix
                cand = _project_simplex(cand)
                sc = _objective(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] mix improve (t={t:.3f}) -> {sc:.6f}")

    # --------------------------------------------------------------
    # Normalise (protect against possible drift) and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"Final objective ≈ {_objective(best_seq):.6f}")
    return [float(x) for x in best_seq]
```
