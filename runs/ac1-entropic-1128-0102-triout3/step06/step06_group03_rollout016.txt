def _adam_step(a, grad, m, v, t, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_proj = _project_simplex(a_new)
    return a_proj, m, v

def _run_softmax_adam(a0, n, deadline, start_time, lr=0.1, beta1=0.9, beta2=0.999):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = np.inf
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # schedule temperature
        it += 1
        # use exponential decay e.g., T = max(1e-6, 0.01 * (0.01)**(it / max_iter))
        # but easier: T = max(1e-6, 0.02 * (0.5 ** (it / 2000))) ...
        T = max(1e-6, 0.02 * (0.5 ** (it / 5000.0))) # decays by half each 5000 iters

        grad = _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        a = _project_simplex(a - step)

    return best_a, best_score

"""
Heuristic optimizer for min max convolution.
Uses multi-stage upsampling and Adam-style gradient descent
with soft‑max smoothing, followed by a targeted local search.
"""

import time, math, numpy as np

def _project_simplex(v):
    # Euclidean projection onto simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_self(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n-1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    return conv

def _conv_rev_weight(a_rev, w):
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n+m-1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[:n+m-1]
    return conv

def _softmax_gradient(a, conv, T):
    # stable soft‑max over conv entries
    maxc = conv.max()
    scaled = (conv - maxc) / T
    exps = np.exp(scaled - np.max(scaled))   # extra stability
    w = exps / np.sum(exps)
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    grad = 2.0 * full[n-1:n-1+n]
    return grad

def _run_adam_gd(a0, n, deadline, start_time,
                 lr=0.15, beta1=0.9, beta2=0.999,

    # pre‑compute exponentials for temperature schedule

        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

            best_score = cur_score
            best_a = a.copy()
            # progress output

        # temperature decays exponentially
        T = T_start * (T_end / T_start) ** (it / 25000.0)

        grad = _softmax_gradient(a, conv, T)

        # Adam update (we minimise, hence subtract)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = _project_simplex(a - step)

    return best_a, best_score

def _upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up_sum = up.sum()
    if up_sum > 0:
        up /= up_sum
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

"""
Heuristic optimiser for the “min‑max self‑convolution’’ problem.

The optimiser works in several stages:

1.  Progressive up‑sampling – start with a modest length and repeatedly
    enlarge the support while keeping the best distribution found so far.
2.  Adam‑style projected gradient descent on a temperature‑annealed
    soft‑max approximation of the L∞‑norm of the linear self‑convolution.
    The temperature is driven to a very small value so that the smooth
    max closely matches the true maximum.
3.  A focused local‑search (“peak shaving’’) that repeatedly moves a
    small amount of mass away from indices that contribute most to the
    currently maximal convolution entry.  An annealed Metropolis rule
    accepts occasional uphill moves.

The whole routine respects the 1000 s budget (a 0.5 s safety margin) and
returns a plain list of non‑negative numbers whose sum is 1 (the scaling
does not affect the evaluation).

    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0

        return np.full_like(v, 1.0 / n)

    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    return conv

        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
