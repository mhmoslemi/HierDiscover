def project_simplex(v):
    # from Duchi: sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def entropic_descent(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)
    tau = 5.0 / n  # initial temperature
    tau_end = tau * 1e-4  # lower final
    decay = 0.9995  # same
    step0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv
        conv = conv_self(a)
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(...)  # log

        # compute softened weights w
        shift = conv.max()
        # avoid overflow
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth surrogate:
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)
        idx = (n - 1) + np.arange(n)   # correct
        grad = 2.0 * corr[idx]

        # multiplicative update
        step = step0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700, 700)
        a = a * np.exp(expo)
        # renormalize:
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # anneal tau
        tau = max(tau_end, tau * decay)

    return best_a, best_sc
```

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        k = conv.argmax()
        # compute gradient: g_i = 2 * a_{k-i} if index valid else 0
        j = k - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # update
        step = step0 / math.sqrt(it)
        a = a - step * grad
        # project onto simplex
        a = project_simplex(a)
        # ensure non-negative with clip (project should guarantee)
        # compute new max and maybe early stop
    return best_a, best_sc

"""
Optimising a non‑negative step‑function for the evaluate_sequence()
objective.

The search proceeds in several stages:

1. **Length schedule** – we start with a small number of coefficients and
   gradually increase to larger sequences (up to 4096).  The best shape
   found for a length is up‑sampled to initialise the next length.

2. **Parametric warm‑starts** – analytic families (product‑power,
   Gaussian, Beta, arcsine, exponential, sparse) are sampled at random.
   The best of these provides a good initial point for each length.

3. **Entropic mirror‑descent on a soft‑max surrogate** – we minimise
   τ·log ∑exp(conv/τ).  The gradient of the smooth objective is obtained
   analytically via a second FFT.  The update is multiplicative
   (exponentiated‑gradient) followed by a renormalisation onto the
   probability simplex.

   **Important fix:** the gradient index is now correctly taken as
   `corr[(n-1)+i]` (the previous implementation used a reversed index).

4. **Exact sub‑gradient descent on the true ∞‑norm** – after the smoothed
   phase we switch to the true max‑convolution objective, using the
   sub‑gradient `g_i = 2·a_{k‑i}` where `k` is the index of the maximal
   convolution entry.  The iterate is projected onto the simplex after
   each Euclidean step.

5. **Light simulated‑annealing polish** – random mass transfers between
   coordinates accept improvements or probabilistic worsening moves.

All convolutions are performed with a power‑of‑two FFT, giving
O(n log n) per iteration.  Time is monitored so that the routine always
returns a valid sequence before the 1000 s budget expires.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (self and cross)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x≥0, Σx=1}. """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring utilities (scale‑invariant)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric families for quick warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term = np.where(term <= 0, 1e-12, term)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term = np.where(term <= 0, 1e-12, term)
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_sparse(n: int, k: int) -> np.ndarray:
    """k uniformly random locations carry equal mass, all others zero."""
    if k >= n:
        k = n
    a = np.zeros(n)
    idx = np.random.choice(n, size=k, replace=False)
    a[idx] = 1.0
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

# Entropic mirror‑descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def entropic_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Minimises the smooth surrogate τ·log ∑exp(conv/τ) with an
    exponentiated‑gradient update.  The gradient index fix (see comment
    above) yields a more faithful descent direction.
    Returns the best vector seen and its exact (non‑smoothed) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # temperature schedule
    tau = 5.0 / n                 # start a bit larger than before
    tau_end = tau * 1e-4          # finish much colder
    decay = 0.9995

    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] entropic improve n={n} it={it} score={cur_sc:.6f}"
            )

        # soft‑max weights
        shift = conv.max()
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (corrected index)
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)                     # length = 3n‑2
        idx = (n - 1) + np.arange(n)                 # <-- correct slice
        grad = 2.0 * corr[idx]

        # exponentiated‑gradient update
        step = step0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)           # avoid overflow
        a = a * np.exp(expo)

        # renormalise onto simplex
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # anneal temperature
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Direct sub‑gradient descent on the true ∞‑norm objective
    (max self‑convolution).  After each Euclidean step the iterate is
    projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 0.2
    it = 0

    while time.time() < deadline:
        it += 1

        conv = conv_self(a)
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}"
            )

        k = conv.argmax()                     # index where max is attained
        j = k - np.arange(n)                  # positions in a that affect the max
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]          # sub‑gradient

        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = project_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap move‑based annealing: transfer a random ε from one index
    to another.  Accept moves that improve the score or according to a
    geometric temperature schedule.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score
