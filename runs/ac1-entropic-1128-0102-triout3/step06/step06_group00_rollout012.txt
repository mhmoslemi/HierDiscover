    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        grad = _subgradient_full(a, conv)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

    return best_a, best_score

"""
Hybrid optimizer for the step‑function bound.

The routine works in four phases (all respecting the overall time budget):

1. **Warm‑up** – a few thousand exponentiated‑gradient updates on a
   log‑sum‑exp (smooth‑max) of the self‑convolution.  This quickly
   drives the shape into a low‑peak region while keeping the vector on
   the probability simplex.

2. **Fine‑tuning** – projected Adam (momentum + adaptive steps) on the
   *exact* max‑convolution sub‑gradient.  The updates are cheap
   (FFT‑based convolution) and the Adam schedule lets us take many
   small steps until the deadline.

3. **Linear‑program refinement** – given the current vector `f` and its
   maximal convolution value `M`, we solve a modest LP:
        maximise   Σ g_j
        subject to f∗g ≤ M  (pointwise) ,   g ≥ 0 .
   Only the *active* convolution indices (those within a tiny tolerance
   of `M`) are kept as constraints, so the LP is very small and solves
   in a fraction of a second.  The solution `g` is simplex‑projected and
   mixed with `f` (convex combination).  Repeating a few times pushes
   the peak down further.

4. **Annealing polish** – a short simulated‑annealing pass that moves
   tiny amounts of mass between entries; it can escape shallow local
   minima left by the deterministic steps.

All iterates stay non‑negative and are re‑projected onto the simplex
after each update.  The best sequence seen throughout the run is
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (using all near‑max entries)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 12000):
    """
    Multiplicative updates based on the log‑sum‑exp smooth approximation
    of max(conv).  Returns the best iterate seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # --- smooth‑max gradient ------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        # correlation a * w_rev (via FFT)
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # --- multiplicative update ---------------------------------
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Adam projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          lr: float = 0.12,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8):
    """
    Adam‑style projected sub‑gradient descent on
        f(a)=2·n·max(conv(a,a)).
    Returns the best vector encountered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score

# Phase‑3 : Linear‑program refinement (small active set)
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (|active| × n) such that
        (A @ g)[r] = (f * g)[active[r]].
    """
    n = f.shape[0]
    rows = active.shape[0]
    A = np.zeros((rows, n))
    for r, k in enumerate(active):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        A[r, j] = f[i]
    return A


def _run_lp_refine(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   t_init: float = 0.12,
                   active_tol: float = 0.02,
                   max_iter: int = 12):
    """
    Small linear‑program based refinement.
    For the current vector f we maximise Σ g_j under the linear
    constraints f∗g ≤ M (only for convolution indices within
    `active_tol` of the current maximum).  The solution g is blended
    into f with mixing weight `t`.  Repeated a few times.
    """
    try:
        import cvxpy as cp
    except Exception as exc:
        print(f"[{time.time() - start_time:.2f}s] cvxpy not available, skipping LP phase.")
        return a0.copy(), float('inf')

    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    t = t_init
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        active = np.where(conv >= M * (1.0 - active_tol))[0]
        if active.size == 0:
            break

        A = _build_constraint_matrix(a, active)

        # Linear program: maximise Σ g_j   s.t.   A g ≤ M,   g ≥ 0
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception as e:
            print(f"[{time.time() - start_time:.2f}s] LP solve error: {e}")
            break

        if g.value is None:
            print(f"[{time.time() - start_time:.2f}s] LP returned no solution.")
            break

        g_opt = np.asarray(g.value).flatten()
        sum_g = g_opt.sum()
        if sum_g < 1e-12:
            # trivial solution – nothing to mix
            break
        g_opt /= sum_g            # bring to simplex

        # convex combination and projection
        new_a = (1.0 - t) * a + t * g_opt
        new_a = _project_simplex(new_a)

        new_conv = _conv_self(new_a)
        new_score = 2.0 * n * new_conv.max()

        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP improve iter={it}: score={new_score:.6f}")

        # accept if we have really reduced the peak
        if new_score < 2.0 * n * M:
            a = new_a
            # tighten tolerance a little for the next round
            active_tol = max(active_tol * 0.85, 0.001)
        else:
            # fall back to a smaller mixing weight
            t *= 0.5
            if t < 1e-4:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 : tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000):
    """
    Simple simulated‑annealing that transfers a small fraction of mass
    between two positions.  The temperature is geometrically cooled.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the evaluation harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if present.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # a length that balances expressiveness and speed
    final_n = 4800
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – exponentiated‑gradient warm‑up (≈8 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    stage1_deadline = time.time() + max(5.0, 0.08 * remaining)

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.4,
        steps_per_alpha=12000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam projected sub‑gradient descent (≈70 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        stage2_deadline = time.time() + max(30.0, 0.70 * remaining)

        a_sub, score_sub = _run_adam_subgradient(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            lr=0.12,
            beta1=0.9,
            beta2=0.999,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – linear‑program refinement (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        stage3_deadline = time.time() + max(10.0, 0.15 * remaining)

        a_lp, score_lp = _run_lp_refine(
            best_seq,
            final_n,
            stage3_deadline,
            start_time,
            t_init=0.12,
            active_tol=0.02,
            max_iter=12,
        )
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 3 improve: score={score_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polish (any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # --------------------------------------------------------------
    # Normalise (protect against drift) and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
