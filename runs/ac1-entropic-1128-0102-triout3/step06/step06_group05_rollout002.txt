"""
Hybrid multi‑scale optimizer for the self‑convolution L∞ problem.

The goal is to minimise
    2·n·max(a * a)   with   a ≥ 0 ,  Σa = 1 .
The code searches over increasing vector lengths, repeatedly
  – warming‑up from a coarser solution,
  – running Adam on a soft‑max approximation of the max‑convolution,
  – refining with a projected sub‑gradient method on the exact max,
  – polishing with a simulated‑annealing / greedy hill‑climb.
Both symmetric and asymmetric initialisations are tried.
The routine respects a global time budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for any length)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Scoring utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Exact objective value (sum(a)=1 assumed)."""
    n = a.shape[0]
    maxb = _conv_self(a).max()
    return 2.0 * n * maxb


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) surrogate gradient
# ----------------------------------------------------------------------


def _grad_softmax(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of L_λ(a) = λ·log Σ exp(conv/λ) .
    Returns a dense gradient of shape (n,).
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    w = np.exp((conv - max_val) / lam)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of weights with reversed a yields derivative of conv wrt a
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]   # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate.
    Returns the best vector found and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule parameters
    lam_factor = 0.5          # initial λ relative to max(conv)
    lam_decay = 0.99
    lam_min = 1e-12

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        maxb = conv.max()
        cur_score = 2.0 * n * maxb
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ
        lam = max(maxb * lam_factor, lam_min)
        lam_factor = max(lam_factor * lam_decay, lam_min)

        grad = _grad_softmax(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base step decays like 1/√it
        base_step = 0.7 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient descent
# ----------------------------------------------------------------------


def _grad_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a * a)[k] at a.
    Returns (gradient, max_value).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)          # length = 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


def _run_subgradient_opt(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step0 = 0.8

    it = 0
    while time.time() < deadline:
        it += 1
        grad, max_val = _grad_max_conv(a)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = _project_simplex(a)

        # optional symmetrisation (randomly applied)
        if random.random() < 0.2:
            a = (a + a[::-1]) * 0.5

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer) and greedy hill‑climb
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Simple annealing: transfer a fraction of mass from one index to another.
    Returns the best sequence discovered and its exact score.
    """
    a = a0.copy()
    score = _score(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12   # up‑to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _score(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


def _run_greedy_hillclimb(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          steps: int = 150000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: only accept moves that strictly improve the exact score.
    """
    a = a0.copy()
    score = _score(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.10
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _score(a_prop)
        if new_score < score:                 # accept only improvements
            a = a_prop
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] HillClimb improve n={n} it={it} score={new_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Shape generators for initial guesses
# ----------------------------------------------------------------------


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a handful of plausible starting vectors derived from a base
    distribution.  All returned vectors are normalised to sum = 1.
    """
    variants = []
    n = base_seq.shape[0]

    # Warm‑start copy
    variants.append(base_seq.copy())

    # Sine shape (positive half wave)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    variants.append(s)

    # Triangular shape
    left = np.linspace(1, n // 2, n // 2, endpoint=False)
    right = np.linspace(n // 2, 1, n - n // 2)
    tri = np.concatenate([left, right])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # Arcsine‑like shape (blows up near the edges)
    t = (np.arange(n) + 0.5) / n
    arc = 1.0 / np.sqrt(t * (1.0 - t) + 1e-12)
    arc /= arc.sum()
    variants.append(arc)

    # Random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    # Piecewise constant (two‑block) shape
    split = n // 3
    block = np.concatenate([np.full(split, 0.5 / split),
                            np.full(n - split, 0.5 / (n - split))])
    block /= block.sum()
    variants.append(block)

    return variants
