def _run_targeted_local_search_multi(a0, deadline, start_time, top_k=5):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2 * n * conv.max()
    cur_score = best_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        # find top K indices
        top_idxs = np.argpartition(conv, -top_k)[-top_k:]
        # compute total contributions per index across those peaks
        contrib = np.zeros(n)
        for k in top_idxs:
            i_arr = np.arange(n)
            j_arr = k - i_arr
            mask = (j_arr >= 0) & (j_arr < n)
            contrib[mask] += a[i_arr[mask]] * a[j_arr[mask]]
        # donor sampling proportional to contribution
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob = contrib / total_contrib
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        low_thr = np.percentile(contrib, 20)

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise a candidate from the previous best construction (if any)
   or from a uniform distribution.
2. Progressive optimisation in four size stages (800 → 2400 → 4800 → 9600):
   * a smooth “soft‑max” projected gradient descent (as in the reference
     implementation) – provides rapid global improvement;
   * a pure sub‑gradient descent on the true max‑convolution – sharpens
     the peak‑shaving step.
3. After the final size stage run an aggressive multi‑peak local search.
   The search repeatedly transfers a small fraction of mass away from indices
   that contribute most to the *K* largest convolution entries and towards
   indices that contribute little.  A Metropolis acceptance rule with a fast
   cooling schedule allows occasional uphill moves.
4. Return the best sequence encountered (normalised to have sum = 1).

The combination of a smooth global optimiser, an exact sub‑gradient step
and a focused multi‑peak “peak‑shaving’’ walk typically reduces the
evaluation score well below the previous 1.512882 bound.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import random

import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: up‑sample (linear interpolation) preserving total mass
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Helper: smooth‑max (soft‑max) gradient of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of Σ_k w_k(conv_k) * conv_k where w_k is the soft‑max of conv/temperature.
    The factor 2 comes from the derivative of aᵀJ_k a (Toeplitz matrix).
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))      # extra stabilisation
    w = exps / exps.sum()

    # weighted gradient: 2 * (rev(a) * w) restricted to the central n entries
    rev_a = a[::-1]
    full = np.convolve(rev_a, w)               # linear convolution (fast enough for our sizes)
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]         # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Optimiser: soft‑max projected gradient descent (as in the reference)
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.95,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected (Nesterov) gradient descent on the smooth max‑convolution.
    Returns the best point seen and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (exponential decay)
    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n}, iter={it}, score={cur_score:.6f}")

        # anneal temperature
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient of the smooth max
        grad = _softmax_gradient(a, conv, T)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimiser: pure sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.5,
                             decay: float = 1e-4) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on  max_k conv_k ."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))

        # sub‑gradient of the peak entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j_arr[mask]]

        step = step0 / (1.0 + decay * it)
        a = _project_simplex(a - step * grad)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted multi‑peak local search (mass‑shifting)
# ----------------------------------------------------------------------
def _run_targeted_local_search_multi(a0: np.ndarray,
                                    deadline: float,
                                    start_time: float,
                                    top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of mass from indices that heavily contribute to
    the K largest convolution entries to indices that contribute little.
    A slow temperature schedule permits occasional uphill moves.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    cur_score = best_score

    # temperature schedule for Metropolis acceptance
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        # indices of the K largest convolution entries
        top_idxs = np.argpartition(conv, -top_k)[-top_k:]

        # total contribution of each position to the selected peaks
        contrib = np.zeros(n)
        for k in top_idxs:
            i_arr = np.arange(n)
            j_arr = k - i_arr
            mask = (j_arr >= 0) & (j_arr < n)
            contrib[mask] += a[i_arr[mask]] * a[j_arr[mask]]

        # choose a donor proportional to contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # choose a recipient from the low‑contribution tail (bottom 20 %)
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # move a random fraction (up to 30 %) of donor mass
        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # cooling schedule
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > random.random()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] multi local improve: n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score

    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Uses a cascade of soft‑max GD → sub‑gradient GD → multi‑peak local search.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Load optional starter (previous best construction)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage configuration (progressive up‑sampling)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]   # final length 9600
    total_len_weight = sum(stage_lengths)

    # Allocate ~70 % of total time to gradient descent, ~30 % to local search
    total_time = deadline - start_time
    grad_budget = total_time * 0.70
    local_budget = total_time - grad_budget

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Gradient descent phase (soft‑max + sub‑gradient)
    # --------------------------------------------------------------
    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # proportion of the *remaining* gradient budget for this stage
        stage_frac = n / total_len_weight
        stage_time = max(2.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for this length
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        a_opt, score_opt = _run_softmax_gd(
            a0, n, stage_deadline, start_time,
            step0=0.6, beta=0.95, max_iter=10 ** 7
        )

        # 2) pure sub‑gradient GD (use the remaining time of this stage)
        if time.time() < stage_deadline:
            sub_deadline = stage_deadline
            a_opt, score_opt = _run_subgradient_descent(
                a_opt, sub_deadline, start_time,
                step0=0.5, decay=1e-4
            )
        else:
            # recompute score for the current a_opt (if we never entered sub‑grad)
            conv = _conv_self(a_opt)
            score_opt = 2.0 * n * conv.max()

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # Update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - local_budget - elapsed)

    # --------------------------------------------------------------
    # Targeted multi‑peak local search
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

            best_seq, deadline, start_time, top_k=5

            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final multi local improve: score={score_loc:.6f}")

    s = best_seq.sum()

    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
