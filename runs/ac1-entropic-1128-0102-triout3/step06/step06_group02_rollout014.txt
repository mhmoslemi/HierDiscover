"""
Proposes a step function (non‑negative vector) minimising ``evaluate_sequence``.
The method consists of:
1. Initialise from the previous best vector or a random simplex point.
2. Run an accelerated soft‑max gradient descent (Adam) to lower the smooth
   max‑convolution.
3. Switch to exact sub‑gradient descent for the true max‑convolution.
4. Perform a targeted simulated‑annealing walk that preferentially moves
   mass away from indices that contribute most to the current worst
   convolution entry.
5. Repeat steps 1‑4 a few times (different random seeds) while respecting
   the overall time budget and keep the best result.
The implementation uses FFT‑based convolutions (O(n log n)) and projects
onto the probability simplex after each update.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max approximation of max(conv)."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))   # improve numerical stability
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _run_softmax_adam(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      lr: float = 0.01,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Adam optimisation on the soft‑max of the self‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    # linear temperature schedule (high -> low)
    T_start = 2.0e-2
    T_end = 5.0e-4
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] adam softmax improve: n={n} iter={it} score={cur_score:.6f}")

        # temperature for smoothing
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

    return best_a, best_score


def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.12,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] subgrad improve: n={n} iter={it} score={cur_score:.6f}")

        k = conv.argmax()
        grad = np.zeros_like(a)
        if k < n:
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 0.3,
                            end_temp: float = 1e-7,
                            steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that preferentially moves mass away from the
    indices that contribute most to the current worst convolution entry."""
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        conv = _conv_self(a)
        k = conv.argmax()
        # indices that actually appear in conv[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # contribution of each i to conv[k]
        contrib = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        contrib[idx] = a[idx] * a[k - idx]

        # pick a source index proportionally to its contribution
        total_contrib = contrib.sum()
        if total_contrib > 0:
            i = np.random.choice(n, p=contrib / total_contrib)
        else:
            i = np.random.randint(0, n)

        # pick a destination index outside the active interval (zero contribution)
        if lo > 0:
            candidates = list(range(0, lo))
        else:
            candidates = []
        if hi < n - 1:
            candidates.extend(range(hi + 1, n))
        if not candidates:
            j = np.random.randint(0, n)
        else:
            j = np.random.choice(candidates)

        if a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.30  # move up to 30 % of the mass at i
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] targeted anneal improve: n={n} iter={it} score={score:.6f}")

    return best_seq, best_score

"""
Proposes a non‑negative step‑function (a probability vector) that minimises the
``evaluate_sequence`` score.

Algorithm
---------
1. Warm‑start from the previously discovered best vector (`height_sequence_1`);
   if unavailable start from a random simplex point of length 9600.
2. Perform several independent optimisation runs while the global time budget
   (≈ 1000 s) allows it.  Each run consists of
   * Adam‑based soft‑max gradient descent (smooth approximation of the max‑convolution).
   * Exact sub‑gradient descent on the true max‑convolution.
   * A *targeted* simulated‑annealing walk that preferentially moves mass away
     from indices that contribute most to the current worst convolution entry.
3. Keep the best sequence over all runs, normalise it to sum = 1 and return it
   as a plain Python list.

All updates respect the simplex constraint (non‑negative, sum = 1) via Euclidean
projection.  FFT‑based convolutions give an O(n log n) cost per evaluation,
allowing many thousands of iterations inside the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution).
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max of the convolution."""
    maxc = conv.max()
    # stabilised exponentials
    scaled = (conv - maxc) / temperature
    scaled -= np.max(scaled)
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max objective.
# ----------------------------------------------------------------------
def _run_softmax_adam(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      lr: float = 0.02,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max of the self‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    # temperature schedule (smooth → sharp)
    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # temperature for the smooth max
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true max‑convolution.
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.15,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected (Nesterov) sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        k = conv.argmax()                     # index attaining the max

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        if k < n:
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement.
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 0.25,
                            end_temp: float = 1e-7,
                            steps: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that moves mass away from the indices that most contribute
    to the current worst convolution entry.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        conv = _conv_self(a)
        k = conv.argmax()
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)

        # contribution of each index to the worst entry
        contrib = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        contrib[idx] = a[idx] * a[k - idx]

        total = contrib.sum()
        if total > 0:
            i = np.random.choice(n, p=contrib / total)
        else:
            i = np.random.randint(0, n)

        # choose a destination outside the active interval (zero contribution)
        candidates = list(range(0, lo)) + list(range(hi + 1, n))
        if candidates:
            j = np.random.choice(candidates)
        else:
            j = np.random.randint(0, n)

        if a[i] <= 0.0 or i == j:
            continue

        # move a random fraction (up to 30 %) of the mass at i
        eps = a[i] * np.random.rand() * 0.30
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] targeted anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score
