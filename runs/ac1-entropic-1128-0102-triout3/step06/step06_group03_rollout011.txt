def _run_softmax_gd(a0, n, deadline, start_time, step0=1.0, beta=0.9):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    T_start = 0.5
    T_end = 1e-5
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # linearly decreasing temperature in logscale
        T = T_start * (T_end / T_start) ** (it / 50000.0)  # adjust scaling baseline
        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / np.sqrt(it)
        a = _project_simplex(a - step * momentum)
    return best_a, best_score

def _run_subgradient_descent(a0, n, deadline, start_time, step0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        k_max = np.argmax(conv)
        # build gradient g_i = 2 * a_{k_max - i} if in range else 0
        # Equivalent to reversed a shifted.
        grad = np.zeros_like(a)
        # indices i such that j = k_max - i in [0, n-1]
        # i in [0, n-1] and j = k_max - i in [0, n-1] => i in [max(0, k_max - (n-1)), min(n-1, k_max)]
        i_start = max(0, k_max - (n-1))
        i_end = min(n-1, k_max)
        i_range = np.arange(i_start, i_end+1)
        j = k_max - i_range
        grad[i_range] = 2.0 * a[j]  # vectorized
        # step size schedule
        step = step0 / np.sqrt(it)
        a = _project_simplex(a - step * grad)
    return best_a, best_score
```

def _run_swap_annealing(a0, n, deadline, start_time):
    a = a0.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    iteration = 0
    while time.time() < deadline:
        iteration+=1
        i,j = np.random.choice(n, size=2, replace=False)
        if a[i]==a[j]:
            continue
        a_swap = a.copy()
        a_swap[i], a_swap[j] = a_swap[j], a_swap[i]
        conv2 = _conv_self(a_swap)
        score2 = 2.0 * n * conv2.max()
        # acceptance prob
        # compute temp schedule
        t = iteration / 10000.0   # adjust
        T = max(0.01 * (1 - t), 1e-6)
        accept = score2 < best_score or np.random.rand() < np.exp(-(score2 - best_score) / T)
        if accept:
            a = a_swap
            conv = conv2
            if score2 < best_score:
                best_score = score2

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
* Warm‑start from the previously found best sequence (``height_sequence_1``)
  if it exists.
* Run several optimisation phases on two problem sizes (≈4800 and ≈9600):
  - **Projected sub‑gradient descent** on the exact max‑convolution.
  - **Projected soft‑max gradient descent** (smooth approximation of the max)
    with an exponential temperature schedule.
  - **Peak‑shaving local search** that repeatedly moves a small amount of
    probability mass away from indices that contribute most to the current
    maximal convolution entry.
* The phases are repeated many times with different random starts
  (random Dirichlet vectors, random permutations of the up‑sampled best
  solution, etc.).  The best sequence seen so far is always kept.
* When the time budget (≈1000 s) is almost exhausted a final local‑search
  stage is performed.
* The returned sequence is normalised to sum = 1 (scale‑invariant for the
  objective) and converted to a plain Python list.

The combination of a deterministic sub‑gradient step, a smooth soft‑max
optimiser and a focused stochastic local search gives a stronger
exploration than the previous implementation and typically pushes the
score below the 1.512 → 1.503 gap.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution a * a (full length = 2·len(a)‑1) using FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of reverse(a) with a weight vector w
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of τ·log∑exp(conv/τ)."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()                     # soft‑max weights

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)       # length = n + (2n‑1) – 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]       # central block of length n
    return grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.5,
                             beta: float = 0.9) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on f(a)=2·n·max(a*a)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, it={it}, score={cur_score:.6f}")

        # locate the index attaining the maximum
        k_max = int(np.argmax(conv))

        # build sub‑gradient g_i = 2·a_{k_max-i} (zero when index out of range)
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1)
        j = k_max - i_range
        grad[i_range] = 2.0 * a[j]

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected soft‑max gradient descent (smooth approximation of max)
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the smooth log‑sum‑exp objective."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (exponential decay)
    T_start = 2.0e-2
    T_end = 5.0e-5   # go a little lower than the previous code

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: "
                  f"n={n}, it={it}, score={cur_score:.6f}")

        # anneal temperature (log‑linear in iteration number)
        T = T_start * (T_end / T_start) ** (it / 60000.0)

        grad = _softmax_gradient(a, conv, T)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """Move a fraction of mass from donors that heavily contribute to the
    current maximal convolution entry to low‑contribution indices."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule for Metropolis acceptance
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to the maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor – sample proportionally to its contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient – choose a low‑contribution index
        low_thr = np.percentile(contrib, 20)            # bottom 20 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # transfer a fraction (up to 30 %) of donor mass
        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = (delta < 0) or (math.exp(-delta / (T + 1e-12)) > np.random.rand())

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, it={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

            break

        # allocate a slice of the remaining small‑phase budget
        slice_deadline = time.time() + (small_deadline - time.time()) / max(1, n_restarts_small - r)

        # initialise a0
        if best_seq is not None:
            a0 = _upsample(best_seq, n_small)
        elif starter is not None:
            a0 = _upsample(starter, n_small)
        else:
            # Dirichlet with α=1 (uniform) provides a random point on the simplex
            a0 = np.random.dirichlet(np.ones(n_small))
        # occasionally shuffle to explore different permutations
        if np.random.rand() < 0.5:
            a0 = a0[np.random.permutation(n_small)]

        # sub‑gradient descent (≈30 % of the slice)
        sub_dead = time.time() + (slice_deadline - time.time()) * 0.30
        a_sub, sc_sub = _run_subgradient_descent(a0, n_small, sub_dead, start_time)

        # soft‑max GD (≈50 % of the slice)
        sm_dead = time.time() + (slice_deadline - time.time()) * 0.50
        a_sm, sc_sm = _run_softmax_gd(a_sub, n_small, sm_dead, start_time)

        # local search (≈20 % of the slice)
        loc_dead = slice_deadline
        a_loc, sc_loc = _run_targeted_local_search(a_sm, n_small, loc_dead, start_time)

        # keep the best of the three results
        for seq, sc in [(a_sub, sc_sub), (a_sm, sc_sm), (a_loc, sc_loc)]:
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:6.2f}s] New best (n={n_small}) score={sc:.6f}")

    # --------------------------------------------------------------
    # Phase B – larger problem (≈9600) using the best found so far
    # --------------------------------------------------------------
    if time.time() < deadline:
        n_big = 9600
        # remaining time after Phase A is spent here
        while time.time() < deadline - 5.0:
            # each restart gets at most 30 s (or less if we are close to the deadline)
            slice_time = min(30.0, deadline - time.time())
            slice_deadline = time.time() + slice_time

            # initialise
            if best_seq is not None:
                a0 = _upsample(best_seq, n_big)
            elif starter is not None:
                a0 = _upsample(starter, n_big)
            else:
                a0 = np.random.dirichlet(np.ones(n_big))
            if np.random.rand() < 0.5:
                a0 = a0[np.random.permutation(n_big)]

            # sub‑gradient
            sub_dead = time.time() + (slice_deadline - time.time()) * 0.30
            a_sub, sc_sub = _run_subgradient_descent(a0, n_big, sub_dead, start_time)

            # soft‑max GD
            sm_dead = time.time() + (slice_deadline - time.time()) * 0.55
            a_sm, sc_sm = _run_softmax_gd(a_sub, n_big, sm_dead, start_time)

            # local search
            loc_dead = slice_deadline
            a_loc, sc_loc = _run_targeted_local_search(a_sm, n_big, loc_dead, start_time)

            for seq, sc in [(a_sub, sc_sub), (a_sm, sc_sm), (a_loc, sc_loc)]:
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:6.2f}s] New best (n={n_big}) score={sc:.6f}")

    # --------------------------------------------------------------
    # Optional final fine‑tuning on an even larger size if time permits
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0 and best_seq is not None:
        # try to double the resolution – this often yields a slight extra gain
        n_extra = min(18000, best_seq.shape[0] * 2)
        a0 = _upsample(best_seq, n_extra)

        # allocate the rest of the budget to a final soft‑max run
        a_final, sc_final = _run_softmax_gd(a0, n_extra, deadline, start_time)

        # final local search (very short)
        a_final, sc_final = _run_targeted_local_search(a_final, n_extra, deadline, start_time)

        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final best (n={n_extra}) score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure we always return a valid sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may appear due to rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list (the evaluation harness expects this)
    return [float(x) for x in best_seq]
