from scipy.optimize import linprog
import scipy.sparse as sp

def compute_b_via_lp(f, M):
    n = len(f)
    K = 2*n-1
    rows = []
    cols = []
    data = []
    for k in range(K):
        # i ranging max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # for i from i_start..i_end, j = k - i
        # So j = k - i ranges from k - i_end to k - i_start
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    c = -np.ones(n) # maximize sum b => minimize negative sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    return b

def _run_subgradient_descent(a0, n, deadline, start_time, max_iters=None):
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2*n - 1)
    best_score = float("inf")
    it = 0
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")
        # gradient computation
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]
        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score

def _solve_lp_for_b(f, M):
    n = f.shape[0]
    K = 2 * n - 1
    rows = []
    cols = []
    data = []
    for k in range(K):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M, dtype=float)
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

"""
Optimiser for the max‑convolution bound.
---------------------------------------

1. **LP‑step** – For a modest size (≈1024) we solve a linear program:
   maximise Σ b_j   subject to   b_j≥0  and  (f∗b)_k ≤ M,
   where M = max(f∗f).  The solution moves probability mass to
   positions that are “quiet’’ for the current shape and strictly
   reduces the maximum convolution value.

2. **Upsampling** – The improved small‑scale shape is linearly
   interpolated to the target length (4096) and re‑normalised.

3. **Sub‑gradient descent** – A projected sub‑gradient method on the
   true objective 2·n·max(conv) further refines the sequence.

4. **Targeted annealing** – Finally a short simulated‑annealing pass
   moves a few percent of mass away from the dominant convolution entry.

The whole pipeline respects the 1000 s budget and returns a list of
non‑negative heights (sum normalised to 1).  No closures or lambdas are
used, all helpers are top‑level functions.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def _solve_lp_for_b(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ b_j   s.t.   b≥0  and  (f∗b)_k ≤ M  for all k.
    Returns the optimal b (may be None if the solver fails).
    """
    n = f.shape[0]
    K = 2 * n - 1

    rows = []
    cols = []
    data = []

    # Build the (K x n) convolution matrix A_ub where
    # (A_ub @ b)_k = Σ_i f_i * b_{k‑i}
    for k in range(K):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M, dtype=float)

    c = -np.ones(n)                     # minimise –Σ b → maximise Σ b
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Phase‑2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    """
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)
    best_score = float("inf")
    it = 0
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient at the maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – Targeted simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-8,
                            steps: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Annealing that repeatedly moves a small amount of mass away from the
    pair of entries that contribute most to the current maximal convolution.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score
