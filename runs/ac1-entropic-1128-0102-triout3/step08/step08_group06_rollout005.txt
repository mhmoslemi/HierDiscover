"""
Propose a step‑function (non‑negative heights) that minimizes the
self‑convolution bound used in ``evaluate_sequence``.

The algorithm proceeds in four stages under the 1000 s budget:

1. **Exponential‑gradient warm‑up** – mirror‑descent on a log‑sum‑exp
   smooth approximation of the maximal convolution entry.  A long schedule
   of smoothing parameters supplies a good initial distribution.

2. **Projected sub‑gradient descent** – exact sub‑gradient of the true
   max (aggregated over all active constraints) with Euclidean simplex
   projection.  A modest step‑size and momentum give a reliable refinement.

3. **Targeted stochastic refinement** – many cheap O(n) moves that shift a
   small fraction of probability mass from indices that contribute most to
   the current worst convolution entry toward indices that contribute the
   least.  The move size decays to 1e‑5, allowing extremely fine adjustments.

4. **Deterministic fine‑refinement** – a greedy hill‑climber that repeatedly
   selects the source index with maximal contribution to the worst entry
   and the destination index with minimal counterpart, then performs a
   line‑search over several ε‑fractions.  This phase usually squeezes the
   last few percent of improvement needed to pass the target bound.

All phases keep track of the best sequence seen so far and finally return
the normalized distribution (sum = 1).  The code respects the time budget
and works from the global warm‑start ``height_sequence_1`` when available.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0          # numerical noise
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smooth max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 15000):
    """Multiplicative mirror‑descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)       # enough for a·w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max weights (log‑sum‑exp)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.2,
                             beta: float = 0.90,
                             max_iter: int = 10**7):
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        # build sub‑gradient (aggregate over all active constraints)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : stochastic targeted refinement (fast, fine‑grained)
# ----------------------------------------------------------------------
def _run_targeted_refine_optimized(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.02,
                                   min_eps_frac: float = 1e-5):
    """
    Stochastic refinement that repeatedly moves a small fraction of mass
    away from indices that contribute most to the current maximal convolution
    entry and toward indices that contribute the least.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linearly decay the move‑size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # source probabilities weighted by contribution to k_max
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # destination – prefer indices that *do not* contribute to k_max
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution (exact up to round‑off)
        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # periodically recompute the convolution from scratch to avoid drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Phase‑4 : deterministic fine‑grained refinement
# ----------------------------------------------------------------------
def _run_fine_refine(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float):
    """
    Greedy hill‑climber: repeatedly pick the index with maximal contribution
    to the current worst convolution entry and move a carefully chosen
    ε‑fraction of its mass toward an index with minimal counterpart.
    Several ε‑fractions are tried (binary‑like search) and the best improving
    move is accepted.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1
        k_max = int(conv.argmax())

        # contributions of each index to the current worst entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]

        src_candidates = np.flatnonzero(contrib > 0)
        if src_candidates.size == 0:
            src = rng.integers(n)
        else:
            src = src_candidates[np.argmax(contrib[src_candidates])]

        # counterpart values a[j] for each possible destination
        counterpart = np.full(n, np.inf)
        counterpart[valid] = a[j[valid]]
        dst = np.argmin(counterpart)
        if dst == src:
            counterpart[dst] = np.inf
            dst = np.argmin(counterpart)

        eps_max = a[src]
        if eps_max <= 0:
            continue

        improved = False
        # try a decreasing list of fractions; the earliest that improves is taken
        for frac in (0.5, 0.25, 0.125, 0.0625, 0.03125,
                     0.015625, 0.0078125, 0.00390625, 0.001953125):
            eps = eps_max * frac
            if eps <= 0:
                continue
            # fast O(n) delta update
            delta = np.zeros(conv_len, dtype=float)
            delta[dst:dst + n] += 2.0 * eps * a
            delta[src:src + n] -= 2.0 * eps * a
            delta[2 * src] += eps * eps
            delta[2 * dst] += eps * eps
            delta[src + dst] -= 2.0 * eps * eps

            conv_new = conv + delta
            new_max = conv_new.max()
            if new_max < max_val - 1e-15:
                # accept move
                a[src] -= eps
                a[dst] += eps
                conv = conv_new
                max_val = new_max
                improved = True
                if new_max < best_max - 1e-15:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] fine refine improve: "
                          f"iter={it:,}, max={best_max:.8f}")
                break

        if not improved:
            # occasional random perturbation as escape mechanism
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-16:
                continue
            eps = a[src] * 0.01 * rng.random()
            if eps <= 0:
                continue
            delta = np.zeros(conv_len, dtype=float)
            delta[dst:dst + n] += 2.0 * eps * a
            delta[src:src + n] -= 2.0 * eps * a
            delta[2 * src] += eps * eps
            delta[2 * dst] += eps * eps
            delta[src + dst] -= 2.0 * eps * eps
            conv_new = conv + delta
            new_max = conv_new.max()
            if new_max < max_val - 1e-15:
                a[src] -= eps
                a[dst] += eps
                conv = conv_new
                max_val = new_max
                if new_max < best_max - 1e-15:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] fine refine random improve: "
                          f"iter={it:,}, max={best_max:.8f}")

        # full recomputation occasionally to avoid accumulated numerical drift
        if it % 60000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Initialise – warm‑start from the optional global variable.
    # ------------------------------------------------------------------
    final_n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            a0 = _upsample(starter, final_n)
        else:
            raise Exception()
    except Exception:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – exponential‑gradient warm‑up
    # ------------------------------------------------------------------
    phase1_end = min(deadline, time.time() + min(30.0, 0.05 * (deadline - time.time())))
    a_warm, sc_warm = _run_exponential_gradient(
        a0,
        final_n,
        phase1_end,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0],
        eta0=0.5,
        steps_per_alpha=15000,
    )
    if sc_warm < best_score:
        best_score = sc_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={sc_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (medium‑scale)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_end = min(deadline, time.time() + max(30.0, 0.40 * (deadline - time.time())))
        a_sub, sc_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_end,
            start_time,
            step0=0.2,
            beta=0.90,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic targeted refinement (lots of cheap moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refine = _run_targeted_refine_optimized(
            best_seq,
            final_n,
            deadline,
            start_time,
            init_eps_frac=0.02,
            min_eps_frac=1e-5,
        )
        conv = _conv_self(a_refine)
        sc_ref = 2.0 * final_n * conv.max()
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_refine.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve: score={sc_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – deterministic fine‑grained refinement (last % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_fine = _run_fine_refine(best_seq, final_n, deadline, start_time)
        conv = _conv_self(a_fine)
        sc_fine = 2.0 * final_n * conv.max()
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 4 improve: score={sc_fine:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return (list of floats)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
