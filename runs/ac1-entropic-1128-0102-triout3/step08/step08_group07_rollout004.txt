"""
Improved step‑function search.

The algorithm follows the same overall structure as the reference
implementation but invests more time in a *global* simulated‑annealing
phase (mass‑reshuffling of several cells at once) and explores a larger
variety of structured initialisations (Gaussian, Hann, Blackman,
random Dirichlet).  The optimisation pipeline is:

 * Stage 0 – many cheap restarts on a modest size (n≈1024) to get a
   promising shape.
 * Stage 1 – upscale to the target size (n=8192) and run a short Adam
   optimisation of the soft‑max surrogate.
 * Stage 2 – projected Nesterov sub‑gradient descent on the true
   max‑convolution for a few minutes.
 * Stage 3 – *heavy* simulated‑annealing that repeatedly reshuffles the
   mass of a random subset of indices; this explores far from the
   current basin.
 * Stage 4 – a final polishing sub‑gradient pass.

All stages respect the global time budget (default 1000 s).  The
returned sequence is a plain Python list of non‑negative floats;
the harness will evaluate it with ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than ``x``."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥0, sum fixed)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.20,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    with Adam and Euclidean projection onto the simplex.
    Returns the best vector seen and its *true* (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline:
        it += 1

        # true objective (for logging)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: "
                  f"iter={it:,} score={cur_score:.6f} τ={tau:.2e}")

        # soft‑max surrogate
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.14) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    Returns the best vector found.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Heavy simulated‑annealing (mass reshuffling)
# ----------------------------------------------------------------------


def _run_heavy_annealing(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         cur_best_score: float,
                         cur_best_seq: np.ndarray,
                         start_temp: float = 2e-2,
                         end_temp: float = 1e-7,
                         steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that repeatedly reshuffles the mass of a random
    subset of indices (size 2‑5).  The acceptance rule uses the true
    objective 2·n·max(conv).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a random subset (size 2–5)
        k = rng.integers(2, 6)            # 2,3,4,5
        idx = rng.choice(n, size=k, replace=False, p=a / a.sum())
        # keep total mass of the subset unchanged, redistribute it randomly
        total = a[idx].sum()
        if total == 0.0:
            continue
        # generate a fresh Dirichlet sample on the subset
        new_weights = rng.random(k)
        new_weights = np.maximum(new_weights, 0.0)
        new_weights /= new_weights.sum()
        a_prop = a.copy()
        a_prop[idx] = total * new_weights

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0.0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisers (Chebyshev, Kaiser, Hann, Blackman, Gaussian,
# random Dirichlet)
# ----------------------------------------------------------------------


def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    The type is chosen with probabilities:
        30 % Chebyshev, 30 % Kaiser, 20 % Hann/Blackman, 20 % Gaussian/Dirichlet.
    """
    # try to import scipy.signal for the classic windows
    try:
        from scipy.signal import chebwin, kaiser, hann, blackman
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.30:
        # Chebyshev window
        att = rng.uniform(30, 120)                # dB attenuation
        w = chebwin(n, at=att, sym=True).astype(float)
    elif have_signal and r < 0.60:
        # Kaiser window
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
    elif have_signal and r < 0.80:
        # Hann or Blackman (random choice)
        if rng.random() < 0.5:
            w = hann(n, sym=True).astype(float)
        else:
            w = blackman(n, sym=True).astype(float)
    else:
        # Gaussian or plain Dirichlet (random choice)
        if rng.random() < 0.5:
            sigma = rng.uniform(0.08, 0.25)
            x = np.linspace(-0.5, 0.5, n)
            w = np.exp(-0.5 * (x / sigma) ** 2)
        else:
            w = rng.random(n)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The search proceeds through several stages
    (quick random restarts → Adam → sub‑gradient descent → heavy
    simulated annealing → final polishing) while respecting the time
    budget.
    The returned list is a plain Python list of floats (∑a_i = 1,
    a_i ≥ 0).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Use a previous solution as a starter if present
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = np.maximum(tmp, 0.0)
                s = starter.sum()
                if s > 0:
                    starter /= s
                else:
                    starter = None
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – rapid cheap restarts on a small size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    remaining = deadline - time.time()
    stage0_time = max(40.0, 0.08 * remaining)       # ~8 % of the budget, at least 40 s
    slice_deadline = time.time() + stage0_time

    # we will perform a handful of restarts (bounded by time)
    r = 0
    while time.time() < slice_deadline:
        # initialise either from the global starter (once) or randomly
        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = _window_initialiser(n0, rng)

        # initial temperature for Adam (based on the current max‑conv)
        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 6.0, 1e-8)

        a_opt, sc = _run_adam_softmax(
            a0, n0, slice_deadline, start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9995,
            step_size=0.22,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r} score={sc:.6f}")
        r += 1

    # ------------------------------------------------------------------
    # Stage 1 – upsample to target size and perform a moderate Adam run
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_target = 8192
    a_target = _upsample(best_seq, n_target)

    conv1 = _conv_self(a_target)
    tau0_1 = max(conv1.max() * 6.0, 1e-10)

    remaining = deadline - time.time()
    stage1_time = max(30.0, 0.18 * remaining)       # about 18 % of what is left
    slice_deadline = time.time() + stage1_time

    a_opt1, sc1 = _run_adam_softmax(
        a_target, n_target, slice_deadline, start_time,
        tau0=tau0_1,
        tau_end=1e-13,
        tau_decay=0.9994,
        step_size=0.18,
    )
    if sc1 < best_score:
        best_score = sc1
        best_seq = a_opt1.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={sc1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent on the true objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        grad_deadline = deadline - 4.0               # keep a few seconds for annealing
        a_opt2, sc2 = _run_subgradient_descent(
            best_seq, n_target, grad_deadline, start_time,
            beta=0.97,
            step0=0.13,
        )
        if sc2 < best_score:
            best_score = sc2
            best_seq = a_opt2.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={sc2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – heavy simulated‑annealing (mass reshuffling)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _run_heavy_annealing(
            best_seq,
            n_target,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=3e-2,
            end_temp=1e-7,
            steps=2_000_000,          # more moves than the reference code
        )
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal final improve: "
                  f"score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – a short polishing sub‑gradient pass (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = deadline
        a_opt4, sc4 = _run_subgradient_descent(
            best_seq, n_target, polish_deadline, start_time,
            beta=0.96,
            step0=0.12,
        )
        if sc4 < best_score:
            best_score = sc4
            best_seq = a_opt4.copy()
            print(f"[{time.time() - start_time:6.2f}s] final polish improve: "
                  f"score={sc4:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation onto the simplex (guard against tiny drift)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list – the harness will call evaluate_sequence
    return [float(x) for x in best_seq]
