def _subgradient_active(a0, n, deadline, start_time, best_score_initial):

    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    eps_rel = 1e-12
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub-Active iter={it:,} score={cur_score:.6f}")
        # Determine active set
        # Use tolerance relative to max.
        tol = max(1e-12, cur_max * 1e-8)  # small tolerance
        active_idx = np.where(conv >= cur_max - tol)[0]
        # accumulate gradient
        grad = np.zeros_like(a)
        # for each active k, add gradient
        for k in active_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(active_idx))  # average
        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)
    return best_a, best_score

"""
Proposes a step‑function (a non‑negative vector that sums to one) that
minimises 2·n·max(conv(a,a)).  

The optimiser follows three stages:

1. **Adam on a smooth surrogate** – we minimise the log‑sum‑exp
   approximation of the max‑convolution, gradually increasing the
   smoothing parameter α (up to 2000).  This gives a high‑quality
   “continuous’’ optimum while staying on the probability simplex.

2. **Projected sub‑gradient with an active‑set** – after the smooth phase
   we switch to the exact objective.  The sub‑gradient is taken as the
   average of the gradients of all convolution entries that are within a
   tiny tolerance of the current maximum; this yields a more stable
   descent direction than using a single index.

3. **Light simulated‑annealing polish** – a cheap mass‑transfer annealing
   finishes the search, moving small amounts of probability mass between
   coordinates while respecting the simplex constraint.

The whole routine respects the given time budget (≈1000 s) and returns
the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_smooth_improved(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alpha_start: float = 0.5,
    alpha_end: float = 2000.0,
    max_iters: int = 20000,
    lr0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on f_α(a)= (1/α)·log∑exp(α·conv(a,a)).
    α grows linearly from ``alpha_start`` to ``alpha_end``.
    Returns the best iterate (lowest true objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    L_grad = _next_pow2(3 * n - 2)          # length for the gradient FFTs

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current objective (exact)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # increase smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_rev = w[::-1]

        # gradient of the smoothed objective
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Projected sub‑gradient descent with active‑set averaging
# ----------------------------------------------------------------------
def _subgradient_active(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The sub‑gradient is the average of the gradients of all
    convolution entries that are within a tiny tolerance of the current
    maximum.  Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0

    # length for FFT‑based gradient computation
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑active iter={it:,} score={cur_score:.6f}")

        # active set: all indices within a relative tolerance of the maximum
        tol = max(1e-12, cur_max * 1e-9)
        active = np.where(conv >= cur_max - tol)[0]

        # build weight vector (average over active set)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        w_rev = w[::-1]

        # gradient = average of gradients of active entries
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : Tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Light mass‑transfer annealing.  Temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportional to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.  The routine
    combines a smooth Adam phase, an active‑set sub‑gradient phase and a
    final annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 works well with FFT‑friendly lengths.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Initialise with the global starter if it exists.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter /= starter.sum()
            else:
                starter = np.full(n, 1.0 / n)
        else:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    best_seq = starter.copy()
    best_score = 2.0 * n * _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:.2f}s] Initial starter score={best_score:.6f}")

    # --------------------------------------------------------------
    # Single long optimisation run (Adam → active‑set SGD → anneal)
    # --------------------------------------------------------------

    # --- Phase‑1 : Adam on smooth surrogate -----------------------
    remaining = deadline - time.time()
    # keep at least ~200 s for the final stages
    adam_deadline = time.time() + max(0.0, remaining - 200.0)

    a_adam, score_adam = _adam_smooth_improved(
        best_seq,
        n,
        adam_deadline,
        start_time,
        alpha_start=0.5,
        alpha_end=2000.0,
        max_iters=20000,
        lr0=0.5,
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] Adam improved: score={score_adam:.6f}")

    # --- Phase‑2 : sub‑gradient with active set -------------------
    if time.time() < deadline - 100.0:
        sub_deadline = time.time() + (deadline - time.time()) * 0.5
        a_sub, score_sub = _subgradient_active(
            best_seq,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑active improve: score={score_sub:.6f}")

    # --- Phase‑3 : annealing polish --------------------------------
    if time.time() < deadline:
        a_final, score_final = _annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=300_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and cast to plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
