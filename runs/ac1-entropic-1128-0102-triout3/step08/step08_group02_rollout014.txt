def subgrad_backtrack(a, deadline):
    best_a = a.copy()
    best_score = compute_score(a)
    step_initial = 0.5
    it = 0
    fft_len = _next_pow2(3*n-2)
    while time.time() < deadline:
        it +=1
        conv = _conv_self(a)
        # compute subgrad
        grad = _max_subgrad(a, conv, rel_tol=1e-6, fft_len=fft_len)
        # try step sizes decreasing until improvement
        eta = step_initial / math.sqrt(it) # base step
        improved = False
        for trial in range(5):
            a_prop = a * np.exp(-eta * grad)
            s = a_prop.sum()
            if s > 0:
                a_prop /= s
            else:
                a_prop = np.full_like(a, 1.0/n)
            score_prop = compute_score(a_prop)
            if score_prop < best_score - 1e-12:
                best_score = score_prop
                best_a = a_prop.copy()
                a = a_prop
                improved = True
                break
            else:
                eta *= 0.5
        if not improved:
            # maybe reduce step size for next iterations
            pass
        # occasionally print
        if it % 500 == 0:
            print(f"[{time.time()-start_time:.2f}s] subgrad backtrack iter={it}, score={best_score:.6f}")
    return best_a, best_score

def greedy_local_search(a, deadline):
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    no_improve = 0
    while time.time() < deadline and no_improve < 2000:
        conv = _conv_self(a)
        k_max = conv.argmax()
        lo = max(0, k_max-(n-1))
        hi = min(n-1, k_max)
        i_idx = np.arange(lo, hi+1)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]   # index of source side (i) where product largest.
        # choose dest as smallest coefficient but not src
        dest_candidates = np.delete(np.arange(n), src)
        dst = dest_candidates[a[dest_candidates].argmin()]
        eps = a[src] * 0.2  # maybe 20% mass
        if eps < 1e-12:
            break
        a_prop = a.copy()

            best_score = score_prop

"""
Hybrid optimiser for the step‑function bound (improved).

The method focuses on a single, strong optimisation pipeline:

1. **Log‑softmax Adam** – the coefficients are represented by logits.
   A smooth log‑sum‑exp approximation of the maximal self‑convolution
   is minimised while the smoothing parameter α is increased from very
   coarse (α≈0.5) to very sharp (α≈2000).  This quickly drives the
   solution into a good basin.

2. **Exponentiated‑gradient descent with back‑tracking line‑search**
   on the exact max‑convolution.  The sub‑gradient (active set) is
   obtained via one FFT‑based cross‑correlation.  At each iteration an
   exponential update `a←a·exp(−η·g)` is tried with several step‑sizes;
   the first that yields a strict improvement is accepted.  This yields
   a more aggressive descent than the plain 1/√t schedule.

3. **Deterministic greedy peak‑reduction** – repeatedly moves a few
   percent of mass from the coefficient that contributes most to the
   current maximal convolution entry to the smallest coefficient.
   The move is accepted only when it lowers the objective.

Multiple diverse starts (previous best, symmetric exponential, U‑shape,
random smooth) are tried, each sharing the overall 1000 s budget.  The
best sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·conv_k) w.r.t. a.
    Returns a vector of shape (n,).
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                       # soft‑max weights
    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def compute_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` (no clipping)."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

def _adam_phase_improved(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> np.ndarray:
    """
    Adam minimisation of a log‑sum‑exp smooth approximation of the max
    self‑convolution.  The smoothing parameter α is increased gradually.
    Returns the best sequence (in the simplex) seen during the phase.
    """
    n = a0.shape[0]
    y = np.log(np.maximum(a0, 1e-16))        # logits
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # schedule of smoothing parameters (more fine‑grained than the original)
    alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
                   80.0, 160.0, 320.0, 640.0, 1280.0, 2560.0]
    iters_per_alpha = 8000
    fft_len = _next_pow2(3 * n - 2)

    best_seq = a0.copy()
    best_score = compute_score(best_seq)

    for alpha in alpha_sched:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq
            a = _softmax_from_logits(y)

            # smooth objective: (1/α)·log‑sum‑exp(α·conv)
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val                     # the true objective proxy

            # gradient w.r.t. a (smooth)
            grad_a = _smooth_max_grad(a, conv, alpha, fft_len)
            grad_a = grad_a * (2.0 * n)                # chain for the factor 2·n

            # chain rule to logits (soft‑max Jacobian)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 500 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                          f"iter={i:,}, score={sc:.6f}")

    return best_seq


def _subgrad_descent_backtrack(a0: np.ndarray,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution with a
    simple back‑tracking line‑search.  Returns the best sequence seen
    and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        grad = _max_subgrad(a, conv, rel_tol=1e-6, fft_len=fft_len)

        # base step decays like 1/√t  (larger early steps)
        base_step = 0.5 / math.sqrt(it)

        improved = False

                improved = True

        if it % 1000 == 0:

def _greedy_peak_reduction(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Deterministic moves that try to lower the current peak of the
    self‑convolution.  A fixed fraction (≈30‑25 %) of the mass from the
    biggest contributor to the peak is transferred to the smallest
    coefficient.  The move is kept only if it strictly reduces the
    objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = compute_score(a)
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]           # index that contributes most

        # destination = smallest coefficient (avoid src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.25                      # move 25 % of src's mass
        if eps < 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = compute_score(a_prop)
        if new_score < best_score - 1e-12:
            a = a_prop
            best_score = new_score
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: score={best_score:.6f}")
        else:
            no_improve += 1

    return a, best_score

    else:
        smooth = np.full(n, 1.0 / n)
    starts.append(smooth)

    # --------------------------------------------------------------
    # Global best tracker
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Optimise each start independently, sharing the global time budget.
    # --------------------------------------------------------------
    for idx_start, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{idx_start + 1}")

        a = np.maximum(a0, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # ----- 1. Adam phase (smooth approximation) -----
        phase_deadline = time.time() + min(0.12 * (deadline - start_time), 150.0)
        a_adam = _adam_phase_improved(a, phase_deadline, start_time)

        # ----- 2. Exponentiated‑gradient with backtrack -----
        if time.time() < deadline:
            remaining = deadline - time.time()
            sub_deadline = time.time() + 0.7 * remaining   # most of the remaining budget
            a_sub, score_sub = _subgrad_descent_backtrack(a_adam, sub_deadline, start_time)
        else:
            a_sub, score_sub = a_adam, compute_score(a_adam)

        # ----- 3. Deterministic greedy peak‑reduction -----
        if time.time() < deadline:
            remaining = deadline - time.time()
            peak_deadline = time.time() + 0.15 * remaining
            a_peak, score_peak = _greedy_peak_reduction(a_sub, peak_deadline, start_time)
        else:
            a_peak, score_peak = a_sub, score_sub

        # ----------------------------------------------------------
        # Pick the best amongst the stages for this start.
        # ----------------------------------------------------------
        stage_scores = [
            (compute_score(a_adam), a_adam),
            (score_sub, a_sub),
            (score_peak, a_peak),
        ]
        local_best_score, local_best_seq = min(stage_scores, key=lambda x: x[0])

        # ----------------------------------------------------------
        # Update the global best if improved.
        # ----------------------------------------------------------
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST *** "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = compute_score(best_seq)
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # Convert to plain Python list of floats as required.
    return [float(x) for x in best_seq.tolist()]
