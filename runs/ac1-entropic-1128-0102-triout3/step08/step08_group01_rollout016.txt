def _run_exponential_gradient(a0, n, deadline, start_time, alphas, eta0=0.2, iters_per_alpha=5000):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2*n-1)).max()
    L = _next_pow2(2*n-1)
    iteration = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            iteration += 1
            conv, grad, _ = _compute_conv_and_grad(a, alpha, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] EG improve: iter={iteration:,}, score={cur_score:.6f}")
            # update
            eta = eta0 / np.sqrt(iteration)
            # clip grad to avoid overflow
            grad_clipped = np.clip(grad, 0, 10)
            a = a * np.exp(-eta * grad_clipped)
            # ensure positivity and renorm
            a = np.maximum(a, 0.0)
            total = a.sum()
            if total > 0:
                a /= total
            else:
                a = np.full_like(a, 1.0/n)
    return best_a, best_score

"""
Hybrid exponentiated‑gradient / sub‑gradient / pair‑annealing optimiser.

The goal is to flatten the self‑convolution a*a as much as possible.
The algorithm works in three stages:

1️⃣  Exponentiated‑gradient descent on a smooth (log‑sum‑exp) approximation
    of the maximum convolution entry.  The update a←a·exp(‑η∇) keeps the
    vector non‑negative and the subsequent renormalisation enforces the
    simplex constraint.

2️⃣  Projected sub‑gradient descent on the true objective
    f(a)=2·n·max(conv(a,a)).  This quickly removes any remaining spikes.

3️⃣  Targeted simulated‑annealing in which the two indices (i,j) that
    contribute most to the current maximal convolution entry are weakened
    by moving a fraction of their mass to random locations.  The temperature
    schedule is exponential and the acceptance rule follows the Metropolis
    criterion.

If time permits the routine performs a few short random restarts, each
starting from a small perturbation of the current best solution.

All helper functions are top‑level; no closures or lambdas are used.
The total running time is limited to the supplied budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv).  The normalization Σa=1 is assumed."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient on the smooth max
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      alphas,
                      eta0: float = 0.3,
                      iters_per_alpha: int = 8000):
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate of max(conv).
    Returns the best iterate (lowest true score) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(a, _next_pow2(2 * n - 1)), n)

    L = _next_pow2(2 * n - 1)
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            # current convolution and score
            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ---- smooth‑max weights (log‑sum‑exp) ----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # ---- gradient of the smooth max (2·corr) ----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ---- exponentiated update ----
            eta = eta0 / math.sqrt(it)
            # Clip gradient to avoid overflow in the exponential
            grad = np.clip(grad, 0.0, 10.0)
            a = a * np.exp(-eta * grad)

            # renormalise onto the simplex
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses a simple momentum scheme and a 1/√t step‑size decay.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient coming from the maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection onto simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted pair‑annealing
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-8,
                        steps: int = 150_000):
    """
    Simulated annealing that always attacks the two entries that contribute
    most to the current maximal convolution coefficient.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # index of the current maximal convolution entry
        k = int(conv.argmax())

        # all i that can participate in this sum
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contribution of each i to the peak (i paired with j = k‑i)
        contrib = a[idx_range] * a[k - idx_range]

        # pick the pair (i,j) with largest product
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max

        # if the two positions coincide we treat it as a single source
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # choose distinct destination indices (avoid the sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose moving a fraction of the source masses
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20   # ≤20 % of source mass
            a_prop[s] -= eps
            a_prop[d] += eps

        # evaluate the proposal
        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score
