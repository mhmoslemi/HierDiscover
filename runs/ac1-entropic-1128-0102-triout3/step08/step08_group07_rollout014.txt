def _softmax_objective(w, n, tau):
    a = _softmax(w) # returns normalized vector (sum=1)
    conv = _conv_self(a)
    # shift for numerical stability
    z = conv / tau
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    sum_exp = np.sum(exp_z)
    obj = 2 * n * tau * (np.log(sum_exp) + z_max)
    # gradient w.r.t a
    p = exp_z / sum_exp  # shape len(conv)
    # compute grad_a as described: grad = 4 * n * inner
    a_rev = a[::-1]
    conv_rev = _conv_fft(a_rev, p) # length n+ len(p) -1 = 3n-2? Actually a_rev length n; p length 2n-1 => conv_rev length 3n-2.
    # inner of length n: conv_rev[n-1 : n-1 + n]
    inner = conv_rev[n-1:n-1+n]
    grad_a = 4 * n * inner
    # Convert grad w.r.t w via softmax Jacobian: g_w = a * (grad_a - avg)
    avg = np.dot(a, grad_a)
    grad_w = a * (grad_a - avg)
    return obj, grad_w

def _run_lbfgs_softmax(a0, n, deadline, start_time, max_iter_per_phase=500, tau0=None):
    """
    Minimize smooth surrogate using L-BFGS-B on logits w (softmax). Returns best a and best 2*n*max(conv) score.
    """
    # compute initial tau if not provided
    if tau0 is None:
        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 8.0, 1e-8)
    # We'll use a continuation schedule: start with tau = tau0, then reduce gradually to tau_end after each phase.
    tau_end = 1e-13
    tau = tau0
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    # Represent current logits
    # Avoid zeros: set w = log(a + eps)
    eps = 1e-12
    w = np.log(a + eps)
    # We'll iteratively run L-BFGS for a number of phases while tau decays
    phase = 0
    while time.time() < deadline and tau > tau_end:
        phase += 1
        # define objective function closure using current tau
        def fun_and_grad(w_vec):
            # compute a = softmax(w)
            # use stable softmax
            w_max = np.max(w_vec)
            exp_w = np.exp(w_vec - w_max)
            a_vec = exp_w / exp_w.sum()
            conv = _conv_self(a_vec)
            # softmax weights for surrogate
            z = conv / tau
            z_max = np.max(z)
            exp_z = np.exp(z - z_max)
            sum_exp = np.exp(z_max) * np.sum(exp_z)  # but we only need log sum exp
            # Actually obj = 2*n*tau*log(sum(exp(z))) ; we have exp(z) = exp(z_max)*exp_z; so log(sum(exp(z))) = z_max + log(sum(exp_z))
            obj = 2.0 * n * tau * (z_max + np.log(np.sum(exp_z)))
            # gradient w.r.t a
            p = exp_z / np.sum(exp_z)   # note p sums to 1
            a_rev = a_vec[::-1]
            conv_rev = _conv_fft(a_rev, p)
            inner = conv_rev[n-1:n-1+n]
            grad_a = 4.0 * n * inner
            # gradient w.r.t w via Jacobian of softmax
            avg = np.dot(a_vec, grad_a)
            grad_w = a_vec * (grad_a - avg)
            return obj, grad_w

        # define callback to monitor improvement and break if time nearly out
        best_obj_local = np.inf
        def callback(w_vec):
            nonlocal best_a, best_score, best_obj_local
            if time.time() >= deadline:
                raise StopIteration
            # compute current a and score
            w_max = np.max(w_vec)
            exp_w = np.exp(w_vec - w_max)
            a_vec = exp_w / exp_w.sum()
            conv = _conv_self(a_vec)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_vec.copy()
                print(f"[{time.time() - start_time:6.2f}s] lbfgs improve (tau={tau:.2e}): score={cur_score:.6f}")
            # also record best obj (not used)
            # optional early stop if not improving
        # run L-BFGS
        try:
            res = minimize(fun=fun_and_grad, x0=w, method='L-BFGS-B', jac=True,
                           callback=callback,
                           options={'maxiter': max_iter_per_phase, 'disp': False})
        except StopIteration:
            # time out
            break
        # update w to solution
        w = res.x
        # compute a from final w
        w_max = np.max(w)
        exp_w = np.exp(w - w_max)
        a = exp_w / exp_w.sum()
        # optional symmetrization
        a = (a + a[::-1]) / 2.0
        a /= a.sum()
        # reduce tau for next phase
        tau *= 0.5  # geometric decay
    return best_a if best_a is not None else a0, best_score

def _gaussian_initialiser(n, rng, sigma_factor=0.15):
    x = np.arange(n)
    center = (n-1)/2
    sigma = sigma_factor * n
    w = np.exp(- ((x-center)**2) / (2*sigma*sigma))
    w += rng.random(n) * 1e-6  # add small noise
    w = np.maximum(w, 0)
    w /= w.sum()
    return w

def _lbfgs_obj_grad(w_vec, n, tau):
    # compute a = softmax
    w_max = np.max(w_vec)
    exp_w = np.exp(w_vec - w_max)
    a = exp_w / exp_w.sum()

    z_max = np.max(z)
    exp_z = np.exp(z - z_max)

"""
Hybrid optimizer for the step‑function challenge.

The search consists of four phases:

1️⃣  Multiple short Adam runs on a soft‑max surrogate (τ‑smoothing).  
    Initial vectors are drawn from several structured windows
    (Chebyshev, Kaiser, Gaussian) and from a Dirichlet draw.

2️⃣  Upsampling to a larger array (≈8192 entries) and a continuation of
    the Adam phase with a finer cooling schedule.

3️⃣  A deterministic L‑BFGS‑B refinement on the same smoothed objective.
    The variables are the logits of a soft‑max parametrisation,
    guaranteeing positivity and Σa_i = 1 for free.
    Between continuation steps we symmetrise the candidate
    (a ← (a + rev(a))/2) which empirically lowers the max‑convolution.

4️⃣  Projected Nesterov sub‑gradient descent on the true objective
    2·n·max(a∗a) followed by a lightweight simulated‑annealing polishing
    that performs mass‑transfer moves.

All phases respect the 1000 s budget; the best sequence seen so far is
returned as a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa = 1)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate Oτ(a)
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9994,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """
    Minimise O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)  with Adam
    and projection onto the probability simplex.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true (unsmoothed) score for logging
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # occasional symmetrisation – helps to flatten the autocorrelation
        if it % 50 == 0:
            a = (a + a[::-1]) / 2.0
            a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

        # symmetrisation every few hundred steps
        if it % 200 == 0:
            a = (a + a[::-1]) / 2.0
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    Acceptance probability uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source chosen proportional to current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisation helpers (Chebyshev / Kaiser / Gaussian)
# ----------------------------------------------------------------------


def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    With probability 0.4 we produce a Chebyshev window,
    with probability 0.4 a Kaiser window,
    otherwise a Gaussian window, finally falling back to a Dirichlet draw.
    """
    # try to import scipy.signal lazily – if unavailable fall back to random
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        # Chebyshev window: attenuation between 30 dB and 120 dB
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
    elif have_signal and r < 0.8:
        # Kaiser window: beta in [0, 14]
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
    elif r < 0.95:
        # Gaussian window – σ chosen as a fraction of the length
        sigma = rng.uniform(0.08, 0.18) * n
        x = np.arange(n) - (n - 1) / 2.0
        w = np.exp(-0.5 * (x / sigma) ** 2)
    else:
        # Simple Dirichlet draw
        w = rng.random(n)

    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# L‑BFGS‑B refinement on the soft‑max surrogate
# ----------------------------------------------------------------------


def _lbfgs_obj_grad(w_vec: np.ndarray, n: int, tau: float):
    """
    Objective and gradient for the soft‑max surrogate
          Oτ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    where a = softmax(w_vec).
    Returns (obj, grad_w) where grad_w respects the soft‑max Jacobian.
    """
    # soft‑max to obtain a (guarantees Σa = 1, a_i ≥ 0)
    w_max = np.max(w_vec)
    exp_w = np.exp(w_vec - w_max)
    a = exp_w / exp_w.sum()

    # linear self‑convolution
    conv = _conv_self(a)

    # soft‑max over the convolution values
    z = conv / tau
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    sum_exp_z = np.sum(exp_z)

    # smooth objective
    obj = 2.0 * n * tau * (z_max + np.log(sum_exp_z))

    # gradient w.r.t. a
    p = exp_z / sum_exp_z                      # shape (2·n‑1,)
    a_rev = a[::-1]
    conv_rev = _conv_fft(a_rev, p)             # length 3·n‑2
    inner = conv_rev[n - 1:n - 1 + n]           # Σ_j a_j·p_{i+j}
    grad_a = 4.0 * n * inner

    # convert ∂O/∂a to ∂O/∂w via the Jacobian of soft‑max
    avg = np.dot(a, grad_a)
    grad_w = a * (grad_a - avg)
    return obj, grad_w


def _run_lbfgs_softmax(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       tau0: float = None,
                       tau_end: float = 1e-13,
                       tau_decay: float = 0.5,
                       max_iter_phase: int = 250) -> tuple[np.ndarray, float]:
    """
    Continuation of the soft‑max surrogate minimisation with L‑BFGS‑B.
    The optimisation variables are the logits w (soft‑max parametrisation).
    Returns the best vector found (still on the simplex) and its true score.
    """
    from scipy.optimize import minimize

    a = a0.copy()
    eps = 1e-12
    w = np.log(a + eps)                     # initialise logits

    # initialise temperature if not supplied
    if tau0 is None:
        tau = max(_conv_self(a).max() * 8.0, 1e-8)
    else:
        tau = tau0

    best_a = a.copy()
    best_score = float('inf')

    while time.time() < deadline and tau > tau_end:
        # L‑BFGS‑B on the smooth surrogate with fixed τ
        try:
            res = minimize(_lbfgs_obj_grad, w,
                           args=(n, tau),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter_phase,
                                    'disp': False})
        except Exception:
            break                                          # any numerical mishap → stop

        w = res.x
        # retrieve a from logits
        w_max = np.max(w)
        exp_w = np.exp(w - w_max)
        a = exp_w / exp_w.sum()

        # symmetrise – a cheap way to reduce the peak of the autocorrelation
        a = (a + a[::-1]) / 2.0
        a = _project_simplex(a)

        # update logits after symmetrisation
        w = np.log(a + eps)

        # true (unsmoothed) score for the bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] lbfgs improve: "
                  f"τ={tau:.2e}, score={cur_score:.6f}")

        tau *= tau_decay

    return best_a, best_score

    # --------------------------------------------------------------
    if time.time() < deadline:
        # give a generous slice of the remaining budget to this phase
        remaining = deadline - time.time()
        lbfgs_deadline = time.time() + max(30.0, 0.30 * remaining)

        a2_opt, score2 = _run_lbfgs_softmax(
            best_seq, n1, lbfgs_deadline, start_time,
            tau0=None,
            tau_end=1e-13,
            tau_decay=0.5,
            max_iter_phase=300,
        )
        if score2 < best_score:
            best_score = score2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 (L‑BFGS) improve: "
                  f"score={score2:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – projected sub‑gradient descent on the true max‑conv
    # --------------------------------------------------------------
    if time.time() < deadline:
        n2 = best_seq.shape[0]
        grad_deadline = deadline - 2.0            # keep a couple of seconds for annealing
        a3_opt, score3 = _run_subgradient_descent(
            best_seq, n2, grad_deadline, start_time,
            beta=0.97,
            step0=0.13,
        )
        if score3 < best_score:
            best_score = score3
            best_seq = a3_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 3 improve: "
                  f"score={score3:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – extended simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=1_200_000,               # a bit more moves than the reference
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation onto the probability simplex
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list – the harness will evaluate it
    return [float(x) for x in best_seq]
