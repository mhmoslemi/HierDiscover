def subgradient_descent(a0, n, max_iter, step0, beta, deadline):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = score_of(a)
    it = 0
    while it < max_iter and time.time() < deadline:
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(it + 1)
        a = _project_simplex(a - step * momentum)
        it += 1
    return best_a, best_score
```

"""
Propose a low‐value step‑function for the self‑convolution bound.

The search proceeds in four stages:

1. **Beta‑shape initialization** – a parametric family
   a_i ∝ (x_i (1‑x_i))^{α‑1} (Beta(α,α) density) possibly mixed with the
   uniform distribution.  A coarse grid over α∈[0.2,1.5] and mixing weight
   λ quickly yields a good starting shape.

2. **Progressive up‑sampling** – the best vector found for a small size
   (256) is linearly interpolated to the final size (default 8192).  This
   gives a fine discretisation of the continuous shape without extra work.

3. **Projected sub‑gradient descent** (Euclidean) followed by
   **exponentiated‑gradient (mirror) descent** on the true objective
   `F(a)=2·n·max(conv(a))`.  Both updates keep the vector on the probability
   simplex (non‑negative, sum = 1).  The sub‑gradient of the max‑convolution
   is computed analytically and the steps are scheduled with a
   diminishing 1/√t step‑size.

4. **Light simulated‑annealing refinement** – a few random mass‑transfer
   moves are accepted with a Metropolis rule to escape tiny local minima.

All phases respect the global time budget (default 1000 s).  The routine
returns a plain Python list that can be fed to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val <= 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Scoring (the true objective, sum is assumed = 1)
# ----------------------------------------------------------------------


def _score_self(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a)) – the quantity we want to minimise."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Simple up‑sampling (linear interpolation) with renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Initialisation by a mixture of Beta(α,α) and the uniform density
# ----------------------------------------------------------------------


def _beta_initial(n: int):
    """Return a good starting vector for size n via a cheap grid search."""
    best_a = None
    best_score = float("inf")

    uniform = np.full(n, 1.0 / n)

    # mid‑point normalisation points in (0,1)
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    logx = np.log(x)
    log1mx = np.log(1.0 - x)

    alphas = np.linspace(0.2, 1.5, 14)      # shape parameter grid
    mixes = [0.0, 0.4, 0.8, 1.0]            # λ – weight of the Beta component

    for alpha in alphas:
        # log‑density of Beta(α,α) up to the normalising constant
        dens_log = (alpha - 1.0) * (logx + log1mx)
        dens = np.exp(dens_log)
        dens_sum = dens.sum()
        if dens_sum <= 0.0:
            continue
        dens /= dens_sum
        for lam in mixes:
            if lam == 0.0:
                a = uniform
            elif lam == 1.0:
                a = dens
            else:
                a = lam * dens + (1.0 - lam) * uniform
                a = a / a.sum()
            sc = _score_self(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:

    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score_self(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score

            print(f"[{time.time() - start_time:6.2f}s] subgrad it={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = 0.95 * momentum + 0.05 * grad
        step = 0.5 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

def _exponentiated_gradient(a0: np.ndarray,
                           deadline: float,
                           start_time: float) -> tuple[np.ndarray, float]:

    n = a.shape[0]

    best_a = a.copy()

def _annealing(a0: np.ndarray,
               deadline: float,
               start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score_self(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        # pick a source proportional to its current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score_self(a_prop)
        delta = new_score - best_score
        # simple Metropolis accept
        temp = 0.02 * (1.0 - it / 300_000)  # linear cooling
        if delta < 0 or math.exp(-delta / max(temp, 1e-12)) > rng.random():
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal it={it:,} score={new_score:.6f}")

    return best_a, best_score

    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1) initialise at a modest size with a cheap parametric search
    # ------------------------------------------------------------------
    init_n = 256
    best_seq, best_score = _beta_initial(init_n)
    print(f"[{time.time() - start_time:6.2f}s] beta‑init n={init_n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) progressive up‑sampling to the final size
    # ------------------------------------------------------------------
    final_n = 8192                     # chosen as a good compromise of quality vs time
    best_seq = _upsample(best_seq, final_n)

    # ------------------------------------------------------------------
    # 3) heavy optimisation on the final size
    # ------------------------------------------------------------------
    # split the remaining budget: 70 % for sub‑grad, 20 % for EG, 10 % for SA
    remaining = deadline - time.time()
    sub_deadline = time.time() + 0.70 * remaining
    eg_deadline = deadline - 0.30 * remaining   # after sub‑grad we still have ~30 % left
    sa_deadline = deadline                       # use whatever is left at the end

    # sub‑gradient descent
    best_seq, sub_score = _subgradient_descent(best_seq, sub_deadline, start_time)
    if sub_score < best_score:
        best_score = sub_score
        print(f"[{time.time() - start_time:6.2f}s] after subgrad best score={best_score:.6f}")

    # exponentiated gradient
    if time.time() < eg_deadline:
        best_seq, eg_score = _exponentiated_gradient(best_seq, eg_deadline, start_time)
        if eg_score < best_score:
            best_score = eg_score
            print(f"[{time.time() - start_time:6.2f}s] after EG best score={best_score:.6f}")

    # light simulated‑annealing for the last seconds
    if time.time() < sa_deadline:
        best_seq, sa_score = _annealing(best_seq, sa_deadline, start_time)
        if sa_score < best_score:
            best_score = sa_score
            print(f"[{time.time() - start_time:6.2f}s] after SA best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # optional sanity check (uses the provided evaluate_sequence)
    try:
        import __main__ as _main_module
        if hasattr(_main_module, "evaluate_sequence"):
            real_score = _main_module.evaluate_sequence(best_seq.tolist())
            print(f"[{time.time() - start_time:6.2f}s] final evaluate_sequence score = {real_score:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
