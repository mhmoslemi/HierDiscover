"""
Propose a step‑function of non‑negative heights that minimises the
self‑convolution bound used by ``evaluate_sequence``.

The search proceeds in three stages within the 1000 s budget:

1. **Warm‑up (exponential‑gradient)** – a few seconds of multiplicative
   updates on a log‑sum‑exp smooth approximation of the maximal convolution
   entry.  This quickly moves the distribution away from pathological spikes.

2. **High‑precision Adam optimisation** – a projected Adam optimiser that
   works on the exact maximal convolution value.  The gradient of the
   max is obtained analytically (∇maxₖ convₖ = 2 a_{k‑i}) and evaluated
   efficiently by a single FFT‐based convolution per iteration.  This
   stage dominates the run‑time and can perform tens of thousands of
   iterations.

3. **Targeted refinement** – an O(n) “mass‑transfer’’ routine that moves a
   tiny fraction of probability mass from indices that contribute most to
   the current maximal entry to indices that contribute the least.  The
   move is accepted only if the maximal convolution value strictly
   decreases; the step‑size decays from a relatively large fraction of the
   source mass to a very small one.

The algorithm works with any previously discovered construction stored in
the global ``height_sequence_1``; otherwise it starts from a random
distribution.  The final sequence is normalised to sum = 1 and all heights
are non‑negative, satisfying the security checks of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (full) self‑convolution a*a using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0                 # clip tiny negatives from round‑off
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 10000):
    """Multiplicative mirror‑descent on the log‑sum‑exp approximation."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)            # enough for a ⋆ w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update (mirror descent on the simplex)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score

# Phase‑2 : Adam optimiser on the exact maximal convolution entry

def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  lr: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  tol: float = 1e-12):
    """
    Projected Adam optimiser on   f(a)=max_k (a*a)[k].
    Gradient of f is 2·conv_rev where conv_rev = mask * a_rev.
    The mask selects all indices whose convolution value is within
    (1‑tol)·max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # --- gradient of the (non‑smoothed) max ---------------------------------
        mask = np.zeros_like(conv)
        mask[conv >= max_val * (1.0 - tol)] = 1.0

        a_rev = a[::-1]
        L = _next_pow2(len(mask) + n - 1)           # length 3·n‑2
        M = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_rev = np.fft.irfft(M * A_rev, n=L)[: len(mask) + n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # shape (n,)

        # --- Adam update ---------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        beta1_pow = beta1 ** it
        beta2_pow = beta2 ** it
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # project back onto the simplex (ensuring non‑negativity and sum = 1)
        a = _project_simplex(a)

        # occasional re‑normalisation to combat numerical drift
        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_score

# ----------------------------------------------------------------------
def _run_targeted_refine(a0: np.ndarray,
                         n: int,

                         start_time: float,

                         min_eps_frac: float = 0.001):

    O(n) refinement that repeatedly moves a tiny fraction of probability mass
    away from indices that contribute most to the current maximal convolution
    entry towards indices that contribute the least.  Step‑size decays
    linearly from ``init_eps_frac`` to ``min_eps_frac``.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linear decay of epsilon fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # --------------------------------------------------------------
        # source index – probability proportional to its contribution
        # --------------------------------------------------------------
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # --------------------------------------------------------------
        # destination – favour indices that barely affect k_max
        # --------------------------------------------------------------
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # --------------------------------------------------------------
        # propose a move
        # --------------------------------------------------------------
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution (exact up to round‑off)
        delta = np.zeros(conv_len, dtype=float)

        # linear part 2·eps·(a_{k‑dst} ‑ a_{k‑src})
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a

        # quadratic correction (only three entries change)
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # periodic full recomputation to avoid drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the verifier
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine follows three phases:
    * a quick exponential‑gradient warm‑up,
    * a long Adam‑based optimisation on the exact max,
    * a final targeted refinement.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give ourselves a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise – upsample a previously known construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
    except Exception:
        starter = None

    # Using a larger support often lets the optimiser push the max lower.
    final_n = 9600                     # ≈2× the previous 4800‑element construction
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – short exponential‑gradient warm‑up (≈5 % of the budget)
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = deadline - time.time()
    phase1_time = min(30.0, 0.05 * remaining)          # at most 30 s
    phase1_deadline = time.time() + phase1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.5,
        steps_per_alpha=10000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: "
              f"score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – heavy Adam optimisation on the exact max.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # keep the bulk of the remaining time for this phase
        remaining = deadline - time.time()
        phase2_time = max(30.0, 0.70 * remaining)    # at least 30 s
        phase2_deadline = time.time() + phase2_time

        a_adam, score_adam = _run_adam_opt(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            lr=0.7,                 # a moderately aggressive learning rate
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            tol=1e-12,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: "
                  f"score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fast targeted refinement (fine‑grained).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_targeted_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            init_eps_frac=0.10,     # start with a fairly large move‑size
            min_eps_frac=0.001,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise the final vector and return it as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # The caller (the evaluation harness) will feed the list into
    # ``evaluate_sequence``; we only need to return the list.
    return [float(x) for x in best_seq]
