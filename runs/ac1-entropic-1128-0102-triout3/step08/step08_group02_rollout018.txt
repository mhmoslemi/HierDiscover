"""
Hybrid search for a low‑peak step function.

The construction is based on three ideas:

1. **Good analytic starts** – the “Catalan” (central binomial) sequence,
   a truncated exponential, a power‑law, a triangular shape and the uniform
   distribution are all normalised to sum‑1 and used as seeds.

2. **Long exponentiated‑gradient descent on the true max‑convolution**
   (the same sub‑gradient routine as the reference code).  The update is
   multiplicative, always stays non‑negative and is followed by a re‑normalisation
   onto the simplex.  The step‑size decays as 0.5/√t and the loop runs for the
   large majority of the time budget, keeping the best vector seen.

3. **Very cheap bookkeeping** – convolution and the sub‑gradient are evaluated
   with one FFT of length the next power‑of‑two (≈ 2·n).  The algorithm prints
   intermediate improvements and finally returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k].
    The active set contains all indices where conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # mask (0/1)
    # cross‑correlation a ⋆ active
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


# ----------------------------------------------------------------------
# Analytic seeds
# ----------------------------------------------------------------------


def _catalan_seed(n: int) -> np.ndarray:
    """
    a_i ∝ C_i = binom(2i,i)/4^i  (central Catalan numbers).
    The recurrence a_{i+1}=a_i·(2i+1)/(2(i+1)) is numerically stable.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(0, n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    # normalise to sum‑1 (the optimisation routine expects a probability vector)
    a /= a.sum()
    return a


def _exponential_seed(n: int, r: float = 0.99) -> np.ndarray:
    """Geometric decay a_i ∝ r^i, normalised."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    a /= a.sum()
    return a


def _powerlaw_seed(n: int, p: float = 0.5) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def _triangular_seed(n: int) -> np.ndarray:
    """Symmetric “U” shape – large at the edges, small in the centre."""
    idx = np.arange(n)
    a = 1.0 / (np.minimum(idx + 1, n - idx) + 1.0)
    a /= a.sum()
    return a


def _random_smooth_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gaussian‑filtered random noise, forced non‑negative and normalised."""
    noise = rng.normal(size=n)
    smooth = np.convolve(noise, np.ones(15) / 15, mode='same')
    smooth = np.clip(smooth, 0, None)
    if smooth.sum() == 0:
        smooth = np.full(n, 1.0 / n)
    else:
        smooth /= smooth.sum()
    return smooth


# ----------------------------------------------------------------------
# Core optimisation – exponentiated‑gradient on the true max‑convolution
# ----------------------------------------------------------------------


def _exponential_gradient_descent(a0: np.ndarray,
                                 deadline: float,
                                 start_time: float,
                                 fft_len: int) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on the exact max‑convolution.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    cur_conv = _conv_self(a)
    best_score = 2.0 * n * cur_conv.max()
    it = 0
    # step‑size schedule: η_t = η0 / sqrt(t+1)
    eta0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s]   GD improve: iter={it:,}, score={cur_score:.6f}")

        # exact sub‑gradient of the max element(s)
        grad = _max_subgrad(a, conv, rel_tol=1e-9, fft_len=fft_len)

        # multiplicative update (mirror descent with KL‑divergence)
        step = eta0 / math.sqrt(it)
        a = a * np.exp(-step * grad)

        # re‑project onto the probability simplex (here simple renorm works because all entries stay non‑negative)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

# Main driver expected by the evaluation framework
# ----------------------------------------------------------------------

    start_time = time.time()
