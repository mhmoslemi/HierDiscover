def propose_candidate(seed=0, budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - 0.2

    n = 4800
    # start candidates
    candidates = []

    # Use height_sequence_1 if present
    try:
        seq1 = np.asarray(height_sequence_1, dtype=float)
        if seq1.ndim == 1 and seq1.size > 0:
            if seq1.shape[0] != n:
                seq1 = _upsample(seq1, n)
            seq1 = seq1.clip(0)
            seq1 = seq1 / seq1.sum()
            candidates.append(seq1)
    except Exception:
        pass

    # uniform
    candidates.append(np.full(n, 1.0/n))

    # exponential
    candidates.append(_exponential_start(n, r=0.985))

    # gaussian
    candidates.append(_gaussian_start(n))

    # Dirichlet
    rng = np.random.default_rng(seed)
    candidates.append(_dirichlet_start(n, conc=2.0, rng=rng))
    candidates.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # Sweep beta shapes
    alphas = np.concatenate([np.linspace(0.3, 0.9, 13), np.linspace(1.0, 2.0, 11)])
    for a in alphas:
        if time.time() >= deadline:

        seq = _beta_shape(n, a)
        candidates.append(seq)

    # Sweep sine powers
    p_vals = np.linspace(0.2, 3.0, 15)
    for p in p_vals:
        if time.time() >= deadline:
            break

        candidates.append(seq)

        seq = _poly_shape(n, p)
        candidates.append(seq)

    # Evaluate all candidates, keep best
    best_seq = None
    best_score = float('inf')
    for idx, seq in enumerate(candidates):
        if time.time() >= deadline:

        score = evaluate_sequence(seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] candidate {idx+1}/{len(candidates)} score={score:.6f}")

            best_seq = seq.copy()

"""
A fast global‑search + local‑polish routine for the self‑convolution
minimisation problem.

* Global phase – enumerate a few low‑dimensional families that are
  known to give nearly flat self‑convolutions (Beta / arcsine,
  sine‑powers, symmetric power‑law).  For each generated vector we
  evaluate ``evaluate_sequence`` and keep the best one.
* Local phase – starting from the best grid candidate we run a
  projected sub‑gradient descent on the exact non‑smooth objective
  ``f(a)=max_k (a*a)[k]`` (the same routine used in the reference
  implementation).  The method works on the probability simplex,
  uses a simple momentum term and a 1/√t step‑size decay.
* Final phase – a cheap greedy “peak‑reduction’’ walk moves a
  fraction of the coefficient that contributes most to the current
  convolution peak towards the smallest coefficient; moves are kept
  only if they strictly decrease the peak.

The whole search respects the 1000 s budget; the global sweep is very
cheap, the heavy work is done in the polishing sub‑gradient loop.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on a discrete grid, normalised."""
    x = (np.arange(n) + 0.5) / n          # mid‑point of each cell
    eps = 1e-15
    a = np.power(x * (1.0 - x) + eps, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def _sine_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·x)^p  with x∈(0,1); symmetric, normalised."""
    x = (np.arange(n) + 0.5) / n
    a = np.sin(np.pi * x) ** p
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def _poly_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) + (n-i)^(-p)  (symmetric power‑law)."""
    i = np.arange(n) + 1
    a = (1.0 / i ** p) + (1.0 / (n - i + 1) ** p)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def _exponential_start(n: int, r: float = 0.98) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised."""
    sigma = n / 6.0
    center = (n - 1) / 2.0
    x = np.arange(n)
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return g / g.sum()


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


# ----------------------------------------------------------------------
# Local optimiser (projected sub‑gradient descent)
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max_k (a*a)[k].
    Returns the best vector seen and its score (2·n·max).
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # --- sub‑gradient (active set) --------------------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # add contributions (each pair counted twice except diagonal)
            g[i] += a[j]
            g[j] += a[i]

        # momentum step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy peak reduction (final cheap polish)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Repeatedly shifts a modest fraction of the coefficient that
    contributes most to the current peak towards the smallest coefficient.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score

# Main driver
# ----------------------------------------------------------------------
