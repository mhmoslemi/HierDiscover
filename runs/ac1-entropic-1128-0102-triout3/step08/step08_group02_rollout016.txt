def _max_subgrad_fft(a, conv, rel_tol, L):
    cur_max = conv.max()
    thresh = cur_max * (1 - rel_tol)
    active = (conv >= thresh).astype(np.float64)
    # cross-correlation a ⋆ active
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * W, n=L)[:a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad

   amount of mass from the pair of coefficients contributing most to the
   current peak to the smallest coefficient.  This often removes isolated
   spikes that the gradient step cannot eliminate efficiently.

3. **Simulated‑annealing fine‑tuning** – a short stochastic mass‑transfer
   phase that explores the neighbourhood of the best solution found so far.

All phases use only NumPy (and the provided ``evaluate_sequence`` for the
final score) and run until the 1000 s budget is exhausted.  The best sequence
found across a few diverse starts is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    # cross‑correlation a ⋆ active
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """Greedy deterministic moves that try to lower the current peak."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a, _next_pow2(3 * n - 2))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (avoid src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.3
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(3 * n - 2)).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a, _next_pow2(3 * n - 2)).max()
    return a, score


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing mass‑transfer moves."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a, _next_pow2(3 * n - 2))
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop, _next_pow2(3 * n - 2)).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Core optimisation for a single start
# ----------------------------------------------------------------------


def _optimise_one_start(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent + peak‑reduction + annealing
    on a single initial vector a0 (already normalised to sum=1).
    Returns the best sequence seen and its score.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() > 0:
        a /= a.sum()
    else:
        a = np.full_like(a, 1.0 / a.shape[0])

    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # initial score (exact, no call to evaluate_sequence)
    cur_conv = _conv_self(a, fft_len)
    cur_score = 2.0 * n * cur_conv.max()
    best_a = a.copy()
    best_score = cur_score

    it = 0
    step0 = 1.0  # initial step magnitude

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # sub‑gradient of the max (active set)
        grad = _max_subgrad(a, conv, rel_tol=1e-9, fft_len=fft_len)

        # diminishing step size with simple back‑tracking
        step = step0 / math.sqrt(it)
        improved = False
        for _ in range(5):  # up to 5 back‑tracking attempts
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)

            new_conv = _conv_self(a_new, fft_len)
            new_score = 2.0 * n * new_conv.max()
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                improved = True
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] GD improve iter={it:,} score={new_score:.6f}")
                break
            else:
                step *= 0.5  # shrink step and retry

        if not improved:
            # occasional random jitter to escape flat regions
            if it % 2000 == 0:
                rng = np.random.default_rng()
                noise = rng.normal(scale=0.01, size=n)
                a = a + noise
                a = np.maximum(a, 0.0)
                a = _project_simplex(a)
                cur_score = 2.0 * n * _conv_self(a, fft_len).max()

        # perform a cheap peak‑reduction every few thousand iterations
        if it % 4000 == 0 and time.time() < deadline:
            a, pr_score = _run_peak_reduction(a, deadline, start_time, max_no_improve=1500)
            if pr_score < best_score:
                best_score = pr_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] after peak‑reduce score={pr_score:.6f}")

    # final annealing phase (uses the remaining time budget)
    if time.time() < deadline:
        a_final, anneal_score = _run_annealing(best_a, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_a = a_final.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The method runs several starts, each consisting of
    a long projected gradient descent, a deterministic peak‑reduction and a
    short simulated‑annealing refinement, all respecting the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine problem size (n)
    # ------------------------------------------------------------------
    try:
        starter_seq = np.asarray(height_sequence_1, dtype=float)
        if starter_seq.ndim == 1 and starter_seq.size > 0:
            n = starter_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4800
        starter_seq = None

    # ------------------------------------------------------------------
    # Build a modest set of diverse starting points
    # ------------------------------------------------------------------
    starts = []

    if starter_seq is not None:
        starts.append(starter_seq / starter_seq.sum())

    starts.append(np.full(n, 1.0 / n))                     # uniform

    # random smooth vector (Gaussian‑filtered noise)
    rng = np.random.default_rng()
    noise = rng.normal(size=n)
    smooth = np.convolve(noise, np.ones(15) / 15, mode='same')
    smooth = np.clip(smooth, 0, None)
    if smooth.sum() == 0:
        smooth = np.full(n, 1.0 / n)
    else:
        smooth /= smooth.sum()
    starts.append(smooth)

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optimise each start independently (still sharing the global budget)
    # ------------------------------------------------------------------
    for idx_start, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{idx_start + 1}")

        a_opt, sc_opt = _optimise_one_start(a0, deadline, start_time)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST *** score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
