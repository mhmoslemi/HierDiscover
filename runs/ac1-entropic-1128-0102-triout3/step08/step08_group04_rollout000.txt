def _run_simulated_annealing(a0, n, deadline, start_time):
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    start = time.time()
    iter = 0

    # Temperature schedule: start T0 = max_val * 0.1 (or maybe max_val)
    T0 = max_val
    while time.time() < deadline:
        # Update temperature linearly decreasing
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / total_time)
        if T < 1e-12:
            T = 1e-12

        # current maximal index
        k_max = int(np.argmax(conv))

        # source selection
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0 and rng.random() < 0.9:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            # fallback uniform based on a
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        # destination selection: random index far from src+... basically similarly to incremental refine but maybe uniform
        dest = rng.integers(n)
        while dest == src:
            dest = rng.integers(n)

        # epsilon step: fraction of src mass, adapt: fraction = 0.1 * (1 - elapsed/total_time) maybe
        max_frac = 0.2 * (1 - elapsed / total_time)  # start 0.2 then go to near zero
        eps = a[src] * rng.random() * max_frac
        if eps <= 0:
            continue

        # compute new conv using copy
        conv_new = conv.copy()
        # slice modifications
        slice_src = slice(src, src + n)

"""
Hybrid search for a low‑peak self‑convolution step function.

The routine used by the grader evaluates

    score = 2 * n * max(a * a) / (sum(a))**2

(and clamps the sequence to [0,1000] and forces a non‑negative sum).
When the sequence is normalised to sum = 1, minimising the score is
equivalent to minimising the maximum entry of the linear self‑convolution
b = a * a.

Our strategy:

1. **Warm‑up** – a short exponentiated‑gradient optimisation on a
   smooth approximation of the max (identical to the reference
   implementation).  This quickly drives the distribution away from the
   trivial uniform shape.

2. **Simulated‑annealing refinement** – millions of tiny mass moves are
   proposed.  A source index is chosen proportionally to its contribution
   to the current maximal convolution entry, a destination index is chosen
   with low contribution, and a small ε‑fraction of the source mass is
   transferred.  The convolution is updated analytically
   (Δconv[k] = ±2·ε·a[·]) in O(n) time, so many iterations fit into the
   1000 s budget.  Moves that *increase* the peak are occasionally
   accepted according to a linearly‑cooling temperature, allowing the
   search to escape local minima.

3. **Polish** – a few iterations of projected sub‑gradient descent on the
   exact max are performed to clean up the final solution.

All helper functions are top‑level, there are no closures or file I/O,
and the final sequence is normalised to sum = 1 before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a using all almost‑max entries."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient warm‑up (smooth max)
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 4000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Simulated‑annealing refinement with analytical conv update
# ----------------------------------------------------------------------


def _run_simulated_annealing(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             eps_start: float = 0.15,
                             eps_end: float = 0.003,
                             temp_scale: float = 1.0) -> np.ndarray:
    """
    Perform a long SA search.  Each proposal moves a small ε‑mass from a
    “source’’ index (chosen proportional to its contribution to the current
    maximal convolution entry) to a “destination’’ index that contributes
    little.  The convolution is updated analytically (Δconv = ±2·ε·a) which
    makes a single iteration O(n).  Moves that raise the peak are accepted
    with probability exp(-Δ / T) where T cools linearly.
    """
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    start = time.time()
    it = 0
    last_recalc = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        # linear cooling schedule; keep a tiny temperature to avoid division by zero
        T = temp_scale * max_val * max(0.0, 1.0 - elapsed / total_time)
        if T < 1e-12:
            T = 1e-12

        # current peak index
        k_max = int(np.argmax(conv))

        # ----- choose source (high contribution to the peak) -----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0 and rng.random() < 0.9:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ----- choose destination (low contribution to the peak) -----
        # Use inverse weighting of the mirror entry a[k_max‑idx]
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0  # avoid trivial move
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ----- epsilon (fraction of source mass) -----
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ----- propose new convolution (analytical O(n) update) -----
        conv_new = conv.copy()
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)

        # linear part of the update (ignore ε² terms)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        delta = new_max - max_val

        # ----- acceptance test -----
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / T)
            if rng.random() < prob:
                accept = True

        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1

            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improve: iter={it:,}, max={best_max:.8f}")

        # Occasionally recompute the convolution from scratch to avoid drift from ignored ε² terms
        if it - last_recalc >= 5000:
            conv = _conv_self(a)
            max_val = conv.max()
            last_recalc = it
            if max_val < best_max:
                best_max = max_val
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA recompute: iter={it:,}, max={best_max:.8f}")

    return best_a


# ----------------------------------------------------------------------
# Phase 3 – projected sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10 ** 6) -> np.ndarray:
    a = a0.copy()
    best_a = a.copy()
    best_max = _conv_self(a).max()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: iter={it:,}, max={best_max:.8f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a

# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.

    The algorithm consists of a brief exponentiated‑gradient warm‑up,
    a long simulated‑annealing refinement with analytical convolution
    updates, and a short projected sub‑gradient polishing phase.
    """
    # ------------------------------------------------------------------
    # General set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the runner

    # ------------------------------------------------------------------
    # Load a starter sequence if a previous construction exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800   # length of the final vector; can be altered if desired

    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – warm‑up (exponentiated gradient on smoothed max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    stage1_time = min(15.0, 0.08 * remaining)   # allocate a small slice
    stage1_deadline = time.time() + stage1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        steps_per_alpha=4000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – massive SA refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # reserve a few seconds for the final polishing
        stage2_deadline = time.time() + max(20.0, 0.85 * remaining)

        a_sa = _run_simulated_annealing(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            eps_start=0.15,
            eps_end=0.003,
            temp_scale=1.0,
        )
        conv_sa = _conv_self(a_sa)
        score_sa = 2.0 * final_n * conv_sa.max()
        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: score={score_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short projected sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        stage3_deadline = deadline
        a_final = _run_subgradient_descent(
            best_seq,
            final_n,
            stage3_deadline,
            start_time,
            step0=0.25,
            beta=0.94,
            max_iter=2 * 10 ** 6,
        )
        conv_final = _conv_self(a_final)
        score_final = 2.0 * final_n * conv_final.max()
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final polish improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return the sequence as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
