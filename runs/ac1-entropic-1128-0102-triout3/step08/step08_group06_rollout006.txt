import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self_fft(a):
    n = a.shape[0]
    L = _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    conv[conv<0] = 0.0
    return conv

def _project_simplex(v):
    # Project v (not necessarily sum to 1) onto probability simplex.
    # Implementation from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho+1)
        w = np.maximum(v - theta, 0)
    else:
        w = np.full_like(v, 1.0/n)
    return w
```

def _run_adam_subgradient(a0, n, deadline, init_lr=0.25, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = _objective(a)
    start_time = time.time()
    while time.time() < deadline:
        t += 1
        conv = _conv_self_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam subgrad improve: iter={t}, score={cur_score:.6f}")

        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)
        # contributions for each active entry
        for k in active:
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            i = np.arange(lo, hi+1)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]
        # subgradient is g (maybe double counts for i=j). That's fine.
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = init_lr / (np.sqrt(t) + 1e-8)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)
        # optional early break if lr gets tiny and no improvement
    return best_a, best_score

def _run_targeted_refine(a0, n, deadline, start_time, init_eps_frac=0.05, min_eps_frac=0.001):
    a = a0.copy()
    conv = _conv_self_fft(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    L = 2*n - 1
    it = 0
    while time.time() < deadline:
        it += 1
        # decay eps_frac
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed/total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())
        # compute contributions for each index i to the max entry
        i_arr = np.arange(n)
        j_arr = k_max - i_arr
        valid = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i_arr[valid]] * a[j_arr[valid]]
        # source = index with max contribution
        src = np.argmax(contrib)
        # find destination candidate with zero contribution (i.e., j out of bounds)
        zero_mask = ~valid  # indices where partner invalid -> zero contribution
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            # otherwise pick index with smallest partner a[j]
            partner = np.zeros(n)
            partner[valid] = a[j_arr[valid]]
            candidates = np.where(partner == partner.min())[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue
        max_eps = a[src]
        if max_eps <= 0:
            continue
        eps = max_eps * eps_frac * rng.random()
        if eps <= 0:
            continue

        # compute diff
        diff = np.zeros(L)
        diff[dst:dst+n] += a
        diff[src:src+n] -= a

        # compute new max efficiently
        # we can compute updated conv
        delta = 2.0 * eps * diff
        # quadratic corrections
        delta[2*src] += eps*eps
        delta[2*dst] += eps*eps
        delta[src+dst] -= 2.0 * eps*eps
        conv_new = conv + delta
        new_max = conv_new.max()
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: iter={it:,}, max={best_max:.8f}")

        # recompute conv occasionally
        if it % 20000 == 0:
            conv = _conv_self_fft(a)
            max_val = conv.max()
    return best_a

def _upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

A three‑phase optimizer for the self‑convolution bound.

Phase 1 – Adam‑accelerated mirror descent on a log‑sum‑exp (smooth)
           approximation of the maximal self‑convolution.

Phase 2 – Adam‑accelerated projected sub‑gradient descent on the exact
           (non‑smoothed) maximum.

Phase 3 – Deterministic targeted refinement that repeatedly moves
           probability mass from the index with the largest contribution
           to the current maximal convolution entry to an index that does
           not contribute (or contributes the least).  For each pair the
           optimal step size is obtained by a cheap golden‑section search.

The routine runs as many iterations as the 1000 s budget permits,
keeps the best sequence seen, and finally returns a normalised
(non‑negative, sum‑to‑one) list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    Result is always non‑negative (tiny negatives are clipped).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _objective(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a normalised probability vector a.
    (The evaluate_sequence routine returns exactly this after normalisation.)
    """
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_adam_smooth(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     alpha_schedule,
                     init_lr: float = 0.6,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = _objective(a)
    start = time.time()

    while time.time() < deadline:
        t += 1
        conv = _conv_self_fft(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start:.2f}s] Adam‑smooth improve: iter={t:,} score={cur_score:.6f}")

        # smoothing parameter (increase with time)
        alpha = alpha_schedule[min(t // 1000, len(alpha_schedule) - 1)]

        # smooth‑max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)        # safe scaling
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # gradient of the smooth max (mirror‑descent on the simplex)
        fft_len = _next_pow2(3 * n - 2)
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w[::-1], n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w) - 1]
        grad = 2.0 * corr[len(w) - 1 - np.arange(n)]

        # Adam update (descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = init_lr / (math.sqrt(t) + 1e-8)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # stay on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Adam on the exact max (sub‑gradient)
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          init_lr: float = 0.30,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = _objective(a)
    start = time.time()

    while time.time() < deadline:
        t += 1
        conv = _conv_self_fft(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start:.2f}s] Adam‑subgrad improve: iter={t:,} score={cur_score:.6f}")

        # exact sub‑gradient of the (non‑smoothed) max
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)

        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = init_lr / (math.sqrt(t) + 1e-8)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted refinement with optimal step size search
# ----------------------------------------------------------------------
def _search_best_eps(diff: np.ndarray,
                     conv: np.ndarray,
                     src: int,
                     dst: int,
                     max_eps: float,
                     max_iter: int = 12) -> tuple[float, float]:
    """
    Golden‑section search for the step size ϵ∈[0,max_eps] that minimises
    max(conv + Δ(ϵ)), where Δ is the exact change caused by moving ϵ
    mass from src to dst.
    Returns (ϵ_opt, new_max).
    """
    phi = (np.sqrt(5) - 1) / 2  # ≈0.618

    # helper – evaluate new maximum for a given ϵ
    def new_max(eps: float) -> float:
        delta = 2.0 * eps * diff
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps
        return (conv + delta).max()

    a = max_eps * phi
    b = max_eps - a
    fa = new_max(a)
    fb = new_max(b)

    for _ in range(max_iter):
        if fa > fb:
            max_eps = b
            b = a
            fb = fa
            a = max_eps * phi
            fa = new_max(a)
        else:
            max_eps = a
            a = b
            fa = fb
            b = max_eps - max_eps * phi
            fb = new_max(b)

    if fa < fb:
        return a, fa
    else:
        return b, fb


def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         init_eps_frac: float = 0.08,
                         min_eps_frac: float = 0.001) -> np.ndarray:
    """
    Deterministic refinement:
    * source = index with the largest contribution to the current maximal
      convolution entry,
    * destination = any index that does **not** participate in that entry
      (or the one with the smallest partner weight),
    * optimal step size is found by a cheap golden‑section search.
    """
    a = a0.copy()
    conv = _conv_self_fft(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    L = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # contributions of each index to the current maximal entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i[valid]] * a[j[valid]]

        src = int(np.argmax(contrib))

        # destination – try to pick an index that does NOT take part
        zero_mask = ~valid
        if np.any(zero_mask):
            dst = int(rng.choice(np.where(zero_mask)[0]))
        else:
            # partner weight = a[j] for indices that do participate
            partner = np.zeros(n)
            partner[valid] = a[j[valid]]
            dst = int(rng.choice(np.where(partner == partner.min())[0]))

        if src == dst:
            continue

        max_eps = float(a[src])
        if max_eps <= 0.0:
            continue

        # linear part of the change (used repeatedly)
        diff = np.zeros(L)
        diff[dst:dst + n] += a
        diff[src:src + n] -= a

        # optimal epsilon for this pair
        eps_opt, new_max = _search_best_eps(diff, conv, src, dst, max_eps)

        # apply a fraction of the optimal step to stay robust
        eps = eps_opt * eps_frac
        if eps <= 0.0:
            continue

        # recompute the exact delta for the chosen epsilon
        delta = np.zeros(L)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_new = conv + delta
        max_val_new = conv_new.max()

        if max_val_new < max_val - 1e-14:
            # accept move
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            max_val = max_val_new
            if max_val_new < best_max - 1e-14:
                best_max = max_val_new
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: iter={it:,} max={best_max:.8f}")

        # occasional full recomputation to avoid numerical drift
        if it % 40000 == 0:
            conv = _conv_self_fft(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the supplied time budget.

    The algorithm consists of three phases (Adam on a smoothed max,
    Adam on the exact max, and a deterministic targeted refinement with
    optimal step‑size search).  It starts from the globally available
    ``height_sequence_1`` when present, otherwise from a fresh random start.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose the length of the vector – a few thousand gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Initialise – warm start from the previously discovered construction if
    # available, otherwise a random probability vector.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = _upsample(starter, n)
        else:
            raise Exception
    except Exception:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # Phase 1 – smooth Adam (≈5 % of the total budget)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, time.time() + 30.0)
    a1, s1 = _run_adam_smooth(
        a0,
        n,
        phase1_deadline,
        alpha_schedule=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        init_lr=0.6,
    )
    if s1 < best_score:
        best_score, best_seq = s1, a1.copy()
        print(f"[{time.time() - start_time:.2f}s] Phase 1 new best score {s1:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact‑max Adam (major part of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + min(remaining * 0.6, 400.0)
        a2, s2 = _run_adam_subgradient(best_seq, n, phase2_deadline, init_lr=0.30)
        if s2 < best_score:
            best_score, best_seq = s2, a2.copy()
            print(f"[{time.time() - start_time:.2f}s] Phase 2 new best score {s2:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic targeted refinement with optimal step size
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a3 = _run_targeted_refine(best_seq, n, deadline, start_time,
                                  init_eps_frac=0.08, min_eps_frac=0.001)
        s3 = _objective(a3)
        if s3 < best_score:
            best_score, best_seq = s3, a3.copy()
            print(f"[{time.time() - start_time:.2f}s] Phase 3 new best score {s3:.6f}")

    # ------------------------------------------------------------------
    # Normalise to a proper probability vector and return.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"Final reported objective (approx): {best_score:.6f}")
    return [float(x) for x in best_seq]
