def solve_lp_max_sum(f, M):
    n = len(f)
    # using scipy.sparse
    rows = []
    cols = []
    data = []
    for j in range(n):
        # column j appears in rows k where k from j to j+n-1
        # i.e., rows = j..j+n-1
        rows.extend(range(j, j+n))
        cols.extend([j]*n)
        data.extend([f[j]]*n)
    # shape (2n-1, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints: A @ b <= M
    # objective: maximize sum(b)
    c = np.full(n, -1) # because linprog does minimize. We'll minimize -sum(b)
    # bounds: b_j >=0
    bounds = [(0, None)] * n
    # Use linprog with high precision.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback: return original f
        return f.copy()
    return res.x

Now writing final code snippet.assistantfinal**Solution Overview**

The objective is  

\[
F(a)=\frac{2n\max_k (a\ast a)[k]}{\bigl(\sum_i a_i\bigr)^2},
\]

for a non‑negative vector `a`.  
`a` can be scaled arbitrarily – the ratio is invariant under a global scaling –
therefore we can restrict the search to the probability simplex
`\sum a_i = 1`.  
The hardest part is the “max‑convolution’’ term.  
We use three complementary optimisation phases:

1. **Good initialisation** – a *Beta(½,½)* (arcsine) shape  
   `a_i ∝ 1/√((i+½)(n‑½‑i))`.  This shape already reduces the peak of the
   convolution compared with the uniform vector.

2. **Smooth max optimisation (Adam)** – we minimise the smooth surrogate  

   \[
   f_\alpha(a)=\frac{1}{\alpha}\log\Bigl(\sum_k e^{\alpha\; (a\ast a)[k]}\Bigr)
   \]

   with a standard Adam optimiser.  The smoothing parameter `α` is increased
   from a small value to a large value, turning the surrogate gradually into
   the true max.

3. **Exact sub‑gradient descent** – after the smooth phase we switch to the
   exact non‑smooth objective.  The sub‑gradient is the vector that appears in
   the convolution at the maximal index.  We use momentum and a step size
   `η_t = η₀/√t` and project the iterate back onto the simplex after every
   step.

4. **Light simulated‑annealing polish** – at the very end we perform a cheap
   annealing that moves a tiny amount of mass between two coordinates, accepting
   downhill moves always and uphill moves with Metropolis probability.

All phases respect a global time budget (≈ 1000 s).  The code is deliberately
written with no closures, no file‑IO and only the allowed libraries.  Progress
is printed with timestamps so that you can see how the bound evolves.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""Propose a step‑function sequence minimizing evaluate_sequence.

The algorithm consists of:
  1. an arcsine / beta(½,½) start vector (good theoretical
     initialisation);
  2. a long Adam optimisation on a smooth log‑sum‑exp surrogate of the
     maximal convolution;
  3. a projected sub‑gradient descent on the exact objective;
  4. a final tiny simulated‑annealing polish.

The whole routine respects a 1000 s wall‑clock budget, returns the best
sequence found (as a plain Python list of floats) and prints progress
information.
"""

import time
import math
import numpy as np
from scipy.fft import rfft, irfft
from numpy.random import default_rng

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam minimisation of a smooth surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alpha_start: float = 0.5,
    alpha_end: float = 500.0,
    max_iters: int = 15000,
    lr0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate

        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k]).

    α grows linearly from *alpha_start* to *alpha_end*.
    Returns the best iterate (lowest true objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()          # exact objective
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-10
    beta1_pow = 1.0
    beta2_pow = 1.0

    # length for the FFTs used in the gradient
    L_grad = _next_pow2(3 * n - 2)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current exact score (for bookkeeping)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # increasing smoothing
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        # reversed weights needed for the gradient of the smooth max
        w_rev = w[::-1]

        # gradient of f_α
        Af = rfft(a, n=L_grad)
        Wrev = rfft(w_rev, n=L_grad)
        corr = irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exact max‑convolution sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    step0: float = 0.4,
    beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    f(a)=2·n·max(conv(a,a)).
    Uses momentum (β) and a 1/√t step‑size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # sub‑gradient: only the shift that attains the maximum matters
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Very light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealing.  Temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        # transfer a small fraction of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

        a0,
        n,
        adam_deadline,
        start_time,
        alpha_start=0.5,
        alpha_end=500.0,
        max_iters=15000,
        lr0=0.6,
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] Adam improved to {score_adam:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Long sub‑gradient descent on the exact objective
    # --------------------------------------------------------------
    sub_deadline = time.time() + (deadline - time.time()) * 0.60
    a_sub, score_sub = _subgradient_descent(
        a_adam,
        n,
        sub_deadline,
        start_time,
        best_score,
        step0=0.4,
        beta=0.9,
    )
    if score_sub < best_score:
        best_score = score_sub
        best_seq = a_sub.copy()
        print(f"[{time.time() - start_time:.2f}s] Sub‑grad improved to {score_sub:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final annealing polish (uses whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=300_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved to {score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time (protect against possible tiny negatives)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return as a plain Python list – the external harness will evaluate it.
    return [float(x) for x in best_seq]
```
