def grad_softmax_surrogate(z, n, tau):
    a = softmax(z)
    conv = _conv_self(a)
    z_exp = np.exp(conv / tau - np.max(conv / tau))
    p = z_exp / z_exp.sum()
    # grad w.r.t a: 4*n*inner as in earlier code
    a_rev = a[::-1]
    inner = _conv_fft(a_rev, p)[n-1:n-1+n]
    grad_a = 4.0 * n * inner
    # grad w.r.t z: a * (grad_a - np.dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    return grad_z, a

    v = np.zeros_like(z)

        v = beta2 * v + (1 - beta2) * (grad_z**2)

"""
Propose a low‑self‑convolution step‑function.

The routine follows a very concise multi‑phase strategy:

1. **Scale selection** – choose a problem size that fits the remaining time.
2. **Simple parametric families** – scan a few one‑parameter families
   (exponential, power‑law and a two‑sided exponential) on the simplex.
   The best point provides a good start.
3. **Adam on a soft‑max surrogate** – optimise the smooth objective
   `O_τ(a)=2·n·τ·log∑exp(conv(a)/τ)` in the log‑softmax (entropy) coordinates.
   A fast τ‑schedule drives the approximation to the true max‑convolution.
4. **Projected sub‑gradient descent** – a few thousand Euclidean sub‑gradient
   steps with momentum refine the solution for the exact objective.
5. **Exponentiated‑gradient (mirror descent)** – multiplicative updates on the
   simplex give a complementary improvement.
6. **Light simulated‑annealing** – mass‑transfer moves escape tiny local minima.

All phases respect the global 1000 s budget and keep the vector on the
probability simplex, which makes the evaluation scale‑invariant.
The final sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Objective utilities (true objective: 2·n·max(conv)  – sum is 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (assumes sum(a)==1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Simple parametric families (used for a fast initial scan)
# ----------------------------------------------------------------------


def _exp_shape(alpha: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-α·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    a = np.exp(-alpha * i)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a.fill(1.0 / n)
    return a


def _powerlaw_shape(beta: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^{-β}, normalised."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-beta)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a.fill(1.0 / n)
    return a


def _double_exp_shape(alpha: float, beta: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-α·i) + exp(-β·(n‑1‑i)), normalised."""
    i = np.arange(n, dtype=float)
    left = np.exp(-alpha * i)
    right = np.exp(-beta * (n - 1 - i))
    a = left + right
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a.fill(1.0 / n)
    return a


def _search_initial_shape(n: int, deadline: float, start_time: float):
    """Very cheap grid search over a few families; returns best (a,score)."""
    best_a = None
    best_score = float("inf")

    # exponential decay
    alphas = np.geomspace(1e-6, 5e-2, num=30)
    for a in alphas:
        if time.time() >= deadline:
            break
        seq = _exp_shape(a, n)
        sc = _objective(seq)
        if sc < best_score:
            best_score, best_a = sc, seq

    # power‑law
    betas = np.linspace(0.0, 2.0, num=30)
    for b in betas:
        if time.time() >= deadline:
            break
        seq = _powerlaw_shape(b, n)
        sc = _objective(seq)
        if sc < best_score:
            best_score, best_a = sc, seq

    # two‑sided exponential (coarse grid)
    alphas2 = np.geomspace(1e-6, 5e-2, num=10)
    betas2 = np.geomspace(1e-6, 5e-2, num=10)
    for a in alphas2:
        for b in betas2:
            if time.time() >= deadline:
                break
            seq = _double_exp_shape(a, b, n)
            sc = _objective(seq)
            if sc < best_score:
                best_score, best_a = sc, seq

    # a few random starts (very cheap)
    rng = np.random.default_rng()
    for _ in range(12):
        if time.time() >= deadline:
            break
        seq = rng.random(n)
        seq /= seq.sum()
        sc = _objective(seq)
        if sc < best_score:
            best_score, best_a = sc, seq

    return best_a, best_score


# ----------------------------------------------------------------------
# Soft‑max surrogate + Adam (optimises in log‑softmax space)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → simplex)."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def _grad_softmax_surrogate(z: np.ndarray,
                            n: int,
                            tau: float):
    """
    Gradient of the smooth surrogate O_τ(a) with respect to the log‑softmax params.
    Returns (grad_z, a) where a = softmax(z).
    """
    a = _softmax(z)
    conv = _conv_self(a)

    # log‑sum‑exp weights
    v = conv / tau
    v_max = np.max(v)
    exp_v = np.exp(v - v_max)
    p = exp_v / exp_v.sum()                     # shape 2·n‑1, sums to 1

    # ∂Oτ/∂a = 4·n·(a * p convolved with reversed a)
    a_rev = a[::-1]
    conv_rev = _conv_fft(a_rev, p)              # length 3·n‑2
    inner = conv_rev[n - 1:n - 1 + n]           # required slice
    grad_a = 4.0 * n * inner

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    return grad_z, a


def _run_adam_softmax(z0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-12,
                      tau_decay: float = 0.995,
                      step_size: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 200_000):
    """
    Adam optimisation of the soft‑max surrogate.
    Returns the best log‑parameters (z) and the associated true score.
    """
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_z = z.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1
        grad_z, a = _grad_softmax_surrogate(z, n, tau)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        z = z - step * m_hat

        # true objective for bookkeeping
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_z = z.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f} τ={tau:.2e}")

        tau = max(tau_end, tau * tau_decay)

    return best_z, best_score


# ----------------------------------------------------------------------
# Euclidean projected sub‑gradient descent (true objective)
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.95,
                             step0: float = 0.4,
                             max_iter: int = 5_000_000):
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad iter={it:,} "
                  f"score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated gradient (mirror descent on the simplex)
# ----------------------------------------------------------------------


def _run_exponentiated_gradient(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               step0: float = 0.20,
                               max_iter: int = 3_000_000):
    """Multiplicative mirror‑descent with entropy regulariser."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG iter={it:,} "
                  f"score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)               # non‑negative
        eta = step0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)                     # multiplicative update
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very light simulated annealing (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 400_000):
    """Mass‑transfer SA; keeps the best sequence seen."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal iter={it:,} "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up.fill(1.0 / new_n)
    return up

    if time.time() < deadline:
        remaining = deadline - time.time()
        eg_deadline = time.time() + remaining * 0.20
        a_eg, _ = _run_exponentiated_gradient(
            a0=best_a,
            n=n,
            deadline=eg_deadline,
            start_time=start_time,
            step0=0.20,
        )
        sc = _objective(a_eg)
        if sc < best_score:
            best_a, best_score = a_eg, sc
            print(f"[{time.time() - start_time:6.2f}s] EG improve score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Light simulated annealing for the final few seconds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, _ = _run_annealing(
            a0=best_a,
            n=n,
            deadline=deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_a,
            start_temp=5e-2,
            end_temp=1e-7,
            steps=400_000,
        )
        sc = _objective(a_sa)
        if sc < best_score:
            best_a, best_score = a_sa, sc
            print(f"[{time.time() - start_time:6.2f}s] anneal final improve score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return a plain Python list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    tot = best_a.sum()
    if tot > 0.0:
        best_a /= tot
    else:
        best_a.fill(1.0 / best_a.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
