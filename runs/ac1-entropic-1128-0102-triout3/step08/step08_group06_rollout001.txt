"""
Optimise the self‑convolution L∞ bound.

We search a one‑parameter family of symmetric “U‑shaped’’ step functions

    a_i ∝ (i+½)^{p}·(n‑½‑i)^{p}      (p<0)

which includes the uniform distribution (p=0) and increasingly edge‑heavy
distributions for more negative p.  The family is cheap to evaluate:
for each p we form the distribution, normalise it to sum = 1,
compute the self‑convolution by FFT and keep the best p.
The coarse grid is refined by a small local search.

Starting from the best shape we run a projected sub‑gradient descent on the
exact max‑convolution (the same descent used in the reference solution) and
finish with the very fast O(n) targeted refinement that moves a tiny amount
of probability mass from indices that contribute most to the current maximal
convolution entry to indices that contribute the least.

The whole pipeline respects the 1000 s budget and returns the best
probability vector found (as a plain Python list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negatives caused by rounding.
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Parameterised family of distributions
# ----------------------------------------------------------------------
def _gen_distribution(p: float, n: int) -> np.ndarray:
    """
    Generate a symmetric distribution a_i ∝ (i+½)^{p}·(n‑½‑i)^{p},
    normalised to sum = 1.  For p=0 the distribution is uniform.
    """
    i = np.arange(n, dtype=np.float64)
    x = i + 0.5
    y = n - 0.5 - i
    # Use logs to avoid overflow/underflow.
    logw = p * (np.log(x) + np.log(y))
    w = np.exp(logw)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full(n, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    """
    Projected sub‑gradient descent on the true maximal convolution entry.
    Uses the same algorithm as the reference implementation.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max (all active indices)
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast targeted refinement (O(n) moves)
# ----------------------------------------------------------------------
def _run_targeted_refine_optimized(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.05,
                                   min_eps_frac: float = 0.001):
    """
    Very fast O(n) refinement.
    Moves a small fraction of probability mass from indices that contribute
    most to the present maximal convolution entry towards indices that
    contribute the least.  Accepts only strict improvements.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # exponential decay of step size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # --------------------------------------------------------------
        # choose source index – probability proportional to its contribution
        # --------------------------------------------------------------
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # --------------------------------------------------------------
        # choose destination – preferably indices that *do not* affect k_max
        # --------------------------------------------------------------
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # --------------------------------------------------------------
        # propose a move
        # --------------------------------------------------------------
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution (exact up to round‑off)
        delta = np.zeros(conv_len, dtype=float)

        # linear part 2*eps*(a_{k-dst} - a_{k-src})
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a

        # quadratic eps² correction (only three entries are affected)
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to avoid drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a

    # ------------------------------------------------------------------

            best_seq = a.copy()

        step = 0.05
        for _ in range(12):
            candidates = [best_p - step, best_p, best_p + step]
            improved = False
            for p in candidates:
                if p >= 0:    # we never need non‑negative p (that would be bell‑shaped)
                    continue
                a = _gen_distribution(p, n)
                conv = _conv_self(a)
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    best_p = p
                    improved = True
                    print(f"[{time.time() - start_time:.2f}s] refine p={p:.4f} score={cur_score:.6f}")
            if not improved:
                step /= 2.0
            if time.time() > deadline - 100:
                break

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        sub_deadline = deadline - 20.0   # keep a few seconds for the final stage
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            n,
            sub_deadline,
            start_time,
            step0=0.30,
            beta=0.90,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad final improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – fast targeted refinement (O(n) moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_targeted_refine_optimized(
            best_seq,
            n,
            deadline,
            start_time,
            init_eps_frac=0.05,
            min_eps_frac=0.001,
        )
        conv = _conv_self(a_refined)
        score_ref = 2.0 * n * conv.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise the final sequence and return it as a list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] candidate final approximated score = {best_score:.6f}")
    return [float(x) for x in best_seq]
