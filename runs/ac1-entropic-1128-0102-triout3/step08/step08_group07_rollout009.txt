"""
Propose a low‑score step‑function.

The algorithm follows a “coarse‑to‑fine” scheme:

1.  Multiple cheap random/structured restarts with a very small size
    (n≈512) using Adam on a soft‑max surrogate of the max‑convolution.
2.  Upsample the best candidate to a medium size (≈8192) and continue
    Adam with a slower cooling schedule.
3.  If time permits, upscale once more to n≈16384 and run Adam again.
4.  Refine the current best vector with a long projected sub‑gradient
    descent on the true objective.
5.  Apply an extended simulated‑annealing stage with larger, adaptive
    mass‑transfer moves.
6.  Finally try a few simple smoothing operators and random‑phase
    perturbations (FFT‑based) – both cheap heuristics that often lower
    the max‑convolution a little further.

All vectors are kept on the probability simplex (∑aᵢ=1,
aᵢ≥0).  The routine respects the global time budget (default 1000 s)
and prints progress information.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two  ≥  x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, sum fixed)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9994,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    with Adam and Euclidean projection onto the simplex.
    Returns the best vector seen (as a np.ndarray) and its *true* score
    2·n·max(conv(a)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true (unsmoothed) score – cheap to compute
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # ---- soft‑max surrogate ----
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ (see derivation in the reference code)
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update + projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    Acceptance uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 35 %) of the source mass
        eps = a[src] * rng.random() * 0.35
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisers (Chebyshev / Kaiser / Gaussian / Beta)
# ----------------------------------------------------------------------


def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random, structured non‑negative vector of length n.
    With probability 0.3 → Chebyshev,
                 0.3 → Kaiser,
                 0.2 → Gaussian,
                 0.2 → Beta‑shaped,
    otherwise a plain Dirichlet draw.
    """
    # optional import of scipy.signal for Chebyshev/Kaiser
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.3:
        # Chebyshev window
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
    elif have_signal and r < 0.6:
        # Kaiser window
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
    elif r < 0.8:
        # Gaussian‑like window (σ chosen so that the support fits)
        sigma = rng.uniform(0.1, 0.4)
        x = np.linspace(-0.5, 0.5, n)
        w = np.exp(-0.5 * (x / sigma) ** 2)
    elif r < 1.0:
        # Beta distribution shape
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        x = np.linspace(0, 1, n)
        w = (x ** (a - 1)) * ((1 - x) ** (b - 1))
        w = np.nan_to_num(w, nan=0.0)
    else:
        # plain Dirichlet
        w = rng.random(n)

    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Helper: cheap objective (no clipping) – assumes sum(a)=1
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Exact objective used in `evaluate_sequence` for a already‑normalised vector."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Helper: random phase perturbation (FFT‑based)
# ----------------------------------------------------------------------


def _random_phase_perturb(a: np.ndarray,
                         rng: np.random.Generator,
                         phase_scale: float = 0.6) -> np.ndarray:
    """
    Keep the magnitude of the FFT of ``a`` but randomise its phases.
    The result is projected back onto the simplex.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    A = rfft(a, n=N)
    mag = np.abs(A)
    phase = np.angle(A)

    # zero‑frequency (DC) must stay unchanged to keep the sum = 1
    delta = rng.normal(scale=phase_scale, size=phase.shape)
    delta[0] = 0.0
    new_A = mag * np.exp(1j * (phase + delta))

    a_new = irfft(new_A, n=N)[:n]
    a_new = np.maximum(a_new, 0.0)
    total = a_new.sum()
    if total > 0:
        a_new /= total
    else:
        a_new = np.full_like(a_new, 1.0 / n)
    return a_new

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The strategy is a multi‑phase optimisation with increasing resolution.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous solution exists, use it as a starter (normalised)
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Phase 0 – many fast Adam runs on a tiny problem (n≈512)
    # ------------------------------------------------------------------
    n0 = 512
    remaining = deadline - time.time()
    phase0_time = max(20.0, 0.12 * remaining)   # ~12 % of total budget
    phase0_deadline = time.time() + phase0_time
    attempts = 0
    while time.time() < phase0_deadline:
        attempts += 1
        if starter is not None and attempts == 1:
            a0 = _upsample(starter, n0)
        else:
            a0 = _window_initialiser(n0, rng)

        tau0 = max(_conv_self(a0).max() * 6.0, 1e-8)
        a_opt, _ = _run_adam_softmax(
            a0, n0, phase0_deadline, start_time,
            tau0=tau0,
            tau_end=1e-6,
            tau_decay=0.9996,
            step_size=0.22,
            max_iter=100_000,
        )
        sc = _score(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase 0 improve: "
                  f"attempt={attempts}, score={sc:.6f}")

    if best_seq is None:
        # fallback – uniform vector
        best_seq = np.full(n0, 1.0 / n0)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase 1 – upscale to medium size (≈8192) and run Adam longer
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    a1 = _upsample(best_seq, n1)
    tau0_1 = max(_conv_self(a1).max() * 8.0, 1e-10)

    remaining = deadline - time.time()
    phase1_time = max(30.0, 0.20 * remaining)   # ~20 % of remaining time
    slice_deadline = time.time() + phase1_time

    a1_opt, _ = _run_adam_softmax(
        a1, n1, slice_deadline, start_time,
        tau0=tau0_1,
        tau_end=1e-12,
        tau_decay=0.9995,
        step_size=0.18,
        max_iter=5_000_000,
    )
    sc1 = _score(a1_opt)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] phase 1 improve: score={sc1:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – optional further up‑sampling (≈16384) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        n2 = 16384
        a2 = _upsample(best_seq, n2)
        tau0_2 = max(_conv_self(a2).max() * 8.0, 1e-10)

        slice_deadline = time.time() + (deadline - time.time()) * 0.25
        a2_opt, _ = _run_adam_softmax(
            a2, n2, slice_deadline, start_time,
            tau0=tau0_2,
            tau_end=1e-14,
            tau_decay=0.9994,
            step_size=0.15,
            max_iter=10_000_000,
        )
        sc2 = _score(a2_opt)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase 2 improve: score={sc2:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        n3 = best_seq.shape[0]
        grad_deadline = deadline - 2.0
        a3_opt, sc3 = _run_subgradient_descent(
            best_seq, n3, grad_deadline, start_time,
            beta=0.97,
            step0=0.13,
        )
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase 3 improve: score={sc3:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – extended simulated‑annealing refinement (multiple cycles)
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:
        a_ref, sc_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-8,
            steps=250_000,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal cycle improve: score={sc_ref:.6f}")

        # ----- occasional smoothing -------------------------------------------------
        for width in (3, 5, 9, 15):
            kernel = np.ones(width) / width
            smooth = np.convolve(best_seq, kernel, mode='same')
            smooth = np.maximum(smooth, 0.0)
            if smooth.sum() > 0:
                smooth /= smooth.sum()
            sc_s = _score(smooth)
            if sc_s < best_score:
                best_score = sc_s
                best_seq = smooth.copy()
                print(f"[{time.time() - start_time:6.2f}s] smoothing improve (w={width}): score={sc_s:.6f}")

        # ----- random phase perturbations -----------------------------------------
        for _ in range(2):
            pert = _random_phase_perturb(best_seq, rng, phase_scale=0.5)
            sc_p = _score(pert)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = pert.copy()
                print(f"[{time.time() - start_time:6.2f}s] phase‑perturb improve: score={sc_p:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (guard against numerical drift)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return plain Python list – the harness will call evaluate_sequence
    return [float(x) for x in best_seq]
