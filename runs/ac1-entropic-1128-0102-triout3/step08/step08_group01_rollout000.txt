import time, math, numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self-convolution a*a via FFT (length 2·len(a)-1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

    """Self‑convolution a*a via FFT (returns length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

            s = pert.sum()
        pert /= s

        pert_seq, pert_score = _run_subgradient_nesterov(
            pert,
            final_n,
            restart_deadline,
            start_time,
            best_score,
        )
        if time.time() < deadline:
            pert_seq, pert_score = _run_targeted_annealing(
                pert_seq,
                final_n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-9,
                steps=100_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
