"""
Multistart optimisation for the evaluate_sequence objective.

The optimiser works on a relatively large discretisation (n=8192) and
combines three complementary phases:

1. **Adam on a smooth log‑sum‑exp surrogate** – the smoothing parameter
   α grows from a small to a large value, giving a differentiable
   approximation of the max‑convolution.  The Adam update respects the
   simplex by Euclidean projection after every step.

2. **Projected sub‑gradient descent on the exact objective** – once a
   good neighbourhood is found, a pure sub‑gradient method (with
   momentum and a 1/√t step‑size) is applied to the non‑smooth
   objective 2·n·max(conv(a,a)).

3. **Light simulated‑annealing polish** – a cheap mass‑transfer move
   (shift a random fraction of the mass from one coordinate to another)
   is performed with an exponentially decaying temperature.  The move
   is accepted according to the Metropolis rule, allowing occasional
   uphill steps.

The algorithm performs many short trials (≈30 s each) with diverse
initialisations (random Dirichlet, triangular, sine, cosine, beta‑shapes)
to explore the landscape, keeps the best candidate and finishes with a
longer sub‑gradient refinement followed by a final annealing polish.
All operations are carried out with FFT‑based convolutions for speed.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam on the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 alpha_start: float = 0.5,
                 alpha_end: float = 500.0,
                 max_iters: int = 8000,
                 lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on  f_α(a)= (1/α)·log∑exp(α·conv(a,a)).
    α rises linearly from ``alpha_start`` to ``alpha_end``.
    Returns the best iterate (lowest true objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    # length for gradient FFTs (covers all needed linear‑convolution lengths)
    L_grad = _next_pow2(3 * n - 2)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # true objective (for bookkeeping)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # update smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient of the smoothed objective (via convolution theorem)
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact max – projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # index of the (currently) maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-6,
               steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Very light mass‑transfer annealing.
    Temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source index proportionally to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 50 % of the source mass (more aggressive)
        eps = a[src] * rng.random() * 0.50
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the allotted time.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # increased problem size – 8192 is still FFT‑friendly
    n = 8192

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try to use a global starter if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if starter.ndim == 1 and starter.size > 0:
            if starter.shape[0] != n:
                # pad with zeros or truncate to match n
                if starter.shape[0] < n:
                    tmp = np.zeros(n)
                    tmp[: starter.shape[0]] = starter
                    starter = tmp
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter /= starter.sum()
                cur = 2.0 * n * _conv_self(starter).max()
                best_seq = starter.copy()
                best_score = cur
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass  # no global starter available

    # ------------------------------------------------------------------
    # Deterministic initial shapes (used for the first few trials)
    # ------------------------------------------------------------------
    def triangular_vector():
        idx = np.arange(n)
        a = np.minimum(idx + 1, n - idx)
        a = a.astype(float)
        a /= a.sum()
        return a

    def sine_vector():
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
        return a

    def cosine_vector():
        a = 0.5 * (1 - np.cos(np.pi * np.arange(n) / (n - 1)))  # half‑cosine bump
        a /= a.sum()
        return a

    def beta_vector(alpha, beta):
        # Beta distribution on (0,1) discretised to n bins
        x = (np.arange(1, n + 1) - 0.5) / n
        a = np.power(x, alpha - 1) * np.power(1 - x, beta - 1)
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a.sum()
        return a

    deterministic_inits = [
        triangular_vector,
        sine_vector,
        cosine_vector,
        lambda: beta_vector(0.5, 0.5),   # U‑shaped (more mass at the ends)
        lambda: beta_vector(2.0, 2.0),   # bell‑shaped
    ]

    trial_count = 0
    min_trial_time = 30.0          # seconds per trial
    final_margin = 60.0             # reserve time for final polishing

    # ------------------------------------------------------------------
    # Repeated short trials (random restarts + a few deterministic seeds)
    # ------------------------------------------------------------------
    while True:
        remaining = deadline - time.time()
        if remaining < min_trial_time + final_margin:
            break

        trial_deadline = time.time() + min_trial_time

        # pick an initial vector
        if trial_count < len(deterministic_inits):
            a0 = deterministic_inits[trial_count]()
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        trial_count += 1

        # Phase‑1: Adam on smoothed objective
        a_adam, score_adam = _adam_smooth(
            a0,
            n,
            trial_deadline,
            start_time,
            alpha_start=0.5,
            alpha_end=500.0,
            max_iters=8000,
            lr0=0.6,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial {trial_count} Adam improve: {score_adam:.6f}")

        # Phase‑2: sub‑gradient descent (use almost all remaining trial time)
        sub_deadline = trial_deadline - 5.0   # keep a few seconds for anneal
        a_sub, score_sub = _subgradient_descent(
            a_adam,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial {trial_count} Sub‑grad improve: {score_sub:.6f}")

        # Phase‑3: quick annealing polish
        a_ann, score_ann = _annealing(
            a_sub,
            n,
            trial_deadline,
            start_time,
            best_score,
            a_sub,
            start_temp=1e-2,
            end_temp=1e-6,
            steps=50_000,
        )
        if score_ann < best_score:
            best_score = score_ann
            best_seq = a_ann.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial {trial_count} Anneal improve: {score_ann:.6f}")

    # ------------------------------------------------------------------
    # Final refinement on the best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()
        print(f"[{time.time() - start_time:.2f}s] No candidate found – using uniform (score={best_score:.6f})")

    # Long sub‑gradient phase (≈60 % of remaining budget)
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.6
        a_sub, score_sub = _subgradient_descent(
            best_seq,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Final sub‑grad improve: {score_sub:.6f}")

    # Final annealing polish (use whatever time is left)
    if time.time() < deadline:
        a_final, score_final = _annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve: {score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
