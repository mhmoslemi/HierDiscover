"""
Improved step‑function search.

The optimisation proceeds in four stages:

  0 – many cheap Adam runs on a modest length (n≈1024) with a *wide* set of
      initialisations (Chebyshev, Kaiser, Tukey, Gaussian, sparse Dirichlet).
  1 – up‑sample the current best to the full scale (n≈8192) and continue the
      Adam optimisation on the log‑sum‑exp surrogate of the max‑convolution.
  2 – long projected (Nesterov) sub‑gradient descent on the true objective
      F(a)=2·n·max(conv(a)).
  3 – an extended simulated‑annealing “mass‑transfer’’ polishing stage.

All stages respect the global time budget (default 1000 s).  The final vector
is projected onto the probability simplex (∑aᵢ=1, aᵢ≥0) – this matches the
scale used by ``evaluate_sequence``.  The implementation is deliberately
self‑contained and uses only `numpy` (and optionally `scipy.signal` for a few
structured windows).
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (linear via FFT)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in the exact polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa fixed)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9994,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    using Adam and Euclidean projection onto the simplex.
    Returns the best vector seen and its *true* (unsmoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true objective for tracking
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # numeric stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """Classic Nesterov projected sub‑gradient descent on F(a)=2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """Very light SA that transfers a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source proportional to its current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """

    Distribution is chosen among:

      – Kaiser window,
      – Tukey window,
      – Gaussian window.
    If SciPy is not available a Gaussian fallback is used.
    The result is normalised to sum = 1.
    """

        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.25:
        # Chebyshev

        w = chebwin(n, at=att, sym=True).astype(float)
    elif have_signal and r < 0.5:

        w = kaiser(n, beta).astype(float)
    elif have_signal and r < 0.75:
        # Tukey
        alpha = rng.uniform(0, 1.0)
        w = tukey(n, alpha).astype(float)
    else:
        # Gaussian – σ chosen as a random fraction of the length
        sigma = rng.uniform(0.08, 0.22) * n

    return w

    n0 = 1024
    remaining = deadline - time.time()
    stage0_time = max(20.0, 0.15 * remaining)
    slice_deadline = time.time() + stage0_time

    max_restarts = 20
    for r in range(max_restarts):
        if time.time() >= deadline:
            break

        a0 = _window_initialiser(n0, rng)

        # Occasionally replace with a sparse Dirichlet draw
        if rng.random() < 0.2:
            alpha = rng.uniform(0.05, 0.3)
            g = rng.gamma(alpha, 1.0, size=n0)
            a0 = g / g.sum()

        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 6.0, 1e-8)

        a_opt, score = _run_adam_softmax(
            a0,
            n0,
            slice_deadline,
            start_time,
            tau0=tau0,
            tau_end=1e-15,
            tau_decay=0.9996,
            step_size=0.25,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage0 improve: "
                  f"restart={r}, score={score:.6f}")

    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # --------------------------------------------------------------
    # Stage 1 – upscale to the full scale (≈8192) and keep Adam
    # --------------------------------------------------------------
    n1 = 8192
    a1 = _upsample(best_seq, n1)

    conv1 = _conv_self(a1)
    tau0_1 = max(conv1.max() * 6.0, 1e-10)

    remaining = deadline - time.time()
    stage1_time = max(30.0, 0.30 * remaining)   # give a generous share
    slice_deadline = time.time() + stage1_time

    a1_opt, score1 = _run_adam_softmax(
        a1,
        n1,
        slice_deadline,
        start_time,
        tau0=tau0_1,
        tau_end=1e-15,
        tau_decay=0.9995,
        step_size=0.20,
        beta1=0.9,
        beta2=0.999,
        eps=1e-8,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage1 improve: score={score1:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – long projected sub‑gradient descent on the true max‑conv
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        grad_deadline = deadline - 2.0
        a2_opt, score2 = _run_subgradient_descent(
            best_seq,
            best_seq.shape[0],
            grad_deadline,
            start_time,
            beta=0.98,
            step0=0.10,
        )
        if score2 < best_score:
            best_score = score2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage2 improve: score={score2:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – extended simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        # allow as many moves as fit into the remaining budget
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-8,
            steps=1_200_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Normalise to the simplex (guard against tiny drift) and return
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
