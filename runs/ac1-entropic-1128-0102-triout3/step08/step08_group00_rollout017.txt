# ----------------------------------------------------------------------
# Basic linear‑algebra helpers

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    beta1_pow = beta2_pow = 1.0

    L_grad = _next_pow2(3 * n - 2)          # length for gradient FFTs

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        cur_score = float(2 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient of the smoothed objective
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Exact max – sub‑gradient descent
# ----------------------------------------------------------------------
def _subgrad_opt(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 best_score_initial: float,
                 step0: float = 0.25,
                 momentum_beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses momentum and a 1/√t step‑size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # gradient of the (unique) maximal entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : Deterministic hill‑climb (always moves mass away from the
#           most harmful indices)
# ----------------------------------------------------------------------
def _hill_climb_opt(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Very cheap greedy improvement: at each iteration move a small fraction
    of mass from the index that contributes most to the current maximal
    convolution entry to the index that contributes least.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # gradient of conv[k] w.r.t. a is simply a reversed slice
        g = np.zeros_like(a)
        g[idx] = a[k - idx]

        src = idx[g[idx].argmax()]          # biggest contributor
        dst = idx[g[idx].argmin()]          # smallest contributor

        if src == dst or a[src] < 1e-14:
            # fall back to a tiny random move
            src = rng.choice(idx, p=a[idx] / a[idx].sum())
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

        eps = min(a[src] * 0.2, 0.005)      # move up to 20 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = float(2 * n * _conv_self(a_prop).max())
        if new_score < best_score:
            best_score, best_a = new_score, a_prop.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb new best={new_score:.6f}")

        # accept if improvement, otherwise keep the old vector
        if new_score < best_score * 1.001:   # a tiny tolerance to avoid stagnation
            a = a_prop
        else:
            # occasional random acceptance to escape tiny plateaux
            if rng.random() < 0.001:
                a = a_prop

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 : Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing_opt(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing with a long exponential temperature schedule.
    """
    a = a0.copy()
    score = float(2 * n * _conv_self(a).max())
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)          # source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 50 % of the source mass
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = float(2 * n * _conv_self(a_prop).max())
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the provided ``evaluate_sequence`` under the given time budget.
    The routine performs many short optimisation trials, keeps the best
    solution, and finishes with a longer refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # problem size – 4096 is FFT‑friendly and was used previously
    n = 4096

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a global starter exists, evaluate it first.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() < 1e-12:
                starter = np.full_like(starter, 1.0 / starter.size)
            else:
                starter = starter / starter.sum()
            cur = float(2 * n * _conv_self(starter).max())
            best_seq = starter.copy()
            best_score = cur
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Repeated short trials (random restarts)
    # --------------------------------------------------------------
    min_trial_time = 80.0          # reserve a chunk for the final polish
    final_margin = 20.0

    while True:
        remaining = deadline - time.time()
        if remaining < min_trial_time + final_margin:
            break

        trial_deadline = time.time() + min_trial_time

        # random start on the simplex
        a0 = np.random.rand(n)
        a0 /= a0.sum()

        # ---- Phase‑1 (Adam smooth) ---------------------------------
        a_adam, score_adam = _adam_opt(
            a0,
            n,
            trial_deadline,
            start_time,
            alpha_start=0.5,
            alpha_end=500.0,
            max_iters=20000,
            lr0=0.6,
        )
        if score_adam < best_score:
            best_score, best_seq = score_adam, a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial Adam improve: score={score_adam:.6f}")

        # ---- Phase‑2 (sub‑gradient) ---------------------------------
        sub_deadline = trial_deadline - 5.0   # keep a few seconds for later phases
        a_sub, score_sub = _subgrad_opt(
            a_adam,
            n,
            sub_deadline,
            start_time,
            best_score,
            step0=0.30,
            momentum_beta=0.9,
        )
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial sub‑grad improve: score={score_sub:.6f}")

        # ---- Phase‑3 (hill‑climb) ----------------------------------
        hill_deadline = sub_deadline - 3.0
        a_hill, score_hill = _hill_climb_opt(
            a_sub,
            n,
            hill_deadline,
            start_time,
            best_score,
        )
        if score_hill < best_score:
            best_score, best_seq = score_hill, a_hill.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial hill‑climb improve: score={score_hill:.6f}")

        # ---- Phase‑4 (quick anneal) --------------------------------
        anneal_deadline = trial_deadline
        a_fin, score_fin = _annealing_opt(
            a_hill,
            n,
            anneal_deadline,
            start_time,
            best_score,
            a_hill,
            start_temp=1e-2,
            end_temp=1e-7,
            steps=80_000,
        )
        if score_fin < best_score:
            best_score, best_seq = score_fin, a_fin.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial anneal improve: score={score_fin:.6f}")

    # ------------------------------------------------------------------
    # Final refinement on the best candidate found so far.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = float(2 * n * _conv_self(best_seq).max())

    # Longer sub‑gradient descent with the remaining budget
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.55
        a_sub, score_sub = _subgrad_opt(
            best_seq,
            n,
            sub_deadline,
            start_time,
            best_score,
            step0=0.20,
            momentum_beta=0.95,
        )
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Final sub‑grad improve: score={score_sub:.6f}")

    # Final annealing polish (uses whatever time is left)
    if time.time() < deadline:
        a_final, score_final = _annealing_opt(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=400_000,
        )
        if score_final < best_score:
            best_score, best_seq = score_final, a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more and cast to plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished. Best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
