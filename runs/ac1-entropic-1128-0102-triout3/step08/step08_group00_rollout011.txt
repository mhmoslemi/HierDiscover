def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n=4096
    L = _next_pow2(2*n-1)

    best_seq = None
    best_score = float('inf')

    # The uniform baseline
    uniform = np.full(n, 1.0/n)
    best_seq = uniform.copy()
    best_score = _evaluate_fast(best_seq)

    # 1. parametric families
    # geometric
    eps_grid = np.logspace(-4, -1, 200)  # epsilon = 1 - r
    for eps in eps_grid:
        r = 1 - eps
        geo = np.power(r, np.arange(n))
        geo /= geo.sum()
        for lam in np.linspace(0,1,11):
            seq = lam*uniform + (1-lam)*geo
            # ensure nonnegative
            seq = np.maximum(seq, 0)
            seq /= seq.sum()
            score = _evaluate_fast(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(...)

    # power family
    p_grid = np.linspace(-1,2,61) # includes negative for decreasing
    idx = np.arange(n)
    for p in p_grid:
        pow_seq = np.power(idx+1, p)
        pow_seq = np.maximum(pow_seq,0)
        pow_seq /= pow_seq.sum()
        score = _evaluate_fast(pow_seq)
        if score < best_score:
            best_score = score
            best_seq = pow_seq.copy()
            print(...)

    # beta family
    p_vals = [0,0.5,1,1.5,2]
    q_vals = [0,0.5,1,1.5,2]
    for p in p_vals:
        for q in q_vals:
            beta_seq = np.power(idx+1, p) * np.power(n - idx, q)
            beta_seq = np.maximum(beta_seq,0)
            beta_seq /= beta_seq.sum()
            score = _evaluate_fast(beta_seq)
            if score < best_score:
                best_score = score
                best_seq = beta_seq.copy()
                print(...)

    # 2. refine via subgradient descent
    if time.time() < deadline - 30:   # keep ~30s for final polish
        sub_deadline = deadline - 20  # leave 20 sec
        a_sub, score_sub = _subgradient_descent(best_seq, n, sub_deadline, start, best_score)
        if score_sub < best_score:
            best_score = score_sub

    # 3. annealing polish

"""
Propose a step‑function (non‑negative list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096 (the same size used by the reference code).

The search consists of three stages:

1️⃣  **Fast parametric sweep** – families of simple analytic shapes
    (geometric, power, Beta‑type and mixtures with the uniform vector)
    are generated, normalised to unit sum and evaluated with a cheap
    FFT‑based convolution.  The best candidate of this cheap sweep is kept.

2️⃣  **Projected sub‑gradient descent** – starting from the best candidate,
    an exact sub‑gradient of the true objective is applied.
    The iterate is always projected onto the probability simplex.
    This phase takes the remaining bulk of the time budget.

3️⃣  **Light simulated‑annealing polish** – a mass‑transfer annealer makes
    tiny random moves between two coordinates, accepting moves that lower
    the objective or with a Metropolis probability.
    The temperature decays exponentially.

All helpers are top‑level functions; no closures or lambdas are used.
The routine respects the 1000 s wall‑clock budget and returns a plain
Python list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (unit‑sum sequences → 2·n·max(conv) )
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Return the exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    # sum(a) should be 1; we keep the denominator for safety.
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑2 : exact max – projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3          # a slightly larger initial step than the reference
    it = 0

    L_grad = _next_pow2(3 * n - 2)   # length for the gradient FFTs

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # gradient of the active diagonal (the one attaining the max)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (moves a small fraction of mass from one
    coordinate to another)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportional to its mass and a random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    Find a non‑negative step‑function (list of heights) that minimises

    """
    np.random.seed(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 0.5

    n = 4096                     # size used by the reference solution
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # If a global starter is provided, evaluate it first.
    # ------------------------------------------------------------------
    try:

    except Exception:

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(n)

    # --- geometric family ------------------------------------------------
    eps_grid = np.logspace(-4, -1, 200)          # epsilon = 1‑r
    for eps in eps_grid:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
