"""
Search for a non‑negative step‑function that minimizes
`evaluate_sequence`.  The algorithm is a multi‑stage pipeline:

  1.  Many cheap Adam runs on a modest size (n≈1024) with
      diverse structured initialisations (random Dirichlet,
      Chebyshev, Kaiser windows).  This stage explores the space
      quickly and yields a handful of promising seeds.

  2.  Progressive up‑sampling: the best seed is linearly
      interpolated to larger sizes (n≈4096 → 8192) and further
      refined with Adam on a soft‑max surrogate of the max‑convolution.

  3.  Projected (Nesterov) sub‑gradient descent on the *exact*
      objective `F(a)=2·n·max(conv(a))` – this is the core polishing
      stage.

  4.  An extended simulated‑annealing phase that repeatedly
      transfers a random fraction of mass between two cells.
      The temperature schedule is very slow (≈2 × 10⁶ steps) to
      allow thorough exploration.

  5.  A final short Adam polish (again on the smoothed objective)
      to settle any remaining tiny improvements.

All stages respect the global time‑budget (default 1000 s) and the
sequence is always projected onto the probability simplex
`Σ a_i = 1, a_i ≥ 0`.  The returned list contains plain Python
floats and will be evaluated by the harness.
"""
import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0, Σa=1)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9994,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    with Adam and Euclidean projection onto the simplex.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # current true score (no smoothing)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

        # early stop if no improvement for a while
        if time.time() - last_improve > 15.0:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that transfers a random fraction of mass
    between two cells.  Uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.45   # up to 45 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to Σ=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisation helpers (Chebyshev / Kaiser windows)
# ----------------------------------------------------------------------


def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    With probability 0.4 we produce a Chebyshev window,
    with probability 0.4 a Kaiser window,
    otherwise a plain random Dirichlet draw.
    """
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
        w = np.maximum(w, 0.0)
    elif have_signal and r < 0.8:
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
        w = np.maximum(w, 0.0)
    else:
        w = rng.random(n)
        w = np.maximum(w, 0.0)

    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w
