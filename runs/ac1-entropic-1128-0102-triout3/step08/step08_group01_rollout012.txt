"""
Iterative LP‑based optimizer for the max‑self‑convolution problem.

The algorithm works in three phases:

1. **LP improvement** – given a current step function `f`,
   solve a linear program that maximises the total mass `∑b_i`
   under the constraint `‖f∗b‖∞ ≤ M`, where `M = ‖f∗f‖∞`.
   After solving, the candidate `b` is scaled back to the
   normalising sum `sqrt(2·n)`.  If the scaled version yields a
   strictly smaller max‑convolution we replace `f` by it;
   otherwise we take a convex combination `f ← (1‑t)f + t·g`.
   Repeating this step monotonically reduces the objective
   `M = 2·n·max(conv(f,f))/(∑f)²`.

2. **Projected sub‑gradient polishing** – after the LP phase we
   run a few thousand iterations of projected sub‑gradient descent
   on the exact objective (the same routine used in the reference
   solution).  This removes any remaining small irregularities.

3. **Targeted simulated‑annealing** – a short annealing run performs
   local mass‑moves from the pair of indices that contribute most to
   the current peak, allowing a final fine‑tune.

All helper functions are top‑level; the optimiser respects the
1000 s time‑budget and constantly prints progress.  The returned
list is a plain Python list of non‑negative heights whose sum is
normalised to 1 (the `evaluate_sequence` routine rescales it
internally, so any positive scaling is equivalent).
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Compute the current objective value (M = max self‑convolution)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Return max(conv(a,a)) given a (sum is irrelevant)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(conv.max())


# ----------------------------------------------------------------------
# Linear‑program: maximise total mass under a convolution bound
# ----------------------------------------------------------------------
def _build_lp_solver(n: int):
    """
    Build a CVXPY problem that, for a given f and bound M,
    maximises sum(b) subject to b≥0 and conv(f,b) ≤ M.
    Returns (problem, b_var, f_param, M_param).
    """
    f_param = cp.Parameter(n, nonneg=True)
    M_param = cp.Parameter(nonneg=True)   # scalar bound
    b = cp.Variable(n, nonneg=True)

    # conv(f,b) has length 2n‑1
    conv_fb = cp.conv(f_param, b)

    constraints = [conv_fb <= M_param, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    return prob, b, f_param, M_param


def _solve_lp(prob, b_var, f_param, M_param, f_curr, M_curr):
    """
    Solve the LP for the given current f and bound M.
    Returns the optimal b (as a dense NumPy array) or None on failure.
    """
    f_param.value = f_curr
    M_param.value = M_curr
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=200_000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9, verbose=False)
    except Exception as e:
        print(f"LP solve exception: {e}")
        return None
    if b_var.value is None:
        print("LP did not produce a solution.")
        return None
    b_val = np.array(b_var.value, dtype=float)
    # numerical safety
    b_val = np.maximum(b_val, 0.0)
    return b_val


# ----------------------------------------------------------------------
# Phase‑2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Gradient is taken at the current maximal index.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.20  # smaller step as we start from a good point
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve (LP‑polish): "
                  f"iter={it:,}, score={cur_score:.6f}")

        # index where the max occurs (break ties arbitrarily)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – Targeted simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-8,
                            steps: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Annealing that always picks the current maximal convolution index
    and moves a small amount of probability mass away from the pair
    that contributes most to that peak.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate current maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        # contribution of each index to the peak
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move at most 20 % of the source mass
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    # Phase 1 – LP‑based monotone improvement
    # ------------------------------------------------------------------
    prob, b_var, f_param, M_param = _build_lp_solver(n)
    t = 0.5                     # initial blending factor
    lp_iter = 0
    # We allocate roughly 30 % of the remaining budget to the LP phase.
    phase1_deadline = time.time() + max(10.0, 0.30 * (deadline - time.time()))

    while time.time() < phase1_deadline:
        lp_iter += 1
        M_curr = _max_conv(f)           # current bound
        b_opt = _solve_lp(prob, b_var, f_param, M_param, f, M_curr)
        if b_opt is None:
            print("LP failed – exiting LP phase.")
            break

        sum_b = b_opt.sum()
        if sum_b <= 0:
            print("LP produced zero mass – exiting LP phase.")
            break

        # Scale to the required sum
        scale = target_sum / sum_b
        g = b_opt * scale

        # New bound after scaling
        M_new = _max_conv(g)

        # Accept the better of (g) or a convex blend.
        if M_new < M_curr - 1e-12:
            f = g
            M_curr = M_new
            action = "replace"
        else:
            # Blend conservatively
            f = (1.0 - t) * f + t * g
            M_curr = _max_conv(f)
            action = "blend"

        if M_curr < best_score - 1e-12:
            best_score = M_curr
            best_seq = f.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter}: {action}, "
                  f"M improved to {M_curr:.6f}")

        # decay blending factor slowly
        t = max(t * 0.95, 0.01)

        # early stop if progress stalls
        if lp_iter > 50 and lp_iter % 10 == 0:
            recent = best_score
            if recent >= best_score * (1 - 1e-5):
                # no meaningful improvement in the last few iterations
                break

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient polishing (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(30.0, 0.40 * remaining)  # ~40% of remaining
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad polish improve: "
                  f"score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted annealing (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_targeted_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=3e-2,
            end_temp=1e-8,
            steps=80_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        # fallback uniform
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished: best M ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
