def _run_optimal_incremental_refine(a0, n, deadline, start_time, epsilon_frac_start=0.04, epsilon_frac_end=0.001):

    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    it = 0

    while time.time() < deadline:
        # compute gradually decreasing epsilon fraction
        elapsed = time.time() - (deadline - total_time)
        eps_frac = max(epsilon_frac_end, epsilon_frac_start * (1 - elapsed/total_time))

        # choose k_max
        k_max = conv.argmax()  # int

        # source indices involved in max bin
        i_start = max(0, k_max - (n-1))
        i_end = min(n-1, k_max)
        i_range = np.arange(i_start, i_end+1)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            src = rng.choice(n, p=a/a.sum())

        if a[src] < 1e-14:
            continue

        # destination: choose an index that has low contribution to max bin (i.e., low a[k_max - dst])? same as previous but also perhaps random
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        # prefer low dest_contrib:
        prob_dst = (1.0/(dest_contrib+1e-12))
        prob_dst[src] = 0  # avoid self
        prob_dst /= prob_dst.sum()
        dst = rng.choice(n, p=prob_dst)

        # compute slope s
        s = np.zeros(2*n-1, dtype=float) # could allocate each iteration, but okay.
        s[src:src+n] -= 2.0 * a
        s[dst:dst+n] += 2.0 * a

        # compute slope at k_max
        s_max = s[k_max]
        if s_max >= 0:
            # not good, try another dst (maybe choose different src/dst)
            continue

        # compute epsilon_star
        # mask where s > s_max
        mask = s > s_max
        if not np.any(mask):
            # No other slope greater; epsilon unlimited (just limited by src mass)
            eps_star = np.inf
        else:
            # For each t in mask, compute eps_i = (conv[k_max] - conv[t]) / (s[t] - s_max)
            # Only consider t where conv[t] < conv[k_max] (since else denominator sign)
            delta_conv = conv[k_max] - conv[mask]  # >=0 presumably
            delta_s = s[mask] - s_max  # >0
            # We need only positive eps
            # Because conv[k_max] might equal conv[t] for some t (multiple max entries); if conv[t]==conv[k_max] then delta_conv=0 => eps=0; ignore those (they're already at same level).
            eps_vals = delta_conv / delta_s

            eps_vals = eps_vals[eps_vals > 1e-16]
            if eps_vals.size == 0:
                eps_star = np.inf
            else:

        eps_limit = a[src] * eps_frac  # i.e., move at most eps_frac fraction of src's mass

"""
Improved search for a low‑max self‑convolution step‑function.

The algorithm consists of three stages:

1️⃣  **Exponentiated‑gradient warm‑up** on a smooth approximation of the
    max (identical to the reference implementation).

2️⃣  **Projected sub‑gradient descent** on the exact ℓ∞‑norm of the
    self‑convolution (again as before, but with a larger iteration budget).

3️⃣  **Optimal incremental refinement** – each move transfers a fraction of
    mass from a “source’’ index (contributing heavily to the current maximal
    convolution bin) to a “destination’’ index (contributing little).  The
    effect of the move on *all* convolution entries is linear in the step
    size ε, so we can compute the largest ε that still reduces the current
    maximum exactly (by solving a simple 1‑D linear programme).  This yields
    far larger improvements per iteration than the tiny random ε used
    previously, yet each iteration remains O(n) and runs in a few
    micro‑seconds.  The procedure runs until the time budget expires,
    keeping the best sequence seen.

The whole search respects a 1000 s wall‑clock budget and returns the best
sequence (a list of non‑negative heights) found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient warm‑up on smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: optimal incremental refinement (large guaranteed moves)
# ----------------------------------------------------------------------
def _run_optimal_incremental_refine(a0: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    start_time: float,
                                    eps_start: float = 0.04,
                                    eps_end: float = 0.001,
                                    rel_improve: float = 1e-12,
                                    recompute_every: int = 5000):
    """
    Perform a huge number of mass‑shifts.  For a chosen source `src` and
    destination `dst` the effect on the whole convolution is linear in ε:
        Δconv = ε·s ,   s[t] = 2·(a[t‑dst] – a[t‑src]) (with out‑of‑range = 0).

    The algorithm computes the largest ε that still reduces the current
    maximal entry (the solution of a 1‑D linear programme).  This gives
    far larger progress per move while keeping the per‑iteration cost O(n).
    """
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # current maximal index
        k_max = int(np.argmax(conv))

        # ---- source: pick an index that really contributes to the max bin ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ---- destination: prefer indices that **do not** appear much in the max bin ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                     # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ---- compute the per‑unit‑ε effect on the whole convolution ----
        # s[t] = -2·a[t‑src] + 2·a[t‑dst]
        s = np.zeros(2 * n - 1, dtype=float)
        s[src:src + n] -= 2.0 * a
        s[dst:dst + n] += 2.0 * a

        s_max = s[k_max]
        if s_max >= 0.0:
            # moving mass would not lower the current max – try a different pair
            continue

        # ---- maximal admissible ε before another entry overtakes the max ----
        mask = s > s_max                     # slopes strictly larger than s_max
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]          # >=0
            delta_s = s[mask] - s_max                     # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        # limit ε by the amount of mass we can take from src
        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 0.0:
            continue

        # ---- apply the move analytically (O(n)) ----
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        # ensure genuine improvement (avoid tiny numerical noise)
        if new_max < max_val - rel_improve:
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            accepted += 1
            it += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional re‑normalisation / recomputation to keep float errors in check
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a
