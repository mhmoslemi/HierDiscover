"""
Non‑negative step‑function optimizer.

The optimizer starts from a collection of diverse initial vectors (including the
previous best one if available) and then performs a dedicated *hill‑climbing*
procedure that repeatedly transfers a small amount of mass from the index that
contributes most to the current maximum convolution peak to the index with the
smallest coefficient.  The transfer size is chosen so that the peak is driven
down close to the second‑largest peak, which yields much larger progress per
iteration than the fixed‑step greedy method used before.

The main loop respects the global time budget (default 1000 s).  The best
sequence found is finally normalised to sum 1 and returned as a plain Python
list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Full linear convolution a*a using FFT (length 2·len(a)‑1).
    Returns a float64 array.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective value 2·n·max(conv)."""
    return float(2.0 * n * conv.max())


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Random starts
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc` and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Discrete Gaussian (σ≈n/6) centred and normalised."""
    sigma = n / 6.0
    centre = (n - 1) / 2.0
    x = np.arange(n, dtype=float)
    g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.98) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


# ----------------------------------------------------------------------
# Core hill‑climbing routine
# ----------------------------------------------------------------------


def _hill_climb(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 1_000_000,
    eps_frac: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing on the true objective.
    Repeatedly moves a fraction of the coefficient that most contributes to the
    maximal convolution peak towards the smallest coefficient.
    The step size is chosen so that the peak is driven close to the second‑largest
    peak, which gives much larger per‑iteration progress.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    # initial score
    conv = _conv_self_fft(a, L)
    best_score = _score_from_conv(conv, n)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # recompute convolution and locate the dominant peak
        conv = _conv_self_fft(a, L)
        max_idx = int(conv.argmax())
        cur_max = float(conv[max_idx])

        # second largest value (used to decide a sensible transfer size)
        if conv.size > 1:
            second_max = float(np.partition(conv, -2)[-2])
        else:
            second_max = cur_max

        # --------------------------------------------------------------
        # 1) Identify the source index that contributes most to the peak.
        # --------------------------------------------------------------
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        i = np.arange(lo, hi + 1, dtype=int)
        j = max_idx - i
        contrib = a[i] * a[j]                     # raw contribution (half of the true weight)
        src = i[int(contrib.argmax())]            # most responsible index

        # --------------------------------------------------------------
        # 2) Destination: the current smallest coefficient (different from src)
        # --------------------------------------------------------------
        dst_candidates = np.delete(np.arange(n), src)
        dst = int(dst_candidates[int(a[dst_candidates].argmin())])

        # --------------------------------------------------------------
        # 3) Compute a safe transfer amount.
        # --------------------------------------------------------------
        # Contribution of src/dst to the maximal convolution entry
        ks = max_idx - src
        kd = max_idx - dst
        a_ks = a[ks] if 0 <= ks < n else 0.0
        a_kd = a[kd] if 0 <= kd < n else 0.0

        denom = 2.0 * (a_ks - a_kd)               # delta = 2*eps*(a_kd - a_ks)
        if denom <= 0.0:
            # fallback: move a modest fraction of src to dst
            eps = eps_frac * a[src]
        else:
            # epsilon that would bring the current peak down to the second peak
            eps_needed = (cur_max - second_max) / denom
            # stay safely away from the exact target to avoid overshoot due to
            # rounding and the quadratic term that we ignore.
            eps_target = min(a[src] * 0.9, eps_needed * 0.9)
            eps = max(eps_target, eps_frac * a[src] * 0.1)   # guarantee progress

        eps = float(eps)
        if eps <= 1e-14 or a[src] <= eps:
            # nothing sensible can be moved – quit this run
            break

        # --------------------------------------------------------------
        # 4) Apply the move and test whether the peak improves.
        # --------------------------------------------------------------
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        conv_new = _conv_self_fft(a_new, L)
        new_max = float(conv_new.max())

        if new_max < cur_max - 1e-15:          # strict improvement
            a = a_new
            cur_score = _score_from_conv(conv_new, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] hill‑climb iter={it:,} "
                    f"score={cur_score:.6f}"
                )
            continue

        # --------------------------------------------------------------
        # 5) If the greedy move failed, try a few alternative destinations.
        # --------------------------------------------------------------
        # Sort a few smallest entries and retry.
        small_idx = np.argsort(a)[:5]
        improved = False
        for d in small_idx:
            if d == src:
                continue
            kd = max_idx - d
            a_kd = a[kd] if 0 <= kd < n else 0.0
            denom2 = 2.0 * (a_ks - a_kd)
            if denom2 <= 0.0:
                continue
            eps_needed2 = (cur_max - second_max) / denom2
            eps2 = min(a[src] * 0.9, eps_needed2 * 0.9)
            eps2 = max(eps2, eps_frac * a[src] * 0.1)
            if eps2 <= 1e-14:
                continue
            a_alt = a.copy()
            a_alt[src] -= eps2
            a_alt[d] += eps2
            conv_alt = _conv_self_fft(a_alt, L)
            if conv_alt.max() < cur_max - 1e-15:
                a = a_alt
                cur_score = _score_from_conv(conv_alt, n)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] hill‑climb alt dest iter={it:,} "
                        f"score={cur_score:.6f}"
                    )
                improved = True
                break
        if improved:
            continue

        # --------------------------------------------------------------
        # 6) As a last resort perform a random transfer (simulated‑annealing‑like).
        # --------------------------------------------------------------
        positive = np.where(a > 1e-16)[0]
        if positive.size < 2:
            break
        src_r = int(np.random.choice(positive))
        dst_r = int(np.random.choice(np.delete(np.arange(n), src_r)))
        eps_r = eps_frac * a[src_r]
        if eps_r <= 1e-14:
            continue
        a_rand = a.copy()
        a_rand[src_r] -= eps_r
        a_rand[dst_r] += eps_r
        conv_rand = _conv_self_fft(a_rand, L)
        if conv_rand.max() < cur_max - 1e-15:
            a = a_rand
            cur_score = _score_from_conv(conv_rand, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] hill‑climb random iter={it:,} "
                    f"score={cur_score:.6f}"
                )
            continue

        # No improvement this iteration – continue with next attempt.
        # (We intentionally do *not* break here to give the algorithm many chances.)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The search consists of a cheap exploration of several diverse starts and a
    dedicated hill‑climbing phase that transfers mass from peak‑responsible
    indices to the smallest ones.  The routine respects the overall budget and
    returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin to avoid being killed right at the deadline
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # Problem size – 4800 proved a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # Build a list of starting vectors (probability vectors, i.e. sum = 1).
    # ------------------------------------------------------------------
    starts = []

    # 1) previous construction, if it exists
    try:
        seq = np.asarray(height_sequence_1, dtype=float)  # noqa: F821
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            seq = seq / seq.sum()
            starts.append(seq)
    except Exception:
        pass

    # 2) a few generic seeds
    starts.append(np.full(n, 1.0 / n))                # uniform
    starts.append(_exponential_start(n, r=0.985))    # exponential decay
    starts.append(_gaussian_start(n))                # bell‑shaped Gaussian
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))   # smooth Dirichlet
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))   # spiky Dirichlet

    # ------------------------------------------------------------------
    # Time allocation:
    #   – 30 % for a quick exploration (no heavy optimisation),
    #   – the remaining 70 % for aggressive hill‑climbing on the best start.
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.30 * budget_s

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Exploration phase – evaluate each start cheaply and keep the best.
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        if time.time() >= explore_deadline:
            break
        score = evaluate_sequence(a0.tolist())
        print(
            f"[{time.time() - start_time:6.2f}s] exploration start #{idx+1} "
            f"score={score:.6f}"
        )
        if score < best_score:
            best_score = score
            best_seq = a0.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (explore) "
                f"score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Hill‑climbing phase – improve the current best for the rest of the budget.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Ensure we really start from a probability vector.
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Remaining budget is spent on a single, long hill‑climb run.
    print(f"[{time.time() - start_time:6.2f}s] hill‑climb phase start")
    final_seq, final_score = _hill_climb(
        best_seq, deadline, start_time, max_iters=10_000_000, eps_frac=0.2
    )
    if final_score < best_score:
        best_score = final_score
        best_seq = final_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (hill‑climb) "
            f"score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Final safety normalisation and reporting.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_report = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_report:.6f}")

    # Return a plain Python list of floats as required.
    return [float(x) for x in best_seq.tolist()]
