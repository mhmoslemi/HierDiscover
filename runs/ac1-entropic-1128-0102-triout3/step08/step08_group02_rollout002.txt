def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine n
    try:
        init_seq = np.asarray(height_sequence_1, dtype=float)
        n = init_seq.shape[0]
    except Exception:
        n = 4800
        init_seq = None

    # Build start vectors
    starts = []
    if init_seq is not None:
        starts.append(init_seq / max(init_seq.sum(), 1e-12))
    # add uniform
    starts.append(np.full(n, 1.0/n))
    # exponential left heavy
    r = 0.99
    exp_seq = np.array([r**i for i in range(n)], dtype=float)
    exp_seq /= exp_seq.sum()
    starts.append(exp_seq)
    # symmetric U-shape
    idx = np.arange(n)
    u_seq = 1.0/(np.minimum(idx+1, n-idx)+1.0)
    u_seq /= u_seq.sum()
    starts.append(u_seq)
    # random Dirichlet
    rng = np.random.default_rng(seed+1)
    for _ in range(2):
        rand_seq = rng.random(n)
        rand_seq = np.clip(rand_seq, 0, None)
        if rand_seq.sum() == 0:
            rand_seq = np.full(n, 1.0/n)
        else:
            rand_seq /= rand_seq.sum()
        starts.append(rand_seq)

    best_seq = None
    best_score = float('inf')
    best_candidates = []   # store top 5

    # allocate for each start proportion may vary based on remaining time.
    for s_idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time()-start_time:6.2f}s] Start {s_idx+1}/{len(starts)}")
        a = np.maximum(a0, 0.0)
        if a.sum() == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a.sum()

        # Adam phase (smooth)
        phase_deadline = min(deadline, time.time() + 0.04 * (deadline - start_time))
        a = _adam_phase(a, phase_deadline, start_time, _next_pow2(3*n-2))

        # Subgradient descent
        if time.time() < deadline:
            sub_deadline = min(deadline, time.time() + 0.45 * (deadline - time.time()))
            a, _ = _subgradient_descent(a, sub_deadline, start_time)

        # local mass transfer hill climbing
        if time.time() < deadline:
            a, _ = _run_local_hill_climb(a, deadline, start_time)

        # Simulated annealing for remaining part of this start
        if time.time() < deadline:
            remaining = deadline - time.time()
            # allocate half for annealing else proceed next start
            anneal_time = remaining * 0.5
            anneal_deadline = time.time() + anneal_time
            a, _ = _run_annealing(a, anneal_deadline, start_time,
                                  start_temp=0.1, end_temp=1e-6,
                                  steps=int(200_000 * (anneal_time/ (deadline-start_time))))

        # final evaluate
        cur_score = evaluate_sequence(a.tolist())
        print(f"[{time.time()-start_time:6.2f}s] End of start {s_idx+1} score {cur_score:.6f}")
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] *** NEW GLOBAL BEST {best_score:.6f}")

        # keep top candidates list (size up to 5)
        best_candidates.append((cur_score, a.copy()))
        best_candidates.sort(key=lambda x: x[0])
        best_candidates = best_candidates[:5]

    # Mixing step: try convex combos of best candidates
    if len(best_candidates) >= 2 and time.time() < deadline:
        print(f"[{time.time()-start_time:6.2f}s] Mixing top candidates")
        mix_best_seq = best_seq
        mix_best_score = best_score
        # pairwise linear interpolation
        for i in range(len(best_candidates)):
            for j in range(i+1, len(best_candidates)):
                seq_i = best_candidates[i][1]
                seq_j = best_candidates[j][1]
                for t in np.linspace(0.0, 1.0, 21):
                    seq_mix = (1-t)*seq_i + t*seq_j
                    # project onto simplex (ensures sum=1)
                    seq_mix = _project_simplex(seq_mix)
                    score = evaluate_sequence(seq_mix.tolist())
                    if score < mix_best_score:
                        mix_best_score = score
                        mix_best_seq = seq_mix.copy()
                        print(f"[{time.time()-start_time:6.2f}s] Mix {i},{j} t={t:.2f} score={score:.6f}")
        # try mixing three top random combos
        if len(best_candidates) >= 3 and time.time() < deadline:
            # generate random convex combos
            for _ in range(30):
                idxs = np.random.choice(len(best_candidates), 3, replace=False)
                w = np.random.random(3)
                w = w / w.sum()
                mix_seq = sum(w[k] * best_candidates[idxs[k]][1] for k in range(3))
                mix_seq = _project_simplex(mix_seq)
                score = evaluate_sequence(mix_seq.tolist())
                if score < mix_best_score:
                    mix_best_score = score
                    mix_best_seq = mix_seq.copy()

            best_seq = mix_best_seq.copy()

    # final sanitisation

    total = best_seq.sum()

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time()-start_time:6.2f}s] final reported score = {final_score:.6f}")
    return [float(x) for x in best_seq.tolist()]

"""
Propose a low‑score step‑function using a multi‑start, multi‑phase
optimiser.

Main ideas
----------
1. Generate several diverse initial probability vectors (uniform,
   exponential, U‑shape, random Dirichlet, and the provided
   ``height_sequence_1`` if any).

2. For each start:
   * **Adam phase** – optimise a smoothed log‑softmax version of the
     objective (log‑sum‑exp of the convolution).  This quickly drives
     the iterate into a good basin.
   * **Exponentiated‑gradient (sub‑gradient) descent** on the true
     max‑convolution objective.  The update is multiplicative, stays
     non‑negative and a Euclidean projection onto the simplex is applied
     each step.
   * **Local hill‑climbing** – deterministic mass transfers that
     specifically target the current peak of the convolution.
   * **Simulated annealing** – a long stochastic mass‑transfer phase
     that refines the solution.

3. Keep the best five candidates across all starts.

4. **Mixing step** – try convex combinations of the top candidates
   (pairwise linear interpolation and a few random three‑way mixes).
   Because the objective is convex in the coefficients, mixing can
   strictly improve the bound.

5. If time remains, a final annealing on the best mixed vector is run.

All operations respect the global 1000 s budget.  The final sequence is
normalised to sum to 1, clipped to the allowed range, and returned as a
list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


# ----------------------------------------------------------------------
# Gradient / sub‑gradient utilities for the max‑convolution objective
# ----------------------------------------------------------------------


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·conv_k) w.r.t. a.
    Returns a vector of shape (n,).
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                       # soft‑max weights
    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    # cross‑correlation a ⋆ active
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray,
                deadline: float,
                start_time: float,
                fft_len: int) -> np.ndarray:
    """
    Adam maximisation on the smooth log‑sum‑exp approximation of the
    max‑convolution.  Returns the best iterate (probability vector) seen.
    """
    y = np.log(np.maximum(a0, 1e-16))        # start from logits
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # smoothing schedule – start very smooth and finish sharp
    alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 1500          # fewer iterations – we have many starts

    best_seq = a0.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    for alpha in alpha_sched:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq
            a = _softmax_from_logits(y)

            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * a.shape[0] * f_val

            # gradient w.r.t. a
            grad_a = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * a.shape[0])

            # chain rule to logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if i % 300 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                          f"iter={i:,}, score={sc:.6f}")

    return best_seq


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Returns best vector seen and its corresponding objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    cur_conv = _conv_self(a)
    best_score = 2.0 * n * cur_conv.max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # sub‑gradient of the max (active set)
        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        # multiplicative update – stay non‑negative
        step = 0.8 / math.sqrt(it + 1)          # slightly larger initial step
        a = a * np.exp(-step * grad)

        # re‑normalise onto simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 4000) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy mass‑transfer moves that aim to lower the current
    maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]               # one side of the biggest contribution

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]  # move to smallest coefficient

        eps = a[src] * 0.35
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Classic simulated‑annealing with mass‑transfer proposals.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move: transfer a random fraction from src to dst
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, "
                  f"score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  The routine follows the multi‑phase
    scheme described in the module docstring and returns the best
    sequence found within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size (n).  Use the length of the supplied
    # ``height_sequence_1`` when available, otherwise fall back to 4800.
    # ------------------------------------------------------------------
    try:
        starter_seq = np.asarray(height_sequence_1, dtype=float)
        if starter_seq.ndim == 1 and starter_seq.size > 0:
            n = starter_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4800
        starter_seq = None

    # ------------------------------------------------------------------
    # Build a collection of diverse initial points.
    # ------------------------------------------------------------------
    starts = []

    if starter_seq is not None:
        starts.append(starter_seq / max(starter_seq.sum(), 1e-12))

    starts.append(np.full(n, 1.0 / n))                     # uniform

    # exponential decay, left‑hand side heavy
    r = 0.985
    exp_vec = np.array([r ** i for i in range(n)], dtype=float)
    starts.append(exp_vec / max(exp_vec.sum(), 1e-12))

    # symmetric U‑shape (large at the edges, small at centre)
    idx = np.arange(n)
    u_shape = 1.0 / (np.minimum(idx + 1, n - idx) + 1.0)
    starts.append(u_shape / max(u_shape.sum(), 1e-12))

    # random Dirichlet draws
    rng = np.random.default_rng(seed + 12345)
    for _ in range(2):
        rand_vec = rng.random(n)
        rand_vec = np.clip(rand_vec, 0, None)
        if rand_vec.sum() == 0:
            rand_vec = np.full(n, 1.0 / n)
        else:
            rand_vec /= rand_vec.sum()
        starts.append(rand_vec)

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    top_candidates = []   # keep the five best sequences seen

    # ------------------------------------------------------------------
    # Optimise each start independently, sharing the global time budget.
    # ------------------------------------------------------------------
    for s_idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{s_idx + 1}/{len(starts)}")
        a = np.maximum(a0, 0.0)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()

        # ----- 1. Brief Adam phase (soft‑max parametrisation) -----
        phase_deadline = min(deadline,
                             time.time() + 0.04 * (deadline - start_time))
        fft_len = _next_pow2(3 * n - 2)
        a = _adam_phase(a, phase_deadline, start_time, fft_len)

        # ----- 2. Long exponentiated‑gradient descent on true max -----
        if time.time() < deadline:
            remaining = deadline - time.time()
            sub_deadline = time.time() + 0.45 * remaining
            a, _ = _subgradient_descent(a, sub_deadline, start_time)

        # ----- 3. Deterministic greedy peak‑reduction -------------
        if time.time() < deadline:
            a, _ = _run_peak_reduction(a, deadline, start_time)

        # ----- 4. Final simulated‑annealing ----------------------
        if time.time() < deadline:
            # keep a fraction of the remaining budget for this start
            remaining = deadline - time.time()
            anneal_deadline = time.time() + min(remaining, 0.25 * (deadline - start_time))
            # scale number of steps with the allocated time
            steps = int(200_000 * (anneal_deadline - time.time()) /
                        (deadline - start_time))
            if steps < 10_000:
                steps = 10_000
            a, _ = _run_annealing(a, anneal_deadline, start_time,
                                  start_temp=0.1, end_temp=1e-6,
                                  steps=steps)

        # ----- Evaluate and possibly update global best -----
        cur_score = evaluate_sequence(a.tolist())
        print(f"[{time.time() - start_time:6.2f}s] End of start {s_idx + 1} score {cur_score:.6f}")
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST *** "
                  f"score={best_score:.6f}")

        # store candidate for later mixing
        top_candidates.append((cur_score, a.copy()))
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:5]

    # ------------------------------------------------------------------
    # Mixing of top candidates (convex combinations)
    # ------------------------------------------------------------------
    if len(top_candidates) >= 2 and time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] Mixing top candidates")
        mix_best_seq = best_seq
        mix_best_score = best_score

        # pairwise linear interpolation
        for i in range(len(top_candidates)):
            for j in range(i + 1, len(top_candidates)):
                seq_i = top_candidates[i][1]
                seq_j = top_candidates[j][1]
                for t in np.linspace(0.0, 1.0, 21):
                    mix = (1.0 - t) * seq_i + t * seq_j
                    mix = _project_simplex(mix)
                    sc = evaluate_sequence(mix.tolist())
                    if sc < mix_best_score:
                        mix_best_score = sc
                        mix_best_seq = mix.copy()
                        print(f"[{time.time() - start_time:6.2f}s] pair mix {i},{j} t={t:.2f} "
                              f"score={sc:.6f}")

        # random three‑way mixes
        if len(top_candidates) >= 3 and time.time() < deadline:
            for _ in range(30):
                idxs = np.random.choice(len(top_candidates), 3, replace=False)
                weights = np.random.random(3)
                weights /= weights.sum()
                mix = (weights[0] * top_candidates[idxs[0]][1] +
                       weights[1] * top_candidates[idxs[1]][1] +
                       weights[2] * top_candidates[idxs[2]][1])
                mix = _project_simplex(mix)
                sc = evaluate_sequence(mix.tolist())
                if sc < mix_best_score:
                    mix_best_score = sc
                    mix_best_seq = mix.copy()
                    print(f"[{time.time() - start_time:6.2f}s] 3‑mix score={sc:.6f}")

        if mix_best_score < best_score:
            best_score = mix_best_score
            best_seq = mix_best_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST AFTER MIXING {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final annealing on the (possibly mixed) best sequence if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 5.0:      # need a few seconds to be worthwhile
            final_deadline = time.time() + remaining
            steps = int(200_000 * remaining / (deadline - start_time))
            if steps < 10_000:
                steps = 10_000
            best_seq, _ = _run_annealing(best_seq, final_deadline, start_time,
                                         start_temp=5e-2, end_temp=1e-7,
                                         steps=steps)

    # ------------------------------------------------------------------
    # Final sanitisation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
