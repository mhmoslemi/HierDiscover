def objective_and_grad(x, n, alpha):
    # compute a = softmax(x)
    max_x = np.max(x)
    ex = np.exp(x - max_x) # stable
    sum_ex = np.sum(ex)
    a = ex / sum_ex
    # self convolution via FFT:
    L = _next_pow2(2 * n - 1)  # for convolution length; but need conv length for gradient also 2n-1.
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2 * n - 1]
    # ensure non-negative small errors: conv[conv < 0] = 0
    # compute smoothed max:
    scaled = alpha * conv
    max_scaled = scaled.max()
    ex_scaled = np.exp(scaled - max_scaled)
    sum_ex_scaled = np.sum(ex_scaled)
    smoothed_max = (max_scaled + np.log(sum_ex_scaled)) / alpha
    # objective f = 2 * n * smoothed_max
    f = 2.0 * n * smoothed_max

    # compute p = softmax of scaled values
    p = ex_scaled / sum_ex_scaled

    # gradient w.r.t a: g_a = 2 * conv_a(p_rev) slice
    # compute convolution via FFT with appropriate length
    L2 = _next_pow2(3 * n)  # for a (len n) and p_rev (len 2n-1)
    Af2 = np.fft.rfft(a, n=L2)
    Pr = np.fft.rfft(p[::-1], n=L2)
    full = np.fft.irfft(Af2 * Pr, n=L2)
    # slice
    start = 2 * n - 2
    g_a = 2.0 * full[start:start + n]
    # gradient w.r.t x:
    dot = np.dot(g_a, a)
    grad = a * (g_a - dot)  # shape n

    return f, grad

def _run_adam(x0, n, alpha, deadline, start_time, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=10**7):
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    t = 0
    best_a = None
    best_obj = np.inf

    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(x, n, alpha)
        # evaluate exact objective every maybe 100 iterations:
        if t % 100 == 0:
            a = softmax(x)
            conv = np.convolve(a, a)
            exact_obj = 2 * n * conv.max()
            if exact_obj < best_obj:
                best_obj = exact_obj
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] alpha {alpha} iter {t:,} exact obj {exact_obj:.6f}")

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

    # return best A and best objective or current a
    if best_a is None:
        best_a = softmax(x)
        conv = np.convolve(best_a, best_a)
        best_obj = 2 * n * conv.max()
    return best_a, best_obj, x

"""
Propose a step‑function (non‑negative heights) that minimizes the
self‑convolution bound used by ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Soft‑max parametrisation** – we represent a probability vector `a`
   as `softmax(x)`.  This guarantees positivity and unit sum without
   explicit projections.

2. **Continuation with Adam optimisation** – we minimise a smooth
   approximation of the max‑convolution using a log‑sum‑exp (LSE)
   surrogate.  The surrogate parameter `alpha` is increased
   progressively (continuation) so that the optimisation first sees a
   very smooth landscape and later approaches the true `max`.
   Adam’s adaptive step‑size handles the non‑convexity well.
   The exact (unsmoothed) objective is evaluated regularly and the
   best iterate is stored.

3. **Targeted refinement** – after the Adam phase we upsample the
   best distribution to the final length (4800) and run a fast
   O(n) local‑move refinement (the same routine used in the reference
   solution) to shave off the last bit of the bound.

All helper functions are defined at the top level, there are no
closures, and the routine respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (same as in the reference code, kept for reuse)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    mx = np.max(x)
    ex = np.exp(x - mx)
    s = np.sum(ex)
    return ex / s


# ----------------------------------------------------------------------
# Objective and its gradient for the smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _objective_and_grad(x: np.ndarray, n: int, alpha: float):
    """
    Returns (f, g) where
        f = 2 * n * smoothed_max( a*a )
        g = gradient of f w.r.t. the unconstrained variable x
    with a = softmax(x).  ``alpha`` is the LSE smoothing parameter.
    """
    # ----- a = softmax(x) -----
    a = _softmax(x)

    # ----- self‑convolution a*a (length 2n-1) -----
    conv_len = 2 * n - 1
    Lc = _next_pow2(conv_len)
    Af = np.fft.rfft(a, n=Lc)
    conv = np.fft.irfft(Af * Af, n=Lc)[:conv_len]

    # ----- smooth approximation of max(conv) -----
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    # stable exponentiation
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    # LSE / alpha = smooth max
    smoothed_max = (max_scaled + np.log(sum_exp)) / alpha
    f = 2.0 * n * smoothed_max

    # ----- softmax weights (gradient of smoothed max w.r.t. conv) -----
    p = exp_scaled / sum_exp                     # shape (2n-1,)

    # ----- gradient w.r.t. a:
    #   g_a = 2 * (a * p_rev)_{segment}
    #   where p_rev = reversed(p), and the needed segment starts at
    #   index (2n-2) of the full convolution a * p_rev.
    p_rev = p[::-1]
    Lg = _next_pow2(3 * n)                       # enough for a (n) + p_rev (2n-1)
    Af_big = np.fft.rfft(a, n=Lg)
    Pr_big = np.fft.rfft(p_rev, n=Lg)
    full = np.fft.irfft(Af_big * Pr_big, n=Lg)   # length Lg
    start = 2 * n - 2
    g_a = 2.0 * full[start:start + n]            # shape (n,)

    # ----- chain rule: x -> a (softmax) -----
    dot = np.dot(g_a, a)                         # scalar = <g_a, a>
    grad_x = a * (g_a - dot)                     # shape (n,)

    return f, grad_x


# ----------------------------------------------------------------------
# Adam optimiser for a given smoothing parameter
# ----------------------------------------------------------------------
def _run_adam(x0: np.ndarray,
              n: int,
              alpha: float,
              deadline: float,
              start_time: float,
              lr: float = 0.5,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              eval_interval: int = 100):
    """
    Runs Adam from the start point ``x0`` until ``deadline``.
    Returns the best probability vector a (sum = 1) observed,
    the corresponding exact objective value, and the final x.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_obj = np.inf
    it = 0

    while time.time() < deadline:
        it += 1
        f, grad = _objective_and_grad(x, n, alpha)

        # occasional exact evaluation (every eval_interval iterations)
        if it % eval_interval == 0:
            a = _softmax(x)
            conv = np.convolve(a, a)                     # exact conv (length 2n-1)
            exact = 2.0 * n * np.max(conv)
            if exact < best_obj:
                best_obj = exact
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"alpha={alpha:.1f} iter={it:,} exact_obj={exact:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        bias_correction1 = 1.0 - beta1 ** it
        bias_correction2 = 1.0 - beta2 ** it
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

    # fallback if no exact evaluation ever improved
    if best_a is None:
        best_a = _softmax(x)
        conv = np.convolve(best_a, best_a)
        best_obj = 2.0 * n * np.max(conv)

    return best_a, best_obj, x


# ----------------------------------------------------------------------
# Targeted refinement (same as reference implementation)
# ----------------------------------------------------------------------
def _run_targeted_refine_optimized(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.05,
                                   min_eps_frac: float = 0.001):
    """Fast O(n) local refinement that strictly reduces the max‑convolution."""
    a = a0.copy()
    conv = np.convolve(a, a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1
        # linearly decay the move‑size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # ---- choose source index (weighted by its contribution) ----
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # ---- choose destination (prefer indices with tiny contribution) ----
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # ---- propose a small mass move ----
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution (exact up to round‑off)
        delta = np.zeros(conv_len, dtype=float)

        # linear part 2*eps*(a_{k-dst} - a_{k-src})
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a

        # quadratic eps² correction (only three entries are affected)
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to avoid drift
        if it % 50000 == 0:
            conv = np.convolve(a, a)
            max_val = conv.max()

    return best_a

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    final_n = 4800           # length of the final construction
    opt_n = 2400             # size used during the smooth Adam phase

    # ------------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # bring it to the optimisation size
            starter_opt = _upsample(starter, opt_n)
            a0 = starter_opt / starter_opt.sum()
            x0 = np.log(a0 + 1e-12)   # softmax inverse
        else:
            raise ValueError
    except Exception:
        # fresh random start
        a0 = np.random.rand(opt_n)
        a0 /= a0.sum()
        x0 = np.log(a0 + 1e-12)

    best_seq_opt = None
    best_obj_opt = np.inf

    # ------------------------------------------------------------------
    # Continuation schedule: start smooth, finish with a very sharp LSE.
    # ------------------------------------------------------------------
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 200.0, 500.0]

    for alpha in alpha_schedule:
        if time.time() >= deadline:
            break

        # allocate a modest slice of time for each continuation step
        remaining = deadline - time.time()
        phase_time = min(30.0, remaining * 0.15)      # at most 30 s or 15 % of remaining
        phase_deadline = time.time() + phase_time

        a_stage, obj_stage, x0 = _run_adam(
            x0,
            opt_n,
            alpha,
            phase_deadline,
            start_time,
            lr=0.5,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            eval_interval=200,
        )

        if obj_stage < best_obj_opt:
            best_obj_opt = obj_stage
            best_seq_opt = a_stage.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"alpha={alpha:.1f} global best obj={best_obj_opt:.6f}")

    # ------------------------------------------------------------------
    # If we never improved (highly unlikely), fall back to uniform.
    # ------------------------------------------------------------------
    if best_seq_opt is None:
        best_seq_opt = np.full(opt_n, 1.0 / opt_n)

    # ------------------------------------------------------------------
    # Upsample to the final length and run a fast targeted refinement.
    # ------------------------------------------------------------------
    seq = _upsample(best_seq_opt, final_n)

    if time.time() < deadline:
        seq = _run_targeted_refine_optimized(
            seq,
            final_n,
            deadline,
            start_time,
            init_eps_frac=0.05,
            min_eps_frac=0.001,
        )

    # ------------------------------------------------------------------
    # Normalise and return as a list of Python floats.
    # ------------------------------------------------------------------
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = np.full(final_n, 1.0 / final_n)

    return [float(v) for v in seq]
