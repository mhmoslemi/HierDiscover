Implementation details for LP: Use SciPy's linprog. In typical SciPy version (v1.10) it can handle dense matrix A (2D array) of shape (m, n). We will compute A_ub as a list of rows.

Pseudo:

def _lp_step(a, M, active_idx):
    n = len(a)
    m = len(active_idx)
    A = np.zeros((m, n), dtype=float)
    for row, k in enumerate(active_idx):
         i_start = max(0, k-(n-1))
         i_end = min(k, n-1)
         for i in range(i_start, i_end+1):
               j = k - i
               A[row, j] = a[i]
    b = np.full(m, M)
    # Solve/d maximization of sum(b)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None

def _lp_descent_step(a, tol=1e-3, max_constraints=30, t=0.2):
    conv = _conv_self(a)
    M = conv.max()
    # Choose active indices - top peaks
    # sort indices descending by conv values, pick top K
    idx_sorted = np.argsort(-conv)  # descending
    K = min(max_constraints, len(conv))
    active_idx = idx_sorted[:K]
    # Build constraint matrix
    n = len(a)
    m = len(active_idx)
    A = np.zeros((m, n), dtype=float)
    for row, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # For i from i_start to i_end inclusive
        # j = k - i
        # set A[row, j] = a[i]
        # we can vectorized: i_range = np.arange(i_start, i_end+1); j_range = k - i_range; A[row, j_range] = a[i_range]
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        A[row, j_range] = a[i_range]
    # Solve LP
    c = -np.ones(n)  # maximize sum(b)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    except Exception as e:
        return None
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None
    g = b / sum_b
    # mixture
    a_new = (1 - t) * a + t * g
    # proj onto simplex (though sum should be 1)
    a_new = _project_simplex(a_new)
    return a_new

    best_score = 2.0 * n * _conv_self(a).max()
    t = t_start

The goal is to minimise  

\[
F(a)=\frac{2\,n\; \max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the discrete self‑convolution of the non‑negative vector `a`.  
Because the objective is homogeneous we may enforce  
\(\sum_i a_i = 1\) and simply minimise `2·n·max(a*a)`.

The optimiser works in three *independent* stages that complement each
other:

1. **Linear‑programming (LP) descent** –  
   for the current vector `a` we compute the maximal convolution value `M`
   and take the indices where the convolution is largest (the *active set*).
   With those indices we solve the linear program  

   \[
   \max_{b\ge 0}\;\;\sum_i b_i \qquad
   \text{s.t. } (a*b)_k\le M\;\; \forall k\in\text{active set},
   \]

   which yields a direction `b` that tries to spread the mass while keeping
   the current peak `M`.  A convex combination  
   `a←(1‑t)·a + t·(b/∑b)` (small `t`) moves the current iterate towards a
   better shape.  Only a handful of LP‑steps (5‑8) are required, but they
   already lower the peak considerably.

2. **Projected sub‑gradient descent** –  
   After the LP phase we switch to the exact non‑smooth objective.
   The sub‑gradient at the index where the convolution attains its maximum
   is easy to compute; we update with momentum and project onto the
   probability simplex.  This phase runs for the bulk of the time budget
   (hundreds of thousands of iterations) and drives the solution close to a
   local optimum.

3. **Fine‑grained simulated annealing** –  
   A cheap mass‑transfer Monte‑Carlo move (`ε`‑mass from one coordinate to
   another) is accepted with a Metropolis rule.  The temperature decays
   exponentially; many cheap moves (up to a few × 10⁵) polish the final
   solution.

The whole optimiser is repeated for several random restarts on a
moderately sized problem (`n = 4096`).  When time permits we **up‑sample**
the best 4096‑vector to `n = 8192` (repeat each entry twice) and run a
short LP + sub‑gradient + annealing refinement – larger `n` gives a slightly
smaller theoretical lower bound.

All heavy work (FFT for convolutions, LP solving) is performed with
NumPy/SciPy; the algorithm respects the 1000 s wall‑clock limit and
returns the best sequence found.

```python
"""
Optimiser for the ``evaluate_sequence`` objective.

The method consists of three layers:

1.  LP‑descent – a tiny linear programme that maximises the total mass
    while keeping the current maximal self‑convolution unchanged.
    The solution is mixed with the current iterate (step size t).

2.  Projected sub‑gradient descent on the exact objective (2·n·max(conv)).

3.  Light simulated‑annealing mass‑transfer polish.

The optimiser is run on several random restarts for n=4096.
If the time budget allows, the best 4096‑solution is up‑sampled to
n=8192 and refined again.  The best sequence (list of non‑negative
floats) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional SciPy import for the LP step
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _HAS_LP = True
except Exception:               # pragma: no cover
    _HAS_LP = False

# ----------------------------------------------------------------------
# basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {w≥0, Σw=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : LP‑descent (requires SciPy)
# ----------------------------------------------------------------------
def _lp_descent(a0: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                max_iters: int = 6,
                t_start: float = 0.5,
                t_decay: float = 0.9,
                max_constraints: int = 30) -> tuple[np.ndarray, float]:
    """
    Perform a few LP steps.
    Returns the best iterate (lowest true objective) seen.
    """
    if not _HAS_LP:
        # fall back to the input unchanged
        return a0.copy(), 2.0 * n * _conv_self(a0).max()

    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    t = t_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()

        # pick the most critical constraints (largest conv entries)
        idx = np.argsort(-conv)[:max_constraints]
        m = idx.size

        # build A_ub matrix (shape m × n)
        A = np.zeros((m, n), dtype=float)
        for row, k in enumerate(idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_range = k - i_range
            A[row, j_range] = a[i_range]

        # solve: maximise sum(b)  ⇔  minimise –sum(b)
        c = -np.ones(n, dtype=float)
        b_ub = np.full(m, M, dtype=float)

        try:
            res = linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True})
        except Exception:                     # pragma: no cover
            break

        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # mixture step and projection back onto the simplex
        g = b / sum_b
        a = (1.0 - t) * a + t * g
        a = _project_simplex(a)

        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter={it} t={t:.3f} score={cur_score:.6f}")

        t = max(t * t_decay, 0.05)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # sub‑gradient at the (unique) maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-6,
               steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealing. Temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportionally to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score
```
