import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self(a):
    n = a.shape[0]
    L = _next_pow2(2*n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    return conv

def _subgradient_full(a, conv, rel_tol=1e-12):
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)
    thresh = max_val * (1 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        i = np.arange(lo, hi+1)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g

def _project_simplex(v):
    # same as earlier
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho+1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0/n)
    return w

def _upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    # maintain mass
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0/new_n)
    return up

def _run_annealing_refine(a0, n, deadline, start_time,

"""
Simple multi‑stage optimizer for the “max‑self‑convolution” problem.

1️⃣  Warm‑up – exponentiated‑gradient on a smooth approximation of the max.
2️⃣  Projected sub‑gradient descent on the exact max (as in the reference).
3️⃣  Simulated‑annealing style incremental refinement: mass is shifted from
    indices that heavily contribute to the current maximal convolution entry
    to indices that contribute little.  Moves that increase the max are
    occasionally accepted according to a temperature schedule, allowing the
    search to escape local minima.  The final step is a greedy refinement that
    keeps only improving moves.
The routine respects a 1000 s budget, works for any length (default 4800),
and returns a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _init_arcsine(n: int) -> np.ndarray:
    """Arcsine‑like probability vector (finite at the boundaries)."""
    i = np.arange(n)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient warm‑up
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)  # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)  # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: simulated‑annealing incremental refinement
# ----------------------------------------------------------------------
def _run_annealing_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         eps_start: float = 0.06,
                         eps_end: float = 0.001,
                         init_temp_factor: float = 0.10,
                         final_temp_factor: float = 1e-6,
                         rel_improve: float = 1e-14):
    """
    Tiny mass‑shifts with a Metropolis‑Hastings acceptance rule.
    The temperature decays linearly from `init_temp_factor*max_val`
    to `final_temp_factor*max_val`.  Moves that reduce the maximal
    convolution entry are always accepted; uphill moves are accepted
    with probability exp(-Δ/T).
    """
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    start_refine = time.time()
    total_time = max(deadline - start_refine, 1e-9)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_refine
        progress = elapsed / total_time
        # shrinking epsilon fraction
        eps_frac = max(eps_end, eps_start * (1.0 - progress))
        # linear temperature decay
        temp = max_val * (init_temp_factor * (1.0 - progress) +
                         final_temp_factor * progress)

        # current maximal index of the convolution
        k_max = int(np.argmax(conv))

        # ----- choose source (high contribution to the max bin) -----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ----- choose destination (low contribution to the max bin) -----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                     # never stay in place
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ----- amount to move -------------------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ----- analytical update of the convolution (O(n)) --------------------
        conv_new = conv.copy()
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        delta = new_max - max_val

        # ----- Metropolis acceptance -----------------------------------------
        accept = False
        if delta < -rel_improve:
            accept = True
        else:
            if temp > 0.0:
                # protect against overflow when delta is huge
                prob = math.exp(-delta / temp) if delta > 0 else 1.0
                if rng.random() < prob:
                    accept = True

        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                      f"iter={it:,}, max={best_max:.8f}")

    return best_a


# ----------------------------------------------------------------------
# Phase‑3b: final greedy refinement (strictly improving moves)
# ----------------------------------------------------------------------
def _run_incremental_refine(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           eps_start: float = 0.04,
                           eps_end: float = 0.0005,
                           rel_improve: float = 1e-14):
    """Same as the original incremental routine, but used only at the very end."""
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    start_refine = time.time()
    total_time = max(deadline - start_refine, 1e-9)
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        k_max = int(np.argmax(conv))

        # source selection (weighted by contribution to the current max bin)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # destination selection (prefer indices that hardly affect the max bin)
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        conv_new = conv.copy()
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        if new_max < max_val - rel_improve:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                      f"iter={it:,}, max={best_max:.8f}")

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem dimensions
    # ------------------------------------------------------------------
    sub_n = 2400        # fast intermediate size
    final_n = 4800      # length of the final answer

    # ------------------------------------------------------------------
    # Initialise – try to reuse the global construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = _upsample(starter, sub_n)
        else:
            raise Exception()
    except Exception:
        # fall back to an arcsine‑like distribution (reasonable start)
        a0 = _init_arcsine(sub_n)

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – exponentiated‑gradient warm‑up
    # ------------------------------------------------------------------
    stage1_end = time.time() + min(15.0, 0.07 * (deadline - time.time()))
    a_warm, score_warm = _run_exponential_gradient(
        a0,
        sub_n,
        stage1_end,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        steps_per_alpha=8000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: "
              f"score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (most of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        stage2_end = time.time() + max(30.0, 0.65 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            sub_n,
            stage2_end,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: "
                  f"score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement on the intermediate size
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_anneal = _run_annealing_refine(
            best_seq,
            sub_n,
            deadline,
            start_time,
            eps_start=0.06,
            eps_end=0.001,
            init_temp_factor=0.10,
            final_temp_factor=1e-6,
        )
        conv = _conv_self(a_anneal)
        score_anneal = 2.0 * sub_n * conv.max()
        if score_anneal < best_score:
            best_score = score_anneal
            best_seq = a_anneal.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"score={score_anneal:.6f}")

    # ------------------------------------------------------------------
    # Upsample to the final resolution and a final greedy polishing pass
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_full = _upsample(best_seq, final_n)

        # a short greedy polishing (strictly improving moves)
        a_final = _run_incremental_refine(
            a_full,
            final_n,
            deadline,
            start_time,
            eps_start=0.03,
            eps_end=0.0005,
        )
        conv_final = _conv_self(a_final)
        score_final = 2.0 * final_n * conv_final.max()
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final greedy improve: "
                  f"score={score_final:.6f}")
        else:
            # if the greedy pass was not beneficial, keep the upsampled version
            best_seq = a_full

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # optional diagnostic output
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] final evaluate_sequence: {val:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
