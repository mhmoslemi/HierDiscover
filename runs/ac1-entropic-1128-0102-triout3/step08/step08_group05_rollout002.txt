    rng = np.random.default_rng()
    cur = a.copy()

    if time.time() < deadline:

"""
Simple multi‑phase optimiser for the max‑self‑convolution objective.

1. Initialise the probability vector (sum = 1) – use the historic best
   construction if available, otherwise uniform.
2. Run a short Adam optimisation on a smooth log‑sum‑exp surrogate of
   the max‑convolution.  This quickly finds a good “shape”.
3. Upsample (if needed) to a larger length and perform a long simulated‑
   annealing (mass‑transfer) phase.  Mass is moved from a source index
   (chosen proportionally to its current weight) to a random destination.
   Moves are accepted with a temperature schedule that drives the
   optimisation toward lower values of 2·n·max(conv(a)).
4. Return the best vector seen, renormalised to sum = 1 (the evaluation
   function is scale‑invariant).

The whole routine respects the provided time budget (default 1000 s) and
prints occasional progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Simple objective utilities
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv(a)) (sum(a) is assumed = 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.25,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise the surrogate
        Oτ(a) = 2·n·τ·log∑exp(conv(a)/τ)
    using Adam with projection onto the probability simplex.
    Returns the best vector (according to the true objective) and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    tau = tau0

    while time.time() < deadline:
        it += 1

        # keep track of true objective
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve iter={it:,} "
                  f"score={cur_score:.6f} τ={tau:.2e}")

        # ---- soft‑max weights (log‑sum‑exp derivative) ----
        conv = _conv_self(a)
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # shape 2·n‑1, sums to 1

        # ---- gradient of Oτ ----
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)     # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]   # slice of length n
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat

        # projection onto the simplex (non‑negative, sum = 1)
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i≥0, Σx_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Heavy simulated‑annealing (mass‑transfer) phase
# ----------------------------------------------------------------------


def _run_heavy_sa(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  start_temp: float = 2e-2,
                  end_temp: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Very light SA that repeatedly moves a random fraction of mass from a
    source cell (chosen proportionally to its current weight) to a random
    destination.  The temperature decays geometrically with elapsed time.
    Returns the best vector seen and its final objective.
    """
    a = a0.copy()
    cur_score = _objective(a)

    best_seq = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        # temperature based on elapsed fraction
        elapsed_frac = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** elapsed_frac

        # pick source proportionally to its mass, destination uniformly
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # move up to 20 % of the source’s mass
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] SA improve iter={it:,} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (sum≈1).  The routine respects the
    global time budget (default 1000 s) and prints progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.
    # Larger sizes give more freedom but each iteration costs more.
    # We adapt the size to the total budget.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 16384
    elif budget_s >= 500:
        n = 8192
    elif budget_s >= 300:
        n = 4096
    else:
        n = 2048

    # ------------------------------------------------------------------
    # Initialise a probability vector (sum = 1).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            starter = np.asarray(globals()["height_sequence_1"], dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = starter / starter.sum()
                a = _upsample(starter, n)
            else:
                raise ValueError
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[{time.time() - start_time:6.2f}s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smooth log‑sum‑exp surrogate.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate roughly 20‑30 % of the remaining budget
        phase_deadline = time.time() + min(remaining * 0.30, 180.0)
        conv0 = _conv_self(a)
        tau0 = max(conv0.max() * 10.0, 1e-8)
        a, _ = _run_adam_softmax(
            a0=a,
            n=n,
            deadline=phase_deadline,
            start_time=start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9995,
            step_size=0.30,
        )
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] after Adam score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Heavy simulated annealing (mass‑transfer) on the full size.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _run_heavy_sa(
            a0=best_seq,
            n=n,
            deadline=deadline,
            start_time=start_time,
            start_temp=2e-2,
            end_temp=1e-7,
        )
        print(f"[{time.time() - start_time:6.2f}s] after SA score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
