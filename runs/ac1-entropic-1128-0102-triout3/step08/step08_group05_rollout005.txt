# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i≥0, Σx_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline:
        it += 1

        # true objective for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f} τ={tau:.2e}")

        # soft‑max weights over the convolution entries
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)         # stability trick
        p = exp_z / exp_z.sum()           # shape (2·n‑1,)

        # gradient of the surrogate
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)    # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]  # slice that aligns with a
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cooling schedule for τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.45) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))   (a lies on the simplex).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad iter={it:,} "
                  f"score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the true objective
# ----------------------------------------------------------------------


def _run_exponentiated_gradient(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               step0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with the entropy (KL) Bregman divergence.
    Update: a ← a ⊙ exp(−η·grad)  then renormalise.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG iter={it:,} "
                  f"score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        eta = step0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportional to its current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move at most 25 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal iter={it:,} "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling + normalisation helpers
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _random_candidate(n: int) -> np.ndarray:
    """Create a random point on the simplex."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.  The routine respects a 1000 s
    time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin to avoid overshooting the judge's timeout
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise from a previously discovered construction if available.
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Progressive up‑sampling schedule – we continue until time runs out.
    # ------------------------------------------------------------------
    size_schedule = [1024, 2048, 4096, 8192, 16384, 32768]  # 32768 is the final target
    best_seq = None
    best_score = float("inf")

    for n in size_schedule:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise the probability vector for this resolution.
        # --------------------------------------------------------------
        if best_seq is not None:
            a = _upsample(best_seq, n)
        elif starter is not None:
            a = _upsample(starter, n)
        else:
            a = _random_candidate(n)

        # --------------------------------------------------------------
        # Quick random‑restart phase (≈10 % of the time left at this size).
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        rand_phase_deadline = time.time() + 0.10 * remaining
        while time.time() < rand_phase_deadline:
            cand = _random_candidate(n)
            cur_score = 2.0 * n * _conv_self(cand).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] Random init improve (n={n}) "
                      f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – Adam on the smooth surrogate.
        # --------------------------------------------------------------
        if time.time() < deadline:
            # a is already on the simplex
            conv0 = _conv_self(a)
            tau0 = max(conv0.max() * 5.0, 1e-8)   # start a bit hotter than before

            # allocate roughly 40 % of the remaining time to this phase
            remaining = deadline - time.time()
            slice_deadline = time.time() + 0.40 * remaining

            a, _ = _run_adam_softmax(
                a0=a,
                n=n,
                deadline=slice_deadline,
                start_time=start_time,
                tau0=tau0,
                tau_end=1e-14,
                tau_decay=0.9999,
                step_size=0.35,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam improve (n={n}) "
                      f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Projected Euclidean sub‑gradient descent.
        # --------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            sub_deadline = time.time() + 0.30 * remaining   # ≈30 % of what is left
            a, _ = _run_subgradient_descent(
                a0=best_seq,
                n=n,
                deadline=sub_deadline,
                start_time=start_time,
                beta=0.94,
                step0=0.45,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Subgrad improve (n={n}) "
                      f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – Exponentiated‑gradient mirror descent.
        # --------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            eg_deadline = time.time() + 0.20 * remaining   # ≈20 % of remaining time
            a, _ = _run_exponentiated_gradient(
                a0=best_seq,
                n=n,
                deadline=eg_deadline,
                start_time=start_time,
                step0=0.25,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] EG improve (n={n}) "
                      f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – Light simulated‑annealing refinement.
        # --------------------------------------------------------------
        if time.time() < deadline:
            a, _ = _run_annealing(
                a0=best_seq,
                n=n,
                deadline=deadline,
                start_time=start_time,
                cur_best_score=best_score,
                cur_best_seq=best_seq,
                start_temp=2e-2,
                end_temp=1e-7,
                steps=500_000,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Anneal final improve (n={n}) "
                      f"score={cur_score:.6f}")

        # end of current size – proceed to the next (larger) one

    # ------------------------------------------------------------------
    # Final normalisation and return.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Report the objective as seen by the official evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] Final evaluate_sequence score = {final_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
