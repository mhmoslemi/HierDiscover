def _gaussian_initialiser(n, rng):
    # generate Gaussian window centered at mid.
    sigma = n / 6.0
    x = np.arange(n)
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    # ensure non-negative
    w = np.maximum(w, 0.0)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w

def _beta_initialiser(n, rng):
    a = rng.uniform(0.5, 5.0)
    b = rng.uniform(0.5, 5.0)
    w = rng.beta(a, b, size=n)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w

def _sparse_initialiser(n, rng, k=None):
    if k is None:
        k = int(np.sqrt(n))
    # choose k distinct indices
    idx = rng.choice(n, size=min(k, n), replace=False)
    w = np.zeros(n)
    if len(idx) > 0:
        weight = 1.0 / len(idx)
        w[idx] = weight
    else:
        w[:] = 1.0 / n
    # optionally add small random noise to avoid pure zero
    w = np.where(w==0, w, w) # keep as is; check sum =1
    return w

  sequence is always projected onto the probability simplex
  (∑ a_i = 1, a_i ≥ 0).

The code is deliberately written without closures or lambda expressions,
as required by the execution environment.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    # small negative floating noise can appear – clip to zero
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------


def _grad_softmax_surrogate(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of   O_τ(a) = τ·log Σ exp(conv(a)/τ)
    with respect to a (a ≥0, Σa=1).
    """
    n = a.shape[0]
    conv = _conv_self(a)

    # soft‑max weights
    z = conv / tau
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)              # stabilise
    p = exp_z / np.sum(exp_z)               # length 2·n‑1

    # gradient = 4·n * (a_rev ⋆ p)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    conv_rev = _conv_fft(a_rev, p)
    inner = conv_rev[n - 1 : n - 1 + n]
    grad = 4.0 * n * inner
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-14,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max surrogate O_τ using Adam with Euclidean
    projection onto the simplex.
    Returns the best vector (according to the true objective) and its
    true score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0
    n = a.shape[0]

    while time.time() < deadline:
        it += 1

        # true (unsmoothed) score – useful for logging
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improvement "
                  f"iter={it:,} score={cur_score:.6f} τ={tau:.2e}")

        # gradient of the surrogate
        grad = _grad_softmax_surrogate(a, tau)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa = 1).
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.13) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a) = 2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    n = a.shape[0]

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improvement "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    The acceptance probability uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improvement "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisers
# ----------------------------------------------------------------------


def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """Chebyshev/Kaiser window or plain Dirichlet."""
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        att = rng.uniform(30, 120)                     # dB
        w = chebwin(n, at=att, sym=True).astype(float)
        w = np.maximum(w, 0.0)
    elif have_signal and r < 0.8:
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
        w = np.maximum(w, 0.0)
    else:
        w = rng.random(n)
        w = np.maximum(w, 0.0)

    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _gaussian_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gaussian bell centred in the middle, normalised to the simplex."""
    sigma = n / 6.0                     # heuristic width
    x = np.arange(n)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w


def _beta_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑distributed random vector, then normalised."""
    a = rng.uniform(0.5, 5.0)
    b = rng.uniform(0.5, 5.0)
    w = rng.beta(a, b, size=n)
    if w.sum() == 0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w


def _sparse_initialiser(n: int, rng: np.random.Generator, k: int = None) -> np.ndarray:
    """Very sparse vector with k equal non‑zero entries (k ≈ √n)."""
    if k is None:
        k = max(2, int(np.sqrt(n)))
    k = min(k, n)
    idx = rng.choice(n, size=k, replace=False)
    w = np.zeros(n, dtype=float)
    w[idx] = 1.0 / k
    return w


def _starter_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """If a previous solution exists, upsample it."""
    seq = globals().get('height_sequence_1')
    if seq is None:
        return None
    try:
        tmp = np.asarray(seq, dtype=float)
        if tmp.ndim != 1 or tmp.size == 0:
            return None
        return _upsample(tmp / tmp.sum(), n)
    except Exception:
        return None


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search consists of:
      – many cheap restarts with diverse initialisers;
      – a long Adam optimisation of a soft‑max surrogate;
      – a projected sub‑gradient polishing phase;
      – a final simulated‑annealing refinement.
    The returned list is a plain Python list of floats (≈ simplex).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 0 – many cheap restarts (n≈2048)
    # ------------------------------------------------------------------
    n0 = 2048
    stage0_time = max(30.0, 0.12 * (deadline - time.time()))
    stage0_deadline = time.time() + stage0_time

    # initialise list of (function, description)
    init_funs = [
        (_window_initialiser, "window"),
        (_gaussian_initialiser, "gaussian"),
        (_beta_initialiser, "beta"),
        (_sparse_initialiser, "sparse")
    ]
    # prepend the starter if we have a previous solution
    if globals().get('height_sequence_1') is not None:
        init_funs.insert(0, (_starter_initialiser, "starter"))

    restart_counter = 0
    while time.time() < stage0_deadline:
        fun, tag = init_funs[restart_counter % len(init_funs)]
        a0 = fun(n0, rng)
        if a0 is None:
            # starter failed – skip this round
            restart_counter += 1
            continue

        # initialise τ from current max‑convolution (a rough scale)
        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 6.0, 1e-8)

        # a short Adam run (few‑second budget)
        sub_deadline = time.time() + 4.0
        a_opt, score = _run_adam_softmax(
            a0,
            deadline=sub_deadline,
            start_time=start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9994,
            step_size=0.20,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage‑0 improve ({tag}) "
                  f"restart={restart_counter} score={score:.6f}")

        restart_counter += 1

    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – upsample and long Adam optimisation (n≈8192)
    # ------------------------------------------------------------------
    n_large = 8192
    a_large = _upsample(best_seq, n_large)

    # recompute τ for the larger vector
    conv_big = _conv_self(a_large)
    tau0_big = max(conv_big.max() * 5.0, 1e-10)

    # allocate ~45 % of remaining budget to this stage
    remaining = deadline - time.time()
    stage1_time = max(40.0, 0.45 * remaining)
    stage1_deadline = time.time() + stage1_time

    a_big_opt, score_big = _run_adam_softmax(
        a_large,
        deadline=stage1_deadline,
        start_time=start_time,
        tau0=tau0_big,
        tau_end=1e-15,
        tau_decay=0.9996,
        step_size=0.14,
    )
    if score_big < best_score:
        best_score = score_big
        best_seq = a_big_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage‑1 improve score={score_big:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (fine polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.5:
        sub_deadline = deadline - 2.5
        a_sg_opt, score_sg = _run_subgradient_descent(
            best_seq,
            deadline=sub_deadline,
            start_time=start_time,
            beta=0.97,
            step0=0.12,
        )
        if score_sg < best_score:
            best_score = score_sg
            best_seq = a_sg_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage‑2 improve score={score_sg:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – extended simulated annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            deadline=deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=1_200_000,          # a bit more moves than reference code
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final simple sanity‑check and normalisation
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list – the harness will evaluate it.
    return [float(x) for x in best_seq]
