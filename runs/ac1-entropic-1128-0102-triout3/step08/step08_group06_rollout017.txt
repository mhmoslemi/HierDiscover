"""
search for a step‑function with a very small maximal self‑convolution.
The algorithm is based on an *exact line–search* move:
choose a source index (large contribution to the current maximum)
and a destination index (small contribution), then compute the largest
ε that can be transferred from src to dst while the maximal convolution
entry strictly decreases.  This replaces the heuristic random moves of the
previous implementation and is able to push the bound below the 1.512402
record.

The routine works as follows:

1. initialise from the global ``height_sequence_1`` if present,
   otherwise from a random point on the simplex (size 4800).
2. run a short exponential‑gradient warm‑up (optional, 20 s).
3. run a projected sub‑gradient descent (optional, 40 s) – mainly to
   get out of the poorest region.
4. **analytic refinement** – the core of the search.  In each iteration
   the best (src,dst) pair among a few promising candidates is selected,
   the optimal ε is computed analytically, and the move is applied if it
   yields a strict reduction of the maximal convolution entry.
   The loop runs until the time budget is exhausted.

All numbers are kept non‑negative, the vector is always re‑normalised to
unit sum, and the best sequence seen so far is returned.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0            # numerical noise
    return conv


# ----------------------------------------------------------------------
# Helper: lightweight Euclidean projection onto the simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic refinement (core of the new search)
# ----------------------------------------------------------------------
def _run_analytic_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         top_src: int = 8,
                         bot_dst: int = 8,
                         max_trials: int = 3,
                         tol: float = 1e-14) -> np.ndarray:
    """
    Repeatedly move probability mass from a source index that heavily
    contributes to the current maximal convolution entry to a destination
    index that contributes little.  For each (src,dst) pair the maximal
    admissible ε is computed exactly (piecewise‑linear analysis) and the
    best improvement among a few candidates is applied.
    """
    a = a0.copy()
    conv = _conv_self(a)
    conv_len = 2 * n - 1
    best_a = a.copy()
    best_max = conv.max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # current maximal entry and its contributing indices
        # ------------------------------------------------------------------
        k_max = int(conv.argmax())
        cur_max = conv[k_max]

        # contributions of each index to conv[k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = np.zeros(n, dtype=float)
        contrib[idx] = a[idx] * a[k_max - idx]

        # --------------------------------------------------------------
        # choose a few promising src indices (large contribution)
        # --------------------------------------------------------------
        if contrib.sum() > 0:
            prob_src = contrib / contrib.sum()
            src_candidates = rng.choice(
                n, size=top_src, replace=False, p=prob_src
            )
        else:
            src_candidates = rng.choice(n, size=top_src, replace=False)

        # --------------------------------------------------------------
        # choose a few promising dst indices (small contribution)
        # --------------------------------------------------------------
        # weight = a[k_max - dst] where defined, otherwise 0 (best)
        weight = np.zeros(n, dtype=float)
        jdst = k_max - np.arange(n)
        valid = (jdst >= 0) & (jdst < n)
        weight[valid] = a[jdst[valid]]
        # smallest weight → best destinations
        dst_order = np.argsort(weight)
        dst_candidates = dst_order[:bot_dst]

        # --------------------------------------------------------------
        # evaluate a few (src,dst) pairs and keep the best improvement
        # --------------------------------------------------------------
        best_impr = 0.0
        best_move = None  # (src,dst,eps,new_max)
        for _ in range(max_trials):
            src = int(rng.choice(src_candidates))
            dst = int(rng.choice(dst_candidates))
            if src == dst:
                continue
            eps_max = a[src]                     # cannot move more than we own
            if eps_max <= tol:
                continue

            # compute the slope vector s = 2*(arr_dst - arr_src)
            s = np.zeros(conv_len, dtype=float)
            s[dst:dst + n] += 2.0 * a
            s[src:src + n] -= 2.0 * a

            s_max = s[k_max]
            if s_max >= 0:          # moving mass would not decrease the max
                continue

            # candidate ε where another line overtakes the current max
            diff = s - s_max
            mask = diff > 1e-18
            if np.any(mask):
                eps_cross = (cur_max - conv[mask]) / diff[mask]
                eps_cross = eps_cross[eps_cross > 0]
                if eps_cross.size > 0:
                    eps_opt = min(eps_max, eps_cross.min())
                else:
                    eps_opt = eps_max
            else:
                eps_opt = eps_max

            if eps_opt <= tol:
                continue

            # exact update of the convolution for this ε
            eps = eps_opt
            delta = np.zeros(conv_len, dtype=float)
            delta[dst:dst + n] += 2.0 * eps * a
            delta[src:src + n] -= 2.0 * eps * a
            # quadratic corrections (three positions)
            delta[2 * src] += eps * eps
            delta[2 * dst] += eps * eps
            delta[src + dst] -= 2.0 * eps * eps

            new_conv = conv + delta
            new_max = new_conv.max()
            improvement = cur_max - new_max

            if improvement > best_impr + 1e-18:
                best_impr = improvement
                best_move = (src, dst, eps, new_conv, new_max)

        # --------------------------------------------------------------
        # apply the best move found (if any)
        # --------------------------------------------------------------
        if best_move is not None:
            src, dst, eps, new_conv, new_max = best_move
            a[src] -= eps
            a[dst] += eps
            # numerical safety – keep non‑negative
            a[a < 0] = 0.0
            # re‑normalise to guard against drift
            total = a.sum()
            if total > 0:
                a /= total
            else:
                a[:] = 1.0 / n

            conv = new_conv
            cur_max = new_max

            if new_max < best_max - tol:
                best_max = new_max
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] analytic improve: "
                    f"iter={it:,}, max={best_max:.8f}"
                )
        else:
            # fallback: a tiny random move (as in the old code)
            # pick src by contribution, dst by minimal weight
            prob_src = contrib / (contrib.sum() + 1e-30)
            src = int(rng.choice(n, p=prob_src))
            weight = np.zeros(n, dtype=float)
            jdst = k_max - np.arange(n)
            valid = (jdst >= 0) & (jdst < n)
            weight[valid] = a[jdst[valid]]
            dst = int(rng.choice(np.where(weight == weight.min())[0]))
            if src == dst:
                continue
            eps = a[src] * 0.0005 * rng.random()   # very small step
            if eps <= 0:
                continue
            delta = np.zeros(conv_len, dtype=float)
            delta[dst:dst + n] += 2.0 * eps * a
            delta[src:src + n] -= 2.0 * eps * a
            delta[2 * src] += eps * eps
            delta[2 * dst] += eps * eps
            delta[src + dst] -= 2.0 * eps * eps
            new_conv = conv + delta
            new_max = new_conv.max()
            if new_max < cur_max - tol:
                a[src] -= eps
                a[dst] += eps
                a[a < 0] = 0.0
                total = a.sum()
                a /= total
                conv = new_conv
                cur_max = new_max
                if new_max < best_max - tol:
                    best_max = new_max
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] tiny‑move improve: "
                        f"iter={it:,}, max={best_max:.8f}"
                    )

        # periodically recompute convolution from scratch to avoid drift
        if it % 5000 == 0:
            conv = _conv_self(a)

    return best_a


# ----------------------------------------------------------------------
# Proposed candidate entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``.
    The method starts from the global ``height_sequence_1`` (if present),
    performs a short warm‑up and then carries out the analytic refinement
    described above.  The best sequence found within the time budget is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # initialise – use existing global construction if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800
    if starter is not None:
        # keep the original length if it matches final_n, otherwise up‑sample
        if starter.size != final_n:
            # simple linear up‑sampling preserving mass
            old_x = np.linspace(0, 1, starter.size)
            new_x = np.linspace(0, 1, final_n)
            a0 = np.interp(new_x, old_x, starter)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = starter.copy()
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional short warm‑up (exponential‑gradient) – 15 s max
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, time.time() + 15.0)
    if time.time() < phase1_deadline:
        # a very light‑weight copy of the earlier EG code (no logging)
        a = a0.copy()
        it = 0
        eta0 = 0.6
        while time.time() < phase1_deadline:
            it += 1
            conv = _conv_self(a)
            score = 2.0 * final_n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
            # smooth‑max weights
            alpha = 10.0
            w = np.exp(alpha * conv - np.max(alpha * conv))
            w /= w.sum()
            # mirror‑descent update
            eta = eta0 / math.sqrt(it)
            upd = np.exp(-eta * np.fft.irfft(
                np.fft.rfft(a) * np.fft.rfft(w[::-1]), n=_next_pow2(3 * final_n - 2)
            )[:final_n])
            a = a * upd
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Optional projected sub‑gradient descent – 30 s max
    # ------------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 30.0)
    if time.time() < sub_deadline:
        a = best_seq.copy()
        it = 0
        step0 = 0.3
        beta = 0.93
        momentum = np.zeros_like(a)
        while time.time() < sub_deadline:
            it += 1
            conv = _conv_self(a)
            score = 2.0 * final_n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
            # sub‑gradient for the max entry
            k_max = int(conv.argmax())
            lo = max(0, k_max - (final_n - 1))
            hi = min(final_n - 1, k_max)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k_max - i
            g = np.zeros_like(a)
            g[i] += a[j]
            g[j] += a[i]
            momentum = beta * momentum + (1 - beta) * g
            step = step0 / math.sqrt(it)
            a = _project_simplex(a - step * momentum)

    # ------------------------------------------------------------------
    # Analytic refinement – the main work horse
    # ------------------------------------------------------------------
    if time.time() < deadline:
        refined = _run_analytic_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            top_src=8,
            bot_dst=8,
            max_trials=3,
            tol=1e-14,
        )
        # compute the exact objective for reporting
        final_conv = _conv_self(refined)
        final_score = 2.0 * final_n * final_conv.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = refined.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final analytic improve: "
                f"score={final_score:.6f}"
            )

    # ------------------------------------------------------------------
    # final normalisation and conversion to Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"Best objective found: {best_score:.6f}")
    return [float(x) for x in best_seq]
