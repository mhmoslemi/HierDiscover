def subgrad_optimize_once(a0, max_iters, start_time, deadline):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = 2*n*_conv_self(a).max()
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # active set as before
        active = np.where(conv >= cur_max * (1 - 1e-12))[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            i = np.arange(lo, hi+1)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]
        # descent direction: negative g (since increase leads to higher conv)
        # Actually gradient of max is g, sub gradient for objective f = max_b is g.
        # To decrease f, we step opposite.
        momentum = beta * momentum + (1 - beta) * g
        it += 1
        step = step0 / (it**0.5)
        a = _project_simplex(a - step * momentum)
    return best_a, best_score

"""
Efficient multi‑restart projected sub‑gradient optimizer with
aggressive greedy polishing.

We keep the sequence on the probability simplex (sum = 1) and
symmetrise it after every update.  Several diverse starts (uniform,
edge‑heavy exponentials, Dirichlet draws) are each optimised for a
short budget using a fast sub‑gradient method.  The best candidate is
finally refined by a deterministic peak‑reduction step that moves a
large fraction of the most responsible coefficient towards the
smallest one while strictly decreasing the convolution peak.

All computation respects the 1000 s total budget.  The returned list
contains non‑negative numbers; the evaluator will normalise it again,
but we keep the sum = 1 to simplify the objective.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Start generators
# ----------------------------------------------------------------------


def _uniform_start(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _exponential_start(n: int, r: float = 0.985) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, mirrored to keep symmetry."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    vec = vec / vec.sum()
    return vec


def _edge_heavy_start(n: int, lam: float) -> np.ndarray:
    """Mass concentrated near the two edges: a_i ∝ exp(-lam * min(i, n-1-i))."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)
    a = np.exp(-lam * d)
    a /= a.sum()
    return a


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (single run)
# ----------------------------------------------------------------------


def _subgrad_optimize_one(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 500_000,
                          beta: float = 0.9,
                          step0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Fast sub‑gradient descent on f(a)=max_k (a*a)[k] with simplex projection.
    Returns best vector seen and its score 2·n·max_b.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)

    it = 0
    # tolerance for active set – a tiny relative slack is enough
    rel_tol = 1e-12

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad iter {it:,} improve {cur_score:.6f}")

        # --- active indices (those attaining the peak) ------------
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # gradient of the max‑convolution (sub‑gradient)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]      # diagonal counted twice – harmless

        # Nesterov‑style momentum and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

        # keep the sequence symmetric – the optimal solution is known to be symmetric
        a = (a + a[::-1]) * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy peak‑reduction (aggressive)
# ----------------------------------------------------------------------


def _greedy_local_improve_aggressive(a0: np.ndarray,
                                     deadline: float,
                                     start_time: float,
                                     max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Move a large fraction of the coefficient most responsible for the current
    convolution peak towards the smallest coefficient, accepting only strict
    improvements.  Several decreasing fractions are tried.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = index with smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        improved = False
        for frac in (0.5, 0.25, 0.1, 0.05):
            eps = frac * a[src]
            if eps < 1e-14:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_max = _conv_self(a_prop).max()
            if new_max < cur_max:
                a = a_prop
                best_score = 2.0 * n * new_max
                improved = True
                print(f"[{time.time() - start_time:6.2f}s] greedy improve: src={src} dst={dst} eps={eps:.3e} new_max={new_max:.6f}")
                break
        if not improved:
            # no improving move found – stop
            break

    return a, best_score

        a = (a + a[::-1]) * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy peak‑reduction (aggressive)
# ----------------------------------------------------------------------


def _greedy_local_improve_aggressive(a0: np.ndarray,
                                     deadline: float,
                                     start_time: float,
                                     max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Move a large fraction of the coefficient most responsible for the current
    convolution peak towards the smallest coefficient, accepting only strict
    improvements.  Several decreasing fractions are tried.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        improved = False
        for frac in (0.5, 0.25, 0.1, 0.05):
            eps = frac * a[src]
            if eps < 1e-14:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_max = _conv_self(a_prop).max()
            if new_max < cur_max:
                a = a_prop
                best_score = 2.0 * n * new_max
                improved = True
                print(f"[{time.time() - start_time:6.2f}s] greedy improve: src={src} dst={dst} eps={eps:.3e} new_max={new_max:.6f}")
                break
        if not improved:
            break

    return a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine builds a pool of diverse starts, runs a fast projected
    sub‑gradient optimiser on each, keeps the best candidate and finally
    polishes it with an aggressive greedy peak‑reduction.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # Problem size – moderate to keep each iteration cheap
    n = 2600

    # ------------------------------------------------------------------
    # Build a collection of starts.
    # ------------------------------------------------------------------
    starts = []

    # uniform
    starts.append(_uniform_start(n))

    # exponential decay (mirrored)
    starts.append(_exponential_start(n, r=0.98))

    # edge‑heavy exponentials with several decay rates
    for lam in [0.001, 0.003, 0.008, 0.015, 0.03]:
        starts.append(_edge_heavy_start(n, lam))

    # Dirichlet draws with a range of concentrations
    for conc in [0.2, 0.5, 1.0, 2.0]:
        starts.append(_dirichlet_start(n, conc, rng))

    # a few completely random Dirichlet vectors
    for _ in range(4):
        conc = 0.1 + 2.0 * rng.random()
        starts.append(_dirichlet_start(n, conc, rng))

    # ------------------------------------------------------------------
    # Time split – 70 % on many short sub‑gradient runs, remaining on greedy.
    # ------------------------------------------------------------------
    subgrad_deadline = start_time + 0.70 * budget_s

    best_seq = None
    best_score = float("inf")

    for idx, a0 in enumerate(starts):
        if time.time() >= subgrad_deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{idx + 1} (sub‑grad)")

        cand_seq, cand_score = _subgrad_optimize_one(
            a0, subgrad_deadline, start_time,
            max_iters=250_000, beta=0.9, step0=0.6
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST from start {idx} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy polishing (uses remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] final greedy polishing")
        final_seq, final_score = _greedy_local_improve_aggressive(
            best_seq, deadline, start_time, max_iters=8000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise one final time and return the plain Python list.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform_start(n)

    best_seq = np.clip(best_seq, 0.0, None)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
