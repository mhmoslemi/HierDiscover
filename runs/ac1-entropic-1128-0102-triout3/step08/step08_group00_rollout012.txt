def solve_lp_for_b(f, M, n):
    # f: np.ndarray, length n, sum = sqrt(2n)
    # Build sparse matrix A shape (2n-1, n)
    offsets = np.arange(n)
    data = [np.full(n, f[d]) for d in range(n)]
    A = sparse.diags(data, offsets, shape=(2*n-1, n), format='csr')
    # Constraints: A @ b <= M
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(b)
    bounds = [(0, None)] * n
    # Solve with highs
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    return res.success, res.x if res.success else None
```

assistantfinal**Solution Overview**

The objective to minimise is  

\[
J(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` denotes the linear self‑convolution of the non‑negative vector `a`.  
Because the score is homogeneous, we can keep the sum of the vector fixed
(e.g. `∑a = 1`).  Then `J(a)=2·n·max(a*a)` and the problem reduces to
*minimise the largest convolution entry*.

The algorithm consists of three phases  

1. **Fast low‑dimensional exploration** – random families (Beta,
   exponential, symmetric exponential) generate a huge number of
   candidates in a fraction of a second.  The best candidate is kept as
   a warm‑start for the next phase.

2. **Refinement with a smooth surrogate** – a log‑sum‑exp (soft‑max)
   approximation of the max is minimised by Adam.  The smoothing
   parameter `α` is increased from `0.5` to `500` while the vector is
   projected onto the probability simplex after every step.

3. **Exact polishing** – a projected sub‑gradient descent works on the
   true non‑smooth objective, followed by a light simulated‑annealing
   that swaps tiny amounts of mass between two coordinates.

All phases respect the global time budget (≈ 1000 s).  The routine
returns a plain Python list of non‑negative floats whose sum is 1,
ready for `evaluate_sequence`.

```python
#!/usr/bin/env python
"""
search for a step function with a very small
max‑convolution value (hence a low evaluate_sequence score).

The algorithm has three stages:
  1. Very fast low‑dimensional exploration using a few parametric families
     (Beta‑distribution, exponential, symmetric exponential) – this yields a
     good starting point for the subsequent optimisation.
  2. Adam on a log‑sum‑exp surrogate of max(conv) with a linearly increasing
     smoothing parameter α (0.5 → 500).  After each Adam step the vector is
     projected onto the probability simplex.
  3. Exact polishing: projected sub‑gradient descent on the true objective,
     finished by a cheap simulated‑annealing mass‑transfer polish.

All phases are time‑aware; the best sequence found so far is returned.
"""

from __future__ import annotations
import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (non‑negative, sum = 1).  Implements the O(n) algorithm of
    Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_smooth(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alpha_start: float = 0.5,
    alpha_end: float = 500.0,
    max_iters: int = 20000,
    lr0: float = 0.6,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on   f_α(a) = (1/α)·log Σ exp(α·conv(a,a)).

    α is raised linearly from ``alpha_start`` to ``alpha_end``.
    The best iterate (lowest true objective) is returned.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    # length for the FFT that is used for the smoothed gradient
    L_grad = _next_pow2(3 * n - 2)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # true objective (for bookkeeping)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smoothness parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights over convolution entries
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient of the smoothed objective
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex (keeps sum = 1)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact max – projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # sub‑gradient at the (any) maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 400_000,
) -> Tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer simulated annealing.
    Temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportional to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.

    The strategy:
      * very fast parametric exploration (Beta / exponential families)
      * Adam on a smooth max‑approximation (long run)
      * projected sub‑gradient descent on the true objective
      * final simulated‑annealing polish
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a power‑of‑two length works well with the FFT.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Keep the best sequence found so far (always kept normalised to sum=1)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")   # objective = 2·n·max(conv)

    # --------------------------------------------------------------
    # Phase 0 : ultra‑fast parametric search (≈30 % of the budget)
    # --------------------------------------------------------------
    param_deadline = start_time + 0.30 * (budget_s - 0.5)
    x_lin = np.linspace(0.0, 1.0, n)   # reuse for Beta family
    while time.time() < param_deadline:
        # pick a family at random
        family = np.random.choice(["beta", "exp", "symexp", "uniform"])
        if family == "beta":
            a_par = 10 ** np.random.uniform(-2, 2)   # 0.01 … 100
            b_par = 10 ** np.random.uniform(-2, 2)
            seq = (x_lin ** (a_par - 1)) * ((1.0 - x_lin) ** (b_par - 1))
            seq[np.isnan(seq)] = 0.0
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
        elif family == "exp":
            lam = np.random.uniform(5e-4, 2e-2)
            seq = np.exp(-lam * np.arange(n))
            seq /= seq.sum()
        elif family == "symexp":
            lam = np.random.uniform(5e-4, 2e-2)
            centre = np.random.randint(0, n)
            dist = np.abs(np.arange(n) - centre)
            seq = np.exp(-lam * dist)
            seq /= seq.sum()
        else:   # uniform
            seq = np.full(n, 1.0 / n)

        conv = _conv_self(seq)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Param‑search best={cur_score:.6f}")

    # --------------------------------------------------------------
    # If nothing was found (should never happen) fall back to uniform.
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()

    # --------------------------------------------------------------
    # Phase 1 : Adam on the smooth max surrogate (≈40 % of the budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate most of the remaining time to Adam, leave some for exact phases
    adam_deadline = time.time() + 0.40 * remaining
    a_adam, score_adam = _adam_smooth(
        best_seq,
        n,
        adam_deadline,
        start_time,
        alpha_start=0.5,
        alpha_end=500.0,
        max_iters=20000,
        lr0=0.6,
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] Adam improved to {score_adam:.6f}")

    # --------------------------------------------------------------
    # Phase 2 : projected sub‑gradient descent (≈20 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.20 * (deadline - time.time())
        a_sub, score_sub = _subgradient_descent(
            best_seq,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad improved to {score_sub:.6f}")

    # --------------------------------------------------------------
    # Phase 3 : final simulated‑annealing polish (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=400_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved to {score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise to sum = 1 and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
