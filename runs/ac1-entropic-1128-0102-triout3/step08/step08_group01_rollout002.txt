"""
Minimal‑max‑convolution optimiser – version 2
-------------------------------------------

We try to push the bound below 1.503 by

* 1️⃣ a richer set of initial sequences (uniform + a family of edge‑heavy
  power‑law shapes),
* 2️⃣ a longer Adam phase on a log‑sum‑exp smooth approximation of the
  maximum convolution coefficient,
* 3️⃣ a high‑iteration projected sub‑gradient descent (Euclidean projection)
  together with an optional KL‑mirror descent pass,
* 4️⃣ a more aggressive targeted simulated‑annealing that moves up to 40 %
  of the source mass and runs many more steps,
* 5️⃣ occasional random‑restart refinements if time permits.

The whole optimisation runs under the supplied 1000 s budget and returns a
probability vector (non‑negative, sum = 1) that can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (result length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _logsumexp_smooth_grad(a: np.ndarray, alpha: float, L: int):
    """
    For a given smoothing factor alpha, return
        * conv = a * a  (full self‑convolution)
        * w    = soft‑max(alpha * conv)
        * grad = ∂/∂a [ (1/alpha) log Σ exp(alpha*conv) ]
    The gradient is 2·corr(a, w_rev) restricted to the first n entries.
    """
    conv = _conv_self_fft(a, L)
    # soft‑max (stable)
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # reverse w for the correlation that yields the gradient
    w_rev = w[::-1]

    Af = np.fft.rfft(a, n=L)
    Wrev = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wrev, n=L)[: a.shape[0] + w_rev.size - 1]
    grad = 2.0 * corr[w_rev.size - 1 - np.arange(a.shape[0])]
    return conv, w, grad


# ----------------------------------------------------------------------
# Phase 1 – Adam on the smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 12000):
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best iterate (lowest true score) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    # initial true score
    conv = _conv_self_fft(a, L)
    best_score = 2.0 * n * conv.max()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1

            # true objective (for bookkeeping)
            conv = _conv_self_fft(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # smooth‑max gradient
            _, _, grad = _logsumexp_smooth_grad(a, alpha, L)

            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)          # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # enforce simplex constraints
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent (Euclidean)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    """
    Projected sub‑gradient descent on the true objective
        f(a) = 2·n·max(conv(a,a)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.3
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self_fft(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient at the maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2b – KL‑mirror descent (optional, usually helps fine‑tuning)
# ----------------------------------------------------------------------
def _run_kl_mirror_descent(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_score_initial: float,
                           max_iters: int = 200_000):
    """
    Exponentiated‑gradient (KL‑mirror) descent using the same sub‑gradient
    as _run_subgradient_descent.  Keeps the iterate on the simplex by
    multiplicative updates.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self_fft(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] KL‑mirror improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # step size schedule – larger early, then 1/√t
        eta = 0.15 / math.sqrt(it)

        # multiplicative update (avoid under‑flow)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a[idx] -= eta * grad[idx]          # only affected coords change
        a = np.exp(log_a)
        a = _project_simplex(a)

    return best_a, best_score

def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-8,
                            steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that repeatedly removes mass from the pair of
    indices that contribute most to the current maximal convolution coefficient.
    The move size may be up to 40 % of the source mass, which is more aggressive
    than the previous implementation.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # cooling schedule (exponential)
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]

        # probability proportional to contribution (if any)
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.40
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self_fft(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Misc – generation of edge‑heavy power‑law initialisations
# ----------------------------------------------------------------------
def _powerlaw_initial(n: int, p: float) -> np.ndarray:
    """
    Returns a symmetric probability vector a_i ∝ ((i+1)*(n-i))^{-p},
    i.e. heavy mass near the two ends, light in the centre.
    """
    idx = np.arange(n, dtype=float) + 1.0        # 1 … n
    # product of distances to the two ends (i+1)*(n-i)
    prod = idx * (n - idx + 1.0)
    # avoid division by zero (should never happen)
    a = np.power(prod, -p)
    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot > 0:
        a /= tot
    else:
        a = np.full(n, 1.0 / n)
    return a

    # Random‑restart loop (lightweight) if there is any time left
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        block = min(15.0, time_left / 2.0)          # allocate ≈15 s per restart
        restart_deadline = time.time() + block

        # perturb current best (small Gaussian noise)
        pert = best_seq + 0.018 * np.random.randn(final_n)
        pert = np.maximum(pert, 0.0)
        pert_sum = pert.sum()
        if pert_sum > 0:
            pert /= pert_sum
        else:
            pert = np.full(final_n, 1.0 / final_n)

        # short sub‑gradient run
        pert_seq, pert_score = _run_subgradient_descent(
            pert,
            final_n,
            restart_deadline,
            start_time,
            best_score,
            max_iters=80_000,
        )
        # brief anneal (very few steps)
        if time.time() < deadline:
            pert_seq, pert_score = _run_targeted_annealing(
                pert_seq,
                final_n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-9,
                steps=60_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (just in case) and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] final score = {best_score:.6f}")
    return [float(x) for x in best_seq]
