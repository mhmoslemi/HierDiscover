"""
Optimiser for the ``evaluate_sequence`` objective.

Main ideas
---------
1. **Smooth‐max L‑BFGS pre‑optimisation** –  
   The non‑differentiable max in the objective is replaced by a
   log‑sum‑exp (smooth max) with a large smoothing parameter α.
   The smooth problem is minimised by L‑BFGS‑B on the *softmax*
   parametrisation (a = softmax(x)), guaranteeing positivity and the
   simplex constraint.  A schedule of increasing α (5 → 50 → 200 → 500 → 1000)
   yields increasingly accurate approximations of the true objective.

2. **Projected sub‑gradient descent** –  
   After the smooth phase we switch to the exact objective.
   The sub‑gradient for the currently maximal convolution entry is used,
   combined with momentum and a 1/√t step‑size, and the iterate is
   projected back onto the probability simplex.

3. **Light simulated‑annealing polish** –  
   A cheap mass‑transfer annealing (source → destination) with an
   exponential temperature schedule further removes remaining peaks.

The whole routine works within the 1000 s budget, performs many cheap
random restarts, keeps the best candidate and finishes with a long
refinement on that candidate.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:

    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.

        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smooth‑max objective (softmax parametrisation) – callable for L‑BFGS
# ----------------------------------------------------------------------
class SmoothObjGrad:
    """
    Callable implementing the smoothed objective

        f_α(a) = 2·n· (1/α)·log Σ_k exp(α·conv_k(a,a))

    where a = softmax(x).  Returns both function value and gradient w.r.t.
    the unconstrained vector x.
    """
    def __init__(self, n: int, alpha: float, L_fft: int):
        self.n = n
        self.alpha = alpha
        self.L = L_fft

    def __call__(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        # ---- softmax to obtain a on the simplex -----------------------
        x_max = np.max(x)
        e = np.exp(x - x_max)
        a = e / e.sum()                      # a ≥ 0, Σa = 1

        # ---- convolution a*a -----------------------------------------
        Af = np.fft.rfft(a, n=self.L)
        conv = np.fft.irfft(Af * Af, n=self.L)[: 2 * self.n - 1]

        # ---- log‑sum‑exp (smooth max) --------------------------------
        scaled = self.alpha * conv
        s_max = scaled.max()
        w = np.exp(scaled - s_max)           # unnormalised weights
        w_sum = w.sum()
        smooth_max = (s_max + np.log(w_sum)) / self.alpha
        f = 2.0 * self.n * smooth_max

        # ---- gradient of smooth max w.r.t. a -------------------------
        w_norm = w / w_sum                    # Σ w_norm = 1
        w_rev = w_norm[::-1]

        # correlation a * w_rev (via FFT)
        W_rev = np.fft.rfft(w_rev, n=self.L)
        corr = np.fft.irfft(Af * W_rev, n=self.L)[: self.n + len(w_rev) - 1]
        grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(self.n)]

        # d f / d a = 2·n·grad_a
        grad_a = grad_a * (2.0 * self.n)

        # ---- chain rule through softmax -------------------------------
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        return f, grad_x


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on
        F(a) = 2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used; after each step the
    iterate is projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.4                # a slightly larger initial step than before
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # sub‑gradient for the *unique* maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-6,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing.  At each step a fraction of the mass at a
    source index is moved to a destination index.  Acceptance follows the
    Metropolis rule with an exponential temperature schedule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportional to its mass, a random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    provided ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – power‑of‑two for efficient FFTs
    n = 4096
    L_fft = _next_pow2(3 * n - 2)                  # length required for gradients

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a global starter exists, evaluate it first.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
            cur = 2.0 * n * _conv_self(starter).max()
            best_seq = starter.copy()
            best_score = cur
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Random‑restart loop
    # ------------------------------------------------------------------
    min_trial_time = 90.0          # keep ~30 s for final refinement
    final_margin = 30.0

    # schedule for the smooth‑max parameter α
    alphas = [5.0, 50.0, 200.0, 500.0, 1000.0]
    maxiter_per_stage = 80         # L‑BFGS iterations per α‑stage

    # try to import SciPy – if unavailable we fall back to the original Adam
    try:
        import scipy.optimize as opt
        have_scipy = True
    except Exception:
        have_scipy = False
        print("[info] SciPy not available – will skip L‑BFGS pre‑optimisation.")

    while True:
        remaining = deadline - time.time()
        if remaining < min_trial_time + final_margin:
            break

        trial_deadline = time.time() + min_trial_time

        # ---- random start on the simplex ----
        a0 = np.random.rand(n)
        a0 /= a0.sum()
        # initialise x such that softmax(x) = a0
        x0 = np.log(a0 + 1e-20)

        # ---- Phase‑1 : L‑BFGS on the smooth‑max surrogate ----
        x = x0.copy()
        if have_scipy:
            for alpha in alphas:
                objgrad = SmoothObjGrad(n, alpha, L_fft)
                # respect the overall deadline for the trial
                maxiter = maxiter_per_stage
                try:
                    x, fval, d = opt.fmin_l_bfgs_b(
                        func=objgrad,
                        x0=x,
                        bounds=[(None, None)] * n,
                        maxiter=maxiter,
                        iprint=-1,
                    )
                except Exception as e:
                    print(f"[warning] L‑BFGS failed ({e}); breaking out of α‑schedule.")
                    break
        else:
            # No SciPy – keep the random start unchanged (skip this phase)
            pass

        # convert final x back to a probability vector
        a = np.exp(x - np.max(x))
        a /= a.sum()

        # evaluate true objective
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial L‑BFGS improve: score={cur_score:.6f}")

        # ---- Phase‑2 : Projected sub‑gradient descent ----
        sub_deadline = trial_deadline - 5.0   # keep a few seconds for a short anneal
        a_sub, score_sub = _subgradient_descent(
            a,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial sub‑grad improve: score={score_sub:.6f}")

        # ---- Phase‑3 : quick annealing (still within trial) ----
        a_fin, score_fin = _annealing(
            a_sub,
            n,
            trial_deadline,
            start_time,
            best_score,
            a_sub,
            start_temp=1e-2,
            end_temp=1e-6,
            steps=40_000,
        )
        if score_fin < best_score:
            best_score = score_fin
            best_seq = a_fin.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial anneal improve: score={score_fin:.6f}")

    # ------------------------------------------------------------------
    # Final refinement on the overall best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()

    # long sub‑gradient phase (≈70 % of the remaining time)
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.70
        a_sub, score_sub = _subgradient_descent(
            best_seq,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Final sub‑grad improve: score={score_sub:.6f}")

    # final annealing polish (use whatever time is left)
    if time.time() < deadline:
        a_final, score_final = _annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
