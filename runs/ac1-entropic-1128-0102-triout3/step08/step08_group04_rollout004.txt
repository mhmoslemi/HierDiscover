def propose_candidate(seed=0, budget_s=1000, **kwargs):
    # set random seed
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n = 6000
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        ...
    except NameError:
        starter = None
    if starter is not None:
        a = _upsample(starter, n)
    else:
        # arcsine distribution
        c = np.empty(n, float)
        c[0] = 1.0
        for i in range(1, n):
            c[i] = c[i-1] * (2*i - 1) / (2*i)
        a = c / np.sum(c)
    # ensure positivity
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= a_sum
    conv = _conv_self(a)  # length 2n-1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    # annealing parameters
    total_time = deadline - start
    eps_start = 0.02  # max fraction of source mass per move at start
    eps_end = 0.0005
    T_start = max(1e-3, max_val * 0.1)
    T_end = 1e-9
    it = 0
    accepted = 0

    # precompute list of indices
    indices = np.arange(n)

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / total_time
        # temperature schedule linear decay
        T = T_start * (1 - fraction) + T_end * fraction
        eps_frac = max(eps_end, eps_start * (1 - fraction))

        # pick source index weighted by contribution to max conv
        k_max = int(np.argmax(conv))
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        range_i = np.arange(i_start, i_end + 1, dtype=int)
        range_j = k_max - range_i
        contrib = a[range_i] * a[range_j]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            src = rng.choice(range_i, p=contrib / total_contrib)
        else:
            src = rng.choice(indices, p=a / a.sum())

        if a[src] < 1e-12:
            continue

        # choose destination index - biased to low value of a
        inv_weights = 1.0 / (a + 1e-12)
        inv_weights[src] = 0.0
        dst = rng.choice(indices, p=inv_weights / inv_weights.sum())

        # epsilon to move
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        # propose update
        conv_new = conv.copy()
        conv_new[src:src+n] -= 2.0 * eps * a
        conv_new[dst:dst+n] += 2.0 * eps * a
        new_max = conv_new.max()
        accept = False
        if new_max < max_val:
            accept = True
        else:
            # accept with probability exp(-(new_max-max_val)/T)
            # if T small, this probability near zero.
            if T > 0:
                prob = math.exp(-(new_max - max_val) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            accepted += 1
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start:.2f}s] iter={it} accept new best max={best_max:.8f}")

        # periodic recompute conv from a to reduce error (e.g., every 5000 accepts)
        if accepted % 5000 == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()
    # final best distribution
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0/n)

    return [float(x) for x in best_a]

"""
Propose a step‑function with a low value of ``evaluate_sequence``.

The method builds a candidate distribution using a known
near‑optimal “arcsine” law (coefficients ∝ C(2i,i)/4ⁱ) and then
refines it with a simulated‑annealing / incremental‑move scheme.
Each move transfers a small amount of mass ε from a source index
to a destination index and updates the self‑convolution analytically
(Δconv = ±2 ε a).  A temperature schedule allows occasional
non‑improving moves to escape local minima, while the step size
ε shrinks over time.  All operations are O(n) per iteration, so
millions of moves can be performed inside the 1000 s budget.
The best sequence seen is returned, normalized to sum‑one and
clipped to the domain required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (not circular) self‑convolution a*a using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _arcsine_distribution(n: int) -> np.ndarray:
    """Return the discrete arcsine law c_i = C(2i,i)/4^i, normalised to sum‑one."""
    c = np.empty(n, dtype=float)
    c[0] = 1.0
    for i in range(1, n):
        # recurrence: c_{i} = c_{i-1} * (2(i-1)+1) / (2*i)
        c[i] = c[i - 1] * (2 * i - 1) / (2.0 * i)
    total = c.sum()
    if total == 0:
        return np.full(n, 1.0 / n)
    return c / total


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    bound for ``evaluate_sequence`` than the previous best (≈1.5124).
    The algorithm uses a deterministic start (arcsine law or a provided
    ``height_sequence_1``) followed by a large‑scale simulated‑annealing
    refinement with O(n) updates.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safe margin

    # ------------------------------------------------------------------
    # Choose problem size and initial sequence
    # ------------------------------------------------------------------
    n = 6000                                 # larger than the previous 4800
    try:
        # ``height_sequence_1`` may be defined in the global scope.
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a = _upsample(starter, n)
        else:
            raise Exception
    except Exception:
        # fall back to the arcsine distribution (good theoretical start)
        a = _arcsine_distribution(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Pre‑compute the convolution and bookkeeping variables
    # ------------------------------------------------------------------
    conv = _conv_self(a)                     # length 2n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    eps_start = 0.02      # initial fraction of the source mass moved
    eps_end = 0.0005      # final fraction (very fine adjustments)
    T_start = max(1e-3, max_val * 0.1)   # temperature for accepting bad moves
    T_end = 1e-9

    indices = np.arange(n)
    it = 0
    accept_cnt = 0
    RECOMP_EVERY = 5000      # recompute convolution from scratch occasionally

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = elapsed / total_time

        # linear schedules for epsilon and temperature
        eps_frac = max(eps_end, eps_start * (1.0 - frac))
        T = T_start * (1.0 - frac) + T_end * frac

        # ------------------------------------------------------------------
        # 1) pick a source index that contributes most to the current max bin
        # ------------------------------------------------------------------
        k_max = int(np.argmax(conv))

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0.0:
            src = rng.choice(i_range, p=contrib / tot_contrib)
        else:
            # fall back to a probability proportional to a itself
            src = rng.choice(indices, p=a / a.sum())

        if a[src] < 1e-15:
            continue

        # ------------------------------------------------------------------
        # 2) pick a destination index that currently carries little mass
        # ------------------------------------------------------------------
        inv_w = 1.0 / (a + 1e-12)
        inv_w[src] = 0.0                     # never move to the same index
        dst = rng.choice(indices, p=inv_w / inv_w.sum())

        # ------------------------------------------------------------------
        # 3) amount of mass to relocate (a small fraction of the source)
        # ------------------------------------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ------------------------------------------------------------------
        # 4) propose the move: update the convolution analytically (O(n))
        # ------------------------------------------------------------------
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a

        new_max = conv_new.max()

        # ------------------------------------------------------------------
        # 5) decide whether to accept the move
        # ------------------------------------------------------------------
        accept = False
        if new_max < max_val:
            accept = True
        else:
            # Metropolis criterion
            if T > 0.0:
                prob = math.exp(-(new_max - max_val) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            # commit the move
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            accept_cnt += 1

            # record the best solution seen so far
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} "
                      f"new best max={best_max:.8f}")

            # occasional re‑computation to keep numerical errors in check
            if accept_cnt % RECOMP_EVERY == 0:
                conv = _conv_self(a)
                max_val = conv.max()
        # ------------------------------------------------------------------
        # End of one annealing iteration
        # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Normalise the best sequence and return it as a plain Python list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    return [float(v) for v in best_a]
