"""
Minimal‑max‑convolution optimiser (improved)

This implementation follows the three‑stage scheme of the reference
optimizer but adds a few enhancements that usually lower the final
objective a little further:

* a larger and more gradual smoothing schedule for the Adam stage;
* a soft‑max (simplex) projection based on the classic Euclidean projection
  (no closures, no lambdas);
* a longer targeted simulated‑annealing phase with a higher initial
  temperature and a 50 % larger “move‑size” bound;
* cheap random restarts that reuse the best sequence found so far.

All helper functions are top‑level; the entry point is
``propose_candidate`` whose signature matches the evaluation harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the linear self‑convolution a*a using FFT.
    ``L`` must be a power of two ≥ 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (without the safety
    checks).  The sequence must already be non‑negative.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    # The denominator is never zero for a valid simplex vector.
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Stage 1 – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.8,
                        iters_per_alpha: int = 20_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best iterate (lowest objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    L = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1

            # ----- current smoothed max -------------------------------------------------
            conv = _conv_self_fft(a, L)
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # smooth‑max weights (log‑sum‑exp)
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)          # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max ----------------------------------------
            Af = np.fft.rfft(a, n=L)
            W_rev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * W_rev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update ---------------------------------------------------------
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)   # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # stay on the simplex (projection)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective.
    Uses the gradient of the currently maximal convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.3
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self_fft(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient coming from the (unique) maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – Targeted simulated annealing (more aggressive)
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 2e-1,
                            end_temp: float = 1e-9,
                            steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Annealing that repeatedly removes probability mass from the pair of
    indices that contributes most to the current maximal convolution entry
    and deposits it at a randomly chosen destination.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, L)
    score = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate current maximal convolution index and its contributors
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]

        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 40 % of the source mass (more aggressive than the baseline)
        eps = a[src] * rng.random() * 0.40
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate the proposed move
        conv_new = _conv_self_fft(a_prop, L)
        new_score = 2.0 * n * conv_new.max() / (a_prop.sum() ** 2)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    deadline = start_time + max(0.0, budget_s - 0.8)

    # ------------------------------------------------------------------
    # Initialise – use the previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4096
    if starter is not None:
        a0 = np.interp(
            np.linspace(0, 1, final_n),
            np.linspace(0, 1, starter.size),
            starter,
        )
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smooth max (≈15 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(5.0, 0.15 * remaining)

    a_adam, score_adam = _run_adam_smoothing(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alphas=[0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        lr0=0.8,
        iters_per_alpha=20_000,
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Sub‑gradient descent (≈65 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.65 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

        # a quick second sub‑gradient run with a larger initial step (helps escape plateaux)
        if time.time() < deadline:
            extra_deadline = time.time() + min(12.0, deadline - time.time())
            a_sub2, score_sub2 = _run_subgradient_descent(
                best_seq,
                final_n,
                extra_deadline,
                start_time,
                best_score,
                max_iters=200_000,
            )
            if score_sub2 < best_score:
                best_score = score_sub2
                best_seq = a_sub2.copy()
                print(f"[{time.time() - start_time:.2f}s] extra sub‑grad improve: score={score_sub2:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Aggressive annealing (use everything that is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_targeted_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-1,
            end_temp=1e-9,
            steps=300_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Random restarts (short bursts) if any time is left.
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        block = min(28.0, time_left / 2.0)          # ~30 s per restart
        restart_deadline = time.time() + block

        # small perturbation of current best
        pert = best_seq + 0.018 * np.random.randn(final_n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # short sub‑gradient phase
        pert_seq, pert_score = _run_subgradient_descent(
            pert,
            final_n,
            restart_deadline,
            start_time,
            best_score,
            max_iters=120_000,
        )
        # quick annealing
        if time.time() < deadline:
            pert_seq, pert_score = _run_targeted_annealing(
                pert_seq,
                final_n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=1e-1,
                end_temp=1e-9,
                steps=100_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
