def _run_adam_smooth(a0, n, deadline, start_time,
                     beta_sched, lr0=0.01,
                     beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    fft_len = _next_pow2(3 * n - 2)
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve: iter={it:,}, score={cur_score:.6f}")

        beta = beta_sched(it) if callable(beta_sched) else beta_sched
        # smoothing: use log-sum-exp approximation
        scaled = beta * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient of smoothed max w.r.t a
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 / math.sqrt(it) * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # project onto simplex
        a = _project_simplex(a)

        # optional projection to non-negative (already ensured)
    return best_a, best_score

def _run_incremental_refine_anneal(a0, n, deadline, start_time,
                                   eps_start=0.04, eps_end=0.002,
                                   rel_improve=1e-12,
                                   temp_start=0.001, temp_end=1e-5):
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0
    accepted = 0
    recompute_interval = 20000  # recompute full conv after this many accepted moves
    last_recompute = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        # epsilon fraction decreasing
        eps_frac = max(eps_end, eps_start * (1.0 - frac))

        # temperature linearly decreasing
        T = max(temp_end, temp_start * (1.0 - frac))

        # find index of current maximum convolution entry
        k_max = int(np.argmax(conv))

        # source from contributions to max
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # destination biased to low contribution
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # epsilon amount moved
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # compute new conv via analytic update
        conv_new = conv.copy()
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        delta = new_max - max_val

        accept = False
        if delta < -rel_improve:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            accepted += 1
            # track best
            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] inc‑refine improve: iter={it:,}, max={best_max:.8f}")

            # periodic recompute exact conv to avoid drift
            if accepted - last_recompute >= recompute_interval:
                conv = _conv_self(a)
                max_val = conv.max()
                last_recompute = accepted

    return best_a

"""
Propose a step function that minimizes evaluate_sequence.

import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: iter={it:,}, score={cur_score:.6f}")

        scaled = alpha * conv

        w_sum = w.sum()
        if w_sum > 0:

        else:

        w_rev = w[::-1]

def _run_adam_smooth(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     beta_sched,
                     lr0: float = 0.02,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve: iter={it:,}, score={cur_score:.6f}")

        # smoothing parameter (log‑sum‑exp temperature)
        beta = beta_sched(it) if callable(beta_sched) else beta_sched

        scaled = beta * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (lr0 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score

def _run_incremental_refine_anneal(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   eps_start: float = 0.04,
                                   eps_end: float = 0.002,
                                   rel_improve: float = 1e-13,
                                   temp_start: float = 0.001,
                                   temp_end: float = 1e-5):
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0
    accepted = 0
    recompute_interval = 20000
    last_recompute = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time if total_time > 0 else 0.0
        eps_frac = max(eps_end, eps_start * (1.0 - frac))
        T = max(temp_end, temp_start * (1.0 - frac))

        k_max = int(np.argmax(conv))

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # destination with low contribution to the max bin
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        conv_new = conv.copy()
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        delta = new_max - max_val

        accept = False
        if delta < -rel_improve:
            accept = True
        else:
            prob = math.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            accepted += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] inc‑refine improve: iter={it:,}, max={best_max:.8f}")

            if accepted - last_recompute >= recompute_interval:
                conv = _conv_self(a)
                max_val = conv.max()
                last_recompute = accepted

    return best_a

Propose a step function that minimises ``evaluate_sequence``.

The algorithm proceeds in three phases:

2️⃣  Adam optimiser on the same smooth objective, gradually increasing the
    smoothing parameter and projecting onto the probability simplex after
    each step.

3️⃣  Massive local‑refinement: tiny amounts of mass are moved from indices
    that contribute most to the current maximal convolution entry to indices
    that contribute little.  Moves are accepted if they improve the max or
    with a Metropolis probability that cools down (simulated‑annealing).
    The convolution is updated analytically in O(n) per move and recomputed
    exactly periodically to avoid drift.

All phases respect the 1000 s budget and the final vector is normalised
to sum = 1, guaranteeing a non‑negative step function suitable for

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) self‑convolution a*a, length 2·len(a)‑1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ---- amount to move -------------------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ---- analytic convolution update ------------------------------------
        conv_new = conv.copy()
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        delta = new_max - max_val

        # ---- acceptance test ------------------------------------------------
        accept = False
        if delta < -rel_improve:
            accept = True
        else:
            prob = math.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            accepted += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] inc‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

            if accepted - last_recompute >= recompute_interval:
                conv = _conv_self(a)
                max_val = conv.max()
                last_recompute = accepted

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Length of the construction – larger values give a finer grid.
    # ------------------------------------------------------------------
    final_n = 6000

    # ------------------------------------------------------------------
    # Initialise – use the global reference if present.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – warm‑up (exponential gradient)
    # ------------------------------------------------------------------
    stage1_time = min(8.0, 0.04 * (deadline - time.time()))
    stage1_deadline = time.time() + stage1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.4,
        steps_per_alpha=5000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimiser on a smoothed max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        stage2_deadline = time.time() + max(30.0, 0.70 * remaining)

        # smoothing schedule (β grows roughly linearly with iteration)
        beta_sched = lambda it: min(300.0, 0.5 + 0.001 * it)

        a_adam, score_adam = _run_adam_smooth(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            beta_sched=beta_sched,
            lr0=0.02,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – massive incremental local refinement with annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_incremental_refine_anneal(
            best_seq,
            final_n,
            deadline,
            start_time,
            eps_start=0.04,
            eps_end=0.001,
            rel_improve=1e-13,
            temp_start=0.001,
            temp_end=1e-6,
        )
        conv = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return the result
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
