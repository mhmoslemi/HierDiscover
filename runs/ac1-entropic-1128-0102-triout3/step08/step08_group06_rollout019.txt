```python
"""
Hybrid optimizer for minimizing the self-convolution L∞ bound.

The algorithm proceeds in several stages designed to work under the 1000 s
budget:

1. Warm‑up by exponentiated‑gradient minimisation of a log‑sum‑exp
   smooth approximation of the maximum of the self‑convolution.

2. Projected sub‑gradient descent on the exact (non‑smoothed) maximum.

3. An active‑set linear‑programming step.  For the current sequence `a`,
   we collect the indices where the convolution is close to its maximum,
   solve a small LP that finds a second sequence `g` with the same
   L1‑norm as `a` and with `a * g` never exceeding the current maximal
   value, and finally mix `a` and the rescaled `g`.  Repeating this step
   strictly reduces the objective.

4. Fast O(n) local refinement together with a short simulated‑annealing
   pass that moves tiny amounts of probability mass from “bad’’ indices
   to “good’’ ones.

The best sequence found during the whole run is returned, normalised so
that its sum equals one (the evaluation routine automatically rescales
the sequence, but the normalisation keeps the intermediate code cheap).
"""
import time
import math
import numpy as np
import scipy.sparse
import scipy.optimize

# ------------------ low‑level utilities ------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0            # eliminate tiny negatives
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # reference: https://arxiv.org/abs/1101.6081
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ------------------ phase 1: exponentiated gradient ------------------

def _run_exponential_gradient(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 10000):
    """Mirror‑descent on the log‑sum‑exp smooth maximum."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)          # for correlation with w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

        # gradient of the smooth max
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update (mirror‑descent on the simplex)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score

# ------------------ phase 2: sub‑gradient descent ------------------

def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**8):
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    n = a.shape[0]
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        max_val = conv.max()
        # indices whose value is within 1e-12 relative tolerance of the max
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        # exact sub‑gradient for the max – sum of two copies of a shifted
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# ------------------ phase 3: active‑set LP refinement ------------------

def _run_active_lp_refine(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         mix_factor: float = 0.2,
                         rel_tol: float = 0.02,
                         max_lp_iters: int = 30):
    """
    Repeatedly improve the sequence by solving a small linear program.

    For the current a we compute its self‑convolution, pick the indices
    whose value is within `rel_tol` of the current maximum, and solve

        max   sum(g)
        s.t.  g >= 0,
              (a * g)_k <= M   for each selected k,
              sum(g) = 1  (enforced later by scaling).

    The solution g is rescaled to have the same ℓ₁‑norm as a and mixed
    with a using `mix_factor`.  The procedure is repeated until time runs
    out or no improvement is observed.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_lp_iters:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP phase improve: "
                  f"iter={it}, score={cur_score:.6f}")

        # active set – indices whose value is close to the maximum
        active = np.where(conv >= M * (1.0 - rel_tol))[0]
        if active.size == 0:
            break

        # Build a sparse matrix representing (a * g)_k for each active k
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            idx_i = np.arange(i_low, i_high + 1, dtype=int)
            coeff = a[i_low:i_high + 1]
            rows.extend([row_idx] * len(idx_i))
            cols.extend(idx_i.tolist())
            data.extend(coeff.tolist())

        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(active.size, n))
        b = np.full(active.size, M)

        # Linear program: maximize sum(g)  <=> minimize -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n

        # Use the high‑performance HiGHS solver that ships with SciPy
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                    method='highs', options={'presolve': True})

        if not res.success:
            # LP failed – fall back to a tiny random perturbation and continue
            print(f"[{time.time() - start_time:.2f}s] LP failed ({res.message})")
            # small random move to escape possible degenerate situation
            eps = 1e-6
            src = np.random.randint(n)
            dst = (src + np.random.randint(1, n)) % n
            a[src] = max(0.0, a[src] - eps)
            a[dst] += eps
            a = _project_simplex(a)
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            # degenerate solution – skip this iteration
            continue

        # Rescale g to have the same ℓ₁‑norm as a (which is 1 after normalisation)
        scale = a.sum() / sum_g
        g_scaled = g * scale

        # Mix the two sequences
        a = (1.0 - mix_factor) * a + mix_factor * g_scaled
        # re‑project onto the simplex to counteract numeric drift
        a = _project_simplex(a)

    return best_a, best_score

# ------------------ phase 4: targeted refinement + annealing ------------------

def _run_targeted_refine_and_anneal(a0: np.ndarray,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.05,
                                   min_eps_frac: float = 0.001,
                                   anneal_temp0: float = 1e-3,
                                   anneal_temp_end: float = 1e-6):
    """
    Fast O(n) refinement (same idea as the original `refine_opt`) followed
    by a short simulated‑annealing pass.  The annealing stage occasionally
    accepts non‑improving moves with a temperature that decays linearly,
    helping to escape tiny local plateaus.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    # --- deterministic refinement (same spirit as previous code) ---
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # source – weighted by its contribution to the current max
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # destination – pick index with minimal contribution to the current max
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # update convolution in O(n)
        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # strict improvement required
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recompute to avoid drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

        # break after a few thousand deterministic moves to give room to annealing
        if it >= 30000:
            break

    # --- simulated annealing stage ---------------------------------
    # continue for the remaining budget
    while time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = anneal_temp0 * (1.0 - elapsed / total) + anneal_temp_end * (elapsed / total)

        # pick a random source proportional to a (so that we move a non‑zero amount)
        src = rng.choice(n, p=a / a.sum())
        if a[src] < 1e-12:
            continue

        # pick a random destination (uniform)
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * 0.01 * rng.random()   # small fraction
        if eps <= 0.0:
            continue

        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()
        delta_obj = new_max - max_val

        accept = False
        if delta_obj < -1e-14:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-delta_obj / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                      f"max={best_max:.8f}")

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if present.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800                     # can be altered; larger n increases cost only modestly
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – a short exponential‑gradient warm‑up
    # ------------------------------------------------------------------
    phase1_time = min(20.0, 0.04 * (deadline - time.time()))
    phase1_deadline = time.time() + phase1_time
    a_warm, score_warm = _run_exponential_gradient(
        a0,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0],
        eta0=0.5,
        steps_per_alpha=5000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] after phase 1: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (heavy lifting)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_time = max(30.0, 0.60 * remaining)
        phase2_deadline = time.time() + phase2_time

"""
Hybrid optimizer for minimizing the self‑convolution L∞ bound.

The algorithm runs under a 1000 s budget and consists of four stages:

1. Warm‑up with exponentiated‑gradient descent on a log‑sum‑exp smooth
   approximation of the maximum self‑convolution.

2. Projected sub‑gradient descent on the exact (non‑smoothed) maximum.

3. An active‑set linear‑programming refinement.  For the current sequence
   `a` we collect indices where the convolution is close to its maximum,
   solve a small LP that finds a second sequence `g` with the same ℓ₁‑norm
   as `a` and with `a * g` never exceeding the current maximal value,
   and finally mix `a` and the scaled `g`.  Repeating this step strictly
   reduces the objective.

4. Fast O(n) targeted refinement followed by a short simulated‑annealing
   pass that moves tiny amounts of probability mass from “bad’’ indices
   to “good’’ ones.

The best sequence discovered during the whole run is returned,
normalised to have ℓ₁‑norm = 1.
"""
import time
import math
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negatives caused by round‑off.
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase 1 – exponentiated gradient on a smooth max
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 10000):
    """Mirror‑descent on the log‑sum‑exp smooth version of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)          # enough for correlation with w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**8):
    """Projected sub‑gradient descent on max(self‑convolution)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    n = a.shape[0]
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – active‑set linear‑programming refinement
# ----------------------------------------------------------------------


def _run_active_lp_refine(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         mix_factor: float = 0.2,
                         rel_tol: float = 0.02,
                         max_lp_iters: int = 30):
    """
    Repeatedly improve the sequence by an LP that maximises the ℓ₁‑norm
    of a secondary sequence `g` while keeping `a * g` under the current
    maximal value `M`.  Only constraints whose convolution value lies
    within `rel_tol` of `M` are kept (active set).  The resulting `g`
    is rescaled to have the same ℓ₁‑norm as `a` and mixed with `a`.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_lp_iters:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP improve: "
                  f"iter={it}, score={cur_score:.6f}")

        active = np.where(conv >= M * (1.0 - rel_tol))[0]
        if active.size == 0:
            break

        # Build sparse matrix for (a * g)_k, k∈active
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            idx_i = np.arange(i_low, i_high + 1, dtype=int)
            coeff = a[i_low:i_high + 1]
            rows.extend([row_idx] * len(idx_i))
            cols.extend(idx_i.tolist())
            data.extend(coeff.tolist())

        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(active.size, n))
        b = np.full(active.size, M)

        # LP: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                    method='highs', options={'presolve': True})

        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
            # fallback: tiny random move
            eps = 1e-6
            src = np.random.randint(n)
            dst = (src + np.random.randint(1, n)) % n
            a[src] = max(0.0, a[src] - eps)
            a[dst] += eps
            a = _project_simplex(a)
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            continue

        scale = a.sum() / sum_g
        g_scaled = g * scale

        a = (1.0 - mix_factor) * a + mix_factor * g_scaled
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – targeted refinement + simulated annealing
# ----------------------------------------------------------------------


def _run_targeted_refine_and_anneal(a0: np.ndarray,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.05,
                                   min_eps_frac: float = 0.001,
                                   anneal_temp0: float = 1e-3,
                                   anneal_temp_end: float = 1e-6):
    """
    First performs the deterministic O(n) refinement from the reference
    implementation, then switches to a short simulated‑annealing pass that
    occasionally accepts non‑improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    # --------------------------------------------------------------
    # Deterministic refinement (same spirit as reference code)
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # source – weighted by contribution to the current maximal entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        tot = contrib.sum()
        prob_src = contrib / tot if tot > 0 else a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # destination – minimise its contribution to the maximal entry
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

        # give the annealer some time
        if it >= 30000:
            break

    # --------------------------------------------------------------
    # Simulated‑annealing stage
    # --------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = anneal_temp0 * (1.0 - elapsed / total) + anneal_temp_end * (elapsed / total)

        src = rng.choice(n, p=a / a.sum())
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * 0.01 * rng.random()
        if eps <= 0:
            continue

        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()
        delta_obj = new_max - max_val

        accept = False
        if delta_obj < -1e-14:
            accept = True
        else:
            prob = math.exp(-delta_obj / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                      f"max={best_max:.8f}")

    return best_a
