"""
Improved hybrid optimiser for the step‑function bound.

The optimiser builds on the previous multi‑phase approach but makes
the following changes that were observed to give a lower final
objective:

* Longer Adam (soft‑max) phase – more iterations per smoothing level.
* Larger step‑size in the exponentiated‑gradient (sub‑gradient) phase.
* Enforced symmetry a_i = a_{n-1-i} after every update (the optimal
  solution is known to be symmetric, this reduces the search space).
* Slightly more time allocated to the sub‑gradient descent (the most
  effective phase) and a modest final annealing.
* Simple peak‑reduction and annealing phases are kept, but they also
  respect the symmetry constraint.

All steps respect a global 1000 s budget.  The function returns a list
of non‑negative heights (already normalised to sum 1) that minimises
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·conv_k) w.r.t. a.
    Returns a vector of shape (n,).
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                       # soft‑max weights
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a_i = a_{n-1-i} and renormalise."""
    a = (a + a[::-1]) * 0.5
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / a.shape[0])
    return a


def _fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a normalised (sum = 1) vector.
    Returns 2·n·max(conv), i.e. the same as evaluate_sequence after
    normalisation.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray, deadline: float, start_time: float,
                fft_len: int) -> np.ndarray:
    """
    Adam optimisation on the smooth log‑sum‑exp approximation of the max.
    Returns the best simplex vector seen.
    """
    y = np.log(np.maximum(a0, 1e-16))  # logits
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.2
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # a *fairly* long schedule
    alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 20000   # increased from 6000

    best_seq = a0.copy()
    best_score = _fast_score(best_seq)

    for alpha in alpha_sched:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq
            a = _softmax_from_logits(y)

            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * a.shape[0] * f_val

            grad_a = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * a.shape[0])

            # chain rule for logits (soft‑max Jacobian)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if _fast_score(a) < best_score:
                best_score = _fast_score(a)
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={best_score:.6f}")

    return best_seq


def _expgrad_subgrad(a0: np.ndarray, deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using the exact max‑subgradient.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = _fast_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: iter={it:,}, score={cur_score:.6f}")

        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        # multiplicative update with a larger constant
        step = 2.0 / math.sqrt(it)          # larger than previous 0.25/√t
        a = a * np.exp(-step * grad)

        # enforce symmetry & simplex
        a = _symmetrize(a)

    return best_a, best_score


def _peak_reduction(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy moves that try to lower the current peak.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # Destination: smallest coefficient distinct from src
        candidates = np.delete(np.arange(n), src)
        dst = candidates[a[candidates].argmin()]

        eps = a[src] * 0.35   # move a bit more than before
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            a = _symmetrize(a)
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    final_score = _fast_score(a)
    return a, final_score


def _annealing(a0: np.ndarray,
               deadline: float,
               start_time: float,
               start_temp: float = 1e-1,
               end_temp: float = 1e-6,
               steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Classic simulated‑annealing with mass‑transfer moves, respecting symmetry.
    """
    a = a0.copy()
    n = a.shape[0]
    score = _fast_score(a)
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _symmetrize(a_prop)

        new_score = _fast_score(a_prop)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.
    The method consists of:
      1. a longer soft‑max Adam warm‑up,
      2. aggressive exponentiated‑gradient descent with symmetry,
      3. greedy peak reduction,
      4. a final annealing pass.
    Several diverse starts are tried; the best sequence found overall
    is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # ensure a tiny safety margin
    deadline = start_time + budget_s - 0.4

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    try:
        starter_seq = np.asarray(height_sequence_1, dtype=float)
        if starter_seq.ndim == 1 and starter_seq.size > 0:
            n = starter_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4800
        starter_seq = None

    # ------------------------------------------------------------------
    # Build a few diverse initial vectors (all normalised to sum 1)
    # ------------------------------------------------------------------
    starts = []

    if starter_seq is not None:
        starts.append(starter_seq / starter_seq.sum())

    starts.append(np.full(n, 1.0 / n))                     # uniform

    # exponential decay, left‑hand side heavy
    r = 0.987
    exp_vec = np.array([r ** i for i in range(n)], dtype=float)
    starts.append(exp_vec / exp_vec.sum())

    # symmetric U‑shape (large at the edges, small in centre)
    idx = np.arange(n)
    u_shape = 1.0 / (np.minimum(idx + 1, n - idx) + 1.0)
    starts.append(u_shape / u_shape.sum())

    # random smooth vector (Gaussian filtered noise)
    rng = np.random.default_rng()
    noise = rng.normal(size=n)
    smooth = np.convolve(noise, np.ones(15) / 15, mode='same')
    smooth = np.clip(smooth, 0, None)
    if smooth.sum() == 0:
        smooth = np.full(n, 1.0 / n)
    else:
        smooth /= smooth.sum()
    starts.append(smooth)

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optimise each start independently, sharing the global time budget.
    # ------------------------------------------------------------------
    for idx_start, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{idx_start + 1}")

        a = np.maximum(a0, 0.0)
        a = _symmetrize(a)               # enforce symmetry from the start

        # ---- 1. Adam warm‑up ---------------------------------------
        phase_deadline = time.time() + min(0.15 * (deadline - start_time), 150.0)
        fft_len = _next_pow2(3 * n - 2)
        a_adam = _adam_phase(a, phase_deadline, start_time, fft_len)

        # ---- 2. Exponentiated‑gradient descent ----------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            grad_deadline = time.time() + 0.6 * remaining
            a_grad, score_grad = _expgrad_subgrad(a_adam, grad_deadline, start_time)
        else:
            a_grad, score_grad = a_adam, _fast_score(a_adam)

        # ---- 3. Deterministic peak reduction -----------------------
        if time.time() < deadline:
            peak_deadline = time.time() + 0.15 * (deadline - time.time())
            a_peak, score_peak = _peak_reduction(a_grad, peak_deadline, start_time)
        else:
            a_peak, score_peak = a_grad, score_grad

        # ---- 4. Final simulated annealing ---------------------------
        if time.time() < deadline:
            a_final, score_final = _annealing(
                a_peak,
                deadline,
                start_time,
                start_temp=1e-1,
                end_temp=1e-6,
                steps=250_000,
            )
        else:
            a_final, score_final = a_peak, score_peak

        # ----------------------------------------------------------------
        # Choose the best stage for this start
        # ----------------------------------------------------------------
        stage_scores = [
            (evaluate_sequence(a_adam.tolist()), a_adam),
            (score_grad, a_grad),
            (score_peak, a_peak),
            (score_final, a_final),
        ]
        local_best_score, local_best_seq = min(stage_scores, key=lambda x: x[0])

        # ----------------------------------------------------------------
        # Update the global best if improved.
        # ----------------------------------------------------------------
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST *** "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
