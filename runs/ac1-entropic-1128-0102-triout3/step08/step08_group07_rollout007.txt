"""
Low‑score non‑negative step function search.

The algorithm works in three phases while respecting the global
budget (default 1000 s):

  1️⃣  Fast exploration on a modest size (≈1 k) using a suite of
      parametric families (uniform, random Dirichlet, Beta‑pdf,
      power‑law, Chebyshev/Kaiser windows).  The best vector is kept.

  2️⃣  Upsample the champion to the target size (8 k) and run a long
      projected Nesterov sub‑gradient descent on the true objective

          F(a) = 2·n·max( a * a )

      (∑a = 1, a ≥ 0).  The step‑size schedule is tuned for a long
      descent and the deadline is honoured.

  3️⃣  Light simulated‑annealing polishing, followed by a few
      deterministic post‑processing tricks (mirroring, convex mixing
      with the uniform vector).  The final vector is projected back
      onto the simplex and returned as a plain Python list.

All helper routines are top‑level (no closures) and only NumPy
(and optionally SciPy for a couple of windows) is used.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Linear‑algebra / projection helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in the exact polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa fixed)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Projected Nesterov sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).

    Returns the best vector seen and its (non‑smoothed) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"it={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    The acceptance probability uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportionally to its current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"it={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    With probability 0.4 we produce a Chebyshev window,
    with probability 0.4 a Kaiser window,
    otherwise a plain random Dirichlet draw.
    """
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        att = rng.uniform(30, 120)                 # attenuation in dB
        w = chebwin(n, at=att, sym=True).astype(float)
        w = np.maximum(w, 0.0)
    elif have_signal and r < 0.8:
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
        w = np.maximum(w, 0.0)
    else:
        w = rng.random(n)
        w = np.maximum(w, 0.0)

    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _beta_window(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑pdf on a regular grid (mid‑points of n bins), normalised to sum = 1.
    Parameters α,β > 0.  Very small α or β create U‑shaped windows.
    """
    # avoid exact 0 or 1 – use mid‑points
    x = (np.arange(n) + 0.5) / n
    # log‑space to stay safe with extreme parameters
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # Clip exponent to avoid overflow
    logw = np.clip(logw, a_min=-700, a_max=700)
    w = np.exp(logw)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _power_window(n: int, p: float, q: float) -> np.ndarray:
    """
    Simple power‑law shape: w_i ∝ (i+1)^p · (n‑i)^q.
    After normalisation the vector lies on the simplex.
    """
    i = np.arange(n, dtype=float)
    w = (i + 1.0) ** p * (n - i) ** q
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _dirichlet_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """Plain Dirichlet draw (parameter 1 for each coordinate)."""
    a = rng.random(n)
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

    # reserve ~30 s for annealing + final post‑processing
    subgrad_deadline = deadline - 30.0
    if subgrad_deadline <= time.time():
        subgrad_deadline = time.time() + 1.0   # fallback, tiny run

    a_opt, best_score = _run_subgradient_descent(
        a_big,
        n_target,
        subgrad_deadline,
        start_time,
        beta=0.96,
        step0=0.15,
    )
    best_seq = a_opt.copy()
    print(f"[{time.time() - start_time:6.2f}s] after sub‑grad best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  simulated‑annealing polishing (light but many moves)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_seq, best_score = _run_annealing(
            best_seq,
            n_target,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-7,
            steps=600_000,
        )
        print(f"[{time.time() - start_time:6.2f}s] after annealing best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  deterministic post‑processing tricks
    # ------------------------------------------------------------------
    # 3a – mirror (average with its reversal)
    if time.time() < deadline:
        mir = (best_seq + best_seq[::-1]) * 0.5
        mir = mir / mir.sum()
        cur = 2.0 * n_target * _conv_self(mir).max()
        if cur < best_score:
            best_seq = mir
            best_score = cur
            print(f"[{time.time() - start_time:6.2f}s] mirror improve score={cur:.6f}")

    # 3b – convex mixing with the uniform vector (coarse line search)
    if time.time() < deadline:
        uniform = np.full(n_target, 1.0 / n_target)
        for t in np.linspace(0.0, 1.0, 11):
            if time.time() >= deadline:
                break
            combo = (1.0 - t) * best_seq + t * uniform
            combo = combo / combo.sum()
            cur = 2.0 * n_target * _conv_self(combo).max()
            if cur < best_score:
                best_seq = combo
                best_score = cur
                print(f"[{time.time() - start_time:6.2f}s] mix t={t:.2f} improve score={cur:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (guard against any tiny drift)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list – the harness will call evaluate_sequence
    return [float(x) for x in best_seq]
