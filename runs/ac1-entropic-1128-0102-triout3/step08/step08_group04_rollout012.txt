The goal is to find a non‑negative sequence `a = (a₀,…,aₙ₋₁)` that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum a_i\bigr)^2},
\]

where `a*a` is the discrete self‑convolution.  
Because the objective is homogeneous we may keep the sequence normalised
(`∑a = 1`).  The problem is non‑convex, therefore a purely deterministic
method is not guaranteed to reach the global optimum.  The algorithm below
combines three complementary heuristics:

1. **Exponentiated‑gradient warm‑up** – optimisation of a smooth
   approximation of the max‑operator (log‑sum‑exp).  This quickly produces a
   well‑shaped distribution.

2. **Projected sub‑gradient descent** – directly minimises the exact
   `max(a*a)` while staying on the probability simplex.

3. **Large‑scale local search** – many cheap “mass‑shift’’ moves are tried.
   Each move transfers a fraction of the mass from an index that contributes
   heavily to the current maximal convolution entry to an index that contributes
   little.  The convolution is updated analytically in **O(n)**, so millions of
   moves can be performed within the time budget.  Moves are accepted if they
   do **not increase** the maximal convolution entry (or with a small
   probability if they increase it – a simple simulated‑annealing scheme).

The three stages run sequentially, each using the remaining time budget.
The best sequence found so far is kept and finally returned as a plain list
of floats.

The code is fully self‑contained, uses only `numpy` and the supplied
`evaluate_sequence` function, and respects the 1000 s time limit.

```python
"""
search for a step‑function with a low self‑convolution peak.

Algorithm
---------
1️⃣  Warm‑up with exponentiated‑gradient on the smooth log‑sum‑exp
    approximation of max(a*a).  This is fast and gives a decent starting
    distribution.

2️⃣  Projected sub‑gradient descent on the exact max; the sub‑gradient is
    built from all entries that are (almost) maximal.  The iterate stays on
    the probability simplex via Euclidean projection.

3️⃣  Massive local search.
    –  pick a “source’’ index that contributes heavily to the current maximal
       convolution entry,
    –  pick a “destination’’ index that contributes little,
    –  move a (shrinking) fraction of the source mass,
    –  update the convolution analytically (Δconv[k] = 2·ε·(a_{k‑dst}−a_{k‑src})),
    –  accept the move if the maximal convolution does not increase,
       otherwise accept with a tiny annealing probability.
    Millions of such O(n) moves are performed in the remaining budget,
    dramatically refining the solution.

The best sequence found is normalised to sum 1 and returned as a list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _score(a: np.ndarray) -> float:
    """Return the exact value of evaluate_sequence for a normalised a."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    n = a.shape[0]
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Phase‑3 : massive local search (tiny mass moves)
# ----------------------------------------------------------------------
def _run_local_search(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      eps_start: float = 0.12,
                      eps_end: float = 0.006,
                      temperature_start: float = 0.0005):
    """
    Perform millions of cheap mass‑shifts.
    Moves are accepted if they do not increase the current maximal convolution
    (or with a small annealing probability when they increase).
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    start_refine = time.time()
    total_time = deadline - start_refine
    iteration = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        # a linearly shrinking epsilon‑fraction
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))
        # temperature for occasional uphill moves
        T = max(1e-12, temperature_start * (1.0 - elapsed / total_time))

        # index of the current maximal convolution entry
        k_max = int(conv.argmax())

        # ----- choose source (high contribution to the max entry) -----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            src = rng.choice(i_range, p=contrib / tot_contrib)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ----- choose destination (low contribution to the max entry) -----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                 # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ----- amount of mass to move -----
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # ----- propose new convolution (analytical O(n) update) -----
        conv_new = conv.copy()
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()

        # ----- acceptance test (mostly downhill, sometimes uphill) -----
        accept = False
        if new_max <= max_val + 1e-14:               # non‑increase
            accept = True
        else:
            # Metropolis‑like acceptance for uphill moves
            prob_up = math.exp(-(new_max - max_val) / T)
            if rng.random() < prob_up:
                accept = True

        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            iteration += 1
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] local‑search improve: "
                      f"iter={iteration:,}, max={best_max:.10f}")

    return best_a

        # recompute exact score for the final candidate
        final_score = _score(a_refined)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final local‑search improve: "
                  f"score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return the best sequence found.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # sanity check – the harness will also call evaluate_sequence,
    # but we can quickly ensure we are not returning an empty list.
    return [float(x) for x in best_seq]
```
