"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

The new search improves on the previous implementation by

  random monotone shapes, and a “U‑shaped” seed);

  sub‑gradient phase;

  (larger move fractions and more passes).

All phases respect the global 1000 s budget.  The final vector is
projected onto the probability simplex before being returned.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of f(a) = (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    centre = (n - 1) / 2.0
    x = np.arange(n)
    g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.985) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _u_shape_start(n: int, eps: float = 1e-3) -> np.ndarray:
    """
    A “U‑shaped’’ distribution: a_i ∝ 1/√((i+1)*(n-i)).
    Slight regularisation (eps) avoids gigantic end values.
    """
    i = np.arange(n, dtype=float) + 1.0
    w = 1.0 / np.sqrt(i * (n - i + eps))
    return w / w.sum()


def _monotone_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random monotone decreasing vector (sorted uniforms)."""
    x = rng.random(n)
    x = np.sort(x)[::-1]          # decreasing
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / s
    return x


# ----------------------------------------------------------------------
# Adam optimisation on the smooth max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   global_deadline: float,
                   start_time: float,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the smooth approximation of max_{k}(a*a)[k].
    Returns the best probability vector seen and its (exact) score.
    """
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – go much sharper than before
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 1024.0]
    iters_per_alpha = 4000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= global_deadline:
                return best_a, best_score

            # soft‑max to enforce the simplex
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            conv = _conv_self(a)
            scaled = α * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α                # smooth max
            obj = 2.0 * n * f_val

            # gradient w.r.t. a
            grad_a = _grad_smooth_max(a, conv, α, fft_len) * (2.0 * n)

            # back‑prop through soft‑max (Jacobian)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam step
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={α:.0f}, score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------
def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Standard projected sub‑gradient descent on f(a)=max_k (a*a)[k].
    Returns the best vector encountered and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # active set: all entries within a tiny tolerance of the maximum
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient computation (exact for the active set)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]          # diag counted twice – harmless

        # momentum & step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy peak reduction (more aggressive)
# ----------------------------------------------------------------------
def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Moves a sizable fraction of the coefficient that most contributes to the
    current maximum convolution peak towards the smallest coefficient.
    Accepts only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]

        src = i_idx[contrib.argmax()]               # most responsible index
        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # move a *larger* fraction than before (up to 40 % of src)
        eps = 0.4 * a[src]
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: iter={it}, max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Small perturbation + short sub‑gradient “annealing’’
# ----------------------------------------------------------------------
def _perturb_and_refine(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       rng: np.random.Generator,
                       temp0: float = 1e-3,
                       steps_per_temp: int = 200) -> tuple[np.ndarray, float]:
    """
    Randomly perturbs the current solution and runs a few sub‑gradient steps.
    The temperature decays geometrically; moves that improve are always kept,
    otherwise accepted with probability exp(-Δ/ T).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    T = temp0
    it = 0

    while time.time() < deadline and T > 1e-9:
        # propose a random move: transfer a bit of mass from a random src to dst
        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = min(a[src] * 0.5, rng.random() * a[src] * 0.5)
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # keep it on the simplex (numerical drift)
        a_prop = _project_simplex(a_prop)

        new_score = evaluate_sequence(a_prop.tolist())
        cur_score = evaluate_sequence(a.tolist())
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < math.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] perturb improve: score={new_score:.6f}")

            # a few sub‑gradient steps to polish the perturbed point
            sub_deadline = time.time() + min(0.8, deadline - time.time())
            a, _ = _subgrad_optimize(a, sub_deadline, start_time, beta=0.96, step0=0.15)

        it += 1
        if it % steps_per_temp == 0:
            T *= 0.7   # geometric cooling

    return best_a, best_score


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine consists of an expanded Adam exploration, rigorous sub‑gradient
    polishing, aggressive greedy peak reduction and a final stochastic polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the evaluator
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 4800 gives a good balance between detail and speed.
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # Build a diverse pool of initial vectors.
    # ------------------------------------------------------------------
    starts = []

    # 1) reuse any previously discovered sequence if present
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) uniform
    starts.append(np.full(n, 1.0 / n))

    # 3) exponential decays (different ratios)
    for r in [0.97, 0.985, 0.99]:
        starts.append(_exponential_start(n, r=r))

    # 4) Gaussian bell
    starts.append(_gaussian_start(n))

    # 5) Dirichlet draws with many concentrations
    for conc in [0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # 6) U‑shaped seed (edges heavy)
    starts.append(_u_shape_start(n))

    # 7) Random monotone decreasing starts
    for _ in range(3):
        starts.append(_monotone_start(n, rng))

    # ------------------------------------------------------------------
    # Time split – give more budget to polishing (45 % exploration,
    # 45 % polishing, 10 % final greedy/perturb).
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.45 * budget_s
    polish_deadline = start_time + 0.90 * budget_s

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Exploration phase – Adam on many starts.
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        if time.time() >= explore_deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] Exploration start #{idx + 1}")
        cand_seq, cand_score = _adam_optimize(a0, explore_deadline, start_time)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (explore) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Polishing phase – long projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < polish_deadline:
        print(f"[{time.time() - start_time:6.2f}s] Polishing phase start")
        polished_seq, polished_score = _subgrad_optimize(
            best_seq, polish_deadline, start_time, beta=0.94, step0=0.25
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (polish) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Greedy peak reduction – more aggressive moves.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] Greedy peak‑reduction start")
        gre_seq, gre_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=8000
        )
        if gre_score < best_score:
            best_score = gre_score
            best_seq = gre_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (greedy) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final stochastic perturb‑and‑refine (simulated annealing style).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] Perturb‑and‑refine start")
        pert_seq, pert_score = _perturb_and_refine(
            best_seq, deadline, start_time, rng, temp0=5e-3, steps_per_temp=300
        )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (final) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and reporting.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
