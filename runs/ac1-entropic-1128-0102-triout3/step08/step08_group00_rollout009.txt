    best_a = a.copy()

    L = next_pow2(3*n) for gradient conv?
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute conv and true objective
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

            best_score = cur_score
            best_a = a.copy()
            print(f"[...] expgrad iter{it} score {cur_score}")

        # compute smoothing factor
        alpha = alpha_start * (alpha_end/alpha_start) ** (it / max_iters)  # geometric
        s = alpha * conv
        max_s = s.max()
        w = np.exp(s - max_s)
        w /= w.sum()
        # gradient = 2 * conv of w_rev with a
        w_rev = w[::-1]
        Lgrad = _next_pow2(3*n-2)
        Af = np.fft.rfft(a, n=Lgrad)
        Wrev = np.fft.rfft(w_rev, n=Lgrad)
        corr = np.fft.irfft(Af * Wrev, n=Lgrad)[: n+len(w_rev)-1]
        grad = 2.0 * corr[len(w_rev)-1 - np.arange(n)]
        # exponentiated gradient update
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        # renormalize to simplex
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full_like(a, 1.0/n)
        else:
            a = a / a_sum
    return best_a, best_score

"""
Propose a step‑function with a low value of evaluate_sequence().

The optimiser works in several complementary stages:

1. **Exponential‑gradient smoothing** – minimise a log‑sum‑exp
   approximation of max(conv(a,a)) while staying on the probability
   simplex.  Multiplicative updates keep the vector non‑negative
   automatically and tend to spread the mass more evenly.

2. **Reordering heuristics** – after each major phase the current vector
   is re‑arranged (descending order and an alternating “large‑small”
   placement).  Large entries are deliberately paired with small ones,
   which empirically reduces the peak of the self‑convolution.

3. **Projected sub‑gradient descent** – a classic sub‑gradient method on
   the exact objective 2·n·max(conv).  Momentum and a 1/√t step size are
   used, and the iterate is projected onto the simplex.

4. **Guided simulated annealing** – tiny mass‑transfer moves are
   performed; source indices are chosen proportionally to the current
   mass while destinations are sampled uniformly.  Moves are accepted
   according to a temperature schedule that slowly cools.

Multiple random restarts are performed within the allotted time budget.
The best sequence found (lowest objective) is returned as a plain Python
list of non‑negative floats (the harness will feed it to evaluate_sequence()).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (full mode) using an FFT of appropriate length.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {w≥0, sum(w)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of evaluate_sequence() for a (ignoring the outer caps)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self(a)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Reordering heuristics
# ----------------------------------------------------------------------
def _reorder_descending(a: np.ndarray) -> np.ndarray:
    """Return a copy of a with entries sorted in non‑increasing order."""
    return np.sort(a)[::-1]


def _alternating_arrangement(v: np.ndarray) -> np.ndarray:
    """
    Place the largest value at the leftmost position, the second largest
    at the rightmost, the third at the next left, etc.
    """
    n = v.shape[0]
    a = np.empty_like(v)
    left, right = 0, n - 1
    i = 0
    while left <= right:
        if i % 2 == 0:
            a[left] = v[i]
            left += 1
        else:
            a[right] = v[i]
            right -= 1
        i += 1
    return a


def _try_reorderings(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """
    Test a few cheap permutations of a (descending, alternating) and keep
    the one with the smallest exact objective.
    """
    best_a = a
    best_score = _objective(a, n)

    # descending order
    a_desc = _reorder_descending(a)
    score_desc = _objective(a_desc, n)
    if score_desc < best_score:
        best_score, best_a = score_desc, a_desc

    # alternating large–small placement
    a_alt = _alternating_arrangement(np.sort(a)[::-1])
    score_alt = _objective(a_alt, n)
    if score_alt < best_score:
        best_score, best_a = score_alt, a_alt

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑1 : Exponential‑gradient on a smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _expgrad_smooth(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    alpha_start: float = 0.5,
                    alpha_end: float = 500.0,
                    max_iters: int = 8000,
                    eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv) by exponentiated
    gradient descent on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a, n)

    L_grad = _next_pow2(3 * n - 2)          # length for the gradient FFTs

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exact objective (for bookkeeping)
        cur_score = _objective(a, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Exp‑grad iter={it:,} score={cur_score:.6f}"
            )

        # geometric schedule for the smoothing parameter α
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max_iters)

        # current self‑convolution
        conv = _conv_self(a)

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_rev = w[::-1]

        # gradient of the smoothed objective:
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update (exponentiated gradient)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact max – projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float,
                         step0: float = 0.5,
                         beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}"
            )

        # index where the maximum is attained
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) (only the active peak contributes)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : guided simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-6,
               max_steps: int = 150_000,
               max_move_frac: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Light mass‑transfer annealing.
    A source index is drawn with probability proportional to its mass,
    a destination uniformly.  A random fraction (up to max_move_frac) of
    the source mass is moved.  Acceptance follows a cooling schedule.
    """
    a = a0.copy()
    score = _objective(a, n)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        src = rng.choice(n, p=a)          # more likely to move mass from larger entries
        dst = rng.integers(n)             # uniform destination
        if src == dst or a[src] < 1e-16:
            continue

        # transfer a random fraction of the source mass
        eps = a[src] * rng.random() * max_move_frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop, n)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a, score = a_prop, new_score

        if score < best_score:
            best_score, best_seq = score, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Locate a non‑negative step‑function (list of heights) that minimises
    evaluate_sequence() under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid overrun
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Choose problem size.
    # 4096 works very fast; if we have a comfortable budget we switch to 8192.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 8192
    else:
        n = 4096

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Global starter (if available) – normalise to the chosen length.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # adapt length
            if starter.size > n:
                starter = starter[:n]
            elif starter.size < n:
                starter = np.concatenate([starter, np.zeros(n - starter.size)])
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = np.full(n, 1.0 / n)
            else:
                starter = starter / starter.sum()
            cur = _objective(starter, n)
            best_seq, best_score = starter.copy(), cur
            print(
                f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}"
            )
    except Exception:
        pass

    # --------------------------------------------------------------
    # Random‑restart loop (short trials)
    # --------------------------------------------------------------
    min_trial_time = 80.0               # seconds reserved for a single trial
    final_margin = 30.0                 # seconds left for the final polish

    while True:
        remaining = deadline - time.time()
        if remaining < min_trial_time + final_margin:
            break

        trial_deadline = time.time() + min_trial_time

        # start from a fresh Dirichlet (uniform) point on the simplex
        a0 = np.random.rand(n)
        a0 /= a0.sum()

        # ---------- Phase‑1 : exponential‑gradient smoothing ----------
        a_smooth, score_smooth = _expgrad_smooth(
            a0,
            n,
            trial_deadline,
            start_time,
            alpha_start=0.5,
            alpha_end=500.0,
            max_iters=8000,
            eta0=0.5,
        )
        if score_smooth < best_score:
            best_score, best_seq = score_smooth, a_smooth.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Trial exp‑grad improve: score={score_smooth:.6f}"
            )

        # ---------- Optional reordering ----------
        a_smooth, score_smooth = _try_reorderings(a_smooth, n)
        if score_smooth < best_score:
            best_score, best_seq = score_smooth, a_smooth.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Reorder after smooth improve: score={score_smooth:.6f}"
            )

        # ---------- Phase‑2 : exact sub‑gradient descent ----------
        sub_deadline = trial_deadline - 5.0   # reserve a few seconds for anneal
        a_sub, score_sub = _subgradient_descent(
            a_smooth,
            n,
            sub_deadline,
            start_time,
            best_score,
            step0=0.5,
            beta=0.9,
        )
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Trial sub‑grad improve: score={score_sub:.6f}"
            )

        # ---------- Reordering after sub‑gradient ----------
        a_sub, score_sub = _try_reorderings(a_sub, n)
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Reorder after sub‑grad improve: score={score_sub:.6f}"
            )

        # ---------- Phase‑3 : light annealing ----------
        a_fin, score_fin = _annealing(
            a_sub,
            n,
            trial_deadline,
            start_time,
            best_score,
            a_sub,
            start_temp=1e-2,
            end_temp=1e-6,
            max_steps=40_000,
        )
        if score_fin < best_score:
            best_score, best_seq = score_fin, a_fin.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Trial anneal improve: score={score_fin:.6f}"
            )

    # ------------------------------------------------------------------
    # Final refinement on the best candidate.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq, n)

    # Long sub‑gradient run
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.70
        a_sub, score_sub = _subgradient_descent(
            best_seq,
            n,
            sub_deadline,
            start_time,
            best_score,
            step0=0.5,
            beta=0.9,
        )
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Final sub‑grad improve: score={score_sub:.6f}"
            )
        # try a reordering once more
        a_sub, score_sub = _try_reorderings(best_seq, n)
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Reorder after final sub‑grad improve: score={score_sub:.6f}"
            )

    # Final annealing polish (use whatever time remains)
    if time.time() < deadline:
        a_final, score_final = _annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            max_steps=200_000,
        )
        if score_final < best_score:
            best_score, best_seq = score_final, a_final.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Final anneal improve: score={score_final:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise (clamp) and return as a plain list of floats.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
