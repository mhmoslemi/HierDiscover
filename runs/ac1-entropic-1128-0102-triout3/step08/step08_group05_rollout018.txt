    import time, numpy as np, scipy.optimize, scipy.sparse as sp

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 2048 (maybe adapt later). Could also choose n based on sqrt(budget) but use base.

    # initial distribution: uniform
    f = np.full(n, 1.0/n)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    iteration = 0
    t = 0.3  # mixing factor
    while time.time() < deadline and iteration < 30:
        iteration += 1
        # compute conv and M
        conv = _conv_self(f)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print("[...]" )

        # Build convolution matrix A
        # f_i values.
        A = np.zeros((2*n-1, n), dtype=float)
        for i in range(n):
            A[i:i+n, i] = f[i]
        # constraints
        b_ub = np.full(2*n-1, M)

        # weight vector: random direction encouraging spread.
        w = np.random.randn(n)  # could also use i-n/2.

        # Set up linprog
        # Minimize -w·g -> c = -w
        c = -w

        # Equality: sum(g)=1
        A_eq = np.ones((1, n))
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n

        # Solve
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if not res.success:
            print("linprog failed.")
            break
        g = res.x
        # Blend
        f = (1 - t) * f + t * g

        # ensure simplex
        f = _project_simplex(f)  # projection onto simplex (though sum may already be 1)

        # optional step size reduction
        t = max(0.01, t * 0.9)

    # optionally run brief gradient descent
    # ... similar to earlier subgradient but limited steps.

    # final evaluation
    best_seq = best_f
    print("Finished best score:", best_score)
    return best_seq.tolist()

"""
search for a low‑self‑convolution step‑function.

The routine combines three ideas that are complementary to the previous
optimiser:

1.  **Linear‑programming mass redistribution** – for a fixed shape `f`,
    the constraints `f * g ≤ max(f*f)` are linear in the unknown `g`.
    Solving a small LP that pushes the mass of `g` to one side while
    respecting these constraints yields a new shape that strictly improves
    the maximum convolution when mixed with the current one.

2.  **Projected sub‑gradient descent** – after a few LP updates a few
    thousand iterations of a standard sub‑gradient method (with momentum)
    are applied to squeeze the bound further.

3.  **Very light simulated‑annealing (mass‑transfer)** – a final short
    SA run escapes tiny local traps.

All vectors are kept on the probability simplex (non‑negative,
∑a=1), which makes `evaluate_sequence` scale‑invariant.
The whole search respects the global time budget (default 1000 s) and
returns a plain Python list of floats.
"""
import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x_i≥0, Σx_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities (linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – needed for the true objective
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Simple projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve (n={n}) "
                f"iter={it:,} score={cur_score:.6f}"
            )

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve (n={n}) "
                f"iter={it:,} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Build the convolution matrix A such that (f * g) = A @ g
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2·n‑1) × n matrix A with A @ g == f * g (linear conv)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # column i of A is f[i] placed on diagonal i
    for i in range(n):
        A[i : i + n, i] = f[i]
    return A

    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine mixes a few LP‑based mass‑redistribution steps with a
    short sub‑gradient descent and a final light simulated annealing.
    It respects the global time budget (default 1000 s) and returns a plain
    Python list of floats (∑a_i ≈ 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise from a previously known construction if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            starter = np.asarray(globals()["height_sequence_1"], dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                f = starter / starter.sum()
            else:
                raise ValueError
        except Exception:
            f = None
    else:
        f = None

    if f is None:
        # start from a random Dirichlet vector (moderately dispersed)
        n = 2048
        alpha = 0.7
        f = np.random.gamma(alpha, 1.0, size=n)
        f /= f.sum()
    else:
        n = f.shape[0]

    best_seq = f.copy()
    best_score = 2.0 * n * _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:6.2f}s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – a handful of LP mass‑redistribution steps.
    # ------------------------------------------------------------------
    lp_steps = 12                     # enough to make a noticeable change
    t = 0.25                          # mixing factor (will decay)
    for step in range(lp_steps):
        if time.time() >= deadline - 5.0:
            break

        conv_f = _conv_self(f)
        M = conv_f.max()

        # build the (2n‑1)×n linear‑convolution matrix for the current f
        A = _build_conv_matrix(f)

        # linear objective: push mass to the right on even steps,
        # to the left on odd steps – this forces an extreme point.
        if step % 2 == 0:
            w = np.arange(n, dtype=float)          # larger indices preferred
        else:
            w = -np.arange(n, dtype=float)         # smaller indices preferred

        # linprog solves   min cᵀx   subject to   A_ub x ≤ b_ub , A_eq x = b_eq .
        c = -w                                     # minimise -w·g  ==  maximise w·g
        b_ub = np.full(2 * n - 1, M)
        A_eq = np.ones((1, n))
        b_eq = np.array([1.0])
        bounds = [(0.0, None)] * n

        # solve the LP
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      A_eq=A_eq,
                      b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            print(f"[{time.time() - start_time:6.2f}s] LP failed at step {step}: {res.message}")
            break

        g = res.x

        # convex combination – this is the “mirror‑descent” style update
        f = (1.0 - t) * f + t * g
        f = _project_simplex(f)                 # numerical safety

        cur_score = 2.0 * n * _conv_self(f).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve step {step} score={cur_score:.6f}")

        # slowly shrink the mixing factor
        t = max(0.02, t * 0.85)

    # ------------------------------------------------------------------
    # Phase 2 – short projected sub‑gradient descent on the true objective.
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        sub_deadline = deadline - 1.5
        f, _ = _run_subgradient_descent(
            a0=f,
            n=n,
            deadline=sub_deadline,
            start_time=start_time,
            beta=0.94,
            step0=0.25,
            max_iter=3_000_000,
        )
        cur_score = 2.0 * n * _conv_self(f).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad final improve score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – light simulated annealing for the last sliver of time.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        f, _ = _run_annealing(
            a0=best_seq,
            n=n,
            deadline=deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=500_000,
        )
        cur_score = 2.0 * n * _conv_self(f).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal final improve score={cur_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation and return a plain list.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
