def _run_annealing_refine(a0, n, deadline, start_time, eps_start=0.05, eps_end=0.001):
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_refine
        # epsilon fraction schedule linear decreasing.
        eps_frac = eps_start * (1 - elapsed/total_time) + eps_end * (elapsed/total_time)
        # temperature schedule similar.
        T = max(1e-9, (max_val) * 0.01 * (1 - elapsed/total_time))
        # select source based on contributions as before.
        k_max = int(np.argmax(conv))
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot = contrib.sum()
        if tot > 0:
            src = rng.choice(i_range, p=contrib / tot)
        else:
            src = rng.choice(n, p=a / a.sum())
        if a[src] < 1e-15:
            continue
        # choose destination uniformly among indices not equal src.
        dst = rng.choice([i for i in range(n) if i != src])
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        new_max = conv_new.max()
        diff = new_max - max_val
        accept = False
        if diff <= 0:
            accept = True
        else:
            if rng.random() < math.exp(-diff / T):
                accept = True
        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print("[...time...] anneal improve: iter=..., max=...")
    return best_a

"""
Proposes a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.  

The search consists of several independent runs inside the 1000 s
budget.  Each run follows three phases:

1️⃣  Warm‑up – exponentiated‑gradient on a smooth (log‑sum‑exp) proxy
    for the maximum self‑convolution.  The schedule is longer than in
    the reference implementation.

2️⃣  Projected sub‑gradient descent – exact max with a momentum term.
    Many more iterations are carried out (the whole remaining budget
    after the warm‑up).

3️⃣  Simulated‑annealing refinement – massive tiny mass moves are
    performed, but now uphill moves are occasionally accepted
    (temperature schedule).  This helps to escape local minima and
    typically yields a lower peak of the self‑convolution.

All phases keep the sequence normalised to sum = 1, and the best
sequence over all runs is returned.  Helper functions are top‑level
and only standard scientific libraries are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 8000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient (log‑sum‑exp)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.3,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : simulated‑annealing incremental refinement
# ----------------------------------------------------------------------
def _run_annealing_refine(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          eps_start: float = 0.05,
                          eps_end: float = 0.001,
                          temp_start: float = 5e-5,
                          temp_end: float = 5e-9,
                          rel_improve: float = 1e-12):
    """
    Tiny mass‑shifts with a simulated‑annealing acceptance rule.
    The move size (fraction of the source mass) shrinks linearly;
    the temperature governing uphill acceptance also decays.
    """
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0

    while time.time() < deadline:
        # schedules
        elapsed = time.time() - start_refine
        frac = elapsed / total_time
        eps_frac = eps_start * (1 - frac) + eps_end * frac
        T = temp_start * (1 - frac) + temp_end * frac

        # current maximal index
        k_max = int(np.argmax(conv))

        # ---- source: proportional to its contribution to the max bin ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            src = rng.choice(i_range, p=contrib / tot_contrib)
        else:
            src = rng.choice(n, p=a / a.sum())

        if a[src] < 1e-15:
            continue

        # ---- destination: any other index (uniform) ----
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # ---- amount to move -------------------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ---- propose new convolution (O(n) analytic update) -----------------
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a

        new_max = conv_new.max()
        diff = new_max - max_val

        # ---- acceptance (annealing) -----------------------------------------
        accept = False
        if diff <= 0:
            accept = True
        elif T > 0 and rng.random() < math.exp(-diff / T):
            accept = True

        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional re‑normalisation of the convolution to erase numerical drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a
