"""
Propose a low‑maximum self‑convolution step‑function.

Algorithm
---------

1. **Parameterisation** – the probabilities `a` are expressed as a
   soft‑max of an unconstrained vector `x`.  This guarantees `a_i ≥ 0`
   and `∑ a_i = 1` without needing explicit projection.

2. **Smooth objective** – minimise a smooth approximation of the true
   objective `2·n·max(conv(a,a))`.  The smooth max is the
   log‑sum‑exp (with a temperature `α`).  Its gradient can be written
   analytically using two FFT‑based convolutions, so each evaluation
   costs O(n log n).

3. **L‑BFGS optimisation** – run a short L‑BFGS‑B optimisation for a
   schedule of increasing smoothing parameters
   `[0.5, 1, 2, 5, 10, 20, 40, 80, 200]`.
   The optimiser works directly on `x`; after each stage the current
   distribution is evaluated with the exact `evaluate_sequence` and the
   best one is remembered.

4. **Targeted simulated annealing** – starting from the best L‑BFGS
   result, perform a classic peak‑removal annealing: repeatedly locate
   the index of the maximal convolution coefficient, pick a contributing
   index proportionally to its contribution, move a small random amount
   of mass to a random destination and accept the move according to an
   exponential cooling schedule.

5. **Random restarts** – if time permits, perturb the current best
   distribution, run a brief L‑BFGS (single high‑α stage) and a short
   anneal.  Improvements are kept.

The whole routine respects the 1000 s budget and returns a plain Python
list of non‑negative heights ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_w_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Convolution needed for the gradient:
    returns the vector g where g[j] = Σ_i w[i]·a[i‑j] (zero‑padded).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(len(w) + n - 1)
    Wf = np.fft.rfft(w, n=L)
    Af = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(Wf * Af, n=L)[: len(w) + n - 1]
    # slice indices n‑1 … n‑1+n‑1  (inclusive)
    return conv[n - 1 : n - 1 + n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability simplex."""
    mx = np.max(x)
    exps = np.exp(x - mx)
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Smooth objective and gradient (log‑sum‑exp approximation)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(x: np.ndarray, n: int, alpha: float):
    """
    Returns (value, gradient) of 2·n·smooth_max(conv(a,a)),
    where a = softmax(x) and smooth_max is the log‑sum‑exp with
    parameter `alpha`.
    """
    # a lies on the simplex by construction
    a = _softmax(x)

    # self‑convolution of a
    conv = _conv_self(a)
    maxc = np.max(conv)

    # soft‑max weights for the smooth max
    w_unnorm = np.exp(alpha * (conv - maxc))
    sum_w = w_unnorm.sum()
    w = w_unnorm / sum_w

    # gradient w.r.t. a : Σ_i w_i·(a_{i‑j}+a_j) = (conv_w_a)_j + a_j
    grad_a = _conv_w_a(w, a) + a

    # chain rule for the soft‑max parametrisation
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

    # smooth approximation of the max
    smooth_max = maxc + np.log(sum_w) / alpha
    value = 2.0 * n * smooth_max
    return value, grad_x


# ----------------------------------------------------------------------
# Targeted simulated annealing (same idea as the reference code)
# ----------------------------------------------------------------------
def _run_annealing(a: np.ndarray, n: int, deadline: float,
                  start_time: float, best_score: float,
                  start_temp: float = 5e-2,
                  end_temp: float = 1e-8,
                  steps: int = 120_000):
    """Moves mass away from the current maximal convolution peak."""
    rng = np.random.default_rng()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best = best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k - idx]
        if contrib.sum() > 0.0:
            src = rng.choice(idx, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best:
            best = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"score={score:.6f}")

    return best_seq, best
