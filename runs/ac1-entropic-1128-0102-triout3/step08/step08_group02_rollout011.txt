"""
Hybrid local‑search optimiser for the step‑function bound.

The optimiser starts from the best known sequence (``height_sequence_1`` if
available) and performs four stages:

1. **Smooth max gradient descent** – minimise a
   log‑sum‑exp approximation of the maximal self‑convolution while staying
   on the probability simplex (soft‑max parametrisation).

2. **Exact sub‑gradient descent** – multiplicative (exponentiated) updates
   using the true sub‑gradient of the max‑convolution, with a back‑tracking
   line‑search that only accepts improving steps.

3. **Refined peak reduction** – deterministic moves that lower the current
   peak by moving mass from the pair that contributes most to the maximal
   lag to the smallest coefficients.

4. **Hill‑climbing random moves** – many small random mass‑transfer moves,
   always keeping the best improvement (no worsening moves).

All stages respect a global 1000 s time budget.  The final sequence is
normalised to sum = 1 and returned as a plain list of ``float``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·conv_k) w.r.t. a.
    Returns a vector of shape (n,).
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                       # soft‑max weights
    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


# ----------------------------------------------------------------------
# Objective helper
# ----------------------------------------------------------------------


def _objective(p: np.ndarray, n: int) -> float:
    """
    Returns the value of the evaluation function for a probability vector p
    (i.e. sum(p)=1, p≥0).  The formula simplifies to 2·n·max(conv(p,p)).
    """
    conv = _conv_self(p)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _smooth_phase(a0: np.ndarray, n: int, deadline: float, start_time: float) -> np.ndarray:
    """
    Gradient descent on the smooth log‑sum‑exp approximation of the max
    convolution.  Operates directly on the probability simplex.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a, n)

    # schedule of smoothness parameters – become increasingly sharp
    alphas = [5.0, 20.0, 80.0, 320.0, 1280.0, 5120.0]
    it_per_alpha = 1500
    total_it = 0
    fft_len = _next_pow2(2 * n - 1)

    for alpha in alphas:
        for _ in range(it_per_alpha):
            if time.time() >= deadline:
                return best
            conv = _conv_self(a)
            # gradient of the smooth max (already includes factor 2·n)
            grad = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * n)
            # simple diminishing step size
            step = 0.12 / (1.0 + 0.0002 * total_it)
            a = a - step * grad
            a = _project_simplex(a)

            total_it += 1
            # occasional exact evaluation
            if total_it % 300 == 0:
                sc = _objective(a, n)
                if sc < best_score:
                    best_score = sc
                    best = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] smooth improve: α={alpha:.0f}, "
                          f"iter={total_it:,}, score={sc:.6f}")
    return best


def _subgrad_phase(a0: np.ndarray, n: int, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent on the true max
    convolution.  A simple back‑tracking line‑search keeps only improving
    moves, guaranteeing monotone descent.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a, n)
    it = 0
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        # base step size (decreases like 1/√it)
        step = 0.25 / math.sqrt(it)

        # propose multiplicative update
        a_prop = a * np.exp(-step * grad)
        a_prop = _project_simplex(a_prop)

        score_prop = _objective(a_prop, n)

        if score_prop < best_score - 1e-12:
            best_score = score_prop
            best = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: iter={it:,}, "
                  f"score={score_prop:.6f}")
        else:
            # no improvement – keep current a (do not accept worsening move)
            # but continue iteration to allow different directions later
            a = a_prop

    return best, best_score


def _peak_reduce_phase(a0: np.ndarray, n: int, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy reduction of the current convolution peak.
    Moves a modest fraction of the most‑responsible source coefficient
    to the smallest coefficient.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a, n)

    no_improve = 0
    max_no_improve = 3000

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        # indices that actually participate in the lag k_max
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]

        src = i_idx[contrib.argmax()]          # source of the biggest contribution
        # destination = smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.25                     # move 25 % of the source mass
        if eps < 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        score_prop = _objective(a_prop, n)

        if score_prop < best_score - 1e-12:
            a = a_prop
            best = a_prop.copy()
            best_score = score_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: score={score_prop:.6f}")
        else:
            no_improve += 1

    return best, best_score


def _hill_climb_phase(a0: np.ndarray, n: int, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomised hill‑climbing: repeatedly transfer a small fraction of mass
    from a relatively large coefficient to a relatively small one, keeping
    only improving moves.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a, n)

    rng = np.random.default_rng()
    no_improve = 0
    max_no_improve = 20000

    while time.time() < deadline and no_improve < max_no_improve:
        # pick source proportional to current weight (bias towards larger coeffs)
        src = rng.choice(n, p=a)
        # pick a destination that is currently small
        dst_candidates = np.where(a < a.mean())[0]
        if dst_candidates.size == 0:
            dst = rng.integers(n)
        else:
            dst = rng.choice(dst_candidates)

        if src == dst:
            continue

        eps = a[src] * rng.random() * 0.10   # up to 10 % of source mass
        if eps < 1e-14:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        score_prop = _objective(a_prop, n)

        if score_prop < best_score - 1e-12:
            a = a_prop
            best = a_prop.copy()
            best_score = score_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve: score={score_prop:.6f}")
        else:
            no_improve += 1

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method consists of:
      1. smooth log‑sum‑exp gradient descent,
      2. exact sub‑gradient descent with back‑tracking,
      3. deterministic peak reduction,
      4. random hill‑climbing.
    All phases respect the global time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine problem size and initialise a probability vector
    # ------------------------------------------------------------------
    try:
        init_seq = np.asarray(height_sequence_1, dtype=float)
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
        n = init_seq.shape[0]
    except Exception:
        n = 4800
        init_seq = np.full(n, 1.0 / n)

    # normalise to a probability simplex
    init_seq = np.maximum(init_seq, 0.0)
    s = init_seq.sum()
    if s <= 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = init_seq / s

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    best_seq = a0.copy()
    best_score = _objective(best_seq, n)

    # ------------------------------------------------------------------
    # Divide the remaining budget among the four phases
    # ------------------------------------------------------------------
    total_budget = deadline - start_time
    smooth_deadline = start_time + total_budget * 0.10
    subgrad_deadline = smooth_deadline + total_budget * 0.55
    peak_deadline = subgrad_deadline + total_budget * 0.20
    hill_deadline = deadline

    # -------- Phase 1 – smooth max gradient descent -------------------
    if time.time() < smooth_deadline:
        a_smooth = _smooth_phase(a0, n, smooth_deadline, start_time)
        sc = _objective(a_smooth, n)
        if sc < best_score:
            best_score = sc
            best_seq = a_smooth.copy()

    # -------- Phase 2 – exact sub‑gradient descent --------------------
    if time.time() < subgrad_deadline:
        a_sub, sc = _subgrad_phase(best_seq, n, subgrad_deadline, start_time)
        if sc < best_score:

    # -------- Phase 3 – deterministic peak reduction ------------------
    if time.time() < peak_deadline:
        a_peak, sc = _peak_reduce_phase(best_seq, n, peak_deadline, start_time)
        if sc < best_score:

    # return a plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
