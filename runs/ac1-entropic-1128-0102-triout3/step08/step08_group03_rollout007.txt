def _expgrad_optimize(a0, deadline, start_time, eta0=3.0):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float("inf")
    t = 0
    while time.time() < deadline:
        t += 1
        # compute convolution
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] expgrad improve: iter={t}, score={cur_score:.6f}")
        # gradient at argmax
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1)
        j = k - i
        grad = np.zeros_like(a)
        grad[i] = 2 * a[j]  # gradient for max shift
        # step size
        eta = eta0 / math.sqrt(t)
        # exponentiated update - move opposite gradient
        # avoid underflow
        update = -eta * grad
        # clip update to not overflow
        # a_i = a_i * exp(update_i)
        a = a * np.exp(update)
        # renormalize to sum 1
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum) or np.isinf(a_sum):
            # reset to uniform if degenerate
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        # occasional exact evaluation
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score

"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
* **Initial pool** – several handcrafted starts (uniform, exponential,
  Gaussian, Dirichlet) and the previously known best sequence.
* **Quick Adam smoothing** – a few Adam iterations on a
  log‑sum‑exp surrogate of the max‑convolution (soft‑max) to
  escape obvious bad regions.
* **Exponentiated‑gradient polishing** – a fast projected mirror‑descent
  on the true objective.  At each iteration the gradient is taken only
  from the *single* shift that currently attains the maximum
  convolution; the update is multiplicative (a ← a·exp(−η·g)) and the
  vector is re‑normalised to stay on the probability simplex.
* **Greedy peak reduction** – a deterministic local move that transfers
  a fraction of the coefficient that contributes most to the current
  peak towards the smallest coefficient, accepting only strict
  improvements.
* **Time budget** – the whole procedure respects the supplied
  ``budget_s`` (default 1000 s).  The final vector is clipped to
  ``[0,1000]`` and normalised before being returned.

The exponentiated‑gradient phase is much cheaper per iteration than the
previous sub‑gradient routine (it uses only the arg‑max shift), allowing
far more iterations in the same wall‑clock time and typically yielding a
lower final score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (full linear convolution a*a)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   global_deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    A few Adam iterations on the smoothed max‑convolution objective.
    Returns the best vector (probability vector) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    # work in the log‑space of the soft‑max (logits)
    y = np.log(np.maximum(a, eps_log))

    # smoothing schedule – start smooth, finish almost exact
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 2000

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= global_deadline:
                return best_a, best_score

            # soft‑max to a valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # true objective (for printing only)
            obj = 2.0 * n * f_val

            # gradient of the smooth surrogate w.r.t. a
            # --- same formula as in the original code -----------------
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing on the true max‑convolution objective
# ----------------------------------------------------------------------


def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on f(a)=max_k (a*a)[k].
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # gradient from the *single* shift attaining the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]  # the sub‑gradient for the active shift

        # step size (diminishing)
        eta = eta0 / math.sqrt(t)

        # multiplicative (exponentiated) update – move opposite to the gradient
        a = a * np.exp(-eta * g)
        # renormalise onto the simplex (ensures positivity and sum‑1)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation to guard against numerical drift
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local improvement (same as the previous version)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine consists of a brief Adam warm‑up, an extensive
    exponentiated‑gradient polishing, and a final greedy peak‑reduction.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # problem size -------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build diverse starts (including the previous best if available)
    # ------------------------------------------------------------------
    starts = []

    # 1) previous construction, if present
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) uniform
    starts.append(np.full(n, 1.0 / n))

    # 3) exponential (very gentle decay)
    starts.append(_exponential_start(n, r=0.9995))

    # 4) exponential (slightly stronger decay)
    starts.append(_exponential_start(n, r=0.985))

    # 5) Gaussian bell
    starts.append(_gaussian_start(n))

    # 6) Dirichlet – moderate concentration
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))

    # 7) Dirichlet – low concentration (more spiky)
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # evaluate the raw starts and keep the best
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1} raw score = {sc:.6f}"
        )

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient polishing (≈80 % of the time)
    # ------------------------------------------------------------------
    polish_deadline = start_time + budget_s * 0.9

        best_seq, polish_deadline, start_time, fft_len, eta0=0.5
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polishing score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak‑reduction (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:

    else:

    # final reported score (for logging)
    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    return [float(x) for x in best_seq.tolist()]
