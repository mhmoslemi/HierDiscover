"""
Hybrid optimiser with an aggressive fast‑annealing phase.

The algorithm proceeds in several stages (all respecting a global 1000 s
budget):

1. **Initialisation** – several diverse starting points are built
   (uniform, exponential, U‑shaped, smooth random, Dirichlet and the
   historic best if available).

2. **Soft‑max Adam** – a short Adam optimisation on the smooth
   log‑sum‑exp approximation of the max‑convolution drives the solution
   quickly toward a low‑peak region.

3. **Exponentiated‑gradient (sub‑gradient) descent** – a long
   multiplicative update on the true max‑convolution further reduces the
   maximal autocorrelation.

4. **Fast simulated annealing** – instead of re‑computing the full
   convolution after each move, the convolution is updated
   incrementally (O(n) vectorised work).  This makes it possible to
   perform millions of annealing steps within the remaining time,
   allowing a much deeper exploration than a standard FFT‑based
   annealer.

5. **Deterministic peak‑reduction** – a greedy “move mass from the
   biggest contribution to the smallest coefficient” pass finishes the
   refinement.

All intermediate sequences are kept on the probability simplex (sum = 1)
and are projected onto the non‑negative orthant when required.  The
best sequence discovered is finally normalised and returned as a list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·conv_k) w.r.t. a.
    Returns a vector of shape (n,).
    """
    scaled = alpha * conv
    m = scaled.max()
    w = np.exp(scaled - m)
    w /= w.sum()
    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray,
                deadline: float,
                start_time: float,
                fft_len: int,
                iters_per_alpha: int = 3000) -> np.ndarray:
    """
    Adam optimisation on the smooth (log‑sum‑exp) approximation of the
    max‑convolution.  Returns the best simplex point seen.
    """
    y = np.log(np.maximum(a0, 1e-16))  # logits
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # schedule – increasing smoothness
    alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
                   80.0, 160.0, 320.0, 640.0]

    best_seq = a0.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    for alpha in alpha_sched:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq
            a = _softmax_from_logits(y)

            conv = _conv_self(a)
            scaled = alpha * conv
            mscaled = scaled.max()
            lse = mscaled + np.log(np.exp(scaled - mscaled).sum())
            f_val = lse / alpha
            obj = 2.0 * a.shape[0] * f_val

            grad_a = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * a.shape[0])
            grad_y = a * (grad_a - np.dot(grad_a, a))

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))

            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if i % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                          f"iter={i:,}, score={sc:.6f}")

    return best_seq


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         step_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Long exponentiated‑gradient descent on the true max‑convolution.
    Returns best sequence and its internal score (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    cur_conv = _conv_self(a)
    best_score = 2.0 * n * cur_conv.max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        step = step_factor / math.sqrt(it)
        a = a * np.exp(-step * grad)

        # Euclidean projection onto simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 5000) -> tuple[np.ndarray, float]:
    """Greedy deterministic moves that try to lower the current peak."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]

        src = i_idx[contrib.argmax()]

        # choose destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.3
        if eps < 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] Peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


def _fast_annealing(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    start_temp: float = 0.5,
                    end_temp: float = 1e-6,
                    max_move_frac: float = 0.3,
                    rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with O(n) incremental convolution updates.
    Returns the best sequence found and its internal score.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    phase_start = time.time()
    total_phase = max(deadline - phase_start, 1e-9)

    # occasional full recalculation to mitigate drift
    full_recalc_every = 20000
    steps = 0

    while time.time() < deadline:
        elapsed = time.time() - phase_start
        # geometric cooling schedule
        T = start_temp * (end_temp / start_temp) ** (elapsed / total_phase)

        # pick source weighted by mass, destination uniformly
        prob = a / a.sum()
        src = rng.choice(n, p=prob)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * max_move_frac
        if eps <= 0:
            continue

        # keep old copies for possible rejection
        a_old = a.copy()
        conv_old = conv.copy()

        # incremental update of convolution using the old a
        conv[src:src + n] -= 2.0 * eps * a_old[:]
        conv[dst:dst + n] += 2.0 * eps * a_old[:]

        # epsilon‑squared correction (exact, but tiny)
        idx2src = 2 * src
        if idx2src < conv.shape[0]:
            conv[idx2src] += eps * eps
        idx2dst = 2 * dst
        if idx2dst < conv.shape[0]:
            conv[idx2dst] += eps * eps
        idxsum = src + dst
        if idxsum < conv.shape[0]:
            conv[idxsum] -= 2.0 * eps * eps

        # move the mass
        a[src] -= eps
        a[dst] += eps

        # enforce non‑negativity (tiny rounding artefacts)
        if a[src] < 0:
            a[src] = 0.0

        new_max = conv.max()
        delta = new_max - cur_max
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis acceptance
            if T > 0 and math.exp(-delta / T) > rng.random():
                accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                # occasional reporting
                if steps % 5000 == 0:
                    print(f"[{time.time() - start_time:6.2f}s] Fast anneal improve: max={cur_max:.6f}")
        else:
            # reject – restore
            a = a_old
            conv = conv_old

        steps += 1

        # periodic full recomputation to avoid numerical drift
        if steps % full_recalc_every == 0:
            conv = _conv_self(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    best_score = 2.0 * n * best_max
    return best_a, best_score


def _random_smooth_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gaussian‑smoothed noise, clipped to non‑negative and normalised."""
    noise = rng.normal(size=n)
    smooth = np.convolve(noise, np.ones(15) / 15, mode='same')
    smooth = np.clip(smooth, 0, None)
    if smooth.sum() == 0:
        smooth = np.full(n, 1.0 / n)
    else:
        smooth /= smooth.sum()
    return smooth

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.
    The optimiser combines Adam on a smooth surrogate, long
    exponentiated‑gradient descent, a fast incremental annealing pass,
    and a final deterministic peak‑reduction.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Determine problem size (n).  Use the length of the supplied
    # ``height_sequence_1`` when available, otherwise fall back to 4800.
    # ------------------------------------------------------------------
    try:
        starter_seq = np.asarray(height_sequence_1, dtype=float)
        if starter_seq.ndim == 1 and starter_seq.size > 0:
            n = starter_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4800
        starter_seq = None

    def _norm(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            v /= s
        else:
            v = np.full_like(v, 1.0 / v.shape[0])
        return v

    # ------------------------------------------------------------------
    # Build a pool of diverse initial points.
    # ------------------------------------------------------------------
    starts = []

    if starter_seq is not None:
        starts.append(lambda: _norm(starter_seq.copy()))

    starts.append(lambda: _norm(np.full(n, 1.0 / n)))                     # uniform

    r = 0.985
    exp_vec = np.array([r ** i for i in range(n)], dtype=float)
    starts.append(lambda: _norm(exp_vec))

    idx = np.arange(n)
    u_shape = 1.0 / (np.minimum(idx + 1, n - idx) + 1.0)
    starts.append(lambda: _norm(u_shape))

    starts.append(lambda: _norm(_random_smooth_start(n, rng)))           # smooth random

    starts.append(lambda: _norm(rng.gamma(0.5, 1.0, size=n)))           # Dirichlet‐like

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optimise each start independently while respecting the global budget.
    # ------------------------------------------------------------------
    for i_start, gen in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{i_start + 1}")

        a0 = gen()

        # ---- Phase 1: short Adam on smooth max ----
        phase_deadline = time.time() + min(0.08 * (deadline - start_time), 35.0)
        fft_len = _next_pow2(3 * n - 2)
        a_adam = _adam_phase(a0, phase_deadline, start_time, fft_len)

        # ---- Phase 2: exponentiated‑gradient (sub‑gradient) descent ----
        if time.time() < deadline:
            sub_deadline = time.time() + min(0.55 * (deadline - time.time()), 500.0)
            a_sub, sub_score = _subgradient_descent(a_adam, sub_deadline, start_time,
                                                    step_factor=0.8)
        else:
            a_sub, sub_score = a_adam, 2.0 * n * _conv_self(a_adam).max()

        # ---- Phase 3: fast annealing with incremental updates ----
        if time.time() < deadline:
            anneal_deadline = time.time() + min(0.25 * (deadline - time.time()), 400.0)
            a_anneal, anneal_score = _fast_annealing(a_sub, anneal_deadline,
                                                     start_time,
                                                     start_temp=0.5,
                                                     end_temp=1e-6,
                                                     max_move_frac=0.3,
                                                     rng=rng)
        else:
            a_anneal, anneal_score = a_sub, 2.0 * n * _conv_self(a_sub).max()

        # ---- Phase 4: deterministic peak‑reduction (polish) ----
        if time.time() < deadline:
            peak_deadline = time.time() + min(0.07 * (deadline - time.time()), 60.0)
            a_peak, peak_score = _run_peak_reduction(a_anneal, peak_deadline, start_time)
        else:
            a_peak, peak_score = a_anneal, 2.0 * n * _conv_self(a_anneal).max()

        # Select the best among the stages for this start
        stage_candidates = [
            (evaluate_sequence(a_adam.tolist()), a_adam),
            (evaluate_sequence(a_sub.tolist()), a_sub),
            (evaluate_sequence(a_anneal.tolist()), a_anneal),
            (evaluate_sequence(a_peak.tolist()), a_peak),
        ]
        local_best_score, local_best_seq = min(stage_candidates, key=lambda x: x[0])

        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST *** score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If time still remains, try a few pure‑annealing random starts.
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a0 = _norm(rng.gamma(0.5, 1.0, size=n))
        anneal_deadline = deadline
        a_fin, _ = _fast_annealing(a0, anneal_deadline, start_time,
                                   start_temp=0.7, end_temp=1e-6,
                                   max_move_frac=0.4, rng=rng)
        fin_score = evaluate_sequence(a_fin.tolist())
        if fin_score < best_score:
            best_score = fin_score
            best_seq = a_fin.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST (random) *** "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
