"""
Propose a step‐function (nonnegative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three stages, all respecting a global time budget
(default 1000 s).

1. **LP balancing** – starting from a good known construction (or the
   uniform vector) we repeatedly solve the linear program

        maximise    sum(b)
        subject to  b ≥ 0,
                    (a * b)(k) ≤ M   for all k,
        where M = max(a * a).

   The optimal b pushes mass into positions where a has little influence.
   A convex combination of a and the normalised b reduces the peak M.
   This step is the “linear‑programming” part described in the statement
   and usually yields a large drop in the objective.

2. **Projected sub‑gradient descent** – we fine‑tune the vector on the
   true non‑smooth objective using a momentum sub‑gradient method with
   Euclidean projection onto the probability simplex.

3. **Exponentiated‑gradient (mirror descent)** – a short entropy‑mirror‑
   descent polishing phase improves the bound further.

All vectors are kept on the simplex (non‑negative, sum = 1) because the
objective is scale‑invariant.  The routine returns a plain Python list
that can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Linear‑algebra helpers (no external dependencies)
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility used by the FFT based convolution."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (a ≥ 0).
    For all k attaining the maximum we add the contribution of a to the
    gradient; the result is averaged over the number of active indices.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# -------------------------------------------------------------
# Main optimisation routine
# -------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine follows the three‑stage
    strategy described in the module docstring and respects the supplied
    time budget (in seconds).  The returned value is a plain list of
    floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # 0.5 s safety margin

    # ---------------------------------------------------------
    # Choose a problem size.  2048 gives a good trade‑off between
    # resolution and the cost of the linear programme.
    # ---------------------------------------------------------
    n = 2048

    # ---------------------------------------------------------
    # Initialise -----------------------------------------------------------------
    # Use the previously discovered construction if it exists,
    # otherwise start from the uniform distribution.
    # ---------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                if tmp.size != n:
                    # linear interpolation to the required length
                    old_x = np.linspace(0.0, 1.0, tmp.size)
                    new_x = np.linspace(0.0, 1.0, n)
                    tmp = np.interp(new_x, old_x, tmp)
                a = tmp / tmp.sum()
            else:
                a = np.full(n, 1.0 / n)
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    best_seq = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    print(f"[{time.time() - start_time:6.2f}s] initial score {best_score:.6f}")

    # ---------------------------------------------------------
    # Helper: build the convolution matrix T such that
    #   (a * b) = T @ b
    # (the matrix has shape (2·n‑1, n) and is simply n shifted copies
    #  of the vector a).
    # ---------------------------------------------------------
    def _build_T(vec: np.ndarray) -> np.ndarray:
        T = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            T[i:i + n, i] = vec                     # shifted copy of vec
        return T

    # ---------------------------------------------------------
    # Stage 1 – Linear‑programming balancing (if cvxpy is available)
    # ---------------------------------------------------------
    try:
        import cvxpy as cp

        # allocate roughly half of the remaining budget to the LP stage
        lp_time = (deadline - time.time()) * 0.5
        lp_deadline = time.time() + lp_time

        iteration = 0
        while time.time() < lp_deadline:
            iteration += 1

            conv_a = _conv_self(a)
            M = conv_a.max()

            # Build the convolution matrix for the current a
            T = _build_T(a)

            # LP: maximise sum(b) s.t. b ≥ 0, T·b ≤ M
            b = cp.Variable(n, nonneg=True)
            constraints = [T @ b <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=300)
            except Exception:       # pragma: no cover
                print("[LP] Solver failure – exiting LP stage.")
                break

            if b.value is None:
                print("[LP] No solution returned – exiting LP stage.")
                break

            b_vec = np.maximum(b.value, 0.0)
            sum_b = b_vec.sum()
            if sum_b <= 0.0:
                break

            # Normalise the partner vector to the simplex
            g = b_vec / sum_b

            # Try a few convex combinations of a and g and keep the best.
            best_local_a = a
            best_local_score = 2.0 * n * conv_a.max()
            for t in (0.2, 0.5, 0.8, 1.0):
                cand = (1.0 - t) * a + t * g
                cand = _project_simplex(cand)
                cand_score = 2.0 * n * _conv_self(cand).max()
                if cand_score < best_local_score:
                    best_local_score = cand_score
                    best_local_a = cand

            a = best_local_a

            # Global bookkeeping
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] LP iter {iteration:<3d} "
                      f"improve → {cur_score:.6f}")

            # a lightweight termination heuristic
            if iteration >= 50:                     # enough for a good start
                break
    except Exception:  # pragma: no cover
        # cvxpy not available – simply skip the LP stage.
        print("[LP] cvxpy not available – skipping LP balancing.")

    # ---------------------------------------------------------
    # Stage 2 – Projected Euclidean sub‑gradient descent (momentum)
    # ---------------------------------------------------------
    sub_deadline = deadline - 2.0               # keep a couple of seconds for stage 3
    if time.time() < sub_deadline:
        a = best_seq.copy()
        momentum = np.zeros_like(a)
        it = 0
        while time.time() < sub_deadline:
            it += 1
            conv_a = _conv_self(a)
            cur_score = 2.0 * n * conv_a.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] subgrad iter {it:<6d} "
                      f"improve → {cur_score:.6f}")

            grad = _subgradient_max(a, conv_a)
            momentum = 0.94 * momentum + 0.06 * grad
            step = 0.15 / math.sqrt(it)
            a = _project_simplex(a - step * momentum)

    # ---------------------------------------------------------
    # Stage 3 – Exponentiated‑gradient (entropy mirror descent)
    # ---------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        it = 0
        while time.time() < deadline:
            it += 1
            conv_a = _conv_self(a)
            cur_score = 2.0 * n * conv_a.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] EG iter {it:<6d} "
                      f"improve → {cur_score:.6f}")

            grad = _subgradient_max(a, conv_a)
            eta = 0.20 / math.sqrt(it)
            # Multiplicative update, protected against under‑flow
            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

    # -----------------------------------------------------------------
    # Final normalisation and return a plain Python list.
    # -----------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
