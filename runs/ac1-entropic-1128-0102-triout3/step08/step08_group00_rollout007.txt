Propose a step‑function (non‑negative list) that minimises
the provided evaluate_sequence() objective.

Algorithm
---------
* Work on a moderate size n (1024) – large enough to get a low bound,
  small enough so that a dense linear program is cheap.
* Repeated random restarts:
    - start from a random point on the probability simplex.
    - iterate a *linear‑programming refinement*:
        + For the current vector a compute M = max(a*a).
        + Solve   maximise   Σ b_i

          The constraint matrix is exactly the convolution matrix of a;

from scipy.optimize import linprog

# Helper functions
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    return np.convolve(a, a)


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective value 2·n·max(conv(a,a))."""
    return float(2.0 * n * _conv_self(a).max())


def _lp_refine(a: np.ndarray, n: int, M: float) -> np.ndarray:
    """
    Solve the LP:
        maximise Σ b_i
        s.t.   (a * b)_k ≤ M   for all k,
               b_i ≥ 0.
    Returns the optimal b (may be None if LP fails).
    """
    # build convolution matrix of a (Toeplitz, shape (2n‑1, n))
    A = diags([a] * n, offsets=np.arange(n), shape=(2 * n - 1, n), format='csr')
    c = -np.ones(n)                      # linprog minimises, we maximise Σb
    b_ub = np.full(2 * n - 1, M)         # RHS of the ≤ constraints
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None
