def _run_targeted_refine(a0, n, deadline, start_time):
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    it = 0
    eps_init = 0.2
    eps_min = 0.001
    rand_prob = 0.02

    while time.time() < deadline:
        it += 1

        # compute epsilon fraction as linear decay
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = eps_init * (1.0 - decay) + eps_min * decay

        # compute k_max
        k_max = int(conv.argmax())

        # decide if random move
        if rng.random() < rand_prob:
            # random src/dst
            src_candidates = np.where(a > 1e-14)[0]
            if src_candidates.size == 0:
                continue
            src = rng.choice(src_candidates)
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)
        else:
            # normal targeted src/dst selection
            i = np.arange(n)
            j = k_max - i
            valid = (j >= 0) & (j < n)
            contrib = np.zeros(n)
            contrib[valid] = a[i[valid]] * a[j[valid]]
            total_contrib = contrib.sum()
            if total_contrib > 0:
                prob_src = contrib / total_contrib
            else:
                prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

            # destination selection
            dst_idx = np.arange(n)
            jdst = k_max - dst_idx
            valid_dst = (jdst >= 0) & (jdst < n)
            dest_weight = np.zeros(n)
            dest_weight[valid_dst] = a[jdst[valid_dst]]
            zero_mask = dest_weight < 1e-14
            if np.any(zero_mask):
                candidates = np.where(zero_mask)[0]
            else:
                min_val = dest_weight.min()
                candidates = np.where(dest_weight == min_val)[0]
            dst = rng.choice(candidates)

        if src == dst:
            continue

        if a[src] < 1e-16:
            continue

        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # delta update
        conv_len = 2*n - 1
        delta = np.zeros(conv_len)
        # (2*eps*a) offset
        delta[dst:dst+n] += 2.0 * eps * a
        delta[src:src+n] -= 2.0 * eps * a
        # quadratic corrections
        delta[2*src] += eps * eps
        delta[2*dst] += eps * eps
        delta[src+dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()
        if new_max < max_val - 1e-14:
            # accept

            a[dst] += eps

"""
Multi‑phase optimizer for the self‑convolution bound.

Phase 1 – exponentiated‑gradient on a smoothed max (log‑sum‑exp).
Phase 2 – projected sub‑gradient descent on the exact max.
Phase 3 – targeted refinement with an adaptive step size and occasional
          random mass‑transfer moves (helps to escape local minima).

The routine respects a 1000 s time budget, records the best sequence found,
and returns it as a plain list of floats.  All helper functions are defined at
module level; no closures or file I/O are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to ``x``."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    Numerical noise that creates tiny negatives is clipped to zero.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of ``seq`` to length ``new_n`` while preserving total mass.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _objective_value(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (ignoring the clamping steps
    that are irrelevant for non‑negative, summed‑to‑one vectors):
    2·n·max(a*a).
    """
    conv = _conv_self(a)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated (mirror) gradient on a smoothed max
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 5000):
    """
    Multiplicative mirror‑descent on the log‑sum‑exp smooth approximation of
    the max convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # length needed for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # --- smoothed‑max weights ------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        # --- gradient via correlation --------------------------------
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # --- multiplicative mirror update ----------------------------
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10 ** 7):
    """
    Projected sub‑gradient descent on the true maximal self‑convolution
    entry.  Momentum (exponential averaging) is used to stabilise the steps.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        max_val = conv.max()
        # tolerance to capture all (numerically) maximal entries
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of the max (sum of gradients of all active entries)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted refinement + occasional random mass moves
# ----------------------------------------------------------------------


def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         init_eps_frac: float = 0.20,
                         min_eps_frac: float = 0.001,
                         random_move_prob: float = 0.02):
    """
    Fast O(n) refinement.
    * A small fraction of probability mass is moved away from indices that
      contribute most to the current maximal convolution entry.
    * With probability ``random_move_prob`` a completely random source/destination
      pair is tried – this helps to escape shallow plateaus.
    * The step‑size (fraction of the source mass) decays linearly from
      ``init_eps_frac`` to ``min_eps_frac`` over the whole phase.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linear decay of epsilon fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        if rng.random() < random_move_prob:
            # ---- completely random move ---------------------------------
            src_candidates = np.where(a > 1e-16)[0]
            if src_candidates.size == 0:
                continue
            src = rng.choice(src_candidates)
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)
        else:
            # ---- targeted source ----------------------------------------
            i = np.arange(n)
            j = k_max - i
            valid = (j >= 0) & (j < n)
            contrib = np.zeros(n, dtype=float)
            contrib[valid] = a[i[valid]] * a[j[valid]]
            total_contrib = contrib.sum()
            if total_contrib > 0:
                prob_src = contrib / total_contrib
            else:
                prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

            # ---- targeted destination -----------------------------------
            dst_idx = np.arange(n)
            jdst = k_max - dst_idx
            valid_dst = (jdst >= 0) & (jdst < n)
            dest_weight = np.zeros(n, dtype=float)
            dest_weight[valid_dst] = a[jdst[valid_dst]]
            zero_mask = dest_weight < 1e-14
            if np.any(zero_mask):
                candidates = np.where(zero_mask)[0]
            else:
                min_val = dest_weight.min()
                candidates = np.where(dest_weight == min_val)[0]
            dst = rng.choice(candidates)

        if src == dst or a[src] < 1e-16:
            continue

        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # ----- fast O(n) update of the convolution --------------------
        delta = np.zeros(conv_len, dtype=float)

        # linear part 2*eps*(a shifted)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a

        # quadratic eps**2 correction (three entries)
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to avoid numeric drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the supplied time budget.

    The search consists of three phases (exponential‑gradient,
    sub‑gradient descent, targeted refinement) and returns the best
    sequence discovered.
    """
    # Seed the global RNG (used by ``np.random.rand`` in the warm‑start)
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem size and initial vector
    # --------------------------------------------------------------
    final_n = 4800                     # a size that balances speed & quality
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = _objective_value(best_seq)
    print(f"[0.00s] initial score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – exponential‑gradient (smooth max)
    # --------------------------------------------------------------
    phase1_time = min(30.0, 0.07 * (deadline - time.time()))
    phase1_deadline = time.time() + phase1_time

    a1, s1 = _run_exponential_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 2, 5, 10, 20, 50, 100, 200, 500, 1000],
        eta0=0.5,
        steps_per_alpha=5000,
    )
    if s1 < best_score:
        best_seq = a1.copy()
        best_score = s1
        print(f"[{time.time() - start_time:.2f}s] after phase 1 best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exact‑max projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_time = max(30.0, 0.60 * remaining)
        phase2_deadline = time.time() + phase2_time

        a2, s2 = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            step0=0.30,
            beta=0.94,
        )
        if s2 < best_score:
            best_seq = a2.copy()
            best_score = s2
            print(f"[{time.time() - start_time:.2f}s] after phase 2 best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – targeted refinement with occasional random moves
    # --------------------------------------------------------------
    if time.time() < deadline:
        a3 = _run_targeted_refine(best_seq, final_n, deadline, start_time,
                                  init_eps_frac=0.20,
                                  min_eps_frac=0.001,
                                  random_move_prob=0.02)
        final_score = _objective_value(a3)
        if final_score < best_score:
            best_seq = a3.copy()
            best_score = final_score
            print(f"[{time.time() - start_time:.2f}s] after refinement best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] final best score: {best_score:.6f}")
    return [float(x) for x in best_seq]
