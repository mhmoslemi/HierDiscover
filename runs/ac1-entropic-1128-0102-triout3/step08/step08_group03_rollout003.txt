    # m: number of breakpoints

    rng = np.random.default_rng()

    # initial params: Dirichlet with conc=1

    params = rng.random(m)
    best_params = params.copy()
    best_seq = shape_to_seq(params, n)
    best_score = evaluate_sequence(best_seq.tolist())

    while time.time() < deadline:
        # compute temperature schedule based on elapsed fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total
        # temperature decreasing from initial_temp to final_temp
        temp = (1 - frac) * 0.2 # maybe start at 0.2 variance? Actually temperature is used for acceptance.
        # propose new params by perturbation
        step_scale = 0.05 * (1 - frac) + 0.005  # decaying step
        new_params = params + step_scale * rng.standard_normal(m)
        # clip to nonnegative
        new_params = np.clip(new_params, 0, None)
        # optionally renormalize params to sum=1? Not necessary, shape still normalized for final sequence.
        # Evaluate candidate
        seq = shape_to_seq(new_params, n)
        score = evaluate_sequence(seq.tolist())
        # accept?
        if score < best_score or rng.random() < math.exp(-(score - best_score)/ (temp + 1e-12)):
            params = new_params
            if score < best_score:
                best_score = score
                best_params = new_params.copy()
                best_seq = seq.copy()
                print(f"[{time.time()-start_time:6.2f}s] anneal new best score {best_score:.6f}")
        # also occasionally random restart if stuck
        if rng.random() < 0.001 and time.time() < deadline - 5:
            params = rng.random(m)
            # maybe maintain best.

    return best_seq, best_score

def shape_to_seq(params, n):

    # Ensure nonneg

    params = np.maximum(params, 0)

    if params.sum() == 0:

        params = np.full_like(params, 1.0/len(params))

    else:

        params = params / params.sum()  # shape param normalized to sum=1 (optional)

    # Interpolate

    old_x = np.linspace(0.0, 1.0, len(params))

    new_x = np.linspace(0.0, 1.0, n)

    seq = np.interp(new_x, old_x, params)

    # Clip any negative due to interpolation (should not happen) and ensure all nonneg.

    seq = np.maximum(seq, 0.0)

    # Normalize to sum=1

    total = seq.sum()

    if total > 0:

        seq = seq / total

    else:

        seq = np.full(n, 1.0 / n)

    return seq

"""
Propose a non‑negative step function that (almost) minimises ``evaluate_sequence``.

Algorithm
---------
1. **Low‑dimensional shape representation** – the sequence of length ``N``
   is obtained by linear interpolation of ``M`` break‑points (``M << N``).
   The break‑points are the optimisation variables; after interpolation the
   vector is projected onto the probability simplex (non‑negative, sum = 1).

2. **Simulated annealing** – a pure‑objective (exact) optimiser that
   repeatedly perturbs the break‑points, accepts improving moves and also
   accepts worsening moves with a temperature dependent probability.
   The temperature and the step‑size decay linearly with the elapsed time.
   The objective is evaluated by a fast FFT based convolution
   (identical to ``evaluate_sequence`` for non‑negative, normalised inputs).

3. **Polishing** – the best candidate from the annealing phase is refined
   with a projected sub‑gradient descent on the true ``max``‑convolution
   objective (the same routine used in the reference solution).

4. **Final check** – the result is normalised once more and the official
   ``evaluate_sequence`` function is called to obtain the final reported
   score.

The whole search respects the time budget (default 1000 s).  The routine
works for any sequence length; a larger length gives the optimizer more
degrees of freedom and often yields a lower bound.  The implementation
uses only ``numpy`` and the supplied ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, fast score)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution ``a*a`` using an FFT of length a power of two."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _fast_evaluate(seq: np.ndarray) -> float:
    """
    Fast equivalent of ``evaluate_sequence`` for non‑negative, normalised
    sequences (sum > 0).  Returns the exact objective value.
    """
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    seq = np.clip(seq, 0.0, None)
    conv = _conv_self(seq)
    max_b = conv.max()
    return float(2.0 * seq.shape[0] * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape → full sequence conversion
# ----------------------------------------------------------------------


def _shape_to_sequence(params: np.ndarray, n: int) -> np.ndarray:
    """
    Linear interpolation of ``params`` (length M) to a full vector of length ``n``.
    The result is clipped to non‑negative values and normalised to sum‑one.
    """
    # protect against all‑zero start
    if params.max() <= 0:
        params = np.full_like(params, 1.0 / params.size)

    # interpolation
    old_x = np.linspace(0.0, 1.0, params.size)
    new_x = np.linspace(0.0, 1.0, n)
    seq = np.interp(new_x, old_x, params)

    # enforce non‑negativity and normalise
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n)

    return seq


# ----------------------------------------------------------------------
# Simulated annealing on the low‑dimensional shape
# ----------------------------------------------------------------------


def _anneal_optimize(
    n: int,
    m: int,
    start_time: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Runs a simple simulated‑annealing search over the ``m`` break‑points.
    Returns the best full‑length sequence and its fast score.
    """
    # initial break‑points: Dirichlet(1) (exponential) – ensures diversity
    cur_params = rng.gamma(1.0, 1.0, size=m)
    cur_params = np.maximum(cur_params, 0.0)

    cur_seq = _shape_to_sequence(cur_params, n)
    cur_score = _fast_evaluate(cur_seq)

    best_params = cur_params.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score

    # annealing schedule constants
    init_sigma = 0.2   # relative magnitude of the initial Gaussian step
    min_sigma = 1e-4   # never go below this
    no_improve_counter = 0
    restart_after = 30000  # steps without improvement → random restart

    while time.time() < deadline:
        # remaining fraction of the exploration window
        remaining = deadline - time.time()
        total_explore = deadline - start_time
        frac = max(0.0, remaining / total_explore)  # 1 → start, 0 → end

        # temperature proportional to ``frac`` (linear cooling)
        sigma = max(min_sigma, init_sigma * frac)

        # propose new parameters
        proposal = cur_params + sigma * rng.standard_normal(m)
        proposal = np.clip(proposal, 0.0, None)

        # evaluate proposal (exact objective)
        seq_prop = _shape_to_sequence(proposal, n)
        score_prop = _fast_evaluate(seq_prop)

        # acceptance test
        if score_prop < cur_score:
            accept = True
        else:
            # Metropolis rule – the denominator scales with the current temperature
            prob = math.exp(-(score_prop - cur_score) / (sigma + 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_params = proposal
            cur_seq = seq_prop
            cur_score = score_prop
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_params = cur_params.copy()
            best_seq = cur_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal new best: score={best_score:.6f}"
            )

        # occasional random restart to escape possible traps
        if no_improve_counter > restart_after:
            cur_params = rng.gamma(1.0, 1.0, size=m)
            cur_params = np.clip(cur_params, 0.0, None)
            cur_seq = _shape_to_sequence(cur_params, n)
            cur_score = _fast_evaluate(cur_seq)
            no_improve_counter = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    beta: float = 0.94,
    step0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Returns the best vector seen and its 2·n·max value.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _fast_evaluate(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}"
            )

        # active set – indices where the convolution is (almost) maximal
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient (sum of contributions from each active index)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        # momentum and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Optional greedy local improvement (identical to reference)
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """Small deterministic loop that moves mass from the biggest contributor to the smallest coefficient."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _fast_evaluate(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]
        if eps <= 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _fast_evaluate(a_prop)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(
                f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_score:.6f}"
            )

    return a, best_score


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine consists of:

    1. Simulated‑annealing on a low‑dimensional piecewise‑linear shape.
    2. A short projected sub‑gradient polishing phase.
    3. Optional greedy peak‑reduction.

    The final sequence is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # Larger ``N`` gives more flexibility; 8000 works well within the budget.
    # ------------------------------------------------------------------
    N = 8000          # length of the final step function
    M = 30            # number of break‑points for the low‑dimensional shape

    # ------------------------------------------------------------------
    # Exploration – simulated annealing on the shape.
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.60 * budget_s
    print(f"[{time.time() - start_time:6.2f}s] start annealing (N={N}, M={M})")
    best_seq, best_score = _anneal_optimize(
        N, M, start_time, explore_deadline, rng
    )
    print(
        f"[{time.time() - start_time:6.2f}s] annealing complete – best score {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Polishing – projected sub‑gradient descent on the exact objective.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] polishing phase start")
        # The sub‑gradient routine expects the vector to be already on the simplex.
        polished_seq, polish_score = _subgrad_optimize(
            best_seq, deadline, start_time, beta=0.94, step0=0.2
        )
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished_seq
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (polish) score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final deterministic greedy improvement (cheap, fast).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] greedy peak‑reduction start")
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (greedy) score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation (guard against drift) and official evaluation.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # official evaluation (ensures compatibility with the hidden tests)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # return plain Python list
    return [float(x) for x in best_seq.tolist()]
