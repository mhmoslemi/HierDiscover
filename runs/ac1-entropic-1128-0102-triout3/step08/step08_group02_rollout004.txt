def _random_local_search(a0, deadline, start_time, max_iter=10000):

   a = a0.copy()
   best_a = a.copy()
   best_score = evaluate_sequence(besta.tolist())
   n = len(a)
   rng = np.random.default_rng()
   while time.time() < deadline and max_iter > 0:
       # pick src weighted by a (to bias moving from high mass)
       src = rng.choice(n, p=a)
       dst = rng.integers(n)
       if src == dst or a[src] < 1e-14:
           continue
       eps = a[src] * rng.random() * 0.2  # up to 20% of src mass
       a_prop = a.copy()
       a_prop[src] -= eps
       a_prop[dst] += eps
       dr = evaluate_sequence(a_prop.tolist())
       # accept if improvement
       if dr < best_score:
           a = a_prop
           best_score = dr
           best_a = a.copy()
           print(f"[{time.time() - start_time:6.2f}s] random-search improve: score={dr:.6f}")
       # else: optionally accept with probability according to temperature? Could skip.

"""
Hybrid multi‑scale optimiser for the step‑function bound.

The algorithm proceeds in several stages while respecting a total
time budget (default 1000 s):

1. Build a few diverse initial vectors (the previous best, uniform,
   a smooth random draw).

2. If the problem size is large, first optimise a down‑sampled version
   (≈ n/4 elements) – a short Adam phase followed by a longer
   exponentiated‑gradient (sub‑gradient) descent.  The result is then
   up‑sampled to the full size.

3. On the full size run a brief Adam smoothing phase, a long
   sub‑gradient descent, a deterministic greedy peak‑reduction and
   finally a random local search (hill‑climbing) that moves small
   amounts of mass between indices only when it lowers the max‑self‑
   convolution.

All phases use fast FFT‑based convolution and keep the vector on the
probability simplex (sum = 1, non‑negative).  The best sequence seen
throughout the run is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


def _smooth_max_grad(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth approximation (1/α)·log Σ exp(α·conv_k)
    with respect to a.
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()
    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _max_subgrad(a: np.ndarray,
                 conv: np.ndarray,
                 rel_tol: float,
                 fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    Active set = entries with conv ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray,
                deadline: float,
                start_time: float,
                fft_len: int) -> np.ndarray:
    """
    Adam optimiser on the smooth (log‑sum‑exp) approximation of the max.
    Returns the best point (in the simplex) seen during the phase.
    """
    y = np.log(np.maximum(a0, 1e-16))        # logits
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # an increasing schedule for the smoothness parameter α
    alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 4000      # a bit shorter than the original version

    best_seq = a0.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    for alpha in alpha_sched:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq
            a = _softmax_from_logits(y)

            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * a.shape[0] * f_val

            # gradient w.r.t. a of the smooth objective
            grad_a = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * a.shape[0])

            # chain rule to logits (soft‑max Jacobian)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                          f"iter={i:,}, score={sc:.6f}")

    return best_seq


def _subgradient_descent_long(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Uses a diminishing step size step0 / sqrt(it) and keeps the best
    vector seen so far.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # exact sub‑gradient of the max
        grad = _max_subgrad(a, conv, rel_tol=1e-9, fft_len=fft_len)

        # diminishing step size
        step = step0 / (math.sqrt(it) + 1.0)

        # multiplicative update (keeps non‑negativity)
        a = a * np.exp(-step * grad)

        # project back onto the simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    # final best
    return best_a, best_score


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 5000) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy moves that try to lower the current peak.
    Stops after max_no_improve consecutive failures or when time runs out.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (avoid src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.3
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


def _random_local_search(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly move a random fraction of mass from a
    source index to a destination index; keep the move only if it reduces
    the objective.  Stops when time runs out or max_iter is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # choose source proportionally to its mass (high‑mass places first)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.20   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = 2.0 * n * _conv_self(a_prop).max()
        if cur_score < best_score:
            a = a_prop
            best_a = a_prop.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:6.2f}s] random‑search improve: iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under a 1000 s time budget.
    The routine combines a multi‑scale Adam+sub‑gradient descent,
    deterministic peak‑reduction and a final random local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Determine the problem size (n) – use the provided sequence if any.
    # --------------------------------------------------------------
    try:
        starter_seq = np.asarray(height_sequence_1, dtype=float)
        if starter_seq.ndim == 1 and starter_seq.size > 0:
            n = starter_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4800
        starter_seq = None

    # --------------------------------------------------------------
    # Assemble a small collection of diverse starting points.
    # --------------------------------------------------------------
    starts = []

    if starter_seq is not None:
        s = np.maximum(starter_seq, 0.0)
        if s.sum() > 0:
            s = s / s.sum()
        starts.append(s)

    starts.append(np.full(n, 1.0 / n))                     # uniform

    # random smooth vector (Gaussian‑filtered noise)
    rng = np.random.default_rng()
    noise = rng.normal(size=n)
    smooth = np.convolve(noise, np.ones(15) / 15, mode='same')
    smooth = np.clip(smooth, 0, None)
    if smooth.sum() == 0:
        smooth = np.full(n, 1.0 / n)
    else:
        smooth /= smooth.sum()
    starts.append(smooth)

    # --------------------------------------------------------------
    # Global best tracker
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Optimise each start independently, sharing the global budget.
    # --------------------------------------------------------------
    for idx_start, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{idx_start + 1}")

        a = np.maximum(a0, 0.0)
        if a.sum() > 0:
            a = a / a.sum()
        else:
            a = np.full_like(a, 1.0 / n)

        # ----------------------------------------------------------
        # (A) Multi‑scale pre‑optimisation for large n
        # ----------------------------------------------------------
        if n > 2400:
            small_n = max(800, n // 4)
            a_small = _upsample(a, small_n)

            # brief Adam on the small problem
            phase_deadline = time.time() + min(0.05 * (deadline - start_time), 30.0)
            fft_len_small = _next_pow2(3 * small_n - 2)
            a_small = _adam_phase(a_small, phase_deadline, start_time, fft_len_small)

            # longer sub‑gradient descent on the small problem
            grad_deadline = time.time() + min(0.10 * (deadline - start_time), 120.0)
            a_small, _ = _subgradient_descent_long(a_small, grad_deadline, start_time)

            # up‑sample back to full resolution
            a = _upsample(a_small, n)

        # ----------------------------------------------------------
        # (B) Adam smoothing on the full size (short)
        # ----------------------------------------------------------
        if time.time() < deadline:
            phase_deadline = time.time() + min(0.05 * (deadline - time.time()), 30.0)
            fft_len_full = _next_pow2(3 * n - 2)
            a_adam = _adam_phase(a, phase_deadline, start_time, fft_len_full)
        else:
            a_adam = a

        # ----------------------------------------------------------
        # (C) Long exponentiated‑gradient descent on the true max
        # ----------------------------------------------------------
        if time.time() < deadline:
            sub_deadline = time.time() + 0.70 * (deadline - time.time())
            a_grad, score_grad = _subgradient_descent_long(a_adam, sub_deadline, start_time)
        else:
            a_grad = a_adam
            score_grad = evaluate_sequence(a_adam.tolist())

        # ----------------------------------------------------------
        # (D) Deterministic greedy peak‑reduction
        # ----------------------------------------------------------
        if time.time() < deadline:
            peak_deadline = time.time() + 0.15 * (deadline - time.time())
            a_peak, score_peak = _run_peak_reduction(a_grad, peak_deadline, start_time,
                                                     max_no_improve=5000)
        else:
            a_peak, score_peak = a_grad, score_grad

        # ----------------------------------------------------------
        # (E) Random local hill‑climbing search (uses remaining time)
        # ----------------------------------------------------------
        if time.time() < deadline:
            a_final, score_final = _random_local_search(a_peak, deadline, start_time,
                                                        max_iter=20000)
        else:
            a_final, score_final = a_peak, score_peak

        # ----------------------------------------------------------
        # Choose the best among the stages for this start
        # ----------------------------------------------------------
        stage_candidates = [
            (evaluate_sequence(a_adam.tolist()), a_adam),
            (score_grad, a_grad),
            (score_peak, a_peak),
            (score_final, a_final)
        ]
        local_best_score, local_best_seq = min(stage_candidates, key=lambda x: x[0])

        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST *** score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
