        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute active set

        active = np.where(conv >= thresh)[0]
        # compute subgradient g

        for k in active:

            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]
        # momentum and step size
        beta = 0.94
        momentum = beta * momentum + (1.0 - beta) * g
        step = 0.5 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)
    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) with a small value of
``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – take the provided `height_sequence_1` if it exists,
   otherwise start from the uniform distribution.  The vector is
   normalised to sum to one (the scale does not matter for the objective).

2. **Exploratory multi‑restart sub‑gradient optimisation** – the bulk of the
   time budget is spent on many short sub‑gradient runs starting from
   random Dirichlet draws (concentration is sampled uniformly from a wide
   range).  Each run performs projected sub‑gradient descent on the true
   max‑convolution objective `max_k (a*a)[k]`.  The current best solution
   is kept globally.

3. **Polishing with a smoothed‑max Adam optimiser** – the best vector found
   in step 2 is refined by a few Adam steps on the smooth log‑sum‑exp
   approximation of the max‑convolution.  This gives a fine‑grained
   improvement while still respecting the simplex constraints.

4. **Deterministic greedy peak reduction** – a cheap loop repeatedly moves a
   small fraction of the coefficient that contributes most to the current
   convolution peak towards the smallest coefficient, accepting the move
   only if the peak strictly decreases.

All phases respect the global time budget (default 1000 s).  The final
vector is projected onto the simplex one last time and returned as a plain
Python list.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (no closures, top level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical safety
        w = np.full_like(v, 1.0 / n)
    return w


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (short runs, many restarts)
# ----------------------------------------------------------------------


def _subgrad_steps(a0: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   beta: float = 0.94,
                   step0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Runs at most ``max_iters`` iterations or until ``deadline``.
    Returns the best vector seen and its score (2·n·max).
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # active set (indices that are (almost) maximal)
        thresh = cur_max * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of the max‑convolution (see analysis)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]          # each ordered pair counted

        # momentum & step‑size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (polishing)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth approximation of max(conv).  Returns the best
    probability vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – they get progressively sharper
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0]
    iters_per_alpha = 1500           # a few thousand cheap Adam steps

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max to enforce simplex
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            conv = _conv_self(a)
            scaled = α * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α        # smooth approximation of max(conv)
            obj = 2.0 * n * f_val   # smoothed objective (ignores denominator)

            # gradient of smooth max w.r.t. a
            w = np.exp(scaled - max_scaled)
            w /= w.sum()            # soft‑max weights over convolution entries
            w_rev = w[::-1]
            Af = rfft(a, n=fft_len)
            Wrev = rfft(w_rev, n=fft_len)
            corr = irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={α:.1f}, score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy peak reduction (final cheap polish)
# ----------------------------------------------------------------------


def _greedy_reduce(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Move a fraction of the most responsible coefficient towards the
    smallest coefficient while the peak strictly decreases."""
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]          # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine consists of a multi‑restart
    sub‑gradient phase, a short Adam polishing phase and a final greedy
    improvement.  It respects the global time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – keep the FFT size moderate; 4800 works well.
    # --------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=float)
        n = base_seq.shape[0] if base_seq.ndim == 1 and base_seq.size > 0 else 4800
    except Exception:
        n = 4800

    # --------------------------------------------------------------
    # Initialise best candidate (uniform fallback)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            seq = np.asarray(height_sequence_1, dtype=float)
            if seq.ndim == 1 and seq.size > 0:
                # if lengths differ, uniformly up‑sample
                if seq.shape[0] != n:
                    x_old = np.linspace(0, 1, seq.shape[0])
                    x_new = np.linspace(0, 1, n)
                    seq = np.interp(x_new, x_old, seq)
                seq = np.maximum(seq, 0.0)
                seq_sum = seq.sum()
                if seq_sum > 0:
                    seq = seq / seq_sum
                else:
                    seq = np.full(n, 1.0 / n)
                best_seq = seq
                best_score = evaluate_sequence(seq.tolist())
            else:
                raise ValueError
        except Exception:
            best_seq = np.full(n, 1.0 / n)
            best_score = evaluate_sequence(best_seq.tolist())
    else:
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"[{time.time() - start_time:6.2f}s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – many short sub‑gradient runs from random starts
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * budget_s   # ~55 % of the budget
    restart_count = 0
    while time.time() < phase1_deadline:
        restart_count += 1
        conc = rng.uniform(0.2, 3.0)                 # random concentration
        a0 = _dirichlet_start(n, conc, rng)

        # give each start a tiny amount of Adam‑style smoothing to avoid
        # pure Dirichlet spikes that tend to be poor
        a0 = _project_simplex(a0)

        # limit each sub‑gradient run to at most 20000 iterations or the
        # remaining time slice (≈10 s)
        sub_deadline = min(phase1_deadline, time.time() + 10.0)
        cand_seq, cand_score = _subgrad_steps(
            a0, max_iters=20000, deadline=sub_deadline,
            beta=0.94, step0=0.7
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (restart {restart_count}) "
                  f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – polishing with a smoothed‑max Adam optimiser
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] polishing phase (Adam) start")
        polished_seq, polished_score = _adam_optimize(
            best_seq, deadline, start_time, eps_log=1e-16
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (polish) "
                  f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic greedy peak reduction
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] greedy peak‑reduction start")
        final_seq, final_score = _greedy_reduce(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (greedy) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (protect against numerical drift) and return.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
