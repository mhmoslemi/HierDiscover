"""
Propose a step‑function (non‑negative vector) that minimises
    2·n·max(conv(a,a)) / (sum(a))**2
where conv is the ordinary linear convolution.
The implementation expands on the previous approach:

* a smooth approximation of the max (log‑sum‑exp) is optimised
  with a limited‑memory BFGS routine (SciPy).  The smoothness
  parameter α is set to a large value (5000) so the gradient points
  almost directly to the true max‑convolution.

* The variables are represented in the unconstrained log‑space and
  mapped to the probability simplex via a soft‑max; this avoids an
  explicit projection step.

* Each BFGS run is a short random restart (≈1000 function evaluations);
  the best result is kept and refined with a projected sub‑gradient
  method (the same gradient as in the reference code) and a light
  simulated‑annealing polish.

* A few deterministic start‑points (geometric decay, a simple Sidon
  set) are also examined and kept if they improve the score.

The whole search respects the 1000 s budget and prints intermediate
scores so that progress can be followed.
"""

import time
import math
import numpy as np
from scipy.optimize import fmin_l_bfgs_b
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t. the probability vector a.
    w is the soft‑max over the convolution entries (same length as conv).
    """
    n = a.shape[0]
    L = _next_pow2(3 * n)                 # safe length for the correlation
    Af = np.fft.rfft(a, n=L)
    Wrev = np.fft.rfft(w[::-1], n=L)      # reverse w for correlation
    corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w) - 1]
    grad = 2.0 * corr[len(w) - 1 - np.arange(n)]
    return grad


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) objective and its gradient (in log‑space)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(x: np.ndarray, n: int, alpha: float):
    """
    Returns (value, grad) where value = 2·n·smooth_max(conv(a,a))
    and grad is the gradient w.r.t. the unconstrained vector x.
    """
    a = _softmax(x)                              # a lies on the simplex
    conv = _conv_self(a)                         # length 2·n‑1
    scaled = alpha * conv
    M = logsumexp(scaled) / alpha                # smooth approximation of max(conv)
    value = 2.0 * n * M

    # soft‑max weights over the convolution entries
    # w sums to 1 and ∂M/∂conv = w
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # gradient w.r.t. a
    grad_a = _grad_wrt_a(a, w)

    # chain rule for the soft‑max (a = softmax(x))
    avg = np.dot(grad_a, a)
    grad_x = a * (grad_a - avg)
    return value, grad_x


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smooth surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    alpha: float,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Run a short L‑BFGS optimisation from the start vector a0.
    Returns the (probability) vector and its true (non‑smooth) score.
    """
    # initialise in log‑space; add a tiny epsilon for stability
    x0 = np.log(a0 + 1e-30)

    # we cap the number of function evaluations so that a single
    # run never exhausts the global time budget.
    maxfun = 1200

    # a tiny callback that aborts the optimisation once the deadline is hit.
    def _callback(xk):
        if time.time() >= deadline:
            raise RuntimeError("deadline reached in L‑BFGS")

    try:
        x_opt, f_opt, _ = fmin_l_bfgs_b(
            func=lambda x: _smooth_obj_and_grad(x, n, alpha),
            x0=x0,
            maxfun=maxfun,
            maxiter=maxfun,
            callback=_callback,
            disp=False,
        )
    except RuntimeError:
        # deadline triggered – fall back to the current best x
        x_opt = x0

    a_opt = _softmax(x_opt)
    true_score = 2.0 * n * _conv_self(a_opt).max()
    return a_opt, true_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  f(a)=2·n·max(conv(a,a)).
    Uses a simple momentum term and a 1/√t step‑size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.45
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # index attaining the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]                # exact sub‑gradient

        # momentum + projection onto the simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # Euclidean projection onto the simplex
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(a - theta, 0.0)
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealing. The temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source (probability ∝ current mass) and a random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move a random fraction (up to 30 %) of the source mass
        delta = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = 2.0 * n * _conv_self(a_prop).max()
        d = new_score - score

        if d < 0.0 or math.exp(-d / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score

def _geometric_start(n: int, ratio: float = 0.999) -> np.ndarray:
    """Geometric decay a[i] ∝ ratio**i, normalised to sum‑1."""
    a = ratio ** np.arange(n, dtype=float)
    a /= a.sum()
    return a


def _sidon_start(n: int) -> np.ndarray:
    """
    Greedy construction of a Sidon set (no duplicate i+j).
    All selected positions receive equal mass.
    """
    used_sums = set()
    positions = []
    for i in range(n):
        conflict = False
        for j in positions:
            if (i + j) in used_sums:
                conflict = True
                break
        if not conflict:
            # also check the self‑sum i+i
            if (2 * i) in used_sums:
                continue
            positions.append(i)
            # record all new sums i+pos
            for j in positions:
                used_sums.add(i + j)
                used_sums.add(j + i)  # redundant but harmless
    a = np.zeros(n)
    if positions:
        a[positions] = 1.0 / len(positions)
    else:
        a = np.full(n, 1.0 / n)
    return a

    a projected sub‑gradient phase and a final simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    n = 4096                                     # FFT‑friendly size used historically
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Use the global starter if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
            cur = 2.0 * n * _conv_self(starter).max()
            best_seq = starter.copy()
            best_score = cur
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Deterministic geometric decay start
    # ------------------------------------------------------------------
    geo = _geometric_start(n, ratio=0.999)
    geo_score = 2.0 * n * _conv_self(geo).max()
    if geo_score < best_score:
        best_seq = geo.copy()
        best_score = geo_score
        print(f"[{time.time() - start_time:.2f}s] Geometric start score={geo_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic Sidon set start (often surprisingly good)
    # ------------------------------------------------------------------
    sidon = _sidon_start(n)
    sidon_score = 2.0 * n * _conv_self(sidon).max()
    if sidon_score < best_score:
        best_seq = sidon.copy()
        best_score = sidon_score
        print(f"[{time.time() - start_time:.2f}s] Sidon start score={sidon_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – short L‑BFGS + sub‑grad + anneal
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:          # leave ~30 s for the final polish
        # random simplex start (Dirichlet‑like)
        a0 = np.random.rand(n)
        a0 /= a0.sum()

        # ---- Phase‑1 : L‑BFGS on the smooth surrogate -----------------
        try:
            a_opt, score_opt = _lbfgs_optimize(
                a0,
                n,
                alpha=5000.0,
                deadline=deadline - 10.0,
                start_time=start_time,
            )
        except Exception:
            a_opt, score_opt = a0.copy(), 2.0 * n * _conv_self(a0).max()

        if score_opt < best_score:
            best_seq = a_opt.copy()
            best_score = score_opt
            print(f"[{time.time() - start_time:.2f}s] L‑BFGS improve: score={score_opt:.6f}")

        # ---- Phase‑2 : projected sub‑gradient descent -----------------
        sub_deadline = deadline - 5.0
        a_sub, score_sub = _subgradient_descent(
            a_opt,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_seq = a_sub.copy()
            best_score = score_sub
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad improve: score={score_sub:.6f}")

        # ---- Phase‑3 : light annealing polish -----------------------
        if time.time() < deadline - 5.0:
            a_fin, score_fin = _annealing(
                a_sub,
                n,
                deadline,
                start_time,
                best_score,
                a_sub,
                start_temp=1e-2,
                end_temp=1e-7,
                steps=100_000,
            )
            if score_fin < best_score:
                best_seq = a_fin.copy()
                best_score = score_fin
                print(f"[{time.time() - start_time:.2f}s] Anneal improve: score={score_fin:.6f}")

    # ------------------------------------------------------------------
    # Final refinement on the best candidate discovered so far.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()

    # Extended sub‑gradient phase (uses the bulk of the remaining budget)
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + 0.7 * remaining
        a_sub, score_sub = _subgradient_descent(
            best_seq,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_seq = a_sub.copy()
            best_score = score_sub
            print(f"[{time.time() - start_time:.2f}s] Final sub‑grad improve: score={score_sub:.6f}")

    # Final annealing polish (any time left)
    if time.time() < deadline:
        a_final, score_final = _annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-7,
            steps=200_000,
        )
        if score_final < best_score:
            best_seq = a_final.copy()
            best_score = score_final
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and cast to a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
