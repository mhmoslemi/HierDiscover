def propose_candidate(...):

    allocated_time = 0.5 * (deadline - start_time)  # 50% of budget to L2 stage
    l2_deadline = time.time() + allocated_time

    max_restarts = 10
    for r in range(max_restarts):

            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] L2 improvement: restart={r}, score={cur_score:.6f}")

    # fallback if best_seq is None
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()

    # Stage subgradient descent (same as before) but with remaining budget - e.g., last 20% or 30%
    if time.time() < deadline:
        sub_deadline = deadline - 2.0
        a_opt2, score2 = _run_subgradient_descent(best_seq, n, sub_deadline, start_time, beta=0.97, step0=0.13)
        if score2 < best_score:
            best_score = score2
            best_seq = a_opt2.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improvement: score={score2:.6f}")

    # Stage annealing
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(best_seq, n, deadline, start_time, best_score, best_seq, start_temp=2e-2, end_temp=1e-7, steps=1_200_000)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # renorm and return
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:

"""
Propose a low‑score step‑function.

The search works in three phases (all respecting the 1000 s budget)

1. **L2‑flattening (Adam on Σ(conv‑μ)²).**  
   The convolution of a probability vector with itself is the distribution
   of the sum of two independent draws.  Making this distribution as flat as
   possible (μ = 1/(2·n‑1)) reduces the peak of the convolution and therefore
   the evaluation score.  The L2 loss is smooth and its gradient can be
   obtained by a single FFT‑based convolution, so we can run many Adam steps
   quickly.  Several random restarts (structured windows + Dirichlet draws)
   explore the landscape.

2. **Projected sub‑gradient descent on the true max‑convolution.**  
   After the L2 phase we polish the best candidate with a classic
   Nesterov‑style sub‑gradient method that directly minimises
   `2·n·max(conv)`.

3. **Light simulated‑annealing refinement** – random mass‑transfer moves
   with geometric cooling.  This often gets out of shallow local minima.

All intermediate vectors are projected onto the probability simplex
(Σ aᵢ = 1, aᵢ ≥ 0).  The final list is returned as plain Python floats.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa fixed)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the L2‑flattening surrogate
# ----------------------------------------------------------------------


def _run_adam_l2(a0: np.ndarray, n: int,
                 deadline: float, start_time: float,
                 step_size: float = 0.15,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8,
                 max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """
    Minimise   L(a) = Σ_k (conv_k(a) – μ)² ,   μ = 1/(2n‑1)
    using Adam with Euclidean projection onto the simplex.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    mu = 1.0 / (2 * n - 1)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)                 # linear convolution a*a
        residual = conv - mu
        loss = np.sum(residual * residual)   # L2 loss (for bookkeeping)

        # true score (what evaluate_sequence would return)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] L2‑adam improve: "
                  f"iter={it:,}, loss={loss:.6e}, score={cur_score:.6f}")

        # gradient of L2 loss w.r.t. a
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, residual)          # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]               # Σ_j a_j·residual_{i+j}
        grad = 4.0 * inner                               # factor 4 from derivation

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    Uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Structured initialisation helpers
# ----------------------------------------------------------------------


def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    40 % Chebyshev window, 40 % Kaiser window, else plain Dirichlet draw.
    """
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
    elif have_signal and r < 0.8:
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
    else:
        # Dirichlet‑type draw with a random concentration parameter
        alpha = rng.uniform(0.2, 2.0, size=n)
        w = rng.gamma(alpha, 1.0)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The method consists of:

      1. several Adam runs on the L2‑flattening surrogate,
      2. a projected sub‑gradient descent on the true max‑convolution,
      3. a light simulated‑annealing polishing stage.

    The returned object is a plain list of floats (≈ simplex).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # Length of the sequence – we keep the tried‑and‑true 8192 for speed.
    # ------------------------------------------------------------------
    n = 8192

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous successful run
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Phase 1 – L2‑flattening (Adam) with several restarts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_time = max(30.0, 0.45 * remaining)   # about 45 % of the total budget
    phase1_deadline = time.time() + phase1_time

    max_restarts = 12
    for r in range(max_restarts):
        if time.time() >= phase1_deadline:
            break

        if starter is not None and r == 0:
            a0 = starter.copy()
        else:
            a0 = _window_initialiser(n, rng)

        a_opt, score = _run_adam_l2(
            a0, n, phase1_deadline, start_time,
            step_size=0.22,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase 1 improve: "
                  f"restart={r}, score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = deadline - 2.0          # keep a couple seconds for SA
        a_opt, score = _run_subgradient_descent(
            best_seq, n, subgrad_deadline, start_time,
            beta=0.97,
            step0=0.13,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase 2 improve: "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score = _run_annealing(
            best_seq, n, deadline, start_time,
            best_score, best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=1_200_000,            # a bit longer than the reference
        )
        if score < best_score:
            best_score = score
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation onto the simplex (guard against tiny drift)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list – the harness will call evaluate_sequence
    return [float(x) for x in best_seq]
