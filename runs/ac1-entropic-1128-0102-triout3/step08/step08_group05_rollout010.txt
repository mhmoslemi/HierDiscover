def O_and_grad(z, n, tau):
    a = softmax(z)
    conv = conv_self(a)  # length 2n-1
    # Compute softmax over conv/tau
    zc = conv / tau
    max_zc = np.max(zc)
    exp_zc = np.exp(zc - max_zc)
    sum_exp = np.exp(zc - max_zc).sum()
    p = exp_zc / sum_exp  # shape 2n-1

    # gradient w.r.t a
    a_rev = a[::-1]
    conv_rev = conv_fft(a_rev, p)  # length 3n-2
    inner = conv_rev[n-1:n-1+n]
    grad_a = 4.0 * n * inner  # shape n

    # compute O
    O = 2.0 * n * tau * (max_zc + np.log(sum_exp))  # because log sum exp = max + log(sum exp)
    # Equivalent: O = 2*n*tau*np.log(np.exp(zc).sum()) but stable compute
    # Actually O = 2*n*tau*np.log(np.exp(zc).sum()).
    # Because we used max_zc subtraction, we should add max_zc back: O = 2*n*tau * (max_zc + np.log(sum_exp))

    # gradient w.r.t z
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    return O, grad_z
```

```
class ObjGrad:
    def __init__(self, n, tau):
        self.n = n
        self.tau = tau
    def __call__(self, z):
        a = softmax(z)
        conv = _conv_self(a)
        # compute O
        zc = conv / self.tau
        max_zc = np.max(zc)
        exp_zc = np.exp(zc - max_zc)
        sum_exp = exp_zc.sum()
        obj = 2.0 * self.n * self.tau * (max_zc + np.log(sum_exp))
        # compute gradient wrt a
        p = exp_zc / sum_exp
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)
        inner = conv_rev[self.n-1:self.n-1+self.n]
        grad_a = 4.0 * self.n * inner
        # gradient w.r.t z
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        return obj, grad_z
```

"""
Search for a non‑negative step‑function that minimises the
max‑self‑convolution objective used by `evaluate_sequence`.

Algorithm
---------
1.  Initialise a probability vector `a` (sum = 1).  If a previous
    construction `height_sequence_1` exists it is up‑sampled, otherwise
    a random vector is used.
2.  Parameterise the vector as a soft‑max of unconstrained real
    variables `z` (i.e. a = softmax(z)).  This automatically enforces
    non‑negativity and the simplex constraint.
3.  Minimise the smooth surrogate

        O_τ(a) = 2·n·τ·log∑_k exp(conv(a)_k / τ)

    using L‑BFGS‑B on the variables `z`.  The gradient of `O_τ` is
    obtained analytically via FFT‑based convolutions.  A continuation
    scheme reduces the temperature `τ` geometrically, warm‑starting each
    L‑BFGS run with the solution from the previous stage.
4.  After the continuation phase a short projected sub‑gradient descent
    (Euclidean projection onto the simplex) and a few exponentiated‑
    gradient steps are performed to fine‑tune the solution.
5.  Finally a very light simulated‑annealing mass‑transfer refinement
    is applied for the last fraction of the time budget.
6.  The best vector found (normalised to sum ≈ 1) is returned as a plain
    Python list.  All operations respect the global time budget
    (default 1000 s).

The code uses only NumPy and SciPy (L‑BFGS) and contains no
closures or lambda functions, satisfying the execution constraints.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i≥0, Σx_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT based linear convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) surrogate and its gradient (L‑BFGS objective)
# ----------------------------------------------------------------------


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


class ObjGrad:
    """
    Callable class returning the smooth surrogate O_τ(a) and its gradient
    w.r.t. the soft‑max parameters `z` (where a = softmax(z)).
    """

    def __init__(self, n: int, tau: float):
        self.n = n
        self.tau = tau

    def __call__(self, z: np.ndarray):
        # a is always on the simplex
        a = _softmax(z)

        # self‑convolution
        conv = _conv_self(a)                     # length 2·n‑1
        zc = conv / self.tau
        max_zc = np.max(zc)

        # stable log‑sum‑exp
        exp_zc = np.exp(zc - max_zc)
        sum_exp = exp_zc.sum()
        obj = 2.0 * self.n * self.tau * (max_zc + np.log(sum_exp))

        # probabilities for the surrogate gradient
        p = exp_zc / sum_exp                     # shape 2·n‑1, sum(p)=1

        # gradient w.r.t. a (see analysis)
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)          # length 3·n‑2
        inner = conv_rev[self.n - 1:self.n - 1 + self.n]
        grad_a = 4.0 * self.n * inner            # shape n

        # chain rule: dz = a * (grad_a - <a,grad_a>)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        return obj, grad_z


# ----------------------------------------------------------------------
# Sub‑gradient of the true (non‑smoothed) objective
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on  F(a)=2·n·max(conv(a))."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient mirror descent on the true objective
# ----------------------------------------------------------------------


def _run_exponentiated_gradient(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               step0: float = 0.20,
                               max_iter: int = 5 * 10 ** 6) -> tuple[np.ndarray, float]:
    """Mirror‑descent with the entropy Bregman divergence."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        eta = step0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve (n={n}) "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def _run_lbfgs_continuation(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float) -> tuple[np.ndarray, float]:
    """
    Runs a geometric τ‑continuation, warm‑starting L‑BFGS at each stage.
    Returns the best vector (as a probability distribution) and its
    true objective value 2·n·max(conv).
    """
    a = a0.copy()
    conv0 = _conv_self(a)
    maxc0 = conv0.max()
    # start with a temperature safely above the current max
    tau = maxc0 * 10.0 if maxc0 > 0 else 1e-3

    best_a = a.copy()
    best_score = 2.0 * n * maxc0
    stage = 0

    while time.time() < deadline and tau > 1e-12:
        stage += 1
        # allocate a modest slice of the remaining time
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.20, max(5.0, remaining * 0.05))
        slice_deadline = time.time() + slice_time

        # number of L‑BFGS iterations – we cap it to keep within slice_time
        # estimate roughly 0.004 s per iteration for n≈4096 (empirical)
        max_iter = max(10, int(slice_time / 0.004))
        max_iter = min(max_iter, 300)

        # initialise L‑BFGS on the soft‑max parameters
        z0 = np.log(best_a + 1e-12)
        objgrad = ObjGrad(n, tau)

        try:
            res = opt.minimize(objgrad,
                               z0,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': max_iter,
                                        'disp': False})
        except Exception as e:
            print(f"[{time.time() - start_time:6.2f}s] L‑BFGS failure: {e}")
            break

        a_new = _softmax(res.x)
        cur_score = 2.0 * n * _conv_self(a_new).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:6.2f}s] L‑BFGS stage {stage} τ={tau:.2e} improve: {cur_score:.6f}")

        # optional cheap sub‑gradient polishing (few hundred steps)
        if time.time() + 2.0 < slice_deadline:
            a_tmp, sub_score = _run_subgradient_descent(
                a_new, n, slice_deadline - 1.0, start_time,
                beta=0.95, step0=0.20, max_iter=500)
            if sub_score < best_score:
                best_score = sub_score
                best_a = a_tmp.copy()
                print(f"[{time.time() - start_time:6.2f}s] post‑L‑BFGS subgrad improve: {sub_score:.6f}")

        # reduce temperature geometrically
        tau *= 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine follows a continuation‑L‑BFGS phase and finishes with
    projected sub‑gradient / exponentiated‑gradient / annealing refinements.
    All work is bounded by `budget_s` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # keep a small safety margin

    # ------------------------------------------------------------------
    # Choose problem size.  Larger n gives a better bound but each iteration
    # costs more.  With the time budget we target n≈4096 (or 8192 if we have
    # ample time left after the continuation phase).
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Initialise – use a previous construction if available, otherwise random.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            starter = np.asarray(globals()["height_sequence_1"], dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                a0 = _upsample(starter, n)
            else:
                raise ValueError
        except Exception:
            a0 = np.random.rand(n)
            a0 /= a0.sum()
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Phase 1 – Continuation L‑BFGS on the smooth log‑sum‑exp surrogate.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_best, best_score = _run_lbfgs_continuation(
            a0, n, deadline, start_time)
    else:
        a_best = a0.copy()
        best_score = 2.0 * n * _conv_self(a_best).max()

    # ------------------------------------------------------------------
    # Phase 2 – Short projected sub‑gradient descent (fine‑tune).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = deadline - 2.0
        a_best, sg_score = _run_subgradient_descent(
            a_best, n, sub_deadline, start_time,
            beta=0.95, step0=0.20, max_iter=2000)
        if sg_score < best_score:
            best_score = sg_score
            print(f"[{time.time() - start_time:6.2f}s] subgrad final improve: {sg_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient mirror descent (optional).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 1.0
        a_best, eg_score = _run_exponentiated_gradient(
            a_best, n, eg_deadline, start_time, step0=0.15)
        if eg_score < best_score:
            best_score = eg_score
            print(f"[{time.time() - start_time:6.2f}s] EG final improve: {eg_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Very light simulated‑annealing mass‑transfer.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_best, sa_score = _run_annealing(
            a_best, n, deadline, start_time,
            cur_best_score=best_score,
            cur_best_seq=a_best,
            start_temp=2e-2, end_temp=1e-7, steps=200_000)
        if sa_score < best_score:
            best_score = sa_score
            print(f"[{time.time() - start_time:6.2f}s] anneal final improve: {sa_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (numerical drift) and return as plain list.
    # ------------------------------------------------------------------
    a_best = np.asarray(a_best, dtype=float)
    total = a_best.sum()
    if total > 0.0:
        a_best /= total
    else:
        a_best = np.full_like(a_best, 1.0 / a_best.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in a_best]
