"""
Hybrid parametric + subgradient optimizer for the max‑self‑convolution bound.
1. Scan a two‑parameter family a_i ∝ exp(−β·d_i)·(d_i+1)^{−α} (d_i distance to nearest edge)
   and keep the best shape.
2. Starting from that shape run a long exponentiated‑gradient (sub‑gradient) descent on the
   exact max‑convolution, always projecting onto the probability simplex.
3. Return the best sequence found before the time budget expires.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step function that minimizes ``evaluate_sequence``.
    The method first explores a small two‑parameter family of edge‑heavy
    shapes, then refines the best candidate with a long exponentiated‑gradient
    descent on the exact max convolution.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # leave a tiny safety margin

    # --------------------------------------------------------------
    # Determine length n (use supplied sequence if present)
    # --------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=float)
        if base_seq.ndim == 1 and base_seq.size > 0:
            n = base_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4800

    # Helpers for the parametric family
    # --------------------------------------------------------------
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))   # distance to closest edge

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1. Coarse grid search over (α,β)
    # --------------------------------------------------------------
    alphas = np.linspace(0.0, 2.5, 27)   # 0 … 2.5 inclusive
    betas = np.linspace(0.0, 0.005, 21)  # 0 … 0.005

        for beta in betas:
            if time.time() >= deadline:
                break

"""
Hybrid parametric + sub‑gradient optimizer for the max‑self‑convolution bound.

Algorithm
---------
1. Scan a simple two‑parameter family of edge‑heavy distributions
   a_i ∝ exp(−β·d_i)·(d_i+1)^{−α},
   where d_i = min(i, n‑1‑i) is the distance to the nearest edge.
   The best shape from this coarse grid is kept.

2. Starting from that shape run a long exponentiated‑gradient (sub‑gradient)
   descent on the exact max‑convolution.  At each iteration we
   * compute the full convolution by FFT,
   * build the active set of indices where the convolution is within 1e‑8 of the
     current maximum,
   * obtain the exact sub‑gradient via a cross‑correlation,
   * perform a multiplicative update a ← a·exp(−step·grad) and project back onto
     the probability simplex.

3. The best sequence seen throughout the run is returned (scaled back to a
   proper probability vector; the scaling does not affect the evaluated score).

The whole procedure respects the supplied time budget (default 1000 s) and uses
only NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w
