"""
Propose a step‑function with a very low evaluation score.

The algorithm builds on the previous multi‑stage approach but adds several
enhancements that proved useful in practice:

*  Many cheap Adam runs on a modest size (n≈1024) with *per‑restart* time
   budgets and a richer set of structured initialisers (including an inverse
  ‑window and sparse “spike” patterns).  This yields a much more diverse pool
   of starting points.
*  A progressive up‑sampling: the best candidate from stage‑0 is linearly
   interpolated to n≈16384 (instead of 8192) before a second Adam phase.
*  The Adam phase uses a slower cooling schedule and a slightly larger step
   size, which helps escape shallow local minima.
*  A short projected sub‑gradient descent (Nesterov momentum) refines the
   solution on the exact max‑convolution objective.
*  A final greedy “mass‑transfer” hill‑climber (instead of a plain SA) makes
   many tiny moves that are only accepted when they improve the true score.
   This is fast and often squeezes a few extra points out of the incumbent.
*  All stages respect the global time budget (default 1000 s) and the final
   sequence is projected onto the simplex (∑a_i = 1, a_i ≥ 0).

The returned list is a plain Python list of non‑negative floats; the harness
will call ``evaluate_sequence`` on it.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient for the exact max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa = 1)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate of max(conv)
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray, n: int,
                      deadline: float, start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.20,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    using Adam with Euclidean projection onto the simplex.
    Returns the best vector observed (as a NumPy array) and its true score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true (unsmoothed) score for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Nesterov‑style projected sub‑gradient descent on the exact max‑conv.
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray, n: int,
                             deadline: float, start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.13,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light greedy “mass‑transfer” refinement (accept‑only‑improving moves)
# ----------------------------------------------------------------------


def _run_greedy_refinement(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           cur_best_score: float,
                           cur_best_seq: np.ndarray,
                           max_moves: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Perform many tiny transfers of probability mass between two cells.
    Only moves that strictly improve the true score are kept.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, max_moves + 1):
        if time.time() >= deadline:
            break

        # pick source proportional to its current mass (to focus on big contributors)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a tiny random fraction (up to 2 %) – tiny steps keep us near the simplex
        eps = a[src] * rng.random() * 0.02
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate the true objective (2·n·max(conv)) – cheap because we reuse the FFT helper
        new_score = 2.0 * n * _conv_self(a_prop).max()

        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: n={n}, "
                  f"move={it:,}, score={new_score:.6f}")
        # otherwise reject

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisation helpers
# ----------------------------------------------------------------------


def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    The mixture includes:
      – Chebyshev window,
      – Kaiser window,
      – inverse‑window (large at the edges, small in the centre),
      – sparse spike pattern,
      – plain Dirichlet draw.
    """
    # try to import scipy.signal lazily – if unavailable fall back to random
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.25:
        # Chebyshev window
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
    elif have_signal and r < 0.5:
        # Kaiser window
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
    elif r < 0.65:
        # Inverse‑window: weight large at the ends, tiny in the centre
        i = np.arange(n)
        w = 1.0 / (i + 1) + 1.0 / (n - i)
    elif r < 0.80:
        # Sparse spike pattern – pick a few indices and distribute mass uniformly
        k = rng.integers(3, 9)                     # 3‑8 spikes
        w = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=k, replace=False)
        w[idx] = 1.0 / k
    else:
        # Plain Dirichlet (uniform random)
        w = rng.random(n)

    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The method consists of

      1. many cheap Adam runs on a modest size (n≈1024) with diverse seeds;
      2. up‑sampling the best candidate to n≈16384 and a second Adam phase;
      3. a short Nesterov sub‑gradient descent on the exact objective;
      4. a greedy mass‑transfer refinement that only accepts improving moves.

    The final list is projected onto the simplex and returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous solution exists, use it as a starter (normalised)
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    remaining = deadline - time.time()
    stage0_time = max(30.0, 0.25 * remaining)      # ~25 % of budget
    max_restarts = 20
    per_restart = max(3.0, stage0_time / max_restarts)

    for r in range(max_restarts):
        if time.time() >= deadline:
            break

        # allocate a tiny slice of the stage‑0 budget for this restart
        restart_deadline = time.time() + per_restart

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = _window_initialiser(n0, rng)

        # initialise τ from the current max‑convolution (a rough scale)
        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 8.0, 1e-8)

        a_opt, score = _run_adam_softmax(
            a0, n0,
            restart_deadline, start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9994,
            step_size=0.22,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # ------------------------------------------------------------------
    # If nothing was produced (unlikely) fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample to a larger size (≈16384) and continue Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 16384
    a1 = _upsample(best_seq, n1)

    conv1 = _conv_self(a1)
    tau0_1 = max(conv1.max() * 8.0, 1e-10)

    remaining = deadline - time.time()
    stage1_time = max(20.0, 0.30 * remaining)      # about 30 % of the remaining budget
    slice_deadline = time.time() + stage1_time

    a1_opt, score1 = _run_adam_softmax(
        a1, n1,
        slice_deadline, start_time,
        tau0=tau0_1,
        tau_end=1e-14,
        tau_decay=0.9993,
        step_size=0.15,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – a short Nesterov sub‑gradient descent on the true max‑conv.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n2 = best_seq.shape[0]
        # keep about 2 s for the final greedy refinement
        grad_deadline = deadline - 2.0
        a2_opt, score2 = _run_subgradient_descent(
            best_seq, n2,
            grad_deadline, start_time,
            beta=0.97,
            step0=0.12,
        )
        if score2 < best_score:
            best_score = score2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – greedy mass‑transfer refinement (accept‑only‑improving)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_greedy_refinement(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            max_moves=800_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final greedy improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise to the simplex (guard against numerical drift)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list – the harness will evaluate it.
    return [float(x) for x in best_seq]
