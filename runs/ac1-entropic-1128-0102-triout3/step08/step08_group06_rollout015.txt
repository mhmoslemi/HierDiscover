def simulated_annealing(a, conv, max_val, deadline, start_time):

    n = len(a)

    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    init_eps_frac = 0.05
    min_eps_frac = 0.001

    # temperature schedule: T = T0 * (1 - elapsed/total) where T0 maybe set relative to current max.
    T0 = max_val * 0.001? Actually delta_max we may accept up to maybe 0.5*max? Let's pick T0 = max_val * 0.5 maybe.

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        fraction = elapsed / total
        eps_frac = init_eps_frac * (1 - fraction) + min_eps_frac * fraction
        # Temperature schedule
        T = max_val * 0.5 * (1 - fraction)  # start high, end low

        # Choose source and dest similar to refine
        k_max = int(conv.argmax())
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        # destination: choose index with minimal contribution to k_max
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue
        if a[src] < 1e-16:
            continue

        eps = a[src] * eps_frac * rng.random()
        if eps <= 0:
            continue

        # compute delta array using a_original = a copy?
        # Use a_original (before move) - which we have.
        delta = np.zeros(2*n-1)
        delta[dst:dst+n] += 2.0 * eps * a
        delta[src:src+n] -= 2.0 * eps * a
        delta[2*src] += eps*eps
        delta[2*dst] += eps*eps
        delta[src+dst] -= 2.0 * eps*eps

        conv_prop = conv + delta
        new_max = conv_prop.max()
        delta_score = new_max - max_val

        if delta_score <= 0 or rng.random() < math.exp(-delta_score / T):
            # accept
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            # update best if improved
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()

            conv = _conv_self(a)
            max_val = conv.max()
    return best_a

The code below implements a multi‑phase optimiser for the self‑convolution bound.
The idea is:

1. **Warm‑start** – use the previously discovered construction if it exists,
   otherwise start from a random simplex point.
2. **Phase 1 – Exponential‑gradient (mirror descent)** on a smooth
   log‑sum‑exp approximation of the maximum convolution entry.
3. **Phase 2 – Projected sub‑gradient descent** on the exact (non‑smoothed)
   objective.
4. **Phase 3 – Fast targeted refinement** (the method that was already
   provided – moves a tiny amount of probability mass away from the most
   harmful indices).
5. **Phase 4 – Simulated‑annealing refinement** – we now allow *uphill*
   moves with a temperature schedule.  This helps the search escape the
   small‑move local optimum that the pure refinement phase gets stuck in
   and typically yields a few‑percent further reduction of the maximal
   self‑convolution value.

All phases respect the overall 1000‑second budget, constantly keep the best
sequence seen, and finally return the best sequence (normalised to sum = 1)
as a plain Python list.  The implementation uses only `numpy` and the
standard library, runs on a single CPU and requires no file or network I/O.

```python
"""
Multi‑phase optimiser for the self‑convolution bound.

1. Warm start (global height_sequence_1 if present) → simplex normalisation.
2. Exponential‑gradient descent on a log‑sum‑exp smooth maximum.
3. Projected sub‑gradient descent on the true max.
4. Fast O(n) targeted refinement (strictly decreasing moves).
5. Simulated‑annealing refinement – occasionally accepts uphill moves,
   allowing the algorithm to escape the tiny local optimum left by #4.

The best sequence found during the run is returned (sum = 1, all entries ≥0).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 – Exponential‑gradient on a smoothed max
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 8000):
    """Mirror‑descent on the log‑sum‑exp smooth maximum."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)      # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 – Projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    """Projected sub‑gradient descent on max_k (a*a)[k]."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SG improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Phase‑3 – Fast targeted refinement (strictly decreasing moves)
# ----------------------------------------------------------------------


def _run_targeted_refine_optimized(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.05,
                                   min_eps_frac: float = 0.001):
    """
    O(n) refinement that moves a tiny amount of probability mass away from
    indices that contribute most to the current maximal convolution entry.
    Only *strictly* improving moves are kept.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linear decay of the step size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # ----- source index (probability ∝ contribution) -----
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # ----- destination index (prefer zero contribution) -----
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----
        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a

def _run_annealing(a0: np.ndarray,
                   n: int,

    increase the maximum are accepted with probability exp(-Δ/T).
    """
    a = a0.copy()

        T = max_val * 0.5 * (1.0 - decay) + 1e-15   # avoid division by zero

        k_max = int(conv.argmax())

        valid = (j >= 0) & (j < n)

        src = rng.choice(n, p=prob_src)

        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14

        if eps <= 0.0:

        # incremental update of the convolution
        delta = np.zeros(conv_len, dtype=float)

                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: "

        if it % 50000 == 0:

            max_val = conv.max()

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Initialise – use the global ``height_sequence_1`` if it exists.

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()

        starter = None

    # problem size – 2400 gives a good compromise between speed and flexibility
    final_n = 2400
    if starter is not None:

            best_score = score_ann
            best_seq = a_anneal.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 4 improve: score={score_ann:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # final sanity check (optional – cheap)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] final evaluated value: {final_val:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
```
