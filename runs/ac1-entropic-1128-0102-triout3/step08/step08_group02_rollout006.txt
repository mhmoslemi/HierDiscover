"""
Hybrid sub‑gradient optimiser for the max‑convolution bound.

The algorithm works in three stages within the 1000 s budget:

1. **Initial candidates** – a handful of simple shapes (uniform,
   exponential, U‑shape, random smooth) and the supplied
   `height_sequence_1` (if any) are normalised and the one with the
   smallest initial `evaluate_sequence` value is selected.

2. **Long exponentiated‑gradient descent** – starting from the best
   candidate we perform multiplicative (exponentiated) updates using
   the exact sub‑gradient of the max‑convolution objective.
   The step size is large at the beginning (≈0.7) and decays as
   1/√t.  If no improvement is observed for a while a small random
   perturbation is added to help escape plateaus.

3. **Deterministic peak reduction** – a short greedy pass that moves a
   fraction of the mass from the coefficient most responsible for the
   current peak to the smallest coefficient, trying to lower the peak
   further.

All steps respect the overall time budget and return the best
non‑negative sequence found (scaled to sum = 1).  The returned list
can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _max_subgrad(a: np.ndarray,
                 conv: np.ndarray,
                 rel_tol: float,
                 fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    Active set contains all indices with conv ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)

    # cross‑correlation a ⋆ active (gradient = 2· that)
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """Greedy moves that try to lower the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        # indices (i,j) that contribute to that peak
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]

        # source = index with the largest contribution in the active pair
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # move a modest fraction of mass
        eps = a[src] * 0.3
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _subgradient_descent_full(a0: np.ndarray,
                              deadline: float,
                              start_time: float) -> tuple[np.ndarray, float]:
    """
    Long exponentiated‑gradient descent on the true max‑convolution.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    cur_conv = _conv_self(a)
    best_score = 2.0 * n * cur_conv.max()

    it = 0
    no_improve = 0
    max_no_improve = 8000          # after this we inject a tiny perturbation

    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: iter={it:,}, score={cur_score:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of the max (tight active set)
        grad = _max_subgrad(a, conv, rel_tol=1e-7, fft_len=fft_len)

        # step size – large initially, decays as 1/√t
        step = 0.7 / math.sqrt(it)

        # multiplicative update (always stays non‑negative)
        a = a * np.exp(-step * grad)

        # projection onto the simplex (sum = 1)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # occasional random jitter if stuck for a while
        if no_improve >= max_no_improve:
            perturb = rng.normal(scale=0.005, size=n)
            perturb = np.where(perturb < 0, np.maximum(perturb, -a), perturb)  # keep non‑negative
            a = a + perturb
            a = np.clip(a, 0, None)
            total = a.sum()
            if total > 0:
                a /= total
            no_improve = 0

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        starter_seq = np.asarray(height_sequence_1, dtype=float)
        if starter_seq.ndim == 1 and starter_seq.size > 0:
            n = starter_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4800
        starter_seq = None

    # --------------------------------------------------------------
    # Build a small pool of diverse initial vectors (normalised)
    # --------------------------------------------------------------
    starts = []

    if starter_seq is not None:
        s = starter_seq.copy()
        s = np.maximum(s, 0.0)
        ss = s.sum()
        if ss > 0:
            s /= ss
        else:
            s = np.full(n, 1.0 / n)
        starts.append(s)

    starts.append(np.full(n, 1.0 / n))                     # uniform

    # left‑hand exponential decay
    r = 0.985
    exp_vec = np.power(r, np.arange(n, dtype=float))
    starts.append(exp_vec / exp_vec.sum())

    # symmetric “U‑shape” (large at the edges, small in centre)
    idx = np.arange(n, dtype=float)
    u_shape = 1.0 / (np.minimum(idx + 1, n - idx) + 1.0)
    starts.append(u_shape / u_shape.sum())

    # random smooth vector (Gaussian‑filtered noise)
    rng = np.random.default_rng()
    noise = rng.normal(size=n)
    smooth = np.convolve(noise, np.ones(31) / 31, mode='same')
    smooth = np.clip(smooth, 0, None)
    if smooth.sum() > 0:
        smooth /= smooth.sum()
    else:
        smooth = np.full(n, 1.0 / n)
    starts.append(smooth)

    # --------------------------------------------------------------
    # Pick the best among the simple starts (cheap evaluation)
    # --------------------------------------------------------------
    best_start = None
    best_start_score = float('inf')
    for idx, a in enumerate(starts):
        if time.time() >= deadline:
            break
        s = evaluate_sequence(a.tolist())
        print(f"[{time.time() - start_time:6.2f}s] init start {idx+1:02d} score={s:.6f}")
        if s < best_start_score:
            best_start_score = s
            best_start = a.copy()

    if best_start is None:
        best_start = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Main optimisation – long sub‑gradient descent
    # --------------------------------------------------------------
    sub_deadline = deadline - 0.3   # reserve a bit for the final tidy‑up
    a_opt, opt_score = _subgradient_descent_full(best_start, sub_deadline, start_time)

    # --------------------------------------------------------------
    # Final short greedy peak‑reduction (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, opt_score = _run_peak_reduction(a_opt, deadline, start_time)

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    a_opt = np.maximum(a_opt, 0.0)
    total = a_opt.sum()
    if total > 0:
        a_opt /= total
    else:
        a_opt = np.full_like(a_opt, 1.0 / n)

    final_score = evaluate_sequence(a_opt.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in a_opt.tolist()]
