"""
Multi‑phase optimizer for the self‑convolution bound.

The algorithm works as follows:
1. Warm‑up with exponentiated‑gradient descent on a smooth log‑sum‑exp
   approximation of the convolution maximum.
2. Fast targeted local refinement that moves a tiny amount of probability
   mass away from indices that contribute most to the current maximal entry.
3. Simulated‑annealing style stochastic mass‑moves that are allowed to
   occasionally increase the objective, helping to escape shallow local minima.
   The annealing schedule slowly reduces the allowed step size and temperature.

The sequence length is increased (upsampled) from any provided warm‑start
(`height_sequence_1`) to a larger size (default 7200) which gives the optimizer
more degrees of freedom.  The best sequence found within the time budget is
returned as a list of non‑negative heights (already normalised to sum = 1).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    Returns a non‑negative array (tiny negatives are clipped to 0).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negative artefacts caused by floating point noise.
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) for a distribution with sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule=None,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 10000):
    """Multiplicative mirror‑descent on the log‑sum‑exp smooth max."""
    if alpha_schedule is None:
        alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]

    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    fft_len = _next_pow2(3 * n - 2)   # large enough for a * w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # --- smooth‑max weights (log‑sum‑exp) ---
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        # gradient of the smooth max w.r.t. a (mirror‑descent)
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update (mirror descent on the simplex)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – fast targeted local refinement
# ----------------------------------------------------------------------
def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         init_eps_frac: float = 0.05,
                         min_eps_frac: float = 0.001):
    """
    O(n) refinement that moves a tiny probability mass away from indices that
    contribute most to the current maximal convolution entry.
    Accepts only strict improvements.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linear decay of allowed step size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # ---- source index weighted by its contribution to the current max ----
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # ---- destination index – prefer indices that *do not* contribute ----
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # ---- propose a move ----
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # --- fast O(n) update of the convolution (exact up to round‑off) ---
        delta = np.zeros(conv_len, dtype=float)

        # linear part 2*eps*(a_{k-dst} - a_{k-src})
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a

        # quadratic eps² correction (only three entries are affected)
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # periodic full recomputation to avoid drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a

# Phase 3 – simulated annealing with mass‑moves
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   init_eps_frac: float = 0.04,
                   min_eps_frac: float = 0.001,
                   init_temp: float = 5e-4,
                   final_temp: float = 1e-7):
    """
    Stochastic optimisation that occasionally accepts uphill moves.
    Operates on the same O(n) move update as the refinement phase but with a
    temperature schedule to escape local minima.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = min(1.0, elapsed / total)

        # temperature and epsilon schedule (linear decay)
        temperature = init_temp * (1.0 - progress) + final_temp * progress
        eps_frac = init_eps_frac * (1.0 - progress) + min_eps_frac * progress

        # current maximal index
        k_max = int(conv.argmax())

        # source probabilities – proportional to contribution to the max entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # destination – favour indices that *do not* contribute to the max entry
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # ---- fast update of the convolution ----
        new_conv = conv.copy()
        new_conv[dst:dst + n] += 2.0 * eps * a
        new_conv[src:src + n] -= 2.0 * eps * a
        new_conv[2 * src] += eps * eps
        new_conv[2 * dst] += eps * eps
        new_conv[src + dst] -= 2.0 * eps * eps

        new_max = new_conv.max()
        delta = new_max - cur_max

        # Metropolis acceptance
        if delta <= 0 or rng.random() < math.exp(-delta / max(temperature, 1e-12)):
            # accept
            a[src] -= eps
            a[dst] += eps
            conv = new_conv
            cur_max = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recompute to keep numerical noise in check
        if it % 40000 == 0:
            conv = _conv_self(a)
            cur_max = conv.max()

    return best_a

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under a 1000 s time budget.

    The routine follows three stages:
      1. Exponentiated‑gradient warm‑up.
      2. Fast targeted refinement.
      3. Simulated‑annealing stochastic optimisation.

    The best sequence discovered throughout the run is returned, normalised
    to have unit sum.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    #  Initialise – use the global ``height_sequence_1`` if it exists.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # Choose a final length considerably larger than the previous 4800.
    # Larger n gives the optimiser more freedom and usually yields a
    # slightly smaller constant.
    final_n = 7200

    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = _score(a0)

    # --------------------------------------------------------------
    # Phase 1 – exponentiated gradient (quick warm‑up)
    # --------------------------------------------------------------
    phase1_time = min(30.0, 0.05 * (deadline - time.time()))
    phase1_deadline = time.time() + phase1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.5,
        steps_per_alpha=8000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_warm:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – fast targeted refinement (deterministic local search)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = deadline - 0.3 * (deadline - time.time())  # keep ~30 % for annealing
        a_refined = _run_targeted_refine(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            init_eps_frac=0.05,
            min_eps_frac=0.001,
        )
        score_ref = _score(a_refined)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing (stochastic escape from local minima)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_annealed = _run_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            init_eps_frac=0.04,
            min_eps_frac=0.001,
            init_temp=5e-4,
            final_temp=1e-7,
        )
        score_ann = _score(a_annealed)
        if score_ann < best_score:
            best_score = score_ann
            best_seq = a_annealed.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve: score={score_ann:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # Return as a plain Python list of floats
    return [float(x) for x in best_seq]
