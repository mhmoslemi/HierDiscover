def _random_local_search(a0, conv0, deadline, start_time, max_iters=200000):
    a = a0.copy()
    conv = conv0.copy()
    cur_max = conv.max()
    n = a.shape[0]
    conv_len = conv.shape[0]
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # choose source index weighted by a (prefer larger)
        if a.sum() == 0:
            break
        src = rng.choice(n, p=a / a.sum())
        # choose dest from low-valued indices (maybe random among bottom 10%)
        # we can pick dest uniformly from n; but try to ensure diff.
        dst = rng.integers(n)
        if dst == src:
            continue
        # epsilon random fraction
        eps = rng.random() * a[src]  # up to full mass
        if eps < 1e-12:
            continue
        # compute shift vectors
        shift_src = np.zeros(conv_len)
        shift_src[src:src+n] = a
        shift_dst = np.zeros(conv_len)
        shift_dst[dst:dst+n] = a
        new_conv = conv + 2 * eps * (shift_dst - shift_src)
        new_max = new_conv.max()
        if new_max < cur_max - 1e-12:
            # accept move
            a[src] -= eps
            a[dst] += eps
            # update conv and cur_max
            conv = new_conv
            cur_max = new_max
            # maybe print improvement
            if it % 1000 == 0:
                print(f"[{time.time() - start_time:6.2f}s] rand search improvement iter={it} new_max={new_max:.6f}")
    return a, cur_max

"""
Propose a non‑negative step function that (almost) minimises ``evaluate_sequence``.

Algorithm
---------

* **Exploration** – several diverse starts are optimised by a soft‑max Adam
  scheme.  The smoothing parameter α is increased slowly (up to 2 048) and a
  modest number of Adam steps are performed.  The best candidate from this
  cheap phase is kept.

* **Polishing** – projected sub‑gradient descent on the true max‑convolution
  objective.  Momentum and a step‑size decay 1/√t are used.  This phase
  receives the bulk of the time budget.

* **Deterministic greedy peak reduction** – a cheap loop moves a fraction of
  the coefficient that contributes most to the current peak towards the
  smallest coefficient, accepting only strict improvements.

* **Random local search** – finally a simple simulated‑annealing style
  neighbourhood search is performed.  A small amount of mass ε is moved
  from a source index (chosen proportionally to its weight) to a random
  destination index; the convolution is updated incrementally and the move
  is kept only if it strictly lowers the peak.  This often squeezes out
  a few extra percent.

All phases respect the overall 1000 s budget.  The result is normalised to
sum to one before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam (soft‑max) optimisation – exploration
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   global_deadline: float,
                   start_time: float,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max‑convolution objective.
    Returns the best probability vector found and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – go further than the original implementation
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]
    iters_per_alpha = 3000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.14
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= global_deadline:
                return best_a, best_score

            # soft‑max to obtain a valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # convolution and smoothed objective
            conv = _conv_self(a)
            scaled = α * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α
            obj = 2.0 * n * f_val

            # gradient w.r.t. a (soft‑max approximation)
            # ---------- soft‑max weights ----------
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            # ---------- cross‑correlation (gradient) ----------
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            grad_a *= 2.0 * n

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={α:.1f}, score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent – polishing
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # active set (indices where conv is close to its maximum)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # double‑count diagonal – harmless

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy peak reduction – deterministic polishing
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Moves a fraction of the coefficient responsible for the current peak
    towards the smallest coefficient, accepting only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.35 * a[src]          # a bit more aggressive than the original 0.25
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max - 1e-15:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Random local search – final fine‑tuning
# ----------------------------------------------------------------------


def _random_local_search(a0: np.ndarray,
                         conv0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing‑style neighbourhood search.
    Moves a random amount ε from a source index (chosen proportionally to a)
    to a random destination index.
    The convolution is updated incrementally; the move is kept only if the
    peak strictly decreases.
    """
    a = a0.copy()
    conv = conv0.copy()
    cur_max = conv.max()
    n = a.shape[0]
    conv_len = conv.shape[0]
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        if a.sum() == 0.0:
            break

        # source index – weighted by current mass (high probability for large entries)
        src = rng.choice(n, p=a / a.sum())
        # destination – uniform random, must differ from src
        dst = rng.integers(n)
        if dst == src:
            continue

        # choose ε ∈ (0, a[src]]
        eps = rng.random() * a[src]
        if eps < 1e-15:
            continue

        # incremental update of the convolution
        shift_src = np.zeros(conv_len, dtype=float)
        shift_src[src:src + n] = a
        shift_dst = np.zeros(conv_len, dtype=float)
        shift_dst[dst:dst + n] = a

        new_conv = conv + 2.0 * eps * (shift_dst - shift_src)
        new_max = new_conv.max()

        if new_max < cur_max - 1e-15:
            # accept move
            a[src] -= eps
            a[dst] += eps
            conv = new_conv
            cur_max = new_max
            if it % 500 == 0:
                print(f"[{time.time() - start_time:6.2f}s] random search improve: iter={it}, max={new_max:.6f}")

    return a, cur_max

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    center = (n - 1) / 2.0
    x = np.arange(n)
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.985) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()

    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))
