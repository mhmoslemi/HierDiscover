def adam_optimize(a0, n, deadline, start_time, init_alpha=0.5, final_alpha=200.0):
    a = a0.copy()
    best = a.copy()
    best_score = compute_objective(a)
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # schedule for alpha: linear in log scale across iterations?
    # We'll compute number of remaining iterations approx (deadline-start_time)/est_time_per_iter (e.g., 0.001 sec) -> maybe large.
    # Instead we increase alpha gradually via factor per iteration.
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = 2 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: iter={t}, score={cur_score:.6f}")
        # alpha schedule
        # We'll have alpha = init_alpha * (final_alpha / init_alpha) ** (min(1, (t * 0.001) / (deadline - start_time)))? Hard.
        # Use exponential increase per iteration:
        alpha = init_alpha * (final_alpha / init_alpha) ** (t / 1000000) # but t may not reach 1e6. Instead compute fraction of time.
        # Better: alpha = init_alpha * (final_alpha / init_alpha) ** ((time.time() - start_time) / (deadline - start_time))
        frac = (time.time() - start_time) / (deadline - start_time)
        alpha = init_alpha * (final_alpha / init_alpha) ** frac
        # Compute weights w
        max_c = conv.max()
        scaled = alpha * (conv - max_c)  # shift to avoid overflow
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # compute gradient
        # w_rev = w[::-1]
        # compute cross-correlation via FFT
        fft_len = _next_pow2(3*n)
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w[::-1], n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w) - 1]  # length n+len(w)-1
        grad = 2.0 * corr[len(w) - 1 - np.arange(n)]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.1 * m_hat / (np.sqrt(v_hat) + eps)  # step size scaling 0.1 maybe tuned
        a = a - step
        # project onto simplex
        a = _project_simplex(a)
    return best, best_score

"""
Propose a step‑function that (approximately) minimises

    2 * n * max_{k} (a * a)_k   (the sum of a is normalised to 1)

The algorithm combines three ideas:

1.  A cheap parametric sweep over simple “U‑shaped’’ families
    (exponential decay from the edges and power‑law decay from the edges)
    to obtain a good starting distribution for a given length *n*.

2.  An Adam‑type projected gradient descent on a smooth approximation of the
    max (log‑sum‑exp) that works directly on the simplex.  The smoothing
    parameter α is increased from a small to a large value during the run.

3.  A fast O(n) “targeted refinement’’ that moves tiny probability masses
    away from indices that contribute most to the current maximal convolution
    entry.  This stage is executed at the very end.

Multiple lengths (1200, 2400, 4800) are tried sequentially within the
overall 1000 s budget; the best sequence found overall is returned.
All numbers are non‑negative and the final vector is normalised to sum = 1.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (same as the reference implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0          # clip tiny negatives from numerical noise
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for cheap initialisation
# ----------------------------------------------------------------------
def _exp_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ lam^{min(i, n-1-i)}   (edges heavy, symmetric)."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)
    a = lam ** d
    return a


def _power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + (n-i)^{-p}   (edges heavy, power law)."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    return a


# ----------------------------------------------------------------------
# Fast objective (exact, no clamping) – used for internal selection
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2*n*max(conv) for a already normalised (sum=1)."""
    if a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return float("inf")
    a = a / s
    conv = _conv_self(a)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on a log‑sum‑exp smooth max
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iter: int = 20000,
                   init_alpha: float = 0.5,
                   final_alpha: float = 200.0,
                   lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed max.
    Returns (best_a, best_score) seen during the run.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # pre‑compute FFT length (enough for a * w_rev)
    fft_len = _next_pow2(3 * n)

    for it in range(1, max_iter + 1):
        now = time.time()
        if now >= deadline:
            break

        # convolution and current objective
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{now - start_time:.2f}s] Adam improve: iter={it:,}, score={cur_score:.6f}")

        # smoothing parameter α (log‑linear schedule)
        frac_time = (now - start_time) / (deadline - start_time)
        alpha = init_alpha * ((final_alpha / init_alpha) ** frac_time)

        # smooth‑max weights (log‑sum‑exp trick)
        max_c = conv.max()
        scaled = alpha * (conv - max_c)          # shift for numerical stability
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_rev = w[::-1]

        # gradient of the smooth max (mirror of Phase‑1)
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        idx = len(w_rev) - 1 - np.arange(n)
        grad = 2.0 * corr[idx]

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)      # stay on the probability simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast O(n) targeted refinement (taken from the reference code)
# ----------------------------------------------------------------------
def _run_targeted_refine_optimized(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.05,
                                   min_eps_frac: float = 0.001) -> np.ndarray:
    """
    Very fast O(n) refinement that moves a tiny amount of probability mass
    away from indices that contribute most to the current maximal convolution.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # -------- source index (probability proportional to contribution) --------
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # -------- destination index (prefer indices that do NOT contribute) --------
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # -------- propose a move --------------------------------------------------
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution
        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # recompute from scratch every so often to avoid drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a
