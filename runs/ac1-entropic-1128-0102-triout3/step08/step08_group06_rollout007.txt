"""
A three‑phase optimizer for the self‑convolution bound.

Phase 1 – Adam on a log‑sum‑exp smooth approximation of the max
          (mirrored‑descent on the simplex).  The smoothing parameter
          α is increased geometrically, allowing the optimiser to first
          make large moves on a very smooth objective and then to
          concentrate on the true max.

Phase 2 – Projected sub‑gradient descent on the exact max (the same
          algorithm as the reference implementation, but with a larger
          step‑size schedule and a longer run).

Phase 3 – Simulated‑annealing refinement that repeatedly moves a tiny
          amount of probability mass from indices that contribute most
          to the current maximal convolution entry to indices that
          contribute the least.  The annealing schedule permits occasional
          uphill moves, helping the search escape shallow local minima.

All phases keep the best sequence ever seen (according to the exact
objective 2·n·max(conv)).  The final sequence is projected onto the
probability simplex and returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0      # eliminate tiny negatives from round‑off
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : Adam on smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_adam_smooth(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     alphas,
                     steps_per_alpha: int = 4000,
                     eta0: float = 0.2):
    """
    Mirror‑descent on the probability simplex with Adam‑style adaption,
    minimising a smoothed approximation of the maximal convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0
    fft_len = _next_pow2(3 * n - 2)   # enough for a and w_rev

    for alpha in alphas:
        for _ in range(steps_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1

            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam  α={alpha:4g}  iter={t:,}  score={cur_score:.6f}")

            # ----- smooth‑max weights (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)   # stable scaling
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ----- gradient of the smoothed max -----
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wr = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wr, n=fft_len)[: n + w_rev.size - 1]
            grad = 2.0 * corr[w_rev.size - 1 - np.arange(n)]

            # ----- Adam update on the simplex -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            eta = eta0 / (math.sqrt(t) + 1.0)   # slowly decaying step size
            a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

            # ----- projection back to simplex -----
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.30,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    """
    Classical projected sub‑gradient descent on the true maximal
    convolution entry, with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter={it:,} score={cur_score:.6f}")

        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of the max (vectorised version)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each pair contributes a[j] + a[i] (see derivation)
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_annealing_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         init_eps_frac: float = 0.10,
                         min_eps_frac: float = 0.001,
                         init_temp: float = 1e-4,
                         final_temp: float = 1e-7):
    """
    Fast O(n) refinement that moves a tiny amount of probability mass.
    Annealing enables occasional uphill moves, increasing the chance to
    escape shallow local minima.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linear temperature decay
        elapsed = time.time() - start_time
        total = deadline - start_time
        temp = init_temp * (1.0 - elapsed / total) + final_temp * (elapsed / total)

        # ----- choose source index weighted by contribution to the current max -----
        k_max = int(conv.argmax())
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # ----- choose destination index (prefer indices that hardly affect the max) -----
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # ----- annealing step size (fraction of src's mass) -----
        elapsed_frac = elapsed / total
        eps_frac = init_eps_frac * (1.0 - elapsed_frac) + min_eps_frac * elapsed_frac
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # ----- compute the convolution change caused by moving eps -----
        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        # quadratic correction (exact)
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # ----- Metropolis acceptance -----
        accept = False
        if new_max < max_val - 1e-14:
            accept = True
        else:
            prob = math.exp(-(new_max - max_val) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: iter={it:,} max={best_max:.8f}")

        # periodic full recomputation to avoid numerical drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed max
    # ------------------------------------------------------------------
    phase1_time = min(40.0, 0.10 * (deadline - time.time()))
    phase1_deadline = time.time() + phase1_time

    alphas = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    a_smooth, score_smooth = _run_adam_smooth(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alphas=alphas,
        steps_per_alpha=4000,
        eta0=0.2,
    )
    if score_smooth < best_score:
        best_score = score_smooth
        best_seq = a_smooth.copy()
        print(f"[{time.time() - start_time:.2f}s] phase‑1 improve: score={score_smooth:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (heavy lifting)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # keep roughly 70 % of the remaining budget for this phase
        phase2_time = max(30.0, 0.70 * remaining)
        phase2_deadline = time.time() + phase2_time

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            step0=0.30,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_annealing_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            init_eps_frac=0.10,   # start with fairly aggressive moves
            min_eps_frac=0.001,
            init_temp=5e-4,
            final_temp=1e-7,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
