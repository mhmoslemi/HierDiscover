        delta[2*dst] += eps*eps
        delta[src+dst] -= 2*eps*eps
        new_conv = conv + delta
        new_max = new_conv.max()

            accept = True

            if rng.random() < prob:

        if accept:
            a[src] -= eps
            a[dst] += eps
            conv = new_conv
            max_val = new_max
            if new_max < best_max - 1e-15:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start:.2f}s] SA improve: max={best_max:.8f}")
        # occasionally recompute conv from scratch
        iter += 1
        if iter % 20000 == 0:

            max_val = conv.max()
    # Final strict refinement (like Phase3 but no uphill)
    a_refined = _run_targeted_refine_optimized(best_a, n, deadline, start)
    # Normalise final a

        a_refined /= total
    else:
        a_refined = np.full(n, 1.0/n)
    # Return list of floats
    return [float(x) for x in a_refined]

"""
Simulated‑annealing optimizer for the self‑convolution bound.

The algorithm works in three steps:

1.  Initialise a probability vector `a` (sum = 1).  If a previous construction
    (`height_sequence_1`) exists it is up‑sampled to the target length,
    otherwise a random simplex point is used.

2.  Simulated‑annealing (SA) loop.
    – At each iteration the index `k*` where the convolution `a*a` attains its
      maximum is identified.
    – A source index is sampled proportionally to its contribution to that
      maximal entry, and a destination index is chosen among those that
      contribute the least.
    – A tiny amount of mass `ε` is moved from source → destination.
      The convolution is updated **in O(n)** using the exact rank‑2 update
      (`Δ = 2ε·(a shifted by dst – a shifted by src) + ε²·δ*δ`).
    – The move is always accepted if it reduces the maximal convolution entry.
      Otherwise it is accepted with Metropolis probability
      `exp(-(new_max‑old_max)/T)`, where the temperature `T` slowly cools.
    – The best vector seen so far is stored.

3.  A final strict refinement (the same “targeted‑refine” routine used in the
    previous three‑phase optimizer) is run with the remaining time budget to
    guarantee that the final answer does not get worse.

The whole procedure respects the 1000 s time budget and returns a list of
non‑negative heights suitable for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negative noise.
    conv[conv < 0] = 0.0
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _run_targeted_refine_optimized(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.05,
                                   min_eps_frac: float = 0.001):
    """
    Fast O(n) refinement that only accepts strictly improving moves.
    This is essentially the Phase‑3 code from the previous optimizer
    (kept unchanged because it works very well for the final polish).
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linearly decay the move‑size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # ---- source index (weighted by its contribution to the current max)
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # ---- destination index (prefer indices not contributing to the max)
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # ---- O(n) exact update of the convolution
        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # ---- accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # periodically recompute from scratch to avoid drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing core
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         init_eps_frac: float = 0.20,
                         min_eps_frac: float = 0.001,
                         init_temp_frac: float = 0.10,
                         min_temp_frac: float = 0.001,
                         report_interval: int = 20000):
    """
    SA loop that moves mass between indices while tracking the best vector.
    Returns the best vector found and its maximal convolution entry.
    """
    rng = np.random.default_rng()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    it = 0

    # Temperature is expressed as a fraction of the *current* maximal value.
    init_temp = max(max_val * init_temp_frac, 1e-12)
    min_temp = max(max_val * min_temp_frac, 1e-12)

    while time.time() < deadline:
        it += 1

        # (1) locate current worst convolution entry
        k_max = int(conv.argmax())

        # (2) source index – proportional to contribution to that entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        # (3) destination index – favour indices that do *not* contribute
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # (4) schedule for step‑size and temperature
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)          # linear progress 0 → 1
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay
        temperature = init_temp * (1.0 - decay) + min_temp * decay

        # (5) propose a move
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # (6) exact O(n) update of the convolution (same as Phase‑3)
        conv_len = 2 * n - 1
        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps
        new_conv = conv + delta
        new_max = new_conv.max()

        # (7) Metropolis acceptance
        accept = False
        if new_max < max_val - 1e-15:                  # strict improvement
            accept = True
        else:
            # probability to accept a worsening move
            prob = math.exp(-(new_max - max_val) / (temperature + 1e-30))
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            conv = new_conv
            max_val = new_max
            if new_max < best_max - 1e-15:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to keep numerical drift bounded
        if it % report_interval == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a, best_max

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.

    The strategy:
        * Initialise from the previous best construction if available.
        * Run a simulated‑annealing loop that moves probability mass between
          indices while sometimes accepting uphill moves.
        * Finish with a strict “targeted‑refine” pass to guarantee no
          deterioration.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Initialise – use the global `height_sequence_1` if present.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # Choose a moderate length (the previous run used 4800; we keep the same)
    n = 4800
    if starter is not None:
        a0 = _upsample(starter, n)
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # 2️⃣ Simulated‑annealing phase (majority of the budget)
    # ------------------------------------------------------------------
    # Allocate ~80 % of the remaining time for SA.
    remaining = deadline - time.time()
    sa_deadline = time.time() + max(30.0, 0.80 * remaining)

    best_a_sa, best_max_sa = _simulated_annealing(
        a0,
        n,
        sa_deadline,
        start_time,
        init_eps_frac=0.20,
        min_eps_frac=0.001,
        init_temp_frac=0.10,
        min_temp_frac=0.001,
        report_interval=20000,
    )
    # Compute the corresponding score (2·n·max) for easy debugging.
    score_sa = 2.0 * n * best_max_sa
    print(f"[{time.time() - start_time:.2f}s] SA finished: score={score_sa:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final strict refinement (Phase‑3 style)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_targeted_refine_optimized(
            best_a_sa,
            n,
            deadline,
            start_time,
            init_eps_frac=0.05,
            min_eps_frac=0.001,
        )
        conv_ref = _conv_self(a_refined)
        best_max = conv_ref.max()
        best_score = 2.0 * n * best_max
        print(f"[{time.time() - start_time:.2f}s] final refine: score={best_score:.6f}")
    else:
        a_refined = best_a_sa
        best_score = score_sa

    # ------------------------------------------------------------------
    # 4️⃣ Normalise and return a plain Python list.
    # ------------------------------------------------------------------
    a_refined = np.maximum(a_refined, 0.0)
    total = a_refined.sum()
    if total > 0:
        a_refined /= total
    else:
        a_refined = np.full(n, 1.0 / n)

    # The harness expects a list of floats, not a NumPy array.
    return [float(x) for x in a_refined]
