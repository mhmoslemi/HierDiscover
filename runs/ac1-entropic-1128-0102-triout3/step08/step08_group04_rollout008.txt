"""
Propose a step‑function that minimises ``evaluate_sequence``.

The search consists of several short restarts.  Each restart runs
  1️⃣  a warm‑up with exponentiated‑gradient on a smoothed max,
  2️⃣  a longer projected sub‑gradient descent on the exact max,
  3️⃣  a massive local refinement that moves tiny amounts of mass.
The refinement uses a simple simulated‑annealing acceptance rule so
that occasional uphill moves are allowed, helping to escape plateaus.
Multiple restarts (different random seeds, different lengths, a
valley‑shaped start etc.) are tried within the 1000 s budget and the
best sequence found is returned.
All numbers are forced to be non‑negative and normalised to unit sum,
which makes the evaluation score equal to ``2·n·max(conv)``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient warm‑up on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient (log‑sum‑exp)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update (mirror descent on the simplex)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : massive incremental refinement with simulated annealing
# ----------------------------------------------------------------------
def _run_incremental_refine_sa(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               eps_start: float = 0.04,
                               eps_end: float = 0.002,
                               init_temp: float = 0.02,
                               final_temp: float = 0.0005,
                               rel_improve: float = 0.0):
    """
    Tiny mass moves with an annealed acceptance probability.
    eps_start / eps_end control the size of the move,
    init_temp / final_temp control the probability of accepting a
    slight increase of the maximal convolution entry.
    """
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0

    while time.time() < deadline:
        # linear schedules
        elapsed = time.time() - start_refine
        frac = elapsed / total_time
        eps_frac = max(eps_end, eps_start * (1.0 - frac))
        temp = max(final_temp, init_temp * (1.0 - frac))

        # current maximal index
        k_max = int(np.argmax(conv))

        # ---- source: an index that contributes a lot to the peak ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ---- destination: an index that contributes little to the peak ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                         # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ---- amount to move -------------------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ---- propose new convolution (O(n) update) -------------------------
        conv_new = conv.copy()
        # slice where the source/destination appear in the sum indices
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a

        new_max = conv_new.max()
        delta = new_max - max_val

        # ---- acceptance ----------------------------------------------------
        accept = False
        if delta < -rel_improve:
            accept = True
        else:
            # allow slight uphill moves with probability exp(-Δ/(temp·max_val))
            if delta <= 0:
                accept = True
            else:
                prob = math.exp(-delta / (temp * max_val + 1e-18))
                if rng.random() < prob:
                    accept = True

        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Multiple short restarts are performed; each restart consists of three
    optimisation phases.  The best sequence found before the deadline is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")
    restart = 0

    # ------------------------------------------------------------------
    # Loop over independent restarts while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:   # leave a tiny buffer
        remaining = deadline - time.time()

        # pick a length; alternating gives diversity without blowing up cost
        n = 4800 if restart % 2 == 0 else 6400

        # split the remaining budget for the three phases
        stage1_time = min(12.0, 0.07 * remaining)
        stage2_time = min(remaining - stage1_time - 6.0, 0.70 * remaining)
        stage3_time = remaining - stage1_time - stage2_time

        # --------------------- initialise a0 ----------------------------
        if restart == 0:
            # try to use the global construction if it exists
            try:
                starter = np.asarray(height_sequence_1, dtype=float)
                if starter.ndim == 1 and starter.size > 0:
                    a0 = _upsample(starter, n)
                else:
                    raise ValueError
            except Exception:
                a0 = np.random.rand(n)
        elif restart % 3 == 1:
            # valley shape (low centre, high edges)
            idx = np.arange(n)
            mid = (n - 1) / 2.0
            a0 = np.abs(idx - mid) + 0.5
        elif restart % 3 == 2:
            # a slight sqrt‑bell shape
            idx = np.arange(n)
            mid = (n - 1) / 2.0
            a0 = np.sqrt(np.abs(idx - mid) + 0.1)
        else:
            a0 = np.random.rand(n)

        a0 = a0.astype(float)
        a0 /= a0.sum()

        # --------------------- Phase 1 : warm‑up -----------------------
        stage1_deadline = time.time() + stage1_time
        a_warm, sc_warm = _run_exponential_gradient(
            a0,
            n,
            stage1_deadline,
            start_time,
            alpha_schedule=[0.5, 2, 5, 10, 20, 40, 80, 200, 400],
            eta0=0.5,
            steps_per_alpha=5000,
        )
        a_cur = a_warm
        cur_score = sc_warm

        # --------------------- Phase 2 : sub‑gradient -----------------
        if time.time() + stage3_time < deadline:
            stage2_deadline = time.time() + stage2_time
            a_sub, sc_sub = _run_subgradient_descent(
                a_cur,
                n,
                stage2_deadline,
                start_time,
                step0=0.32,
                beta=0.96,
            )
            if sc_sub < cur_score:
                a_cur = a_sub
                cur_score = sc_sub

        # --------------------- Phase 3 : SA refinement ----------------
        if time.time() < deadline - 1.0:
            a_ref = _run_incremental_refine_sa(
                a_cur,
                n,
                deadline,
                start_time,
                eps_start=0.06,
                eps_end=0.001,
                init_temp=0.02,
                final_temp=0.0005,
                rel_improve=0.0,
            )
            conv_ref = _conv_self(a_ref)
            sc_ref = 2.0 * n * conv_ref.max()
            if sc_ref < cur_score:
                a_cur = a_ref
                cur_score = sc_ref

        # --------------------- update global best --------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_cur.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart {restart} "
                  f"new global best score={best_score:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no sequence was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4800, 1.0 / 4800)

    # ------------------------------------------------------------------
    # Normalise and convert to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
