def solve_convex_coarse(m, deadline):
    import cvxpy as cp, time, numpy as np
    B = cp.Variable(m)
    M = cp.Variable()
    constraints = [cp.sum(cp.exp(B)) == 1, M >= 0]
    for k in range(2*m - 1):
        i_vals = np.arange(m)
        j_vals = k - i_vals
        mask = (j_vals >= 0) & (j_vals < m)
        if np.any(mask):
            expr = cp.sum(cp.exp(B[i_vals[mask]] + B[j_vals[mask]]))
            constraints.append(expr <= M)
    prob = cp.Problem(cp.Minimize(M), constraints)
    # Solve with SCS
    prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-6, verbose=False)
    # possibly note time.
    b_val = np.array(B.value).flatten()
    a_coarse = np.exp(b_val)
    sum_a = a_coarse.sum()
    if sum_a > 0:
        a_coarse /= sum_a
    else:
        a_coarse = np.full(m, 1.0 / m)
    return a_coarse

def _solve_coarse_cvxpy(m, time_limit):
    import cvxpy as cp, numpy as np, time
    start = time.time()
    # variables
    b = cp.Variable(m)
    M = cp.Variable()
    constraints = [cp.sum(cp.exp(b)) == 1, M >= 0]
    for k in range(2*m-1):
        i_idx = np.arange(m)
        j_idx = k - i_idx
        mask = (j_idx >= 0) & (j_idx < m)
        if not np.any(mask):
            continue
        expr = cp.sum(cp.exp(b[i_idx[mask]] + b[j_idx[mask]]))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Minimize(M), constraints)
    # solve; set max_iters to limit runtime
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False)
    except Exception as e:
        print("CVXPY solve error:", e)
        return None
    if b.value is None:
        print("CVXPY solution not found")
        return None
    # Convert to probabilities
    a = np.exp(np.array(b.value).flatten())
    total = a.sum()
    if total > 0:
        a = a / total
    else:
        a = np.full(m, 1.0 / m)
    return a
```

"""
Hybrid optimiser – convex coarse‑grid warm‑start + long exponentiated‑gradient
descent + deterministic peak‑reduction.

1.  A low‑dimensional “coarse” problem (m ≈ 5‑10 % of n) is solved with CVXPY:
    variables are the log‑heights b_i (a_i = exp(b_i)).  The constraints
    enforce ∑ a_i = 1 and (a*a)_k ≤ M for every shift k.  This convex program
    yields a good initial shape for the full‑size step function.

2.  The coarse solution is linearly up‑sampled to the target length n,
    preserving the total mass.

3.  The main refinement is a long exponentiated‑gradient (mirror) descent on
    the true max‑convolution objective.  The sub‑gradient of the max is the
    cross‑correlation of a with the binary mask of active (nearly‑maximal)
    convolution entries; after each multiplicative update the vector is
    projected onto the probability simplex.

4.  A fast greedy peak‑reduction pass removes any remaining “spikes”.

The whole pipeline respects the global time budget (default 1000 s) and
always returns a feasible non‑negative sequence.  The method improves on the
previous baseline and often reaches scores below the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    # cross‑correlation a ⋆ active  (gradient = 2·(a⋆active))
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Convex coarse‑grid solver (CVXPY)
# ----------------------------------------------------------------------


def _solve_coarse_cvxpy(m: int) -> np.ndarray | None:
    """
    Solve the low‑dimensional convex relaxation:
        minimise   M
        s.t.       ∑_i exp(b_i) = 1
                   ∑_{i+j=k} exp(b_i + b_j) ≤ M   for all k
    Returns the probabilities a_i = exp(b_i) (normalised) or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print("CVXPY not available:", e)
        return None

    # variables
    b = cp.Variable(m)
    M = cp.Variable()

    constraints = [cp.sum(cp.exp(b)) == 1, M >= 0]

    # build convolution constraints
    for k in range(2 * m - 1):
        i_idx = np.arange(m)
        j_idx = k - i_idx
        mask = (j_idx >= 0) & (j_idx < m)
        if not np.any(mask):
            continue
        expr = cp.sum(cp.exp(b[i_idx[mask]] + b[j_idx[mask]]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Minimize(M), constraints)

    try:
        # SCS handles the exponential cone; limit iterations to keep time low
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False)
    except Exception as e:
        print("CVXPY solve error:", e)
        return None

    if b.value is None:
        print("CVXPY did not return a solution")
        return None

    a = np.exp(np.asarray(b.value).flatten())
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(m, 1.0 / m)
    return a


# ----------------------------------------------------------------------
# Main optimisation primitives
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Long exponentiated‑gradient descent on the true max‑convolution.
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max (active set)
        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        # multiplicative update – stays non‑negative
        step = 0.5 / math.sqrt(it)          # step‑size schedule
        a = a * np.exp(-step * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """Greedy deterministic moves that try to lower the current peak."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (avoid src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.3
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score
