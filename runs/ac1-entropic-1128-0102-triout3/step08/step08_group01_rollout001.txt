"""
Propose a low‑max‑convolution step function.

The algorithm consists of three loosely coupled phases:

1. **Shape exploration** –  a small set of parametric families
   (symmetric exponential, Gaussian and polynomial “tent” shapes) are
   sampled, interpolated to the working size and evaluated with a fast
   FFT based score routine.  The best shape is kept as a starter.

2. **Iterative refinement** – the starter is improved by two
   complementary first‑order methods that respect the simplex
   constraints:
     • Projected sub‑gradient descent (additive updates, momentum,
       clipping + renormalisation).
     • Exponential‑gradient descent (multiplicative updates, i.e.
       mirror descent with the KL‑divergence Bregman potential).

3. **Targeted simulated annealing** – repeatedly moves a small amount
   of probability mass away from the indices that contribute most to
   the current maximal convolution coefficient.  An exponential cooling
   schedule is used and every accepted move updates the convolution
   via FFT.

If time permits, short random restarts are performed (perturbation +
refinement) to escape possible local minima.

All heavy work is guarded by a deadline (`budget_s` seconds) and the
code never exceeds the allowed CPU time.  The final sequence is
normalised to sum‑to‑one and returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures / no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _score(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(seq)^2 (no security checks)."""
    a = seq.astype(np.float64, copy=False)
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is guaranteed > 0 in our usage
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families (all symmetric, non‑negative, summed later)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _sym_exponential(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-alpha * |i - centre|)."""
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * np.abs(idx - centre))
    a /= a.sum()
    return a


def _sym_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i‑centre)^2 / (2σ²))."""
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def _sym_polynomial(n: int, p: float, q: float = 1.0) -> np.ndarray:
    """
    a_i ∝ max(0, 1 - (|i-centre|/centre)^p) ^ q.
    The shape is a “tent” with polynomial decay.
    """
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    d = np.abs(idx - centre) / centre
    base = np.maximum(0.0, 1.0 - d ** p)
    a = base ** q
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# First‑order optimisation helpers
# ----------------------------------------------------------------------


def _run_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    init_step: float = 0.5,
    momentum_coeff: float = 0.94,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    After each additive step we clip negatives and renormalise.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _score(a)

    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the active maximal entry
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # 2·a_{k-i}

        # momentum update
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * grad

        step = init_step / math.sqrt(it)
        a = a - step * momentum

        # keep feasible (non‑negative simplex)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                f"iter={it:,}, score={cur_score:.6f}"
            )
    return best_a, best_score


def _run_exponential_gradient(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    init_eta: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent with KL‑divergence (multiplicative updates).
    a ← a ⊙ exp(-η·∇f)  and re‑normalise.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        eta = init_eta / math.sqrt(it)
        # multiplicative update; avoid overflow/underflow
        a = a * np.exp(-eta * grad)

        # renormalise (still non‑negative)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = _uniform(n)
        else:
            a = a / s

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                f"iter={it:,}, score={cur_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted simulated annealing
# ----------------------------------------------------------------------


def _run_targeted_annealing(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    start_temp: float = 5e-2,
    end_temp: float = 1e-9,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Annealing that repeatedly moves a tiny amount of probability mass
    away from the pair of indices that contribute most to the current
    maximal convolution coefficient.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # locate current maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1, dtype=int)

        # contribution of each index to the peak (a_i * a_{k-i})
        contrib = a[idx_range] * a[k - idx_range]
        contrib_sum = contrib.sum()
        if contrib_sum > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib_sum)
        else:
            src = rng.integers(n)

        # random destination, preferably away from the peak
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move at most 30 % of source mass, scaled by temperature
        eps = a[src] * rng.random() * min(0.30, T * 10.0 + 0.05)

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            conv = conv_new
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] anneal improve: "
                f"iter={it:,}, score={cur_score:.6f}"
            )

    return best_a, best_score

    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = _uniform(n)

    # Return as a list of floats (the outer harness will call evaluate_sequence)
    return [float(x) for x in best_seq]
