"""
Optimiser for the step‑function objective 2·n·max(conv(a,a)) / (sum a)**2.

The algorithm combines several complementary heuristics:

1.  A short Adam phase on a smooth log‑sum‑exp approximation of the max‑convolution.
2.  A global search over the 2‑parameter beta family (a flexible smooth shape).
3.  Projected sub‑gradient descent with a correct gradient of the max‑convolution.
4.  An LP‑based “peak‑removal” step that solves a small linear programme
    (maximise total mass while keeping all currently maximal convolution entries
    below their current value).  The LP direction is mixed into the current
    distribution.
5.  Targeted simulated‑annealing that moves a small amount of probability mass
    away from the pair of entries that contributes most to the current peak.
6.  Light‑weight random‑restart perturbations.

All heavy work (convolution, projection) uses FFTs and an O(n) simplex projection,
so the optimiser can comfortably run for the full 1000 s budget on n≈8000.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _evaluate_obj(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase‑1 – Adam on a log‑sum‑exp smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 5000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max weights -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)      # stability trick
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of smoothed max (via FFT) -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)   # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 – Projected sub‑gradient descent (correct gradient)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    it = 0
    L = _next_pow2(2 * n - 1)

    # momentum parameters
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.35

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # index of the (first) maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # correct sub‑gradient of b_k = sum_i a_i a_{k-i}
        grad = np.zeros_like(a)
        grad[idx] = a[idx] + a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – LP‑based peak‑removal step
# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray,
             M: float,
             active_idx: np.ndarray,
             n: int) -> np.ndarray | None:
    """
    Solve:
        maximise   sum(g_j)
        subject to conv(a,g)[k] <= M  for k in active_idx,
                   sum(g) <= 1,
                   g >= 0.
    Returns the feasible vector g (not normalised).  If LP fails, returns None.
    """
    rows = len(active_idx) + 1
    A = np.zeros((rows, n), dtype=float)
    b = np.empty(rows, dtype=float)

    # constraints coming from the active convolution indices
    for i, k in enumerate(active_idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        A[i, idx] = a[k - idx]  # a_{k-j}
        b[i] = M

    # total‑mass constraint (≤ 1)
    A[-1, :] = 1.0
    b[-1] = 1.0

    # maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception:               # pragma: no cover – linprog should be available
        return None

    if res.success:
        g = res.x
        # if the LP returns essentially zero mass, treat as failure
        if g.sum() < 1e-12:
            return None
        return g
    return None


def _run_lp_improvement_loop(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP improve: iter={it:,}, score={cur_score:.6f}")

        # pick a handful of the currently highest convolution entries
        top_k = np.argsort(-conv)[:5]
        M = conv[top_k[0]]

        g_dir = _lp_step(a, M, top_k, n)
        if g_dir is None:
            # fallback: tiny random redistribution to keep moving
            src = np.random.randint(n)
            dst = np.random.randint(n)
            if src != dst and a[src] > 1e-12:
                eps = a[src] * 0.07
                a[src] -= eps
                a[dst] += eps
                a = _project_simplex(a)
            continue

        # mix the LP direction with current iterate
        t = 0.18
        a = (1.0 - t) * a + t * g_dir
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 – Targeted simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-8,
                            steps: int = 150_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate current maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move at most 20 % of the source mass
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# Helper – a quick random search over the Beta family

def _beta_family_candidate(alpha: float, beta: float, n: int) -> np.ndarray:
    """Return a discretised Beta(α,β) density on n points, normalised to sum=1."""
    # avoid exact 0 or 1 to keep finite values
    x = np.linspace(1e-8, 1 - 1e-8, n)
    pdf = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    if pdf.sum() == 0.0:
        pdf = np.ones_like(pdf)
    pdf /= pdf.sum()
    return pdf

            # log‑uniform sampling of α,β ∈ [0.1, 10]
            loga = np.random.uniform(-1, 1)
            logb = np.random.uniform(-1, 1)
            a = 10 ** loga
            b = 10 ** logb
            cand = _beta_family_candidate(a, b, n)
            cand_score = _evaluate_obj(cand, n)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] beta sweep improve: score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smoothed objective (≈10 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        phase1_deadline = time.time() + max(5.0, 0.10 * remaining)
        a_adam, score_adam = _run_adam_smoothing(
            best_seq,
            n,
            phase1_deadline,
            start_time,
            alphas=[0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0],
            lr0=0.6,
            iters_per_alpha=5000,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (major chunk)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.65 * remaining)
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP‑based peak‑removal (takes the remaining bulk of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase3_deadline = time.time() + max(15.0, 0.55 * remaining)
        a_lp, score_lp = _run_lp_improvement_loop(
            best_seq,
            n,
            phase3_deadline,
            start_time,
            best_score,
        )
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 (LP) improve: score={score_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted annealing polish (uses whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_targeted_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=180_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Light random‑restart perturbations if any time remains.
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        block = min(25.0, time_left / 2.0)

        # small random perturbation of the current best
        pert = best_seq + 0.012 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        sub_deadline = time.time() + block
        pert_seq, pert_score = _run_subgradient_descent(
            pert,
            n,
            sub_deadline,
            start_time,
            best_score,
            max_iters=120_000,
        )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"best final score (internal eval): {_evaluate_obj(best_seq, n):.6f}")
    return [float(x) for x in best_seq]
