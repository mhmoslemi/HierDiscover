def _run_incremental_refine_opt(a0, n, deadline, start_time, n_candidates=8, rel_improve=1e-13):
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute src
        k_max = int(np.argmax(conv))
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        if contrib.sum() > 0:
            src = rng.choice(i_range, p=contrib / contrib.sum())
        else:
            src = rng.choice(n, p=a / a.sum())
        if a[src] <= 1e-15:
            continue

        best_eps = 0.0
        best_dst = None

        for _ in range(n_candidates):
            dst = rng.integers(n)
            if dst == src:
                continue
            # compute base delta
            base_delta = np.zeros(2*n - 1)
            base_delta[src:src+n] -= 2.0 * a
            base_delta[dst:dst+n] += 2.0 * a

            pos = base_delta > 0
            if pos.any():
                eps_allowed = ((cur_max - conv[pos]) / base_delta[pos]).min()
                if eps_allowed <= 0:
                    continue
            else:
                eps_allowed = np.inf

            eps_allowed = min(eps_allowed, a[src])
            if eps_allowed > best_eps:
                best_eps = eps_allowed
                best_dst = dst

        if best_dst is None or best_eps <= 1e-15:
            continue

        # choose epsilon as fraction of best_eps, maybe random
        eps = best_eps * (0.5 + 0.5 * rng.random())
        # apply move
        a_old = a.copy()
        a[src] -= eps
        a[best_dst] += eps

        # update conv analytically using old a
        conv[src:src+n] -= 2.0 * eps * a_old
        conv[best_dst:best_dst+n] += 2.0 * eps * a_old

        cur_max = conv.max()
        if cur_max < best_max - rel_improve:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] inc‑refine-opt improve: max={best_max:.8f}")

    return best_a

"""
Improved step‑function search.

The algorithm follows three phases:

1️⃣  Warm‑up with an exponentiated‑gradient on a smoothed max (log‑sum‑exp).
    This quickly moves the mass from a random start towards a low‑peak
    autocorrelation.

2️⃣  Projected sub‑gradient descent on the exact max, using a momentum term
    and Euclidean projection onto the simplex.  This refines the shape.

3️⃣  Massive incremental local refinement with *analytic* step size
    computation.  For each move we choose a source index that contributes
    heavily to the current maximal convolution entry and several candidate
    destination indices.  For each candidate we construct the exact linear
    effect on the convolution (the “base_delta” vector) and compute the
    largest admissible ε that guarantees the new maximal value is strictly
    smaller.  The move is then performed with ε a random fraction of this
    bound.  This allows moves that are orders of magnitude larger than the
    tiny ε‑shifts used previously, dramatically accelerating convergence.

If time permits a second independent start (a sine‑shaped distribution)
is refined and the better of the two candidates is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray,
                      conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient warm‑up on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient (log‑sum‑exp trick)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : massive incremental refinement with analytic step size
# ----------------------------------------------------------------------
def _run_incremental_refine_opt(a0: np.ndarray,
                                n: int,
                                deadline: float,
                                start_time: float,
                                n_dest_candidates: int = 8,
                                rel_improve: float = 1e-13):
    """
    Perform millions of analytically‑computed mass moves.
    Each move selects a source index that heavily participates in the
    current maximal convolution entry and a destination index that yields
    the largest admissible ε (computed from the exact linear effect on the
    convolution).  The move is accepted only if the maximal entry strictly
    drops.
    """
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # ---------- pick source (high contribution to current max) ----------
        k_max = int(np.argmax(conv))
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            src = rng.choice(i_range, p=contrib / tot_contrib)
        else:
            src = rng.choice(n, p=a / a.sum())

        if a[src] < 1e-15:
            continue

        # ---------- choose destination (candidate set) ----------
        # deterministic candidate: the index with smallest current mass
        # (excluding src)
        idx_excluding_src = np.arange(n) != src
        dst_det = int(np.argmin(np.where(idx_excluding_src, a, np.inf)))
        candidates = [dst_det]

        # random candidates
        while len(candidates) < n_dest_candidates:
            dst = rng.integers(n)
            if dst != src and dst not in candidates:
                candidates.append(dst)

        # ---------- evaluate each candidate to obtain maximal feasible ε ----------
        best_eps = 0.0
        best_dst = None
        a_old = a  # a hasn't been modified yet

        for dst in candidates:
            # linear effect on the convolution (base_delta = ε·Δ)
            base_delta = np.zeros(2 * n - 1, dtype=float)
            base_delta[src:src + n] -= 2.0 * a_old
            base_delta[dst:dst + n] += 2.0 * a_old

            # indices where the move would increase the convolution
            pos = base_delta > 0
            if pos.any():
                # maximal ε that keeps every entry ≤ cur_max
                eps_allowed = ((cur_max - conv[pos]) / base_delta[pos]).min()
                if eps_allowed <= 0:
                    continue
            else:
                eps_allowed = np.inf

            eps_allowed = min(eps_allowed, a_old[src])
            if eps_allowed > best_eps:
                best_eps = eps_allowed
                best_dst = dst

        if best_dst is None or best_eps <= 1e-15:
            continue

        # take a random fraction of the admissible step (0.5 – 1.0)
        eps = best_eps * (0.5 + 0.5 * rng.random())

        # ---------- apply the move (use original a for the linear update) ----------
        a_old = a.copy()
        a[src] -= eps
        a[best_dst] += eps

        conv[src:src + n] -= 2.0 * eps * a_old
        conv[best_dst:best_dst + n] += 2.0 * eps * a_old

        cur_max = conv.max()
        it += 1
        if cur_max < best_max - rel_improve:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] inc‑refine‑opt improve: "
                  f"iter={it:,}, max={best_max:.8f}")

    return best_a
