"""
Improved step‑function search.

The algorithm follows the same multi‑stage scheme that proved successful
before, but with a different budget allocation and a more aggressive
final annealing stage.  The main ideas are:

* many cheap Adam runs on a relatively small problem (n≈1024) to explore
  the landscape;
* up‑sampling the best candidate to the full size (n=8192) and a longer
  Adam optimisation on the soft‑max surrogate of the max‑convolution;
* a long projected sub‑gradient descent on the true objective;
* an extended simulated‑annealing stage that mixes small mass‑transfer
  moves with occasional “swap” moves; this stage receives a sizeable
  part of the total budget;
* a short final Adam run at a very low temperature to polish the
  solution.

All vectors are kept on the probability simplex (∑a_i = 1, a_i ≥ 0) so
that the evaluation function reduces to `2·n·max(conv(a))`.
The code is written to respect a 1000 s wall‑time budget.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa = 1).
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val <= 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    # each convolution index k contributes a_i * a_{k-i}
    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max surrogate
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9993,
                      step_size: float = 0.20,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective
          O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    with Adam and projection onto the simplex.
    Returns the best vector (according to the true score) encountered.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline:
        it += 1

        # true objective for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weighting
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stabilise
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ (see derivation in the reference implementation)
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)     # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]   # contribution for each a_i
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Classical Nesterov‑style projected sub‑gradient descent on
        F(a) = 2·n·max(conv(a))   (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass‑transfer + swaps)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that performs two kinds of moves:
      – a fractional mass transfer between two distinct positions,
      – an occasional swap of the whole masses of two positions.
    Acceptance uses the true objective.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # decide move type
        if rng.random() < 0.9:          # mass‑transfer (most frequent)
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            eps = a[src] * rng.random() * 0.40   # up to 40 % of source mass
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
        else:                           # swap move
            i, j = rng.choice(n, size=2, replace=False)
            a_prop = a.copy()
            a_prop[i], a_prop[j] = a[j], a[i]

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to length `new_n`, then renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisation helpers (Chebyshev / Kaiser / sine)
# ----------------------------------------------------------------------


def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    Distribution:
        0.35 – Chebyshev window,
        0.35 – Kaiser window,
        0.20 – sinusoidal window (half‑cycle),
        0.10 – plain Dirichlet draw.
    """
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.35:
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
    elif have_signal and r < 0.70:
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
    elif r < 0.90:
        # one half of a sine window, shifted to be non‑negative
        x = np.linspace(0, np.pi, n)
        w = np.sin(x) ** 2
    else:
        w = rng.random(n)

    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises `evaluate_sequence`.  The routine respects the time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Use a previous solution as a warm start, if available
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    remaining = deadline - time.time()
    stage0_time = max(25.0, 0.12 * remaining)   # ~12 % of the budget
    slice_deadline = time.time() + stage0_time

    max_restarts = 20
    for r in range(max_restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = _window_initialiser(n0, rng)

        # initialise τ from a rough scale
        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 6.0, 1e-8)

        a_opt, score = _run_adam_softmax(
            a0, n0, slice_deadline, start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9995,
            step_size=0.22,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # fall‑back in the unlikely case that nothing was produced
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – upsample to the target size (n≈8192) and continue Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_target = 8192
    a_target = _upsample(best_seq, n_target)

    # a short re‑evaluation of τ for the larger vector
    conv1 = _conv_self(a_target)
    tau0_1 = max(conv1.max() * 6.0, 1e-10)

    remaining = deadline - time.time()
    stage1_time = max(30.0, 0.20 * remaining)   # ~20 % of the remaining budget
    slice_deadline = time.time() + stage1_time

    a_opt, score = _run_adam_softmax(
        a_target, n_target, slice_deadline, start_time,
        tau0=tau0_1,
        tau_end=1e-14,
        tau_decay=0.9994,
        step_size=0.15,
    )
    if score < best_score:
        best_score = score
        best_seq = a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long projected sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n2 = best_seq.shape[0]                # stays at 8192
        # leave a few seconds for the final polishing stages
        grad_deadline = deadline - 30.0
        a_opt, score = _run_subgradient_descent(
            best_seq, n2, grad_deadline, start_time,
            beta=0.97,
            step0=0.13,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – extended simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline - 5.0          # keep a few seconds for final polish
        a_ref, score = _run_annealing(
            best_seq,
            best_seq.shape[0],
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-7,
            steps=2_000_000,
        )
        if score < best_score:
            best_score = score
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve (final): "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final Adam polishing at a very low temperature
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # we give the remaining (few seconds) to a quick Adam run with a tiny τ
        final_deadline = deadline
        conv_final = _conv_self(best_seq)
        tau0_f = max(conv_final.max() * 4.0, 1e-9)
        a_opt, score = _run_adam_softmax(
            best_seq, best_seq.shape[0],
            final_deadline, start_time,
            tau0=tau0_f,
            tau_end=1e-15,
            tau_decay=0.9991,
            step_size=0.10,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] final Adam polish improve: "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time (guard against numerical drift)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list – the harness will evaluate it
    return [float(x) for x in best_seq]
