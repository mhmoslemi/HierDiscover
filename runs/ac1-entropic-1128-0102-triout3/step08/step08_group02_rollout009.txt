"""
Hybrid optimiser for the self‑convolution bound – improved version.

Key ideas
---------
* Use several diverse initialisations (uniform, exponential, U‑shape,
  random‑smooth and a *sparse* start with few equally‑weighted spikes).
* A longer “soft‑max Adam” phase with many more smoothing levels
  (α up to 2560) steers the solution sharply toward a region of low
  max‑convolution.
* Exponentiated‑gradient (multiplicative) descent on the *exact* max
  (active‑set sub‑gradient) with a larger initial step‑size
  (η₀≈0.5) and a 1/√t schedule.  This phase gets the lion’s share of the
  time budget because it provides the biggest improvements.
* A more aggressive deterministic peak‑reduction pass (larger
  no‑improvement limit) and a longer simulated‑annealing refinement.
* All phases respect a global 1000 s time budget and are timed out cleanly.

The returned list is a non‑negative step‑function (probability simplex)
that minimises ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·conv_k) w.r.t. a.
    Returns shape (n,).
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                       # soft‑max weights
    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    Active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Optimisation primitives

def _adam_phase(a0: np.ndarray,
                deadline: float,
                start_time: float,
                fft_len: int) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp approximation of the max.
    Returns the best simplex vector seen during the phase.
    """
    y = np.log(np.maximum(a0, 1e-16))        # logits
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # smoothing schedule – very smooth → very sharp (α up to 2560)
    alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                   40.0, 80.0, 160.0, 320.0, 640.0, 1280.0, 2560.0]
    # more iterations for the low‑α part, fewer for high‑α (still cheap)
    iters_per_alpha = [8000, 8000, 6000, 5000, 5000, 4000,
                       4000, 3000, 3000, 2500, 2500, 2000, 2000]

    best_seq = a0.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    for alpha, nit in zip(alpha_sched, iters_per_alpha):
        for i in range(nit):
            if time.time() >= deadline:
                return best_seq
            a = _softmax_from_logits(y)

            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            # log‑sum‑exp (numerically stable)
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * a.shape[0] * f_val

            # gradient w.r.t. a
            grad_a = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * a.shape[0])

            # chain rule to logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                          f"iter={i:,}, score={sc:.6f}")

    return best_seq


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Long exponentiated‑gradient descent on the exact max‑convolution.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    cur_conv = _conv_self(a)
    best_score = 2.0 * n * cur_conv.max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # exact sub‑gradient of the (non‑smooth) max
        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        # exponentiated (multiplicative) update
        step = eta0 / math.sqrt(it)
        a = a * np.exp(-step * grad)

        # project back onto the simplex (simple renormalisation)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 5000) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy transfers that reduce the current maximal peak.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # index contributing most to the peak

        # pick the smallest coefficient different from src as the destination
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.30
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-8,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement with mass‑transfer moves.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, "
                  f"score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  The method runs several starts,
    each consisting of:

    1. a longer Adam smoothing phase with α up to 2560,
    2. an aggressive exponentiated‑gradient descent on the true max‑convolution,
    3. deterministic peak‑reduction,
    4. a final annealing pass.

    The best sequence over all starts is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Determine problem size (n)
    # ------------------------------------------------------------------
    try:
        starter_seq = np.asarray(height_sequence_1, dtype=float)
        if starter_seq.ndim == 1 and starter_seq.size > 0:
            n = starter_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4800
        starter_seq = None

    # ------------------------------------------------------------------
    # Build a diverse collection of starting points
    # ------------------------------------------------------------------
    starts = []

    if starter_seq is not None:
        starts.append(starter_seq / starter_seq.sum())

    starts.append(np.full(n, 1.0 / n))                     # uniform

    # exponential decay – left‑hand heavy
    r = 0.985
    exp_vec = np.array([r ** i for i in range(n)], dtype=float)
    starts.append(exp_vec / exp_vec.sum())

    # symmetric U‑shape (large at edges)
    idx = np.arange(n)
    u_shape = 1.0 / (np.minimum(idx + 1, n - idx) + 1.0)
    starts.append(u_shape / u_shape.sum())

    # random smooth vector (Gaussian‑filtered noise)
    rng = np.random.default_rng()
    noise = rng.normal(size=n)
    smooth = np.convolve(noise, np.ones(15) / 15, mode='same')
    smooth = np.clip(smooth, 0, None)
    if smooth.sum() == 0:
        smooth = np.full(n, 1.0 / n)
    else:
        smooth /= smooth.sum()
    starts.append(smooth)

    # sparse random start – few equally‑weighted spikes (approx. √n spikes)
    m = max(2, int(math.sqrt(n)))
    sparse_idx = rng.choice(n, size=m, replace=False)
    sparse = np.zeros(n, dtype=float)
    sparse[sparse_idx] = 1.0 / m
    starts.append(sparse)

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optimise each start, sharing the global time budget
    # ------------------------------------------------------------------
    total_starts = len(starts)
    for i_start, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        remaining_starts = total_starts - i_start
        # allocate a fraction of the remaining budget to this start
        remaining_time = deadline - time.time()
        per_start_budget = remaining_time / remaining_starts

        print(f"[{time.time() - start_time:6.2f}s] start #{i_start + 1}/{total_starts}")

        # Normalise start onto the simplex
        a = np.maximum(a0, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # ------------------------------------------------------------------
        # Phase 1 – Adam (soft‑max) smoothing
        # ------------------------------------------------------------------
        phase_deadline = time.time() + min(0.1 * per_start_budget, 60.0)  # at most 60 s
        fft_len = _next_pow2(3 * n - 2)
        a_adam = _adam_phase(a, phase_deadline, start_time, fft_len)

        # ------------------------------------------------------------------
        # Phase 2 – Exponentiated‑gradient descent on the true max
        # ------------------------------------------------------------------
        if time.time() < deadline:
            grad_deadline = time.time() + min(0.65 * (deadline - time.time()), per_start_budget * 0.65)
            a_grad, score_grad = _subgradient_descent(a_adam, grad_deadline, start_time, eta0=0.5)
        else:
            a_grad, score_grad = a_adam, evaluate_sequence(a_adam.tolist())

        # ------------------------------------------------------------------
        # Phase 3 – Deterministic peak reduction
        # ------------------------------------------------------------------
        if time.time() < deadline:
            peak_deadline = time.time() + min(0.15 * (deadline - time.time()), per_start_budget * 0.15)
            a_peak, score_peak = _run_peak_reduction(a_grad, peak_deadline, start_time,
                                                     max_no_improve=5000)
        else:
            a_peak, score_peak = a_grad, score_grad

        # ------------------------------------------------------------------
        # Phase 4 – Simulated annealing refinement
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_final, score_final = _run_annealing(
                a_peak,
                deadline,
                start_time,
                start_temp=5e-2,
                end_temp=1e-8,
                steps=200_000,
            )
        else:
            a_final, score_final = a_peak, score_peak

        # ------------------------------------------------------------------
        # Choose the best stage for this start
        # ------------------------------------------------------------------
        stage_scores = [
            (evaluate_sequence(a_adam.tolist()), a_adam),
            (score_grad, a_grad),
            (score_peak, a_peak),
            (score_final, a_final),
        ]
        local_best_score, local_best_seq = min(stage_scores, key=lambda x: x[0])

        # ------------------------------------------------------------------
        # Update the global best if improved
        # ------------------------------------------------------------------
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST *** "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
