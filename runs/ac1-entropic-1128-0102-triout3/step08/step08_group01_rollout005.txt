Propose a low‑score step‑function.

Algorithm
---------
1. **Exploration** – generate many candidate shapes (random Dirichlet draws
   and a few parametric families such as U‑shaped, Gaussian, sine‑powers).
   Keep the best according to a fast FFT‑based score.

2. **Adam smoothing** – on the current best vector run Adam on a

   maximum.

   sub‑gradient method on the true objective (2·n·max‑conv).  A
   diminishing step‑size and momentum are used.

4. **Targeted simulated annealing** – repeatedly move a small amount of

   the current peak.  An exponential cooling schedule allows occasional
   uphill moves.

5. **Late random restarts** – while time permits, perturb the best
   solution and repeat steps 2‑4 on the perturbed start.  This helps to
   escape any remaining local minima.

    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length L (>=2·len(a)-1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _fast_score(a: np.ndarray, L: int) -> float:
    """Score = 2·n·max(conv(a,a))  (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    conv = _conv_self_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth approximation of the max
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   alphas,
                   lr0: float = 0.6,
                   iters_per_alpha: int = 12000) -> tuple[np.ndarray, float]:
    """Adam optimiser on log‑sum‑exp smoothing of the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self_fft(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self_fft(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smoothed max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)      # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)   # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          best_score_initial: float,
                          max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self_fft(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient taken at a (could be multiple maximisers – we just pick one)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted simulated annealing
# ----------------------------------------------------------------------
def _targeted_anneal(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     cur_best_score: float,
                     cur_best_seq: np.ndarray,
                     start_temp: float = 5e-2,
                     end_temp: float = 1e-9,
                     steps: int = 250_000) -> tuple[np.ndarray, float]:
    """Move mass away from the pair that creates the current peak."""
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate current maximal convolution index and its contributors
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        total = contrib.sum()
        if total > 0.0:
            src = rng.choice(idx_range, p=contrib / total)
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a modest amount (up to 20 % of source mass)
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self_fft(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Small parametric families (used in the exploration phase)
# ----------------------------------------------------------------------
def _make_u_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped (mass at the edges).  p < 0 gives a peak at the ends."""
    i = np.arange(n, dtype=float)
    a = ((i + 0.5) * (n - 0.5 - i)) ** p
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _make_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Centered Gaussian; sigma_frac is σ / n."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _make_sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^p(π (i+½)/n).  p>0 concentrates in the middle,
       0<p<1 yields flatter shape."""
    i = np.arange(n, dtype=float)
    a = np.sin(math.pi * (i + 0.5) / n) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    provided ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25  # small safety margin

    # ------------------------------------------------------------------
    # Problem size – we use a fairly large n to give the optimiser room.
    # ------------------------------------------------------------------
    n = 8192
    L = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Phase 0 – quick random / parametric exploration
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n)
    best_score = _fast_score(best_seq, L)

    # number of exploratory candidates (budget‑dependent)
    explore_until = start_time + 0.12 * (deadline - start_time)

    while time.time() < explore_until:
        # Choose a family at random
        r = np.random.rand()
        if r < 0.4:          # Dirichlet / Gamma draw
            conc = 10 ** np.random.uniform(-2, 1)   # between 0.01 and 10
            g = np.random.gamma(shape=conc, scale=1.0, size=n)
            a = g / g.sum()
        elif r < 0.6:        # U‑shape
            p = np.random.uniform(-2.0, -0.2)
            a = _make_u_shape(n, p)
        elif r < 0.8:        # Gaussian
            sigma_frac = np.random.uniform(0.08, 0.30)
            a = _make_gaussian(n, sigma_frac)
        else:                # Sine‑power
            p = np.random.uniform(0.3, 4.0)
            a = _make_sin_power(n, p)

        sc = _fast_score(a, L)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] explore improve: "
                  f"score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (≈15 % of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(5.0, 0.15 * remaining)

    a_adam, score_adam = _adam_optimize(
        best_seq,
        n,
        phase1_deadline,
        start_time,
        alphas=[0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0],
        lr0=0.6,
        iters_per_alpha=12000,
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] phase‑1 improve: "
              f"score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (major chunk)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.60 * remaining)

        a_sub, score_sub = _subgradient_optimize(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑2 improve: "
                  f"score={score_sub:.6f}")

        # a short second sub‑gradient run with a larger initial step
        if time.time() < deadline:
            extra_deadline = time.time() + min(12.0, deadline - time.time())
            a_sub2, score_sub2 = _subgradient_optimize(
                best_seq,
                n,
                extra_deadline,
                start_time,
                best_score,
                max_iters=200_000,
            )
            if score_sub2 < best_score:
                best_score = score_sub2
                best_seq = a_sub2.copy()
                print(f"[{time.time() - start_time:.2f}s] extra sub‑grad improve: "
                      f"score={score_sub2:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Long targeted annealing (use up the rest)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _targeted_anneal(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-9,
            steps=300_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Opportunistic random restarts (if any time left)
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:
        # allocate a small block (≈20 s) for each restart
        block = min(20.0, (deadline - time.time()) / 2.0)
        restart_deadline = time.time() + block

        # perturb current best a little
        pert = best_seq + 0.02 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # short Adam + sub‑gradient + anneal sequence
        a_tmp, sc_tmp = _adam_optimize(
            pert,
            n,
            restart_deadline,
            start_time,
            alphas=[0.2, 1.0, 5.0],
            lr0=0.4,
            iters_per_alpha=4000,
        )
        if time.time() < deadline:
            a_tmp, sc_tmp = _subgradient_optimize(
                a_tmp,
                n,
                deadline,
                start_time,
                best_score,
                max_iters=80_000,
            )
        if time.time() < deadline:
            a_tmp, sc_tmp = _targeted_anneal(
                a_tmp,
                n,
                deadline,
                start_time,
                best_score,
                a_tmp,
                start_temp=2e-2,
                end_temp=1e-9,
                steps=120_000,
            )
        if sc_tmp < best_score:
            best_score = sc_tmp
            best_seq = a_tmp.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: "
                  f"score={sc_tmp:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return as a plain list of floats
    return [float(x) for x in best_seq]
