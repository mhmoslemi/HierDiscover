"""
Multi‑restart, multi‑scale optimiser for the max‑self‑convolution bound.

The routine works inside the provided time budget (~1000 s) and tries
to beat the previous best (≈ 1.5124).  The main ideas are

* start from the known construction (``height_sequence_1``) if it exists,
* explore a few simple parametric families (power‑law, exponential,
  Gaussian) on several lengths (1024 → 8192) and keep the best,
* run a short Adam optimisation on a smooth log‑sum‑exp surrogate of
  the max‑convolution, followed by a brief Euclidean sub‑gradient
  descent, an exponentiated‑gradient (mirror) step and a light
  simulated‑annealing refinement,
* all vectors are kept on the probability simplex (non‑negative,
  Σaᵢ = 1) so that the evaluation function reduces to
  ``2·n·max(conv(a))``.

The algorithm is deliberately lightweight in each phase to leave enough
time for the larger problem sizes, which are the ones that can push the
bound below 1.51.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (true objective)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.25,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Minimise   Oτ(a)=2·n·τ·log∑exp(conv/τ)   with Adam and simplex projection."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true score for bookkeeping (no projection here)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f} τ={tau:.2e}")

        # soft‑max weights over the 2·n‑1 convolution entries
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # numerical stabilisation
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # shape (2·n‑1,)

        # gradient of Oτ: 4·n * (a_rev * p) trimmed to length n
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)    # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]  # the needed slice
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cooling schedule for τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent (true objective)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Classical sub‑gradient descent on F(a)=2·n·max(conv(a))."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the true objective
# ----------------------------------------------------------------------
def _run_exponentiated_gradient(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               step0: float = 0.20,
                               max_iter: int = 5 * 10 ** 6) -> tuple[np.ndarray, float]:
    """Mirror‑descent with entropy Bregman divergence."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        eta = step0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing (mass‑transfer)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source proportional to its current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30   # move at most 30 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve (n={n}) "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) and normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, keeping Σ=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine follows a multi‑restart, multi‑scale strategy and returns
    a plain Python list of floats (Σ≈1).  All work respects the supplied
    ``budget_s`` time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Use the previously discovered construction if it exists.
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                tmp = np.maximum(tmp, 0.0)
                if tmp.sum() > 0.0:
                    starter = tmp / tmp.sum()
        except Exception:
            starter = None

    best_seq = None
    best_score = float("inf")

    # optional: evaluate starter immediately
    if starter is not None:
        cur = 2.0 * starter.shape[0] * _conv_self(starter).max()
        best_seq = starter.copy()
        best_score = cur
        print(f"[{time.time() - start_time:6.2f}s] starter score={cur:.6f}")

    # ------------------------------------------------------------------
    # Candidate lengths – we increase the size only if time permits.
    # ------------------------------------------------------------------
    size_schedule = [1024, 2048, 4096, 8192]

    for n in size_schedule:
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        if remaining < 5.0:          # not enough time for a meaningful attempt
            break

        # ------------------------------------------------------------------
        # Initialise a vector for this size.
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] <= n:
            a = _upsample(best_seq, n)
        elif starter is not None:
            a = _upsample(starter, n)
        else:
            a = rng.random(n)
            a /= a.sum()

        # ------------------------------------------------------------------
        # 1.  Very cheap parametric sweep (power‑law, exponential, Gaussian)
        # ------------------------------------------------------------------
        # power‑law family a_i ∝ (i+1)^α (n‑i)^β
        alphas = np.linspace(-1.2, 1.2, 9)
        betas = np.linspace(-1.2, 1.2, 9)
        for alpha in alphas:
            for beta in betas:
                if time.time() >= deadline:
                    break
                idx = np.arange(n)
                seq = ((idx + 1) ** alpha) * ((n - idx) ** beta)
                seq = np.maximum(seq, 0.0)
                s = seq.sum()
                if s == 0.0:
                    continue
                seq /= s
                conv = _conv_self(seq)
                score = 2.0 * n * conv.max()
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:6.2f}s] n={n} pow a={alpha:.2f} b={beta:.2f} score={score:.6f}")

        # exponential decay a_i ∝ exp(-λ·i), made symmetric
        lambdas = np.linspace(0.0, 0.001, 7)
        for lam in lambdas:
            if time.time() >= deadline:
                break
            idx = np.arange(n)
            seq = np.exp(-lam * idx)
            seq = (seq + seq[::-1]) / 2.0
            seq /= seq.sum()
            conv = _conv_self(seq)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:6.2f}s] n={n} exp λ={lam:.6f} score={score:.6f}")

        # Gaussian‑like family centred in the middle
        sigma_fracs = np.linspace(0.05, 0.25, 5)
        for sf in sigma_fracs:
            if time.time() >= deadline:
                break
            idx = np.arange(n)
            centre = (n - 1) / 2.0
            sigma = sf * n
            seq = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
            seq /= seq.sum()
            conv = _conv_self(seq)
            score = 2.0 * n * conv.max()
            if score < best_score:

                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] n={n} subgrad refine score={score:.6f}")

            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] n={n} SA final score={score:.6f}")

        print(f"[{time.time() - start_time:6.2f}s] after size {n} best score ≈ {best_score:.6f}")
