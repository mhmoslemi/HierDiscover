    i = np.arange(n)

    if a_sum == 0:

    return a / a_sum

def _beta_seq(n, alpha):

"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

The search consists of three stages:

1. **Fast parametric sweep** – we generate many candidate shapes
   (Gaussian, symmetric Beta, sine‑power and random block‑constant
   profiles) for the fixed length n = 4096 and keep the best one.
2. **Projected sub‑gradient refinement** – starting from the best
   candidate we run a long sub‑gradient descent on the exact objective
   (2·n·max(conv(a,a))) while staying on the probability simplex.
3. **Light simulated‑annealing polish** – a cheap mass‑transfer anneal
   swaps small amounts of probability between two coordinates and
   accepts improving moves (or occasionally worsening ones) until the
   time budget is exhausted.

All phases respect the global 1000 s budget.  The routine returns a
plain ``list[float]`` that can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical fallback
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Parametric families of candidate sequences
# ----------------------------------------------------------------------
def _gaussian_seq(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _beta_seq(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the discrete grid.
    For α=1 we obtain the uniform distribution.
    """
    x = (np.arange(n) + 0.5) / n                     # points in (0,1)
    # avoid overflow/underflow for extreme α
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _sine_seq(n: int, power: float) -> np.ndarray:
    """sin‑powered profile: a_i ∝ sin(π·(i+0.5)/n)^power."""
    i = np.arange(n, dtype=float) + 0.5
    a = np.sin(np.pi * i / n) ** power
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _block_seq(rng: np.random.Generator, n: int, blocks: int) -> np.ndarray:
    """Random piece‑wise constant profile with ``blocks`` contiguous segments."""
    # random break‑points
    cuts = np.sort(rng.integers(0, n + 1, size=blocks - 1))
    edges = np.concatenate(([0], cuts, [n]))
    heights = rng.random(blocks) + 1e-6          # ensure strictly positive
    a = np.empty(n, dtype=float)
    for b in range(blocks):
        a[edges[b] : edges[b + 1]] = heights[b]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Fast objective (identical to evaluate_sequence when sum=1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a sequence already normalised to sum = 1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact objective)
# ----------------------------------------------------------------------
def _subgradient_refine(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Long sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses momentum and a 1/√t step‑size.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    cur_best = best_score

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < cur_best:
            cur_best = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} best={cur_score:.6f}")

        # sub‑gradient at the (unique) maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, cur_best


# ----------------------------------------------------------------------
# Tiny mass‑transfer annealing (polish)
# ----------------------------------------------------------------------
def _anneal_refine(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Very light simulated annealing that moves a small fraction of mass
    between two coordinates.  Temperature decays exponentially.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    # exponential schedule from 5·10⁻² to 10⁻⁶
    start_T = 5e-2
    end_T = 1e-6
    it = 0
    while time.time() < deadline:
        it += 1
        T = start_T * (end_T / start_T) ** (it / 1e6)   # denominator is a large constant; the exact shape is not crucial

        src = rng.choice(n, p=a)                     # source weighted by mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30           # move up to 30 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - best_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_a, best_score

                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective(starter)
                if sc < best_score:
                    best_score = sc
                    best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Fast parametric sweep (Gaussian, Beta, sine‑power, random blocks)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    # a small reserve for the final refinement
    reserve = 30.0

    # pre‑computations – we will keep generating candidates until time is low
    while time.time() < deadline - reserve:
        # pick a family at random
        family = rng.choice(['gaussian', 'beta', 'sine', 'block'])
        if family == 'gaussian':
            sigma = rng.uniform(n / 30.0, n / 1.5)          # roughly 0.03 n … 0.66 n
            seq = _gaussian_seq(n, sigma)
        elif family == 'beta':
            # sample α on a log‑scale between 0.1 and 10
            log_alpha = rng.uniform(-1.0, 1.0)
            alpha = 10 ** log_alpha
            seq = _beta_seq(n, alpha)
        elif family == 'sine':
            power = rng.uniform(0.5, 4.0)                    # sin‑power exponent
            seq = _sine_seq(n, power)
        else:   # block
            blocks = rng.integers(2, 9)                      # between 2 and 8 blocks
            seq = _block_seq(rng, n, blocks)

        # normalise (should already sum to 1, but guard against numerics)
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            continue
        seq /= total

        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New candidate ({family}) score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Local refinement via projected sub‑gradient descent (as much time as left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.65
        a_refined, sc_refined = _subgradient_refine(best_seq,
                                                    sub_deadline,
                                                    start_time,
                                                    best_score)
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad refinement improved to {sc_refined:.6f}")

    # ------------------------------------------------------------------
    # Final annealing polish (uses whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _anneal_refine(best_seq,
                                           deadline,
                                           start_time,
                                           best_score)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved to {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and cast to a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
