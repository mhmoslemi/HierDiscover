"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------

* **Stage 0 – cheap parametric starts** – try a few simple families
  (exponential, Gaussian, uniform, Dirichlet) and keep the best one.

* **Stage 1 – conditional‑gradient (Frank–Wolfe) on the max‑convolution**
  using a linear program.
  For the current candidate `a` we compute `M = max(a*a)`.  The LP
  maximises `∑ b_i` subject to the *linear* constraints
  `(a * b)[k] ≤ M` for the (few) *active* indices where the bound is tight.
  The optimal direction `b` is mixed with `a` (step ≈ 0.2).  Repeating this
  strictly decreases `M` and therefore the evaluation score.

* **Stage 2 – short projected sub‑gradient polishing** – a few thousand
  iterations of the true non‑smooth objective fine‑tune the final vector.

All phases respect the global 1000 s budget.  The returned vector is
normalised to sum 1 (the evaluation function is scale‑invariant).

The implementation uses only NumPy, SciPy’s `linprog` (high‑s) and the
provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Returns the best vector seen and its 2·n·max value.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # active set (indices attaining the current maximum, up to tolerance)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient for the max‑convolution (sum of gradients of active entries)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        # momentum (Nesterov‑style) and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        # Euclidean projection onto the probability simplex
        # (the classic O(n) algorithm)
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(a - theta, 0.0)
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming direction (conditional‑gradient step)
# ----------------------------------------------------------------------


def _lp_maximising_direction(a: np.ndarray,
                             M: float,
                             active_idxs: np.ndarray) -> np.ndarray:
    """
    Solve   max ∑ b_i   s.t.   (a * b)[k] ≤ M   for k in ``active_idxs``,
    b_i ≥ 0.
    Returns the optimal b (may be all zeros if LP fails).
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []

    for r, k in enumerate(active_idxs):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_vals = np.arange(lo, hi + 1, dtype=int)
        j_vals = k - i_vals
        coeffs = a[j_vals]                     # a_{k-i}
        rows.extend([r] * len(i_vals))
        cols.extend(i_vals.tolist())
        data.extend(coeffs.tolist())

    A = sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idxs), n))
    c = -np.ones(n)                            # linprog minimises, so -sum(b)
    b_ub = np.full(len(active_idxs), M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return np.zeros_like(a)
    return res.x

    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – the original 4800 works well with FFTs and LP size.
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # Stage 0 – generate a handful of cheap parametric starts.
    # ------------------------------------------------------------------
    starts = []

    # exponential family, a few decay rates
    for r in np.linspace(0.90, 0.999, 12):
        vec = np.array([r ** i for i in range(n)], dtype=float)
        vec /= vec.sum()
        starts.append(vec)

    # Gaussian bell
    sigma = n / 6.0
    center = (n - 1) / 2.0
    x = np.arange(n)
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    starts.append(g / g.sum())

    # uniform
    starts.append(np.full(n, 1.0 / n))

    # Dirichlet with moderate & low concentration
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # previous construction if available
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # up‑sample to required length (preserving the sum)
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
                seq = np.maximum(seq, 0.0)
                seq /= seq.sum()
            starts.append(seq)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose the best start according to the true evaluation function.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1} score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
            print(f"   *** new global best from starts (score={sc:.6f})")

    # ------------------------------------------------------------------
    # Stage 1 – conditional gradient (LP) refinement.
    # ------------------------------------------------------------------
    # Parameters controlling the LP/CG loop
    lp_step = 0.2                # mixing weight
    active_tol = 1e-3            # relative tolerance for active set

    while time.time() < deadline:
        conv = _conv_self(best_seq)
        M = conv.max()

        # pick the constraints that are (almost) binding
        active = np.where(conv >= M * (1.0 - active_tol))[0]
        if active.size == 0:
            active = np.arange(conv.size)          # fallback – all constraints

        # solve the LP to obtain a direction b
        b = _lp_maximising_direction(best_seq, M, active)
        sum_b = b.sum()
        if sum_b < 1e-12:
            # nothing useful can be added – exit the loop
            break
        b_norm = b / sum_b

        # mix a little towards the LP direction
        cand = (1.0 - lp_step) * best_seq + lp_step * b_norm

        # evaluate candidate
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improvement: score = {cand_score:.6f}")
        else:
            # if no improvement, reduce the mixing weight a little
            lp_step = max(lp_step * 0.9, 0.02)

        # safety check – stop if we are very close to the time limit
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Stage 2 – short sub‑gradient polishing (a few hundred thousand steps).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] polishing with sub‑gradient")
        polished_seq, polished_score = _subgrad_optimize(
            best_seq, deadline, start_time, beta=0.94, step0=0.2
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** polishing gave new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and reporting.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]


# ----------------------------------------------------------------------
# Helper to create Dirichlet starts (used above)
# ----------------------------------------------------------------------
def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x
```
This implementation follows the described multi‑stage optimisation, uses a linear‑programming convex direction to lower the maximal convolution, and finishes with a short projected sub‑gradient polish. It stays within the 1000 s budget and returns a valid non‑negative step‑function list.
