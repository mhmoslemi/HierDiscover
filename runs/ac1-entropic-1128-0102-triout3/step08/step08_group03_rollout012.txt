"""
Improved search for a non‑negative step function minimising ``evaluate_sequence``.

Algorithm
---------
* **Initialisation** – build a diverse pool of start vectors (previous best,
  uniform, Dirichlet, exponential, Gaussian).  The pool is normalised to sum‑one.
* **Exploration** – repeatedly run a fast projected sub‑gradient descent from
  random Dirichlet starts.  Each run stops after a short time slice; the best
  vector seen overall is kept.
* **Deterministic greedy refinement** – starting from the best sub‑gradient
  result we repeatedly move a fraction of the coefficient that contributes most
  to the current maximal convolution peak towards the coefficient that would
  increase that peak the least.  The move size is computed analytically
  (maximal admissible step that cannot create a new larger peak) and a line‑search
  guarantees a strict decrease of the peak.
* **Final polishing** – a few extra sub‑gradient steps are performed on the
  remaining time budget, then the final vector is normalised and returned.

All arithmetic respects the simplex (non‑negative, sum = 1) and the convolution
is evaluated via an FFT for speed (≈ O(n log n) per evaluation).  The whole
search respects the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    Returns a real numpy array.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ``conc`` and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Discrete centred Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    center = (n - 1) / 2.0
    x = np.arange(n)
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.98) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Returns the best vector seen and its score 2·n·max.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # --- sub‑gradient (active set) --------------------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        # momentum (Nesterov‑style) and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy refinement with optimal line‑search
# ----------------------------------------------------------------------


def _derivative_contrib(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Compute ∂(a*a)_k / ∂a_idx for all k (0 ≤ k < 2·n‑1) as a vector.
    Formula: d_k = a_idx + a_{k‑idx} (second term only where valid).
    """
    n = a.shape[0]
    m = 2 * n - 1
    d = np.full(m, a[idx], dtype=float)            # the constant a_idx term
    lo = idx
    hi = idx + n - 1
    # slice where k‑idx is inside [0, n‑1]
    d[lo:hi + 1] += a                                 # a_{k‑idx} = a[0:n]
    return d


def _greedy_optimize_optimal(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Greedy peak‑reduction: move mass from the coefficient that most
    contributes to the current maximal convolution peak towards the one that
    contributes the least.  The step size is computed analytically (largest
    delta that cannot create a new larger peak) and a back‑tracking line‑search
    guarantees a strict decrease.
    """
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1

    conv = _conv_self(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        cur_max = conv.max()
        k_max = conv.argmax()

        # derivative of the current peak with respect to each a_i
        idxs = np.arange(n)
        mask = (k_max - idxs >= 0) & (k_max - idxs < n)
        T = a.copy()
        T[mask] += a[k_max - idxs[mask]]

        src = int(np.argmax(T))                     # most responsible index
        # destination – index that would increase the peak the least
        dst_candidates = np.delete(np.arange(n), src)
        dst = int(dst_candidates[np.argmin(T[dst_candidates])])

        # compute derivative vectors for src and dst
        d_src = _derivative_contrib(a, src)
        d_dst = _derivative_contrib(a, dst)
        s = d_dst - d_src                           # change of each convolution entry per unit delta
        s_max = s[k_max]                            # negative (peak will go down)

        # maximal admissible delta (cannot exceed source mass)
        delta = a[src]

        # constraints from other entries that might become new max
        diff = s - s_max                           # >0  means entry rises faster than peak
        pos = diff > 1e-12
        if np.any(pos):
            bounds = (cur_max - conv[pos]) / diff[pos]
            # due to numeric noise bounds can be negative; ignore those
            bounds = bounds[bounds > 0]
            if bounds.size > 0:
                delta = min(delta, bounds.min())

        # safety factor
        delta *= 0.98

        if delta <= 1e-15:
            break

        # line‑search: ensure the move really reduces the peak
        while True:
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            # keep numeric feasibility
            if a_new[src] < 0:
                delta *= 0.5
                if delta <= 1e-15:
                    break
                continue
            conv_new = _conv_self(a_new)
            new_max = conv_new.max()
            if new_max < cur_max - 1e-14:
                # accept move
                a = a_new
                conv = conv_new
                new_score = 2.0 * n * new_max
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] greedy optimal improve: iter={it}, score={new_score:.6f}")
                break
            else:
                # step too large – halve and retry
                delta *= 0.5
                if delta <= 1e-15:
                    break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The driver runs several short sub‑gradient runs from random starts,
    keeps the best vector, then applies the deterministic greedy optimiser.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin before the forced termination
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 4800 has proved a good compromise between expressivity
    # and computational cost.
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # Build a set of diverse initial vectors.
    # ------------------------------------------------------------------
    starts = []

    # 1) previous construction, if available
    try:
        seq = np.asarray(height_sequence_1, dtype=float)   # may be undefined – guarded by try/except
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) uniform
    starts.append(np.full(n, 1.0 / n))

    # 3) exponential decay
    starts.append(_exponential_start(n, r=0.985))

    # 4) Gaussian bell
    starts.append(_gaussian_start(n))

    # 5) Dirichlet (smooth)
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))

    # 6) Dirichlet (spiky)
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # ------------------------------------------------------------------
    # Exploration phase – many short sub‑gradient runs from random starts.
    # ------------------------------------------------------------------
    subgrad_deadline = start_time + 0.60 * budget_s   # 60 % of time budget
    best_seq = None
    best_score = float("inf")

    # keep the best among the hand‑crafted starts as an initial baseline
    for a0 in starts:
        sc = 2.0 * n * _conv_self(a0).max()
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()

    # additional random starts
    while time.time() < subgrad_deadline:
        a0 = _dirichlet_start(n, conc=1.0, rng=rng)
        # each run gets at most 30 s (or the remaining time)
        run_deadline = min(subgrad_deadline, time.time() + 30.0)
        cand_seq, cand_score = _subgrad_optimize(a0, run_deadline, start_time,
                                                 beta=0.94, step0=0.5)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (subgrad run) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Greedy deterministic refinement – uses analytic line‑search.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] greedy optimal refinement start")
        final_seq, final_score = _greedy_optimize_optimal(best_seq, deadline,
                                                          start_time,
                                                          max_iters=20000)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (greedy optimal) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (guard against drift) and reporting.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # Return a plain Python list – the evaluation harness expects this.
    return [float(x) for x in best_seq.tolist()]
