def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    n=a.shape[0]
    max_val = conv.max()
    eps = max(rel_tol*max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_inds:
        i = k - idx
        mask = (i >=0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]
    if max_inds.size > 1:
        grad /= max_inds.size
    return grad

"""
Search for a non‑negative step function (vector of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv is the linear self‑convolution.  The algorithm works inside the
time budget (default 1000 s) and consists of three stages

1.  A short projected sub‑gradient descent on the true objective creates a
    non‑uniform shape with a slack in the convolution bound.
2.  A linear‑programming refinement (one LP per iteration) maximises the
    total mass of a new vector g under the current bound
    `a * g ≤ max(conv(a, a))`.  Normalising g gives a strictly better
    candidate whenever the bound is not already tight.
3.  The candidate is up‑sampled to a larger resolution and a final
    sub‑gradient phase fine‑tunes it.

All iterates live on the probability simplex (non‑negative, sum = 1), which
makes the evaluation function scale‑invariant.  The implementation uses only
NumPy and SciPy and respects the 1000 s limit without any file or network IO.
"""

import time
import math
import numpy as np
from scipy import optimize, sparse

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i ≥ 0, Σx_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv)  (true objective)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0)."""
    n = a.shape[0]
    max_val = conv.max()
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on
        F(a) = 2·n·max(conv(a))   (a lies on the simplex).
    Returns the best point seen and its true score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad n={n} iter={it:,} score={cur_score:.6f}"
            )
        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray,
               n: int,
               max_iters: int,
               deadline: float,
               start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves the LP

        max Σ g_i
        s.t.  conv(a, g) ≤ M   (element‑wise),
              g_i ≥ 0

    where M = max(conv(a, a)).  The solution is normalised to sum = 1 and
    replaces a if it improves the true objective.
    """
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        M = conv.max()

        # Build the convolution constraint matrix A (size (2n‑1)×n)
        offsets = np.arange(n)
        # each diagonal offset p contains the constant a[p]
        data = [np.full(n, a[p]) for p in range(n)]
        A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
        c = -np.ones(n)                     # minimise –Σg_i  ⇔ maximise Σg_i
        b = np.full(2 * n - 1, M)

        res = optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
        )
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # no slack – the bound is already tight
            break

        a_new = g / sum_g
        new_score = 2.0 * n * _conv_self(a_new).max()
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] LP n={n} iter={it+1} score={new_score:.6f}"
            )
        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (sum ≈ 1) ready for the evaluator.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # a tiny safety margin

    # ------------------------------------------------------------------
    # Obtain an initial construction if the harness provides one.
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = np.maximum(tmp, 0.0)
                s = starter.sum()
                if s > 0.0:
                    starter = starter / s
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Small and large resolution we will work with.
    # ------------------------------------------------------------------
    n_small = 2048          # LP works comfortably at this size
    n_large = 4096          # final fine‑tuning at a larger resolution

    # ------------------------------------------------------------------
    # Stage 1 – initialise at the small resolution.
    # ------------------------------------------------------------------
    if starter is not None and starter.size == n_small:
        a = starter.copy()
    elif starter is not None:
        a = _upsample(starter, n_small)
    else:
        a = np.random.rand(n_small)
        a /= a.sum()

    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Stage 2 – quick sub‑gradient descent to obtain a non‑uniform shape.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        sub_deadline = time.time() + min(remaining * 0.25, 40.0)
        a, _ = _run_subgradient_descent(
            a0=a,
            n=n_small,
            deadline=sub_deadline,
            start_time=start_time,
            beta=0.94,
            step0=0.25,
        )

    # ------------------------------------------------------------------
    # Stage 3 – LP refinement (several iterations).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a, _ = _lp_refine(
            a=a,
            n=n_small,
            max_iters=6,
            deadline=deadline,
            start_time=start_time,
        )

    # ------------------------------------------------------------------
    # Stage 4 – upsample to a larger resolution.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = _upsample(a, n_large)

    # ------------------------------------------------------------------
    # Stage 5 – final sub‑gradient descent at the larger resolution.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline2 = deadline
        a, _ = _run_subgradient_descent(
            a0=a,
            n=n_large,
            deadline=sub_deadline2,
            start_time=start_time,
            beta=0.94,
            step0=0.20,
        )

    # ------------------------------------------------------------------
    # Normalise one last time and return a plain list.
    # ------------------------------------------------------------------
    a = np.asarray(a, dtype=float)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full_like(a, 1.0 / a.size)

    # final logging
    final_score = 2.0 * a.size * _conv_self(a).max()
    print(
        f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {final_score:.6f}"
    )
    return [float(x) for x in a]
