"""
Propose a step‑function (non‑negative vector whose sum is 1) that minimises

    f(a) = 2·n·max(conv(a,a))

where conv is the linear self‑convolution.  The optimisation proceeds in
three stages:

1.  Many short trials (≈60 s each).  A random (or mildly structured)
    start vector is refined by an Adam optimiser on a smooth log‑sum‑exp
    surrogate of the max‑convolution.  The smoothing parameter α is
    increased from a modest to a very large value.

2.  A longer projected sub‑gradient descent (with momentum) is applied
    to the exact objective.  The iterate is kept on the probability
    simplex after every step.

3.  Finally a lightweight simulated‑annealing polish and a series of
    cheap random perturbations are performed until the time budget
    expires.

All heavy work is done with FFTs (O(n log n)) and the code respects the
global 1000 s budget.  The best sequence found is returned as a plain
Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using an FFT.
    Returns a real 1‑D array.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (non‑negative,
    sum = 1).  Returns a new array.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 alpha_start: float = 1.0,
                 alpha_end: float = 800.0,
                 max_iters: int = 12000,
                 lr0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on   f_α(a)= (1/α)·log Σ_k exp(α·conv_k(a)),
    where conv_k(a) is the self‑convolution.
    α is increased linearly from `alpha_start` to `alpha_end`.
    Returns the best iterate (lowest true objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0

    # length needed for the gradient convolution (a * w_rev)
    L_grad = _next_pow2(3 * n - 2)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current exact score (used for bookkeeping)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient of the smoothed objective
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.4
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # index where the max is attained
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-3,
               end_temp: float = 1e-6,
               steps: int = 50000) -> tuple[np.ndarray, float]:
    """
    Very light mass‑transfer annealing.  Temperature decays exponentially.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportionally to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30          # move up to 30 % of src
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Misc initialisers (random + a few deterministic shapes)
# ----------------------------------------------------------------------
def _init_random(rng: np.random.Generator, n: int) -> np.ndarray:
    a = rng.random(n)
    a /= a.sum()
    return a


def _init_cosine_u(rng: np.random.Generator, n: int) -> np.ndarray:
    """U‑shaped cosine: high at the ends, low in the middle."""
    x = np.arange(n)
    a = 1.0 + np.cos(2.0 * np.pi * x / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _init_double_gauss(rng: np.random.Generator, n: int) -> np.ndarray:
    """Two Gaussian bumps placed off‑centre."""
    sigma = n / 20.0
    centres = [n / 4.0, 3 * n / 4.0]
    x = np.arange(n)
    a = sum(np.exp(-0.5 * ((x - c) / sigma) ** 2) for c in centres)
    a /= a.sum()
    return a

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
    `evaluate_sequence`.  The routine respects the 1000 s time budget,
    runs many short trials with Adam + sub‑gradient + annealing,
    then performs an intensive polish on the best found sequence.
    Returns the sequence as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5               # small safety margin

    # problem size – 4096 is FFT‑friendly and offers enough freedom
    n = 4096

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Global starter (if the harness provided one)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = starter / starter.sum()
            cur = 2.0 * n * _conv_self(starter).max()
            best_seq = starter.copy()
            best_score = cur
            print(f"[{time.time() - start:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Short trials (≈60 s each).  We keep a pool of different
    # initialisers to increase diversity.
    # ------------------------------------------------------------------
    min_trial_time = 60.0
    final_margin = 30.0
    init_funcs = [_init_random, _init_cosine_u, _init_double_gauss]

    while True:
        remaining = deadline - time.time()
        if remaining < min_trial_time + final_margin:
            break

        trial_deadline = time.time() + min_trial_time

        # pick an initialiser (occasionally perturb the current best)
        if best_seq is not None and rng.random() < 0.25:
            a0 = best_seq + 0.005 * rng.standard_normal(n)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-12:
                a0 = _init_random(rng, n)
            else:
                a0 /= a0.sum()
        else:
            init = rng.choice(init_funcs)
            a0 = init(rng, n)

        # --------------------------------------------------------------
        # Phase‑1 : Adam on smooth max
        # --------------------------------------------------------------
        a_adam, score_adam = _adam_smooth(
            a0,
            n,
            trial_deadline,
            start,
            alpha_start=1.0,
            alpha_end=800.0,
            max_iters=12000,
            lr0=0.4,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start:.2f}s] Trial Adam improve: {score_adam:.6f}")

        # --------------------------------------------------------------
        # Phase‑2 : exact sub‑gradient descent
        # --------------------------------------------------------------
        sub_deadline = trial_deadline - 3.0           # keep a few seconds for polish
        a_sub, score_sub = _subgradient_descent(
            a_adam,
            n,
            sub_deadline,
            start,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start:.2f}s] Trial sub‑grad improve: {score_sub:.6f}")

        # --------------------------------------------------------------
        # Phase‑3 : quick annealing polish
        # --------------------------------------------------------------
        a_fin, score_fin = _annealing(
            a_sub,
            n,
            trial_deadline,
            start,
            best_score,
            a_sub,
            start_temp=5e-3,
            end_temp=1e-6,
            steps=50000,
        )
        if score_fin < best_score:
            best_score = score_fin
            best_seq = a_fin.copy()
            print(f"[{time.time() - start:.2f}s] Trial anneal improve: {score_fin:.6f}")

    # ------------------------------------------------------------------
    # Final intensive refinement on the best candidate.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()

    # long sub‑gradient pass (≈70 % of the remaining time)
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.65
        a_sub, score_sub = _subgradient_descent(
            best_seq,
            n,
            sub_deadline,
            start,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start:.2f}s] Final sub‑grad improve: {score_sub:.6f}")

    # high‑α Adam refinement (the last few seconds)
    if time.time() < deadline:
        a_ref, score_ref = _adam_smooth(
            best_seq,
            n,
            deadline,
            start,
            alpha_start=200.0,
            alpha_end=2000.0,
            max_iters=18000,
            lr0=0.2,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start:.2f}s] Final high‑α Adam improve: {score_ref:.6f}")

    # ------------------------------------------------------------------
    # Random perturb‑and‑refine loop (as long as time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 0.5:
        # tiny Gaussian perturbation of the current best
        a0 = best_seq + 0.02 * rng.standard_normal(n)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-12:
            continue
        a0 /= a0.sum()

        # a few quick sub‑gradient steps (≈0.8 s)
        sub_deadline = time.time() + 0.8
        a_loc, score_loc = _subgradient_descent(
            a0,
            n,
            sub_deadline,
            start,
            best_score,
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start:.2f}s] Perturb‑refine improve: {score_loc:.6f}")
            continue

        # if sub‑gradient didn't improve, try a very short anneal
        anneal_deadline = time.time() + 0.6
        a_loc, score_loc = _annealing(
            a0,
            n,
            anneal_deadline,
            start,
            best_score,
            a0,
            start_temp=1e-2,
            end_temp=1e-6,
            steps=15000,
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start:.2f}s] Perturb‑anneal improve: {score_loc:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
