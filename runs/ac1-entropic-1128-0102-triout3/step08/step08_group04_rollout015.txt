def _adam_optimize(a0, n, deadline, start_time):
    a = a0.copy()
    best = a.copy()
    best_score = float('inf')
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.1
    it = 0
    alpha_schedule = [1.0, 3.0, 10.0, 30.0, 100.0]
    alpha_idx = 0
    steps_per_alpha = 4000
    while time.time() < deadline:
        it += 1
        alpha = alpha_schedule[min(alpha_idx, len(alpha_schedule)-1)]
        conv, grad = _gradient_smoothed_max(a, alpha)  # returns conv and grad
        # compute score
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: iter={it}, score={cur_score:.6f}")
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / math.sqrt(it)
        upd = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - upd
        # project onto simplex
        a = _project_simplex(a)
        # update alpha schedule index
        if it % steps_per_alpha == 0 and alpha_idx < len(alpha_schedule)-1:
            alpha_idx += 1
    return best, best_score

"""
Hybrid Adam‑gradient + incremental refinement for minimizing
evaluate_sequence.  Multiple initialisations (uniform, random,
down‑sampled previous best) are tried; each run uses Adam on the
smoothed max (soft‑max of the self‑convolution) with a simplex
projection, then a massive local‑move refinement that updates the
convolution analytically.  The routine respects the 1000 s budget
and returns the best sequence found.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT (output length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _resample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n`, preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   lr0: float = 0.15,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation of the smoothed max of the
    self‑convolution.  Returns the best sequence seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    it = 0
    alpha_schedule = [1.0, 3.0, 10.0, 30.0, 80.0, 200.0]
    alpha_idx = 0
    steps_per_alpha = 5000

    while time.time() < deadline:
        it += 1
        # ----- smoothed max and its gradient -----
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, len(alpha_schedule) - 1)]
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_rev = w[::-1]

        L = _next_pow2(3 * n)  # large enough for correlation
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)
        upd = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - upd
        a = _project_simplex(a)

        # schedule change
        if it % steps_per_alpha == 0 and alpha_idx < len(alpha_schedule) - 1:
            alpha_idx += 1

    return best_a, best_score

def _run_incremental_refine(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           eps_start: float = 0.2,
                           eps_end: float = 0.001,
                           rel_improve: float = 1e-12) -> np.ndarray:
    """
    Tiny‑mass moves that analytically update the self‑convolution.
    An epsilon that shrinks from eps_start to eps_end is used.
    Only moves that strictly lower the current maximum are kept.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        # linear decay of epsilon fraction
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        k_max = int(np.argmax(conv))

        # source: weighted by its contribution to the current max bin
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            # fallback: pick proportionally to the current mass
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # destination: prefer indices that contribute little to the max bin
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                     # never stay in place
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # amount of mass to move – a fraction of the source mass
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # analytical update of the convolution
        conv_new = conv.copy()
        # indices where `src` and `dst` appear in the convolution sums
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        if new_max < max_val - rel_improve:
            # accept move
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] inc‑refine improve: iter={it:,}, max={best_max:.8f}")

    return best_a

        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)
        upd = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - upd
        a = _project_simplex(a)

        # ----- advance smoothing schedule -----
        if it % steps_per_alpha == 0 and alpha_idx < len(alpha_schedule) - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑B : Massive incremental refinement (tiny mass moves)
# ----------------------------------------------------------------------
def _run_incremental_refine(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           eps_start: float = 0.2,
                           eps_end: float = 0.001,
                           rel_improve: float = 1e-14) -> np.ndarray:
    """
    Perform millions of tiny mass‑shifts away from the coordinates that
    contribute most to the current maximal convolution entry.
    Updates of the convolution are done analytically, making a single
    iteration O(n) and allowing a huge number of iterations inside the
    budget.
    """
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        k_max = int(np.argmax(conv))

        # ---- choose source (high contribution to the max bin) ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ---- choose destination (low contribution to the max bin) ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                         # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ---- amount to move -------------------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ---- propose new convolution (O(n) update) -------------------------
        conv_new = conv.copy()
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        if new_max < max_val - rel_improve:
            # accept move
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] inc‑refine improve: iter={it:,}, max={best_max:.8f}")

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Use a moderate length – more iterations fit into the time budget.
    n = 2400

    # ------------------------------------------------------------------
    # Build a collection of initial guesses.
    # ------------------------------------------------------------------
    init_seqs = []

    # 1) uniform distribution
    init_seqs.append(np.full(n, 1.0 / n))

    # 2) random Dirichlet vector
    init_seqs.append(np.random.dirichlet(np.ones(n)))

    # 3) down‑sampled previous construction if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        if old.ndim == 1 and old.size > 0:
            init_seqs.append(_resample(old, n))
    except Exception:
        pass

    best_seq = None
    best_score = float("inf")
    total_seeds = len(init_seqs)

    # ------------------------------------------------------------------
    # Optimise each start in turn, giving each a fair share of the budget.
    # ------------------------------------------------------------------
    for seed_idx, seq in enumerate(init_seqs):
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # Allocate time proportionally to the number of seeds still to try.
        time_for_this = remaining / (total_seeds - seed_idx)
        sub_deadline = time.time() + time_for_this

        # Phase A – Adam optimisation on the smoothed max
        a_opt, _ = _adam_optimize(seq, n, sub_deadline, start_time)

        # Phase B – massive incremental refinement (use any time left)
        if time.time() < sub_deadline:
            a_ref = _run_incremental_refine(
                a_opt,
                n,
                sub_deadline,
                start_time,
                eps_start=0.2,
                eps_end=0.001,
                rel_improve=1e-14,
            )
        else:
            a_ref = a_opt

        # Evaluate the candidate
        conv = _conv_self(a_ref)
        final_score = 2.0 * n * conv.max()
        print(f"[{time.time() - start_time:.2f}s] seed {seed_idx} final score: {final_score:.6f}")

        if final_score < best_score:
            best_score = final_score
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] NEW GLOBAL BEST (score={best_score:.6f})")

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing succeeded.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Normalise to sum‑to‑one (the objective is scale‑invariant).
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return as plain Python list.
    return [float(x) for x in best_seq]
