"""
Propose a step‑function (non‑negative list of heights) that minimizes the
``evaluate_sequence`` objective.

The algorithm combines several complementary heuristics:

1. **Multiple random restarts** – each trial starts from a Dirichlet sample
   (or the supplied global starter) and runs a short Adam optimisation on a
   smoothed log‑sum‑exp surrogate of the max‑convolution.  The smoothing
   parameter α is increased up to a large value, so the optimiser gradually
   focuses on the true L∞‑objective.

2. **Projected sub‑gradient descent** – after the smooth phase a classic
   sub‑gradient step (with momentum) is taken on the exact objective
   ``2·n·max(conv(a,a))``.  The iterate is projected onto the probability
   simplex after every step.

3. **Exponentiated‑gradient descent** – a mirror‑descent step with KL‑
   divergence (multiplicative updates) explores an alternative search
   direction that respects non‑negativity and the simplex.

4. **Candidate mixing** – the best few candidates obtained from the restarts
   are linearly mixed (convex combinations).  Because the convolution is
   quadratic, mixing different shapes can cancel peaks and improve the bound.

5. **Light simulated‑annealing polish** – finally a cheap annealing that moves
   mass between two coordinates is applied until the time budget is exhausted.

All phases respect the global 1000 s budget (a 0.5 s safety margin).  The
function returns a plain Python list of floats that sum to 1 (the evaluation
routine rescales anyway).  The implementation uses only NumPy and the
standard library, runs on a single CPU and conforms to the required API.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear (non‑circular) self‑convolution a*a using FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 alpha_start: float = 0.5,
                 alpha_end: float = 800.0,
                 max_iters: int = 6000,
                 lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on f_α(a)=1/α·log Σ exp(α·conv(a,a)).
    α is increased linearly from alpha_start to alpha_end.
    Returns the best iterate (lowest true objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    # length for FFTs used in gradient
    L_grad = _next_pow2(3 * n - 2)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # linearly increase smoothing
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights over convolution entries
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient of the smoothed objective (see derivations in original code)
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact max – projected sub‑gradient descent (additive updates)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Simple momentum (β≈0.9) and step size η_t = step0/√t.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # identify one maximising shift (any will do)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2‑b : exponentiated‑gradient (mirror descent with KL)
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the L∞‑objective using the KL‑divergence
    (multiplicative updates).  Works directly on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp‑grad it={it:,} score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # step size decays like 1/√t
        eta = 0.5 / math.sqrt(it + 1.0)
        # multiplicative update
        a = a * np.exp(-eta * grad)
        a = _project_simplex(a)

    return best_a, best_score

# Phase‑3 : very light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-6,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealing.  Temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # sample a source proportionally to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase‑4 : mixing of top candidates
# ----------------------------------------------------------------------
def _mix_candidates(candidates: list[np.ndarray],
                    n: int,
                    deadline: float,
                    start_time: float,
                    cur_best_seq: np.ndarray,
                    cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Random convex combinations of the best few candidates.
    Keeps any improvement found.
    """
    if not candidates:
        return cur_best_seq, cur_best_score

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.choice(len(candidates), size=2, replace=False)
        lam = rng.random()
        mix = lam * candidates[i] + (1.0 - lam) * candidates[j]
        # mixing already preserves the simplex, but numerical errors are removed
        mix = np.maximum(mix, 0.0)
        mix_sum = mix.sum()
        if mix_sum == 0:
            continue
        mix /= mix_sum

        score = 2.0 * n * _conv_self(mix).max()
        if score < best_score:
            best_score = score
            best_seq = mix.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix improve: score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` objective within the given time budget.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – leave a little buffer before the harness cuts us off
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Determine problem size (use the known global starter if present)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        n = starter.size
        use_global = True
        starter = np.maximum(starter, 0.0)
        starter_sum = starter.sum()
        if starter_sum > 0:
            starter /= starter_sum
    except Exception:
        # fallback – a power‑of‑two size that is FFT‑friendly
        n = 4096
        use_global = False
        starter = None

    # Book‑keeping for the overall best solution
    best_seq = None
    best_score = float("inf")
    top_candidates = []          # list of (score, sequence) – keep a few best

    # Helper to evaluate a sequence and possibly update the global best
    def _maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_score, top_candidates
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0.0:
            return
        seq = seq / s
        score = 2.0 * n * _conv_self(seq).max()
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best: {score:.6f}")
        # keep up to 8 top candidates
        if len(top_candidates) < 8:
            top_candidates.append(seq.copy())
        else:
            # replace a random worst if we improve
            worst_idx = np.argmax([2.0 * n * _conv_self(c).max() for c in top_candidates])
            worst_score = 2.0 * n * _conv_self(top_candidates[worst_idx]).max()
            if score < worst_score:
                top_candidates[worst_idx] = seq.copy()

    # ------------------------------------------------------------------
    # Phase A – many short random‑restart trials
    # ------------------------------------------------------------------
    trial_time = 35.0               # seconds allocated per trial
    mixing_margin = 30.0            # left for mixing + final polish

    while True:
        remaining = deadline - time.time()
        if remaining < trial_time + mixing_margin:
            break

        trial_deadline = time.time() + trial_time

        # ----- initial point -------------------------------------------------
        if use_global and np.random.rand() < 0.15:
            a0 = starter.copy()
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ----- Phase‑1 : Adam on smooth max ---------------------------------
        a_adam, _ = _adam_smooth(
            a0,
            n,
            trial_deadline,
            start_time,
            alpha_start=0.5,
            alpha_end=800.0,
            max_iters=6000,
            lr0=0.6,
        )
        _maybe_update(a_adam)

        # ----- Phase‑2 : projected sub‑gradient descent ----------------------
        sub_deadline = time.time() + trial_time * 0.45
        a_sub, _ = _subgradient_descent(
            a_adam,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        _maybe_update(a_sub)

        # ----- Phase‑2‑b : exponentiated‑gradient descent --------------------
        exp_deadline = time.time() + trial_time * 0.20
        a_exp, _ = _exp_gradient_descent(
            a_sub,
            n,
            exp_deadline,
            start_time,
            best_score,
        )
        _maybe_update(a_exp)

        # ----- Brief annealing to escape possible tiny plateaus -------------
        anneal_deadline = time.time() + trial_time * 0.10
        a_ann, _ = _annealing(
            a_exp,
            n,
            anneal_deadline,
            start_time,
            best_score,
            a_exp,
            start_temp=1e-2,
            end_temp=1e-6,
            steps=30_000,
        )
        _maybe_update(a_ann)

    # ------------------------------------------------------------------
    # Phase B – mixing of the best few candidates discovered so far
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and len(top_candidates) >= 2:
        mix_deadline = time.time() + remaining * 0.45
        best_seq, best_score = _mix_candidates(
            top_candidates,
            n,
            mix_deadline,
            start_time,
            best_seq if best_seq is not None else top_candidates[0],
            best_score,
        )
        _maybe_update(best_seq)

    # ------------------------------------------------------------------
    # Phase C – final simulated‑annealing polish (as much time as left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _annealing(
            best_seq if best_seq is not None else np.full(n, 1.0 / n),
            n,
            deadline,
            start_time,
            best_score,
            best_seq if best_seq is not None else np.full(n, 1.0 / n),
            start_temp=5e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        _maybe_update(final_seq)

    # ------------------------------------------------------------------
    # Normalise one last time and convert to plain Python list.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
