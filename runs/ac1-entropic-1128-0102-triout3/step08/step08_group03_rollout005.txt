"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Diverse starts** – a handful of analytically‑motivated initial
   sequences (uniform, exponential, Gaussian, Dirichlet, plus the
   previous best if available) are normalised to a probability simplex.

2. **Soft‑max Adam** – each start is refined by a continuation method:
   a smooth log‑sum‑exp approximation of the max–convolution is minimised
   with Adam while the smoothing parameter α is increased geometrically.
   The best Adam result is kept.

3. **Projected sub‑gradient descent with restarts** – the objective
   ``f(a)=max_k (a*a)[k]`` is tackled directly.  The sub‑gradient of the
   active constraints is computed efficiently via convolution, a
   momentum term is added and the step‐size decays as 1/√t.  If the
   method stagnates for a while the momentum is cleared (hard restart)
   and a tiny random perturbation is added to escape flat regions.

4. **Random local search** – a cheap hill‑climbing phase that repeatedly
   transfers a small amount of mass from the coefficient that most
   contributes to the current peak towards the smallest coefficient.
   Only moves that strictly lower the objective are kept.

All phases respect a global time budget (default 1000 s).  The final
vector is clipped, re‑normalised to sum‑one and returned as a plain
Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    # https://arxiv.org/abs/1101.6081 (section 2)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


def _score_from_seq(a: np.ndarray) -> float:
    """Exact evaluation of the objective (faster than evaluate_sequence)."""
    conv = _conv_self(a)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * a.shape[0] * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max Adam (smooth max optimisation)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    alpha_schedule=None,
    iters_per_alpha: int = 3000,
    lr0: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Adam on the smoothed max‑convolution objective."""
    if alpha_schedule is None:
        alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]

    a = a0.copy()
    n = a.shape[0]
    # work in the log‑space to stay on the simplex via a soft‑max
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_sc = _score_from_seq(a)

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_sc

            # soft‑max ↔ probability vector
            maxy = np.max(y)
            expy = np.exp(y - maxy)
            a = expy / expy.sum()

            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            # log‑sum‑exp safely
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha           # smooth approximation of max
            obj = 2.0 * n * f_val

            # gradient of the smooth max (soft‑max weights)
            w = np.exp(scaled - max_scaled)
            w /= w.sum()                   # shape (2n‑1,)
            # cross‑correlation a ⋆ w  (grad = 2·(a⋆w))
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            grad_a *= 2.0 * n               # chain rule for the outer factor

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 200 == 0:
                sc = _score_from_seq(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with restarts
# ----------------------------------------------------------------------


def _subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    step0: float = 0.3,
    beta: float = 0.94,
    stagnation_limit: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max_k (a*a)[k].

    A hard restart is performed when the objective has not improved for
    ``stagnation_limit`` iterations: momentum is cleared and a tiny random
    perturbation is added to the current iterate.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_sc = _score_from_seq(a)
    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            stagn = 0
            print(
                f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_sc:.6f}"
            )
        else:
            stagn += 1

        # ---- sub‑gradient of the max (active set) --------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # double‑counting is intentional (gradient of a_i a_j)

        # momentum (Nesterov‑style) and step‑size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # hard restart if we have been stuck too long
        if stagn >= stagnation_limit:
            # clear momentum and add a small jitter
            momentum[:] = 0.0
            jitter = np.random.default_rng().normal(scale=1e-4, size=n)
            a = _project_simplex(a + jitter)
            stagn = 0
            continue

        a = _project_simplex(a - step * momentum)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Random local improvement (hill climbing)
# ----------------------------------------------------------------------


def _random_local_search(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly move a modest fraction of mass from
    the coefficient that contributes most to the current peak towards the
    currently smallest coefficient.  Only moves that strictly lower the
    objective are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    best_sc = _score_from_seq(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_moves:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        # indices that participate in the current maximal convolution term
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]  # most responsible coefficient

        # destination = smallest non‑source coefficient
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # move a modest fraction (capped to keep both entries non‑negative)
        eps = min(0.5 * a[src], 0.001)
        if eps <= 1e-14:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_sc = _score_from_seq(a_prop)
        if new_sc < best_sc - 1e-12:
            a = a_prop
            best_sc = new_sc
            print(
                f"[{time.time() - start_time:6.2f}s] random local improve: iter={it:,}, score={new_sc:.6f}"
            )

    return a, best_sc

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Discrete centred Gaussian (σ≈n/6)."""
    sigma = n / 6.0
    centre = (n - 1) / 2.0
    x = np.arange(n)
    g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.98) -> np.ndarray:
    """Geometric decay a_i ∝ r**i."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()
