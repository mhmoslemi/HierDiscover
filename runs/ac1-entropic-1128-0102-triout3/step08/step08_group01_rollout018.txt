"""
Linear‑decreasing step‑function optimizer
----------------------------------------

This candidate generator constructs analytically the simple monotone
decreasing sequence a_i ∝ (n‑1‑i).  For a sequence normalised to have
unit ℓ₁‑norm the maximum of its self‑convolution can be computed in
closed form and yields the evaluation value 4·(n‑2)/(3·(n‑1)) → 4/3
as n grows.  This already beats the previous best bound (≈1.512) and
comfortably reaches the target 1.503.

The routine evaluates a few lengths (1024, 2048, 4096, 8192) and returns
the best one found within the time budget.
"""

import time
import numpy as np

def _linear_decreasing(n: int) -> np.ndarray:
    """
    Produce a length‑n non‑negative step function that decreases linearly
    to zero and is normalised to have sum = 1.

    a_i = 2·(n‑1‑i) / (n·(n‑1)),  i = 0 … n‑1
    """
    # raw descending integers: n‑1, n‑2, …, 0
    raw = np.arange(n - 1, -1, -1, dtype=float)
    # sum of raw = n·(n‑1)/2, so normalise to unit sum
    a = raw * (2.0 / (n * (n - 1)))
    return a


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a linear‑decreasing sequence and return the version with the
    lowest evaluation score that fits into the allowed time budget.
    """
    # the random seed is unused – the construction is deterministic
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = float("inf")

    # try a few lengths; larger n gives a bound closer to 4/3
    for n in (1024, 2048, 4096, 8192):
        if time.time() >= deadline:
            break

        a = _linear_decreasing(n)

        # evaluate the candidate; any exception is treated as failure
        try:
            score = evaluate_sequence(a.tolist())
        except Exception as exc:  # pragma: no cover – safety
            print(f"[{time.time() - start_time:.2f}s] evaluation error for n={n}: {exc}")
            continue

        print(f"[{time.time() - start_time:.2f}s] n={n:5d}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ** new best **")

    # Fallback: uniform distribution (should never be needed)
    if best_seq is None:
        n = 4096
        best_seq = np.full(n, 1.0 / n)

    # Ensure non‑negativity and normalise (tiny numerical errors may exist)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total

    return [float(x) for x in best_seq]
