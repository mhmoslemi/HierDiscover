"""
Propose a step‑function that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq))/sum(seq)^2

The algorithm works on the probability simplex (∑a_i = 1) and
optimises the smooth log‑sum‑exp surrogate of the max‑convolution
by a continuation method (decreasing τ).  At each τ a full
L‑BFGS‑B run (gradient supplied analytically) finds a high‑quality
local optimum.  The solution is then refined by a short projected
sub‑gradient descent and a light simulated‑annealing mass‑transfer.
The whole search respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Averaged sub‑gradient of   max(conv)   w.r.t. a   (a ≥ 0).
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Surrogate (log‑sum‑exp) objective and its gradient w.r.t. the
# unconstrained soft‑max parameters w (a = softmax(w)).
# ----------------------------------------------------------------------


def _surrogate_val_and_grad(w: np.ndarray,
                            tau: float,
                            n: int) -> tuple[float, np.ndarray]:
    """
    Returns (value, gradient) of
        O_τ(w) = 2·n·τ·log∑exp(conv(softmax(w))/τ)
    and its gradient w.r.t. w.
    """
    # soft‑max mapping a = exp(w)/∑exp(w)
    max_w = np.max(w)
    e = np.exp(w - max_w)
    a = e / e.sum()

    # convolution a*a
    conv = _conv_self(a)

    # log‑sum‑exp in a numerically stable way
    z = conv / tau
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    sum_exp = exp_z.sum()
    val = 2.0 * n * tau * (max_z + np.log(sum_exp))

    # p = softmax over convolution entries
    p = exp_z / sum_exp

    # gradient of O_τ w.r.t. a (see paper notes)
    a_rev = a[::-1]
    conv_rev = _conv_fft(a_rev, p)                     # length = n + (2n‑1)‑1 = 3n‑2
    inner = conv_rev[n - 1: n - 1 + n]                 # slice of length n
    grad_a = 4.0 * n * inner

    # chain rule through soft‑max: ∇_w O = a ⊙ (grad_a − ⟨a,grad_a⟩)
    dot = np.dot(a, grad_a)
    grad_w = a * (grad_a - dot)
    return float(val), grad_w.astype(np.float64)


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a plain Python list of floats (sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose a problem size according to the remaining budget.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n = 16384
    elif remaining > 300:
        n = 8192
    else:
        n = 4096

    # --------------------------------------------------------------
    # Initialise a feasible probability vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=float)
            init = _upsample(init, n)
        except Exception:
            init = None
    else:
        init = None

    if init is None:
        # a simple symmetric triangular shape (peaked in the centre)
        i = np.arange(n, dtype=float)
        init = np.minimum(i + 1, n - i)
    init = np.maximum(init, 0.0)
    if init.sum() == 0.0:
        init = np.full(n, 1.0 / n)
    else:
        init = init / init.sum()

    # Parameters for the soft‑max representation
    eps = 1e-16
    w = np.log(init + eps)                # w such that a = softmax(w)

    # --------------------------------------------------------------
    # Continuation on τ : start rather smooth, finish very tight.
    # --------------------------------------------------------------
    conv0 = _conv_self(init)
    tau = max(conv0.max() * 10.0, 1e-6)    # initial temperature
    min_tau = 1e-11

    # bookkeeping of the best solution seen
    best_a = init.copy()
    best_score = 2.0 * n * conv0.max()

    # small helper to compute the true objective quickly
    def true_score(a_vec: np.ndarray) -> float:
        return 2.0 * n * _conv_self(a_vec).max()

    # ------------------------------------------------------------------
    # Main optimisation loop over decreasing τ.
    # ------------------------------------------------------------------
    while time.time() < deadline and tau > min_tau:
        time_left = deadline - time.time()
        # allocate a fraction of the remaining time to this τ‑stage
        stage_time = min(time_left * 0.3, max(5.0, time_left * 0.1))
        # estimate a generous number of L‑BFGS iterations (each iteration ≈ one fn+grad eval)
        max_iter = int(min(4000, max(200, stage_time * 200)))   # 200 it/s is a safe upper bound

        # L‑BFGS‑B on the smooth surrogate (value & gradient supplied together)
        try:
            from scipy.optimize import minimize
            res = minimize(fun=lambda w_vec: _surrogate_val_and_grad(w_vec, tau, n),
                           x0=w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter,
                                    'disp': False})
            if res.success:
                w = res.x
                a = np.exp(w - np.max(w))
                a = a / a.sum()
                cur = true_score(a)
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] τ={tau:.2e} L‑BFGS improve: score={cur:.6f}")
            else:
                # fall‑back to a single gradient step if L‑BFGS failed
                val, grad = _surrogate_val_and_grad(w, tau, n)
                step = 0.2 / (1.0 + np.log(1.0 / tau))
                w = w - step * grad
        except Exception:
            # SciPy not available – simple gradient descent fallback
            val, grad = _surrogate_val_and_grad(w, tau, n)
            step = 0.2 / (1.0 + np.log(1.0 / tau))
            w = w - step * grad

        # tighten τ
        tau *= 0.3

    # --------------------------------------------------------------
    # Final refinement with projected sub‑gradient descent (simplex).
    # --------------------------------------------------------------
    a = best_a.copy()
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.15
    max_iters = 3000

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        cur = 2.0 * n * conv.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad refine iter={it} score={cur:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    # --------------------------------------------------------------
    # Very light simulated‑annealing “mass‑transfer” for the last seconds.
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_a.copy()
        cur_best = best_score
        best_seq = a.copy()
        rng = np.random.default_rng()
        steps = 200_000
        start_temp = 3e-2
        end_temp = 1e-7

        for it in range(1, steps + 1):
            if time.time() >= deadline:
                break
            # exponential cooling schedule
            T = start_temp * (end_temp / start_temp) ** (it / steps)

            # choose a source proportional to its mass
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

            eps_transfer = a[src] * rng.random() * 0.25
            a_prop = a.copy()
            a_prop[src] -= eps_transfer
            a_prop[dst] += eps_transfer

            new_score = true_score(a_prop)
            delta = new_score - cur_best
            if delta < 0 or math.exp(-delta / T) > rng.random():
                a = a_prop
                cur_best = new_score
                if cur_best < best_score:
                    best_score = cur_best
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={cur_best:.6f}")

        best_a = best_seq

    # ------------------------------------------------------------------
    # Normalise finally (to avoid any tiny numerical drift) and return.
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
