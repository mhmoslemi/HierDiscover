def _anneal_optimize(a0, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    n = a.shape[0]
    # parameters
    T0 = 0.01  # initial temperature (objective units)
    # but we need to calibrate: typical Δ ~ 0.001 maybe.
    # Use temperature scaling relative to initial best_score.
    # Or use exponential cooling: T = T0 * (remaining_time/total_time)
    total_time = deadline - start_time
    it = 0
    while time.time() < deadline:
        it += 1
        # choose source i with a_i > eps
        src = rng.integers(n)
        while a[src] < 1e-12:
            src = rng.integers(n)
        # Choose destination j != src
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1
        # choose eps fraction of source (e.g., 10% of a[src]) but ensure not too small.
        delta = a[src] * rng.uniform(0.01, 0.2)  # 1% to 20% of src
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # compute new score
        new_score = evaluate_sequence(a_new.tolist())
        # acceptance
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={new_score:.6f}")
            continue
        # else maybe accept
        # compute temperature
        remaining = deadline - time.time()
        # linear schedule: T = T0 * (remaining / total_time)
        T = T0 * (remaining / total_time)
        # probability
        if rng.random() < math.exp(-(new_score - best_score) / T):
            a = a_new
    return best_a, best_score
```

"""
Propose a non–negative step function that (almost) minimises
evaluate_sequence.

Algorithm
---------

1. **Initialisation** – Start from the previously found construction
   (global variable ``height_sequence_1``) or from a uniform vector.
   Normalise to a probability vector (sum = 1).

2. **Exploration** – A few Adam steps on a smoothed version of the
   max‑convolution objective.  The smoothing parameter α is increased
   gradually (0.5 → … → 32768) so that the landscape becomes progressively
   closer to the true max.  The best candidate from this phase is kept.

3. **Polishing** – Projected sub‑gradient descent on the true objective
   ``f(a)=max_k (a*a)[k]`` with Nesterov‑style momentum.  The step size
   decays as 1/√t.  This phase runs for most of the remaining budget.

4. **Stochastic local search** – Simulated annealing on the simplex.
   In each iteration a random source index donates a random fraction of
   its mass to a random destination index.  Moves that improve the
   objective are always kept; uphill moves are accepted with a
   temperature‑controlled probability.  This cheap “shake‑and‑refine”
   often pushes the solution a little further than pure sub‑gradient
   descent.

All phases respect the global time limit (default 1000 s).  The final
vector is clamped to non‑negative values, renormalised and returned as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    Returns a real array.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    # pick the n entries that correspond to the gradient indices
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform a few Adam iterations on the smooth approximation of the
    max‑convolution objective.  Returns the best vector found and its
    exact score (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]
    # work in the log‑space to enforce positivity and simplex constraints
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule (α grows → sharper approximation of max)
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0, 32768.0]
    iters_per_alpha = 2500

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max → probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # convolution and smoothed objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val  # same scaling as the true objective

            # gradient wrt a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # back‑prop through soft‑max (Jacobian)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 200 == 0:
                cur_score = 2.0 * n * _conv_self(a).max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                          f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.95,
                      step0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max_k (a*a)[k].
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # --- sub‑gradient (active set) --------------------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – correct derivative

        # momentum and step‑size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Simulated annealing on the simplex
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     rng: np.random.Generator,
                     min_frac: float = 0.01,
                     max_frac: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic local search.  Randomly move a fraction of the mass
    from a source index to a destination index.  Accept improvements
    always; uphill moves with Metropolis probability.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    total_time = max(deadline - start_time, 1e-6)
    it = 0

    # initial temperature – a small fraction of the current score
    T0 = 0.02 * best_score if best_score > 0 else 1e-3

    while time.time() < deadline:
        it += 1
        # pick source with non‑zero mass
        src = rng.integers(n)
        while a[src] <= 1e-12:
            src = rng.integers(n)

        # pick destination different from source
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # choose a fraction of src to move
        frac = rng.uniform(min_frac, max_frac)
        delta = a[src] * frac
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # exact score (sum of a stays 1, so just 2n*max(conv))
        new_score = 2.0 * n * _conv_self(a_new).max()

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={new_score:.6f}")
            continue

        # uphill move – Metropolis criterion
        remaining = deadline - time.time()
        T = T0 * (remaining / total_time)   # linear cooling
        if T > 0 and rng.random() < math.exp(-(new_score - best_score) / T):
            a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine consists of three phases:
      * Adam exploration on a smoothed max‑convolution,
      * long sub‑gradient polishing,
      * final simulated‑annealing refinement.
    The best sequence found before the deadline is returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 4800 entries work well with the FFT sizes used.
    # --------------------------------------------------------------
    n = 4800

    # --------------------------------------------------------------
    # Build initial candidate (use previous construction if available)
    # --------------------------------------------------------------
    try:
        seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if seq.ndim != 1 or seq.size == 0:
            raise ValueError
        if seq.shape[0] != n:
            # linear interpolation to required length, preserving total mass
            old_x = np.linspace(0.0, 1.0, seq.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            seq = np.interp(new_x, old_x, seq)
        seq = np.clip(seq, 0.0, None)
        s = seq.sum()
        if s > 0:
            seq = seq / s
        else:
            seq = np.full(n, 1.0 / n)
        init = seq.astype(float)
    except Exception:
        init = np.full(n, 1.0 / n)

    best_seq = init.copy()
    best_score = 2.0 * n * _conv_self(best_seq).max()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam exploration (≈20 % of the budget)
    # --------------------------------------------------------------
    explore_deadline = start_time + 0.20 * budget_s
    if time.time() < explore_deadline:
        print(f"[{time.time() - start_time:6.2f}s] entering Adam exploration")
        cand_seq, cand_score = _adam_optimize(best_seq, explore_deadline, start_time)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW BEST (Adam) "
                  f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈65 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] entering sub‑gradient polishing")
        polished_seq, polished_score = _subgrad_optimize(
            best_seq, deadline, start_time, beta=0.95, step0=0.5)
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW BEST (polish) "
                  f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Simulated annealing refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] entering annealing refinement")
        final_seq, final_score = _anneal_optimize(
            best_seq, deadline, start_time, rng)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW BEST (anneal) "
                  f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # sanity check with the official evaluator

    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
