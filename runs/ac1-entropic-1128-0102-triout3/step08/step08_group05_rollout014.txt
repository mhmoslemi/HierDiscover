def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # schedule length
    length_schedule = [1024, 2048, 4096, 8192]

    best_seq = None
    best_score = float('inf')
    
    for n in length_schedule:
        if time.time() >= deadline:
            break
        # initialize a
        if best_seq is None:
            a = np.full(n, 1.0/n)
        else:
            a = _upsample(best_seq, n)
        # convert to logits
        eps = 1e-12
        x = np.log(a + eps)  # avoid log 0

        # compute initial tau
        conv = _conv_self(a)
        tau = max(conv.max() * 10.0, 1e-4)  # ensure not too tiny
        # set tau schedule
        tau_factors = np.geomspace(tau, 1e-8, num=6)  # decreasing geometric
        for tau_curr in tau_factors:
            if time.time() >= deadline:
                break
            # define objective/grad closure using partial function
            def fun_and_grad(x_vec):
                val, grad = _smooth_obj_grad(x_vec, n, tau_curr)
                return val, grad
            # run minimize
            res = minimize(fun=lambda x_vec: _smooth_obj_grad(x_vec, n, tau_curr)[0],
                           x0=x,
                           jac=lambda x_vec: _smooth_obj_grad(x_vec, n, tau_curr)[1],
                           method='L-BFGS-B',
                           options={'maxiter': 120, 'disp': False})
            x = res.x
            # update a
            a = _softmax(x)
            # evaluate true score
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start:.2f}s] n={n} tau={tau_curr:.2e} improve {cur_score:.6f}")
        # optional subgradient refinement
        if time.time() < deadline:
            a_ref, score_ref = _run_subgradient_descent(a, n, deadline - 5.0, start,
                                                        beta=0.9, step0=0.2, max_iter=200000)
            cur_score = 2.0 * n * _conv_self(a_ref).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_ref.copy()
                print(f"[{time.time() - start:.2f}s] subgrad refine n={n} improve {cur_score:.6f}")

    # final normalization
    best_seq = np.array(best_seq)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0/len(best_seq))
    print(f"[{time.time() - start:.2f}s] final best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]

    log‑sum‑exp surrogate

        Oτ(a) = 2·n·τ·log Σ exp(conv(a)/τ)

    using L‑BFGS‑B on the *log‑its* of the probabilities (softmax
    parametrisation).  The surrogate is convex in the logits and its
    gradient can be written in closed form using FFT‑based convolutions.
    The temperature τ is geometrically decreased (≈6 levels) so that the
    optimisation gradually approaches the true max‑convolution objective.

3.  **Fine‑grained projected sub‑gradient descent** – a short Euclidean
    sub‑gradient phase (with momentum) is applied to the true objective
    to polish the result.

All work respects the global 1000 s budget.  The final sequence is
renormalised to sum to 1 (the objective is scale‑invariant) and returned
as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max surrogate objective (value + gradient) on logits
# ----------------------------------------------------------------------


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    return e / s


def _smooth_obj_grad(x: np.ndarray, n: int, tau: float):
    """
    Returns (value, gradient) of the smooth surrogate
        Oτ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    where a = softmax(x).  Gradient is taken w.r.t. the logits x.
    """
    a = _softmax(x)                       # a lies on the simplex
    conv = _conv_self(a)                  # length 2·n‑1

    # soft‑max over conv/τ  (stable)
    max_c = np.max(conv)
    exp_arg = (conv - max_c) / tau
    e = np.exp(exp_arg)
    sum_e = e.sum()
    p = e / sum_e                         # shape (2·n‑1,)

    # gradient w.r.t a  (see paper derivation)
    a_rev = a[::-1]
    conv_rev = _conv_fft(a_rev, p)        # length 3·n‑2
    inner = conv_rev[n - 1 : n - 1 + n]    # slice of length n
    grad_a = 4.0 * n * inner               # ∂O/∂a

    # gradient w.r.t logits x  (soft‑max Jacobian)
    c = np.dot(grad_a, a)                 # scalar = <grad_a , a>
    grad_x = a * (grad_a - c)             # ∂O/∂x_i = a_i (grad_a_i - c)

    # surrogate value
    obj = 2.0 * n * tau * (np.log(sum_e) + max_c / tau)

    return obj, grad_x


def _smooth_obj(x: np.ndarray, n: int, tau: float) -> float:
    """Helper for scipy – returns only the function value."""
    val, _ = _smooth_obj_grad(x, n, tau)
    return val


def _smooth_grad(x: np.ndarray, n: int, tau: float) -> np.ndarray:
    """Helper for scipy – returns only the gradient."""
    _, grad = _smooth_obj_grad(x, n, tau)
    return grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))   (a on the simplex).
    Returns the best vector seen and the corresponding true score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by normalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        out = seq.copy()
    else:
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, new_n)
        out = np.interp(new_x, old_x, seq)
    # avoid exact zeros (log‑softmax would break)
    out = np.maximum(out, 1e-12)
    out /= out.sum()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine follows a multi‑scale L‑BFGS‑B optimisation of a smooth surrogate
    and finishes with a short projected sub‑gradient phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # length schedule – start small to obtain a good shape quickly
    size_schedule = [512, 1024, 2048, 4096, 8192, 16384]

    best_seq = None
    best_score = float("inf")

    for n in size_schedule:
        if time.time() >= deadline:
            break

        # initialise a (on the simplex) for this size
        if best_seq is None:
            a = np.full(n, 1.0 / n)
        else:
            a = _upsample(best_seq, n)

        # convert to logits (avoid log(0))
        x = np.log(a)

        # initial temperature – a bit larger than current max conv
        conv0 = _conv_self(a)
        tau = max(conv0.max() * 10.0, 1e-4)

        # geometric schedule for τ (≈6 levels)
        tau_vals = np.geomspace(tau, 1e-8, num=6)

        for tau_cur in tau_vals:
            if time.time() >= deadline:
                break

            # L‑BFGS‑B on the smooth surrogate
            res = minimize(fun=_smooth_obj,
                           x0=x,
                           args=(n, tau_cur),
                           jac=_smooth_grad,
                           method='L-BFGS-B',
                           options={'maxiter': 120, 'disp': False})

            x = res.x
            a = _softmax(x)                     # back to the simplex

            # evaluate the true (non‑smooth) objective
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] n={n} τ={tau_cur:.2e} "
                      f"improve score={cur_score:.6f}")

        # short projected sub‑gradient polishing (uses only the remaining budget)
        if time.time() < deadline:
            sub_deadline = deadline - 2.0  # keep a couple of seconds for final steps
            a_ref, score_ref = _run_subgradient_descent(
                a0=best_seq,
                n=n,
                deadline=sub_deadline,
                start_time=start_time,
                beta=0.90,
                step0=0.20,
                max_iter=5_000_000,
            )
            cur_score = 2.0 * n * _conv_self(a_ref).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] subgrad final (n={n}) "
                      f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
