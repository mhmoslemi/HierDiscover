```
def _subgrad(a, n, deadline, start_time, best_score):
    a = a.copy()
    best_a = a.copy()
    best = best_score
    it = 0
    momentum = np.zeros_like(a)
    beta = 0.9
    while time.time() < deadline:
        it+=1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0*n*cur_max
        if cur_score < best:
            best = cur_score
            best_a = a.copy()
        # compute gradient at max index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]
        # update
        momentum = beta * momentum + (1 - beta) * grad
        step = 0.5 / np.sqrt(it)
        a = a - step * momentum
        # projection onto simplex
        a = _project_simplex(a)
    return best_a, best
```

assistantfinal**Solution Overview**

The objective is  

\[
\min_{a\ge 0,\;\sum a_i=1}\;2\,n\max_{k}(a*a)_k ,
\qquad (a*a)_k = \sum_i a_i\,a_{k-i},
\]

i.e. minimise the $L_\infty$‑norm of the auto‑convolution while staying on the probability simplex.
The previous optimiser relied only on smooth‑max surrogates, Adam and a tiny annealing loop.
Below we add a **linear‑programming (LP) step** that is exactly the procedure described in the
state‑of‑the‑art paper:

* for the current sequence `f` compute `M = max(conv(f,f))`;
* solve  

  \[
  \max_{b\ge0}\; \mathbf 1^{\!\top}b
  \quad\text{s.t.}\quad (f*b)_k\le M\;\; \forall k,
  \]

  which is a linear program because `f` is fixed;
* normalise the LP‑solution `b` to a probability vector `g = b/∑b` and mix  

  \[
  f\leftarrow(1-t)f+t\,g ,
  \]

  with a moderate mixing weight `t`.

The LP points the optimisation into a direction that **strictly reduces the
max‑convolution** – it is a much stronger descent direction than a plain
sub‑gradient.  After a few LP‑updates we polish the result with the proven
sub‑gradient descent and finish with a light simulated‑annealing pass.

The whole search runs under the 1000 s budget, performs several random
re‑starts and keeps the best sequence found.

```python
"""
LP‑guided optimiser for `evaluate_sequence`.

* Randomly initialise a probability vector.
* Repeatedly
    – compute M = max(self‑convolution)
    – solve a linear programme (maximise sum(b)  s.t.  conv(f,b) ≤ M)
    – mix a little towards the normalised LP solution.
* After the LP phase polish with projected sub‑gradient descent.
* End with a tiny mass‑transfer annealing.
The method follows the “active‑set LP” idea from the reference paper but
adds a fast FFT‑based convolution and a simple simplex projector.
"""
import time
import math
import numpy as np
from scipy import optimize, sparse as sp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score(a: np.ndarray, n: int) -> float:
    """Exact objective value 2·n·max(conv(a,a))."""
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Linear‑programming direction
# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max 1ᵀb   s.t.  (a * b) ≤ M ,  b ≥ 0.
    Returns the optimal b (not normalised) or None on failure.
    """
    n = a.shape[0]
    L = 2 * n - 1

    # avoid zero columns – otherwise the LP would be unbounded
    a_pos = np.maximum(a, 1e-12)

    # Build the Toeplitz matrix K where K @ b = conv(a, b).
    # Each column j contains a_pos on the diagonal shifted by +j.
    diagonals = [a_pos] * n                # n identical diagonals
    offsets = np.arange(n)                 # 0 … n‑1
    K = sp.diags(diagonals, offsets, shape=(L, n), format='csc')

    # Linear programme: minimise -1ᵀb  (i.e. maximise 1ᵀb)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    ub = np.full(L, M, dtype=np.float64)

    res = optimize.linprog(
        c_obj,
        A_ub=K,
        b_ub=ub,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'disp': False},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Momentum sub‑gradient method on  f(a)=2·n·max(conv(a,a)).
    The iterate is projected onto the simplex after each step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}"
            )

        # gradient at a maximal index (any maximal index works)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = 0.25 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap mass‑transfer annealing (final polish)
# ----------------------------------------------------------------------
def _annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny annealing that moves a small amount of mass between two coordinates.
    Temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)                 # source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30       # move ≤30 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score
```
