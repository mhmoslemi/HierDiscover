def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.05

    best_seq = None
    best_score = float('inf')
    
    # Stage 0: multiple restarts
    n0 = 1024  # maybe smaller to allow many restarts
    
    # allocate 15% time

    # possible initializers: window, beta, random Dirichlet
    def init_seq():
        r = rng.random()
        if r < 0.4:
            return _window_initialiser(n0, rng)
        elif r < 0.7:
            # Beta shape
            alpha = rng.uniform(0.5, 5.0)
            beta = rng.uniform(0.5, 5.0)
            x = np.linspace(0, 1, n0)
            w = (x**(alpha-1) * (1-x)**(beta-1))
            w = np.nan_to_num(w)  # handle 0^neg
            if w.sum() == 0:
                w = np.ones_like(w)
            w = w / w.sum()
            return w
        else:
            # Dirichlet random
            a = rng.random(n0)
            a = np.maximum(a, 0)
            a = a / a.sum()
            return a
    
    while time.time() < stage0_deadline:
        a0 = init_seq()
        # Apply Adam softmax with moderate tau and time budget for this restart
        # Using early stopping: allocate small slice of stage0 time per restart

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage0 improve score {score:.6f}")

    if time.time() < deadline:

        a1 = _upsample(best_seq, n1)
        conv1 = _conv_self(a1)

"""
Propose a low‑score non‑negative step function.

The pipeline is a more aggressive version of the reference solver:

  1.  Multiple short Adam runs on a modest size (n≈1024) with a
      heterogeneous set of initialisers (window, beta‑shape, Dirichlet).
  2.  Upsample to 4096 and refine with a longer Adam run.
  3.  Upsample to the final size (8192) and perform a *very* low‑temperature
      soft‑max Adam optimisation (τ→0) to approach the true max‑convolution.
  4.  Projected Nesterov sub‑gradient descent on the exact objective.
  5.  An extended simulated‑annealing phase (mass‑transfer moves) that
      runs until the global time budget is exhausted.

All stages respect the budget (default 1000 s) and keep the vector on the
probability simplex (∑aᵢ = 1, aᵢ ≥ 0), which makes the returned value
directly comparable to the score used by the harness.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa = 1)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-15,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.20,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    with Adam + Euclidean projection onto the simplex.
    Returns the best vector seen and its *true* (unsmoothed) score
    2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true objective for logging
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # numerical stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # slice that corresponds to ∂/∂a_i
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.98,
                             step0: float = 0.30,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Nesterov‑style projected sub‑gradient descent on
        F(a) = 2·n·max(conv(a))   (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-8,
                   steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    Acceptance uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source chosen proportionally to its mass (more mass = higher chance)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move up to 40 % of src mass
        delta = a[src] * rng.random() * 0.40
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta_obj = new_score - score

        if delta_obj < 0 or math.exp(-delta_obj / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisation helpers
# ----------------------------------------------------------------------


def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    With probability 0.4 we produce a Chebyshev window,
    with probability 0.4 a Kaiser window,
    otherwise a plain random Dirichlet draw.
    """
    # Try to import scipy.signal – if unavailable fall back to random / Hann.
    try:
        from scipy.signal import chebwin, kaiser, hann
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        att = rng.uniform(30, 120)               # Chebyshev attenuation (dB)
        w = chebwin(n, at=att, sym=True).astype(float)
    elif have_signal and r < 0.8:
        beta = rng.uniform(0, 14)                # Kaiser beta
        w = kaiser(n, beta).astype(float)
    elif have_signal:
        # Hann window (always available via scipy if present)
        w = hann(n).astype(float)
    else:
        # Pure random vector (Dirichlet) as fallback
        w = rng.random(n)

    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _beta_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Symmetric beta‑type shape:  w[i] ∝ (i/(n‑1))**(α‑1)·(1‑i/(n‑1))**(β‑1)
    with α,β drawn uniformly from a modest range.
    """
    alpha = rng.uniform(0.5, 5.0)
    beta = rng.uniform(0.5, 5.0)
    x = np.linspace(0.0, 1.0, n)
    w = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    w[np.isnan(w)] = 0.0
    if w.sum() == 0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The method combines:
      – heterogeneous short‑run Adam restarts on a modest size,
      – progressive up‑sampling,
      – a very low‑temperature Adam refinement,
      – Nesterov sub‑gradient descent,
      – a long simulated‑annealing polishing phase.
    The returned list is a plain Python list of floats (≈1‑simplex).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Use a previously found construction if present
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                s = tmp.sum()
                if s > 0:
                    starter = tmp / s
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    stage0_alloc = max(30.0, 0.15 * (deadline - start_time))
    stage0_deadline = start_time + stage0_alloc

    # allow up to 30 restarts, each gets a few seconds
    while time.time() < stage0_deadline:
        if starter is not None and best_seq is None:
            a0 = _upsample(starter, n0)
        else:
            # heterogeneous initialiser
            r = rng.random()
            if r < 0.33:
                a0 = _window_initialiser(n0, rng)
            elif r < 0.66:
                a0 = _beta_initialiser(n0, rng)
            else:
                a0 = rng.random(n0)
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()

        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 8.0, 1e-8)

        # give each restart about 2‑3 s (or less if time is low)
        sub_deadline = min(time.time() + 3.0, stage0_deadline)
        a_opt, score = _run_adam_softmax(
            a0, n0, sub_deadline, start_time,
            tau0=tau0,
            tau_end=1e-15,
            tau_decay=0.9998,
            step_size=0.25,
            max_iter=200_000,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"score={score:.6f}")

    # fallback – should never happen
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – upsample to 4096 and continue Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)
    conv1 = _conv_self(a1)
    tau1 = max(conv1.max() * 8.0, 1e-10)

    rem = deadline - time.time()
    stage1_alloc = max(20.0, 0.20 * rem)
    stage1_deadline = time.time() + stage1_alloc

    a1_opt, score1 = _run_adam_softmax(
        a1, n1, stage1_deadline, start_time,
        tau0=tau1,
        tau_end=1e-15,
        tau_decay=0.9996,
        step_size=0.18,
        max_iter=500_000,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() >= deadline:

    if total > 0:

    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
