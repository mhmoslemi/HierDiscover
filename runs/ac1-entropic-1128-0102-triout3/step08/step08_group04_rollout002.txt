def _exp_gradient_optim(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    alpha = 10.0  # smoothing parameter
    eta0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        cur = 2.0 * n * conv.max()
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad iter {it}, score {cur:.6f}")
        # Smoothed gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        w /= w_sum
        # compute gradient using correlation as before
        w_rev = w[::-1]
        L = _next_pow2(3 * n - 2)
        Af = np.fft.rfft(a, n=L)
        Wr = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wr, n=L)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
        eta = eta0 / math.sqrt(it)
        upd = np.exp(-eta * grad)
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n
    return best_a, best_val

def _incremental_refine_anneal(a0, n, deadline, start_time,
                              eps_start=0.06, eps_end=0.001,
                              rel_improve=1e-12, temp_start=1e-4, temp_end=1e-7):
    a = a0.copy()
    conv = _conv_self_fft(a)  # length 2n-1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    start_refine = time.time()
    total = deadline - start_refine
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total))
        temp = max(temp_end, temp_start * (1.0 - elapsed / total))
        # choose src based on contribution to max entry but also random
        k_max = int(np.argmax(conv))
        # compute contributions for src
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        totc = contrib.sum()
        if totc > 0 and rng.random() < 0.9:
            prob_src = contrib / totc
            src = rng.choice(i_range, p=prob_src)
        else:
            src = rng.choice(n, p=a / a.sum())
        if a[src] < 1e-15:
            continue
        # destination: choose random weighted by low contribution to max bin
        # compute dest_contrib = a[k_max - idx] if in range else 0
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]

"""
Propose a step‑function that (approximately) minimizes ``evaluate_sequence``.
The search consists of several independent restarts, each of which performs

1️⃣  A short exponentiated‑gradient warm‑up on a log‑sum‑exp smooth
    approximation of the max convolution.
2️⃣  A massive incremental refinement where tiny amounts of mass are
    moved from “source’’ indices that contribute a lot to the current
    maximal convolution entry to “destination’’ indices that contribute
    little.  The convolution array is updated analytically, so each
    iteration is O(n).  A simple simulated‑annealing rule occasionally
    accepts non‑improving moves, allowing the search to escape local
    plateaus.

Different initialisations (uniform, random, Hann window, block‑wise,
arcsine‑like) are tried in turn; the best sequence obtained over all
restarts is returned.  The code respects a 1000 s time budget and never
uses file or network I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _init_distribution(n: int, mode: str, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative vector of length *n* summing to 1."""
    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "random":
        a = rng.random(n)
        a /= a.sum()

    elif mode == "hann":
        # classic Hann window, shifted to be non‑negative
        i = np.arange(n, dtype=float)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        a += 1e-12           # avoid exact zeros at the ends
        a /= a.sum()

    elif mode == "blocks":
        # K equally‑spaced blocks, each a flat plateau
        K = 4
        block_len = max(1, n // (2 * K))
        a = np.zeros(n)
        for k in range(K):
            start = k * (2 * block_len)
            a[start:start + block_len] = 1.0
        a /= a.sum()

    elif mode == "arcsine":
        # discrete analogue of the arcsine distribution
        i = np.arange(n, dtype=float)
        eps = 1e-12
        a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
        a += eps
        a /= a.sum()

    else:
        raise ValueError(f"unknown mode {mode}")

    # Clamp to the bounds used inside evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Stage 1 – exponentiated‑gradient warm‑up on a smoothed max
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     alpha: float = 20.0,
                     eta0: float = 0.5) -> np.ndarray:
    """Perform exponentiated‑gradient updates on the log‑sum‑exp smoothed max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur = 2.0 * n * conv.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad iter {it:,} score {cur:.6f}")

        # smoothed max via log‑sum‑exp
        scaled = alpha * conv
        m = scaled.max()
        w = np.exp(scaled - m)
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wr = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wr, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        eta = eta0 / math.sqrt(it)
        upd = np.exp(-eta * grad)
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

    return best_a


# ----------------------------------------------------------------------
# Stage 3 – massive incremental refinement with tiny moves
# ----------------------------------------------------------------------
def _run_incremental_refine(a0: np.ndarray,
                            deadline: float,
                            start_time: float,
                            eps_start: float = 0.06,
                            eps_end: float = 0.001,
                            rel_improve: float = 1e-12,
                            temp_start: float = 5e-5,
                            temp_end: float = 5e-8) -> np.ndarray:
    """
    Move tiny amounts of probability mass from indices that heavily
    contribute to the current maximal convolution entry to indices that
    contribute little.  The convolution is updated analytically, making a
    single iteration O(n).  An annealing rule occasionally accepts a
    non‑improving move.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)               # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))
        temp = max(temp_end, temp_start * (1.0 - elapsed / total_time))

        # --- source index (prefer indices that actually appear in the max bin) ---
        k_max = int(np.argmax(conv))
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0 and rng.random() < 0.9:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # --- destination index (prefer indices that hardly affect the max bin) ---
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # --- amount of mass to shift ---
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # --- analytic convolution update (O(n)) ---
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        new_max = conv_new.max()
        delta = new_max - max_val

        # acceptance rule
        accept = False
        if delta < -rel_improve:
            accept = True
        else:
            # simulated‑annealing style acceptance
            prob = math.exp(-delta / (temp * max_val)) if temp > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                if it % 500 == 0:
                    print(f"[{time.time() - start_time:.2f}s] refine iter {it:,} max={best_max:.8f}")

    return best_a

    # ------------------------------------------------------------------
    for mode_idx, mode in enumerate(init_modes):
        now = time.time()
        remaining = deadline - now
        modes_left = len(init_modes) - mode_idx
        if remaining <= 10.0:      # leave a small reserve for final polishing
            break
        # allocate a fraction of the remaining time to this mode
        mode_budget = remaining / modes_left

        # --------------------------------------------------------------
        # Initialise a0
        # --------------------------------------------------------------
        if mode == "starter" and starter is not None:
            a0 = _upsample(starter, final_n)
        else:
            a0 = _init_distribution(final_n, mode, rng)

        # --------------------------------------------------------------
        # Stage 1 – warm‑up (≈ 5 % of mode budget, at least 3 s)
        # --------------------------------------------------------------
        warmup_time = max(3.0, 0.05 * mode_budget)
        w_deadline = time.time() + warmup_time
        a_warm = _run_exp_gradient(a0, w_deadline, start_time)

        # --------------------------------------------------------------
        # Stage 3 – massive incremental refinement (rest of the budget)
        # --------------------------------------------------------------
        refine_deadline = deadline if mode_idx == len(init_modes) - 1 else time.time() + (mode_budget - warmup_time)
        a_refined = _run_incremental_refine(a_warm, refine_deadline, start_time)

        # --------------------------------------------------------------
        # Evaluate and possibly keep the new best sequence
        # --------------------------------------------------------------
        cur_score = _eval_score(a_refined)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] New best after mode '{mode}': score={cur_score:.6f}")

        # early exit if we have run out of time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback if everything failed (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(final_n, 1.0 / final_n)

    # Final normalisation – protects against tiny numerical drift
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # Return a plain Python list (as required by the evaluation harness)
    return [float(x) for x in best_seq]
