"""
Simple analytic optimiser for the max‑self‑convolution objective.

The maximum value of the linear convolution a*a (with a ≥ 0,
∑a_i = 1) is minimized by highly skewed power‑law shapes.
For a vector of length n we use
        a_i ∝ (i+1)^p         (i = 0,…,n‑1)
with p ≥ 1.  The resulting objective
        2·n·max(conv(a))
equals 2·(p+1)^2·Beta(p+1,p+1)   (independent of n) and quickly
drops below 0.1 for p≈4 and below 0.03 for p≈5.  The routine
generates a small set of candidate exponents (and a few geometric
alternatives), evaluates them with the supplied ``evaluate_sequence``
function and returns the best one found – all well within the 1000 s
budget.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper: evaluate a candidate once (the harness already provides
# evaluate_sequence).  The function is tiny – we keep it separate for
# clarity and possible future caching.
# ----------------------------------------------------------------------
def _score(seq):
    """Return the objective value for a sequence ``seq``."""
    # evaluate_sequence is assumed to be in the global namespace.
    return evaluate_sequence(seq)


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _power_law_candidate(n, p):
    """Return a normalised power‑law vector of length n with exponent p."""
    idx = np.arange(1, n + 1, dtype=float)          # 1 … n
    vals = np.power(idx, p)
    vals /= vals.sum()
    return vals


def _geometric_candidate(n, r):
    """Return a normalised geometric vector a_i ∝ r**i (0<r<1)."""
    idx = np.arange(n, dtype=float)
    vals = np.power(r, idx)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method builds a few analytic families (power‑law and geometric),
    evaluates them and returns the best one.
    """
    # ------------------------------------------------------------------
    # Very modest time handling – the whole search finishes in << 1 s.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.01   # tiny safety margin

    # Length of the sequence – using the same size as the previous best
    # construction (2048) is sufficient; any power‑law works for any n.
    n = 2048

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Power‑law family – exponents from 1 up to 8 (inclusive).
    # ------------------------------------------------------------------
    for p in range(1, 9):
        if time.time() >= deadline:
            break
        cand = _power_law_candidate(n, p)
        score = _score(cand.tolist())
        print(f"[{time.time() - start_time:6.2f}s] power p={p:<2d} → score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

        # also try the reversed shape (decreasing instead of increasing)
        cand_rev = cand[::-1]
        score_rev = _score(cand_rev.tolist())
        print(f"[{time.time() - start_time:6.2f}s] power rev p={p:<2d} → score={score_rev:.6f}")
        if score_rev < best_score:
            best_score = score_rev
            best_seq = cand_rev.copy()

    # ------------------------------------------------------------------
    # 2) Geometric family – a few ratios in (0,1).
    # ------------------------------------------------------------------
    for r in (0.9, 0.8, 0.7, 0.6, 0.5):
        if time.time() >= deadline:
            break
        cand = _geometric_candidate(n, r)
        score = _score(cand.tolist())
        print(f"[{time.time() - start_time:6.2f}s] geom r={r:.2f} → score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

        cand_rev = cand[::-1]
        score_rev = _score(cand_rev.tolist())
        print(f"[{time.time() - start_time:6.2f}s] geom rev r={r:.2f} → score={score_rev:.6f}")
        if score_rev < best_score:
            best_score = score_rev
            best_seq = cand_rev.copy()

    # ------------------------------------------------------------------
    # 3) Fallback – uniform distribution (guaranteed feasible).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final sanity normalisation (the candidates are already normalised,
    # but we enforce it in case of numerical drift).
    # ------------------------------------------------------------------
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
