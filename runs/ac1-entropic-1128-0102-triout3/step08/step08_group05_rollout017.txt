"""
Propose a step‑function sequence with a very low evaluation score.

The algorithm consists of three stages:

1. **Quick parametric sweep** – generate a family of symmetric Beta‑type
   densities  a_i ∝ x^{α‑1}(1‑x)^{α‑1}  (x = (i+½)/n) for several values of α.
   After normalising the vector to have ℓ₁‑norm √(2n) the true objective
   (which equals max (conv) for this normalisation) is evaluated with the
   provided ``evaluate_sequence``.  The best candidate is kept.

2. **Linear‑programming refinement** – for the current vector f we solve the
   LP

        maximise    Σ b_j
        subject to  (f * b)_k ≤ M   (k = 0,…,2n‑2)
                    b_j ≥ 0

   where M = max(conv(f)).  The convolution (f * b) is linear in b, so the
   LP can be built directly.  The optimal b is rescaled to the same ℓ₁‑norm
   √(2n) and mixed with f (f ← (1‑t) f + t b) with a decreasing step size t.
   A few iterations quickly lower the maximal self‑convolution.

3. **Projected sub‑gradient descent** – a final Euclidean projected sub‑gradient
   descent (with momentum) is run for the remaining time to polish the result.

All vectors stay on the simplex (non‑negative, fixed ℓ₁‑norm), which makes the
objective scale‑invariant.  The routine respects the global time budget
(`budget_s` seconds) and returns a plain Python list of floats ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i≥0, Σx_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT for speed)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """A (averaged) sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]
    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent (real objective)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on  F(a)=max(conv(a))  (a stays on the simplex)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

            best_score, best_seq = final_score, refined_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] final subgrad improved to {final_score:.6f}")

    # --------------------------------------------------------------
    # Return a plain list (evaluate_sequence is scale‑invariant)
    # --------------------------------------------------------------
    out = best_seq.tolist()
    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return out
