def _run_exponentiated_gradient(a0, n, deadline, start_time, max_iters=None):
    a = a0.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    L = _next_pow2(2*n - 1)
    it = 0
    beta = 0.5  # start temperature param
    eta0 = 0.5
    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        # compute convolution
        conv = _conv_self(a, L)
        cur_score = 2.0*n*conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter {it} improve {cur_score:.6f}")
        # gradually increase beta
        # e.g., beta = 0.5 + 0.01 * it
        beta = min(80.0, 0.5 + it * 0.001)  # adjust
        # compute softmax weights
        scaled = beta * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        # gradient: 2 * conv of a with reversed w
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w[::-1], n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w) - 1]  # length n + len(w) -1 = n + 2n-2 ??
        # Wait w length = 2n-1, so len(w)-1 = 2n-2. So corr length = n + 2n-2 = 3n-2? Not needed.
        # Actually for gradient we need first n elements of length n+len(w)-1:
        # In earlier code, they used corr[len(w)-1 - np.arange(n)].
        grad = 2.0 * corr[len(w) - 1 - np.arange(n)]  # shape (n,)
        # update
        eta = eta0 / np.sqrt(it)

        # renorm to simplex
        total = a.sum()
        if total <= 0:
            a = np.random.rand(n)
            a /= a.sum()
            continue
        a /= total
    return best_a, best_score

"""
Convolution‑max minimiser (≈1 000 s)

The objective 2·n·max(conv(a,a))/sum(a)² is invariant under scaling,
so we optimise a probability vector a (sum = 1, a_i≥0) and return it
as a plain list.

The optimiser consists of three cheap, complementary stages that run
within the 1000 s budget:

1. **Exponentiated‑gradient (EG) on a smooth max**  
   A log‑sum‑exp (soft‑max) surrogate of the max‑convolution is used.
   The gradient of the surrogate is obtained by a single FFT‑based
   convolution.  Multiplicative updates (`a_i←a_i·exp(−η·grad_i)`) keep
   the variables on the simplex automatically.  The smoothing
   parameter β is increased slowly, turning the surrogate into the
   true max in later iterations.

2. **Greedy local mass‑shifting**  
   While the current peak index `k` of the convolution is known,
   the pair `(i, k‑i)` that contributes most to this peak is identified.
   A small fraction of probability mass is moved from this source index
   to a destination where its counterpart `a_{k‑dest}` is minimal,
   which provably reduces the peak to first order.  The step size is a
   fixed fraction (10 %) of the source mass; after each move the vector
   is renormalised.

3. **Random‑restart perturbations** (optional, time‑permitting)  
   A short EG run from a slightly perturbed copy of the best sequence
   helps to escape shallow plateaux.

All heavy work is performed with FFTs of length a power‑of‑two, so each
iteration costs `O(n log n)` (≈1 ms for n = 4096).  The algorithm respects
the time budget by checking `deadline` after every iteration.

The final vector is clipped to non‑negative values, normalised to sum = 1,
and returned as a list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the a*a convolution (length 2·len(a)‑1) via FFT.
    The result is a real array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# 1. Exponentiated‑gradient optimiser on a smooth max
# ----------------------------------------------------------------------
def _run_exponentiated_gradient(a0: np.ndarray,
                                n: int,
                                deadline: float,
                                start_time: float,
                                max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on the log‑sum‑exp
    surrogate of max(conv(a,a)).  Returns the best iterate seen.
    """
    a = a0.copy()
    best_a = a.copy()
    # the objective for a sum‑normalised vector is simply 2·n·max(conv)
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    L = _next_pow2(2 * n - 1)

    it = 0
    eta0 = 0.6                # initial learning‑rate factor
    beta = 0.5                # initial smoothing factor

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1

        # forward pass
        conv = _conv_self(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,} improve={cur_score:.6f}")

        # increase smoothing (soft‑max temperature) slowly
        beta = min(80.0, beta + 0.005)   # final β≈80 gives a tight max

        # soft‑max weights over convolution entries
        scaled = beta * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient of the smoothed objective (same as in the reference code)
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update and re‑normalisation onto the simplex
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # numerical safety – keep values non‑negative and renormalise
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total <= 0.0:                     # fallback to uniform if everything vanished
            a = np.full(n, 1.0 / n)
        else:
            a /= total

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Greedy local mass‑shifting optimiser
# ----------------------------------------------------------------------
def _run_greedy_local(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small fraction of probability mass away from the
    pair of indices that contributes most to the current maximal
    convolution coefficient.  The move is accepted unconditionally (it
    is a descent direction to first order).  The best sequence encountered
    is returned.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} improve={cur_score:.6f}")

        # index of the current peak
        k = int(conv.argmax())

        # indices that actually appear in this peak
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)

        # contributions a_i * a_{k-i}
        contrib = a[idx] * a[k - idx]
        if contrib.size == 0:
            break

        # source index = one with the largest contribution to the peak
        i_src = idx[np.argmax(contrib)]

        # destination: index where the counterpart a_{k-d} is minimal
        complement = np.zeros_like(a)
        # fill only the range where k-d is a valid index
        complement[lo:hi + 1] = a[k - np.arange(lo, hi + 1)]

        # exclude the source from the candidate set
        mask = np.arange(n) != i_src
        candidates = np.arange(n)[mask]
        comp_vals = complement[mask]
        i_dst = candidates[np.argmin(comp_vals)]

        # move a fraction (10 %) of the source mass to the destination
        eps = min(a[i_src] * 0.10, a[i_src] - 1e-14)
        if eps <= 0.0:
            # source exhausted – break to avoid endless loop
            break
        a[i_src] -= eps
        a[i_dst] += eps

        # renormalise (the sum stays 1 up to round‑off)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Random‑restart driver (lightweight)
# ----------------------------------------------------------------------
def _run_restart(a_best: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform a short EG run from a slightly perturbed copy of a_best.
    Returns a possibly improved sequence.
    """
    # small Gaussian perturbation
    pert = a_best + 0.015 * np.random.randn(n)
    pert = np.clip(pert, 0.0, None)
    if pert.sum() <= 0.0:
        pert = np.full(n, 1.0 / n)
    else:
        pert /= pert.sum()

    # allocate a modest time slice (≈6 s) for the restart
    time_left = deadline - time.time()
    if time_left < 2.0:
        return a_best.copy(), 2.0 * n * _conv_self(a_best, _next_pow2(2 * n - 1)).max()
    sub_deadline = time.time() + min(6.0, time_left / 2.0)

    a_new, score_new = _run_exponentiated_gradient(
        pert, n, sub_deadline, start_time, max_iters=5000
    )
    return a_new, score_new
