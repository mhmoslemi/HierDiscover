"""
Propose a non‑negative step function that (almost) minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – use the best known construction (``height_sequence_1``)
   if it exists, otherwise fall back to a uniform vector.  The vector is
   normalised to sum‑to‑one.

2. **Hill‑climbing on the true max‑convolution objective** – the objective
   ``f(a)=max_k (a*a)[k]`` is scale‑invariant.  For a current vector ``a``,
   its full linear convolution ``c = a*a`` is kept explicitly.  A single move
   transfers a small amount ``ε`` from a *source* index (the element that most
   contributes to the current maximal convolution entry) to a *destination*
   index (the currently smallest coefficient).  The effect on the full
   convolution can be updated in O(n) time via a slice update, and the move
   is accepted if it strictly reduces the current maximum.  A simple
   simulated‑annealing rule occasionally accepts slightly worsening moves to
   escape local plateaus.

3. **Random restarts** – if no improvement is observed for a while,
   a tiny random perturbation (re‑normalised) is applied and the search
   continues.

4. **Final normalisation** – the best vector found is clamped to
   non‑negative values, re‑normalised, and returned as a plain Python list.

The whole procedure respects the 1000 s time budget and prints progress
messages.  No external I/O or network access is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based for the initial full convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Small helper to up‑sample a vector while preserving its sum
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Core hill‑climbing optimiser (max‑convolution reduction)
# ----------------------------------------------------------------------


def _hill_climb(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_no_improve: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Perform a simple hill‑climbing / simulated‑annealing search that
    repeatedly moves mass from a coefficient contributing heavily to the
    current max convolution peak to a small coefficient.
    Returns the best vector seen and its evaluation score.
    """
    n = a0.shape[0]
    a = a0.copy()
    conv = _conv_self(a)                     # length 2*n-1
    max_c = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * max_c
    no_improve = 0
    rng = np.random.default_rng(12345)

    # pre‑allocate a view of the convolution array for slice updates
    conv_len = 2 * n - 1

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1. identify the worst (max) convolution entry and the most
        #    responsible index pair.
        # --------------------------------------------------------------
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[int(contrib.argmax())]          # source index to shrink
        # --------------------------------------------------------------
        # 2. choose a destination (currently smallest coefficient)
        # --------------------------------------------------------------
        dst_candidates = np.where(np.arange(n) != src)[0]
        dst = int(dst_candidates[a[dst_candidates].argmin()])

        # --------------------------------------------------------------
        # 3. decide how much mass to move.
        # --------------------------------------------------------------
        src_val = a[src]
        if src_val <= 0.0:
            # nothing to move – stagnation, break out
            break
        eps = min(src_val * 0.25, 0.01)   # modest fraction, capped
        if eps < 1e-12:
            break

        # --------------------------------------------------------------
        # 4. compute the convolution update using slice operations
        # --------------------------------------------------------------
        a_old = a.copy()                 # needed for the shift term
        # slice update:  conv += 2*eps*(shift_dst - shift_src)
        conv[src : src + n] -= 2.0 * eps * a_old
        conv[dst : dst + n] += 2.0 * eps * a_old

        # epsilon‑squared contribution (very small, but kept for exactness)
        idx2src = 2 * src
        idx2dst = 2 * dst
        idxsum = src + dst
        if idx2src < conv_len:
            conv[idx2src] += eps * eps
        if idx2dst < conv_len:
            conv[idx2dst] += eps * eps
        if src != dst and idxsum < conv_len:
            conv[idxsum] -= 2.0 * eps * eps

        # --------------------------------------------------------------
        # 5. apply the move to the vector itself
        # --------------------------------------------------------------
        a[src] -= eps
        a[dst] += eps

        # --------------------------------------------------------------
        # 6. evaluate the new maximal convolution entry
        # --------------------------------------------------------------
        new_max_c = conv.max()
        new_score = 2.0 * n * new_max_c

        # --------------------------------------------------------------
        # 7. acceptance test (strict improvement preferred; otherwise SA)
        # --------------------------------------------------------------
        improved = new_max_c < max_c - 1e-14
        if improved:
            max_c = new_max_c
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] hill‑climb iter {int(time.time())} "
                    f"improved score={new_score:.6f}"
                )
            no_improve = 0
        else:
            # Simulated‑annealing fallback – occasionally accept a worsening move
            # to escape shallow plateaus.
            temp = max(0.01, (deadline - time.time()) / (budget_s := 1000))
            prob = np.exp(-(new_max_c - max_c) / (temp * max_c))
            if rng.random() < prob:
                max_c = new_max_c
                no_improve = 0
            else:
                # revert the move (undo everything)
                a[src] += eps
                a[dst] -= eps
                conv[src : src + n] += 2.0 * eps * a_old
                conv[dst : dst + n] -= 2.0 * eps * a_old
                if idx2src < conv_len:
                    conv[idx2src] -= eps * eps
                if idx2dst < conv_len:
                    conv[idx2dst] -= eps * eps
                if src != dst and idxsum < conv_len:
                    conv[idxsum] += 2.0 * eps * eps
                no_improve += 1

        # --------------------------------------------------------------
        # 8. random restart if we have been stuck for a while
        # --------------------------------------------------------------
        if no_improve >= max_no_improve:
            # add a very small random jitter and re‑normalise
            jitter = rng.normal(scale=1e-4, size=n)
            a = best_a + jitter
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            conv = _conv_self(a)
            max_c = conv.max()
            no_improve = 0
            print(
                f"[{time.time() - start_time:6.2f}s] random restart triggered"
            )

    final_score = 2.0 * n * max_c
    return best_a, final_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine initialises from a known good construction (if available) and
    then performs a fast hill‑climbing / simulated‑annealing optimisation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    n = 4800  # length used in the reference solution

    # ------------------------------------------------------------------
    # Initialise from the global best if present, else fall back to uniform.
    # ------------------------------------------------------------------
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            a0 = np.maximum(seq, 0.0)
            a0_sum = a0.sum()
            if a0_sum == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0_sum
        else:
            raise ValueError
    except Exception:
        a0 = np.full(n, 1.0 / n)  # uniform start

    # ------------------------------------------------------------------
    # Run the hill‑climbing optimiser for the remaining budget.
    # ------------------------------------------------------------------
    best_seq, best_score = _hill_climb(a0, deadline, start_time)

    # ------------------------------------------------------------------
    # Final normalisation (guard against tiny numerical drift).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}"
    )
    # return a plain Python list
    return [float(x) for x in best_seq.tolist()]
