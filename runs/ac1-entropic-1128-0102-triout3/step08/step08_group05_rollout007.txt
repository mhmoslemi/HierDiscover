"""
Multi‑stage optimizer for the max‑self‑convolution problem.

The search proceeds on a sequence of increasing lengths (powers of two).
For each length we perform:

1. A short Adam run on a smooth log‑sum‑exp surrogate of the max‑convolution.
2. Projected Euclidean sub‑gradient descent on the true objective.
3. (If SciPy is available) a few linear‑programming refinements that
   maximise the ℓ₁‑norm of a partner vector under the same max‑convolution bound.
   This step provably reduces the objective when it succeeds.
4. A second Adam run to exploit any new curvature.
5. Another sub‑gradient stage.
6. Entropy‑mirrored (exponentiated‑gradient) descent.
7. A very light simulated‑annealing mass‑transfer for the last seconds.

All iterates are kept on the probability simplex (non‑negative,
sum = 1), which makes the final evaluation invariant to scaling.
The routine respects the global time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i≥0, Σx_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for true objective)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑sum‑exp) surrogate of max‑conv
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.25,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Minimise the smooth surrogate Oτ(a)=2·n·τ·log∑exp(conv/τ) via Adam."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f} τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # shape 2·n‑1, sums to 1

        # gradient of Oτ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # required slice
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Classical projected sub‑gradient descent on F(a)=2·n·max(conv(a))."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the true objective
# ----------------------------------------------------------------------


def _run_exponentiated_gradient(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               step0: float = 0.20,
                               max_iter: int = 5 * 10 ** 6) -> tuple[np.ndarray, float]:
    """Mirror‑descent with the entropy Bregman divergence."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)               # non‑negative
        eta = step0 / math.sqrt(it)                    # diminishing step
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve (n={n}) "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional – needs SciPy)
# ----------------------------------------------------------------------
try:
    import scipy.sparse as _sp
    from scipy.optimize import linprog as _linprog
    _HAVE_SCIPY = True
except Exception:  # pragma: no cover
    _HAVE_SCIPY = False


def _lp_refine(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               max_iter: int = 5,
               mix: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.

    For the current vector a, let M = max(a*a).  Solve the LP
        maximise   Σ b_j
        subject to a * b ≤ M  (componentwise)
                   b_j ≥ 0.
    If Σ b_j > 1 we can move a towards the normalised b, which
    provably reduces the objective.  The routine performs a few such
    updates (mix controls the step size of the convex combination).
    """
    if not _HAVE_SCIPY:
        return a0, float("inf")

    import time
    a = a0.copy()
    best = a.copy()
    best_score = float("inf")

    # Pre‑compute index arrays for the sparse convolution matrix.
    i_idx = np.repeat(np.arange(n), n)          # row contribution of a_i
    j_idx = np.tile(np.arange(n), n)            # column (b index)
    rows = i_idx + j_idx                         # actual row numbers (0 … 2n‑2)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()

        # Build sparse matrix A where A_{k,j} = a_{k-j} (or 0 if out of range)
        data = a[i_idx]
        A = _sp.csr_matrix((data, (rows, j_idx)), shape=(2 * n - 1, n))

        # Linear program: minimise -Σb  ⇒  c = -1
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = _linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                           method='highs', options={'presolve': True})
        except Exception as e:  # pragma: no cover
            print(f"[{time.time() - start_time:6.2f}s] LP failed: {e}")
            break

        if res.status != 0 or res.x is None:
            # Infeasible or numerical problem.
            break

        b_opt = res.x
        B = b_opt.sum()
        if B <= 1.0 + 1e-12:
            # No improvement possible.
            break

        # Convex combination and re‑normalisation.
        g = b_opt / B
        a = (1 - mix) * a + mix * g
        a = _project_simplex(a)

        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve (n={n}) "
                  f"iter={it} Σb={B:.6f} score={cur_score:.6f}")

    return best, best_score

# Main entry point required by the harness

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    The method follows the multi‑stage scheme described above, trying a few
    different resolutions and employing Adam, sub‑gradient, LP refinement,
    exponentiated‑gradient and a final annealing pass.  It respects the
    global time budget (default 1000 s) and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Optional initialisation from a previously discovered sequence.
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Progressive up‑sampling schedule.
    # ------------------------------------------------------------------
    size_schedule = [256, 512, 1024, 2048, 4096]

    best_seq = None
    best_score = float("inf")

    for n in size_schedule:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a vector of length n (up‑sample or random)
        # --------------------------------------------------------------
        if best_seq is not None:
            a = _upsample(best_seq, n)
        elif starter is not None:
            a = _upsample(starter, n)
        else:
            a = np.random.rand(n)
            a /= a.sum()

        # --------------------------------------------------------------
        # Phase 1 – Adam on soft‑max surrogate (short burst)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        slice_time = min(remaining * 0.12, 30.0)   # a few seconds, at most 30 s
        tau0 = max(_conv_self(a).max() * 10.0, 1e-8)
        a, _ = _run_adam_softmax(
            a0=a,
            n=n,
            deadline=time.time() + slice_time,
            start_time=start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9995,
            step_size=0.30,
        )
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] size {n} Adam improve: "
                  f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Projected sub‑gradient descent (medium burst)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        sub_deadline = time.time() + min(remaining * 0.30, 90.0)
        a, _ = _run_subgradient_descent(
            a0=a,
            n=n,
            deadline=sub_deadline,
            start_time=start_time,
            beta=0.95,
            step0=0.30,
        )
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] size {n} subgrad improve: "
                  f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – Linear‑programming refinement (if SciPy is available)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0 and n <= 2048 and _HAVE_SCIPY:
            a, _ = _lp_refine(
                a0=a,
                n=n,
                deadline=deadline,
                start_time=start_time,
                max_iter=5,
                mix=0.30,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] size {n} LP improve: "
                      f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – Second Adam to exploit newly created curvature
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            slice_time = min(remaining * 0.10, 20.0)
            tau0 = max(_conv_self(a).max() * 10.0, 1e-8)
            a, _ = _run_adam_softmax(
                a0=a,
                n=n,
                deadline=time.time() + slice_time,
                start_time=start_time,
                tau0=tau0,
                tau_end=1e-13,
                tau_decay=0.9995,
                step_size=0.25,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] size {n} Adam‑2 improve: "
                      f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 5 – Second sub‑gradient pass (heavy lifting)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            sub_deadline = time.time() + min(remaining * 0.15, 60.0)
            a, _ = _run_subgradient_descent(
                a0=a,
                n=n,
                deadline=sub_deadline,
                start_time=start_time,
                beta=0.94,
                step0=0.20,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] size {n} subgrad‑2 improve: "
                      f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 6 – Exponentiated‑gradient (mirror descent)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            eg_deadline = time.time() + min(remaining * 0.10, 30.0)
            a, _ = _run_exponentiated_gradient(
                a0=a,
                n=n,
                deadline=eg_deadline,
                start_time=start_time,
                step0=0.20,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] size {n} EG improve: "
                      f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 7 – Light simulated annealing for the very last seconds
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 0.5:
            a, _ = _run_annealing(
                a0=a,
                n=n,
                deadline=deadline,
                start_time=start_time,
                cur_best_score=best_score,
                cur_best_seq=best_seq,
                start_temp=2e-2,
                end_temp=1e-7,
                steps=200_000,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] size {n} anneal final improve: "
                      f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform vector of length 2048.
        best_seq = np.full(2048, 1.0 / 2048)

    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
