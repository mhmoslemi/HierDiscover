"""
Propose a step‐function that (approximately) equalises its self‑convolution.
The algorithm consists of three stages:

1️⃣  Initialise the distribution with the discrete arcsine law (heavy mass at
    the ends – the continuous analogue gives a flat sum distribution).
2️⃣  Run a fast Adam–style projected (sub‑)gradient descent on the exact
    max‑convolution objective.  The gradient is the exact sub‑gradient of the
    ℓ∞‑norm of the convolution.
3️⃣  Greedy massive local refinement: pick a source index that contributes
    heavily to the current maximal convolution entry, a destination that
    contributes little, and move the *largest* admissible amount of mass.
    The admissible step is found analytically by a simple linear programme,
    which makes each iteration O(n) while guaranteeing a strict decrease of
    the maximum convolution value.

All three stages respect the global time budget (default 1000 s) and keep the
vector non‑negative and normalised (projection onto the probability simplex).
The resulting sequence is returned as a list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    All indices whose convolution value lies within rel_tol of the maximum
    are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # derivative w.r.t. a[i] is a[j]
        g[i] += a[j]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _arcsine_distribution(n: int) -> np.ndarray:
    """
    Discrete arcsine law on {0,…,n‑1}:
    a_i ∝ 1 / sqrt(x_i·(1−x_i)), where x_i = (i+0.5)/n.
    The vector is normalised to sum to 1.
    """
    i = np.arange(n) + 0.5
    x = i / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    # Guard against possible infinities (should not happen with 0.5 offset)
    a[np.isinf(a)] = 0.0
    a = np.nan_to_num(a)
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Stage 2 – Adam projected sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.25,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of  max(conv) subject to a∈simplex.
    Returns the best vector found and its (un‑scaled) score 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # decreasing step size (the division by sqrt(it) avoids overly large steps)
        lr = step0 / math.sqrt(it)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)

        # move opposite to the gradient (we minimise the max)
        a = a - update

        # projection onto simplex (ensures non‑negativity and unit sum)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – Greedy massive incremental refinement (analytic line‑search)
# ----------------------------------------------------------------------
def _run_large_incremental_refine(a0: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 start_time: float,
                                 safety: float = 0.95) -> np.ndarray:
    """
    Perform very many mass‑shifts.  For each move we compute the
    *largest* admissible epsilon that keeps the current maximal
    convolution entry strictly decreasing.  The update of the convolution
    is performed analytically (Δconv = 2·ε·(a_dst – a_src) over the appropriate
    slices), so each iteration is O(n).
    """
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # current maximal index (break ties randomly)
        k_max = int(np.argmax(conv))

        # -------- source selection (indices contributing to the max entry) ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            # fallback – choose proportionally to the mass itself
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue    # nothing to move

        # -------- destination selection – favour low contribution ----------
        # For the same max index compute the counterpart value a[k_max‑idx] ;
        # small values → larger probability.
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                     # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ------------------- compute analytic delta array -------------------
        delta = np.zeros_like(conv)
        # contribution of moving mass from src
        delta[src:src + n] -= 2.0 * a
        # contribution of adding mass to dst
        delta[dst:dst + n] += 2.0 * a

        delta_kmax = delta[k_max]

        # ------------------- maximal feasible epsilon ----------------------
        # We need conv[k] + eps*delta[k] ≤ conv[k_max] + eps*delta_kmax  for all k
        # → eps ≤ (conv[k_max] - conv[k]) / (delta[k] - delta_kmax) for every
        #    k with delta[k] > delta_kmax.
        mask = delta > delta_kmax + 1e-20          # strictly larger
        if np.any(mask):
            numer = conv[k_max] - conv[mask]
            denom = delta[mask] - delta_kmax
            # avoid division by zero, numer is non‑negative by definition
            eps_bounds = numer / denom
            # due to rounding eps_bounds can be negative – discard those
            eps_max = np.min(eps_bounds[eps_bounds > 0.0]) if np.any(eps_bounds > 0.0) else 0.0
        else:
            eps_max = a[src]      # no limiting other index

        # also cannot move more mass than is present at src
        eps_max = min(eps_max, a[src])

        # apply a safety factor and ensure a minimal step size
        eps = eps_max * safety
        if eps <= 0.0:
            continue

        # ------------------- accept move ----------------------------------
        conv += eps * delta
        a[src] -= eps
        a[dst] += eps
        max_val = conv.max()
        it += 1

        if max_val < best_max - 1e-15:
            best_max = max_val
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] large‑refine improve: "
                  f"iter={it:,}, max={best_max:.8f}")

    return best_a
