"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Random sparse support generation** – we repeatedly draw a random
   set of indices S ⊂ {0,…,n‑1} while keeping the number of ordered
   pairs (i,j) with i+j = k bounded by a small constant L (L=2,3 or 4).
   Small L forces the convolution to stay flat; the greedy construction
   is performed many times and the largest feasible set is kept.

2. **Local refinement** – the uniform distribution on the chosen support
   (a_i = 1/|S| for i∈S) is refined by a short projected sub‑gradient
   descent on the true objective f(a)=max_k (a*a)[k].  This is the same
   “polishing” routine used in the reference solution but run for only
   a few seconds per candidate, allowing thousands of independent tries.

3. **Final polishing** – after the time budget is almost exhausted the
   current best candidate is polished for the remaining time with a
   longer sub‑gradient run and a cheap deterministic greedy peak‑reduction
   step.

All phases respect the global deadline (default 1000 s).  The result is
normalised to sum to one before being returned.

The search explores many diverse sparse starts, which is fundamentally
different from the pure Adam‑then‑sub‑gradient pipeline of the
reference code and is well suited to lowering the peak of the
self‑convolution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _compute_score(a: np.ndarray) -> float:
    """Exact evaluation of the objective (identical to evaluate_sequence
    for admissible vectors)."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s ** 2)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.2) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=max_k (a*a)[k]."""
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional log
            if it % 500 == 0:
                print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

        # active set (indices attaining the peak, within tolerance)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient for the active peak(s)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]                 # diagonal counted twice – harmless

        # momentum / step‑size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Shift a modest fraction of the coefficient responsible for the
    current max convolution peak towards the smallest coefficient,
    accepting only strictly improving moves."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _compute_score(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]                     # most responsible

        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]      # move a modest fraction
        if eps <= 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _compute_score(a_prop)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_score:.6f}")

    return a, best_score

def _can_add(i: int, S: list[int], counts: np.ndarray, max_contrib: int) -> bool:
    """Check whether index i can be added without violating the per‑sum
    bound ``max_contrib`` (ordered pair count)."""
    if counts[i + i] + 1 > max_contrib:
        return False
    for j in S:
        if counts[i + j] + 2 > max_contrib:
            return False
    return True


def _add_index(i: int, S: list[int], counts: np.ndarray) -> None:
    """Insert index i into the support and update the pair‑count table."""
    counts[i + i] += 1
    for j in S:
        counts[i + j] += 2
    S.append(i)


def _build_random_support(n: int,
                          max_contrib: int,
                          rng: np.random.Generator,
                          attempts: int = 5) -> tuple[list[int], np.ndarray]:
    """Greedy construction of a sparse index set.
    Repeats ``attempts`` times with different random orderings and returns
    the largest feasible set found."""
    best_S: list[int] = []
    best_counts = None
    for _ in range(attempts):
        S: list[int] = []
        counts = np.zeros(2 * n - 1, dtype=int)
        order = rng.permutation(n)
        for i in order:
            if _can_add(i, S, counts, max_contrib):
                _add_index(i, S, counts)
        if len(S) > len(best_S):
            best_S = S.copy()
            best_counts = counts.copy()
    return best_S, best_counts

                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:6.2f}s] seed seq improved: score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Random sparse‑support exploration phase.
    # ------------------------------------------------------------------
    # We continuously generate candidates while we have at least ~2 s left.
    while time.time() < deadline - 2.5:
        # random bound on ordered‑pair counts (2 → 4 works well in practice)
        max_contrib = int(rng.choice([2, 3, 4]))
        S, _ = _build_random_support(n, max_contrib, rng, attempts=7)
        if len(S) < 30:          # ignore trivially tiny supports
            continue

        # start with a uniform distribution on the support
        a0 = np.zeros(n)
        a0[S] = 1.0 / len(S)

        # short polishing – only a couple of seconds per candidate
        sub_deadline = time.time() + 1.4
        a_ref, sc_ref = _subgrad_optimize(a0, sub_deadline, start_time,
                                          beta=0.94, step0=0.25)

        # exact score (avoids any tiny clipping differences)
        sc = _compute_score(a_ref)

        if sc < best_score:
            best_score = sc
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] NEW GLOBAL BEST (random) "
                  f"score={sc:.6f}  | support size={len(S)}  max_contrib≤{max_contrib}")

    # ------------------------------------------------------------------
    # Final polishing of the overall best candidate.
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print(f"[{time.time() - start_time:6.2f}s] final polishing start")
        best_seq, best_score = _subgrad_optimize(best_seq, deadline, start_time,
                                                beta=0.94, step0=0.2)

    # ------------------------------------------------------------------
    # Final deterministic greedy peak‑reduction (cheap but often useful).
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print(f"[{time.time() - start_time:6.2f}s] greedy peak‑reduction start")
        best_seq, best_score = _greedy_local_improve(best_seq, deadline,
                                                     start_time, max_iters=5000)

    # ------------------------------------------------------------------
    # Normalise once more (protect against numerical drift) and return.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = _compute_score(best_seq)
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
