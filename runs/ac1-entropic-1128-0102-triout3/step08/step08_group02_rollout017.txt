"""
Hybrid optimiser – enriched with a cheap parametric random‑search stage.
The routine works inside a single time budget (default 1000 s).

Phases
------
1. **Diverse starts** – a short random search creates many
   plausible step‑functions.  The search uses three lightweight
   parametric families:
      * power‑law edge‑heavy shape a_i ∝ ((i+0.5)*(n-i-0.5))⁻ᵅ,
      * piece‑wise constant vectors with a random Dirichlet base,
      * a smoothed Dirichlet vector (Gaussian kernel).
   The best of the generated candidates is kept.

2. **Soft‑max Adam phase** – the chosen start is refined using a
   log‑softmax parametrisation and a smooth log‑sum‑exp
   approximation of the max‑convolution.  The schedule of the
   smoothing parameter α is extended (up to 1280) and the number of
   iterations per α is increased when time permits.

3. **Long exponentiated‑gradient (mirror) descent** – the exact
   sub‑gradient of the max‑convolution is applied with an
   exponentiated update (always stays non‑negative) and a
   projection onto the probability simplex.

4. **Deterministic peak reduction** – a greedy move shifts a small
   amount of mass from the coefficient that contributes most to
   the current peak towards the smallest coefficient.

5. **Brief simulated‑annealing finish** – a classic annealing
   sweep makes a last pass of stochastic mass‑transfer moves.

All phases respect the overall deadline.  The best sequence seen
throughout the whole run is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.stats import dirichlet

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) w.r.t. a.
    Returns a vector of shape (n,).
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                          # soft‑max weights
    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Random‑search helpers (three light parametric families)

def _shape_powerlaw(n: int, alpha: float) -> np.ndarray:
    """Edge‑heavy power‑law shape a_i ∝ ((i+0.5)*(n-i-0.5))⁻ᵅ."""
    i = np.arange(n, dtype=float) + 0.5
    prod = i * (n + 0.5 - i)               # (i+0.5)*(n-i-0.5)
    v = prod ** (-alpha)
    v = np.clip(v, 0.0, None)
    v_sum = v.sum()
    if v_sum == 0:
        v = np.full(n, 1.0 / n)
    else:
        v /= v_sum
    return v


def _shape_dirichlet_blocks(n: int, rng: np.random.Generator,
                            blocks: int = 20) -> np.ndarray:
    """Random Dirichlet base of `blocks` pieces, linearly interpolated."""
    base = rng.dirichlet(np.ones(blocks))
    return _upsample(base, n)


def _smooth_gaussian(v: np.ndarray, sigma: float) -> np.ndarray:
    """1‑D Gaussian smoothing (periodic boundary handling via padding)."""
    if sigma <= 0.0:
        return v
    # kernel size ≈ 6σ, make it odd
    k = int(np.ceil(sigma * 6))
    if k % 2 == 0:
        k += 1
    x = np.arange(k) - k // 2
    g = np.exp(-0.5 * (x / sigma) ** 2)
    g /= g.sum()
    # use mode='same' with reflexion at the boundaries
    smooth = np.convolve(v, g, mode='same')
    smooth = np.clip(smooth, 0.0, None)
    sm_sum = smooth.sum()
    if sm_sum > 0:
        smooth /= sm_sum
    else:
        smooth = np.full_like(smooth, 1.0 / v.shape[0])
    return smooth


def _random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random start using a mixture of the three families.
    """
    choice = rng.integers(3)
    if choice == 0:
        # power‑law
        alpha = rng.uniform(0.0, 4.0)
        a = _shape_powerlaw(n, alpha)
    elif choice == 1:
        # Dirichlet blocks
        blocks = rng.integers(5, 30)
        a = _shape_dirichlet_blocks(n, rng, blocks=blocks)
    else:
        # Dirichlet + Gaussian smoothing
        blocks = rng.integers(5, 30)
        a = _shape_dirichlet_blocks(n, rng, blocks=blocks)
        sigma = rng.uniform(0.0, 5.0)
        a = _smooth_gaussian(a, sigma)

    # ensure positivity & normalization
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a

# Optimisation primitives (Adam on smooth max, exponentiated GD, etc.)

def _adam_phase(a0: np.ndarray,
                deadline: float,
                start_time: float,
                fft_len: int) -> np.ndarray:
    """
    Adam optimiser on the smooth log‑sum‑exp approximation of the max
    convolution.  Returns the best simplex‑feasible vector seen.
    """
    y = np.log(np.maximum(a0, 1e-16))          # logits for soft‑max
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # extended schedule – smoother beginning, finer end
    alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
                  80.0, 160.0, 320.0, 640.0, 1280.0]
    iters_per_alpha = 8000   # a bit more than the original 6000

    best_seq = a0.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    for alpha in alpha_sched:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq
            a = _softmax_from_logits(y)

            # smooth objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha                # smooth approximation of max(conv)
            obj = 2.0 * a.shape[0] * f_val

            # gradient wrt a
            grad_a = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * a.shape[0])

            # chain rule to log‑softmax
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 300 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                          f"iter={i:,}, score={sc:.6f}")

    return best_seq


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated (mirror) gradient descent on the true max‑convolution.
    Returns the best vector seen and its corresponding score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    cur_conv = _conv_self(a)
    best_score = 2.0 * n * cur_conv.max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        # step‑size schedule – a bit more aggressive at the start
        step = 0.30 / math.sqrt(it)
        a = a * np.exp(-step * grad)

        # projection onto simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """Greedy deterministic moves that try to lower the current peak."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = index of the smallest coefficient (avoid src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.3
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Classic simulated‑annealing mass‑transfer moves."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, "
                  f"score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method consists of
        * a fast random‑search stage to obtain a promising start,
        * a longer Adam smoothing phase,
        * exponentiated‑gradient descent on the exact max‑convolution,
        * a brief greedy peak‑reduction,
        * a short simulated‑annealing finish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    try:
        starter_seq = np.asarray(height_sequence_1, dtype=float)
        if starter_seq.ndim == 1 and starter_seq.size > 0:
            n = starter_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4800
        starter_seq = None

    # ------------------------------------------------------------------
    # Phase 0 – cheap random search for a good seed
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    random_phase_deadline = start_time + min(0.15 * budget_s, 180.0)

    while time.time() < random_phase_deadline:
        cand = _random_candidate(n, rng)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:6.2f}s] random‑search improve: "
                  f"score={sc:.6f}")

    # fallback to uniform if random search produced nothing
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth max‑convolution
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + min(0.25 * (deadline - time.time()), 200.0)
    fft_len = _next_pow2(3 * n - 2)
    a_adam = _adam_phase(best_seq, phase1_deadline, start_time, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient (sub‑gradient) descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + 0.55 * remaining
        a_grad, score_grad = _subgradient_descent(a_adam, phase2_deadline, start_time)
    else:
        a_grad, score_grad = a_adam, evaluate_sequence(a_adam.tolist())

    # ------------------------------------------------------------------
    # Phase 3 – deterministic peak reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        peak_deadline = time.time() + 0.12 * (deadline - time.time())
        a_peak, score_peak = _run_peak_reduction(a_grad, peak_deadline, start_time)
    else:
        a_peak, score_peak = a_grad, score_grad

    # ------------------------------------------------------------------
    # Phase 4 – brief simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            a_peak,
            deadline,
            start_time,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=200_000,
        )
    else:
        a_final, score_final = a_peak, score_peak

    # ------------------------------------------------------------------
    # Choose the best among the stages
    # ------------------------------------------------------------------
    stage_scores = [
        (evaluate_sequence(a_adam.tolist()), a_adam),
        (score_grad, a_grad),
        (score_peak, a_peak),
        (score_final, a_final),
    ]
    local_best_score, local_best_seq = min(stage_scores, key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    local_best_seq = np.maximum(local_best_seq, 0.0)
    total = local_best_seq.sum()
    if total > 0:
        local_best_seq /= total
    else:
        local_best_seq = np.full_like(local_best_seq, 1.0 / n)

    final_score = evaluate_sequence(local_best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in local_best_seq.tolist()]
