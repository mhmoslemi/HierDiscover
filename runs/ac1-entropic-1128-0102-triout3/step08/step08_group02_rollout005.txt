"""
Hybrid multi‑scale optimiser for the step‑function bound.

The algorithm proceeds in two resolution levels:
  1. A low‑resolution (≈600 points) optimisation where many cheap
     iterations of a smooth‑max Adam phase and a long AdaGrad‑style
     exponentiated‑gradient descent are performed.
  2. The best low‑resolution solution is up‑sampled to the target length
     (the length of ``height_sequence_1`` or 4800) and refined with the
     same three phases (Adam → AdaGrad sub‑gradient → deterministic peak‑
     reduction → simulated‑annealing).

All updates keep the vector on the probability simplex, guaranteeing a
positive sum.  The schedule of the Adam phase explores a large range of
smoothness parameters (α up to 1280), while the AdaGrad descent uses a
per‑coordinate step size that automatically decays.  Deterministic
peak‑reduction moves mass away from the largest convolution contribution,
and a final annealing pass performs random mass‑transfer moves with a
geometric cooling schedule.

Multiple diverse starts (uniform, exponential, U‑shape, random smooth,
and a down‑sampled previous best) are tried; the globally best sequence
found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x >=0, sum(x)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) w.r.t. a.
    Returns a vector of shape (n,).
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()  # soft‑max weights, length 2n‑1

    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)
    n = a.shape[0]
    Lw = w_rev.shape[0]
    grad = 2.0 * corr[Lw - 1 - np.arange(n)]
    return grad


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the (non‑smooth) max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)
    n = a.shape[0]
    Lw = w_rev.shape[0]
    grad = 2.0 * corr[Lw - 1 - np.arange(n)]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _downsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation (down‑sampling) preserving total sum."""
    return _upsample(seq, new_n)   # same routine works both ways

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray,
                phase_deadline: float,
                start_time: float,
                fft_len: int,
                n: int,
                iters_per_alpha: int = 6000) -> np.ndarray:
    """
    Adam optimisation on a smoothed version of the max‑convolution.
    Returns the best simplex‑feasible vector seen.
    """
    y = np.log(np.maximum(a0, 1e-16))        # logits
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # schedule of smoothing parameters
    alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
              40.0, 80.0, 160.0, 320.0, 640.0, 1280.0]

    best_seq = a0.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    for alpha in alphas:
        for i in range(iters_per_alpha):
            if time.time() >= phase_deadline:
                return best_seq
            a = _softmax_from_logits(y)

            conv = _conv_self(a)
            # smooth objective (log‑sum‑exp)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val

            # gradient w.r.t. a
            grad_a = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * n)

            # chain rule back to logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 300 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                          f"iter={i:,}, score={sc:.6f}")

    return best_seq


def _subgradient_adagrad(a0: np.ndarray,
                         subgrad_deadline: float,
                         start_time: float,
                         n: int) -> tuple[np.ndarray, float]:
    """
    Long exponentiated‑gradient descent on the true max‑convolution,
    using an AdaGrad per‑coordinate step‑size schedule.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    G = np.zeros_like(a)          # accumulated squared gradients
    eta0 = 0.5                    # base learning rate
    it = 0
    while time.time() < subgrad_deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # exact sub‑gradient of the max
        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        # AdaGrad accumulated steps
        G += grad ** 2
        step_vec = eta0 / (np.sqrt(G) + 1e-12)

        # multiplicative update (mirror descent w.r.t. KL)
        a = a * np.exp(-step_vec * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Greedy deterministic moves that try to lower the current peak.
    Moves a small fraction of the coefficient that participates most in
    the maximal convolution entry to the smallest coefficient.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]   # source of the largest contribution

        # destination is the smallest coefficient not equal to src
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.3
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] Peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing mass‑transfer moves.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: iter={it:,}, "
                  f"score={score:.6f}")

    return best_seq, best_score


def _optimise_at_resolution(a_start: np.ndarray,
                            deadline: float,
                            start_time: float,
                            n: int,
                            low_res: bool = False) -> tuple[np.ndarray, float]:
    """
    Run the four optimisation phases (Adam → AdaGrad sub‑gradient → peak‑reduction
    → annealing) on a vector of length ``n``.  The amount of time allocated to
    each phase is a fixed fraction of the remaining time.
    """
    # 1. Adam phase (smooth max)
    remaining = deadline - time.time()
    if remaining < 5.0:
        # Not enough time for anything else – just return the start.
        return a_start, evaluate_sequence(a_start.tolist())

    adam_time = min(remaining * 0.12, 120.0)
    fft_len = _next_pow2(3 * n - 2)

    # for low‑resolution runs we can afford more inner iterations
    iters_per_alpha = 8000 if low_res else 4000
    a_adam = _adam_phase(a_start, time.time() + adam_time, start_time,
                         fft_len, n, iters_per_alpha)

    # 2. AdaGrad sub‑gradient descent
    remaining = deadline - time.time()
    if remaining <= 0:
        return a_adam, evaluate_sequence(a_adam.tolist())
    subgrad_time = remaining * 0.55
    a_sub, score_sub = _subgradient_adagrad(a_adam,
                                            time.time() + subgrad_time,
                                            start_time, n)

    # 3. Deterministic peak reduction
    remaining = deadline - time.time()
    if remaining <= 0:
        return a_sub, score_sub
    peak_time = remaining * 0.15
    a_peak, score_peak = _run_peak_reduction(a_sub,
                                              time.time() + peak_time,
                                              start_time)

    # 4. Simulated annealing
    remaining = deadline - time.time()
    if remaining <= 0:
        return a_peak, score_peak
    a_final, score_final = _run_annealing(a_peak,
                                          deadline,
                                          start_time,
                                          start_temp=5e-2,
                                          end_temp=1e-6,
                                          steps=300_000)

    # Pick the best among the stages
    stage_scores = [
        (evaluate_sequence(a_adam.tolist()), a_adam),
        (score_sub, a_sub),
        (score_peak, a_peak),
        (score_final, a_final)
    ]
    best_score, best_seq = min(stage_scores, key=lambda x: x[0])
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine uses a two‑scale optimisation strategy with several diverse
    starting points and returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine the target length
    # ------------------------------------------------------------------
    try:
        starter_seq = np.asarray(height_sequence_1, dtype=float)
        if starter_seq.ndim == 1 and starter_seq.size > 0:
            n_target = starter_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n_target = 4800
        starter_seq = None

    # ------------------------------------------------------------------
    # Low‑resolution size (≈600 points, but not larger than target)
    # ------------------------------------------------------------------
    n_low = min(600, n_target)

    # ------------------------------------------------------------------
    # Build a diverse list of starting vectors (at low resolution)
    # ------------------------------------------------------------------
    starts = []

    # uniform
    starts.append(np.full(n_low, 1.0 / n_low))

    # exponential decay (left‑heavy)
    r = 0.985
    exp_vec = np.array([r ** i for i in range(n_low)], dtype=float)
    exp_vec /= exp_vec.sum()
    starts.append(exp_vec)

    # symmetric U‑shape (large at edges)
    idx = np.arange(n_low)
    u_shape = 1.0 / (np.minimum(idx + 1, n_low - idx) + 1.0)
    u_shape /= u_shape.sum()
    starts.append(u_shape)

    # random smooth vector (Gaussian‑filtered noise)
    rng = np.random.default_rng()
    noise = rng.normal(size=n_low)
    smooth = np.convolve(noise, np.ones(15) / 15, mode='same')
    smooth = np.clip(smooth, 0, None)
    if smooth.sum() == 0:
        smooth = np.full(n_low, 1.0 / n_low)
    else:
        smooth /= smooth.sum()
    starts.append(smooth)

    # down‑sampled previous best (if available)
    if starter_seq is not None:
        starts.append(_downsample(starter_seq, n_low))

    # ------------------------------------------------------------------
    # Global best holder
    # ------------------------------------------------------------------
    best_seq_global = None
    best_score_global = float('inf')

    # ------------------------------------------------------------------
    # Optimise each start independently (low‑resolution → up‑sample → high‑res)
    # ------------------------------------------------------------------
    for idx_start, a0_low in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{idx_start + 1}")

        # ensure proper simplex normalisation
        a0_low = np.maximum(a0_low, 0.0)
        if a0_low.sum() > 0:
            a0_low /= a0_low.sum()
        else:
            a0_low = np.full_like(a0_low, 1.0 / n_low)

        # ------------------------------------------------------------------
        # Phase 1 – low‑resolution optimisation
        # ------------------------------------------------------------------
        low_deadline = time.time() + min(0.25 * (deadline - start_time), 120.0)
        a_opt_low, _ = _optimise_at_resolution(a0_low,
                                               low_deadline,
                                               start_time,
                                               n_low,
                                               low_res=True)

        # ------------------------------------------------------------------
        # Phase 2 – up‑sample to target resolution
        # ------------------------------------------------------------------
        a_up = _upsample(a_opt_low, n_target)
        a_up = np.maximum(a_up, 0.0)
        if a_up.sum() > 0:
            a_up /= a_up.sum()
        else:
            a_up = np.full_like(a_up, 1.0 / n_target)

        # ------------------------------------------------------------------
        # Phase 3 – high‑resolution optimisation (remaining budget)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_opt_high, score_high = _optimise_at_resolution(
                a_up,
                deadline,
                start_time,
                n_target,
                low_res=False
            )
        else:
            a_opt_high = a_up
            score_high = evaluate_sequence(a_opt_high.tolist())

        # ------------------------------------------------------------------
        # Update global best
        # ------------------------------------------------------------------
        if score_high < best_score_global:
            best_score_global = score_high
            best_seq_global = a_opt_high.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST *** "
                  f"score={best_score_global:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation
    # ------------------------------------------------------------------
    if best_seq_global is None:
        best_seq_global = np.full(n_target, 1.0 / n_target)

    best_seq_global = np.maximum(best_seq_global, 0.0)
    total = best_seq_global.sum()
    if total > 0:
        best_seq_global /= total
    else:
        best_seq_global = np.full_like(best_seq_global, 1.0 / n_target)

    final_score = evaluate_sequence(best_seq_global.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # Return as plain Python list of floats
    return [float(x) for x in best_seq_global.tolist()]
