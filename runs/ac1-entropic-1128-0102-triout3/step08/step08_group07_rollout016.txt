"""
Improved step‑function search for the ``evaluate_sequence`` objective.

The algorithm builds on the reference implementation but adds
   • a larger and more diverse set of structured initialisers
     (Chebyshev, Kaiser, Hann, Blackman, Gaussian);
   • more aggressive multi‑restart Adam optimisation with a slower
     geometric cooling of the soft‑max temperature τ;
   • a refined sub‑gradient descent (larger initial step, adaptive
     momentum, early‑stop when no improvement for a while);
   • a longer simulated‑annealing polishing phase with an enlarged
     move budget (mass‑transfer, simple swaps and three‑way splits).

All stages respect the global 1000 s budget, and the sequence is
projected onto the probability simplex (∑ aᵢ = 1, aᵢ ≥ 0) after each
update – exactly the scaling used by ``evaluate_sequence``.
The returned list contains plain Python floats.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σa = 1)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth soft‑max surrogate
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9993,
                      step_size: float = 0.20,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max proxy
        Oτ(a)=2·n·τ·log Σexp(conv(a)/τ)
    with Adam and Euclidean projection onto the simplex.
    Returns the best vector observed and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline:
        it += 1

        # true (unsmoothed) score for logging
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stabilise
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of Oτ : 4·n·(a ⊛ p)
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)         # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]       # central n entries
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.14,
                             patience: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).

    ``patience`` stops the loop early if no improvement was seen for that
    many iterations – useful to re‑allocate time to the annealing stage.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

        # early exit if stagnant for a long time
        if it - last_improve > patience:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers, swaps, splits)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-8,
                   steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that mixes three simple moves:
        – transfer a random fraction of mass between two cells,
        – swap the contents of two cells,
        – split a fraction of mass from one cell onto two others.
    The true objective (2·n·max(conv)) is used for acceptance.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        move_type = rng.integers(3)          # 0 – transfer, 1 – swap, 2 – split

        if move_type == 0:   # simple transfer
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue
            eps = a[src] * rng.random() * 0.30      # up to 30 % of source mass
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

        elif move_type == 1:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            a_prop = a.copy()
            a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        else:                # split a mass from src onto two destinations
            src = rng.choice(n, p=a / a.sum())
            if a[src] < 1e-12:
                continue
            dst1, dst2 = rng.integers(n, size=2)
            if src in (dst1, dst2):
                continue
            eps = a[src] * rng.random() * 0.20   # up to 20 % of source
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst1] += eps * 0.5
            a_prop[dst2] += eps * 0.5

        # evaluate the proposal
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length and re‑normalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisation helpers (extra windows)
# ----------------------------------------------------------------------


def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    Probabilities:
      0.3 Chebyshev, 0.3 Kaiser, 0.2 Hann, 0.1 Gaussian, 0.1 plain Dirichlet.
    """
    # optional import of scipy.signal
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.3:
        # Chebyshev window (attenuation 30–120 dB)
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
    elif have_signal and r < 0.6:
        # Kaiser window (beta 0–14)
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
    elif r < 0.8:
        # Hann window
        i = np.arange(n, dtype=float)
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif r < 0.9:
        # Gaussian window (σ ≈ n/6)
        i = np.arange(n, dtype=float)
        sigma = n / 6.0
        w = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    else:
        # plain Dirichlet draw
        w = rng.random(n)

    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.
    The pipeline consists of
        1. many short Adam runs on a modest size (n≈4096) with diverse starts,
        2. up‑sampling to the target size (n≈8192) and a longer Adam phase,
        3. a projected sub‑gradient phase,
        4. a heavyweight simulated‑annealing polishing stage.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Use a previous solution as a starter if it exists
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest size (n≈4096)
    # ------------------------------------------------------------------
    n0 = 4096
    # allocate ~20 % of the remaining budget (but at least 30 s)
    remaining = deadline - time.time()
    stage0_time = max(30.0, 0.20 * remaining)
    slice_deadline = time.time() + stage0_time

    max_restarts = 20
    for r in range(max_restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = _window_initialiser(n0, rng)

        # initialise τ from the current max‑convolution (a rough scale)
        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 8.0, 1e-9)

        a_opt, score = _run_adam_softmax(
            a0, n0, slice_deadline, start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9993,
            step_size=0.22,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # Fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – upscale to a larger size (≈8192) and continue Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    a1 = _upsample(best_seq, n1)
    conv1 = _conv_self(a1)
    tau0_1 = max(conv1.max() * 8.0, 1e-10)

    remaining = deadline - time.time()
    stage1_time = max(20.0, 0.25 * remaining)
    slice_deadline = time.time() + stage1_time

    a1_opt, score1 = _run_adam_softmax(
        a1, n1, slice_deadline, start_time,
        tau0=tau0_1,
        tau_end=1e-14,
        tau_decay=0.9993,
        step_size=0.15,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (shortened if stagnates)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n2 = best_seq.shape[0]
        grad_deadline = deadline - 5.0               # keep a few seconds for SA
        a2_opt, score2 = _run_subgradient_descent(
            best_seq, n2, grad_deadline, start_time,
            beta=0.97,
            step0=0.14,
            patience=250_000,
        )
        if score2 < best_score:
            best_score = score2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – extended simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-8,
            steps=1_200_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (protect against tiny drift)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list – the harness will invoke ``evaluate_sequence``
    return [float(x) for x in best_seq]
