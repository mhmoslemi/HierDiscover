"""
Propose a step‑function that beats the previous bound (≈1.5097).

Approach
--------
1. **Analytic family** –  
   Build a simple symmetric “tent” shape on a discrete grid:
   a_i ∝ min(i, n−1−i)^p   (p>0).  
   This mimics the continuous triangular density whose self‑convolution
   attains an L∞‑norm of 3/4, i.e. an objective of exactly 1.5.
   We sweep a few exponents p and keep the best candidate.

2. **Refinement** –  
   Starting from the best analytic candidate we run a projected
   sub‑gradient descent (the same method used in the reference code)
   for the remaining time budget.  The update projects onto the
   probability simplex after every step.

3. **Final normalisation** –  
   After the optimisation the vector is projected back onto the simplex
   (non‑negative, sum = 1) and returned as a plain Python list.

The algorithm works for any length `n`.  The default `n = 4096`
matches the previous best solution, but the code is written so that the
budget is respected even on a single‑CPU machine.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper linear‑algebra utilities (FFT based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Returns the Euclidean projection of v onto {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the true objective.
    Returns the best iterate seen before `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # index attaining the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a (see reference code)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep a on the simplex
        a = _project_simplex(a)

    return best_a, best_score
