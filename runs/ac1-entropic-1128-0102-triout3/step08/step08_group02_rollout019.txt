"""
Hybrid optimiser with extended phases.

The optimiser builds on the previous “Adam → sub‑gradient → peak‑reduction →
annealing” pipeline but adds a few simple yet effective tricks:

* **More diverse starts** – besides the five deterministic starts we also draw
  several Dirichlet samples with different concentration parameters.
* **Longer Adam smoothing** – the α‑schedule is extended (up to 1280) and the
  number of iterations per α is doubled.
* **More aggressive exponentiated‑gradient descent** – a larger initial step
  (0.5 instead of 0.25) and a slower decay (1/√it) allow a deeper search.
* **Stronger peak reduction** – the greedy “move mass from the biggest
  contributor of the current peak to the smallest coefficient” is performed
  for many more iterations.
* **Heavier annealing** – 600 000 moves with a very slow cooling schedule
  give the final phase plenty of time to escape shallow minima.
* **Local refinement** – after annealing a tiny hill‑climbing pass (random
  mass‑transfer moves accepted only if they improve the score) is executed.

All phases respect a single global time budget (default 1000 s).  The final
sequence is normalised to unit sum and all entries are forced to be non‑negative.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max  (1/α)·log Σ exp(α·conv_k) w.r.t. a.
    Returns a vector of shape (n,).
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                # soft‑max weights
    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _max_subgrad(a: np.ndarray, conv: np.ndarray,
                 rel_tol: float, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    # cross‑correlation a ⋆ active
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with the given concentration (>0)."""
    if concentration <= 0:
        concentration = 1.0
    draws = rng.gamma(concentration, 1.0, size=n)
    draws /= draws.sum()
    return draws

def _adam_phase(a0: np.ndarray,
                deadline: float,
                start_time: float,
                fft_len: int) -> np.ndarray:
    """
    Adam optimisation on the smooth approximation of the max convolution.
    Returns the best sequence (in simplex) seen during the phase.
    """
    y = np.log(np.maximum(a0, 1e-16))        # logits
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # extended schedule – a bit finer at the end
    alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
                  80.0, 160.0, 320.0, 640.0, 1280.0]
    iters_per_alpha = 8000

    best_seq = a0.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    for alpha in alpha_sched:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq
            a = _softmax_from_logits(y)

            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            # log‑sum‑exp (smooth max)
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * a.shape[0] * f_val

            # gradient w.r.t. a
            grad_a = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * a.shape[0])

            # chain rule to the logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if i % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                          f"iter={i:,}, score={sc:.6f}")

    return best_seq


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> np.ndarray:
    """
    Long exponentiated‑gradient descent on the true max‑convolution.
    Returns the best sequence seen (sum is kept equal to 1).
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max (active set)
        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        # multiplicative update (always stays non‑negative)
        step = 0.5 / math.sqrt(it)          # a slightly larger step than before
        a = a * np.exp(-step * grad)

        # renormalise onto the simplex (exact sum = 1)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 5000) -> np.ndarray:
    """Greedy deterministic moves that try to lower the current peak."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # where the biggest product lives

        # destination = index with smallest coefficient (but not src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.3
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    return a


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 600_000) -> np.ndarray:
    """Classic simulated‑annealing mass‑transfer moves."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq


def _local_hill_climb(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      attempts: int = 100_000) -> np.ndarray:
    """
    Very cheap hill‑climbing pass after annealing.
    Moves a small amount of mass from a random source to a random destination
    and keeps the change only if it improves the score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = evaluate_sequence(a.tolist())
    rng = np.random.default_rng()

    for it in range(attempts):
        if time.time() >= deadline:
            break
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.1
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        sc = evaluate_sequence(a_prop.tolist())
        if sc < best_score:
            a = a_prop
            best_score = sc
            print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve: iter={it:,}, score={sc:.6f}")

    return a

    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # Convert to plain Python floats for the caller
    return [float(x) for x in best_seq.tolist()]
