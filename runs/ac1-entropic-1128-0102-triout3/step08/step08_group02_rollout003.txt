"""
Hybrid search for a low‑max‑convolution step function.

The method consists of three stages:

1. **Broad shape exploration** – generate a collection of parametrised
   families (exponential, power‑law, U‑shape, triangular, random‑smooth)
   and keep the one with the smallest evaluation value.

2. **Adaptive exponentiated‑gradient descent** – starting from the best
   shape, repeatedly move mass in the direction of the exact sub‑gradient
   of the max‑convolution (active set).  A back‑tracking line search with
   multiplicative updates guarantees that each accepted step strictly
   reduces the objective.  The step size shrinks as 1/√iteration and is
   halved until improvement is observed.

3. **Deterministic peak‑reduction** – a greedy pass that moves a small
   fraction of the heaviest contributor to the current peak towards the
   currently smallest coefficient.  This often yields a final polish.

All phases respect the overall time budget (default 1000 s).  The final
sequence is normalised to sum = 1 (the objective is scale‑invariant) and
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _max_subgrad(a: np.ndarray,
                 conv: np.ndarray,
                 rel_tol: float,
                 fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k].
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    Returns a vector of shape (n,).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)  # mask of active peaks
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad

def _generate_candidate_shapes(n: int) -> list[np.ndarray]:
    """Return a list of diverse normalised non‑negative vectors of length n."""
    shapes = []

    # uniform
    shapes.append(np.full(n, 1.0 / n))

    # exponential left‑hand heavy
    for r in np.linspace(0.90, 0.999, 6):
        vec = np.array([r ** i for i in range(n)], dtype=float)
        shapes.append(vec / vec.sum())

    # exponential right‑hand heavy
    for r in np.linspace(0.90, 0.999, 6):
        vec = np.array([r ** (n - 1 - i) for i in range(n)], dtype=float)
        shapes.append(vec / vec.sum())

    # symmetric exponential (heavy at both ends)
    for r in np.linspace(0.90, 0.999, 6):
        vec = np.array([r ** min(i, n - 1 - i) for i in range(n)], dtype=float)
        shapes.append(vec / vec.sum())

    # power‑law left‑hand heavy
    for p in np.linspace(0.5, 2.5, 5):
        vec = np.array([1.0 / ((i + 1) ** p) for i in range(n)], dtype=float)
        shapes.append(vec / vec.sum())

    # power‑law edge‑heavy (U‑shape)
    for p in np.linspace(0.5, 2.5, 5):
        vec = np.array(
            [1.0 / ((min(i + 1, n - i)) ** p) for i in range(n)], dtype=float
        )
        shapes.append(vec / vec.sum())

    # symmetric triangular (peak in the centre)
    center = n // 2
    left = np.arange(1, center + 1)
    right = np.arange(center, 0, -1)[: n - center]
    tri = np.concatenate([left, right])
    shapes.append(tri.astype(float) / tri.sum())

    # smooth random (Gaussian‑filtered noise)
    rng = np.random.default_rng(0)

    smooth = np.clip(smooth, 0, None)
    if smooth.sum() > 0:
        shapes.append(smooth / smooth.sum())
    else:
        shapes.append(np.full(n, 1.0 / n))

    return shapes


# ----------------------------------------------------------------------
# Greedy peak‑reduction (deterministic final polishing)
# ----------------------------------------------------------------------


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """Greedy deterministic moves that try to lower the current peak."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (avoid src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.3
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(
                f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}"
            )
        else:
            no_improve += 1

    final_score = 2.0 * n * _conv_self(a).max()
    return a, final_score

    # 1.  Shape exploration
    # ------------------------------------------------------------------
    candidates = _generate_candidate_shapes(n)
    if starter_seq is not None:
        # include the provided sequence as one more candidate
        cand = np.maximum(starter_seq, 0.0)
        if cand.sum() > 0:
            cand = cand / cand.sum()
        else:
            cand = np.full_like(cand, 1.0 / n)
        candidates.append(cand)

    best_seq = None
    best_score = float("inf")
    print(f"[{time.time() - start_time:6.2f}s] evaluating {len(candidates)} shape candidates")
    for idx, vec in enumerate(candidates):
        if time.time() >= deadline:
            break
        # evaluate via the true objective (scale‑invariant)
        sc = evaluate_sequence(vec.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = vec.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] new shape best #{idx+1}: score={sc:.6f}"
            )

    # If for some reason no candidate survived, fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 2.  Adaptive exponentiated‑gradient descent on true max‑convolution
    # ------------------------------------------------------------------
    a = best_seq.copy()
    fft_len = _next_pow2(3 * n - 2)
    it = 0
    rng = np.random.default_rng(seed + 12345)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] grad descend improve: iter={it:,}, score={cur_score:.6f}"
            )

        # compute exact sub‑gradient of the max (active set)
        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        # step size schedule (decreasing) – start relatively large
        base_step = 0.5 / math.sqrt(it)
        step = base_step
        improved = False

        # back‑tracking line search (max 6 reductions)
        for _ in range(6):
            a_new = a * np.exp(-step * grad)
            s = a_new.sum()
            if s > 0:
                a_new /= s
            else:
                a_new = np.full_like(a, 1.0 / n)

            new_max = _conv_self(a_new).max()
            if new_max < cur_max - 1e-15:  # strict improvement
                a = a_new
                improved = True
                break
            step *= 0.5

        if not improved:
            # small stochastic perturbation to escape possible plateaus
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst and a[src] > 1e-14:
                eps = a[src] * 0.03  # move 3 % of source mass
                a_prop = a.copy()
                a_prop[src] -= eps
                a_prop[dst] += eps
                a_prop /= a_prop.sum()
                new_max = _conv_self(a_prop).max()
                if new_max < cur_max:
                    a = a_prop
                    # (no need to set improved flag – we already recorded score at top)

    # ------------------------------------------------------------------
    # 3.  Deterministic greedy peak‑reduction (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_peak, score_peak = _run_peak_reduction(
            best_seq, deadline, start_time, max_no_improve=3000
        )
        if score_peak < best_score:
            best_score = score_peak
            best_seq = a_peak.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] post‑peak best score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 4.  Sanity & return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # Return a plain Python list of floats (as required by the evaluator)
    return [float(v) for v in best_seq.tolist()]
