def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Load starter if available
    try: 
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except:
        starter = None

    n = 4800  # maybe adjustable
    if starter is not None:
        a = _upsample(starter, n)
    else:
        a = np.random.rand(n)
        a /= a.sum()

    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0

    # hyperparameters
    eps_start = 0.04
    eps_end = 0.001  # lower for fine tuning
    T_start = 1e-4  # temperature
    # perhaps we use exponentially decreasing eps and T

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # schedule
        frac = elapsed / total_time
        eps_frac = max(eps_end, eps_start * (1 - frac)**2)   # square for faster decay
        T = max(1e-8, T_start * (1 - frac)**2)

        # find current max index
        k_max = int(np.argmax(conv))

        # compute source probabilities: contributions line.
        i0 = max(0, k_max - (n - 1))
        i1 = min(n - 1, k_max)
        i_range = np.arange(i0, i1 + 1)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            src = rng.choice(i_range, p=contrib / total_contrib)
        else:
            src = rng.choice(n, p=a / a.sum())
        if a[src] < 1e-15:
            continue

        # Choose destination: bias to low a and far from src maybe
        # compute weights inversely proportional to a[i] and also distance to src
        weight = 1.0 / (a + 1e-12)
        # avoid src
        weight[src] = 0.0
        # maybe also incorporate distance weighting: distance = abs(i - src) + 1
        # weight = weight / distance
        dist = np.abs(np.arange(n) - src) + 1.0
        weight = weight / dist
        prob_dst = weight / weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # epsilon
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        # compute new conv
        conv_new = conv.copy()
        conv_new[src : src + n] -= 2.0 * eps * a
        conv_new[dst : dst + n] += 2.0 * eps * a
        new_max = conv_new.max()

        accept = False
        if new_max < max_val - 1e-14:
            accept = True
        else:
            # uphill acceptance
            delta = new_max - max_val
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            # apply move
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improve: iter={it:,}, max={best_max:.8f}")

    # final normalize and return best
    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()
    return best_a.tolist()

"""
A simulated‑annealing / local‑move search for a non‑negative step function
that minimises ``evaluate_sequence``.

The algorithm works entirely in the probability simplex (the sum of the
heights is kept equal to 1, which makes the objective
    2·n·max(conv)   (conv = a⋆a)
independent of any scaling).  Starting from the best known construction
(if the global ``height_sequence_1`` is present) we repeatedly perform two
kinds of elementary moves:

* **small mass shift** – move a tiny fraction of the mass from an index that
  contributes heavily to the current maximal convolution entry to an index
  that contributes little.  The convolution array is updated analytically,
  which is O(n) per move.

* **swap** – exchange the whole masses of a heavy index and a light index.
  This is implemented as a single large mass‑shift move (ε = a_heavy−a_light).

Moves are accepted if they lower the maximal convolution value, otherwise
they are accepted with a temperature‑dependent probability (simulated
annealing).  Both the step size (ε‑fraction) and the temperature decay
smoothly during the run.  The best sequence seen before the deadline is
returned.

The whole routine respects the 1000 s time budget, uses only one CPU core,
and does not perform any I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Main entry point
# ----------------------------------------------------------------------

    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """

    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # try to start from the best known construction, otherwise random
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800                                 # problem size
    if starter is not None:
        a = _upsample(starter, final_n)
    else:
        a = np.random.rand(final_n)
        a /= a.sum()

    # convolution of the current candidate
    conv = _conv_self(a)                           # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Hyper‑parameters (all decay with time)
    # ------------------------------------------------------------------
    eps_start = 0.04           # initial fraction of a[src] moved
    eps_end   = 0.001          # final fraction
    T_start   = 1e-4           # initial temperature for simulated annealing
    p_swap    = 0.02           # probability of performing a full swap move
    rel_improve = 1e-12        # numerical tolerance for improvement

    total_time = deadline - start_time
    it = 0
    it_since_best = 0

    # ------------------------------------------------------------------
    # Main SA loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time

        # annealing schedules
        eps_frac = max(eps_end, eps_start * (1.0 - frac) ** 3)
        T = max(1e-12, T_start * (1.0 - frac) ** 3)

        # --------------------------------------------------------------
        # Choose move type
        # --------------------------------------------------------------
        if rng.random() < p_swap:
            # ---------- swap move (full exchange of two entries) ----------
            # source biased to large entries, destination biased to small ones
            prob_src = a / a.sum()
            src = rng.choice(final_n, p=prob_src)

            # inverse weighting for destination (avoid src)
            inv_weight = 1.0 / (a + 1e-12)
            inv_weight[src] = 0.0
            prob_dst = inv_weight / inv_weight.sum()
            dst = rng.choice(final_n, p=prob_dst)

            if src == dst:
                continue

            # ensure src has the larger mass
            if a[src] < a[dst]:
                src, dst = dst, src

            diff = a[src] - a[dst]          # amount to move (full swap)
            if diff <= 0.0:

            eps = diff                       # whole difference

        else:

            # the shift that currently attains the maximal convolution entry
            k_max = int(np.argmax(conv))

            j_range = k_max - i_range
            contrib = a[i_range] * a[j_range]
            tot_contrib = contrib.sum()
            if tot_contrib > 0:

            dist = np.abs(np.arange(final_n) - src) + 1.0

            dst = rng.choice(final_n, p=prob_dst)

        if accept:
            # apply the move
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            it_since_best += 1

            # track the overall best solution
            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                it_since_best = 0
                score = 2.0 * final_n * best_max          # sum(a)=1 → exact score
                print(f"[{time.time() - start_time:.2f}s] "
                      f"new best: iter={it:,}, max={best_max:.8f}, score={score:.6f}")

        # occasional re‑normalisation to avoid drift
        if (it & 0xFFF) == 0:               # every 4096 iterations
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
                # recompute the convolution from scratch to kill accumulated error
                conv = _conv_self(a)
                max_val = conv.max()
            else:
                a = np.full(final_n, 1.0 / final_n)
                conv = _conv_self(a)
                max_val = conv.max()

        # simple restart if we have been stuck for a long time
        if it_since_best > 300_000 and (time.time() - start_time) < (deadline - 30):
            # re‑initialise from a noisy copy of the current best
            a = best_a + 0.001 * np.random.rand(final_n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            conv = _conv_self(a)
            max_val = conv.max()
            it_since_best = 0
            print(f"[{time.time() - start_time:.2f}s] restart after stagnation")

    # ------------------------------------------------------------------
    # Return the best sequence seen (already normalised)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()
    # optional sanity check (uncomment if you wish to see the final score)
    # try:
    #     from __main__ import evaluate_sequence
    #     print("final evaluate_sequence =", evaluate_sequence(best_a.tolist()))
    # except Exception:
    #     pass
    return [float(x) for x in best_a]
