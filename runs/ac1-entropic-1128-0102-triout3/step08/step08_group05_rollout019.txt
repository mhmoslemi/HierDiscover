def hill_climb(a, deadline):
    n = len(a)
    # conv: linear convolution using FFT for accuracy, but after that we will update manually.
    # Use direct convolution? but we need conv array for initial values.

    conv = np.convolve(a, a) # shape 2n-1

    # Build heap
    heap = [(-val, idx) for idx, val in enumerate(conv)]
    heapq.heapify(heap)

    best_a = a.copy()
    best_score = 2 * n * conv.max()

    # main loop
    while time.time() < deadline:
        # clean heap to ensure top is up-to-date
        while heap:
            neg_val, idx = heap[0]
            if conv[idx] != -neg_val:
                heapq.heappop(heap) # stale
                continue
            break
        # top
        max_val = -heap[0][0]
        max_idx = heap[0][1]

        # Determine src range
        src_low = max(0, max_idx - (n - 1))
        src_high = min(max_idx, n - 1)

        # try up to 5 random attempts
        accepted = False
        for _ in range(5):
            src = rng.integers(src_low, src_high + 1)
            dst = max_idx - src
            if src == dst or a[src] < 1e-12:
                continue
            # choose direction: we need a[src] < a[dst] to reduce conv
            if a[src] >= a[dst]:
                continue
            eps = a[src] * (0.5 * rng.random())  # up to half of src mass
            if eps <= 0.0:
                continue

            # compute updated conv entries
            idx1 = 2 * src
            idx2 = 2 * dst
            idx3 = max_idx  # src + dst = max_idx
            # note idx3 same as max_idx
            # Use original a values
            a_src = a[src]
            a_dst = a[dst]
            # compute new conv entries
            c1 = conv[idx1] - 2.0 * eps * a_src + eps * eps
            c2 = conv[idx2] + 2.0 * eps * a_dst + eps * eps
            c3 = conv[idx3] + 2.0 * eps * (a_src - a_dst) - 2.0 * eps * eps

            # compute max_other: find max among heap excluding idx1, idx2, idx3 (which may be stale)
            changed_set = {idx1, idx2, idx3}
            # peek for max_other
            # temporarily store popped items
            temp = []
            max_other = -np.inf
            while heap:
                negv, ii = heap[0]
                curv = conv[ii]
                if curv != -negv:
                    heapq.heappop(heap)  # stale
                    continue
                if ii in changed_set:
                    temp.append(heapq.heappop(heap))
                    continue
                max_other = -negv
                break
            # push back temporarily popped items (they are stale but we will push new values later)
            for item in temp:
                heapq.heappush(heap, item)

            # new max after update
            new_max = max(max_other, c1, c2, c3)

            if new_max < max_val - 1e-12:
                # accept
                a[src] -= eps
                a[dst] += eps
                conv[idx1] = c1
                conv[idx2] = c2
                conv[idx3] = c3
                # push new values onto heap
                heapq.heappush(heap, (-c1, idx1))
                heapq.heappush(heap, (-c2, idx2))
                heapq.heappush(heap, (-c3, idx3))
                # update best
                cur_score = 2 * n * new_max
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] hillclimb improve: score={cur_score:.6f}")
                accepted = True
                break
        if not accepted:
            # maybe random perturbation not possible; skip
            pass

    return best_a, best_score

def hill_climb(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)   # shape 2n-1
    max_val = conv.max()
    max_idx = int(conv.argmax())
    best_a = a.copy()
    best_score = 2.0 * n * max_val
    rng = np.random.default_rng()
    no_improve = 0
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        # attempt limited number of moves
        for _ in range(3):
            # Choose src such that src+dst = max_idx
            src_low = max(0, max_idx - (n - 1))
            src_high = min(max_idx, n - 1)
            if src_low > src_high:
                break  # shouldn't happen
            src = rng.integers(src_low, src_high + 1)
            dst = max_idx - src
            if src == dst:
                continue
            if a[src] < 1e-12:
                continue
            # to reduce conv[max_idx], we need a[src] < a[dst]
            if a[src] >= a[dst]:
                continue
            eps = a[src] * rng.random() * 0.4  # up to 40%
            if eps <= 0.0:
                continue
            # compute updated conv values
            idx1 = 2 * src
            idx2 = 2 * dst
            idx3 = max_idx
            a_src = a[src]
            a_dst = a[dst]
            c1 = conv[idx1] - 2.0 * eps * a_src + eps * eps
            c2 = conv[idx2] + 2.0 * eps * a_dst + eps * eps
            c3 = conv[idx3] + 2.0 * eps * (a_src - a_dst) - 2.0 * eps * eps

            # compute tentative new max as max of c1,c2,c3 and possibly other entries unchanged but ≤ old max

"""
search‑candidate – a multi‑stage optimiser for the max‑self‑convolution
objective used by `evaluate_sequence`.

The routine works inside a fixed time budget (default 1000 s).  It
1) starts from the known best construction (if any) or a random vector,
2) quickly shapes the distribution with Adam on a smooth log‑sum‑exp
   surrogate,
3) performs a heavy Euclidean sub‑gradient descent on the true objective,
4) finishes with a specialised hill‑climbing local search that updates the
   convolution in O(1) time (the convolution changes only at three indices
   when a small amount of mass is moved between two positions).  This last
   phase can perform millions of cheap moves and usually squeezes a few
   extra points out of the bound.

All vectors are kept on the probability simplex (non‑negative,
sum ≈ 1) – the evaluation function is scale‑invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x_i≥0, Σx_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for true objective)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.25,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Minimise  Oτ(a)=2·n·τ·log∑exp(conv/τ)  with Adam and simplex projection."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f} τ={tau:.2e}")

        # ----- log‑sum‑exp weights -----
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # shape 2·n‑1, sums to 1

        # ----- gradient of Oτ -----
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)        # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]      # required slice
        grad = 4.0 * n * inner

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on  F(a)=2·n·max(conv(a))."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climbing local optimiser (fast O(1) updates)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Fine‑grained local search.  A move transfers a small ε from a smaller
    entry to a larger one such that the index of the current maximal
    convolution value changes; the convolution is updated analytically
    (only three positions are affected), making millions of cheap moves
    possible.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2·n‑1
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    best_a = a.copy()
    best_score = 2.0 * n * max_val
    rng = np.random.default_rng()
    no_improve = 0
    max_iter_without_progress = 5000

    L = conv.shape[0]                        # = 2·n‑1

    while time.time() < deadline:
        # try a few random “targeted” moves
        improved = False
        for _ in range(3):
            # indices that sum to the current maximal position
            src_low = max(0, max_idx - (n - 1))
            src_high = min(max_idx, n - 1)
            if src_low > src_high:
                break
            src = rng.integers(src_low, src_high + 1)
            dst = max_idx - src
            if src == dst or a[src] < 1e-12:
                continue
            # moving from a smaller entry to a larger one reduces the
            # convolution value at `max_idx`
            if a[src] >= a[dst]:
                continue
            eps = a[src] * rng.random() * 0.40      # up to 40 % of src mass
            if eps <= 0.0:
                continue

            # indices that will change in the convolution
            idx1 = 2 * src
            idx2 = 2 * dst
            idx3 = max_idx

            a_src = a[src]
            a_dst = a[dst]

            c1 = conv[idx1] - 2.0 * eps * a_src + eps * eps
            c2 = conv[idx2] + 2.0 * eps * a_dst + eps * eps
            c3 = conv[idx3] + 2.0 * eps * (a_src - a_dst) - 2.0 * eps * eps

            # compute the maximum over the unchanged entries
            # (a tiny O(n) scan – acceptable because n ≤ 16384)
            max_other = -np.inf
            for i in range(L):
                if i == idx1 or i == idx2 or i == idx3:
                    continue
                v = conv[i]
                if v > max_other:
                    max_other = v

            new_max = max(max_other, c1, c2, c3)

            if new_max < max_val - 1e-12:          # genuine improvement
                # accept the move
                a[src] -= eps
                a[dst] += eps
                conv[idx1] = c1
                conv[idx2] = c2
                conv[idx3] = c3
                max_val = new_max
                max_idx = int(conv.argmax())
                cur_score = 2.0 * n * max_val
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] hillclimb improve: "
                          f"score={cur_score:.6f}")
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1

        # occasional random perturbation to escape possible plateaus
        if no_improve >= max_iter_without_progress:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src != dst and a[src] > 1e-12:
                eps = a[src] * rng.random() * 0.05
                a[src] -= eps
                a[dst] += eps
                conv = _conv_self(a)                # recompute fully (rare)
                max_idx = int(conv.argmax())
                max_val = float(conv[max_idx])
            no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
