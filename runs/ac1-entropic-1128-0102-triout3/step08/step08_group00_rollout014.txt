def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, math, random
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Choose n based on remaining time
    if deadline - start > 600:
        n = 8192
    else:
        n = 4096

    best_seq = None
    best_score = float('inf')

    # Helper to evaluate (fast) and update best
    def eval_and_update(seq):
        nonlocal best_seq, best_score
        # Normalize seq to sum 1
        seq = np.maximum(seq, 0)
        total = seq.sum()
        if total <= 0:
            return
        seq = seq / total
        conv = _conv_self(seq)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start:.2f}s] New best score = {score:.6f}")
        return seq

    # Starting from global if exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.shape[0] == n:
            seq = eval_and_update(starter)
    except Exception:
        pass

    # Parametric shapes: power law and symmetric variant
    for p in np.linspace(0.2, 2.0, 19):  # 0.2 to 2.0 step 0.1
        if time.time() >= deadline:
            break
        idx = np.arange(n)
        # simple power law decreasing from start
        a = (idx + 1) ** (-p)
        # optional symmetrize: reflect to get symmetric shape
        # Let's use symmetric: a = (idx+1)**(-p) + (n-idx)**(-p)
        # But this double counts, may blow up; just keep simple.
        seq = eval_and_update(a)

    # Random restarts with gradient descent
    while time.time() < deadline - 30:  # leave 30s for final polishing
        remaining = deadline - time.time()
        # simple random start
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()
        # short Adam smooth run
        a1, s1 = _adam_smooth(a0, n, time.time() + min(remaining, 120), start,
                               alpha_start=0.5, alpha_end=300.0, max_iters=8000, lr0=0.5)
        if s1 < best_score:
            best_score = s1
            best_seq = a1.copy()
        # subgradient on result
        sub_deadline = time.time() + min(remaining-10, 200)
        a2, s2 = _subgradient_descent(a1, n, sub_deadline, start, best_score)
        if s2 < best_score:
            best_score = s2
            best_seq = a2.copy()
        # quick anneal
        anneal_deadline = time.time() + 30
        a3, s3 = _annealing(a2, n, anneal_deadline, start, best_score, a2,
                             start_temp=1e-2, end_temp=1e-6, steps=20000)
        if s3 < best_score:
            best_score = s3
            best_seq = a3.copy()

    # Final refinement
    if best_seq is not None:
        # longer subgradient
        final_sub_deadline = time.time() + (deadline - time.time()) * 0.6
        a_final, s_final = _subgradient_descent(best_seq, n, final_sub_deadline, start, best_score)
        if s_final < best_score:
            best_score = s_final
            best_seq = a_final.copy()
        # final anneal
        if time.time() < deadline:
            a_final2, s_final2 = _annealing(best_seq, n, deadline, start, best_score, best_seq,
                                            start_temp=5e-2, end_temp=1e-6, steps=150000)
            if s_final2 < best_score:
                best_score = s_final2
                best_seq = a_final2.copy()

    # Normalize final sequence and output as list
    best_seq = np.maximum(best_seq, 0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)
    return [float(x) for x in best_seq]

"""
A fast heuristic optimiser for the “evaluate_sequence” objective.

The algorithm combines three ideas that complement each other:

1. **Simple analytic initialisations** – power‑law distributions
   a_i ∝ (i+1)^(-p) (with a small grid of p values) give a good
   starting shape (they already produce a fairly flat self‑convolution).

2. **Continuously‑smoothed Adam optimisation** – we minimise a
   log‑sum‑exp surrogate of max(conv(a,a)).  The smoothing parameter α
   is increased from 0.5 to a large value (≈300) during the run,
   turning the surrogate gradually into the true objective.
   Adam’s adaptive step‑size together with a projection onto the
   probability simplex yields fast progress.

3. **Exact projected sub‑gradient + light annealing** – after the
   smooth phase we switch to the exact non‑smooth objective.
   Sub‑gradient descent with momentum and a 1/√t step‑size finishes the
   optimisation, and a final tiny simulated‑annealing polish moves a
   small amount of mass between two coordinates to escape possible
   plateaus.

All phases respect the global time budget (default 1000 s).
The routine returns a list of non‑negative floats; the harness will
feed this list to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 alpha_start: float = 0.5,
                 alpha_end: float = 300.0,
                 max_iters: int = 12000,
                 lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on f_α(a)= 1/α·log∑exp(α·conv(a,a)).
    α is increased linearly from ``alpha_start`` to ``alpha_end``.
    Returns the best iterate (lowest true objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    L_grad = _next_pow2(3 * n - 2)          # length for the gradient FFTs

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # true objective (no smoothing)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights (log‑sum‑exp derivative)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient of the smoothed objective
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 – exact sub‑gradient descent on 2·n·max(conv)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum β≈0.9, step ∝ 1/√t.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        # gradient of the active convolution entry
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – tiny simulated annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-6,
               steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing.  Temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportional to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score
