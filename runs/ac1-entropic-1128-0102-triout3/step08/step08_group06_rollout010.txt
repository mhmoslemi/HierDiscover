    v = np.zeros_like(a)
    t = 0

        conv = _conv_self(a)
        cur_score = 2.0*n*conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)
        # subgradient
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi+1)
            j = k - i
            g[i] += a[j]

"""
Propose a high‑quality step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  

The algorithm consists of four stages, each respecting the global time
budget (default 1000 s) and always keeping the best sequence seen so far.

1. **Warm start** – either up‑sample the global ``height_sequence_1`` or
   start from a random probability vector.
2. **Exponential‑gradient (mirrored) descent** on a log‑sum‑exp
   approximation of the maximal self‑convolution.  A schedule of smoothing
   parameters ``alpha`` is used; the update is performed on the probability
   simplex.
3. **Adam‑style projected sub‑gradient descent** on the *exact* maximal
   convolution entry.  The sub‑gradient is obtained from the active set
   (indices attaining the current maximum) and an Adam‑like adaptive step
   size is applied before projection onto the simplex.
4. **Aggressive local refinement** – a fast O(n) move which transfers a
   tiny amount of mass from a source index that contributes heavily to the
   current maximum towards a destination index that contributes little.
   The move size decays from ``init_eps_frac`` to ``min_eps_frac``.
   Afterwards a cheap random‑move stage is run to exploit any remaining
   improvement opportunities.

All heavy work (convolutions) uses FFTs for speed.  The best sequence found
throughout the run is normalised to sum = 1 and returned as a plain Python
list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negative values caused by round‑off.
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponential‑gradient on a smoothed max (mirror descent)
# ----------------------------------------------------------------------


def _run_exponential_gradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alpha_schedule,
    eta0: float = 0.5,
    steps_per_alpha: int = 5000,
):
    """
    Mirror‑descent on the log‑sum‑exp (soft‑max) of the self‑convolution.
    Returns the best vector found and its corresponding score
    (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)  # length sufficient for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                f"iter={it:,}, score={cur_score:.6f}"
            )

        # current smoothing parameter
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # soft‑max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient = 2 * a ⊛ w_rev   (cross‑correlation)
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative (mirror) update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= upd
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a[:] = 1.0 / n

        # advance smoothing schedule slowly
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Adam‑style projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------


def _run_subgradient_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    lr0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """
    Adam‑like adaptive projected sub‑gradient descent on the true maximal
    convolution entry.  Returns the best vector seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] adam‑subgrad improve: "
                f"iter={t:,}, score={cur_score:.6f}"
            )

        # sub‑gradient of the max (active set)
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # Adam update on the simplex
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate decay similar to 1/√t
        lr = lr0 / math.sqrt(t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : aggressive targeted refinement (fast, O(n))
# ----------------------------------------------------------------------


def _run_targeted_refine_aggressive(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    init_eps_frac: float = 0.2,
    min_eps_frac: float = 0.001,
):
    """
    Fast O(n) refinement that repeatedly moves a tiny fraction of probability
    mass away from indices that contribute most to the current maximal
    convolution entry.  The allowed move size decays linearly from
    `init_eps_frac` to `min_eps_frac`.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linear decay of move‑size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        # current maximiser
        k_max = int(conv.argmax())

        # ----- choose a source index, weighted by its contribution -----
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ----- choose a destination that contributes little to k_max -----
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # ----- propose a move -----
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # O(n) exact update of the convolution (linear + quadratic part)
        delta = np.zeros(conv_len, dtype=float)

        # linear contribution 2*eps*(a_{k-dst} - a_{k-src})
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a

        # quadratic correction (only three entries)
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] refine‑aggr improve: "
                    f"iter={it:,}, max={best_max:.8f}"
                )

        # occasional full recomputation to avoid accumulated rounding error
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Phase‑4 : cheap random local search (fallback)
# ----------------------------------------------------------------------


def _run_random_local_search(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    max_eps_frac: float = 0.02,
):
    """
    Very simple random move: pick two distinct indices, move a random
    fraction of the source mass to the destination and keep the move only
    if the maximal convolution entry strictly decreases.
    """
    a = a0.copy()
    conv = _conv_self(a)
    best_a = a.copy()
    best_max = conv.max()
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * max_eps_frac * rng.random()
        if eps <= 0.0:
            continue

        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()
        if new_max < best_max - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            best_max = new_max
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] rand‑local improve: "
                f"iter={it:,}, max={best_max:.8f}"
            )
        # occasional full recompute
        if it % 100000 == 0:
            conv = _conv_self(a)
            best_max = conv.max()
    return best_a

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Initialise – warm‑start from the global construction if possible.
    # ------------------------------------------------------------------
    final_n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            a0 = _upsample(starter, final_n)
        else:
            raise ValueError
    except Exception:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – exponential‑gradient warm‑up.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_time = min(20.0, 0.05 * remaining)
    phase1_deadline = time.time() + phase1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.5,
        steps_per_alpha=5000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(
            f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_warm:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑style projected sub‑gradient descent on the true max.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_time = max(30.0, 0.70 * remaining)  # keep most of the budget
        phase2_deadline = time.time() + phase2_time

        a_sub, score_sub = _run_subgradient_adam(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            lr0=0.12,
            beta1=0.9,
            beta2=0.999,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(
                f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – aggressive targeted refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_targeted_refine_aggressive(
            best_seq,
            final_n,
            deadline,
            start_time,
            init_eps_frac=0.2,
            min_eps_frac=0.001,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(
                f"[{time.time() - start_time:.2f}s] phase 3 improve: score={score_ref:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 4 – cheap random local search for any remaining slack.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_rand = _run_random_local_search(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_eps_frac=0.02,
        )
        conv_rand = _conv_self(a_rand)
        score_rand = 2.0 * final_n * conv_rand.max()
        if score_rand < best_score:
            best_score = score_rand
            best_seq = a_rand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] phase 4 improve: score={score_rand:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise final vector and return as a plain list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
