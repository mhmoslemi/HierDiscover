def _local_refine(a, deadline, start_time, max_iters=5000):
    n = len(a)
    conv = _conv_self(a)
    best_score = 2*n*conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # compute current max
        cur_max = conv.max()
        kmax = conv.argmax()
        # find contributions for this kmax
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = kmax - i_idx
        prod = a[i_idx] * a[j_idx]
        source = i_idx[prod.argmax()]  # source index with largest product
        # dest as index with minimum a (excluding source)
        # optional: random among low indices
        dest_candidates = np.where(np.arange(n) != source)[0]
        # Choose dest with smallest a

        eps = 0.2 * a[source]  # 20% moved

    i = np.arange(n)

    # linear increase to center then decrease

    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum > 0:

def _beta_start(n, alpha):
    # gamma pdf of Beta(alpha,alpha) discrete
    x = (np.arange(n) + 0.5) / n   # center of each bin

"""
A multi‑phase optimizer for the step‑function objective.

The sequence a = (a_0,…,a_{n‑1}) is treated as a probability vector
(sum = 1, a_i ≥ 0).  The objective is
    score(a) = 2·n·max_k (a*a)[k] .
The algorithm consists of

1. **Diverse deterministic / random starts** (uniform, triangular,
   beta‑shaped, Gaussian, exponential, Dirichlet).

2. **Exploration** – a few Adam iterations on a smooth approximation of
   the max‑convolution (soft‑max with a schedule of smoothing parameters).

3. **Polishing** – projected sub‑gradient descent on the true (non‑smooth)
   objective.

4. **High‑α Adam refinement** – the same Adam routine with very large
   smoothing parameters (α up to 1024) to push the solution closer to the
   true max.

5. **Local greedy refinement** – a fast, exact‑convolution update that
   moves a small fraction of the coefficient most responsible for the
   current peak towards the smallest coefficient, accepting only
   improvements.

All phases respect a global time budget (default 1000 s).  The final
sequence is normalized to sum 1 and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()
    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Start‑point generators
# ----------------------------------------------------------------------


def _uniform_start(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _triangular_start(n: int) -> np.ndarray:
    """Symmetric triangle: 0 at the ends, peak in the centre."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    a = np.where(i <= center, i / center, (n - 1 - i) / center)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform_start(n)


def _beta_start(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(α,α) shape.
    For α<1 the distribution is U‑shaped (large weight at the ends),
    for α>1 it is bell‑shaped.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n  # bin centres in (0,1)
    a = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform_start(n)


def _exponential_start(n: int, r: float = 0.985) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    center = (n - 1) / 2.0
    x = np.arange(n, dtype=float)
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return g / g.sum()


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform_start(n)


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   iters_per_alpha: int = 2500,
                   alpha_schedule: list = None,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the smoothed max‑convolution objective.
    Returns the best probability vector seen and its exact score.
    """
    if alpha_schedule is None:
        alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0]

    a = a0.copy()
    n = a.shape[0]
    # work in log‑space to keep positivity
    y = np.log(np.maximum(a, eps_log))

    fft_len = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max to retrieve a valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # evaluate smoothed objective
            conv = _conv_self(a)
            scaled = α * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α
            obj = 2.0 * n * f_val  # not used for optimisation, only for logging

            # gradient w.r.t. a
            grad_a = _grad_smooth_max(a, conv, α, fft_len) * (2.0 * n)

            # back‑prop to logits y (soft‑max Jacobian)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # periodic exact evaluation
            if t % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={α:.1f}, score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max_k (a*a)[k].
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # --- sub‑gradient (active set) --------------------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each term a_i * a_j contributes a_j to ∂b_k/∂a_i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        # momentum (Nesterov‑style) and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    # finally evaluate the best found vector
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, final_score

def _high_alpha_adam(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:

    """

    alpha_sched = [128, 256, 512, 1024]
    # fewer iterations per level than the main exploration; the high α

        deadline,

        alpha_schedule=alpha_sched,
    )

# Local greedy refinement (exact convolution updates)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a fraction of the coefficient that most contributes
    to the current convolution peak towards a small coefficient.
    The convolution is recomputed exactly after every accepted move.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)

    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        cur_max = conv.max()
        kmax = conv.argmax()

        # indices that participate in the peak kmax
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = kmax - i_idx
        prod = a[i_idx] * a[j_idx]
        src = i_idx[prod.argmax()]          # most responsible index

        # destination = smallest coefficient (excluding src)
        candidates = np.delete(np.arange(n), src)
        dst = candidates[a[candidates].argmin()]

        # move a modest fraction of the source mass
        eps = 0.10 * a[src]                  # 10 % of the source value
        if eps < 1e-14:
            break

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # recompute convolution exactly (FFT version)
        conv_new = _conv_self(a_new)
        new_max = conv_new.max()

        if new_max + 1e-12 < cur_max:        # strict improvement
            a = a_new
            conv = conv_new
            best_score = 2.0 * n * new_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] local refine improve: iter={it}, max={new_max:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    The routine employs several complementary heuristics and respects a
    1000 s time budget.
    """

    start_time = time.time()

    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    n = 4800

    # Build a collection of diverse starting points.
    # ------------------------------------------------------------------
    starts = []

    # 1) previous best construction (if available)
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    starts.append(_uniform_start(n))
    starts.append(_triangular_start(n))
    starts.append(_beta_start(n, 0.5))   # U‑shaped (α<1)
    starts.append(_beta_start(n, 0.75))
    starts.append(_beta_start(n, 1.5))   # bell‑shaped

    # 3) random Dirichlet draws (smooth & spiky)

        starts.append(_dirichlet_start(n, conc, rng))

    # 4) exponential and Gaussian profiles
    starts.append(_exponential_start(n, r=0.985))
    starts.append(_gaussian_start(n))

    # ------------------------------------------------------------------
    # Time allocation
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.30 * budget_s   # 30 % for cheap Adam
    polish_deadline  = deadline - 0.20 * budget_s    # leave 20 % for later phases

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Exploration phase – cheap Adam runs on each start.
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        if time.time() >= explore_deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] exploration start #{idx + 1}")

        cand_seq, cand_score = _adam_optimize(
            a0,
            explore_deadline,
            start_time,

        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (explore) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Polishing phase – projected sub‑gradient descent.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < polish_deadline:
        print(f"[{time.time() - start_time:6.2f}s] polishing phase start")
        polished_seq, polished_score = _subgrad_optimize(
            best_seq, polish_deadline, start_time, beta=0.95, step0=0.3
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (polish) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # High‑α Adam refinement – a few hundred iterations with α up to 1024.
    # ------------------------------------------------------------------
    refine_deadline = deadline - 0.1 * budget_s
    if best_seq is not None and time.time() < refine_deadline:
        print(f"[{time.time() - start_time:6.2f}s] high‑α Adam refinement start")
        refined_seq, refined_score = _high_alpha_adam(
            best_seq, refine_deadline, start_time
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (high‑α) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final local greedy refinement – exact convolution updates.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] final local refinement start")
        final_seq, final_score = _local_refine(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (local) "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (guard against numerical drift) and return.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
