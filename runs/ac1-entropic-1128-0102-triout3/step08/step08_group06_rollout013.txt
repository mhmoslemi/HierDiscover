"""
Propose a step‑function with a low self‑convolution bound.

The algorithm works in four stages:

1.  **Parameter sweep** – try a few analytically‑motivated families
    (exponential, power‑law and a U‑shape) and a few random starts.
    The best of these is kept as a warm‑start.

2.  **Exponential‑gradient (smooth max)** – mirror‑descent on a
    log‑sum‑exp approximation of the maximal convolution entry.
    The step is projected onto the probability simplex.

3.  **Projected sub‑gradient descent** – use the exact (non‑smoothed)
    max‑convolution and a momentum‑accelerated sub‑gradient method.

4.  **Annealed targeted refinement** – move a tiny amount of probability
    mass from indices that contribute most to the current maximal
    convolution entry to indices that contribute the least.
    A simple simulated‑annealing acceptance rule lets the search escape
    shallow local minima.

All phases respect the overall time budget (≈1000 s) and the best
sequence observed throughout the run is returned (already normalised to
sum = 1).  The code works with the optional global `height_sequence_1`
as a warm start but also works from scratch.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage 1 – quick parametric sweep / random warm‑starts
# ----------------------------------------------------------------------
def _initial_candidates(n: int, seed: int = 0):
    rng = np.random.default_rng(seed)

    candidates = []

    # 1. Global warm‑start if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            candidates.append(_upsample(starter, n))
    except Exception:
        pass

    # 2. Simple random Dirichlet
    candidates.append(rng.random(n))
    candidates.append(rng.random(n) ** 2)   # a bit more skewed
    candidates.append(rng.random(n) ** 0.5) # less skewed

    # 3. Exponential family a_i ∝ exp(λ·i)
    for lam in np.linspace(-4.0, 4.0, 9):
        a = np.exp(lam * np.arange(n))
        candidates.append(a)

    # 4. Power‑law family a_i ∝ (i+1)^p
    for p in np.linspace(-2.0, 2.0, 9):
        a = (np.arange(n) + 1) ** p
        candidates.append(a)

    # 5. U‑shape (large at the ends, small in the middle)
    i = np.arange(n, dtype=float)
    a = 1.0 / ((i + 1) * (n - i))
    candidates.append(a)

    # Normalise all candidates
    normalised = []
    for a in candidates:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)
        normalised.append(a)

    return normalised


def _bound_from_sequence(a: np.ndarray) -> float:
    """Return the objective 2·n·max(conv) for a normalised (sum=1) sequence."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2 * n * conv.max())


# ----------------------------------------------------------------------
# Stage 2 – exponential‑gradient on smooth max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 8000):
    """Mirror‑descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # Smoothed max – log‑sum‑exp weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Multiplicative / mirror‑descent update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – projected sub‑gradient descent on exact max
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 5_000_000):
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    n = a.shape[0]
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # Sub‑gradient of the max entry
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # Momentum & step‑size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 4 – annealed targeted refinement (fast, fine‑grained)
# ----------------------------------------------------------------------
def _run_annealed_refine(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         init_eps_frac: float = 0.05,
                         min_eps_frac: float = 0.001,
                         init_temp: float = 5e-6,
                         min_temp: float = 1e-8,
                         random_prob: float = 0.15):
    """
    Very fast O(n) refinement with a simple simulated‑annealing rule.
    Tiny mass is moved from indices that contribute most to the current
    maximal convolution entry to indices that contribute the least.
    Occasionally a purely random move is tried (probability `random_prob`).
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    n = a.shape[0]
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linear schedules for eps and temperature
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)

        eps_frac = init_eps_frac * (1.0 - frac) + min_eps_frac * frac
        temp = init_temp * (1.0 - frac) + min_temp * frac

        # --------------------------------------------------------------
        # choose source and destination
        # --------------------------------------------------------------
        if rng.random() < random_prob:
            # fully random move
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
        else:
            # targeted move – source proportional to its contribution to the max entry
            k_max = int(conv.argmax())
            i = np.arange(n)
            j = k_max - i
            valid = (j >= 0) & (j < n)
            contrib = np.zeros(n, dtype=float)
            contrib[valid] = a[i[valid]] * a[j[valid]]
            total_contrib = contrib.sum()
            if total_contrib > 0:
                prob_src = contrib / total_contrib
                src = rng.choice(n, p=prob_src)
            else:
                src = rng.choice(n, p=a / a.sum())

            # destination – prefer indices that *do not* participate in k_max
            dst_idx = np.arange(n)
            jdst = k_max - dst_idx
            valid_dst = (jdst >= 0) & (jdst < n)
            dest_weight = np.zeros(n, dtype=float)
            dest_weight[valid_dst] = a[jdst[valid_dst]]
            zero_mask = dest_weight < 1e-14
            if np.any(zero_mask):
                candidates = np.where(zero_mask)[0]
            else:
                candidates = np.where(dest_weight == dest_weight.min())[0]
            dst = rng.choice(candidates)

        if src == dst or a[src] < 1e-20:
            continue

        # --------------------------------------------------------------
        # propose a move
        # --------------------------------------------------------------
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution
        delta = np.zeros(conv_len, dtype=float)

        # linear part 2*eps*(a_{k-dst} - a_{k-src})
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a

        # quadratic eps² correction
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept if improvement or according to Metropolis rule
        accept = False
        if new_max < max_val - 1e-14:
            accept = True
        else:
            # simulated‑annealing acceptance
            prob = math.exp(-(new_max - max_val) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # every few thousand steps recompute conv from scratch – guard against drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    The routine follows a four‑phase scheme:
      1) cheap parametric sweep + random starts,
      2) exponential‑gradient on a smoothed max,
      3) projected sub‑gradient descent,
      4) annealed targeted refinement.
    The best sequence seen throughout the run is returned (normalised to have sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a moderate length – 4800 worked well before and is a good
    # trade‑off between quality and per‑iteration cost.
    # ------------------------------------------------------------------
    final_n = 4800

    # ------------------------------------------------------------------
    # Phase‑0 – initial candidates (parameter sweep + random)
    # ------------------------------------------------------------------
    init_candidates = _initial_candidates(final_n, seed=seed)
    best_seq = None
    best_score = float("inf")

    for a in init_candidates:
        # quick bound (no evaluate_sequence call) – all candidates are already normalised
        score = _bound_from_sequence(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
    print(f"[{time.time() - start_time:.2f}s] init best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient warm‑up
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase1_deadline = time.time() + min(60.0, 0.05 * (deadline - time.time()))
        a_warm, score_warm = _run_exponential_gradient(
            best_seq,
            phase1_deadline,
            start_time,
            alpha_schedule=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            eta0=0.5,
            steps_per_alpha=8000,
        )
        if score_warm < best_score:
            best_score = score_warm
            best_seq = a_warm.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact‑max projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(120.0, 0.70 * remaining)

            start_time,

    # ------------------------------------------------------------------
    # Phase 3 – annealed targeted refinement (fine‑grained)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_annealed_refine(
            best_seq,

            start_time,

            min_eps_frac=0.001,
            init_temp=5e-6,
            min_temp=1e-9,
            random_prob=0.12,
        )
        # compute the *exact* objective for the final candidate
        conv_ref = _conv_self(a_refined)
        score_ref = float(2.0 * final_n * conv_ref.max())
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return – everything is already non‑negative.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:

    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # final sanity check – call the official evaluator once (cheap)

    except Exception:
        # if the evaluator misbehaves we simply ignore – the sequence is still valid
        pass

    return [float(x) for x in best_seq]
