def _run_nesterov_subgrad(a0, n, deadline, start_time, step0=0.5):
    a = a0.copy()
    y = a.copy()
    t = 1.0
    best_a, best_score = a.copy(), 2.0 * n * _conv_self(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(y)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = y.copy()
        grad = _subgradient_max(y, conv)
        step = step0 / math.sqrt(it)
        a_next = _project_simplex(y - step * grad)
        t_next = (1 + math.sqrt(1 + 4 * t * t)) / 2
        y = a_next + ((t - 1) / t_next) * (a_next - a)
        a = a_next
        t = t_next
    return best_a, best_score
```

def _perturb(a, scale):
    n = a.size

    noise = np.random.randn(n) * scale

    a_pert = np.maximum(a_pert, 0)

        a_pert /= total
    else:
        a_pert = np.full_like(a_pert, 1.0 / n)

"""
Proposes a step‑function (non‑negative vector) that minimises
    2·n·max(conv(a)) / (sum(a)^2)

Algorithm (≈100 s total budget, works up to 1000 s):
1.  Initialise a small symmetric vector (size 1024) – either from the
    previously discovered construction `height_sequence_1` or from a
    random simplex point.
2.  A brief Adam optimisation on the smooth log‑sum‑exp surrogate
    `Oτ(a)=2·n·τ·log∑exp(conv/τ)` improves the shape.
3.  Upsample the vector to a larger size (8192) and symmetrise it.
4.  A long Adam run on the same surrogate (τ is driven to ~1e‑15) on
    the large size gives a very good approximation of the true
    max‑convolution objective.
5.  A Nesterov‑accelerated projected sub‑gradient descent on the
    true objective refines the solution.
6.  A light simulated‑annealing mass‑transfer finishes the search.
All iterates are projected onto the probability simplex and forced to
be symmetric (a_i = a_{n‑1‑i}), which is natural for the autocorrelation
problem and reduces the effective search space.
The function respects the overall time budget and prints intermediate
scores.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i≥0, Σx_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Symmetry helper
# ----------------------------------------------------------------------


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Return the symmetric version (v_i = v_{n-1-i}) of a vector."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for true objective)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp surrogate of max‑conv
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-15,
                      tau_decay: float = 0.9997,
                      step_size: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise the surrogate
        Oτ(a) = 2·n·τ·log∑exp(conv(a)/τ)
    using Adam and projection onto the simplex, with enforced symmetry.
    Returns the best vector (according to the true objective) and its
    true score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true score for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f} τ={tau:.2e}")

        # ---- soft‑max weights ----
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # shape 2·n‑1, sums to 1

        # ---- gradient of Oτ ----
        # grad_i = 4·n·∑_j a_j·p_{i+j}
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # required slice
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        # projection + symmetry
        a = _project_simplex(a)
        a = _symmetrize(a)
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Nesterov accelerated projected sub‑gradient descent (true objective)
# ----------------------------------------------------------------------


def _run_nesterov_subgrad(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Nesterov‑accelerated projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))   (a lies on the simplex and is kept symmetric).
    """
    a = a0.copy()
    y = a.copy()
    t = 1.0

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(y)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = y.copy()
            print(f"[{time.time() - start_time:6.2f}s] Nesterov improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(y, conv)
        step = step0 / math.sqrt(it)

        a_next = _project_simplex(y - step * grad)
        a_next = _symmetrize(a_next)
        a_next = _project_simplex(a_next)

        t_next = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_next + ((t - 1.0) / t_next) * (a_next - a)

        a = a_next
        t = t_next

    return best_a, best_score

# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve (n={n}) "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
