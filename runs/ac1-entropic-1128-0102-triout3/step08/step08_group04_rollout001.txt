"""
Algorithm to minimise ``evaluate_sequence``.

1️⃣  Initialise a non‑negative probability vector `a` of length `N`
    (default 7200).  If the global ``height_sequence_1`` exists we
    up‑sample it, otherwise we start from a random Dirichlet sample.

2️⃣  Warm‑up – exponentiated‑gradient on a smooth (soft‑max) surrogate
    of the max‑convolution.  This is the same idea as the reference code
    but we give it a slightly larger time budget and a few more steps.

3️⃣  Projected sub‑gradient descent on the *exact* max‑convolution to
    obtain a solid starting point.

4️⃣  Massive incremental refinement:
      • At each iteration move a fraction ε of mass from a “source’’ index
        that contributes heavily to a large convolution entry to a “dest’’
        index that contributes little.
      • ε is large at the beginning (≈30 % of the source weight) and
        shrinks linearly to ≈0.1 % at the end of the phase.
      • The peak index is sampled among the top‑K convolution entries
        (default K=5) so several peaks are flattened.
      • After a configurable number of successful moves we recompute the
        full convolution with FFT to avoid drift of the analytic update.
      • Only moves that *strictly* decrease the global maximum of the
        self‑convolution are kept (monotone descent).

5️⃣  The best sequence seen during the whole run is returned (normalised
    to unit sum).  All intermediate progress is printed so the user can
    see the bound decreasing.

The routine respects the supplied time budget (default 1000 s) and uses
only a single CPU core.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient warm‑up (soft‑max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 8000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.ifft(Af * Wrev)[: n + len(w_rev) - 1].real
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Phase‑3 : massive incremental refinement (improved)
# ----------------------------------------------------------------------
def _run_incremental_refine_improved(a0: np.ndarray,
                                     n: int,
                                     deadline: float,
                                     start_time: float,
                                     eps_start: float = 0.30,
                                     eps_end: float = 0.001,
                                     top_k: int = 5,
                                     recompute_every: int = 20000,
                                     rel_improve: float = 1e-12):
    """
    Aggressive flattening of the self‑convolution.
    * eps_start / eps_end define the shrinking fraction of the source mass.
    * top_k – number of largest convolution entries from which a peak is drawn.
    * recompute_every – full FFT recomputation after this many accepted moves.
    """
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    accepted = 0
    total_iters = 0
    start_refine = time.time()
    total_time = deadline - start_refine

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # ---- pick a peak index among the top‑k ----
        if top_k >= len(conv):
            cand_idxs = np.arange(len(conv))
        else:
            # argpartition gives unsorted top‑k; we keep them unsorted – fine.
            cand_idxs = np.argpartition(-conv, top_k - 1)[:top_k]
        k_max = int(rng.choice(cand_idxs))

        # ---- source selection (high contribution to that peak) ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = int(rng.choice(i_range, p=prob_src))
        else:
            # fallback – choose proportionally to a itself
            prob_src = a / a.sum()
            src = int(rng.choice(n, p=prob_src))

        if a[src] < 1e-15:
            continue

        # ---- destination selection (low contribution to that peak) ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                         # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = int(rng.choice(n, p=prob_dst))

        # ---- amount to move -------------------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0 or eps > a[src]:
            continue

        # ---- evaluate the move (analytic O(n) update) --------------------
        conv_new = conv.copy()
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        if new_max < max_val - rel_improve:
            # accept
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            accepted += 1
            total_iters += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] inc‑refine improve: "
                      f"accepted={accepted:,}, max={best_max:.8f}")

            # periodic full recomputation to avoid drift
            if accepted % recompute_every == 0:
                conv = _conv_self(a)
                max_val = conv.max()
        else:
            total_iters += 1

    return best_a

            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: "
                  f"score={score_sub:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – massive incremental local refinement (aggressive)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_incremental_refine_improved(
            best_seq,
            final_n,
            deadline,
            start_time,
            eps_start=0.30,      # start with moving up to 30 % of source mass
            eps_end=0.001,       # end with sub‑percent moves
            top_k=5,
            recompute_every=20000,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # Final log using the official evaluator (if desired)
    try:
        from __main__ import evaluate_sequence  # the platform provides it
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] final reported score: {final_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
