"""
Improved search for a non‑negative step function that minimises
evaluate_sequence.

Algorithm
---------
1. **Warm start** – take the previously discovered best sequence
   (if available) and normalise it to sum = 1.  Fall back to a uniform
   start.

2. **Smooth‑max Adam optimisation** – perform a long projected‑gradient
   descent on the smooth approximation
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
   with a schedule of increasing smoothing parameters α.
   Each iteration projects the iterate onto the probability simplex.
   This phase occupies the majority of the time budget and yields a
   very good candidate.

3. **Exact sub‑gradient polishing** – starting from the best Adam
   candidate, run a few thousand iterations of classic sub‑gradient
   descent on the true objective max_k (a*a)[k] (with momentum).
   The step‑size is much smaller than in the reference code, giving a
   finer finish.

4. **Fast greedy peak‑reduction** – keep the convolution array and
   perform cheap O(n) moves: a small fraction of the coefficient that
   contributes most to the current peak is transferred to the smallest
   coefficient.  The move is accepted only if it strictly lowers the
   peak.  Because the convolution is updated incrementally this loop
   can perform many thousands of iterations in the remaining seconds.

The routine respects the 1000 s time limit, prints progress messages
and finally returns the best sequence found (as a plain Python list).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max function
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # the centre of the correlation corresponds to indices 0..n-1
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – long Adam optimisation on the smoothed max
# ----------------------------------------------------------------------


def _adam_optimize_long(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed max function.
    Runs as many iterations as time permits.
    Returns the best vector (as a probability vector) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # pre‑compute FFT length for the gradient routine
    fft_len = _next_pow2(3 * n - 2)

    # schedule of smoothing parameters – the larger α, the closer to the true max
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        # a fixed α is used for many tiny Adam steps
        while time.time() < deadline:
            # ---- objective and gradient (smooth) ----
            conv = _conv_self(a)
            scaled = α * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α                     # smoothed max
            obj = 2.0 * n * f_val               # the quantity returned by evaluate_sequence

            # gradient w.r.t. a (including the outer factor 2·n)
            grad_a = _grad_smooth_max(a, conv, α, fft_len) * (2.0 * n)

            # Adam update on the raw vector (no logits); project after update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)                     # simple decay

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 500 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={α:.0f}) "
                          f"score={sc:.6f}")

            # safety break if we exhausted the allotted time for this α
            # (the outer while loop will also stop when the global deadline is reached)
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – classic sub‑gradient descent (shorter, fine‑grained)
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Fewer iterations than the reference implementation, but a much
    smaller initial step gives a finer final result.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # --- sub‑gradient (active set) --------------------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        # momentum + decreasing step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – fast greedy peak reduction using an incremental update
# ----------------------------------------------------------------------


def _fast_greedy_improve(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small fraction of the coefficient that most
    contributes to the current convolution peak towards the smallest
    coefficient.  The convolution array is updated incrementally
    (O(n) per move) which makes the loop very cheap.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = conv.argmax()
        cur_max = conv[k_max]

        # indices that participate in this peak
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # most responsible index

        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # transfer a modest fraction (decrease over time)
        eps = 0.05 * a[src]                     # 5 % of the source value
        if eps < 1e-14:
            break

        # ---- incremental convolution update ----
        # keep a copy of conv only if the move is accepted
        conv_tmp = conv.copy()
        # linear part: -2·eps·a shifted by src  + 2·eps·a shifted by dst
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        # quadratic part
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        if new_max < cur_max:                      # accept only strict improvements
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            best_score = 2.0 * n * new_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] fast‑greedy improve: iter={it:,}, max={new_max:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0,

    """

    ``evaluate_sequence``.  The routine respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
