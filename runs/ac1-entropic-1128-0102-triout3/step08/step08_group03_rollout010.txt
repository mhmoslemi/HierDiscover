def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, math, numpy as np
    try:
        from scipy.optimize import linprog
        SCIPY_AVAILABLE = True
    except Exception:
        SCIPY_AVAILABLE = False

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # parameters
    small_n = 600   # can adjust based on time
    large_n = 4800

    best_small = None
    best_M = float('inf')
    best_score = float('inf')
    rng = np.random.default_rng(seed)

    # Stage 1: LP iterative search (if SciPy)
    if SCIPY_AVAILABLE:
        stage1_deadline = start_time + 0.2 * budget_s
        # generate starts (Dirichlet)
        starts = []
        starts.append(np.full(small_n, 1.0/small_n))
        starts.append(_gaussian_start(small_n))
        for conc in [0.5, 1.0, 2.0, 5.0]:
            starts.append(_dirichlet_start(small_n, conc, rng))
        # also exponential
        starts.append(_exponential_start(small_n, r=0.98))

        for s_idx, a0 in enumerate(starts):
            if time.time() >= stage1_deadline:
                break
            a = a0.copy()
            # ensure positivity
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            print(f"[{time.time()-start_time:6.2f}s] LP start #{s_idx+1}")

            for it in range(15):
                if time.time() >= stage1_deadline:
                    break
                # compute convolution and M
                conv = _conv_self(a)
                M = conv.max()
                # solve LP
                b = _lp_optimize(a, M)
                if b is None or np.any(np.isnan(b)):
                    print("LP failed, break")
                    break
                sum_b = b.sum()
                if sum_b == 0:
                    break
                g = b / sum_b
                # blending
                t = 0.3
                a = (1 - t) * a + t * g
                # maintain positivity and renorm
                a = np.maximum(a, 1e-12)
                a /= a.sum()
                # evaluate current M
                conv_new = _conv_self(a)
                new_M = conv_new.max()
                # track best
                if new_M < best_M:
                    best_M = new_M
                    best_small = a.copy()
                    best_score = 2 * small_n * new_M  # sum=1, so evaluate = 2n*max_b
                    print(f"[{time.time()-start_time:6.2f}s] new best small M={new_M:.6f}, score={best_score:.6f}")

    # If no small best found, fallback to uniform
    if best_small is None:
        best_small = np.full(small_n, 1.0 / small_n)

    # Stage 2: Upsample to large_n
    a_big = _upsample(best_small, large_n)
    # Ensure positivity and renorm

"""
Multi‑stage optimizer for the step‑function attack.

1. **Coarse LP phase (small size)**
   – Work with a modest length (≈600) where a linear program
     `max Σ b_j  s.t.  f∗b ≤ M` can be solved quickly (SciPy hiGHS).
   – Starting from several random Dirichlet / Gaussian / exponential
     vectors we iterate:
        * compute M = max(f∗f)
        * solve the LP for b
        * normalise b and blend   f ← (1‑t)·f + t·b   (t≈0.3)
        * repeat 10‑15 times, keeping the best f (smallest M).

2. **Upsampling**
   – The best small‑size vector is linearly interpolated to the target
     length (default 4800) and re‑normalised to a probability vector.

3. **Smooth Adam exploration**
   – A few Adam steps on a soft‑max approximation of the max‑convolution
     (increasing smoothing α) quickly improve the large‑scale candidate.

4. **Projected sub‑gradient polishing**
   – True sub‑gradient descent on `max_k (a∗a)[k]` with momentum and a
     decreasing step size drives the peak down further.

5. **Greedy peak reduction**
   – Repeatedly shift a fraction of the coefficient that contributes most
     to the current peak towards the smallest coefficient, accepting only
     strict improvements.

All phases respect the overall time budget (default 1000 s).  The final
vector is clipped to non‑negative values, re‑normalised and returned as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector (positive) with given concentration."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    center = (n - 1) / 2.0
    x = np.arange(n)
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.98) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   global_deadline: float,
                   start_time: float,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    A few Adam iterations on the smoothed max‑convolution objective.
    Returns the best probability vector and its raw evaluate_sequence score.
    """
    a = a0.copy()
    n = a.shape[0]
    # initialise logits for the soft‑max parametrisation
    y = np.log(np.maximum(a, eps_log))

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – grows increasingly non‑smooth
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 1500

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= global_deadline:
                return best_a, best_score

            # re‑project onto simplex via soft‑max
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # smoothed objective
            conv = _conv_self(a)
            scaled = α * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α
            obj = 2.0 * n * f_val

            # gradient wrt a
            grad_a = _grad_smooth_max(a, conv, α, fft_len) * (2.0 * n)

            # back‑prop through soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on y
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={α:.1f}, score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on f(a)=max_k (a*a)[k] with Euclidean projection onto
    the probability simplex.  Returns best seen vector and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # active set (all indices whose conv value is within a tiny tolerance)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient (sum of gradients of all active quadratics)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each ordered pair (i,j) contributes a_j to gradient w.r.t a_i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        # momentum update (Nesterov‑style) and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy deterministic peak reduction
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Repeatedly shift a modest fraction of the coefficient that most contributes
    to the current maximum convolution peak towards the smallest coefficient.
    Moves are kept only if they strictly lower the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # move a quarter of the source weight
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (small size)
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray, M: float):
    """
    Solve  max Σ b_j   s.t.   (f * b)[k] ≤ M  for all k,
    with b_j ≥ 0.  Returns the optimal b (may be dense).  Uses SciPy hiGHS.
    """
    try:
        from scipy.optimize import linprog
    except Exception as e:
        print("SciPy linprog not available – skipping LP step.")
        return None

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix A_ub (rows × n).  Row k corresponds to
    # Σ_i f_i * b_{k-i} ≤ M.
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        A[k, j] = f[i]

    c = -np.ones(n)                # maximise Σ b_j  → minimise −Σ b_j
    bounds = [(0, None)] * n       # b_j ≥ 0
    ub = np.full(rows, M)

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    else:
        print("LP failed:", res.message if hasattr(res, "message") else "unknown")
        return None

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine proceeds through a small‑size LP phase, up‑sampling,
    a short Adam exploration, a long sub‑gradient polish and a final greedy
    refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    small_n = 600                 # size for the LP phase (fast)
    large_n = 4800                # final length (used by the original benchmark)

    # ------------------------------------------------------------------
    # Stage 1 – LP based shape discovery (if SciPy is present)
    # ------------------------------------------------------------------
    best_small = None
    best_M = float("inf")
    stage1_deadline = start_time + 0.20 * budget_s   # ~200 s for LP work

    # assemble a few diverse initial vectors (all positive, sum=1)
    starts = [
        np.full(small_n, 1.0 / small_n),          # uniform
        _gaussian_start(small_n),                # smooth bell
        _exponential_start(small_n, r=0.98),    # geometric decay
        _dirichlet_start(small_n, conc=0.5, rng=rng),
        _dirichlet_start(small_n, conc=2.0, rng=rng),
    ]

    # optional: use the existing construction if available
    if 'height_sequence_1' in globals():
        try:
            seq = np.asarray(height_sequence_1, dtype=float)
            if seq.ndim == 1 and seq.size > 0:
                seq = _upsample(seq, small_n)
                seq = np.maximum(seq, 1e-12)
                seq /= seq.sum()
                starts.append(seq)
        except Exception:
            pass

    for s_idx, a0 in enumerate(starts):
        if time.time() >= stage1_deadline:
            break
        a = a0.copy()
        # ensure strict positivity to keep LP bounded
        a = np.maximum(a, 1e-12)
        a /= a.sum()
        print(f"[{time.time() - start_time:6.2f}s] LP start #{s_idx + 1}")

        for it in range(12):                # limited number of LP iterations
            if time.time() >= stage1_deadline:
                break
            conv = _conv_self(a)
            M = conv.max()
            b = _lp_optimize(a, M)
            if b is None or np.isnan(b).any():
                break
            sum_b = b.sum()
            if sum_b == 0:
                break
            g = b / sum_b                     # normalise to a probability vector
            # blend with a modest mixing factor
            t = 0.3
            a = (1 - t) * a + t * g
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            # evaluate new max
            new_M = _conv_self(a).max()
            if new_M < best_M:
                best_M = new_M
                best_small = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] ** new best small M={new_M:.6f}")

    # fallback in case LP phase never produced a candidate
    if best_small is None:
        best_small = np.full(small_n, 1.0 / small_n)

    # ------------------------------------------------------------------
    # Stage 2 – upsample to the target size
    # ------------------------------------------------------------------
    a_big = _upsample(best_small, large_n)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    init_score = evaluate_sequence(a_big.tolist())
    print(f"[{time.time() - start_time:6.2f}s] upsampled to n={large_n}, init score={init_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – lightweight Adam exploration (≈30 % of total budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.35 * budget_s
    cand_seq, cand_score = _adam_optimize(a_big, explore_deadline, start_time)
    best_seq = cand_seq.copy()
    best_score = cand_score
    print(f"[{time.time() - start_time:6.2f}s] after Adam best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – projected sub‑gradient polishing (the bulk of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = deadline - 20.0   # keep ~20 s for the final greedy pass
        polished_seq, polished_score = _subgrad_optimize(
            best_seq, polish_deadline, start_time,
            beta=0.94, step0=0.3
        )
        best_seq = polished_seq.copy()
        best_score = polished_score
        print(f"[{time.time() - start_time:6.2f}s] after polishing best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – greedy peak reduction (quick deterministic clean‑up)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        best_seq = final_seq.copy()
        best_score = final_score
        print(f"[{time.time() - start_time:6.2f}s] after greedy best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and reporting
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    final_eval = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_eval:.6f}")

    # return a plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
