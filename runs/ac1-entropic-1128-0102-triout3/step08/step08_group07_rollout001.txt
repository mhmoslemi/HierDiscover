"""
Propose a low‑score non‑negative step function.

Algorithm (≈ 1000 s total):
  1. Small‑scale “Frank–Wolfe” iterations on n≈1024.
     – Solve a linear program that maximises Σ g_j under the
       constraint (f * g)_k ≤ max(f * f) for all k.
     – Mix the solution g with the current f (convex combination)
       and project onto the probability simplex.
  2. Upsample the best small‑scale solution to n≈4096 and run a
     long Adam optimisation on the smooth soft‑max surrogate of the
     max‑convolution.
  3. Upsample to the final size n≈8192 and polish with a
     Nesterov‑style projected sub‑gradient descent.
  4. Finish with an extended simulated‑annealing stage that moves
     mass between cells; a final short Adam run is applied after
     annealing for extra polishing.
The combination of a global LP‑based direction (stage 1) and the
gradient‑based refinements (stages 2‑4) improves the bound beyond the
reference implementation.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ------------------------------------------------------------
# Linear algebra helpers
# ------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ------------------------------------------------------------

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ------------------------------------------------------------

def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, sum fixed)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate O_τ(a)
# ------------------------------------------------------------

def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9994,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    with Adam and Euclidean projection onto the simplex.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true score for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # for stability
        p = exp_z / np.sum(exp_z)          # (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ------------------------------------------------------------

def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ------------------------------------------------------------

def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    The acceptance probability uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportionally to its weight
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ------------------------------------------------------------

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ------------------------------------------------------------
# Structured initialisation helpers (Chebyshev / Kaiser windows)
# ------------------------------------------------------------

def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    With probability 0.4 we produce a Chebyshev window,
    with probability 0.4 a Kaiser window,
    otherwise a plain random Dirichlet draw.
    """
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
        w = np.maximum(w, 0.0)
    elif have_signal and r < 0.8:
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
        w = np.maximum(w, 0.0)
    else:
        w = rng.random(n)
        w = np.maximum(w, 0.0)

    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# ------------------------------------------------------------
# Frank–Wolfe + LP helper (small‑scale)
# ------------------------------------------------------------

def _run_frank_wolfe(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     max_iters: int = 8,
                     alpha: float = 0.5) -> np.ndarray:
    """
    Perform a few Frank‑Wolfe style steps on a small problem (n≈1024):
    maximise Σ g_j  subject to (a * g)_k ≤ M  (M = max(a*a)),
    g ≥ 0.  The solution g is mixed with the current a.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        # SciPy not available – fall back to a plain copy
        print("SciPy not available – skipping Frank‑Wolfe stage.")
        return a0.copy()

    a = a0.copy()
    n = a.shape[0]
    # pre‑allocate the convolution matrix (dense – acceptable for n≈1024)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # column i is f shifted by i positions
    for i in range(n):
        A[i:i + n, i] = a[:]
    # Note: A will be rebuilt after each iteration because a changes.

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()

        # rebuild A for the current a
        for i in range(n):
            A[i:i + n, i] = a[:]

        # LP: maximise Σ g   → minimise -Σ g
        c = -np.ones(n, dtype=float)
        b_ub = np.full(rows, M, dtype=float)
        bounds = [(0.0, None) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"LP failed at FW iteration {it}, message: {res.message}")
            break

        g = res.x
        g_sum = g.sum()
        if g_sum > 0:
            g = g / g_sum
        else:
            g = np.full_like(g, 1.0 / n)

        # convex mix and project back onto simplex
        a = (1.0 - alpha) * a + alpha * g
        a = _project_simplex(a)

        # occasional logging
        if (it + 1) % 2 == 0:
            cur_score = 2.0 * n * _conv_self(a).max()
            print(f"[{time.time() - start_time:6.2f}s] FW iter {it+1}/{max_iters}, "
                  f"score={cur_score:.6f}")

    return a


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The pipeline mixes a small‑scale LP‑based Frank–Wolfe stage,
    up‑sampling, Adam optimisation on a soft‑max surrogate,
    sub‑gradient descent and a final simulated‑annealing polish.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------
    # Optional starter from a previous run
    # --------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # --------------------------------------------------------
    # Stage 0 – small‑scale Frank‑Wolfe (n≈1024)
    # --------------------------------------------------------
    n_small = 1024
    remaining = deadline - time.time()
    stage0_time = max(30.0, 0.12 * remaining)   # allocate ~12 % of the rest
    slice_deadline = time.time() + stage0_time

    max_restarts = 8
    for r in range(max_restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n_small)
        else:
            a0 = _window_initialiser(n_small, rng)

        a_opt = _run_frank_wolfe(a0, slice_deadline, start_time,
                                 max_iters=8, alpha=0.5)

        cur_score = 2.0 * n_small * _conv_self(a_opt).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={cur_score:.6f}")

    # fallback in the unlikely case the above never succeeded
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)
        best_score = 2.0 * n_small * _conv_self(best_seq).max()

    # --------------------------------------------------------
    # Stage 1 – upsample to medium size (n≈4096) and Adam on soft‑max
    # --------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_mid = 4096
    a_mid = _upsample(best_seq, n_mid)

    conv_mid = _conv_self(a_mid)
    tau0_mid = max(conv_mid.max() * 8.0, 1e-10)

    remaining = deadline - time.time()
    stage1_time = max(20.0, 0.25 * remaining)   # about 25 % of what's left
    slice_deadline = time.time() + stage1_time

    a_mid_opt, score_mid = _run_adam_softmax(
        a_mid, n_mid, slice_deadline, start_time,
        tau0=tau0_mid,
        tau_end=1e-14,
        tau_decay=0.9994,
        step_size=0.15,
    )
    if score_mid < best_score:
        best_score = score_mid
        best_seq = a_mid_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score_mid:.6f}")

    # --------------------------------------------------------
    # Stage 2 – upsample to final size (n≈8192) and sub‑gradient descent
    # --------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_big = 8192
    a_big = _upsample(best_seq, n_big)

    # sub‑gradient descent (leave a couple of seconds for final anneal)
    grad_deadline = deadline - 2.0
    a_big_opt, score_big = _run_subgradient_descent(
        a_big, n_big, grad_deadline, start_time,
        beta=0.97,
        step0=0.13,
    )
    if score_big < best_score:
        best_score = score_big
        best_seq = a_big_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score_big:.6f}")

    # --------------------------------------------------------
    # Stage 3 – extended simulated‑annealing refinement + final Adam polish
    # --------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=800_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={score_ref:.6f}")

    # One short Adam polish after annealing (if a little time remains)
    if time.time() + 5.0 < deadline:
        # brief warm‑up with a tiny τ to polish any remaining roughness
        conv_final = _conv_self(best_seq)
        tau0_final = max(conv_final.max() * 6.0, 1e-12)
        polishing_deadline = deadline - 0.5
        best_seq, _ = _run_adam_softmax(
            best_seq,
            best_seq.shape[0],
            polishing_deadline,
            start_time,
            tau0=tau0_final,
            tau_end=1e-15,
            tau_decay=0.9992,
            step_size=0.12,
        )

    # --------------------------------------------------------
    # Normalise onto the simplex one last time
    # --------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return plain Python list
    return [float(x) for x in best_seq]
