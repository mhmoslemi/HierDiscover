"""
Step‑function optimizer (LP + smooth descent).

The algorithm consists of three stages, all respecting a global time
budget (default 1000 s).

Stage 1 – Linear‑programming “mass‑inflation’’ step.
    Starting from a probability vector `f` (sum = 1) we solve the LP

        maximise   Σ g_i
        subject to  (f * g)[k] ≤ M   for every k,
                    g_i ≥ 0,

    where `M = max(f * f)`.  
    If the optimum Σ g_i is larger than 1 we can scale the solution
    down to a probability vector and obtain a new `f` with a strictly
    smaller self‑convolution bound.  This is the exact analogue of the
    linear‑programming step described in the statement, but it is coded
    from scratch using `scipy.optimize.linprog`.

Stage 2 – Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
    Using the same tools as the reference implementation we run a few
    Adam iterations that minimise

        O_τ(f) = 2·n·τ·log Σ exp(conv(f)/τ)

    while projecting onto the simplex after each step.  The temperature
    τ is decreased geometrically, turning the surrogate gradually into
    the true `max` operator.

Stage 3 – Short Euclidean projected sub‑gradient descent (heavy lifting)
    and finally a tiny simulated‑annealing mass‑transfer that tries to
    escape the last local traps.

The whole pipeline works for any length `n`.  In practice we use
`n = 2048` (the same size that the reference code reached before the
time out) because the LP constraints become too large for `n = 4096`
on a single CPU.  All vectors are kept on the probability simplex,
so the search is invariant to the overall scale of the step‑function.

The function `propose_candidate` returns a plain Python list of non‑negative
floats ready to be fed to `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x_i≥0, Σx_i=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using a real FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Linear‑programming mass‑inflation step
# ----------------------------------------------------------------------


def _lp_mass_inflation(f: np.ndarray,
                       M: float,
                       active_ratio: float = 0.90,
                       min_active: int = 200) -> np.ndarray | None:
    """
    Solve the LP that maximises total mass of g under the constraint
    (f * g)[k] ≤ M.  Only the most active constraints are kept
    (those where conv(f,f) is close to M) to keep the problem small.
    Returns the optimal g if Σg > 1 (i.e. an improving direction)
    otherwise returns None.
    """
    n = f.shape[0]

    # ---- active set of constraints ---------------------------------
    conv_ff = _conv_self(f)
    thresh = max(M * active_ratio, M - 1e-12)
    active = np.where(conv_ff >= thresh)[0]
    if active.size < min_active:
        # ensure we have enough constraints for a well‑posed LP
        # take the largest values even if they are slightly below the thresh
        sorted_idx = np.argsort(conv_ff)[::-1]
        active = sorted_idx[:min_active]

    # ---- build A_ub, b_ub -----------------------------------------
    rows = []
    for k in active:
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # the slice of f that contributes to conv at index k
        row[i_start:i_end + 1] = f[k - np.arange(i_start, i_end + 1)]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M, dtype=float)

    # ---- linear program (maximise sum(g)  <=> minimise -sum(g)) ---
    c = -np.ones(n)                      # minimise -Σg_i
    bounds = [(0.0, None)] * n           # g_i ≥ 0

    # Using HiGHS (the default for recent SciPy) – fast for moderate sizes.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None

    g = res.x
    total = g.sum()
    if total <= 1.0 + 1e-9:          # no improvement possible
        return None
    # Scale to a probability vector (sum = 1)
    g = g / total
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.25,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        Oτ(a) = 2·n·τ·log∑exp(conv(a)/τ)
    using Adam and projection onto the simplex.
    Returns the best vector (according to the true objective) and its
    true score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f} τ={tau:.2e}")

        # ---- soft‑max weights ----
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                       # length 2n‑1, sums to 1

        # ---- gradient of Oτ ----
        a_rev = a[::-1]
        conv_rev = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(p), n=_next_pow2(3 * n - 2))
        inner = conv_rev[n - 1:n - 1 + n]          # slice of length n
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        F(a)=2·n·max(conv(a))   (a lies on the simplex).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,

                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve (n={n}) "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine combines a linear‑programming
    “mass‑inflation’’ step, a smooth Adam phase and a final sub‑gradient
    descent + simulated annealing.  All work respects the given time
    budget (default 1000 s).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise a probability vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            starter = np.asarray(globals()["height_sequence_1"], dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                init = starter / starter.sum()
        except Exception:
            init = None
    else:
        init = None

    n = 2048  # fixed size for the LP stage (larger would be too heavy)

    if init is not None and init.size == n:
        a = init.copy()
    elif init is not None:
        # linearly interpolate any provided seed to length n
        old_x = np.linspace(0.0, 1.0, init.size)
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, init)
        a = _project_simplex(a)
    else:
        a = np.random.rand(n)
        a = _project_simplex(a)

    best_seq = a.copy()
    best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 2.  LP‑mass‑inflation phase (aggressive early improvement)
    # ------------------------------------------------------------------
    lp_phase_deadline = start_time + (deadline - start_time) * 0.30
    lp_iter = 0
    while time.time() < lp_phase_deadline:
        lp_iter += 1
        conv = _conv_self(a)
        M = conv.max()
        g = _lp_mass_inflation(a, M,
                               active_ratio=0.92,
                               min_active=250)
        if g is None:
            # no improving LP direction found; break early
            break
        # Blend the new distribution with a (tuned step size)
        t = 0.3
        a = (1.0 - t) * a + t * g
        a = _project_simplex(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve (iter={lp_iter}) "
                  f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Adam on the smooth surrogate (soft‑max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        conv0 = _conv_self(a)
        tau0 = max(conv0.max() * 8.0, 1e-8)
        remaining = deadline - time.time()
        # allocate roughly half of the remaining time to Adam
        adam_deadline = time.time() + remaining * 0.45
        a, _ = _run_adam_softmax(
            a0=a,
            n=n,
            deadline=adam_deadline,
            start_time=start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9995,
            step_size=0.30,
        )
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam final improve: "
                  f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Projected sub‑gradient descent (heavy lifting)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = deadline - 2.0  # keep a couple of seconds for annealing
        a, _ = _run_subgradient_descent(
            a0=best_seq,
            n=n,
            deadline=sub_deadline,
            start_time=start_time,
            beta=0.95,
            step0=0.30,
        )
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Sub‑grad final improve: "
                  f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Light simulated‑annealing refinement (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a, _ = _run_annealing(
            a0=best_seq,
            n=n,
            deadline=deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=200_000,
        )
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Annealing final improve: "
                  f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (extra safety) and return.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
