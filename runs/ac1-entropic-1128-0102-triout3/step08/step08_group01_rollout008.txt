def _lbfgs_obj_grad(x, alpha, n, L):
    # Softmax conversion
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    a = exp_x / exp_x.sum()
    # Compute FFT of a
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    # Smoothed max
    scaled = alpha * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    # Objective
    f = 2.0 * n * (max_scaled + np.log(sum_exp)) / alpha
    # Compute w
    w = exp_scaled / sum_exp
    # Gradient with respect to a

    Wrev = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wrev, n=L)

    # gradient to x via softmax
    dot = np.dot(a, grad_J)
    grad_x = a * (grad_J - dot)
    return f, grad_x

"""
Propose candidate for the step‑function optimisation.

Algorithm (≈1 000 s budget)

1. Initialise a probability vector on the simplex (use the global
   ``height_sequence_1`` if available, otherwise random).

2. **L‑BFGS on a smooth approximation of the max‑convolution**
   – Optimise the soft‑max of the self‑convolution via a continuation
   scheme over the smoothing parameter α.
   – Variables are the logits of a softmax (no simplex constraints needed).
   – Gradient is obtained analytically using FFT‑based convolutions.
   – After each α‐stage the true objective 2·n·max(conv) is evaluated
   and the best sequence is retained.

3. **Projected sub‑gradient descent** (as in the previous optimiser)
   – Further reduces the true max‑convolution while keeping the vector
   on the probability simplex.

4. **Targeted simulated annealing** – Moves mass away from the pairs that
   contribute most to the current peak.

5. **Random restarts** – Small perturbations of the current best are
   re‑optimised with a short L‑BFGS + annealing run.

All helper functions are top‑level (no closures).  The routine respects the
time budget and returns a list of non‑negative heights that sum to one.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# L‑BFGS smoothing optimisation
# ----------------------------------------------------------------------
def _lbfgs_obj_grad(x: np.ndarray, alpha: float, n: int,
                    Lc: int, Lcorr: int) -> tuple[float, np.ndarray]:
    """
    Objective and gradient for L‑BFGS.
    x  – logits (unconstrained)
    α  – smoothing parameter
    n  – length of the step function
    Lc – FFT length for the self‑convolution (≥2·n‑1)
    Lcorr – FFT length for the correlation (≥ n + 2·n‑2)
    Returns (f, grad) where f is the smooth approximation
    of 2·n·max(conv) and grad is ∂f/∂x.
    """
    # ----- soft‑max to obtain a on the simplex -----
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    a = exp_x / exp_x.sum()

    # ----- self‑convolution (exact) -----
    Af = np.fft.rfft(a, n=Lc)
    conv = np.fft.irfft(Af * Af, n=Lc)[: 2 * n - 1]

    # ----- smooth max via log‑sum‑exp -----
    scaled = alpha * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    # f = 2·n·max_smooth,  max_smooth = (max_scaled + log(sum_exp)) / α
    f = 2.0 * n * (max_scaled + np.log(sum_exp)) / alpha

    # ----- gradient w.r.t a (smooth max part) -----
    w = exp_scaled / sum_exp                     # soft‑max weights
    w_rev = w[::-1]

    # correlation a * w_rev (linear convolution)
    Af_big = np.fft.rfft(a, n=Lcorr)
    Wrev_big = np.fft.rfft(w_rev, n=Lcorr)
    corr = np.fft.irfft(Af_big * Wrev_big, n=Lcorr)

    offset = len(conv) - 1                       # = 2·n‑2
    idx = offset - np.arange(n)                  # indices for gradient
    grad_max = 2.0 * corr[idx]                    # ∂max_smooth/∂a

    # gradient of the true objective J = 2·n·max_smooth
    grad_J = 2.0 * n * grad_max

    # ----- back‑propagate through the soft‑max (∂x) -----
    dot = np.dot(a, grad_J)
    grad_x = a * (grad_J - dot)                  # shape (n,)

    return f, grad_x


def _run_lbfgs_smoothing(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         alphas,
                         max_iter_per_alpha: int = 800) -> tuple[np.ndarray, float]:
    """
    Continuation L‑BFGS on the smooth max‑convolution.
    Returns the best sequence (lowest true objective) found.
    """
    Lc = _next_pow2(2 * n - 1)               # for self‑convolution
    Lcorr = _next_pow2(3 * n - 2)            # for correlation a * w_rev

    # initialise logits from the starting probability vector
    x = np.log(a0 + 1e-12)

    best_a = a0.copy()
    conv = _conv_self(best_a, Lc)
    best_score = 2.0 * n * conv.max()

    for alpha in alphas:
        if time.time() >= deadline:
            break

        # minimise the smooth objective with L‑BFGS‑B
        args = (alpha, n, Lc, Lcorr)
        options = {'maxiter': max_iter_per_alpha, 'disp': False}
        res = minimize(_lbfgs_obj_grad,
                       x,
                       args=args,
                       method='L-BFGS-B',
                       jac=True,
                       options=options)

        x = res.x

        # rebuild the probability vector
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # evaluate the true (non‑smoothed) objective
        conv = _conv_self(a, Lc)
        score = 2.0 * n * conv.max()

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] L‑BFGS improve (α={alpha}):"
                  f" score={score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Gradient is taken at the current maximal index.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient from the (first) maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – Targeted simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-8,
                            steps: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Annealing that always picks the current maximal convolution index
    and moves a small amount of probability mass away from the pair
    that contributes most to that peak.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase1_deadline = time.time() + max(5.0, 0.12 * remaining)

        alphas = [0.5, 2.0, 5.0, 15.0, 40.0]
        a_lbfgs, score_lbfgs = _run_lbfgs_smoothing(
            best_seq,
            final_n,
            phase1_deadline,
            start_time,
            alphas,
            max_iter_per_alpha=800,
        )
        if score_lbfgs < best_score:
            best_score = score_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 1 (L‑BFGS) improve: "
                  f"score={score_lbfgs:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (major chunk)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.70 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: "
                  f"score={score_sub:.6f}")

        # a second brief sub‑gradient run with a larger initial step
        if time.time() < deadline:
            extra_deadline = time.time() + min(8.0, deadline - time.time())
            a_sub2, score_sub2 = _run_subgradient_descent(
                best_seq,
                final_n,
                extra_deadline,
                start_time,
                best_score,
                max_iters=200_000,
            )
            if score_sub2 < best_score:
                best_score = score_sub2
                best_seq = a_sub2.copy()
                print(f"[{time.time() - start_time:.2f}s] extra sub‑grad improve: "
                      f"score={score_sub2:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted annealing (use the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_targeted_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=150_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Optional random restarts with a tiny L‑BFGS + anneal budget
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        block = min(30.0, time_left / 2.0)
        restart_deadline = time.time() + block

        # perturb current best (Gaussian jitter)
        pert = best_seq + 0.02 * np.random.randn(final_n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # short L‑BFGS to re‑focus
        pert_seq, pert_score = _run_lbfgs_smoothing(
            pert,
            final_n,
            restart_deadline,
            start_time,
            alphas=[0.5, 2.0, 5.0],
            max_iter_per_alpha=300,
        )

        # short anneal finish
        if time.time() < deadline:
            pert_seq, pert_score = _run_targeted_annealing(
                pert_seq,
                final_n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-8,
                steps=80_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: "
                  f"score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] finished – final score: "
          f"{2.0 * final_n * _conv_self(best_seq, _next_pow2(2 * final_n - 1)).max():.6f}")

    return [float(x) for x in best_seq]
