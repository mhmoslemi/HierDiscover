# -*- coding: utf-8 -*-
"""
Improved step‑function search.

The algorithm follows a multi‑stage pipeline:
  1. Many cheap restarts with Adam on a smooth soft‑max surrogate.
  2. Progressive up‑sampling to a larger length and a second Adam run.
  3. Projected Nesterov sub‑gradient descent on the true max‑convolution.
  4. An extended simulated‑annealing polishing phase.
  5. A short post‑processing phase that tries a few simple reorderings
     (sorted, reversed, tiny smoothing) and runs a final few sub‑gradient
     steps on the best of them.

All stages are time‑aware; the whole routine stops a few hundred
milliseconds before the global budget expires and returns the best
sequence found.  The returned list satisfies the non‑negativity and
simplicity requirements of ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, sum fixed)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9994,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    with Adam and Euclidean projection onto the simplex.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true (unsmoothed) score for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # for numerical stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    The acceptance probability uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisation helpers (Chebyshev / Kaiser / Beta)
# ----------------------------------------------------------------------
def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    With probability 0.4 we produce a Chebyshev window,
    with probability 0.4 a Kaiser window,
    otherwise a plain random Dirichlet draw.
    """
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
        w = np.maximum(w, 0.0)
    elif have_signal and r < 0.8:
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
        w = np.maximum(w, 0.0)
    else:
        w = rng.random(n)
        w = np.maximum(w, 0.0)

    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _beta_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a discretised Beta(α,β) density on n points,
    with α,β sampled uniformly in a modest range.
    """
    a = rng.uniform(0.2, 5.0)
    b = rng.uniform(0.2, 5.0)
    xs = np.linspace(0.0, 1.0, n)
    # Beta pdf (unnormalised)
    w = np.power(xs, a - 1) * np.power(1.0 - xs, b - 1)
    w = np.nan_to_num(w, posinf=0.0, neginf=0.0)
    w += 1e-12   # avoid exact zeros that would break the simplex projection
    w /= w.sum()
    return w.astype(float)

# Post‑processing helpers: reordering + tiny smoothing

def _postprocess_candidates(base_seq: np.ndarray, n: int, start_time: float,

    candidates.append(np.sort(base_seq))

    candidates.append(np.sort(base_seq)[::-1])

    kernel = np.array([0.25, 0.5, 0.25])

    smooth = np.maximum(smooth, 0.0)
    smooth /= smooth.sum()
    candidates.append(smooth)

    # reverse of the smoothed version
    candidates.append(smooth[::-1])

    # evaluate each candidate with the official function (may be slightly slower)
    best = base_seq
    best_score = float("inf")
    for cand in candidates:

        cand_sum = cand.sum()
        if cand_sum == 0:

        cand = cand / cand_sum

All stages are time‑aware; the whole routine stops a few hundred
milliseconds before the global budget expires and returns the best
sequence found.  The returned list satisfies the non‑negativity and
simplicity requirements of ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, sum fixed)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9994,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    with Adam and Euclidean projection onto the simplex.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true (unsmoothed) score for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # for numerical stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    The acceptance probability uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisation helpers (Chebyshev / Kaiser / Beta)
# ----------------------------------------------------------------------
def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    With probability 0.4 we produce a Chebyshev window,
    with probability 0.4 a Kaiser window,
    otherwise a plain random Dirichlet draw.
    """
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
        w = np.maximum(w, 0.0)
    elif have_signal and r < 0.8:
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
        w = np.maximum(w, 0.0)
    else:
        w = rng.random(n)
        w = np.maximum(w, 0.0)

    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _beta_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a discretised Beta(α,β) density on n points,
    with α,β sampled uniformly in a modest range.
    """
    a = rng.uniform(0.2, 5.0)
    b = rng.uniform(0.2, 5.0)
    xs = np.linspace(0.0, 1.0, n)
    w = np.power(xs, a - 1) * np.power(1.0 - xs, b - 1)
    w = np.nan_to_num(w, posinf=0.0, neginf=0.0)
    w += 1e-12
    w /= w.sum()
    return w.astype(float)


# ----------------------------------------------------------------------
# Post‑processing helpers: reordering + tiny smoothing
# ----------------------------------------------------------------------
def _postprocess_candidates(base_seq: np.ndarray, n: int, start_time: float,
                            deadline: float) -> np.ndarray:
    """
    Try a handful of cheap transformations (sorted, reversed, mild smoothing)
    and keep the one with the lowest true objective.  Afterwards a very short
    sub‑gradient polish is performed.
    """
    candidates = []

    # original sequence
    candidates.append(base_seq.copy())

    # sorted ascending (minimises a·J·a)
    candidates.append(np.sort(base_seq))

    # sorted descending
    candidates.append(np.sort(base_seq)[::-1])

    # tiny 3‑point moving‑average smoothing
    kernel = np.array([0.25, 0.5, 0.25])
    smooth = np.convolve(base_seq, kernel, mode="same")
    smooth = np.maximum(smooth, 0.0)
    smooth /= smooth.sum()
    candidates.append(smooth)
    candidates.append(smooth[::-1])

    best_seq = base_seq
    best_score = float("inf")

    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0:
            continue
        cand /= cand.sum()
        score = evaluate_sequence(cand.tolist())
        print(f"[{time.time() - start_time:6.2f}s] postprocess candidate score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    # final ultra‑short sub‑gradient polish
    if time.time() < deadline:
        a_opt, _ = _run_subgradient_descent(
            best_seq, n, deadline, start_time,
            beta=0.97, step0=0.08, max_iter=200_000
        )
        cand_score = evaluate_sequence(a_opt.tolist())
        print(f"[{time.time() - start_time:6.2f}s] final polish score={cand_score:.6f}")
        if cand_score < best_score:
            best_seq = a_opt.copy()
            best_score = cand_score

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The pipeline is:
      – many cheap Adam restarts on a moderate size,
      – up‑sample to a larger size and run Adam again,
      – a focussed Nesterov sub‑gradient phase,
      – an extended simulated‑annealing polishing phase,
      – a short post‑processing that tests sorted / smoothed variants
        and performs a final quick polish.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous solution exists, use it as a starter (normalised)
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a moderate size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    remaining = deadline - time.time()
    stage0_time = max(20.0, 0.12 * remaining)
    slice_deadline = time.time() + stage0_time

    max_restarts = 30
    for r in range(max_restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            if rng.random() < 0.6:
                a0 = _window_initialiser(n0, rng)
            else:
                a0 = _beta_initialiser(n0, rng)

        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 6.0, 1e-8)

        a_opt, score = _run_adam_softmax(
            a0, n0, slice_deadline, start_time,
            tau0=tau0,
            tau_end=1e-14,
            tau_decay=0.9996,
            step_size=0.20,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: restart={r}, score={score:.6f}")

    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – upscale to a larger size (≈8192) and continue Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    a1 = _upsample(best_seq, n1)
    conv1 = _conv_self(a1)
    tau0_1 = max(conv1.max() * 6.0, 1e-10)

    remaining = deadline - time.time()
    stage1_time = max(30.0, 0.25 * remaining)
    slice_deadline = time.time() + stage1_time

    a1_opt, score1 = _run_adam_softmax(
        a1, n1, slice_deadline, start_time,
        tau0=tau0_1,
        tau_end=1e-15,
        tau_decay=0.9995,
        step_size=0.15,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – aggressive sub‑gradient descent (still on n1)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        grad_deadline = deadline - 3.0   # keep a few seconds for later stages
        a2_opt, score2 = _run_subgradient_descent(
            best_seq, n1, grad_deadline, start_time,
            beta=0.97,
            step0=0.10,
        )
        if score2 < best_score:
            best_score = score2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – extended simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=1_000_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – cheap post‑processing (reordering + tiny smoothing) + final polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq = _postprocess_candidates(best_seq, best_seq.shape[0],
                                          start_time, deadline)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] final postprocess score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise to the simplex (guard against tiny numerical drift)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list – the harness will call evaluate_sequence
    return [float(x) for x in best_seq]
