def differential_evolution(m, blocksize, n, budget_end):

    popsize = 30

    init_pop = np.random.dirichlet(np.ones(m), size=popsize) # each row sum 1.

    scores = np.empty(popsize)

    for i in range(popsize):
        a = np.repeat(init_pop[i], blocksize) / blocksize
        conv = _conv_self(a)
        scores[i] = 2 * n * conv.max()

    best_idx = scores.argmin()
    best_w = init_pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < budget_end:
        for i in range(popsize):
            # select distinct r1,r2,r3 indices not equal i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, size=3, replace=False)
            mutant = init_pop[r1] + F * (init_pop[r2] - init_pop[r3])
            # clip
            mutant = np.maximum(mutant, 0.0)
            # project onto simplex
            mutant = _project_simplex(mutant)

            # crossover
            crossover_mask = np.random.rand(m) < Cr
            # ensure at least one dimension from mutant
            if not np.any(crossover_mask):
                crossover_mask[np.random.randint(m)] = True
            trial = np.where(crossover_mask, mutant, init_pop[i])
            trial = _project_simplex(trial)  # ensures sum=1

            # evaluate trial
            a_trial = np.repeat(trial, blocksize) / blocksize
            conv = _conv_self(a_trial)
            trial_score = 2 * n * conv.max()
            if trial_score < scores[i]:
                init_pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_w = trial.copy()
                    print(f"[{time.time()-start_time:.2f}s] DE new best score {best_score:.6f}")
            # optionally break earlier if time
            if time.time() >= budget_end:
                break

    return best_w, best_score

"""
Hybrid optimiser for ``evaluate_sequence``.

The algorithm works in three stages:

1️⃣  Coarse global search – a simple Differential‑Evolution (DE) runs on a
    low‑dimensional “block” representation of the step‑function.  Each block
    contains the same height, the blocks are projected onto the probability
    simplex, and the DE loop is limited only by the global time budget.

2️⃣  Smooth local refinement – the best candidate from DE is expanded to the
    full resolution (4096 points) and optimised with an Adam optimiser applied
    to a smooth log‑sum‑exp surrogate of the maximal convolution.  The smoothing
    parameter α is increased gradually (continuation).

3️⃣  Exact polishing – finally a projected sub‑gradient descent and a light
    simulated‑annealing polish are performed on the true objective.

All phases respect the 1000 s wall‑time limit and always return the best
sequence seen.  The returned list is a non‑negative step‑function normalised
to sum 1, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective: 2·n·max(conv(a,a))  (sum(a) is assumed = 1).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Phase‑1 : Differential evolution on a block‑wise representation
# ----------------------------------------------------------------------
def _differential_evolution(m: int,
                            blocksize: int,
                            n: int,
                            start_time: float,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    DE on the simplex of dimension ``m`` (each dimension is a block weight).
    The full sequence is obtained by repeating each weight ``blocksize`` times.
    Returns the best block‑vector and its objective value.
    """
    popsize = 30
    F = 0.8
    Cr = 0.9
    rng = np.random.default_rng()

    # initialise population on the simplex
    pop = rng.dirichlet(np.ones(m), size=popsize)   # shape (popsize,m)
    scores = np.empty(popsize, dtype=float)

    # evaluate initial population
    for i in range(popsize):
        a = np.repeat(pop[i], blocksize) / blocksize
        scores[i] = _objective(a)

    best_idx = int(np.argmin(scores))
    best_w = pop[best_idx].copy()
    best_score = float(scores[best_idx])
    print(f"[{time.time() - start_time:.2f}s] DE init best={best_score:.6f}")

    # main DE loop – stop when time expires
    while time.time() < deadline:
        for i in range(popsize):
            # choose three distinct individuals different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)
            mutant = _project_simplex(mutant)

            # binomial crossover
            cross_mask = rng.random(m) < Cr
            if not np.any(cross_mask):
                cross_mask[rng.integers(m)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = _project_simplex(trial)

            a_trial = np.repeat(trial, blocksize) / blocksize
            trial_score = _objective(a_trial)

            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = float(trial_score)
                    best_w = trial.copy()
                    print(f"[{time.time() - start_time:.2f}s] DE new best={best_score:.6f}")

            # early exit if we run out of time
            if time.time() >= deadline:
                break

    return best_w, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 alpha_start: float = 0.5,
                 alpha_end: float = 2000.0,
                 max_iters: int = 20000,
                 lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on f_α(a)= 1/α·log∑exp(α·conv(a,a)).
    α is increased linearly from ``alpha_start`` to ``alpha_end``.
    Returns the best iterate (lowest true objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    L_grad = _next_pow2(3 * n - 2)          # length for the gradient FFTs

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current true objective (used only for bookkeeping)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # smoothing parameter schedule
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # smooth‑max weights
        conv = _conv_self(a)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient of the smoothed objective
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # stay on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum with β≈0.9 and step size 0.4/√t are used.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.4
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 : Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-6,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealing. Temperature decays exponentially.
    """
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    n = a.shape[0]

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)          # source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30   # move up to 30 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score
