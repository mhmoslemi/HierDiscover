def annealing_refine(a, conv, max_val, n, deadline, start_time):
    best_a = a.copy()
    best_max = max_val
    # parameters
    T0 = max_val * 0.1   # start T maybe 10% of max_val? Actually we want small.
    eps_max_frac_initial = 0.2
    eps_max_frac_final = 0.01
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        fraction = elapsed / total
        T = T0 * (1 - fraction)  # linear cooling
        eps_frac = eps_max_frac_initial * (1 - fraction) + eps_max_frac_final * fraction
        
        # choose src
        kmax = int(conv.argmax())
        i = np.arange(n)
        j = kmax - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        if contrib.sum() > 0:
            src = np.random.choice(n, p=contrib/ contrib.sum())
        else:
            src = np.random.choice(n, p=a)
        # choose dest with low contribution
        jdst = kmax - np.arange(n)
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n)
        dest_weight[valid_dst] = a[jdst[valid_dst]]

        else:
            candidates = np.where(dest_weight == dest_weight.min())[0]

        if src == dst: continue

        if eps <= 0: continue

        conv_new[2*src] += eps*eps

        a_refined = _run_targeted_refine_optimized(best_seq, n, deadline, start, init_eps_frac=0.02, min_eps_frac=0.001)
        conv_ref = _conv_self(a_refined)
        score_ref = 2 * n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined

    # Normalise to sum 1
    best_seq = np.maximum(best_seq, 0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return as list of floats
    return [float(x) for x in best_seq]

"""
Simple multi‑phase optimiser for the self‑convolution bound.

Phase 0 – generate a handful of sensible initial shapes (uniform,
Beta‑family, random and the optional warm‑start `height_sequence_1`);
keep the best by the exact objective 2·n·max(conv).

Phase 1 – simulated‑annealing style local search.
Each iteration moves a tiny amount of probability mass from a
source index (chosen proportionally to its contribution to the current
maximal convolution entry) to a destination index that contributes
little to that entry.  The step size decays and a small temperature
allows occasional uphill moves, helping the search escape shallow
local minima.

Phase 2 – fast greedy refinement (the same O(n) “targeted‑refine”
used in the previous state‑of‑the‑art code) to polish the result.

The whole routine respects the supplied time budget (default 1000 s)
and returns the best sequence found, normalised to sum = 1.
All numbers are non‑negative as required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT‑based convolution, simplex projection, up‑sampling)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0               # numerical noise
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric families for quick initialisation
# ----------------------------------------------------------------------
def _beta_shape(alpha: float, n: int) -> np.ndarray:
    """
    Symmetric Beta(α,α) density sampled on a regular grid.
    Works for any α>0; for α<1 the density blows up at the edges
    (handled by normalisation).
    """
    x = (np.arange(n) + 0.5) / n          # points in (0,1)
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a[np.isnan(a)] = 0.0                  # protects against 0**negative
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def _random_shape(n: int) -> np.ndarray:
    """Pure random positive vector normalised to a probability simplex."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – simulated‑annealing style local search
# ----------------------------------------------------------------------
def _annealing_refine(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eps_start: float = 0.20,
                      eps_end: float = 0.02,
                      temp_start: float = 1e-6,
                      temp_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from a “high‑contribution’’ index to a
    “low‑contribution’’ index.  The step size decays from ``eps_start`` to
    ``eps_end`` while a tiny temperature allows occasional uphill moves.
    Returns the best sequence discovered and its maximal convolution entry.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()

    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    i_arr = np.arange(n)

    it = 0
    while time.time() < deadline:
        it += 1
        # linear schedules
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)

        eps_frac = eps_start * (1.0 - frac) + eps_end * frac
        temperature = temp_start * (1.0 - frac) + temp_end * frac

        # index of current maximal convolution entry
        kmax = int(conv.argmax())

        # --- source index (probability proportional to contribution) ---
        j = kmax - i_arr
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i_arr[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            src = rng.choice(n, p=contrib / total_contrib)
        else:                               # fallback to uniform weighting
            src = rng.choice(n, p=a)

        # --- destination index (prefer indices that do *not* touch kmax) ---
        jdst = kmax - i_arr
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        # avoid degenerate src == dst
        candidates = candidates[candidates != src]
        if candidates.size == 0:
            continue
        dst = rng.choice(candidates)

        # --- propose a move ------------------------------------------------
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution (exact up to round‑off)
        delta = np.zeros_like(conv)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # --- acceptance test ------------------------------------------------
        accept = False
        if new_max < cur_max - 1e-14:
            accept = True
        else:
            # Metropolis rule (uphill moves allowed while temperature > 0)
            if temperature > 0.0 and rng.random() < math.exp(-(new_max - cur_max) / temperature):
                accept = True

        if accept:
            # commit the move
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            cur_max = new_max

            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: iter={it:,}, max={best_max:.8f}")

        # every 50000 iterations recompute the convolution from scratch
        # (accumulated rounding errors could otherwise drift)
        if it % 50000 == 0:
            conv = _conv_self(a)
            cur_max = conv.max()

    return best_a, best_max


# ----------------------------------------------------------------------
# Phase 2 – fast greedy “targeted‑refine” (copied from the reference code)
# ----------------------------------------------------------------------
def _run_targeted_refine_optimized(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.02,
                                   min_eps_frac: float = 0.001):
    """
    Fine‑grained greedy optimisation.
    Moves a tiny fraction of probability mass away from indices that
    contribute most to the currently maximal convolution entry.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linear decay of step size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # source index – weighted by its contribution to the current max
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # destination – favours indices that *do not* contribute to k_max
        jdst = k_max - np.arange(n)
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # propose a move
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution (exact up to round‑off)
        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: iter={it:,}, max={best_max:.8f}")

        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine explores a few parametric families,
    then performs a simulated‑annealing local search followed by a fast
    greedy refinement.  The best sequence found within the time budget is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a working length.  4800 was used in the reference solution;
    # it provides a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # Phase 0 – build a small pool of initial candidates.
    # ------------------------------------------------------------------
    candidates = []

    # 0) warm‑start from the previously discovered construction, if any
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            if starter.shape[0] != n:
                starter = _upsample(starter, n)
            candidates.append(starter)
    except Exception:
        pass

    # 1) a few Beta‑shapes (α in a sensible range)
    for alpha in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0, 1.2, 1.5]:
        candidates.append(_beta_shape(alpha, n))

    # 2) a pure random start
    candidates.append(_random_shape(n))

    # ------------------------------------------------------------------
    # Pick the best of the initial pool (exact objective, no smoothing).
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    best_max = None
    for a in candidates:
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            best_max = cur_max
            print(f"[{time.time() - start_time:.2f}s] init candidate score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – simulated‑annealing style local search.
    # Allocate roughly 55 % of the remaining budget.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        sa_deadline = time.time() + 0.55 * remaining
        a_sa, max_sa = _annealing_refine(
            best_a,
            n,
            sa_deadline,
            start_time,
            eps_start=0.20,
            eps_end=0.02,
            temp_start=1e-6,
            temp_end=1e-9,
        )
        score_sa = 2.0 * n * max_sa
        if score_sa < best_score:
            best_score = score_sa
            best_a = a_sa.copy()
            best_max = max_sa
            print(f"[{time.time() - start_time:.2f}s] annealing improve: score={score_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – fast greedy “targeted‑refine” to polish the solution.
    # Use the rest of the time budget.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_targeted_refine_optimized(
            best_a,
            n,
            deadline,
            start_time,
            init_eps_frac=0.02,
            min_eps_frac=0.001,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_a = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return the best sequence found.
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    return [float(x) for x in best_a]
