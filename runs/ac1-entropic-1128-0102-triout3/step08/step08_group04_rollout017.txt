"""
Improved optimisation for the step‑function bound.

The algorithm consists of three phases:

1️⃣  A very short exponentiated‑gradient warm‑up on a smoothed max (the same
    idea as the reference code, but we limit it to a few seconds so that the
    bulk of the budget can be spent on the next phase).

2️⃣  A long projected Adam‑style sub‑gradient descent on the *exact* max of
    the self‑convolution.  Adam gives an adaptive learning rate and usually
    drives the objective much lower than the simple diminishing‑step
    sub‑gradient used previously.

3️⃣  Aggressive local refinement: many tiny mass‑shifts are performed.
    The convolution update is carried out analytically (Δconv[k] = 2·ε·(a_dst‑a_src))
    which makes a single iteration O(n).  The schedule for ε shrinks from
    4 % of the source mass down to 0.1 % and every accepted move that lowers
    the current maximum is kept.  This phase often squeezes a further few
    percent out of the bound.

All phases respect the global 1000 s budget.  The best sequence encountered
throughout the run is returned (normalised to sum = 1).  The code works for
any length – the default is 4800, the same as the reference implementation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (same as in the reference implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # each ordered pair (i,j) contributes a[j] to ∂/∂a_i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : very short exponentiated‑gradient warm‑up (smoothed max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 3000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Adam‑accelerated projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          lr: float = 0.2,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          rel_tol: float = 1e-12):
    """
    Adam‑style sub‑gradient method on `max(conv)`.  The iterate is kept on
    the probability simplex via Euclidean projection after every step.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0
    best_a = a.copy()
    best_score = float("inf")
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑SG improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the exact max
        grad = _subgradient_full(a, conv, rel_tol=rel_tol)

        # Adam updates (with bias correction)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

    return best_a, best_score

# Phase‑3 : aggressive analytical incremental refinement
# ----------------------------------------------------------------------
def _run_incremental_refine(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           eps_start: float = 0.04,
                           eps_end: float = 0.001,
                           rel_improve: float = 0.0):
    """
    Perform a massive number of tiny mass‑shifts, updating the convolution
    analytically (Δconv[k] = 2·ε·(a_dst – a_src)).
    """
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # current maximal convolution index
        k_max = int(np.argmax(conv))

        # ---- choose a source index (high contribution to the max bin) ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ---- choose a destination index (low contribution to the max bin) ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                         # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ---- amount to move -------------------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ---- analytical convolution update (O(n)) -------------------------
        conv_new = conv.copy()
        # indices where the source (resp. destination) participates as left term
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        if new_max < max_val - rel_improve:
            # accept move
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] inc‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Initialise – try to reuse a previously found construction if
    # the global ``height_sequence_1`` exists.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800                       # length of the step‑function
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – very short warm‑up (exponentiated gradient on smoothed max)
    # --------------------------------------------------------------
    stage1_time = min(6.0, 0.05 * (deadline - time.time()))
    stage1_deadline = time.time() + stage1_time
    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        steps_per_alpha=3000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: "
              f"score={score_warm:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long Adam‑accelerated projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly 75 % of the remaining budget to this phase
        stage2_deadline = time.time() + max(30.0, 0.75 * remaining)

        a_adam, score_adam = _run_adam_subgradient(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            lr=0.18,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            rel_tol=1e-12,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: "
                  f"score={score_adam:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – massive incremental local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_incremental_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            eps_start=0.05,   # a little more aggressive than before
            eps_end=0.001,
            rel_improve=0.0,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
