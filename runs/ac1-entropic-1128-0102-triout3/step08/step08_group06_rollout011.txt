"""
Propose a step‑function by simulated‑annealing on the *exact* maximal
self‑convolution value.

The algorithm starts from the global ``height_sequence_1`` (if present) or
from a random uniform start, normalises it to a probability vector and
computes its self‑convolution by FFT.  Then a large number of tiny mass
transfers “src → dst” are attempted.  A move is always accepted when it
strictly reduces the current maximal convolution entry, otherwise it is
accepted with a Metropolis probability that depends on a temperature
schedule.  The temperature starts relatively high (allowing occasional
up‑hill moves) and cools linearly to near zero.  The source index is
chosen proportionally to its contribution to the current maximal
convolution entry, while the destination is biased towards indices that
contribute little.  The move size is a small random fraction of the source
mass; the fraction decays together with the temperature.

Periodically the full convolution is recomputed from scratch to avoid
numerical drift.  The best sequence seen during the run is kept and a
final fast “targeted‑refine” phase (tiny greedy moves) is applied in the
remaining time budget.

The whole search respects the 1000 s budget and returns a non‑negative
sequence whose sum is 1 (the evaluator will then clamp to the allowed
range).  The method is deliberately different from the previously used
exponential‑gradient / sub‑gradient pipeline and is capable of escaping
local minima that those deterministic refinements can get stuck in.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negative noise that can appear because of floating‑point errors.
    conv[conv < 0] = 0.0
    return conv


# ----------------------------------------------------------------------
# Simulated‑annealing move handling
# ----------------------------------------------------------------------
def _propose_move(a: np.ndarray,
                  conv: np.ndarray,
                  n: int,
                  k_max: int,
                  src_probs: np.ndarray,
                  dst_candidates: np.ndarray,
                  eps_frac: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, np.ndarray, float, int, int, float]:
    """
    Generates a candidate move (src → dst) and returns the updated
    convolution (as a delta array), the new maximal value, and the
    indices and epsilon used.
    """
    src = rng.choice(n, p=src_probs)

    if a[src] <= 0.0:
        return None  # nothing to move

    dst = rng.choice(dst_candidates)
    if src == dst:
        return None

    eps = a[src] * eps_frac * rng.random()
    if eps <= 0.0:
        return None

    # Build the delta to the convolution (exact up‑to round‑off).
    conv_len = 2 * n - 1
    delta = np.zeros(conv_len, dtype=float)

    # Linear part 2*eps*(a_{k‑dst} – a_{k‑src})
    delta[dst:dst + n] += 2.0 * eps * a
    delta[src:src + n] -= 2.0 * eps * a

    # Quadratic correction (three entries)
    delta[2 * src] += eps * eps
    delta[2 * dst] += eps * eps
    delta[src + dst] -= 2.0 * eps * eps

    new_conv = conv + delta
    new_max = new_conv.max()
    return new_conv, delta, new_max, src, dst, eps

# Main entry point
# ----------------------------------------------------------------------
