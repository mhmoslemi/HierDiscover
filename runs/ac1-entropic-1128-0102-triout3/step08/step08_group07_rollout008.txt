def _lp_improve(f, M):
    # Build A
    n = len(f)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    result = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if not result.success:
        return None
    b = result.x
    sum_b = b.sum()
    if sum_b == 0:
        return f
    g = b / sum_b
    return g

def _beta_window(n, alpha, beta):
    i = np.arange(n) + 0.5
    x = i / n
    w = (x**(alpha - 1)) * ((1 - x)**(beta - 1))

"""
Propose a low‑score step‑function.

The search merges three ideas that are not used together in the reference
implementation:

1. **LP smoothing on a small problem** – on a modest size (≈1024) we solve
   a linear program that maximises the total mass of a new vector `b`
   while keeping the convolution `f*b` below the current worst value `M`.
   After scaling `b` to the probability simplex we mix it with the current
   vector.  Repeating this a handful of times drives the autocorrelation
   towards a near‑flat shape (the continuous optimum is `f*f ≡ const`).

2. **Upsampling & fine optimisation** – the improved small vector is
   linearly interpolated to a large size (8192) and refined with the
   soft‑max Adam phase, a projected Nesterov sub‑gradient descent and a
   light simulated‑annealing polishing stage (the same tools as in the
   previous state‑of‑the‑art code).

3. **Richer initialisers** – besides Chebyshev/Kaiser windows we also
   try Beta‑shaped windows (`β<1` gives strong endpoints) and Dirichlet
   draws with small concentration, which are known to be good seeds for
   the LP stage.

All stages respect the global 1000 s budget.  The returned list is a plain
Python list of non‑negative floats whose sum is (numerically) 1.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, sum fixed)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9994,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    with Adam and Euclidean projection onto the simplex.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    The acceptance probability uses the true objective (2·n·max(conv)).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random but structured non‑negative vector of length n.
    With prob. 0.4 we give a Chebyshev window,
    with prob. 0.4 a Kaiser window,
    otherwise a plain Dirichlet draw.
    """
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
        w = np.maximum(w, 0.0)
    elif have_signal and r < 0.8:
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
        w = np.maximum(w, 0.0)
    else:
        w = rng.random(n)
        w = np.maximum(w, 0.0)

    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _beta_window(n: int, alpha: float, beta_: float) -> np.ndarray:
    """
    Beta‑shaped (Beta(α,β)) window on the simplex.
    α,β < 1 give strong mass at the two ends,
    α=β=1 is the flat uniform window.
    """
    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    # avoid exact 0/1 which would raise inf for α/β<1
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1)) * ((1 - x) ** (beta_ - 1))
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _dirichlet_initialiser(n: int, rng: np.random.Generator, concentration: float) -> np.ndarray:
    """Draw a Dirichlet vector with the given concentration parameter."""
    alpha = np.full(n, concentration)
    w = rng.dirichlet(alpha)
    return w

def _lp_improve(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ b_j   s.t.  f * b ≤ M ,  b ≥ 0.
    Return the scaled feasible vector g = b / Σb (or None on failure).
    """
    n = f.shape[0]
    # Build the (2n‑1)×n convolution matrix of f.
    # Column j is f shifted by j positions.
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    # Linear programme: minimise -1ᵀb   (i.e. maximise 1ᵀb)
    c = -np.ones(n, dtype=np.float64)
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=ub,
                  bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None

    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    g = b / total
    return g


def _lp_smoothing_stage(a0: np.ndarray,
                        max_iters: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve `a` by the LP step described above.
    Returns the best vector found and its max‑convolution value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = _conv_self(a).max()

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = _conv_self(a).max()
        g = _lp_improve(a, M)
        if g is None:
            break
        # mixing parameter – a bit aggressive at first, then decays
        t = 0.4 * (0.97 ** it)
        a_new = (1.0 - t) * a + t * g
        M_new = _conv_self(a_new).max()
        if M_new < M:
            a = a_new
            if M_new < best_M:
                best_M = M_new
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] lp improve: iter={it}, M={M_new:.6e}")
        else:
            # if no improvement, shrink the step size more aggressively
            if t < 1e-3:
                break
    return best_a, best_M

                tau0=tau0,
                tau_end=1e-13,
                tau_decay=0.9995,
                step_size=0.22,
            )
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] stage 0 fallback Adam: "
                      f"restart={r}, score={cur_score:.6f}")

    # safety net – extremely unlikely to be hit
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)
        best_score = 2.0 * n_small * _conv_self(best_seq).max()

    # --------------------------------------------------------------
    # Stage 1 – upsample to a large size and continue fine optimisation
    # --------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_large = 8192
    a_large = _upsample(best_seq, n_large)

    # a quick Adam pass to adapt the shape at the new resolution
    conv1 = _conv_self(a_large)
    tau0_large = max(conv1.max() * 8.0, 1e-10)

    remaining = deadline - time.time()
    stage1_time = max(20.0, 0.25 * remaining)
    slice_deadline = time.time() + stage1_time

    a_large_opt, score1 = _run_adam_softmax(
        a_large, n_large, slice_deadline, start_time,
        tau0=tau0_large,
        tau_end=1e-14,
        tau_decay=0.9994,
        step_size=0.15,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a_large_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 Adam improve: score={score1:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent on the true objective
    # --------------------------------------------------------------
    if time.time() < deadline:
        n2 = best_seq.shape[0]
        grad_deadline = deadline - 2.0   # leave a couple of seconds for SA
        a2_opt, score2 = _run_subgradient_descent(
            best_seq, n2, grad_deadline, start_time,
            beta=0.97,
            step0=0.13,
        )
        if score2 < best_score:
            best_score = score2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 subgrad improve: score={score2:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – extended simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=800_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # final normalisation (guard against tiny drift)
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain list for the harness
    return [float(x) for x in best_seq]
