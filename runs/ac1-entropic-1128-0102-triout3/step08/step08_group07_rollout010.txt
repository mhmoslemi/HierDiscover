"""
Propose a low‑score non‑negative step function.

The algorithm builds on the previous successful pipeline but adds:
  * more diverse deterministic initialisations,
  * an extra up‑sampling stage to a larger length (16384),
  * a longer projected sub‑gradient phase on the exact max‑convolution,
  * a more aggressive simulated‑annealing phase that combines
    local mass‑transfer moves with occasional global Gaussian perturbations.
All stages respect a global 1000 s budget, always keep the vector in the
probability simplex (∑a_i = 1, a_i ≥ 0), and return a plain Python list.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for exact polishing)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, sum fixed)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9994,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """
    Minimise   O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ)
    with Adam, projecting onto the simplex after each step.
    Returns the best vector seen and its *true* (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # evaluate true score occasionally (every 10 iterations)
        if it % 10 == 0:
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                      f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max of the convolution
        conv = _conv_self(a)
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ (see original script)
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # slice of length n
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    # final true score for the returned vector
    conv = _conv_self(best_a)
    best_score = 2.0 * n * conv.max()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    # final score
    conv = _conv_self(best_a)
    best_score = 2.0 * n * conv.max()
    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers + Gaussian jumps)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that mixes:
       – a local mass‑transfer move (source → destination),
       – a global Gaussian perturbation (optional, 10 % of moves).
    Acceptance uses the true objective 2·n·max(conv).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if rng.random() < 0.9:
            # ---- Local mass‑transfer move ----
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue
            eps = a[src] * rng.random() * 0.35   # up to 35 % of source mass
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            a_prop = _project_simplex(a_prop)
        else:
            # ---- Global Gaussian perturbation ----
            sigma = 0.2 * (1.0 - it / steps) + 1e-5   # linear cooling of step size
            a_prop = a + sigma * rng.standard_normal(n)
            a_prop = np.maximum(a_prop, 0.0)
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0.0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with minimal
    `evaluate_sequence` value.
    The pipeline:
      0. Grab a previously‑found solution if available.
      1. Multiple short Adam runs on a modest size (n≈2048) with diverse
         deterministic initialisations.
      2. Progressive up‑sampling (2048 → 4096 → 8192 → 16384) with short
         Adam refinements.
      3. A long projected sub‑gradient descent on the final length.
      4. An extended simulated‑annealing polishing phase that combines
         local mass transfers and occasional global Gaussian jumps.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # 0 – initialise from a previously discovered construction if possible
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # 1 – many quick Adam runs on a modest size (n≈2048)
    # ------------------------------------------------------------------
    n0 = 2048
    best_seq = None
    best_score = float("inf")
    stage0_deadline = min(deadline, start_time + 0.12 * budget_s)

    for restart in range(12):
        if time.time() >= deadline:
            break

        # deterministic plus random initialisers
        if starter is not None and restart == 0:
            a0 = _upsample(starter, n0)
        else:
            mode = restart % 4
            if mode == 0:
                # decreasing quadratic
                idx = np.arange(n0, dtype=float)
                a0 = (n0 - idx) ** 2
            elif mode == 1:
                # exponential tail
                a0 = np.exp(-0.004 * np.arange(n0))
            elif mode == 2:
                # sine‑window like
                a0 = np.sin(np.pi * (np.arange(n0) + 0.5) / n0) ** 2
            else:
                # pure random Dirichlet draw
                a0 = rng.random(n0)
            a0 = _project_simplex(a0)

        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 8.0, 1e-8)

        a_opt, true_score = _run_adam_softmax(
            a0, n0, stage0_deadline, start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9995,
            step_size=0.22,
        )
        if true_score < best_score:
            best_score = true_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage0 improve: restart={restart}, "
                  f"score={true_score:.6f}")

    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # 2 – progressive up‑sampling with short Adam refinements
    # ------------------------------------------------------------------
    up_sizes = [4096, 8192, 16384]                # final target is 16384
    for target_n in up_sizes:
        if time.time() >= deadline:
            break
        # linear interpolation to new size
        a_up = _upsample(best_seq, target_n)
        a_up = _project_simplex(a_up)

        conv_up = _conv_self(a_up)
        tau0_up = max(conv_up.max() * 8.0, 1e-10)

        # allocate a small slice of remaining time (≈15 % each up‑sample)
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.15, 30.0)   # cap to keep enough time later
        slice_deadline = time.time() + slice_time

        a_opt, true_score = _run_adam_softmax(
            a_up, target_n, slice_deadline, start_time,
            tau0=tau0_up,
            tau_end=1e-14,
            tau_decay=0.9994,
            step_size=0.15,
        )
        cur_score = true_score
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] upsample improve: n={target_n}, "
                  f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3 – long projected sub‑gradient descent on the final length
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_final = best_seq.shape[0]
        grad_deadline = deadline - 2.0               # keep a couple seconds for annealing
        a_sgd, sgd_score = _run_subgradient_descent(
            best_seq, n_final, grad_deadline, start_time,
            beta=0.97,
            step0=0.13,
        )
        cur_score = 2.0 * n_final * _conv_self(a_sgd).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_sgd.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad final improve: score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4 – extended annealing (mass‑transfer + Gaussian jumps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # a generous number of moves; real count is limited by remaining time
        moves = 1_500_000
        a_anneal, anneal_score = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=moves,
        )
        cur_score = 2.0 * best_seq.shape[0] * _conv_self(a_anneal).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_anneal.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation onto the simplex (guard against tiny drift)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the harness
    return [float(x) for x in best_seq]
