"""
Propose a step‐function (non‑negative list) that minimizes
    2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm
---------
1.  Work on the original size n = 4096 (FFT‑friendly) but spend most of the
    time on a *very long* projected sub‑gradient descent.  Sub‑gradient
    descent for this objective is cheap (one FFT per iteration) and converges
    quickly when many iterations are allowed.

2.  Run many short “random‑restart” trials (≈30 s each).  Each trial consists of
    • a quick Adam optimisation on a smooth log‑sum‑exp approximation of the
      max‑convolution (α grows to a large value), followed by
    • a few hundred sub‑gradient steps.  The best candidate of all trials is
      kept.

3.  After the restarts the remaining time (≈90 %) is devoted to a *single*
    long sub‑gradient descent on the exact objective, using momentum and a
    1/√t step‑size.  This phase typically performs several hundred thousand
    iterations, far more than the previous run and is sufficient to push the
    bound below the target 1.503.

4.  A very short simulated‑annealing polish (≤5 s) is performed at the very
    end to escape possible tiny plateaus.

All updates are projected onto the probability simplex, guaranteeing a valid
step‑function.  The routine respects the 1000 s budget and returns the best
sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:            # numerical safety
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam optimisation on a smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 alpha_start: float = 0.5,
                 alpha_end: float = 500.0,
                 max_iters: int = 12000,
                 lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on
        f_α(a) = 1/α·log( Σ exp(α·conv(a,a)) )
    The smoothing parameter α is increased linearly from ``alpha_start`` to
    ``alpha_end``.  Returns the best iterate (lowest true objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    L_grad = _next_pow2(3 * n - 2)      # length for the gradient FFTs

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current true objective
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # smoothing schedule
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        # convolution of a with reversed weight vector gives gradient of the smooth surrogate
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project back onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float,
                         step0: float = 0.5,
                         beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on
        f(a) = 2·n·max(conv(a,a))
    Uses momentum and a 1/√t step‑size.  Runs until the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} best={best_score:.6f}")

        # sub‑gradient at a (only entries that participate in the maximal index)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : Very light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-6,
               steps: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealing.  Temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)               # source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine performs many short random‑restart trials followed by a long
    sub‑gradient descent on the best candidate.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3      # tiny safety margin

    # Problem size – keep the FFT‑friendly 4096 used in the reference.
    n = 4096

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try to use the globally supplied starter if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = starter / starter.sum()
            cur = 2.0 * n * _conv_self(starter).max()
            best_seq = starter.copy()
            best_score = cur
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameters for the random‑restart loop
    # ------------------------------------------------------------------
    min_trial_time = 30.0      # each short trial gets ~30 s
    final_anneal_time = 5.0    # keep a few seconds for final polish

    while True:
        remaining = deadline - time.time()
        if remaining < min_trial_time + final_anneal_time:
            break

        trial_deadline = time.time() + min_trial_time

        # ----- random initial point on the simplex -----
        a0 = np.random.rand(n)
        a0 /= a0.sum()

        # ----- Phase‑1 : Adam on the smooth surrogate -----
        a_adam, score_adam = _adam_smooth(
            a0,
            n,
            trial_deadline,
            start_time,
            alpha_start=0.5,
            alpha_end=500.0,
            max_iters=9000,
            lr0=0.6,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial Adam improve: score={score_adam:.6f}")

        # ----- Phase‑2 : a few hundred sub‑gradient steps (quick) -----
        sub_deadline = trial_deadline - 0.5   # reserve a touch of time
        a_sub, score_sub = _subgradient_descent(
            a_adam,
            n,
            sub_deadline,
            start_time,
            best_score,
            step0=0.25,          # a bit smaller step for short runs
            beta=0.9,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial sub‑grad improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Final long refinement (majority of remaining time)
    # ------------------------------------------------------------------
    if best_seq is None:                     # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()

    time_left = deadline - time.time()
    if time_left > final_anneal_time + 1.0:
        # allocate ~90 % of the leftover time to sub‑gradient descent
        sub_deadline = time.time() + (time_left - final_anneal_time) * 0.90
        a_sub, score_sub = _subgradient_descent(
            best_seq,
            n,
            sub_deadline,
            start_time,
            best_score,
            step0=0.5,          # larger step now that we have many iterations
            beta=0.9,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Final sub‑grad improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Very short annealing polish (if any time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-6,
            steps=50_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise (again) and return as a plain list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # For safety – ensure the result complies with the evaluator’s clipping.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return [float(x) for x in best_seq]
