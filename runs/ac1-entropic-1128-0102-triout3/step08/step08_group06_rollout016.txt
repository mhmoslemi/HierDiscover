"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm consists of several stages:

1.  Warm‑up with exponentiated‑gradient on a smoothed (log‑sum‑exp)
    approximation of the maximal self‑convolution.
2.  Projected sub‑gradient descent on the exact max of the convolution.
3.  A fast O(n) targeted refinement that moves a tiny amount of mass
    from indices that contribute most to the current maximal entry
    to indices that contribute the least.
4.  A linear‑programming refinement:  using `scipy.optimize.linprog`
    we solve a linear approximation of the problem
    (minimise the maximal convolution entry) on the active set of
    constraints, then take a line‑search step.  Repeating this a few
    times yields a measurable improvement over the previous three‑phase
    scheme.

The whole search respects a 1000 s budget (a 0.5 s safety margin) and
keeps the best sequence found throughout the run.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negatives that appear because of numerical noise.
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 10000):
    """Multiplicative mirror‑descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max weights (log‑sum‑exp)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : cheap targeted refinement (fast, fine‑grained)
# ----------------------------------------------------------------------
def _run_targeted_refine_optimized(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.05,
                                   min_eps_frac: float = 0.001):
    """Fast O(n) refinement moving a tiny amount of probability mass."""
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # Linear decay for step size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # source – weighted by its contribution to the current max
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # destination – favour indices that hardly participate in k_max
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution (exact up to round‑off)
        delta = np.zeros(conv_len, dtype=float)
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Phase‑4 : linear‑programming based refinement
# ----------------------------------------------------------------------
def _run_lp_refine(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iters: int = 12,
                   active_tol: float = 1e-6):
    """
    Repeatedly solve a linear approximation of the max‑convolution
    problem on the currently active constraints and take a line‑search
    step.  The loop stops when time is exhausted or no further improvement
    is possible.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        # pick active constraints (those close to the current maximum)
        thresh = cur_max * (1.0 - active_tol)
        active_idx = np.where(conv >= thresh)[0]
        if active_idx.size == 0:
            active_idx = np.array([conv.argmax()])

        m = active_idx.size
        # Build sparse matrix for the inequality 2*a*d - t <= -conv
        rows = []
        cols = []
        data = []

        for row, k in enumerate(active_idx):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1)
            j = k - i
            coeff = 2.0 * a[j]          # coeff for d_i
            rows.extend([row] * i.size)
            cols.extend(i.tolist())
            data.extend(coeff.tolist())

        # column for the scalar variable t (last column)
        rows.extend(range(m))
        cols.extend([n] * m)
        data.extend([-1.0] * m)

        A_ub = csr_matrix((data, (rows, cols)), shape=(m, n + 1))
        b_ub = -conv[active_idx].astype(float)

        # Equality: sum(d) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        # Bounds: d_i >= -a_i  (to keep a_i + d_i >= 0), t >= 0
        bounds = [(-float(a[i]), None) for i in range(n)]
        bounds.append((0.0, None))  # bound for t

        # Linear program: minimise t
        res = linprog(c=np.concatenate([np.zeros(n), [1.0]]),
                      A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # LP failed – exit the refinement loop.
            print(f"[{time.time() - start_time:.2f}s] LP solve failed, breaking.")
            break

        d = res.x[:n]
        t_opt = res.x[-1]

        # If the LP does not propose a lower bound, we are stuck.
        if t_opt >= cur_max - 1e-12:
            print(f"[{time.time() - start_time:.2f}s] LP unable to improve (t={t_opt:.6f}).")
            break

        # Line‑search to respect the quadratic terms ignored by the LP.
        lam = 1.0
        improved = False
        while lam > 1e-6:
            a_trial = a + lam * d
            if np.any(a_trial < -1e-12):
                lam *= 0.5
                continue
            a_trial = np.maximum(a_trial, 0.0)
            total = a_trial.sum()
            if total == 0:
                lam *= 0.5
                continue
            a_trial /= total
            conv_trial = _conv_self(a_trial)
            max_trial = conv_trial.max()
            if max_trial < cur_max - 1e-14:
                a = a_trial
                cur_max = max_trial
                improved = True
                if max_trial < best_score:
                    best_score = max_trial
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] LP refine improve: "
                          f"iter={it:,}, max={max_trial:.8f}")
                break
            lam *= 0.5

        if not improved:
            # Could not take a step; stop.
            print(f"[{time.time() - start_time:.2f}s] LP line‑search failed, breaking.")
            break

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.

    The optimiser proceeds through four phases:
        1. Exponentiated‑gradient warm‑up.
        2. Projected sub‑gradient descent.
        3. Fast targeted refinement.
        4. Linear‑programming based refinement + a final targeted pass.
    The best sequence encountered is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated gradient (quick warm‑up)
    # ------------------------------------------------------------------
    phase1_time = min(45.0, 0.07 * (deadline - time.time()))
    phase1_deadline = time.time() + phase1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0,
                        40.0, 80.0, 160.0, 320.0, 640.0],
        eta0=0.5,
        steps_per_alpha=8000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact‑max projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_time = max(30.0, 0.65 * remaining)
        phase2_deadline = time.time() + phase2_time

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fast targeted refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_targeted_refine_optimized(
            best_seq,
            final_n,
            deadline,
            start_time,
            init_eps_frac=0.05,
            min_eps_frac=0.001,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] refined improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – LP‑based refinement (more aggressive)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_lp = _run_lp_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_iters=20,
            active_tol=1e-7,
        )
        conv_lp = _conv_self(a_lp)
        score_lp = 2.0 * final_n * conv_lp.max()
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"[{time.time() - start_time:.2f}s] LP‑refine improve: score={score_lp:.6f}")

    # ------------------------------------------------------------------
    # Final tiny targeted clean‑up (one last pass)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final = _run_targeted_refine_optimized(
            best_seq,
            final_n,
            deadline,
            start_time,
            init_eps_frac=0.02,
            min_eps_frac=0.0005,
        )
        conv_final = _conv_self(a_final)
        score_final = 2.0 * final_n * conv_final.max()
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final tidy improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # Convert to plain Python floats for the caller.
    return [float(x) for x in best_seq]
