def _lbfgs_obj(z, n, tau):
    a = np.exp(z)
    Z = a.sum()
    a = a / Z
    conv = _conv_self(a)
    # compute stable log-sum-exp
    scaled = conv / tau
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    O = 2.0 * n * tau * (max_scaled + np.log(sum_exp))
    # gradient w.r.t. a
    p = exp_scaled / sum_exp
    a_rev = a[::-1]
    inner = _conv_fft(a_rev, p)[n-1:n-1+n]
    grad_a = 4.0 * n * inner
    # gradient w.r.t. z
    avg = np.dot(a, grad_a)
    grad_z = a * (grad_a - avg)
    return O, grad_z

def _upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up = up / total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

"""
Propose a step‑function (non‑negative sequence) that minimises the
objective 2·n·max(conv(a,a)) / (∑a)².

The algorithm proceeds in several stages, all respecting the global
time budget (default 1000 s).

1.  Initialise a probability vector (sum = 1) – either from the previous
    best construction (`height_sequence_1`) or from a random vector.
2.  Upsample the vector through a schedule of lengths
    (256 → 512 → 1024 → 2048 → 4096 → 8192).  At each size we run a
    *smooth surrogate* optimisation:
        O_τ(a)=2 n τ log ∑ exp(conv(a)/τ) .
    The surrogate is convex; we minimise it w.r.t. the logits
    `z_i = log a_i` using L‑BFGS‑B.  The temperature τ is gradually
    reduced (factors 5, 1, 0.2, 0.04, 0.008, 0.0016) so that the
    optimisation tracks the true L∞‑objective.
3.  The best vector from the surrogate phase is refined by a short
    projected Euclidean sub‑gradient descent on the true objective.
4.  Finally a light simulated‑annealing mass‑transfer is applied to
    escape tiny local traps.
All vectors are kept on the probability simplex (non‑negative,
∑=1), which makes the evaluation function scale‑invariant.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x_i≥0, Σx_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for true objective)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve (n={n}) "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth surrogate O_τ
# ----------------------------------------------------------------------


def _objective_and_grad_lbfgs(z: np.ndarray, n: int, tau: float) -> tuple[float, np.ndarray]:
    """
    Compute O_τ and its gradient w.r.t. logits `z` where
    a_i = exp(z_i)/∑exp(z) (i.e. soft‑max).
    Returns (value, grad) where grad has shape (n,).
    """
    # a = softmax(z)
    a = np.exp(z)
    Z = a.sum()
    a = a / Z

    conv = _conv_self(a)

    # stable log‑sum‑exp of conv / tau
    scaled = conv / tau
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    O = 2.0 * n * tau * (max_scaled + np.log(sum_exp))

    # gradient w.r.t. a : 4·n·(a_rev * p)[slice]
    p = exp_scaled / sum_exp                     # shape 2·n‑1, sums to 1
    a_rev = a[::-1]
    inner = _conv_fft(a_rev, p)[n - 1:n - 1 + n]
    grad_a = 4.0 * n * inner

    # gradient w.r.t. z using soft‑max Jacobian
    avg = np.dot(a, grad_a)
    grad_z = a * (grad_a - avg)

    return O, grad_z


def _run_lbfgs_schedule(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Successively minimise the smooth surrogate O_τ for a decreasing schedule
    of τ‑factors using L‑BFGS‑B.  Returns the best vector seen and its true
    score (2·n·max(conv)).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = float("inf")

    # τ‑factors (larger → smoother, smaller → closer to max)
    tau_factors = [5.0, 1.0, 0.2, 0.04, 0.008, 0.0016]

    for factor in tau_factors:
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_conv = conv.max()
        tau = max_conv * factor
        if tau <= 0.0:
            tau = 1e-8

        # logits (avoid log(0))
        eps = 1e-12
        z0 = np.log(a + eps)

        # allocate a modest slice of remaining time for this factor
        remaining = deadline - time.time()
        max_iter = 200  # safe upper bound; L‑BFGS decides when to stop
        # protect against extremely short remaining time
        if remaining < 5.0:
            max_iter = max(10, int(remaining * 10))

        res = scipy.optimize.minimize(fun=_objective_and_grad_lbfgs,
                                      x0=z0,
                                      args=(n, tau),
                                      method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': max_iter, 'disp': False})

        # retrieve the new a
        z_opt = res.x
        a_opt = np.exp(z_opt)
        a_opt = a_opt / a_opt.sum()

        cur_score = 2.0 * n * _conv_self(a_opt).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] LBFGS factor={factor:.4f} "
                  f"iter={res.nit} score={cur_score:.6f}")

        a = a_opt  # continue with the refined distribution

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine follows a multi‑scale L‑BFGS refinement, a short sub‑gradient
    polishing phase and a final simulated‑annealing massage.
    Returns a plain Python list of floats (sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise – use the previously found construction if it exists.
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # Start from a modest size; later we will up‑sample.
    init_n = 256
    if starter is not None:
        best_seq = _upsample(starter, init_n)
    else:
        a = np.random.rand(init_n)
        best_seq = a / a.sum()

    best_score = float("inf")

    # ------------------------------------------------------------------
    # Progressive up‑sampling schedule.
    # ------------------------------------------------------------------
    size_schedule = [256, 512, 1024, 2048, 4096, 8192]

    for n in size_schedule:
        if time.time() >= deadline:
            break

        # Upsample the current best vector to the new size
        if best_seq.shape[0] != n:
            a = _upsample(best_seq, n)
        else:
            a = best_seq.copy()

        # --------------------------------------------------------------
        # Phase 1 – L‑BFGS on the smooth surrogate with decreasing τ.
        # --------------------------------------------------------------
        a, score = _run_lbfgs_schedule(a0=a,
                                        n=n,
                                        deadline=deadline,
                                        start_time=start_time)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] size {n} LBFGS best score {score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Projected Euclidean sub‑gradient descent on the true objective.
        # --------------------------------------------------------------
        if time.time() < deadline:
            sub_deadline = deadline - 2.0   # keep a few seconds for later phases
            a, sub_score = _run_subgradient_descent(a0=best_seq,
                                                    n=n,
                                                    deadline=sub_deadline,
                                                    start_time=start_time,
                                                    beta=0.95,
                                                    step0=0.20)
            if sub_score < best_score:
                best_score = sub_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] size {n} subgrad improve {sub_score:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – Light simulated annealing for the final few seconds.
        # --------------------------------------------------------------
        if time.time() < deadline:
            a, ann_score = _run_annealing(a0=best_seq,
                                          n=n,
                                          deadline=deadline,
                                          start_time=start_time,
                                          cur_best_score=best_score,
                                          cur_best_seq=best_seq,
                                          start_temp=2e-2,
                                          end_temp=1e-7,
                                          steps=200_000)
            if ann_score < best_score:
                best_score = ann_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] size {n} anneal final improve {ann_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and return a plain list.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
