def _run_adam_smooth(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    steps_per_alpha = 5000
    L = _next_pow2(2 * n - 1)
    while time.time() < deadline:
        it += 1
        # Determine alpha
        alpha_idx = min((it - 1) // steps_per_alpha, len(alpha_schedule) - 1)
        alpha = alpha_schedule[alpha_idx]
        # FFT of a
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        # compute smooth max weight w
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()
        # gradient
        w_rev = w[::-1]
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = 0.1 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_simplex(a)
        # evaluate score
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improved: iter={it}, alpha={alpha:.1f}, score={cur_score:.6f}")
    return best_a, best_score

"""
Propose a step‐function (non‑negative sequence) that minimizes

    2 * n * max(convolve(seq, seq)) / (sum(seq)**2)

The optimizer consists of three stages:

1. **Adam on a smooth approximation of the maximum** – we minimise the
   log‑sum‑exp of the self‑convolution (a smooth max) using Adam with a
   simplex projection after each update.  The smoothing parameter α is
   increased gradually (continuation) so the optimisation first sees a
   gentle landscape and later focuses on the true max.

2. **Fast targeted refinement** – a greedy O(n) move that shifts a tiny
   amount of probability mass away from indices that contribute most to
   the current maximal convolution entry, accepting only strict
   improvements.  This is the same idea as the “refine_opt” phase in the
   reference code but runs after the Adam stage.

3. **Optional short restarts** – while time permits we perturb the best
   solution found, run a brief Adam + refinement cycle and keep any
   improvement.  This helps to escape tiny local minima.

All stages respect the global 1000 s budget and the best sequence found
throughout the run is returned (normalised to sum = 1).  The code works
with the optional global variable ``height_sequence_1`` as a warm start.
"""

import time
import math
import numpy as np
import sys

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values – clip.
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase 1 – Adam on smooth max
# ----------------------------------------------------------------------
def _run_adam_smooth(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smoothed maximum of the self‑convolution.
    Returns the best sequence seen and its (unsmoothed) objective value.
    """
    # hyper‑parameters
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0,
                      40.0, 80.0, 160.0, 320.0, 640.0]
    steps_per_alpha = 5000          # after this many steps we move to the next α
    lr = 0.2                        # base learning‑rate for Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros(n)
    v = np.zeros(n)

    L = _next_pow2(2 * n - 1)       # FFT length, reused each iteration
    it = 0

    while time.time() < deadline:
        it += 1
        # current smoothing parameter
        alpha_idx = min((it - 1) // steps_per_alpha, len(alpha_schedule) - 1)
        alpha = alpha_schedule[alpha_idx]

        # ----- compute convolution and smooth‑max weights -----
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # log‑sum‑exp (smooth max)
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)          # shift for stability
        w = exp_scaled / exp_scaled.sum()                 # softmax weights

        # ----- gradient of the smooth max w.r.t. a -----
        w_rev = w[::-1]
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # ----- Adam update with simplex projection -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep steps from being too aggressive (clip to a fraction of the current mass)
        step = np.clip(step, -0.5, 0.5)

        a = a - step
        a = _project_simplex(a)          # back to the probability simplex

        # ----- keep the best sequence seen -----
        cur_score = 2.0 * n * conv.max()   # exact objective (sum(a)=1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                  f"iter={it:,}, alpha={alpha:.1f}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – fast targeted refinement (same spirit as the reference)
# ----------------------------------------------------------------------
def _run_targeted_refine_optimized(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.05,
                                   min_eps_frac: float = 0.001) -> np.ndarray:
    """
    Very fast O(n) refinement.
    Moves a tiny amount of probability mass away from indices that contribute
    most to the current maximal convolution entry.  The move is accepted only
    if the maximal value strictly decreases; the step‑size decays linearly.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linear decay of the move size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # ---- choose source index (weighted by its contribution to the max) ----
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # ---- choose destination (prefer indices that hardly affect the offending sum) ----
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # ---- propose a move ----
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution (exact up to round‑off)
        delta = np.zeros(conv_len, dtype=float)

        # linear part 2*eps*(a_{k-dst} - a_{k-src})
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a

        # quadratic eps² correction (only three entries are affected)
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # periodic recompute from scratch to avoid drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Initialise – warm start from a global construction if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            # normalise to a probability vector
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # We use a larger support than the reference (9600) to obtain a finer
    # discretisation, which often yields a slightly smaller constant.
    final_n = 9600
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + 0.20 * (deadline - time.time())
    a_adam, score_adam = _run_adam_smooth(a0, final_n,
                                          phase1_deadline, start_time)
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – fast targeted refinement (≈60 % of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + 0.60 * (deadline - time.time())
        a_refined = _run_targeted_refine_optimized(best_seq, final_n,
                                                   phase2_deadline, start_time,
                                                   init_eps_frac=0.05,
                                                   min_eps_frac=0.001)
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short random restarts with quick Adam + refine cycles
    # ------------------------------------------------------------------
    while time.time() + 8.0 < deadline:   # leave a few seconds for finalisation
        # small random perturbation of the current best
        pert = best_seq + 0.02 * np.random.rand(final_n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # brief Adam run (≈4 s)
        sub_deadline = time.time() + 4.0
        a_tmp, _ = _run_adam_smooth(pert, final_n, sub_deadline, start_time)

        # brief refinement (≈4 s)
        sub_deadline = time.time() + 4.0
        a_tmp = _run_targeted_refine_optimized(a_tmp, final_n,
                                                sub_deadline, start_time,
                                                init_eps_frac=0.05,
                                                min_eps_frac=0.001)

        conv_tmp = _conv_self(a_tmp)
        score_tmp = 2.0 * final_n * conv_tmp.max()
        if score_tmp < best_score:
            best_score = score_tmp
            best_seq = a_tmp.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={score_tmp:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return the final candidate
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # (optional) sanity check with the provided evaluator
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] final evaluate_sequence={val:.6f}")
    except Exception as exc:
        print("evaluate_sequence call failed:", exc, file=sys.stderr)

    return [float(x) for x in best_seq]
