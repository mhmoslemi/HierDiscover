The optimizer works in three stages  

1. **Convex QCQP warm‑start** – a tiny convex program (size ≈ 256) directly minimises the maximal convolution value `t`.  The problem  

```
minimise   t
subject to a ≥ 0 ,  Σ a = 1 ,
           aᵀ Qₖ a ≤ t   for all k = 0 … 2n‑2
```

is convex because every matrix `Qₖ` (ones on the anti‑diagonal) is PSD.  Solving it with **CVXPY/ECOS** gives a solid starting distribution that is already far below the naïve uniform bound.

2. **Refinement on the simplex** – the warm‑start is polished with three classic optimisation phases that proved very effective in the reference code:

   * **Adam on a log‑sum‑exp smooth approximation** of the max‑convolution,
   * **Projected sub‑gradient descent** on the true objective,
   * **Targeted simulated annealing** that repeatedly moves probability mass away from the pair of indices that contribute most to the current peak.

   All steps stay on the probability simplex (`a ≥ 0 , Σa = 1`) via Euclidean projection.

3. **Random‑restart polishing** – if time remains we perform short restarts: a slight random perturbation of the current best solution followed by a short sub‑gradient + annealing run.

The whole routine respects the 1000 s budget, constantly checks the remaining time and returns the best sequence found.  The code contains no closures or lambdas and works with only the public `evaluate_sequence` function.

```python
"""
Convolution‑max minimiser
-------------------------

* Warm‑start with a small convex QCQP (CVXPY) to obtain a
  distribution that already lowers max(conv) below the uniform bound.
* Refine the distribution on the probability simplex using
  Adam (log‑sum‑exp smoothing), projected sub‑gradient descent
  and targeted simulated‑annealing.
* Optional random restarts for the last seconds of the budget.

All helpers are top‑level, no closures or lambdas are used.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 – Adam on a smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.5,
                        iters_per_alpha: int = 6000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best iterate (lowest 2·n·max(conv)) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # smooth‑max weights
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # gradient of smooth‑max
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)   # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # stay on the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Gradient is taken at the current maximal index.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # index of the maximal convolution coefficient
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]                     # sub‑gradient of conv[k]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – Targeted simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-8,
                            steps: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Annealing that always picks the current maximal convolution index
    and moves a small amount of probability mass away from the pair
    that contributes most to that peak.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate current maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move at most 20 % of the source mass
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Small convex QCQP warm‑start (size≈256)
# ----------------------------------------------------------------------
def _solve_qcqp(n: int) -> tuple[np.ndarray, float]:
    """
    Solve   min   t
            s.t. a ≥ 0 , Σ a = 1 , aᵀ Qₖ a ≤ t  for all k
    using CVXPY/ECOS.  Returns the optimal a and the optimal t.
    """
    try:
        import cvxpy as cp
        import scipy.sparse as sp
    except Exception as e:
        raise RuntimeError("CVXPY or SciPy not available") from e

    a = cp.Variable(n)
    t = cp.Variable()
    constraints = [a >= 0, cp.sum(a) == 1, t >= 0]

    # each Qₖ has ones on the anti‑diagonal (partial permutation matrix)
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        rows, cols, data = [], [], []
        for i in range(lo, hi + 1):
            j = k - i
            rows.append(i); cols.append(j); data.append(1.0)
            if i != j:
                rows.append(j); cols.append(i); data.append(1.0)
        Qk = sp.coo_matrix((data, (rows, cols)), shape=(n, n)).tocsc()
        constraints.append(cp.quad_form(a, Qk) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    # ECOS handles the quadratic constraints as rotated second‑order cones
    prob.solve(solver=cp.ECOS, max_iters=4000,
               abstol=1e-9, reltol=1e-9, feastol=1e-9, verbose=False)

    if prob.status not in ("optimal", "optimal_inaccurate"):
        raise RuntimeError(f"QCQP not solved to optimality (status={prob.status})")

    a_opt = np.array(a.value).flatten()
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    if s > 0:
        a_opt /= s
    else:
        a_opt = np.full(n, 1.0 / n)
    return a_opt, float(t.value)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    provided ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start with a tiny QCQP (n≈256).  If CVXPY is unavailable we
    #    fall back to a random start.
    # ------------------------------------------------------------------
    n_qcqp = 256
    best_seq = None
    best_score = float("inf")

    try:
        # give the QCQP at most a few seconds; it is usually solved instantly
        a_qcqp, t_opt = _solve_qcqp(n_qcqp)
        # evaluate the obtained vector
        seq = a_qcqp.tolist()
        best_score = evaluate_sequence(seq)
        best_seq = a_qcqp.copy()
        print(f"[{time.time() - start_time:.2f}s] QCQP warm‑start score={best_score:.6f}")
    except Exception as exc:
        print(f"[{time.time() - start_time:.2f}s] QCQP failed ({exc}), using random start.")
        # fall back to a random simplex point of size 4096
        fallback_n = 4096
        a_rand = np.random.rand(fallback_n)
        a_rand /= a_rand.sum()
        best_seq = a_rand
        best_score = float("inf")

    # ------------------------------------------------------------------
    # 2️⃣ Refinement phases (Adam → sub‑gradient → annealing)
    # ------------------------------------------------------------------
    n = best_seq.shape[0]

    # ---- Phase‑1 : Adam on smoothed max --------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase1_deadline = time.time() + max(5.0, 0.12 * remaining)
        a_adam, score_adam = _run_adam_smoothing(
            best_seq,
            n,
            phase1_deadline,
            start_time,
            alphas=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            lr0=0.6,
            iters_per_alpha=6000,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_adam:.6f}")

    # ---- Phase‑2 : projected sub‑gradient descent ------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(15.0, 0.70 * remaining)
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

        # a brief second sub‑gradient run with a larger step size
        if time.time() < deadline:
            extra_deadline = time.time() + min(12.0, deadline - time.time())
            a_sub2, score_sub2 = _run_subgradient_descent(
                best_seq,
                n,
                extra_deadline,
                start_time,
                best_score,
                max_iters=180_000,
            )
            if score_sub2 < best_score:
                best_score = score_sub2
                best_seq = a_sub2.copy()
                print(f"[{time.time() - start_time:.2f}s] extra sub‑grad improve: score={score_sub2:.6f}")

    # ---- Phase‑3 : targeted annealing ------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_targeted_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=180_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Random restarts (short bursts) if time is left
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        block = min(25.0, time_left / 2.0)
        restart_deadline = time.time() + block

        pert = best_seq + 0.018 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        pert_seq, pert_score = _run_subgradient_descent(
            pert,
            n,
            restart_deadline,
            start_time,
            best_score,
            max_iters=130_000,
        )
        if time.time() < deadline:
            pert_seq, pert_score = _run_targeted_annealing(
                pert_seq,
                n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-8,
                steps=90_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score = {best_score:.6f}")
    return [float(x) for x in best_seq]
