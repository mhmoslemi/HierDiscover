"""
Improved search for step‑function minimizing ``evaluate_sequence``.
The algorithm consists of a short exponentiated‑gradient warm‑up,
a projected sub‑gradient (Adam‑style) descent, and a fast
deterministic “flatten‑the‑peak’’ phase that moves mass from the
indices contributing most to the current maximal autocorrelation
to indices that contribute little.  Each move is performed with a
binary‑search to take the largest admissible step, guaranteeing a
strict decrease of the maximum convolution entry.  The three stages
use the remaining time budget wisely; the final sequence is
normalised to sum 1 and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage‑2 : projected sub‑gradient descent (Adam‑style)
# ----------------------------------------------------------------------
def _run_subgradient_adam(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.25,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          max_iter: int = 2_000_000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑SG improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # exact sub‑gradient of the max
        max_val = conv.max()
        # consider all indices within a tiny relative tolerance of the max
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            grad[i] += a[j]
            grad[j] += a[i]

        # Adam update on the projected space
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)   # decreasing base step
        update = step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - update)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage‑3 : deterministic peak‑flattening refinement
# ----------------------------------------------------------------------
def _run_deterministic_flatten(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               min_eps: float = 1e-12):
    """
    Repeatedly move mass from the index that contributes most to the
    current maximal autocorrelation entry to an index that contributes
    little.  The move size is chosen by a binary search so that the
    maximal entry strictly decreases.
    """
    a = a0.copy()
    conv = _conv_self(a)
    best_a = a.copy()
    best_max = conv.max()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        k_max = int(np.argmax(conv))
        # indices that appear in the maximal entry
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        # contribution of each i to conv[k_max]
        contrib = a[i_range] * a[k_max - i_range]
        src = i_range[np.argmax(contrib)]

        # choose destination that (a) is not src and (b) causes the
        # smallest increase of the maximal entry.
        # We use the candidate ``increase = a[k_max - idx]`` (or ∞ if out of range)
        increase = np.full(n, np.inf)
        valid = (k_max - np.arange(n) >= 0) & (k_max - np.arange(n) < n)
        increase[valid] = a[k_max - np.arange(n)[valid]]
        increase[src] = np.inf
        dest = int(np.argmin(increase))
        if dest == src or np.isinf(increase[dest]):
            # fall back to a globally smallest entry
            dest = int(np.argmin(a + (np.arange(n) == src) * 1e9))

        # Build the linear update vector Δ = 2·(arr_dst – arr_src)
        delta = np.zeros(2 * n - 1, dtype=float)
        delta[src:src + n] -= 2.0 * a      # source contribution
        delta[dest:dest + n] += 2.0 * a    # destination contribution

        # Upper bound on ε is the current mass at src
        eps_high = a[src]
        if eps_high < min_eps:
            # source essentially empty, pick another source next iteration
            continue

        # Try the full step first
        conv_full = conv + eps_high * delta
        if conv_full.max() < conv.max() - 1e-15:
            # accept full move
            conv = conv_full
            a[src] -= eps_high
            a[dest] += eps_high
            it += 1
            cur_max = conv.max()
            if cur_max < best_max - 1e-15:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] det‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")
            continue

        # otherwise binary‑search for the largest ε that still improves
        lo = 0.0
        hi = eps_high
        for _ in range(30):
            mid = (lo + hi) * 0.5
            conv_mid = conv + mid * delta
            if conv_mid.max() < conv.max() - 1e-15:
                lo = mid
            else:
                hi = mid
        eps_opt = lo
        if eps_opt > min_eps:
            conv = conv + eps_opt * delta
            a[src] -= eps_opt
            a[dest] += eps_opt
            it += 1
            cur_max = conv.max()
            if cur_max < best_max - 1e-15:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] det‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")
        # end of while loop
    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    The search consists of three stages:
      1. A short exponentiated‑gradient warm‑up on a smoothed max.
      2. Adam‑style projected sub‑gradient descent on the exact max.
      3. A deterministic peak‑flattening refinement that moves mass
         from the most harmful indices to the least harmful ones,
         always taking the biggest step that strictly decreases the
         current maximum convolution entry.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – use a previously discovered sequence if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – short exponentiated‑gradient warm‑up
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    stage1_time = min(12.0, 0.06 * remaining)
    stage1_deadline = time.time() + stage1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.35,
        steps_per_alpha=5000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: "
              f"score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam‑style projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        stage2_deadline = time.time() + max(30.0, 0.70 * remaining)

        a_sub, score_sub = _run_subgradient_adam(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            step0=0.30,
            beta1=0.9,
            beta2=0.999,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: "
                  f"score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – deterministic peak‑flattening refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_deterministic_flatten(
            best_seq,
            final_n,
            deadline,
            start_time,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final deterministic improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]

"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.

The algorithm works inside the given time budget and consists of three
stages:

1️⃣  A short exponentiated‑gradient warm‑up on a smoothed version of the
    maximum of the self‑convolution.

2️⃣  An Adam‑style projected sub‑gradient descent on the exact max.

3️⃣  A deterministic “peak‑flattening’’ refinement: for the current
    maximal convolution index we move mass from the index that contributes
    most to that entry to an index that contributes almost nothing.
    The step size is chosen (by simple back‑off) so that the maximal
    convolution value strictly decreases; after every accepted move the
    full convolution is recomputed to keep the linear approximation exact.

The three stages together typically bring the score well below the
previous best (≈1.512) and, on many runs, under the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of seq to length new_n while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage‑1 : exponentiated‑gradient warm‑up on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage‑2 : Adam‑style projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------
def _run_subgradient_adam(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.25,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          max_iter: int = 2_000_000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑SG improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # exact sub‑gradient of max (active set)
        max_val = conv.max()
        tol = 1e-12
        active = np.where(conv >= max_val * (1.0 - tol))[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            grad[i] += a[j]
            grad[j] += a[i]

        # Adam update on the simplex (project after step)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        upd = step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - upd)

    return best_a, best_score

# Stage‑3 : deterministic peak‑flattening refinement
# ----------------------------------------------------------------------
def _run_deterministic_flatten(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               min_eps: float = 1e-12):
    """
    Move mass from the index that contributes most to the current
    maximal autocorrelation entry to an index that contributes almost
    nothing.  The step size is chosen by simple back‑off until a
    strict decrease of the maximal entry is observed (verified with the
    exact convolution).  The procedure runs until the time budget is
    exhausted.
    """
    a = a0.copy()
    conv = _conv_self(a)
    best_a = a.copy()
    best_max = conv.max()
    it = 0

    while time.time() < deadline:
        k_max = int(np.argmax(conv))

        # indices that appear in the maximal entry
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)

        # contribution of each index to conv[k_max]
        contrib = a[i_range] * a[k_max - i_range]
        src = i_range[np.argmax(contrib)]

        # choose a destination that is (almost) empty
        # global smallest entry, but never the source itself
        mask = np.arange(n) != src
        dest = int(np.argmin(np.where(mask, a, np.inf)))

        # linear‑approx update vector Δ = 2·(arr_dst – arr_src)
        delta = np.zeros(2 * n - 1, dtype=float)
        delta[src:src + n] -= 2.0 * a
        delta[dest:dest + n] += 2.0 * a

        prev_max = conv.max()
        eps = a[src] * 0.5                # start with a reasonably large step
        moved = False

        while eps > min_eps:
            # cheap linear check – if this already cannot improve, give up early
            if (conv + eps * delta).max() >= prev_max - 1e-12:
                eps *= 0.5
                continue

            # try the move with the exact convolution
            old_src = a[src]
            old_dst = a[dest]
            a[src] = old_src - eps
            a[dest] = old_dst + eps
            conv_new = _conv_self(a)
            if conv_new.max() < prev_max - 1e-12:
                # successful move
                conv = conv_new
                moved = True
                cur_max = conv.max()
                if cur_max < best_max - 1e-12:
                    best_max = cur_max
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] det‑refine improve: "
                          f"iter={it:,}, max={best_max:.8f}")
                it += 1
                break
            else:
                # revert and try a smaller step
                a[src] = old_src
                a[dest] = old_dst
                eps *= 0.5

        if not moved:
            # nothing worked – give the algorithm a chance to reshuffle a bit
            # tiny random jitter (kept extremely small to keep feasibility)
            jitter = np.random.default_rng().uniform(-1e-12, 1e-12, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            conv = _conv_self(a)

    return best_a
