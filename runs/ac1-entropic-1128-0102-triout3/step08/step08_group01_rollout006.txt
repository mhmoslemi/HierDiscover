def _run_exponential_gradient(a0, n, deadline, start_time, alphas, steps_per_alpha):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    L = _next_pow2(2*n-1)
    it = 0
    for alpha in alphas:
        for _ in range(steps_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] expgrad improve: iter={it}, alpha={alpha}, score={cur_score:.6f}")
            # smoothed max weights
            w = np.exp(alpha * conv - np.max(alpha * conv))
            w = w / w.sum()  # shape length 2n-1
            # compute gradient = 2 * (a * reversed w) convolution
            # grad_i = 2 * sum_{j} w_{i+j} * a_j? Let's derive:
            # We have smoothed objective H = (1/alpha) log sum exp(alpha * conv_k). Gradient dH/ d a_i = sum_k w_k * d(conv_k)/d a_i.
            # d(conv_k)/ da_i = a_{k-i} + a_{i-(k-i)}? Actually conv_k = sum_j a_j a_{k-j}. Derivative wrt a_i = a_{k-i} + a_{i - (k-i)}? 
            # But due to symmetry we can compute via correlation: grad = 2 * (a * w_rev) convolution.
            # Use method similar to previous: compute FFT of a and w_rev and do product.
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev)-1 - np.arange(n)]
            # mirror descent update: a_i <- a_i * exp(-eta * grad_i)
            eta = 0.1 / math.sqrt(it + 1) # schedule
            a = a * np.exp(-eta * grad)
            # renorm to simplex
            a = _project_simplex(a)
            it += 1
    return best_a, best_score

Symmetric mirror‑descent + sub‑gradient + annealing optimiser
-----------------------------------------------------------

* Enforce a≈a[::-1] at every iteration – the optimal step function is
  symmetric, and this reduces the search space dramatically.
* Stage 1 – Mirror descent on a log‑sum‑exp smooth approximation of the
  max‑convolution.  The update is performed on the simplex using the
  entropic (exponential) mirror map, which yields larger, more stable
  moves than vanilla gradient descent.
* Stage 2 – Projected sub‑gradient descent with momentum on the true
  objective (2·n·max(conv)), again symmetrised after each step.
* Stage 3 – Targeted simulated‑annealing that moves probability mass
  away from the pairs that dominate the current peak.
* Optional short random‑restart loops if time permits.
* The whole pipeline respects the 1000 s budget and returns a plain
  Python list suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution a*a via FFT (length 2·len(a)‑1).
    Result is a real numpy array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """
    Enforces a_i = a_{n-1-i}.  The sum is preserved.
    """
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Stage 1 – Mirror descent on smoothed max
# ----------------------------------------------------------------------
def _run_mirror_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        steps_per_alpha: int = 12000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponential‑gradient) on the log‑sum‑exp
    approximation of max(conv(a,a)).  Returns the best iterate seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)

    it = 0
    for alpha in alphas:
        for _ in range(steps_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # current objective (exact, not smoothed)
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] mirror improve: "
                      f"iter={it:,} alpha={alpha:.2f} score={cur_score:.6f}")

            # ----- smoothed max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)      # stability trick
            w /= w.sum()                         # size 2n‑1, sums to 1

            # ----- gradient of the smoothed max -----
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- exponential‑gradient (mirror descent) update -----
            # learning‑rate schedule – aggressive early, 1/√t later
            eta = 0.8 / math.sqrt(it + 1.0)
            a = a * np.exp(-eta * grad)

            # keep on simplex and enforce symmetry
            a = _project_simplex(a)
            a = _symmetrize(a)

            it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent (with momentum) on f(a)=2·n·max(conv(a,a)).
    The iterate is symmetrised after each step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.30          # a bit larger than the former 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,} score={cur_score:.6f}")

        # gradient from the current maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        a = _project_simplex(a)
        a = _symmetrize(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – Targeted simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-8,
                            steps: int = 180_000) -> tuple[np.ndarray, float]:
    """
    Annealing that always picks the current maximal convolution index
    and moves a small amount of probability mass away from the pair of
    entries that contribute most to that peak.  The sequence is forced
    to stay symmetric.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate current maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contributions of each pair to the peak
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n // 2)   # only need left half because of symmetry

        # mirror index of src (the partner in the pair)
        partner = k - src
        if partner < 0 or partner >= n:
            # fallback – pick a random index on the left half
            src = rng.integers(n // 2)
            partner = k - src

        # choose a destination that is not part of this dominant pair
        dst = rng.integers(n // 2)
        if dst == src or dst == partner:
            continue

        # amount to move (up to 15 % of the source mass)
        eps = a[src] * rng.random() * 0.15
        if eps <= 1e-15:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[partner] -= eps
        a_prop[dst] += eps
        a_prop[n - 1 - dst] += eps   # keep symmetry

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper – generate simple analytic shapes (always symmetrised)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _shape_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, zero at the ends, peak in the centre."""
    m = (n - 1) // 2
    left = np.arange(1, m + 2)          # 1 .. m+1
    right = left[:-1][::-1] if n % 2 == 0 else left[::-1]
    a = np.concatenate([left, right])
    a = a.astype(float)
    a /= a.sum()
    return a


def _shape_half_sine(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n) – strictly positive and symmetric."""
    xs = np.arange(n) + 0.5
    a = np.sin(np.pi * xs / n)
    a = _symmetrize(a)          # enforce exact symmetry
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _shape_exponential(n: int, beta: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    centre = (n - 1) / 2.0
    xs = np.arange(n)
    a = np.exp(-beta * np.abs(xs - centre))
    a = _symmetrize(a)
    a /= a.sum()
    return a


def _shape_gaussian(n: int, beta: float) -> np.ndarray:
    """Symmetric discrete Gaussian."""
    centre = (n - 1) / 2.0
    xs = np.arange(n)
    a = np.exp(-beta * (xs - centre) ** 2)
    a = _symmetrize(a)
    a /= a.sum()
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the supplied
    ``evaluate_sequence`` under the given time budget (≈1000 s).
    The routine combines analytic pilots, a mirror‑descent stage,
    a sub‑gradient stage and a targeted annealing polish, all with
    enforced symmetry.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem dimension (the previous best used 4096)
    # ------------------------------------------------------------------
    final_n = 4096

    # ------------------------------------------------------------------
    # 0) Build a small library of analytic starter shapes and pick the best.
    # ------------------------------------------------------------------
    candidates = []

    # uniform & triangular & half‑sine are always worth a look
    candidates.append(_shape_uniform(final_n))
    candidates.append(_shape_triangular(final_n))
    candidates.append(_shape_half_sine(final_n))

    # sweep a few exponential decays
    for beta in [0.0005, 0.001, 0.0025, 0.005, 0.01, 0.02]:
        candidates.append(_shape_exponential(final_n, beta))

    # sweep a few Gaussians
    for beta in [0.0005, 0.001, 0.0025, 0.005, 0.01]:
        candidates.append(_shape_gaussian(final_n, beta))

        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            prev = _symmetrize(prev)
            candidates.append(prev)
    except Exception:
        pass

    # evaluate all pilots (cheap, no heavy loops)
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        # make sure we have a proper simplex vector
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        cand = _symmetrize(cand)
        score = evaluate_sequence(cand.tolist())
        print(f"[{time.time() - start_time:.2f}s] pilot {idx} score = {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 1) Mirror‑descent stage on smoothed max (≈10 % of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        phase1_deadline = time.time() + max(5.0, 0.12 * remaining)
        a_mirror, score_mirror = _run_mirror_descent(
            best_seq,
            final_n,
            phase1_deadline,
            start_time,
            alphas=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0],
            steps_per_alpha=15000,
        )
        if score_mirror < best_score:
            best_score = score_mirror
            best_seq = a_mirror.copy()
            print(f"[{time.time() - start_time:.2f}s] mirror stage improve: "
                  f"score={score_mirror:.6f}")

    # ------------------------------------------------------------------
    # 2) Sub‑gradient descent (major chunk of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.70 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad stage improve: "
                  f"score={score_sub:.6f}")

        # a brief second run with a larger initial step to hop out of plateaus
        if time.time() < deadline:
            extra_deadline = time.time() + min(12.0, deadline - time.time())
            a_sub2, score_sub2 = _run_subgradient_descent(
                best_seq,
                final_n,
                extra_deadline,
                start_time,
                best_score,
                max_iters=250_000,
            )
            if score_sub2 < best_score:
                best_score = score_sub2
                best_seq = a_sub2.copy()
                print(f"[{time.time() - start_time:.2f}s] extra sub‑grad improve: "
                      f"score={score_sub2:.6f}")

    # ------------------------------------------------------------------
    # 3) Targeted annealing polish (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_targeted_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-9,
            steps=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal final improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional random restarts while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        block = min(28.0, (deadline - time.time()) / 2.0)
        restart_deadline = time.time() + block

        # random perturbation of the current best (little Gaussian noise)
        pert = best_seq + 0.02 * np.random.randn(final_n)
        pert = np.maximum(pert, 0.0)
        pert = _symmetrize(pert)
        pert /= pert.sum()

        # short sub‑gradient burst
        pert_seq, pert_score = _run_subgradient_descent(
            pert,
            final_n,
            restart_deadline,
            start_time,
            best_score,
            max_iters=140_000,
        )
        # quick anneal
        if time.time() < deadline:
            pert_seq, pert_score = _run_targeted_annealing(
                pert_seq,
                final_n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-8,
                steps=100_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: "
                  f"score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
