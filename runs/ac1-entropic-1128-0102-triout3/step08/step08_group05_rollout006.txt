"""
Multi‑scale stochastic optimiser for the max‑self‑convolution objective.

The routine works inside the global time budget (default 1000 s) and
combines several ideas that differ from the reference implementation:

1.  **Larger problem sizes.**  The search proceeds up to 16384 points.
    More degrees of freedom allow the self‑convolution to be spread
    more evenly, which can lower the 2·n·max(conv) bound.

2.  **Multiple initial shapes.**  For each size we start from several
    qualitatively different distributions (uniform, random, exponential
    decay, cosine‑square) and from the best solution of the previous
    size.

3.  **Three optimisation phases per restart**
    – Adam on a smooth log‑sum‑exp surrogate of max(conv) (continuation)
    – Projected sub‑gradient descent on the true objective
    – Exponentiated‑gradient (mirror‑descent) on the true objective.

4.  **Final light simulated‑annealing** that moves a random fraction of
    mass between two cells; this helps escape tiny local traps.

All vectors are kept on the probability simplex (non‑negative, sum = 1),
so the evaluation function `evaluate_sequence` is scale‑invariant.
The routine respects the time limit; if the budget expires it simply
returns the best vector found so far.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the true objective)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0).
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad

# Adam optimiser on the soft‑max (log‑sum‑exp) surrogate of max‑conv
# ----------------------------------------------------------------------

                      n: int,

                      tau0: float,

                      tau_decay: float = 0.9995,
                      step_size: float = 0.25,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        Oτ(a) = 2·n·τ·log∑exp(conv(a)/τ)
    using Adam + projection onto the simplex.
    Returns the best vector (according to the true objective) and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true score for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam iter={it:,} "
                  f"n={n} score={cur_score:.6f} τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)         # numerical stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # p sums to 1, length 2·n‑1

        # gradient of Oτ : 4·n·∑ a_j·p_{i+j}
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)    # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]  # slice of length n
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))   (a lies on the simplex).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad iter={it:,} "
                  f"n={n} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (entropy‑mirror) descent on the true objective
# ----------------------------------------------------------------------


def _run_exponentiated_gradient(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               step0: float = 0.20,
                               max_iter: int = 5 * 10 ** 6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with the entropy Bregman divergence.
    Update: a ← a ⊙ exp(−η·grad)  followed by renormalisation.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG iter={it:,} "
                  f"n={n} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        eta = step0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)               # multiplicative update
        s = a.sum()
        if s > 0.0:
            a /= s
        else:                                     # rare numerical under‑flow
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that moves a random fraction of mass between two cells.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source proportionally to its current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal iter={it:,} "
                  f"n={n} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Helper to generate a few deterministic initial shapes
# ----------------------------------------------------------------------


def _initial_shapes(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list of plausible starting vectors of length `n`."""
    shapes = []

    # uniform
    u = np.full(n, 1.0 / n)
    shapes.append(u)

    # random Dirichlet‑like
    r = rng.random(n)
    r /= r.sum()
    shapes.append(r)

    # exponential decay
    x = np.arange(n)
    lam = 5.0 / n   # controls decay speed
    e = np.exp(-lam * x)
    e /= e.sum()
    shapes.append(e)

    # cosine‑square (a smooth bell)
    c = np.sin(np.pi * (x + 0.5) / n) ** 2
    c /= c.sum()
    shapes.append(c)

    # linear ramp (0 → 1)
    l = x.astype(float)
    l /= l.max()
    l /= l.sum()
    shapes.append(l)

    return shapes


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine follows a multi‑scale, multi‑restart strategy described in the
    module docstring and respects the global time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Load a previously known construction if it exists.
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Progressive up‑sampling schedule (larger sizes improve the bound).
    # ------------------------------------------------------------------
    size_schedule = [4096, 8192, 16384]

    best_seq = None
    best_score = float("inf")

    for n in size_schedule:
        if time.time() >= deadline:
            break

        print(f"[{time.time() - start_time:6.2f}s] === Starting size {n} ===")

        # ------------------------------------------------------------------
        # Base vector for this size (upsampled best so far or a starter)
        # ------------------------------------------------------------------
        if best_seq is not None:
            base_vec = _upsample(best_seq, n)
        elif starter is not None:
            base_vec = _upsample(starter, n)
        else:
            base_vec = np.full(n, 1.0 / n)

        # ------------------------------------------------------------------
        # Run a handful of different initialisations / restarts.
        # ------------------------------------------------------------------
        init_vectors = _initial_shapes(n, rng)
        # also add a noisy copy of the upsampled best (if any)
        if best_seq is not None:
            noisy = best_seq.copy()
            noisy += 0.02 * rng.random(n)            # small perturbation
            noisy = np.maximum(noisy, 0.0)
            noisy /= noisy.sum()
            init_vectors.append(noisy)

        # Allocate roughly equal time to each restart (but never exceed the deadline)
        for restart_id, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break

            # Slight random perturbation to avoid identical runs
            a = a0.copy()
            a += 1e-3 * rng.random(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

            # --------------------------------------------------------------
            # Phase 1 – Adam on the smoothed surrogate
            # --------------------------------------------------------------
            conv0 = _conv_self(a)
            tau0 = max(conv0.max() * 10.0, 1e-8)
            phase_deadline = min(deadline,
                                 time.time() + (deadline - time.time()) * 0.15)
            a, _ = _run_adam_softmax(a0=a,
                                     n=n,
                                     deadline=phase_deadline,
                                     start_time=start_time,
                                     tau0=tau0,
                                     tau_end=1e-13,
                                     tau_decay=0.9995,
                                     step_size=0.30)
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam improve (n={n}) "
                      f"restart={restart_id} score={cur_score:.6f}")

            # --------------------------------------------------------------
            # Phase 2 – Projected Euclidean sub‑gradient descent
            # --------------------------------------------------------------
            if time.time() < deadline:
                phase_deadline = min(deadline,
                                     time.time() + (deadline - time.time()) * 0.35)
                a, _ = _run_subgradient_descent(a0=a,
                                                n=n,
                                                deadline=phase_deadline,
                                                start_time=start_time,
                                                beta=0.9,
                                                step0=0.30)
                cur_score = 2.0 * n * _conv_self(a).max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Subgrad improve (n={n}) "
                          f"restart={restart_id} score={cur_score:.6f}")

            # --------------------------------------------------------------
            # Phase 3 – Exponentiated‑gradient mirror descent
            # --------------------------------------------------------------
            if time.time() < deadline:
                phase_deadline = min(deadline,
                                     time.time() + (deadline - time.time()) * 0.30)
                a, _ = _run_exponentiated_gradient(a0=a,
                                                   n=n,
                                                   deadline=phase_deadline,
                                                   start_time=start_time,
                                                   step0=0.20)
                cur_score = 2.0 * n * _conv_self(a).max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] EG improve (n={n}) "
                          f"restart={restart_id} score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – Light simulated‑annealing refinement (final seconds)
        # --------------------------------------------------------------
        if time.time() < deadline:
            a, _ = _run_annealing(a0=best_seq,
                                  n=n,
                                  deadline=deadline,
                                  start_time=start_time,
                                  cur_best_score=best_score,
                                  cur_best_seq=best_seq,
                                  start_temp=2e-2,
                                  end_temp=1e-7,
                                  steps=250_000)
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Final anneal improve (n={n}) "
                      f"score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # Keep the best vector of the current size for up‑sampling.
        # ------------------------------------------------------------------
        if best_seq is not None:
            # Normalise once more (numerical drift)
            best_seq = np.maximum(best_seq, 0.0)
            s = best_seq.sum()
            if s > 0.0:
                best_seq /= s
            else:
                best_seq = np.full_like(best_seq, 1.0 / n)

        print(f"[{time.time() - start_time:6.2f}s] End of size {n} – "
              f"best score ≈ {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] Finished – global best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
