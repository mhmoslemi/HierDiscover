"""
search for a non‑negative step function that (almost) minimises
evaluate_sequence.

Strategy
--------
1.  Assemble a diverse pool of seeds (uniform, exponential, Gaussian,
    several Dirichlet draws and the previous best if available).  All
    seeds are symmetrised (a ⇢ (a+a[::-1])/2) and normalised to sum‑1.

2.  For every seed run a *smooth‑max* Adam optimiser.
    The max‑convolution is approximated by a log‑sum‑exp with a
    temperature α that is raised from 0.5 to 5000.  This yields a smooth
    gradient that gradually focuses on the true maximum.  The best
    candidate of all runs is kept.

3.  Refine the survivor with a *projected sub‑gradient* method that
    uses the exact max‑convolution (no smoothing).  The sub‑gradient is
    computed analytically (no double‑counting) and a Nesterov‑style
    momentum with a relatively large initial step (0.6) is employed.
    If no improvement is observed for a while the step size is
    re‑initialised to escape possible plateaus.

4.  A deterministic greedy “peak‑reduction’’ sweep moves a modest
    fraction of the coefficient that contributes most to the current
    convolution peak towards the smallest coefficient, accepting only
    strictly improving moves.

All phases respect the global time budget (default 1000 s).  The final
vector is projected onto the probability simplex, clipped to [0,∞) and
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + w_rev.shape[0] - 1]

    # Extract the centred part of length n
    offset = w_rev.shape[0] - 1
    grad = 2.0 * corr[offset:offset + a.shape[0]]
    return grad


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i}."""
    return (v + v[::-1]) * 0.5


def _normalise(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and normalise to sum‑one."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    centre = (n - 1) / 2.0
    x = np.arange(n)
    g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.98) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution (log‑sum‑exp) objective.
    Returns the best vector (probability simplex) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    # work in log‑space to enforce simplex via soft‑max
    y = np.log(np.maximum(a, eps_log))

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – increasingly sharp approximation of the max
    alphas = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 5000.0]
    iters_per_alpha = 3000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15          # a slightly larger base learning rate
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # obtain a valid simplex vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # compute smoothed objective and its gradient
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha           # smooth approximation of max(conv)
            obj = 2.0 * n * f_val

            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # back‑prop through the soft‑max (Jacobian of soft‑max)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.0f}, score={sc:.6f}")

    # final evaluation on the latest a
    sc = evaluate_sequence(a.tolist())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
        print(f"[{time.time() - start_time:6.2f}s] Adam final improve: score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – exact projected sub‑gradient descent (correct gradients)
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.6,
                      restart_patience: int = 120000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses the exact sub‑gradient (no double counting) and restarts the
    step size when progress stalls.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    it = 0
    last_improve_iter = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve_iter = it
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # --------------------------------------------------------------
        # exact sub‑gradient of max(conv) – sum of gradients of active peaks
        # --------------------------------------------------------------
        rel_tol = 1e-12
        active = np.where(conv >= cur_max * (1.0 - rel_tol))[0]

        # gradient for a single peak k is 2 * a shifted by k
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # contribution 2 * a[j] to gradient of a[i]
            g[i] += 2.0 * a[j]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

        # ------------------------------------------------------------------
        # restart logic – if no improvement for a while, increase step0 a bit
        # ------------------------------------------------------------------
        if it - last_improve_iter > restart_patience:
            step0 = min(step0 * 1.3, 1.2)   # modest boost
            last_improve_iter = it
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad restart: step0 now {step0:.3f}")

    # final evaluation
    final_score = evaluate_sequence(best_a.tolist())
    if final_score < best_score:
        best_score = final_score
        best_a = best_a.copy()
    return best_a, best_score

# Phase 3 – greedy peak reduction (more aggressive)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of the coefficient that most contributes
    to the current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing peaks.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # index that contributes most

        # destination = index with smallest value (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # move a modest fraction (tuned for aggressive descent)
        eps = 0.35 * a[src]                     # 35 % of source mass
        if eps < 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max - 1e-12:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: iter={it}, max={new_max:.6f}")

    return a, best_score

    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin so we never exceed the budget
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 4800 works well with the FFT‑based convolutions.
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # Build a pool of start vectors.
    # ------------------------------------------------------------------
    starts = []

    # a) previous construction if available
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear up‑sample to the required length
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = _normalise(seq)
            seq = _symmetrize(seq)
            starts.append(seq)
    except Exception:
        pass

    # b) deterministic diversifiers
    starts.append(_normalise(np.full(n, 1.0 / n)))           # uniform
    starts.append(_normalise(_exponential_start(n, r=0.985)))  # exponential
    starts.append(_normalise(_gaussian_start(n)))          # Gaussian bell
    starts.append(_normalise(_dirichlet_start(n, conc=2.0, rng=rng)))  # smooth Dirichlet
    starts.append(_normalise(_dirichlet_start(n, conc=0.5, rng=rng)))  # spiky Dirichlet

    # c) a few extra random Dirichlet draws with varying concentration
    for conc in [0.3, 0.7, 1.5, 3.0]:
        for _ in range(2):
            vec = _normalise(_dirichlet_start(n, conc=conc, rng=rng))
            starts.append(_symmetrize(vec))

    # ------------------------------------------------------------------
    # Exploration – Adam on the smooth max‑convolution.
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.30 * budget_s   # 30 % of total budget
    best_seq = None
    best_score = float("inf")

    for idx, a0 in enumerate(starts):
        if time.time() >= explore_deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] Exploration start #{idx + 1}")
        cand_seq, cand_score = _adam_optimize(a0, explore_deadline, start_time)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (explore) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Polishing – exact sub‑gradient descent with corrected gradients.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] Polishing phase start")
        polished_seq, polished_score = _subgrad_optimize(
            best_seq,
            deadline,
            start_time,
            beta=0.94,
            step0=0.6,
            restart_patience=120000,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (polish) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy peak‑reduction.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] Greedy peak‑reduction start")
        final_seq, final_score = _greedy_local_improve(
            best_seq,
            deadline,
            start_time,
            max_iters=8000,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (greedy) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and safety checks.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = _normalise(_symmetrize(best_seq))

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
