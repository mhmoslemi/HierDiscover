def _compress(seq, new_n):
    old_n = seq.shape[0]
    if new_n >= old_n:
        return seq.copy()
    compressed = np.empty(new_n, dtype=float)
    for i in range(new_n):
        start = int(i * old_n / new_n)
        end = int((i+1) * old_n / new_n)
        if end <= start:
            end = start + 1
        compressed[i] = seq[start:end].mean()
    # normalize sum
    s = compressed.sum()
    if s>0:
        compressed /= s
    else:
        compressed = np.full_like(compressed, 1.0/new_n)
    return compressed

"""
Multi‑scale optimizer for the step‑function bound.

The algorithm proceeds in three stages under a single 1000 s budget:

1. **Low‑dimensional search** (≈20 % of the time):
   – Reduce the problem to a modest size (≈800 coefficients) by block‑averaging.
   – Run a short Adam phase on a smooth log‑softmax surrogate of the max‑convolution,
     followed by a long exponentiated‑gradient (sub‑gradient) descent on the true max.
   – Keep the best low‑dim solution.

2. **Upsampling**:
   – Linearly interpolate the best low‑dim solution to the full length (default 4800)
     and renormalise it to the probability simplex.

3. **Full‑dim refinement** (≈80 % of the time):
   – Exponentiated‑gradient descent on the true max‑convolution (most of the remaining time).
   – Deterministic greedy peak‑reduction to eliminate the dominant contributions.
   – A short simulated‑annealing pass for final fine‑tuning.

All intermediate steps respect non‑negativity, normalise the vector to sum = 1,
and never let any entry exceed the evaluator’s 1000‑limit.  The routine prints
progress and finally returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _softmax_from_logits(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


def _smooth_max_grad(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth approximation (log‑sum‑exp) of max(conv)
    w.r.t. a.
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                       # soft‑max weights
    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _max_subgrad(a: np.ndarray,
                 conv: np.ndarray,
                 rel_tol: float,
                 fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution objective.
    The active set contains all entries conv[k] ≥ (1‑rel_tol)·max(conv).
    """
    cur_max = conv.max()
    thresh = cur_max * (1.0 - rel_tol)
    active = (conv >= thresh).astype(np.float64)   # binary mask
    w_rev = active[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum (renormalised afterwards)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _compress(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Block‑averaging compression from len(seq) → new_n (preserves sum)."""
    old_n = seq.shape[0]
    if new_n >= old_n:
        return seq.copy()
    compressed = np.empty(new_n, dtype=float)
    for i in range(new_n):
        start = int(i * old_n / new_n)
        end = int((i + 1) * old_n / new_n)
        if end <= start:
            end = start + 1
        block = seq[start:end]
        compressed[i] = block.mean() if block.size else 0.0
    total = compressed.sum()
    if total > 0:
        compressed /= total
    else:
        compressed = np.full_like(compressed, 1.0 / new_n)
    return compressed

def _adam_phase(a0: np.ndarray,
                deadline: float,
                start_time: float,
                fft_len: int) -> np.ndarray:
    """
    Adam optimiser on the smooth (log‑sum‑exp) surrogate of max(conv).
    Returns the best simplex point seen during the phase.
    """
    y = np.log(np.maximum(a0, 1e-16))        # logits
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # increasingly sharp approximations
    alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 4000          # fewer than the original – we rely on the later phase

    best_seq = a0.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    for alpha in alpha_sched:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq
            a = _softmax_from_logits(y)

            # smooth objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * a.shape[0] * f_val

            # gradient wrt a
            grad_a = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * a.shape[0])

            # chain rule to logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f},"
                          f" iter={i:,}, score={sc:.6f}")

    return best_seq


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Long exponentiated‑gradient descent on the true max‑convolution.
    Returns the best sequence seen and its (2·n·max_conv) score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    best_a = a.copy()
    cur_conv = _conv_self(a)
    best_score = 2.0 * n * cur_conv.max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: iter={it:,},"
                  f" score={cur_score:.6f}")

        # sub‑gradient of the max (active set)
        grad = _max_subgrad(a, conv, rel_tol=1e-8, fft_len=fft_len)

        # multiplicative step (always stays non‑negative)
        step = 0.5 / math.sqrt(it)          # a slightly larger initial step
        a = a * np.exp(-step * grad)

        # Euclidean projection onto the simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """Deterministic greedy moves that lower the current convolution peak."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # choose destination with the smallest mass that is not src
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.4              # a bit more aggressive than the original 0.3
        if eps < 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """Classic simulated‑annealing mass‑transfer moves."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.35   # allow slightly larger jumps
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,},"
                  f" score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The method uses a multi‑scale scheme:
    a cheap low‑dim optimisation (Adam + sub‑gradient), upsampling,
    then an extensive full‑dim sub‑gradient phase, deterministic peak‑reduction,
    and a short annealing pass.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the host environment
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        starter_seq = np.asarray(height_sequence_1, dtype=float)
        if starter_seq.ndim == 1 and starter_seq.size > 0:
            n = starter_seq.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4800
        starter_seq = None

    # --------------------------------------------------------------
    # Low‑dimensional stage (≈20 % of total budget)
    # --------------------------------------------------------------
    m = min(800, n)                     # low‑dim size (fast FFTs)
    low_starts = []

    low_starts.append(np.full(m, 1.0 / m))                # uniform
    r = 0.985
    exp_vec = np.exp(np.arange(m) * np.log(r))
    low_starts.append(exp_vec / exp_vec.sum())           # exponential decay
    idx = np.arange(m)
    u_shape = 1.0 / (np.minimum(idx + 1, m - idx) + 1.0)
    low_starts.append(u_shape / u_shape.sum())           # symmetric U‑shape
    rng = np.random.default_rng()
    noise = rng.normal(size=m)
    smooth = np.convolve(noise, np.ones(15) / 15, mode='same')
    smooth = np.clip(smooth, 0, None)
    if smooth.sum() == 0:
        smooth = np.full(m, 1.0 / m)
    else:
        smooth = smooth / smooth.sum()
    low_starts.append(smooth)                            # random smooth

    # optionally compress the provided global solution
    if starter_seq is not None and n != m:
        low_starts.append(_compress(starter_seq, m))

    low_deadline = start_time + budget_s * 0.2
    best_low = None
    best_low_score = float('inf')
    for idx_start, a0 in enumerate(low_starts):
        if time.time() >= low_deadline:
            break
        a = np.maximum(a0, 0.0)
        if a.sum() > 0:
            a /= a.sum()
        else:
            a = np.full_like(a, 1.0 / m)

        # short Adam phase (more aggressive schedule)
        adam_deadline = time.time() + (low_deadline - time.time()) * 0.15
        fft_len_low = _next_pow2(3 * m - 2)
        a_adam = _adam_phase(a, adam_deadline, start_time, fft_len_low)

        # long sub‑gradient descent on the low‑dim problem
        if time.time() < low_deadline:
            a_opt, score_opt = _subgradient_descent(a_adam, low_deadline, start_time)
        else:
            a_opt = a_adam
            score_opt = 2.0 * m * _conv_self(a_opt).max()

        if score_opt < best_low_score:
            best_low_score = score_opt
            best_low = a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] low‑dim run {idx_start+1},"
              f" score={score_opt:.6f}")

    if best_low is None:
        best_low = np.full(m, 1.0 / m)

    # --------------------------------------------------------------
    # Upsample to full dimension
    # --------------------------------------------------------------
    a_full = _upsample(best_low, n)
    a_full = np.maximum(a_full, 0.0)
    if a_full.sum() > 0:
        a_full /= a_full.sum()
    else:
        a_full = np.full_like(a_full, 1.0 / n)

    # --------------------------------------------------------------
    # Full‑dim refinement (≈80 % of total budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        # time ran out – return whatever we have
        final_seq = a_full
        final_score = evaluate_sequence(final_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] time exhausted, score={final_score:.6f}")
        return [float(x) for x in final_seq.tolist()]

    # allocate sub‑gradient, peak‑reduction and annealing proportions
    sub_deadline = time.time() + remaining * 0.68
    a_opt, score_opt = _subgradient_descent(a_full, sub_deadline, start_time)

    # peak reduction – a tighter final polishing
    if time.time() < deadline:
        peak_deadline = time.time() + (deadline - time.time()) * 0.2
        a_opt, score_opt = _run_peak_reduction(a_opt, peak_deadline, start_time)

    # short annealing pass
    if time.time() < deadline:
        anneal_deadline = deadline
        # we limit the iteration count; the deadline will anyway stop earlier
        a_opt, score_opt = _run_annealing(a_opt, anneal_deadline, start_time,
                                          steps=120_000)

    # --------------------------------------------------------------
    # Final sanitisation and reporting
    # --------------------------------------------------------------
    best_seq = np.maximum(a_opt, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
