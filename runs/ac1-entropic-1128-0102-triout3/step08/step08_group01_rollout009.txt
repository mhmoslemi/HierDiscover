# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used after every additive update
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Linear convolution via FFT (self‑convolution a*a)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear self‑convolution of ``a`` using FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())           # next power of two ≥ 2·n‑1
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _adam_smoothing(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    alphas: list[float],
                    lr0: float = 0.6,
                    iters_per_alpha: int = 4000) -> np.ndarray:

    a = a0.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return a
            it += 1

            conv = _conv_self(a)
            # --- smooth max weights ---------------------------------
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)      # stability
            w /= w.sum()
            w_rev = w[::-1]

            # --- gradient of the smooth max -------------------------
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # --- Adam update ----------------------------------------

            beta2_pow *= beta2

        while time.time() < deadline - 5.0:

            block = min(20.0, (deadline - time.time()) / 2.0)
            restart_deadline = time.time() + block

            # perturb current best (or uniform if none yet)
            if best_seq is not None:
                a_restart = best_seq + 0.02 * np.random.randn(final_n)
            else:
                a_restart = np.random.rand(final_n)
            a_restart = np.clip(a_restart, 0.0, None)
            a_restart /= a_restart.sum()

            # short Adam run
            a_restart = _adam_smoothing(a_restart,
                                         deadline=restart_deadline,
                                         start_time=start_time,
                                         alphas=[0.5, 2.0, 10.0],
                                         lr0=0.5,
                                         iters_per_alpha=2000)

            # short mirror descent
            a_restart = _mirror_subgradient(a_restart,
                                            deadline=restart_deadline,
                                            start_time=start_time,
                                            max_iters=200_000)

            # quick anneal
            a_restart = _anneal_targeted(a_restart,
                                          deadline=restart_deadline,
                                          start_time=start_time,
                                          steps=50_000,
                                          start_temp=2e-2,
                                          end_temp=1e-8)

            cur_score = 2.0 * final_n * _conv_self(a_restart).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_restart.copy()
                print(f"[{time.time() - start_time:.2f}s] restart improve "
                      f"(n={final_n}) score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a plain Python list.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 1024
        best_seq = np.full(1024, 1.0 / 1024)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
