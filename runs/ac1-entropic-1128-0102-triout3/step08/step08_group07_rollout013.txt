"""
Propose a step‑function with a lower evaluation score.

Algorithm
---------
1. **Initial sampling** – generate several structured or sparse
   probability vectors of moderate length (n≈2048) and keep the best one.
2. **Short Adam smoothing** – run a few Adam steps on the smooth
   log‑sum‑exp surrogate of the max‑convolution to get a cleaner start.
3. **LP‑based refinement** – for the current vector `f` solve the
   linear program  

        max   Σ b_i
        s.t.  (f * b)[k] ≤ M   ∀k,   b_i ≥ 0

   where `M = max(f * f)`.  The solution `b` is normalised to a
   probability vector and mixed with `f`.  Repeating this step drives the
   true max‑convolution down quickly.
4. **Upsample** the best vector to the final size (n=8192) and keep it
   normalised.
5. **Projected sub‑gradient descent** on the exact objective
   `2·n·max(conv(f,f))` (Nesterov momentum) for a large fraction of the
   remaining budget.
6. **Extended simulated annealing** – many random mass‑transfer moves
   (with a geometric temperature schedule) to escape the final local
   minima.
7. Return the best vector found (always projected onto the simplex).

All stages respect the global time budget (default 1000 s).  The code
uses only NumPy, SciPy (for the linear program) and the provided
`evaluate_sequence` function.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple linear‑algebra helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` for a vector already on the simplex."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for the exact polishing stage)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (∑a = 1, a ≥ 0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser of the soft‑max surrogate O_τ(a)
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9994,
                      step_size: float = 0.18,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """
    Minimise O_τ(a) = 2·n·τ·log Σ exp(conv(a)/τ) by Adam,
    projecting onto the simplex after each step.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true objective for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # for stability
        p = exp_z / np.sum(exp_z)          # shape (2·n‑1,)

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # Σ_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # geometric cooling of τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (Nesterov) sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.96,
                             step0: float = 0.12,
                             max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))    (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Simple SA that transfers a random fraction of mass between two cells.
    Acceptance uses the true objective 2·n·max(conv).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Structured initialisation helpers
# ----------------------------------------------------------------------
def _window_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random but structured non‑negative vector of length n.
    40 % Chebyshev, 40 % Kaiser, otherwise a simple Dirichlet draw.
    """
    try:
        from scipy.signal import chebwin, kaiser
        have_signal = True
    except Exception:
        have_signal = False

    r = rng.random()
    if have_signal and r < 0.4:
        att = rng.uniform(30, 120)
        w = chebwin(n, at=att, sym=True).astype(float)
        w = np.maximum(w, 0.0)
    elif have_signal and r < 0.8:
        beta = rng.uniform(0, 14)
        w = kaiser(n, beta).astype(float)
        w = np.maximum(w, 0.0)
    else:
        w = rng.random(n)
        w = np.maximum(w, 0.0)

    w /= w.sum() if w.sum() > 0 else 1.0
    return w


def _dirichlet_initialiser(n: int, rng: np.random.Generator, alpha: float = 0.5) -> np.ndarray:
    """Sparse Dirichlet draw (alpha < 1 favours sparsity)."""
    draws = rng.gamma(alpha, 1.0, size=n)
    if draws.sum() == 0:
        draws[:] = 1.0 / n
    else:
        draws /= draws.sum()
    return draws


# ----------------------------------------------------------------------
# Linear‑programming refinement (max‑convolution bound)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray):
    """
    Build the matrix A (shape (2·n‑1, n)) such that
        (A @ b)[k] = (f * b)[k]    (linear convolution, zero padding).
    Each column i contains the constant value f[i] on the rows i … i+n‑1.
    """
    from scipy.sparse import csr_matrix

    n = f.shape[0]
    # rows for each column i: i, i+1, …, i+n‑1
    row_idx = np.arange(n)[:, None] + np.arange(n)          # (n, n)
    rows = row_idx.ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(f, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_for_g(f: np.ndarray, M: float):
    """
    Solve    max Σ b_i   s.t.  (f * b)[k] ≤ M  ∀k,  b_i ≥ 0.
    Returns the optimal b (may be all‑zero if the LP fails) .
    """
    from scipy.optimize import linprog

    # tiny regularisation – prevents unbounded columns when f_i ≈ 0
    f_reg = np.maximum(f, 1e-12)

    A = _build_conv_matrix(f_reg)
    b_ub = np.full(2 * len(f) - 1, M)
    c = -np.ones(len(f))                     # maximise Σb  → minimise -Σb
    bounds = [(0, None) for _ in range(len(f))]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None
    return res.x


def _lp_refinement(f0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Alternate:

        f ← (1‑t)·f + t·(b/Σb)

    where b solves the LP described above.
    Returns the best vector seen during the process.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _objective(f)
    iteration = 0
    t_mix = 0.5   # convex‑combination weight

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(f)
        M = conv.max()

        b = _solve_lp_for_g(f, M)
        if b is None:
            break

        s = b.sum()
        if s <= 0:
            break
        b_norm = b / s

        f = (1 - t_mix) * f + t_mix * b_norm
        f = _project_simplex(f)

        cur_score = _objective(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve: iter={iteration}, score={cur_score:.6f}")

    return best_f, best_score

    best_score = float("inf")

    # ------------------------------------------------------------------

            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)

                starter = tmp / tmp.sum()

                                             tau0=tau0, tau_end=1e-13,

        if score_opt < best_score:

    # ------------------------------------------------------------------
    n_final = 8192
    best_seq = _upsample(best_seq, n_final)

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient descent on the true objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # reserve a couple of seconds for the final annealing stage
        grad_deadline = deadline - 2.0
        a2, score2 = _run_subgradient_descent(best_seq, n_final,
                                              grad_deadline, start_time,
                                              beta=0.97, step0=0.13)
        if score2 < best_score:
            best_score = score2
            best_seq = a2.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 subgrad improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – extended simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(best_seq, n_final,
                                          deadline, start_time,
                                          best_score, best_seq,
                                          start_temp=2e-2, end_temp=1e-7,
                                          steps=1_200_000)   # more moves than before
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation onto the simplex (guard against tiny drift)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
