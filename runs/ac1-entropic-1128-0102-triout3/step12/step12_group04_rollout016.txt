        n = 4096

"""
Improved search for a low‑objective non‑negative step sequence.

The algorithm consists of three phases

1. **Warm‑start** – generate a few analytic shapes (tent, cosine,
   half‑circle, Gaussian, power‑tent) and a handful of random
   Dirichlet vectors.  The best of them (according to the exact
   objective) is taken as the initial point.

2. **Soft‑max sub‑gradient descent** – minimise a smooth proxy of the
   true objective:
        G(a) = (2·n/α)·log Σ_k exp(α·conv_k(a,a))
   whose gradient is a weighted sum of the individual convolution
   entries with soft‑max weights.  α is increased gradually so the
   proxy approaches the true max.  After each gradient step the vector
   is projected onto the probability simplex (non‑negative, sum = 1).

   A diminishing step size η_t = η₀ / √t together with Euclidean
   projection provides stable progress and allows the method to
   escape shallow plateaus.

3. **Light simulated‑annealing polish** – a Metropolis walk that
   moves a random packet of mass between two indices.  It finishes
   the remaining time budget and can improve the result further.

All heavy work (self‑convolution and the weighted convolution needed
for the gradient) is performed with NumPy FFTs, which are fast for
powers‑of‑two.  The whole search respects the 1000 s budget and returns
a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)
    return conv_full[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a normalised (Σ a = 1) vector.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        return w
    # fallback – uniform point
    return np.full_like(v, 1.0 / n)


# ----------------------------------------------------------------------
# Warm‑starts
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Analytic candidate vectors (already normalised to sum = 1)."""
    x = np.linspace(0.0, 1.0, n, dtype=float)

    # 1) symmetric linear tent
    tent = np.minimum(x, 1.0 - x)

    # 2) half‑cosine
    cosine = np.sin(np.pi * x)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

    # 5) power‑tent (flatter than linear)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Soft‑max gradient of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      alpha: float,
                      L2: int) -> np.ndarray:
    """
    Gradient of the soft‑max weighted sum of convolution entries.
    Returns the vector  ∂/∂a Σ w_k·conv_k  where
        w_k = exp(α·conv_k) / Σ_j exp(α·conv_j).
    The gradient is computed by an FFT‑based convolution.
    """
    max_c = conv.max()
    # stable soft‑max
    exp_vals = np.exp(alpha * (conv - max_c))
    w = exp_vals / exp_vals.sum()

    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L2)
    Af = np.fft.rfft(a_rev, n=L2)
    conv_wr = np.fft.irfft(Wf * Af, n=L2)

    n = a.shape[0]
    # gradient of the weighted sum (no outer 2·n factor yet)
    grad = 2.0 * conv_wr[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Soft‑max sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth approximation of the objective using a
    diminishing‑step soft‑max sub‑gradient method.
    Returns the best iterate encountered and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT sizes
    L_self = _next_pow2(2 * n - 1)          # for a*a
    L_grad = _next_pow2(3 * n)              # for weighted convolution

    best_a = a.copy()
    best_score = _objective(a)

    # hyper‑parameters
    step0 = 0.6            # initial step size
    alpha = 5.0            # soft‑max temperature
    alpha_max = 200.0
    alpha_inc = 1.02       # multiplicative increase per iteration
    it = 0

    while time.time() < deadline:
        it += 1

        # self‑convolution and current objective
        conv = _conv_self(a, L_self)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad iter={it:,} "
                  f"score={cur_score:.6f}")

        # gradient of the smooth proxy
        grad = _softmax_gradient(a, conv, alpha, L_grad)

        # gradient of the proxy G(a) = (2·n/α)·log Σ exp(α·conv)
        grad_obj = (2.0 * n) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # update + projection onto the simplex
        a = a - step * grad_obj
        a = _project_simplex(a)

        # slowly sharpen the soft‑max
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_inc)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same idea as the reference code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk that moves a random packet of mass between two indices."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of proposals – scaled to the remaining time budget
    total_steps = 80_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # very fast exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)            # source proportional to current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Determine problem size – if a global starter exists use its length,
    # otherwise fall back to 4096 (power of two → fast FFTs).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
        else:
            n = 4096
    except Exception:
        n = 4096

    # --------------------------------------------------------------
    # 1) Warm‑start – a collection of simple analytic shapes + random draws.
    # --------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # uniform vector
    warm_starts.append(np.full(n, 1.0 / n))

    # a few Dirichlet‑type random vectors (α = 1 gives exponential tails)
    for _ in range(12):
        x = np.random.gamma(1.0, 1.0, size=n)
        if x.sum() > 0:
            warm_starts.append(x / x.sum())

    # include the global starter if present and valid
    try:
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            hs = np.maximum(hs, 0.0)
            if hs.sum() > 0:
                warm_starts.append(hs / hs.sum())
    except Exception:
        pass

    # evaluate all warm starts and keep the best
    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Main optimisation – soft‑max sub‑gradient descent.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                     # keep a few seconds for polishing
        gd_deadline = time.time() + remaining * 0.85
        a_opt, score_opt = _subgradient_optimize(best_a, gd_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad best score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing polish (if any time remains).
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _anneal(best_a, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal best score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation & conversion to plain list.
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    return [float(v) for v in best_a]
