"""
Optimise a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

using a simple multi‑start projected sub‑gradient method with a
final simulated‑annealing polish.  The algorithm works on a fairly
large power‑of‑two length (default 2**16) so that the discretisation
error of the underlying continuous optimisation is tiny.  An initial
pool of analytic shapes (power ramps, beta‑type profiles) seeds the
search; the best candidate is refined by many fast FFT‑based
sub‑gradient steps on the probability simplex and then polished by a
light Metropolis‑type annealer.  The routine respects the 1000 s time
budget and returns a plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:                     # should never happen for a probability vector
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape built from two power ramps joining a flat plateau.
    Normalised to sum = 1.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # fix possible off‑by‑one
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1.
    The continuous beta pdf ∝ x^{a‑1} (1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (active‑set variant)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         top_k: int = 10) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent on the probability simplex.
    At each iteration the sub‑gradient is averaged over the
    ``top_k`` largest convolution entries.
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # momentum
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.7
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,}  score={cur_score:.6f}")

        # pick active indices – either the K biggest or all within a small tolerance
        if top_k >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient of max(conv) w.r.t. a (the factor 2 comes from symmetry)
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # momentum + diminishing step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move tiny packets of mass between positions and accept
    according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000  # adapt to remaining time automatically

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # pick source proportionally to its mass (more likely to move)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 50 %) of the source mass
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function.  The routine builds a pool of
    analytic shapes, refines the best one with projected sub‑gradient
    descent (top‑k active set), and finishes with a simulated‑annealing
    polish.  The returned list contains non‑negative floats and respects
    the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # 1) Choose a power‑of‑two length for fast FFTs
    # --------------------------------------------------------------
    n = 65536    # 2¹⁶ – good trade‑off between discretisation and speed

    # --------------------------------------------------------------
    # 2) Build an initial pool of promising candidates
    # --------------------------------------------------------------
    pool = []                # list of (score, vector)
    pool_size = 8
    best_seq = None
    best_score = float("inf")

    # ---- power‑ramp family -------------------------------------------------
    p_grid = np.linspace(0.2, 2.0, 19)            # 0.2 … 2.0
    alpha_grid = np.linspace(0.0, 0.5, 11)       # 0 … 0.5
    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_power_shape(n, p, alpha)
            sc = _objective(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
            if len(pool) < pool_size:
                pool.append((sc, a.copy()))
                pool.sort(key=lambda x: x[0])
            elif sc < pool[-1][0]:
                pool[-1] = (sc, a.copy())
                pool.sort(key=lambda x: x[0])
        # occasional progress report
        if (time.time() - start_time) % 30 < 0.2:
            print(f"[{time.time() - start_time:.2f}s] power‑grid progress → best={best_score:.6f}")

    # ---- beta‑distribution family -----------------------------------------
    a_grid = np.linspace(0.2, 4.0, 13)   # shape parameters
    b_grid = np.linspace(0.2, 4.0, 13)
    for a_par in a_grid:
        for b_par in b_grid:
            a_vec = _make_beta_shape(n, a_par, b_par)
            sc = _objective(a_vec)
            if sc < best_score:
                best_score, best_seq = sc, a_vec.copy()
            if len(pool) < pool_size:
                pool.append((sc, a_vec.copy()))
                pool.sort(key=lambda x: x[0])
            elif sc < pool[-1][0]:
                pool[-1] = (sc, a_vec.copy())
                pool.sort(key=lambda x: x[0])

    # ---- a few random perturbations ---------------------------------------
    rng = np.random.default_rng()
    for _ in range(250):
        if time.time() >= deadline:
            break
        # random convex combination of two beta shapes
        a1, b1 = rng.uniform(0.1, 3.0, 2)
        a2, b2 = rng.uniform(0.1, 3.0, 2)
        w = rng.random()
        shape1 = _make_beta_shape(n, a1, b1)
        shape2 = _make_beta_shape(n, a2, b2)
        a = w * shape1 + (1 - w) * shape2
        a /= a.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
        if len(pool) < pool_size:
            pool.append((sc, a.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, a.copy())
            pool.sort(key=lambda x: x[0])

    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Possibly use a globally supplied starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter /= starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    if len(pool) < pool_size:
                        pool.append((sc, starter.copy()))
                        pool.sort(key=lambda x: x[0])
                    elif sc < pool[-1][0]:
                        pool[-1] = (sc, starter.copy())
                        pool.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4) Local refinement – projected sub‑gradient descent
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 40.0:
        sub_deadline = time.time() + 0.92 * remaining
        refined_seq, refined_score = _subgradient_descent(best_seq,
                                                          sub_deadline,
                                                          start_time,
                                                          top_k=10)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score, best_seq = anneal_score, anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improve score = {anneal_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalisation & conversion to a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
