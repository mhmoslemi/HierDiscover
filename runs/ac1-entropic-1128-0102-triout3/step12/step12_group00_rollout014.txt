def propose_candidate(...):

    n = 65536
    # compute lengths for FFT
    L_self = _next_pow2(2*n -1)

    # initial pool via shape generator

            a = _make_shape(n, p, alpha)
            score = _objective(a)
            if score < best_score:
                best_seq = a.copy(); best_score = score
    # also incorporate random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(200):
        a = rng.random(n)
        a = a / a.sum()
        score = _objective(a)
        if score < best_score:
            best_seq = a.copy(); best_score = score
    # maybe also use height_sequence_1 normalized with padding
    # embed into n
    try:
        hs = np.asarray(height_sequence_1, dtype=float)
        # embed central region into length n: zero pad and maybe upsample.
        # For simplicity: place at start indices and replicate or stretch?
        # We'll just pad to length n by repeating to fill.
        # Or we can upscale using interpolation: use np.interp.
        # Let's upscale with linear interpolation to length n.
        len_h = hs.size
        x_old = np.linspace(0, 1, len_h)
        x_new = np.linspace(0, 1, n)
        hs_interp = np.interp(x_new, x_old, hs)
        hs_interp = np.maximum(hs_interp, 0)
        hs_interp /= hs_interp.sum()
        score = _objective(hs_interp)
        if score < best_score:
            best_seq = hs_interp.copy(); best_score = score
    except Exception:
        pass

    # now run smooth gradient descent with Adam
    a = best_seq
    # Adam parameters
    m = np.zeros_like(a); v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    step0_lr = 0.5   # base learning rate
    # schedule for smoothing c
    c_start = 2.0; c_end = 60.0
    total_iters_target = 20000

    it = 0
    best_a = a.copy(); best_obj = _objective(a)
    while time.time() < deadline - 5.0:
        # compute c based on iteration count
        c = c_start + (c_end - c_start) * min(1.0, it/total_iters_target)
        # compute conv self
        conv = _conv_self_fft(a)   # length 2n-1
        # smooth weights
        maxc = conv.max()
        scaled = c * (conv - maxc)
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        # compute gradient via cross convolution
        a_rev = a[::-1]
        # Convolution w * a_rev (full)
        conv2 = np.fft.irfft(np.fft.rfft(w, n=L_cross) * np.fft.rfft(a_rev, n=L_cross), n=L_cross)
        start = n-1
        grad = 2.0 * conv2[start:start+n]

        # Adam update
        it += 1
        lr = step0_lr / np.sqrt(it)  # decreasing
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        a = _project_simplex(a)

        # evaluate true objective periodically
        if it % 200 == 0:
            cur_obj = _objective(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] iter={it} score={cur_obj:.6f}")
        # optionally break if time near deadline

    # same as earlier _anneal but adapt for size.
    if time.time() < deadline - 2:

    # final normalization and return
    best_a = np.maximum(best_a, 0)
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a]

import numpy as np, time, math

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self_fft(a):
    n = a.shape[0]
    L = _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)
    return conv[:2*n-1]

def _cross_conv_fft(w, a_rev, L_cross):
    # w length 2n-1, a_rev length n
    Fw = np.fft.rfft(w, n=L_cross)
    Fa = np.fft.rfft(a_rev, n=L_cross)
    conv = np.fft.irfft(Fw * Fa, n=L_cross)
    return conv

def _objective(seq):
    # fast version expects sum ~1
    n = seq.shape[0]
    conv = _conv_self_fft(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v):
    """Projection onto the simplex {x>=0, sum x =1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        return w
    else:
        return np.full_like(v, 1.0/n)

def _make_shape(n, p, alpha):
    # same as earlier (symmetric shape)
    alpha = max(0., min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len)//2
    if side_len <=0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len+1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape = shape / shape.sum() if shape.sum()>0 else np.full(n, 1.0/n)
    return shape

def _softmax_descent(a0, deadline, start_time, L_self, L_cross):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.6
    it = 0
    # smoothing schedule parameters
    c_start = 2.0
    c_end = 70.0
    total_iters_for_c = 20000
    while time.time() < deadline:
        it += 1
        # self convolution
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        # objective
        cur_score = float(2.0 * n * cur_max / (a.sum()**2))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] softmax iter={it:,} score={cur_score:.6f}")
        # smoothing coefficient
        c = c_start + (c_end - c_start) * min(1.0, it/total_iters_for_c)
        # softmax weights
        shifted = c * (conv - cur_max)   # non-positive
        exp_shift = np.exp(shifted)
        w = exp_shift / exp_shift.sum()
        # gradient via cross conv
        a_rev = a[::-1]
        conv2 = _cross_conv_fft(w, a_rev, L_cross)
        # extract relevant part
        start = n - 1
        grad = 2.0 * conv2[start:start + n]
        # momentum and step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # projection onto simplex
        a = _project_simplex(a)
    return best_a, best_score

def _anneal(seq, deadline, start_time, max_steps=200000):
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_steps:
        it += 1
        # linear cooling
        T = 0.02 * (1e-6/0.02) ** (it/max_steps)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")
    return best_a, best_score

"""
Propose a step‑function (probability vector) with a low ``evaluate_sequence``
score.

The algorithm proceeds in four stages:
1. Build a tiny pool of promising initial shapes (analytic power‑ramps,
   random Dirichlet draws and, if present, the global starter).
2. Refine the best candidates with a *soft‑max* approximation of the
   ``max`` in the objective.  The gradient of the smooth objective is obtained
   by convolving the soft‑max weights with the reversed current vector.
   FFT‑based convolutions give O(n log n) per iteration; after each step the
   vector is projected onto the probability simplex.
3. A short simulated‑annealing polish moves tiny packets of mass between
   random positions.
4. Normalise the final vector and return it as a plain Python list.

All helper functions are defined at top level so they can be pickled if the
harness runs the solution in a child process.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * n - 1]


def _cross_conv_fft(w: np.ndarray, a_rev: np.ndarray, L_cross: int) -> np.ndarray:
    """Convolution w * a_rev (full linear convolution) using FFT."""
    Fw = np.fft.rfft(w, n=L_cross)
    Fa = np.fft.rfft(a_rev, n=L_cross)
    conv = np.fft.irfft(Fw * Fa, n=L_cross)
    return conv


# ----------------------------------------------------------------------
# Fast objective (sum is assumed ≈ 1, i.e. a probability vector)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)²) – fast version."""
    n = seq.shape[0]
    conv = _conv_self_fft(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generator – symmetric power ramp + optional plateau
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape (length n) built from two power ramps that meet a
    plateau of height 1.  The vector is normalised to sum = 1.
    Parameters
    ----------
    p     – exponent of the side ramps (p>0).
    alpha – relative plateau width ∈[0,0.5]; 0 → pure tent,
            0.5 → uniform.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                     # rises from 0…1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]

    total = shape.sum()
    if total > 0.0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Soft‑max (smooth max) gradient descent on the simplex
# ----------------------------------------------------------------------
def _softmax_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_self: int,
                     L_cross: int) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using a smooth approximation of the max‑convolution.
    Returns the best iterate seen together with its fast objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.6
    it = 0

    # smoothing schedule: start gentle, end sharp
    c_start = 2.0
    c_end = 70.0
    c_iters = 20000  # after this the smoothing coefficient stays at c_end

    while time.time() < deadline:
        it += 1

        # self‑convolution and current objective
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max / (a.sum() ** 2))

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] soft‑max iter={it:,} score={cur_score:.6f}")

        # smoothing coefficient (linear increase)
        c = c_start + (c_end - c_start) * min(1.0, it / c_iters)

        # soft‑max weights on the convolution entries
        shifted = c * (conv - cur_max)               # ≤ 0, avoids overflow
        exp_shift = np.exp(shifted)
        w = exp_shift / exp_shift.sum()              # weight vector, sum = 1

        # gradient of the smooth objective:
        #   ∇ = 2 * (w ⋆ a)  where ⋆ is cross‑correlation.
        a_rev = a[::-1]
        conv2 = _cross_conv_fft(w, a_rev, L_cross)
        start = n - 1                                 # index of zero‑lag in full conv
        grad = 2.0 * conv2[start:start + n]

        # momentum & diminishing step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a -= step * momentum
        a = _project_simplex(a)                       # stay on the simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float,
            max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions; Metropolis rule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # source weighted by its current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score
