"""
Improved optimizer for the step‑function objective 2·n·max(conv(a,a))/sum(a)².

The method consists of three stages that run inside the 1000 s budget:

1. **Adam‑mirrored smooth‑max descent** – an exponentiated‑gradient (mirror)
   version of Adam on the log‑sum‑exp approximation of the max.  The smoothing
   parameter α is increased geometrically up to 2·10⁴, allowing the algorithm
   to first smooth the max and then focus on the true ℓ∞‑norm.

2. **Direct max‑subgradient descent** – a fast exponentiated‑gradient step on the
   exact max (the “hard” sub‑gradient).  This drives the solution past any
   plateau left by the smooth phase.

3. **Stochastic hill‑climbing (annealing)** – small random mass transfers with
   a Metropolis acceptance rule for a final polish.

The optimizer starts from a small portfolio of seeds (the previous best,
a uniform distribution and a couple of random Dirichlet draws) and keeps the
best candidate found throughout all phases.  The whole routine respects the
time budget by checking the deadline at every iteration.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: exact objective (scale‑invariant – we keep sum(a)=1)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector a with sum≈1."""
    n = a.size
    # length for self‑convolution
    L_self = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑mirrored smooth‑max descent
# ----------------------------------------------------------------------
def _adam_smooth_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the log‑sum‑exp (soft‑max) approximation of max(conv).
    Uses Adam‑style adaptive step sizes on the exponentiated‑gradient update.
    """
    a = a0.copy()
    n = a.size
    # FFT lengths
    L_self = 1 << ((2 * n - 2).bit_length())
    L_grad = 1 << ((3 * n - 3).bit_length())

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.15           # base learning‑rate before the 1/√t decay
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _exact_score(best_a)

    # geometric schedule for the smoothing parameter α
    alphas = np.geomspace(0.5, 2e4, num=80)

    it = 0
    for alpha in alphas:
        # break if we are close to the deadline
        if time.time() >= deadline - 0.3:
            break

        # inner loop – run as many iterations as time permits for this α
        while time.time() < deadline - 0.3:
            it += 1

            # 1) self‑convolution a*a
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights w = softmax(α·conv)
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)          # stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)   (central n entries)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam update on the *log‑space* (mirrored) variables
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            # bias‑correction
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # effective step (the 1/√t decay is inherited from base_lr)
            step = (base_lr / math.sqrt(it)) * (m_hat / (np.sqrt(v_hat) + eps))

            # exponentiated‑gradient update on the simplex
            # clip step to avoid overflow in the exponent
            step = np.clip(step, -20.0, 20.0)
            a = a * np.exp(-step)
            a = np.maximum(a, 1e-30)          # keep strict positivity
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            else:
                a = np.full_like(a, 1.0 / n)

            # occasional exact evaluation
            if it % 5000 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Adam α={alpha:.1e} it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Direct max‑subgradient descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _max_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Push mass away from the index attaining the current max (hard sub‑gradient)."""
    a = a0.copy()
    n = a.size
    L_self = 1 << ((2 * n - 2).bit_length())

    best_a = a.copy()
    best_score = _exact_score(best_a)

    it = 0
    while time.time() < deadline - 0.3:
        it += 1

        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # g_i = 2·a_{k‑i}

        eta = 0.5 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        if it % 4000 == 0:
            cur_score = _exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (final polish)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """Small random mass transfers with Metropolis acceptance."""
    cur = a0.copy()
    cur_score = _exact_score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    start_temp = 5e-2
    end_temp = 1e-10

    n = cur.size

    while time.time() < deadline:
        it += 1

        # linear temperature decay
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if cur[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 30 % of the source mass
        eps = cur[src] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum > 0.0:
            cand /= cand_sum
        else:
            continue

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score

        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} best={best_score:.6f}")

    return best_seq, best_score
