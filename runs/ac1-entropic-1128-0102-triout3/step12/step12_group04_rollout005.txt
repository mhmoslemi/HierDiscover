```
def _adam_smoothmax(a0, deadline, start_time, beta0, beta_inc, beta_max, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(4 * n)   # safe
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        beta = beta0 + it * beta_inc
        if beta > beta_max:
            beta = beta_max

        # compute conv and gradient
        conv = _conv_self(a, L)
        maxconv = conv.max()
        # compute softmax weights
        # shift conv by max for numerical stability
        exp_vals = np.exp(beta * (conv - maxconv))
        w = exp_vals / exp_vals.sum()
        # convolution of w with reversed a to get gradient slice
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L)
        A_rev_f = np.fft.rfft(a_rev, n=L)
        grad_slice = np.fft.irfft(Wf * A_rev_f, n=L)[n - 1 : n - 1 + n]
        # gradient of smooth max times objective scaling
        grad = 4.0 * n * grad_slice  # factor 4n as derived

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto simplex
        a = _project_simplex(a)

        # evaluate objective
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it} beta={beta:.1f} score={cur_score:.6f}")

    return best_a, best_score
```

assistantfinal**Solution Overview**

The evaluation function is  

\[
E(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2}\qquad 
a_i\ge 0 .
\]

Because the ratio is *scale‑invariant* we may (and do) keep the vector on the
probability simplex \( \sum_i a_i = 1\).  
With this normalisation  

\[
E(a)=2n\;\max (a*a) .
\]

Hence we have to flatten the self‑convolution as much as possible.

The code below follows three ideas that improve the previous
implementation.

1. **Warm‑starts** – a richer family of analytic shapes (tent, cosine,
   half‑circle, Gaussian, power‑tents, β‑U‑shapes) and a few random Dirichlet
   samples are generated, the best of them (exact objective) is kept.

2. **Projected Adam on a *smooth* approximation of the max** – instead of the
   crude top‑k averaging we use the soft‑max  

   \[
   \operatorname{SM}_\beta (c)=\frac1\beta\log\!\Bigl(\sum_k e^{\beta c_k}\Bigr)
   \]
   with a temperature β that is increased during the run.
   The gradient of SM can be written as a single convolution, so each
   iteration costs only a few FFTs ( O(n log n) ).

3. **Light simulated‑annealing polish** – after the main Adam phase a cheap
   Metropolis walk moves a random packet of mass between two indices,
   helping to escape shallow minima.

The whole search respects the 1000 s budget:  
≈ 80 % of the time is spent in Adam, the remainder in annealing.
The final vector is projected back onto the simplex before returning.

```python
"""
Optimise a non‑negative step‑function a∈ℝⁿ (∑a=1) to minimise

        2·n·max(conv(a,a))

by
  – warm‑starting from several analytic shapes and random Dirichlet samples,
  – Adam on a *smooth* max (soft‑max) gradient with a β‑schedule,
  – a short simulated‑annealing polish.

All heavy operations use real FFTs (O(n log n)).  The routine fits
the 1000 s budget and returns a plain Python list ready for
evaluate_sequence().
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective `2·n·max(conv(a,a))` for a normalised (∑a=1) vector.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – analytic families + a few random Dirichlet vectors
# ----------------------------------------------------------------------


def _make_warm_starts(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a collection of candidate vectors."""
    x = np.arange(n, dtype=float) / (n - 1)               # normalised coordinate in [0,1]

    # 1) symmetric linear tent
    tent = np.minimum(x, 1.0 - x)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)                           # zero at the ends

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

    # 5) power‑tent (flatter than the linear tent)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    # 6) β‑U‑shape  (U‑shaped Beta distributions, p,q<1)
    beta_shapes = []
    for a in (0.4, 0.5, 0.6):
        beta_shapes.append(x ** a * (1 - x) ** a)

    # 7) random Dirichlet samples (sparse start)
    rand_samples = []
    for _ in range(3):
        s = rng.exponential(scale=1.0, size=n)
        rand_samples.append(s)

    candidates = [
        tent, cosine, circ, gauss, power_tent,
        *beta_shapes,
        *rand_samples,
    ]

    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Adam optimiser using a *smooth* max (soft‑max) gradient
# ----------------------------------------------------------------------


def _adam_smoothmax(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    beta0: float = 0.5,
                    beta_inc: float = 0.0,
                    beta_max: float = 5000.0,
                    lr: float = 0.015,
                    beta1: float = 0.9,
                    beta2: float = 0.999,
                    eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex.
    The loss is `2·n·SM_β(conv)`, SM_β being the soft‑max approximation of the
    maximum.  β is increased linearly (`beta0 + it*beta_inc`) up to `beta_max`.
    Returns the best iterate and its exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(4 * n)            # safe length for all FFTs

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # ------------ temperature schedule for the soft‑max ------------
        beta = beta0 + it * beta_inc
        if beta > beta_max:
            beta = beta_max

        # ----------------- self‑convolution -----------------
        conv = _conv_self(a, L)               # length 2·n‑1
        maxconv = conv.max()

        # --------------- soft‑max weights -----------------
        # exp(β*(conv - max)) ∈ (0,1]; stable because the maximum term becomes 1
        exp_vals = np.exp(beta * (conv - maxconv))
        w = exp_vals / exp_vals.sum()         # shape (2·n‑1,)

        # --------------- gradient of SM -----------------
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L)
        Revf = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(Wf * Revf, n=L)
        # slice that corresponds to ∂/∂a_i
        slice_ = grad_full[n - 1 : n - 1 + n]

        # gradient of the *objective* (4·n factor derived in the analysis)
        grad = 4.0 * n * slice_

        # ------------------- Adam update -------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ------------------- projection -------------------
        a = _project_simplex(a)

        # ------------------- bookkeeping -------------------
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"β={beta:.1f} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (identical to the reference code)
# ----------------------------------------------------------------------


def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk that moves a random packet of mass between two
    indices.  Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 40_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (very fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)            # source proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
```
