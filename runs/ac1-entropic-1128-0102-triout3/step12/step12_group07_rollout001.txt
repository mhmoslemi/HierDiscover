Projection onto simplex algorithm:

def proj_simplex(v):
    # Return projection of v onto {x >=0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

This algorithm returns projection onto L1-ball of radius 1 with non-negative constraints, which is simplex.

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # init p (height_sequence_1 if available else arcsine)
    try:
        p0 = np.asarray(height_sequence_1, dtype=float)
        if p0.ndim != 1 or p0.size==0:
            raise ValueError
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s==0:
            raise ValueError
        p = p0 / s
        n = p.shape[0]
        print(f"Using height_sequence_1 (n={n})")
    except Exception:
        n = 4096
        x = (np.arange(n, dtype=float) + 0.5)/n
        alpha=0.38
        EPS=1e-6
        w = (x*(1-x) + EPS) ** (-alpha)
        p = w / w.sum()
        print(f"Generated arcsine start (n={n})")

    # compute conv
    conv = _conv_self(p)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max
    best_p = p.copy()
    print(f"Initial score {best_score:.6f}")

    iteration = 0
    last_improve_iter = 0
    # parameters
    max_lr = 0.1
    min_lr = 0.001
    # jitter frequency
    jitter_interval = 5000
    # plateau detection
    plateau_limit = 20000

    while time.time() < deadline:
        iteration += 1
        # compute worst shift
        k = int(conv.argmax())
        cur_max = conv.max()
        # gradient
        idx = np.arange(n)
        source_idx = k - idx  # potential g indices
        mask = (source_idx >= 0) & (source_idx < n)
        g = np.zeros_like(p)
        g[mask] = 2.0 * p[source_idx[mask]]

        # step size schedule
        elapsed = time.time() - start
        # linear decay for lr
        lr = max_lr * (1.0 - elapsed / budget_s)
        if lr < min_lr:
            lr = min_lr

        # candidate via additive step
        p_candidate = p - lr * g
        # projection onto simplex
        p_candidate = proj_simplex(p_candidate)

        # evaluate
        conv_candidate = _conv_self(p_candidate)
        new_max = conv_candidate.max()
        if new_max < cur_max - 1e-14:
            # accept
            p = p_candidate
            conv = conv_candidate
            cur_max = new_max
            score = 2.0 * n * cur_max
            if score < best_score:
                best_score = score
                best_p = p.copy()
                last_improve_iter = iteration
                print(f"[{time.time()-start:.2f}s] iter={iteration} new best score={best_score:.6f}")
        else:
            # try smaller step by halving lr up to 3 times
            improved = False
            lr_tmp = lr * 0.5
            for _ in range(3):
                if lr_tmp < min_lr * 0.1:
                    break
                p_candidate = p - lr_tmp * g
                p_candidate = proj_simplex(p_candidate)
                conv_candidate = _conv_self(p_candidate)
                new_max = conv_candidate.max()
                if new_max < cur_max - 1e-14:
                    improved = True
                    break
                lr_tmp *= 0.5
            if improved:
                p = p_candidate
                conv = conv_candidate
                cur_max = new_max
                score = 2.0 * n * cur_max
                if score < best_score:
                    best_score = score
                    best_p = p.copy()
                    last_improve_iter = iteration
                    print(f"[{time.time()-start:.2f}s] iter={iteration} after line search new best score={best_score:.6f}")
            else:
                # no improvement, possibly plateau
                pass

        # jitter occasional or after plateau
        if iteration % jitter_interval == 0:
            # add small jitter to p and re-project
            jitter_scale = 0.001  # relative jitter amplitude
            jitter = jitter_scale * p * (np.random.rand(n) - 0.5)
            p = np.maximum(p + jitter, 0.0)
            p = proj_simplex(p)
            conv = _conv_self(p)

        if iteration - last_improve_iter > plateau_limit:
            # re‑jitter from best
            jitter = 0.005 * best_p * (np.random.rand(n) - 0.5)
            p = np.maximum(best_p + jitter, 0.0)
            p = proj_simplex(p)
            conv = _conv_self(p)
            last_improve_iter = iteration
            print(f"[{time.time()-start:.2f}s] plateau – re‑jittered at iter {iteration}")

    # final best as list
    best_p = best_p / best_p.sum()

def proj_simplex(v):
    """ Euclidean projection of vector v onto the simplex {x >= 0, sum x = 1}. """
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all entries are zero or negative -> return unit vector at max entry
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # ensure sum = 1 (might be off by tiny epsilon)
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        # fallback to unit vector at index of max original v
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
    return w

"""
Simple projected sub‑gradient descent for the autocorrelation peak.

The score of a non‑negative sequence `p` (any scaling) is
    score = 2·n·max(conv(p,p)) / (sum(p)**2)
which is invariant to scaling, so we can keep `p` normalised to sum = 1.

We iteratively:
  1. Compute the linear self‑convolution `c = p * p`.
  2. Find the index `k` where `c` attains its maximum.
  3. The sub‑gradient of the maximal entry `c[k]` w.r.t. `p` is
        g_i = 2·p_{k-i}   (zero if the index is out of range).
     Moving `p` a little opposite to this gradient reduces `c[k]`.
  4. Take a step `p ← p – η·g`, clip negatives to zero and renormalise.
  5. Accept the move only if the new maximal convolution value decreased.
  6. Occasionally add a tiny random jitter (to escape plateaus) and
     restart from the best solution seen so far.

The learning rate η decays linearly from a relatively large value
(to make fast early progress) down to a small floor.  The whole
procedure runs until the time budget expires and returns the best
vector encountered (as a plain Python list).  The algorithm is
pure‑Python + NumPy and respects the 1‑CPU constraint.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: linear self‑convolution via FFT (fast for moderate sizes)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv
