def _subgrad_optimize(a0, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()

        L = _next_pow2(2*len(a)-1)  # precomputed maybe outside loop

def propose_candidate(seed=0,budget_s=1000,...):
   start_time = time.time(); deadline = start_time + budget_s - 0.5.
   n=4096
   rng = np.random.default_rng(seed)
   # generate warm starts (include previous sequences)
   warm = _make_warm_starts(n)
   # add param shapes
   warm += _generate_parametric_candidates(n, rng)
   # include height_sequence_1 if global var exists.
   # evaluate each
   best_a = None; best_score = np.inf
   for ws in warm:
       sc = _objective(ws)
       if sc < best_score: best_score=sc; best_a=ws.copy()
   print("Warm start best:", best_score)
   # allocate time for subgradient
   remaining = deadline - time.time()
   if remaining > 5: # at least for subgrad
       subgrad_deadline = time.time() + remaining * 0.7
       a_opt, score_opt = _subgrad_optimize(best_a, subgrad_deadline, start_time)
       if score_opt < best_score:
           best_score = score_opt; best_a = a_opt.copy()
           print("Subgrad best:", best_score)
       remaining = deadline - time.time()
   # polishing anneal
   if time.time() < deadline:
       anneal_deadline = deadline
       a_fin, score_fin = _anneal(best_a, anneal_deadline, start_time)
       if score_fin < best_score:
           best_score=score_fin; best_a = a_fin.copy()
           print("Anneal best:", best_score)
   # final normalization
   best_a = np.maximum(best_a, 0)
   total = best_a.sum()
   if total <=0:
       best_a = np.full(n, 1/n)
   else:
       best_a/=total
   return best_a.tolist()

"""
Low‑autocorrelation step‑function search.

The goal is to minimise 2·n·max(conv(a,a)) for a non‑negative vector a with
∑a=1.  The code builds a diverse set of analytic warm‑starts, selects the best
one, then performs a fast sub‑gradient descent on the exact max‑convolution
objective (the gradient is an average of shifted copies of a).  After the
gradient phase a cheap simulated‑annealing polish is applied.  The whole
search respects a 1000 s budget and returns a plain Python list ready for
``evaluate_sequence``.
"""

import math
import time

import numpy as np


# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_full(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is assumed non‑negative)."""
    n = a.shape[0]
    conv = _conv_full(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Warm‑starts (analytic shapes)
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """A small library of simple analytic candidates."""
    x = np.arange(n, dtype=float) / (n - 1)

    # 1) symmetric linear tent
    tent = np.minimum(x, 1.0 - x)

    # 2) raised cosine
    cosine = np.sin(np.pi * x)

    # 3) half‑circle
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))

    # 4) Gaussian
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

    # 5) power‑tent (flatter)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


def _parametric_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate extra candidates: exponentials, power‑laws and random vectors."""
    cand = []

    # exponential decays
    for lam in np.linspace(0.0005, 0.025, 20):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        cand.append(a)

    # decreasing power‑law
    for p in np.linspace(0.2, 3.0, 20):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        cand.append(a)

    # increasing power‑law (mirror of the previous)
    for p in np.linspace(0.2, 3.0, 20):
        a = (np.arange(n, 0, -1, dtype=float) ** (-p))
        a /= a.sum()
        cand.append(a)

    # a few random probability vectors (as a sanity check)
    for _ in range(10):
        a = rng.random(n)
        a /= a.sum()
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Uses the true sub‑gradient of the max‑convolution (average of the
    shifted copies of a for all maximal shift positions).
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    # first convolution (needed for the first gradient)
    conv = _conv_full(a, L)
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        max_val = conv.max()
        ks = np.where(conv == max_val)[0]

        # ---- exact sub‑gradient of the max entry ----
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            grad[idx] += a[k - idx]
        grad = (2.0 / len(ks)) * grad

        # simple diminishing step size
        step = 0.5 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        # new convolution – will be used in the next iteration
        conv = _conv_full(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Subgrad iter={it:,} "
                f"score={cur_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(
    seq: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk that transfers a random packet of mass."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    step = 0
    while time.time() < deadline:
        step += 1

        # exponential cooling – relatively gentle
        T = 0.05 * math.exp(-step / 1e5) + 1e-9

        src = rng.choice(n, p=a)            # source weighted by current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        eps = a[src] * rng.random() * 0.5   # move up to 50 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Anneal step={step:,} "
                    f"best={new_score:.6f}"
                )

    return best_a, best_score

    # --------------------------------------------------------------
    warm_starts = _make_warm_starts(n)
    warm_starts.extend(_parametric_candidates(n, rng))

    # optional global starter (e.g. a previously found solution)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                warm_starts.append(starter)
    except Exception:
        pass

    # evaluate all warm starts and keep the best
    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Sub‑gradient descent (the heavy work)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:  # keep a few seconds for the final polish
        # roughly 70 % of the remaining budget for the gradient phase
        grad_deadline = time.time() + remaining * 0.70
        a_opt, score_opt = _subgrad_optimize(a0, grad_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad best score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing polish (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_fin, score_fin = _anneal(a0, deadline, start_time)
        if score_fin < best_score:
            best_score = score_fin
            a0 = a_fin.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal best={score_fin:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation & conversion to plain list
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    return [float(v) for v in a0]
