def _grad_softmax_weights(conv, a, beta):
    # compute softmax weights p
    max_c = conv.max()
    shifted = conv - max_c
    exps = np.exp(beta * shifted)
    p = exps / exps.sum()
    # compute correlation: g_i = 2 * sum_k p_k * a_{k-i}
    rev_a = a[::-1]
    L = _next_pow2(len(p) + len(a) - 1)
    # FFT conv of p and rev_a
    # Use np.fft.rfft for real
    Pf = np.fft.rfft(p, n=L)
    Af = np.fft.rfft(rev_a, n=L)
    conv_pa = np.fft.irfft(Pf * Af, n=L)
    # slice: g = conv_pa[n-1: n-1+n] if conv_pa length >= 2n-1 else but conv_pa length is L; we want indices n-1 to n-1+n-1 inclusive (size n)
    start = len(a) - 1
    g = conv_pa[start:start + len(a)]
    g = 2.0 * g
    return g, p

"""Optimisation of step‑function heights via smoothed Adam descent,
random warm–starts and a greedy mass‑transfer refinement.

The algorithm builds several analytic and random initial probability
vectors (the “heights” are a probability distribution, i.e. they sum to 1).
A smoothed‑max objective is minimised by mirror‑descent on the simplex:
the loss is ‑‖a‖₁⁻²·max(conv(a,a)), approximated with a soft‑max
over the convolution entries.  The gradient is obtained analytically
and an Adam optimiser on the soft‑max parameters is run.  The best
vector is then polished by a simple greedy redistribution of mass
targeting the indices that currently dominate the convolution,
followed by a lightweight simulated‑annealing walk.

The whole search respects the 1000 s budget and returns a plain list
of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution using FFT, length = len(x)+len(y)-1."""
    L = _next_pow2(len(x) + len(y) - 1)
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)
    return conv[: len(x) + len(y) - 1]


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic warm‑start shapes
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)  # in [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _collect_warm_starts(n: int) -> list[np.ndarray]:
    """Gather a modest library of diverse initial vectors."""
    cand = []
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    for lc in (0.25, 0.5, 0.75):
        centre = _shape_power_center(n, 2.0)
        edge = _shape_power_edge(n, 2.0)
        a = lc * centre + (1 - lc) * edge
        a = a / a.sum()
        cand.append(a)

    # optional global starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # remove NaNs
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq

# Smoothed‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam on the soft‑max (log‑it) parametrisation.
    ``a0`` must already be a probability vector (non‑negative, sum=1).
    """
    n = a0.shape[0]
    # initialise log‑its; add a tiny constant to avoid log(0)
    x = np.log(a0 + 1e-20)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.3          # base learning‑rate for the log‑its
    beta0 = 5.0        # initial smoothing for the soft‑max over conv
    beta_inc = 0.02    # increase of smoothing per iteration
    max_beta = 80.0

    best_a = np.exp(x) / np.exp(x).sum()
    best_score = _objective(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        # probability vector from current logits
        max_x = np.max(x)
        e = np.exp(x - max_x)
        a = e / e.sum()

        # current convolution and score
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # smoothed max via soft‑max weights
        beta = min(max_beta, beta0 + beta_inc * it)
        # stable soft‑max
        shift = conv.max()
        exparg = beta * (conv - shift)
        w = np.exp(exparg)
        p = w / w.sum()               # shape (2n‑1,)

        # gradient w.r.t a (2 * correlation of p with a)
        rev_a = a[::-1]
        L = _next_pow2(len(p) + len(a) - 1)
        Pf = np.fft.rfft(p, n=L)
        Af = np.fft.rfft(rev_a, n=L)
        corr = np.fft.irfft(Pf * Af, n=L)
        start = n - 1
        grad_a = 2.0 * corr[start:start + n]

        # chain rule to obtain gradient w.r.t logits x
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # decaying learning rate (helps stability later on)
        lr = lr0 / (np.sqrt(it) + 1.0)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits numerically stable (translation does not change softmax)
        x -= np.max(x)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy mass‑transfer refinement
# ----------------------------------------------------------------------
def _greedy_refine(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:

    Deterministic hill‑climbing: repeatedly move a fraction of mass from the
    index that contributes most to the current maximal convolution entry to a
    random destination.  Accept only improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    cur_score = best_score

    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a)
        k_max = int(conv.argmax())

        # contribution of each i to conv[k_max]
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        src = int(np.argmax(contrib))
        # pick a destination different from src
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move a modest fraction of the source mass
        eps = a[src] * 0.25
        if eps < 1e-14:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        new_score = _objective(a_new)

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Greedy improve score={new_score:.6f}")

    return best_a, best_score

"""Optimisation of step‑function heights via smoothed Adam descent,
random warm–starts and a greedy mass‑transfer refinement.

The algorithm builds several analytic and random initial probability
vectors (the “heights” are a probability distribution, i.e. they sum to 1).
A smoothed‑max objective is minimised by mirror‑descent on the simplex:
the loss is ‑‖a‖₁⁻²·max(conv(a,a)), approximated with a soft‑max over the
convolution entries.  The gradient is obtained analytically and an Adam
optimiser on the soft‑max parameters is run.  The best vector is then
polished by a simple greedy redistribution of mass targeting the indices
that currently dominate the convolution, followed by a lightweight
simulated‑annealing walk.

The whole search respects the 1000 s budget and returns a plain list
of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution using FFT, length = len(x)+len(y)-1."""
    L = _next_pow2(len(x) + len(y) - 1)
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)
    return conv[: len(x) + len(y) - 1]


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic warm‑start shapes
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)  # in [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _collect_warm_starts(n: int) -> list[np.ndarray]:
    """Gather a modest library of diverse initial vectors."""
    cand = []
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    for lc in (0.25, 0.5, 0.75):
        centre = _shape_power_center(n, 2.0)
        edge = _shape_power_edge(n, 2.0)
        a = lc * centre + (1 - lc) * edge
        a = a / a.sum()
        cand.append(a)

    # optional global starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam on the soft‑max (log‑it) parametrisation.
    ``a0`` must already be a probability vector (non‑negative, sum=1).
    """
    n = a0.shape[0]
    # initialise log‑its (tiny offset avoids log(0))
    x = np.log(a0 + 1e-20)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.3          # base learning‑rate for the logits
    beta0 = 5.0        # initial smoothing for the soft‑max over conv
    beta_inc = 0.02    # increase of smoothing per iteration
    max_beta = 80.0

    # current best
    best_a = np.exp(x) / np.exp(x).sum()
    best_score = _objective(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        # probability vector from current logits
        max_x = np.max(x)
        e = np.exp(x - max_x)
        a = e / e.sum()

        # current convolution and score
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # soft‑max weights approximating the max
        beta = min(max_beta, beta0 + beta_inc * it)
        shift = conv.max()
        exparg = beta * (conv - shift)
        w = np.exp(exparg)
        p = w / w.sum()                    # shape (2n‑1,)

        # gradient w.r.t a (2 * correlation of p with a)
        rev_a = a[::-1]
        L = _next_pow2(len(p) + len(a) - 1)
        Pf = np.fft.rfft(p, n=L)
        Af = np.fft.rfft(rev_a, n=L)
        corr = np.fft.irfft(Pf * Af, n=L)
        start = n - 1
        grad_a = 2.0 * corr[start:start + n]

        # chain rule to obtain gradient w.r.t logits x
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # slowly decaying step size
        lr = lr0 / (np.sqrt(it) + 1.0)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # translation does not change the soft‑max – keep numbers moderate
        x -= np.max(x)

    return best_a, best_score
