"""
Propose a step‑function (non‑negative probability vector) with a low
evaluation score.

Algorithm
---------
1. **Problem size** – use a power‑of‑two length `n = 65536`.  This gives a
   fine discretisation while keeping FFTs fast.

2. **Initial candidates** – generate a small pool of analytic shapes
   (power‑ramps, flat plateaux, Gaussians).  Keep the best few.

3. **Smooth ‑L∞ optimisation** – minimise a smooth approximation of the
   L∞‑norm of the self‑convolution:
        f(a) = log‑sum‑exp(β·(a*a)) / β .
   For a given β the gradient is
        ∇f = 2·conv_lin(a, w_rev)   with   w_i = softmax_i(β·(a*a)_i) .
   The optimisation uses projected (simplex) stochastic gradient descent
   with momentum and a schedule that slowly increases β from 0.5 to 20.
   The loop runs until the time budget is exhausted.

4. **Light simulated‑annealing polish** – a few thousand mass‑transfer
   moves (Metropolis rule) are performed to escape possible local minima.

5. **Finalisation** – clip, renormalise, and return a plain Python list.

All helper functions are top‑level so they can be pickled if the harness
runs the solution in a child process.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: smallest power of two ≥ x
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear convolution using real FFT (fast, zero‑padded)
# ----------------------------------------------------------------------
def _conv_linear(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b (both 1‑D)."""
    n = a.shape[0] + b.shape[0] - 1
    L = _next_pow2(n)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Self‑convolution (a*a) – used for the objective and for the smooth
# approximation.
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators – power ramp + plateau, and Gaussian
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric shape built from two power ramps that meet a plateau of height 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete centred Gaussian, normalised to sum = 1."""
    sigma = max(1e-3, sigma_frac * n)
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Smooth L∞‑approximation descent (soft‑max of the convolution)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on f(a)=log‑sum‑exp(β·(a*a))/β."""
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n - 2)            # length needed for a*w_rev convolution
    best_a = a.copy()
    best_score = _objective(a)

    # momentum & step‑size parameters
    momentum = np.zeros_like(a)
    beta = 0.5
    beta_max = 20.0
    beta_rate = (beta_max - beta) / 150_000  # increase slowly
    step0 = 0.4
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- FFT of the current vector (shared for both convolutions) ----
        Af = np.fft.rfft(a, n=L)

        # ---- self‑convolution (a*a) for the soft‑max weights ------------
        conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        max_c = conv_full.max()
        # stable soft‑max
        scaled = beta * (conv_full - max_c)
        exp_scaled = np.exp(scaled)
        sum_exp = exp_scaled.sum()
        w = exp_scaled / sum_exp                     # probability vector, length 2n‑1

        # ---- gradient = 2·conv_lin(a, w_rev) ---------------------------
        w_rev = w[::-1]
        conv2_full = np.fft.irfft(Af * np.fft.rfft(w_rev, n=L), n=L)
        # slice the relevant n entries and reverse to obtain the gradient
        start = n - 1
        end = start + n
        grad = 2.0 * conv2_full[start:end][::-1]

        # ---- momentum, step‑size and projection ------------------------
        momentum = 0.9 * momentum + 0.1 * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

        # ---- keep the best exact score seen so far --------------------
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[{time.time() - start_time:.2f}s] smooth iter={it:,} "
                      f"score={cur_score:.6f}")

        # ---- update β -------------------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta + beta_rate)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (keeps the simplex constraint)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # choose source proportionally to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer a random fraction (≤25 %) of source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) with a low
    value of ``evaluate_sequence``.  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Problem size – power‑of‑two for fast FFTs
    # --------------------------------------------------------------
    n = 65536

    # --------------------------------------------------------------
    # 2) Build a small pool of promising analytic shapes
    # --------------------------------------------------------------
    top_k_pool = 6
    pool = []                     # list of (score, vector)
    best_seq = None
    best_score = float("inf")

    # power‑ramp + plateau grid
    p_grid = np.linspace(0.2, 2.0, 13)      # 0.2 … 2.0
    alpha_grid = np.linspace(0.0, 0.5, 9)   # 0 … 0.5
    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_power_shape(n, p, alpha)
            sc = _objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
            # keep the pool sorted
            if len(pool) < top_k_pool:
                pool.append((sc, a.copy()))
                pool.sort(key=lambda x: x[0])
            elif sc < pool[-1][0]:
                pool[-1] = (sc, a.copy())
                pool.sort(key=lambda x: x[0])

    # Gaussian shapes
    sigma_grid = np.linspace(0.03, 0.30, 10)   # fraction of n
    for sf in sigma_grid:
        a = _make_gaussian_shape(n, sf)
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        if len(pool) < top_k_pool:
            pool.append((sc, a.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, a.copy())
            pool.sort(key=lambda x: x[0])

    # a few random mixtures (rand‑moving‑average of two Gaussians)
    rng = np.random.default_rng()
    for _ in range(200):
        if time.time() >= deadline:
            break
        sf1, sf2 = rng.uniform(0.03, 0.30, size=2)
        w = rng.random()
        g1 = _make_gaussian_shape(n, sf1)
        g2 = _make_gaussian_shape(n, sf2)
        a = w * g1 + (1 - w) * g2
        a = _project_simplex(a)          # ensure exact simplex
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        if len(pool) < top_k_pool:
            pool.append((sc, a.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, a.copy())
            pool.sort(key=lambda x: x[0])

    print(f"[{time.time() - start_time:.2f}s] initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) If a global starter is available, inject it into the pool
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            # if the starter length differs from n, resize by simple interpolation
            if starter.size != n:
                # nearest‑neighbour up‑sampling (fast, preserves non‑negativity)
                repeat = int(np.ceil(n / starter.size))
                starter = np.tile(starter, repeat)[:n]
            starter = _project_simplex(starter)
            cur = _objective(starter)
            print(f"[{time.time() - start_time:.2f}s] global starter score = {cur:.6f}")
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                if len(pool) < top_k_pool:
                    pool.append((cur, starter.copy()))
                    pool.sort(key=lambda x: x[0])
                elif cur < pool[-1][0]:
                    pool[-1] = (cur, starter.copy())
                    pool.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4) Smooth descent from the best candidate (majority of the budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + 0.85 * remaining
        refined_seq, refined_score = _smooth_descent(best_seq,
                                                     sub_deadline,
                                                     start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑descent best = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalise and convert to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
