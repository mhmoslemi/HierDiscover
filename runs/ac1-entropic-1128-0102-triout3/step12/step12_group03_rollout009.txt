"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The strategy is a multi‑stage search:

1. **Problem size** – pick a length N (larger N gives finer granularity,
   but we keep it moderate to stay within the time budget).

2. **Diverse initial pool** – generate candidates from several analytical
   families:
        * “edge‑heavy’’ power‑law (a_i ∝ (min(i,N‑1‑i)+1)^‑p)
        * Dirichlet draws with various concentration parameters
        * the original power‑ramp shapes (low‑edge / high‑centre)
        * a few pure random vectors.
   All candidates are normalised to unit ℓ₁‑norm and their exact objective
   is evaluated.  The best few (top_k) are retained.

3. **Local optimisation** – each top candidate is refined by two fast
   gradient‑based methods.
        * *Smooth GD* – minimise a soft‑max surrogate of the max‑convolution
          using an Adam update and projection onto the probability simplex.
        * *Sub‑gradient GD* – directly follows the sub‑gradient of the true
          max‑convolution (the same update used in the reference code) but
          with a larger initial step size.
   The time budget is split evenly amongst the retained candidates.

4. **Final polishing** – a short simulated‑annealing walk is performed on
   the overall best sequence found so far.

All heavy work (convolution, soft‑max gradients) is performed with real
FFT for speed.  The routine respects the 1000 s time limit; if the budget
runs out the best sequence obtained up to that point is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of candidate shapes
# ----------------------------------------------------------------------
def _make_ramp_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric “ramp‑with‑plateau’’ shape, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_edge_shape(n: int, p: float) -> np.ndarray:
    """Edge‑heavy power‑law: a_i ∝ (min(i,n‑1‑i)+1)^‑p, normalised to sum 1."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    shape = d ** (-p)
    shape /= shape.sum()
    return shape


def _make_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(alpha) vector (symmetric concentration)."""
    a = rng.gamma(alpha, 1.0, size=n)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient‑based refinements
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate and temperature schedule
    step0 = 0.15
    T0 = 0.05
    T_end = 1e-4
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} score={cur_score:.6f}")

        # ----- temperature (linear decay) -----
        frac = it / (it + 500.0)
        T = max(T_end, T0 * (1.0 - frac))

        # ----- soft‑max weights -----
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # ----- gradient via correlation (w * a_rev) -----
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a = _project_simplex(a)

    return best_a, best_score


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.7   # larger than in the reference code
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad iter={it:,} score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

    return best_a, best_score


def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Light simulated annealing (random mass transfers)."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 250_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = 0.03 * (1e-6 / 0.03) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.3   # move up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose a problem size – larger N gives better resolution.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 250:
        N = 32768
    else:
        N = 16384

    top_k = 7
    top = []   # list of (score, vector)

    rng = np.random.default_rng()

    # ----- 1) analytic families -------------------------------------------------
    # a) edge‑heavy power‑law
    for p in np.linspace(0.1, 2.5, 15):
        a = _make_edge_shape(N, p)
        sc = _objective(a)
        if len(top) < top_k:
            top.append((sc, a))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, a)
            top.sort(key=lambda x: x[0])

    # b) power‑ramp (low‑edge / high‑centre) – same family as the reference code
    p_grid = np.linspace(0.2, 2.0, 9)
    alpha_grid = np.linspace(0.0, 0.5, 6)
    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_ramp_shape(N, p, alpha)
            sc = _objective(a)
            if len(top) < top_k:
                top.append((sc, a))
                top.sort(key=lambda x: x[0])
            elif sc < top[-1][0]:
                top[-1] = (sc, a)
                top.sort(key=lambda x: x[0])

    # c) Dirichlet draws with several concentration parameters
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.0, 2.0]:
        a = _make_dirichlet_shape(N, alpha, rng)
        sc = _objective(a)
        if len(top) < top_k:
            top.append((sc, a))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, a)
            top.sort(key=lambda x: x[0])

    # d) a few plain uniform‑random vectors (Dirichlet with α=1)
    for _ in range(30):
        if time.time() >= deadline:
            break
        a = rng.random(N)
        a /= a.sum()
        sc = _objective(a)
        if len(top) < top_k:
            top.append((sc, a))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, a)
            top.sort(key=lambda x: x[0])

    # ----- 2) incorporate the global starter if available -----------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                # simple linear interpolation to required length
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                sc = _objective(starter_up)
                if len(top) < top_k:
                    top.append((sc, starter_up))
                    top.sort(key=lambda x: x[0])
                elif sc < top[-1][0]:
                    top[-1] = (sc, starter_up)
                    top.sort(key=lambda x: x[0])
    except Exception:
        pass

    # pick the current best
    best_score, best_seq = top[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Local refinement of each retained candidate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30 and len(top) > 0:
        # split the remaining budget evenly among the top candidates,
        # giving each candidate two refinement phases.
        per_candidate = remaining / (len(top) * 2.0)
        for sc, seq in top:
            # smooth GD phase
            sub_deadline = time.time() + per_candidate
            refined_seq, refined_score = _smooth_gradient_descent(seq, sub_deadline, start_time)
            if refined_score < best_score:
                best_score, best_seq = refined_score, refined_seq
                print(f"[{time.time() - start_time:.2f}s] SmoothGD improved best = {best_score:.6f}")

            # sub‑gradient GD phase
            sub_deadline = time.time() + per_candidate
            refined_seq, refined_score = _subgradient_descent(best_seq, sub_deadline, start_time)
            if refined_score < best_score:
                best_score, best_seq = refined_score, refined_seq
                print(f"[{time.time() - start_time:.2f}s] SubGrad improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        annealed_seq, annealed_score = _anneal(best_seq, deadline, start_time)
        if annealed_score < best_score:
            best_score, best_seq = annealed_score, annealed_seq
            print(f"[{time.time() - start_time:.2f}s] Anneal improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
