"""
Propose a non‑negative step function (list of floats) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv(a, a) is the linear self‑convolution of the list.
The algorithm consists of three phases:

1️⃣  A very broad random parametric sweep over several analytic families
    (geometric, power‑law, beta‑type, Gaussian, cosine‑window, Tukey,
    Cauchy).  Each candidate is normalised to sum = 1 and the true
    objective is evaluated with an FFT‑based convolution.  The best
    candidate from this stage seeds the next phase.

2️⃣  A long projected Adam optimisation on the probability simplex.
    Instead of the sub‑gradient of the hard max, a *smooth* approximation
    of the max (log‑sum‑exp) is used.  For a temperature parameter α the
    smooth objective is

        S_α(a) = (1/α)·log Σ_k exp(α·conv_k)

    whose gradient can be written in closed‑form as

        ∇S_α = 2·(a * w) ,   w_k ∝ exp(α·conv_k) .

    increased slowly from a small value (≈5) to a large one (≈500) so the
    optimiser first follows a smooth landscape and then focuses on the
    true max.  After each Adam step the vector is projected onto the
    probability simplex.  The best iterate w.r.t. the *true* objective
    (using the hard max) is kept.

3️⃣  A stochastic hill‑climb that repeatedly transfers a tiny random
    fraction of mass from one index to another and accepts only improving
    moves.  This phase runs for the last part of the time budget.

All work is done with FFT‑based convolutions (real FFT) and the code
honours the 1000 s time limit.  The returned list is ready for
``evaluate_sequence``.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a * b using FFT.
    Returns the first len(a)+len(b)-1 entries (the linear result).
    """
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Fa * Fb, n=L)
    return conv[: len(a) + len(b) - 1]


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (length 2·len(a)-1)."""
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

    """True objective 2·n·max(conv) (assuming sum(a)=1)."""
    return float(2.0 * n * conv.max())

    """Compute the true objective for a normalised sequence."""
    conv = _conv_self(seq, L_conv_self)
    return _objective_from_conv(conv, n)

    idx = np.arange(n, dtype=np.float64)

    families = ['geometric', 'power', 'beta', 'gaussian',
                'cosine', 'tukey', 'cauchy']

    # number of random draws – cheap, we can afford a few thousand
    max_trials = 6000
    for trial in range(max_trials):
        # stop a little early to keep time for optimisation phases
        if time.time() >= deadline - 300.0:
            break

        family = np.random.choice(families)

        if family == 'geometric':
            r = np.random.uniform(0.5, 0.9999)
            base = np.power(r, idx)

        elif family == 'power':
            p = np.random.uniform(-2.0, 4.0)
            base = np.power(idx + 1.0, p)

        elif family == 'beta':
            p = np.random.uniform(-0.5, 4.5)
            q = np.random.uniform(-0.5, 4.5)
            base = np.power(idx + 1.0, p) * np.power(n - idx, q)

        elif family == 'gaussian':
            # sigma as a fraction of n, centre may be shifted
            sigma_rel = np.random.uniform(0.04, 0.30)
            mu = np.random.uniform(0, n - 1)
            base = np.exp(-0.5 * ((idx - mu) / (sigma_rel * n)) ** 2)

        elif family == 'cosine':
            p = np.random.uniform(0.5, 6.0)
            base = np.sin(np.pi * (idx + 0.5) / (n + 1)) ** p

        elif family == 'tukey':
            alpha = np.random.uniform(0.0, 1.0)
            # Efficient Tukey window computation
            x = idx / (n - 1)
            base = np.empty_like(x)
            mask1 = x < alpha / 2
            mask2 = (x >= alpha / 2) & (x <= 1 - alpha / 2)
            mask3 = x > 1 - alpha / 2
            base[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * x[mask1] / alpha - 1)))
            base[mask2] = 1.0
            base[mask3] = 0.5 * (1 + np.cos(np.pi * (2 * x[mask3] / alpha - 2 / alpha + 1)))

        elif family == 'cauchy':
            mu = np.random.uniform(0, n - 1)
            gamma_rel = np.random.uniform(0.02, 0.25)
            gamma = gamma_rel * n
            base = 1.0 / ((idx - mu) ** 2 + gamma ** 2)

        else:
            continue

        # Mix with uniform to keep a certain amount of flatness
        lam = np.random.rand()  # weight for uniform component
        seq = lam * uniform + (1.0 - lam) * base
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s

        cur = _score(seq, L_self, n)
        if cur < best_score:
            best_score = cur
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New best from {family} trial "
                  f"score={cur:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam optimiser on the simplex
    # --------------------------------------------------------------
    # We'll reserve ~150 s for the final hill‑climb
    hill_time = 150.0
    adam_deadline = deadline - hill_time
    a = best_seq.copy()
    best_a = a.copy()
    best_obj = best_score

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15                     # relatively aggressive
    alpha_start = 5.0
    alpha_max = 500.0

    it = 0
    while time.time() < adam_deadline:
        it += 1

        # FFT of current iterate
        Fa = np.fft.rfft(a, n=L_grad)

        # self‑convolution (first 2n‑1 entries are the linear result)
        conv_full = np.fft.irfft(Fa * Fa, n=L_grad)[: 2 * n - 1]
        max_conv = conv_full.max()

        # true objective for logging / tracking
        cur_obj = float(2.0 * n * max_conv)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"best_obj={best_obj:.6f}")

        # smooth max (log‑sum‑exp) – temperature α grows slowly
        alpha = min(alpha_max, alpha_start * (1.0 + it * 1e-4))
        # shift by max to avoid overflow
        shifted = conv_full - max_conv
        w = np.exp(alpha * shifted)          # max weight == 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of the smooth max: 2·(a * w_norm)
        Fw = np.fft.rfft(w_norm, n=L_grad)
        grad_full = np.fft.irfft(Fa * Fw, n=L_grad)[:n]   # first n entries = (a * w) restricted
        grad = 2.0 * grad_full

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

        # occasional tiny perturbation to escape plateaus
        if it % 2500 == 0:
            rng = np.random.default_rng()
            noise = 5e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    # keep the best iterate found by Adam
    a = best_a.copy()
    best_obj = best_obj   # already stored

    # --------------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climb (more aggressive transfers)
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = np.random.randint(n)
        if a[src] < 1e-13:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        # transfer a random fraction (up to a few percent) of src's mass
        frac = np.random.rand() * 0.04 + 1e-6   # ≈[1e‑6, 0.04]
        eps_transfer = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps_transfer
        a_prop[dst] += eps_transfer

        cur_obj = _score(a_prop, L_self, n)
        if cur_obj < best_obj:
            a = a_prop
            best_obj = cur_obj
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 0:
        a = a / total
    else:
        a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_obj:.6f}")
    return [float(x) for x in a]
```
