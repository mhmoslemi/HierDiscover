def _weighted_gradient(a, w):
    n = a.shape[0]
    a_rev = a[::-1]
    # term1
    term1_full = np.convolve(w, a_rev, mode='full')
    # term1 slice length n
    start = n - 1
    term1 = term1_full[start:start + n]
    # window sum of w length n (sliding)
    # compute prefix sum: cum[0]=0, cum[i+1]=cum[i]+w[i] etc.
    cum = np.concatenate(([0.0], np.cumsum(w)))
    # window sum of w indexes [j, j+n-1], for j=0..n-1
    win = cum[n:] - cum[:n]
    term2 = a * win
    grad = term1 + term2
    return grad

"""
Propose a non‑negative step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

using a smooth‑max gradient descent (Adam) on the probability simplex.
The algorithm consists of:

1️⃣  Warm‑start catalogue – many analytically‑motivated shapes (uniform,
    tent, power‑law, exponential, Gaussian, arcsine) – are normalised to
    sum 1 and the best of them (by the exact objective) is taken.

2️⃣  Smooth‑max Adam optimisation – we approximate the max by a soft‑max
    (log‑sum‑exp) with a temperature that slowly increases.
    The gradient of the smoothed objective can be written in closed form
    as a weighted convolution, which we compute efficiently with NumPy.
    Adam with Euclidean projection onto the simplex drives the solution
    towards a flat self‑convolution.

3️⃣  A tiny simulated‑annealing polish (optional, run only if time permits).

The whole search respects the 1000 s budget, works for a single CPU and
returns a plain list of non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Primitive numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Equivalent to ``evaluate_sequence`` (without defensive checks).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑start shape generators (all return a probability vector)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _collect_warm_starts(n: int) -> list[np.ndarray]:
    """Build a modest library of diverse initial vectors."""
    cand = []

    # deterministic analytic families
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # random Dirichlet draws – they often give surprises
    rng = np.random.default_rng()
    for _ in range(6):
        a = rng.gamma(0.5, 1.0, size=n)   # small shape -> spiky draws
        a = a / a.sum()
        cand.append(a)

    # optional starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                cand.append(starter)
    except Exception:
        pass

    # deduplicate / remove NaNs
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Gradient of the (soft)‑max of the self‑convolution
# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a, where w is a probability vector
    over the 2·n‑1 convolution entries.
    """
    n = a.shape[0]
    # term 1 : Σ_k w_k a_{k-j}
    a_rev = a[::-1]                         # reversed copy
    full = np.convolve(w, a_rev, mode='full')
    start = n - 1
    term1 = full[start:start + n]           # length n

    # term 2 : a_j Σ_{k=j}^{j+n-1} w_k   (sliding window sum of w)
    # prefix sums for O(n) sliding window
    cum = np.concatenate(([0.0], np.cumsum(w)))   # length 2·n
    win = cum[n:] - cum[:n]                        # w[j] + ... + w[j+n-1]
    term2 = a * win

    return term1 + term2


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   alpha0: float = 5.0,
                   lr0: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   alpha_max: float = 50.0,
                   inc_every: int = 1000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth‑max of the self‑convolution.
    Returns the best sequence seen and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    alpha = alpha0
    lr = lr0

    while time.time() < deadline:
        t += 1

        # self‑convolution
        conv = _conv_self(a)
        cur_max = conv.max()

        # smooth‑max weights (soft‑max centred at the current maximum)
        # avoids overflow by subtracting cur_max
        w = np.exp(alpha * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of the smoothed objective
        grad = _weighted_gradient(a, w)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # stay on the simplex
        a = _project_simplex(a)

        # evaluate true objective (expensive but cheap enough)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} "
                  f"score={cur_score:.6f}")

        # schedule: increase the temperature (alpha) and decay learning rate
        if t % inc_every == 0:
            alpha = min(alpha_max, alpha * 1.07)
            lr *= 0.93

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap Metropolis walk that moves a random packet of mass."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # a modest number of trials – we run only while time permits
    it = 0
    while time.time() < deadline:
        it += 1
        # exponential cooling; starts at 0.02 and ends near 1e‑6
        T = 0.02 * (1e-6 / 0.02) ** (it / 200_000)

        src = rng.choice(n, p=a)          # source chosen proportionally to mass
        dst = rng.integers(n)             # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30   # up‑to 30 % of source mass
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps

        new_score = _objective(prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal step={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    The returned list contains non‑negative floats and will be fed to
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # 4096 is a good balance; we also allow a larger size if time permits.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start catalogue
    # ------------------------------------------------------------------
    candidates = _collect_warm_starts(n)

    # ------------------------------------------------------------------
    # 2️⃣  Pick the best warm‑start (exact objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates, 1):
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] Warm‑start {idx}/{len(candidates)} "
              f"score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # 3️⃣  Adam optimisation (smooth‑max) – allocate most of the time.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:               # reserve a few seconds for polishing
        opt_deadline = time.time() + remaining * 0.88
        refined_seq, refined_score = _adam_optimize(
            best_seq,
            opt_deadline,
            start_time,
            alpha0=5.0,
            lr0=0.12,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha_max=50.0,
            inc_every=800
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Light annealing polish (if anything is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final sanity checks and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
