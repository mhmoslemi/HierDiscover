"""
Advanced three‑phase optimizer for the step‑function objective

The algorithm consists of:
  1. Smooth‑max mirror descent using an Adam‑style update on the
     log‑sum‑exp approximation of the maximum convolution value.
  2. Direct max‑subgradient descent (exponentiated‑gradient) on the
     true ℓ∞‑norm of the self‑convolution.
  3. Targeted greedy hill‑climbing that always moves mass away from the
     pair of entries that contribute most to the current maximal
     convolution value.

Multiple short runs (different random seeds / initialisations) are
performed within the time budget; the best sequence found is polished
with a final hill‑climb before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Problem size (the same size as the previous best solution)
# ----------------------------------------------------------------------
n = 4096                     # length of the step function
# FFT lengths – powers of two, large enough for the required convolutions
_L_SELF = 1 << ((2 * n - 2).bit_length())   # >= 2n‑1
_L_GRAD = 1 << ((3 * n - 3).bit_length())   # >= 3n‑2

# ----------------------------------------------------------------------
# Fast exact evaluation (used internally; identical to evaluate_sequence
# up to the harmless clipping of values to [0,1000]).
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using FFT (a must be non‑negative)."""
    Af = np.fft.rfft(a, n=_L_SELF)
    conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : Smooth‑max mirror descent with Adam on the simplex
# ----------------------------------------------------------------------
def _smooth_descent_adam(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑accelerated mirror descent on the log‑sum‑exp surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _exact_score(best_a)

    # geometric schedule for the smoothing parameter
    alphas = np.geomspace(0.5, 5000.0, num=60)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.025                           # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    for alpha in alphas:
        while time.time() < deadline - 0.2:
            t += 1
            beta1_pow *= beta1
            beta2_pow *= beta2

            # 1) self‑convolution a*a
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]

            # 2) smooth‑max (softmax) weights
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)      # stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient of the smooth‑max surrogate:
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_GRAD)
            Revf = np.fft.rfft(a_rev, n=_L_GRAD)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam update on the (log‑space) mirror‑descent direction
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            # step size adaption – larger α ⇒ smaller step (as in the original code)
            lr = lr0 / max(1.0, math.sqrt(alpha))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            # 5) exponentiated‑gradient update on the simplex
            #   a_i ← a_i * exp(−step_i) , then renormalise
            a = a * np.exp(-step)
            a = np.maximum(a, 1e-30)          # avoid exact zeros
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            else:                              # numerical safeguard
                a.fill(1.0 / n)

            # occasional exact evaluation
            if t % 5000 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] smooth‑adam α={alpha:.1f} "
                          f"it={t:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Max‑subgradient descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _max_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Push mass away from the index that realises the current max convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _exact_score(best_a)

    it = 0
    while time.time() < deadline - 0.2:
        it += 1

        # current self‑convolution and its maximum index
        Af = np.fft.rfft(a, n=_L_SELF)
        conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
        k = int(conv.argmax())

        # sub‑gradient for (a*a)[k] = Σ a_i a_{k-i}
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # g_i = 2·a_{k‑i}

        # exponentiated‑gradient step (step size decays like 1/√t)
        eta = 0.5 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a.fill(1.0 / n)

        # periodic exact evaluation
        if it % 3000 == 0:
            cur_score = _exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : Targeted greedy hill‑climbing
# ----------------------------------------------------------------------
def _targeted_hill_climb(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves that always attack the pair of entries
    that most contribute to the current maximal convolution value.
    """
    cur = a0.copy()
    cur_score = _exact_score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1

        # 1) current convolution and its maximum index
        Af = np.fft.rfft(cur, n=_L_SELF)
        conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
        k = int(conv.argmax())

        # 2) find the pair (i, j) that contributes most to this maximum
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)                # all i with a_i a_{k-i}
        contrib = cur[idx] * cur[k - idx]                     # element‑wise products
        imax = idx[np.argmax(contrib)]                        # first index of the worst pair
        jmax = k - imax

        # 3) choose two destination indices that currently hold little mass
        #    (preferably far away from the source indices)
        low_mask = cur < np.median(cur)
        low_indices = np.where(low_mask)[0]
        # guarantee we have at least two candidates
        if low_indices.size < 2:
            dst = rng.choice(np.arange(n), size=2, replace=False)
        else:
            dst = rng.choice(low_indices, size=2, replace=False)

        # ensure destinations are distinct from the sources
        dst = [d for d in dst if d not in (imax, jmax)]
        while len(dst) < 2:
            cand = rng.integers(n)
            if cand not in (imax, jmax) and cand not in dst:
                dst.append(cand)
        d1, d2 = dst[:2]

        # 4) move a small fraction of the mass from the worst pair to the destinations
        eps_frac = rng.random() * 0.2 + 0.05           # between 5 % and 25 %
        eps = min(cur[imax], cur[jmax]) * eps_frac
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[imax] -= eps
        cand[jmax] -= eps
        cand[d1] += eps
        cand[d2] += eps

        # renormalise (numerical drift)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            continue

        cand_score = _exact_score(cand)
        if cand_score < cur_score:                     # greedy acceptance
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] targeted‑hill it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: run a single optimisation “run” (smooth → sub‑grad → hill‑climb)
# ----------------------------------------------------------------------
def _run_optimizer(initial: np.ndarray,
                   run_deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """Execute all three phases on a given initial vector."""
    # Phase‑1
    a, sc1 = _smooth_descent_adam(initial, run_deadline, start_time)
    if time.time() >= run_deadline - 5.0:
        return a, sc1

    # Phase‑2
    sub_deadline = min(run_deadline, time.time() + (run_deadline - time.time()) * 0.4)
    a, sc2 = _max_subgradient_descent(a, sub_deadline, start_time)
    best_sc = min(sc1, sc2)

    # Phase‑3
    if time.time() < run_deadline - 0.5:
        a, sc3 = _targeted_hill_climb(a, run_deadline, start_time)
        best_sc = min(best_sc, sc3)

    return a, best_sc

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine performs several short optimisation runs
    (different random initialisations) within the time budget and returns the
    best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Determine a reasonable number of independent runs
    # ------------------------------------------------------------------
    # we keep a couple of seconds for the final polishing pass
    max_runs = 4
    for run_idx in range(max_runs):
        # decide whether enough time is left for another run
        remaining = deadline - time.time()
        if remaining < 25.0:                     # each run needs at least ~20 s
            break

        # allocate a slice of the remaining budget to this run
        run_budget = remaining * 0.3
        run_deadline = time.time() + run_budget

        # --------------------------------------------------------------
        # initialise the starting vector for this run
        # --------------------------------------------------------------
        if run_idx == 0:
            # try to reuse the previously known best sequence if available
            try:
                init = np.asarray(height_sequence_1, dtype=float)
                init = np.maximum(init, 0.0)
                if init.size != n:
                    if init.size > n:
                        init = init[:n]
                    else:
                        init = np.pad(init, (0, n - init.size), mode='constant')
                s = init.sum()
                if s > 0.0:
                    init /= s
                else:
                    raise Exception
            except Exception:
                init = np.full(n, 1.0 / n)
        elif run_idx == 1:
            # pure uniform start
            init = np.full(n, 1.0 / n)
        else:
            # random Dirichlet start – concentration tuned for a slightly sparse shape
            conc = 0.6
            rng = np.random.default_rng()
            init = rng.dirichlet(np.full(n, conc))

        # --------------------------------------------------------------
        # Run the three‑phase optimiser on this start vector
        # --------------------------------------------------------------
        a, sc = _run_optimizer(init, run_deadline, start_time)

        # update global best
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Run {run_idx + 1}/{max_runs} "
                  f"new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        final_seq, final_sc = _targeted_hill_climb(best_seq, deadline, start_time)
        if final_sc < best_score:
            best_seq = final_seq
            best_score = final_sc
            print(f"[{time.time() - start_time:.2f}s] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and convert to plain Python list
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to a simple uniform vector (should never happen)
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # return a plain Python list as required by the framework
    return [float(v) for v in best_seq]
