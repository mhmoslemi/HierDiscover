def _subgradient_descent(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    lr = 0.5  # start
    min_lr = 1e-4
    decay = (min_lr / lr) ** (1 / 1e5) perhaps.

    L_fft = _next_pow2(2*n-1) maybe for conv.

    while time.time()<deadline:
        # compute conv and M
        conv = _conv_self(a, L=L_fft)  # length 2n-1
        M = conv.max()
        # compute w (weights)
        # Choose indices where conv close to M (within tolerance 1e-12 fraction)
        # Tolerance: rr = (np.abs(conv - M) < 1e-12) maybe relative.
        mask = np.isclose(conv, M, rtol=1e-12, atol=1e-15)
        idxs = np.where(mask)[0]
        if len(idxs) ==0:
            # fallback to simple
            idxs = [np.argmax(conv)]
        w = np.zeros_like(conv)
        w[idxs] = 1.0 / len(idxs) # uniform over max positions

        # Compute gradient via FFT correlation
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=_next_pow2(3*n-2))
        A = np.fft.rfft(a_rev, n=_next_pow2(3*n-2))
        corr = np.fft.irfft(W*A, n=_next_pow2(3*n-2))
        start = n-1
        grad = 2 * corr[start:start+n]

        # descent step
        a = a - lr * grad
        a = _project_simplex(a)
        # update best
        cur_score = 2.0 * n * M / (a.sum()**2)  # but a sum ≈ 1
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad iteration ... cur={cur_score:.6f}")

        # adapt lr (maybe reduce when not improving)
        # simple schedule: lr = max(min_lr, lr * 0.9999)
        lr = max(min_lr, lr * 0.9999)

def propose_candidate(seed=0,budget_s=1000,**kwargs):

    import time, numpy as np

    start_time = time.time()

    np.random.seed(seed)

    # Problem size
    n = 4096
    L_fft = _next_pow2(2*n-1)
    L_corr = _next_pow2(3*n-2)

    # Baseline uniform
    uniform = np.full(n, 1.0/n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time()-start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # Optionally use global starter
    # same as earlier.

    # Parametric sweep
    sweep_deadline = deadline - 400.0
    idx = np.arange(1,n+1,dtype=np.float64)
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-3.0, 2.0, 41)
    lam_vals = [0.0,0.2,0.5,0.8,1.0]
    for r in r_grid:
        if time.time() > sweep_deadline:
            break
        geo = r**np.arange(n, dtype=np.float64)
        for p in p_grid:
            if time.time() > sweep_deadline:
                break
            shape = geo * (idx**p)
            s=shape.sum()
            if s==0: continue
            shape /= s
            for lam in lam_vals:
                seq = lam*uniform + (1-lam)*shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time()-start_time:.2f}s] sweep r={r:.4f} p={p:.2f} lam={lam:.2f} score={sc:.6f}")

    # Subgradient descent
    subgrad_deadline = deadline - 150.0
    print(f"[{time.time()-start_time:.2f}s] Starting subgradient descent")
    a, sc = _subgradient_descent(best_seq, n, subgrad_deadline, start_time, best_score, L_fft, L_corr)
    if sc < best_score:
        best_seq = a.copy()
        best_score = sc
        print(f"[{time.time()-start_time:.2f}s] subgradient improvement score={sc:.6f}")

    # Hill climbing (simulated annealing)
    hc_deadline = deadline
    print(f"[{time.time()-start_time:.2f}s] Starting hill climbing")

def _subgradient_descent(a0, n, deadline, start_time, best_score_initial, L_fft, L_corr):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    lr = 0.2
    min_lr = 1e-5
    decay = 0.9995

    while time.time() < deadline:
        # compute conv
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        # compute active mask
        # tolerance:
        tol = max(1e-12, M*1e-12)
        mask = np.abs(conv - M) <= tol
        w = np.zeros_like(conv)
        active = np.where(mask)[0]
        if active.size == 0:
            active = [np.argmax(conv)]
        w[active] = 1.0 / active.size

        # compute gradient (correlation)
        a_rev = a[::-1]
        # w FFT
        W = np.fft.rfft(w, n=L_corr)
        A_rev = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A_rev, n=L_corr)
        start = n - 1
        grad = 2.0 * corr[start:start+n]

        # gradient step (additive)
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a == 0:
            # fallback: revert to uniform
            a = np.full(n, 1.0/n)
        else:
            a = a / sum_a

        # evaluate
        cur_score = 2.0 * n * M  # sum a ~1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] subgrad it={len(best_seq?)?} score={cur_score:.6f}")

        # increase iteration
        lr = max(min_lr, lr * decay)

        # occasional random noise
        if np.random.rand() < 0.001:
            # add small uniform noise
            noise = 1e-5 * np.random.randn(n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
    return best_a, best_score

def _simulated_hill_climb(a0, n, deadline, start_time, L_fft):
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    T0 = 1e-3  # initial temperature in objective scale
    # Use linear schedule: T = T0 * (time_left / total_time)
    total_time = deadline - start_time
    while time.time() < deadline:
        # fraction of time elapsed
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T0 * (1 - frac)  # linearly decay to zero

        # propose move
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # magnitude up to 0.1 * src mass for moderate moves; occasionally larger
        if rng.random() < 0.1:
            max_frac = 0.3
        else:
            max_frac = 0.05
        eps = a[src] * rng.random() * max_frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # renormalize (sum unchanged)
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T,1e-12)):
            # accept
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] hill climb score={best_score:.6f}")
    return best_a, best_score

probability simplex (sum = 1).  The algorithm consists of three phases:

1️⃣  A quick parametric sweep over geometric‑power families (including a
    mixture with the uniform distribution) to obtain a good starting
    point.

2️⃣  A fast sub‑gradient descent on the exact max‑convolution (no
    soft‑max approximation).  For the current active indices where the
    convolution attains its maximum we compute the sub‑gradient
    𝑔 = 2 * corr(w, a_rev) with an indicator weight vector w supported
    on those indices.  An additive step a ← a – η g is taken, negatives
    are clamped to zero and the vector is re‑normalised onto the simplex.
    The learning rate η decays geometrically.

3️⃣  A simulated‑annealing hill‑climber that repeatedly transfers a
    random fraction of mass between two coordinates.  Improvements are
    always kept; worsening moves are accepted with probability
    exp(−Δ/T) where the temperature T is slowly reduced to zero.
    The final best sequence is returned as a plain list of floats.

All heavy operations use FFT‑based convolutions (O(n log n)).
The routine respects the 1000 s time budget and prints progress
information.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {w≥0, Σw=1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact evaluation of the objective for a vector with sum≈1.
    Returns 2·n·max(conv(a,a)).
    """
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 2 – sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    L_fft: int,
    L_corr: int,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on J(a)=2·n·max(conv(a,a)) with the simplex constraint.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # learning‑rate schedule
    eta = 0.20
    eta_min = 1e-5
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        it += 1
        # 1. current convolution and its maximum
        conv = _conv_self(a, L=L_fft)
        M = conv.max()

        # 2. sub‑gradient (uniform on the active set)
        tol = max(1e-12, M * 1e-12)
        active = np.where(np.abs(conv - M) <= tol)[0]
        if active.size == 0:
            active = [np.argmax(conv)]
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # 3. gradient = 2 * corr(w, a_rev)
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)
        start = n - 1
        grad = 2.0 * corr[start : start + n]

        # 4. gradient step, clamp negatives, renormalise onto simplex
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # 5. keep the best seen so far
        cur_score = 2.0 * n * M     # sum(a)≈1 by construction
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                f"score={cur_score:.6f}"
            )

        # 6. learning‑rate decay
        eta = max(eta_min, eta * decay)

        # occasional tiny random shake‑up to leave plateaus
        if it % 3000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – simulated‑annealing hill climb (exact objective)
# ----------------------------------------------------------------------
def _simulated_hill_climb(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers between two coordinates.
    Accept improving moves; accept worsening moves with probability
    exp(−Δ/T) where T is linearly cooled to zero.
    """
    a = a0.copy()
    cur_score = _eval_fast(a, L_fft)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    total_time = deadline - start_time

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        # temperature decays linearly from 1e-3 to 0
        T = max(1e-12, 1e-3 * (1.0 - frac))

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # occasionally allow larger jumps (up to 30 % of src mass)
        max_frac = 0.3 if rng.random() < 0.1 else 0.05
        eps = a[src] * rng.random() * max_frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # sum unchanged, all entries stay non‑negative by construction

        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < math.exp(-delta / T):
            # accept the move
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – lightweight parametric sweep
# ----------------------------------------------------------------------
def _parametric_sweep(
    n: int,
    uniform: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Sweep over sequences of the form a_i ∝ r^i·(i+1)^p
    optionally mixed with the uniform distribution.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)  # i+1
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-3.0, 2.0, 41)
    lam_vals = (0.0, 0.2, 0.5, 0.8, 1.0)

    for r in r_grid:
        if time.time() >= deadline:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_vals:
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} "
                        f"λ={lam:.2f} score={sc:.6f}"
                    )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the given
    evaluation function.  The routine respects a 1000 s budget and
    returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # problem size – 4096 gives a good trade‑off between flexibility and speed
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # baseline: uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optional global starter (if the execution environment provides it)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / s
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (quick)
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 400.0   # keep ~6 min for later phases
    best_seq, best_score = _parametric_sweep(
        n, uniform, best_seq, best_score, sweep_deadline, start_time, L_fft
    )

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent (core optimisation)
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 150.0   # leave ~2½ min for hill‑climb
    print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
    a_sg, sc_sg = _subgradient_descent(
        best_seq,
        n,
        subgrad_deadline,
        start_time,
        best_score,
        L_fft,
        L_corr,
    )
    if sc_sg < best_score:
        best_score = sc_sg
        best_seq = a_sg.copy()
        print(f"[{time.time() - start_time:.2f}s] sub‑gradient improvement score={sc_sg:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing hill‑climb (fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        a_hc, sc_hc = _simulated_hill_climb(best_seq, n, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb final best score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return as a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
