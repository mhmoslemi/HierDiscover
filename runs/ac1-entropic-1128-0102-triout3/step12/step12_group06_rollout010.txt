"""
Hybrid optimiser for the step‑function objective

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear (full) convolution.
The optimiser works on a reduced size (default n = 1024) to allow many
iterations inside the 1000 s budget.

Main ideas
---------
1.  **Smooth‑max mirror descent** – exponentiated‑gradient on a
    log‑sum‑exp approximation of the max, with a wide geometric schedule
    for the smoothing parameter α (up to 1e5).  This quickly drives the
    distribution away from the worst convolution peak.

2.  **Direct max‑subgradient descent** – after the smoothing phase we
    use the exact sub‑gradient at the current maximal index.  An
    exponentiated‑gradient step on the simplex with a slower decay
    `η ∝ t⁻⁰·⁶` gives finer improvement.

3.  **Targeted hill‑climb** – instead of random source/destination
    transfers we concentrate mass on the indices that actually *contribute*
    to the maximal convolution entry.  Mass is moved from the highest
    contributors to random low‑weight locations, with a simple Metropolis
    acceptance rule for occasional uphill moves.

4.  **Multiple restarts** – the whole pipeline is executed from several
    random Dirichlet starts (including the supplied `height_sequence_1`);
    the best solution found overall is returned.

All helper functions are top‑level (no closures) and the code stays fully
on the probability simplex (`∑a = 1`) while respecting non‑negativity.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (no closures)
# ----------------------------------------------------------------------
def _fft_len(full_len: int) -> int:
    """Smallest power of two >= full_len."""
    return 1 << (full_len - 1).bit_length()


def _conv_full(a: np.ndarray, n: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    L = _fft_len(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _exact_score(a: np.ndarray, n: int) -> float:
    """Exact objective value (no smoothing)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_full(a, n)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Log‑sum‑exp (smooth‑max) weights for a convolution array."""
    exp_arg = alpha * conv
    max_exp = exp_arg.max()
    w = np.exp(exp_arg - max_exp)          # improve numerical stability
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    return w


def _smooth_gradient(a: np.ndarray, w: np.ndarray, n: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective:
        g = 2·(w ⋆ rev(a))
    where `⋆` denotes linear convolution and rev(a)[i] = a[n‑1‑i].
    """
    # prepare sizes for the convolution w * rev(a)
    Lg = _fft_len(3 * n - 2)                # length needed for full product
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=Lg)
    Revf = np.fft.rfft(a_rev, n=Lg)
    grad_full = np.fft.irfft(Wf * Revf, n=Lg)[: 3 * n - 2]
    # central part corresponds to the gradient w.r.t. the original a
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _subgradient_at_max(a: np.ndarray, n: int) -> np.ndarray:
    """
    Sub‑gradient of the true max‑conv objective at the current maximal index.
    """
    conv = _conv_full(a, n)
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]            # includes the factor 2 for the diagonal term
    return grad, conv.max()


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max mirror descent
# ----------------------------------------------------------------------
def _smooth_descent(a: np.ndarray, n: int, deadline: float,
                    start_time: float, best_score: float):
    """Exponentiated‑gradient on a log‑sum‑exp approximation of the max."""
    best_a = a.copy()
    it = 0
    # α spans from 0.5 up to 1e5 (geometric schedule)
    alphas = np.geomspace(0.5, 1e5, num=70)

    for alpha in alphas:
        if time.time() >= deadline - 0.2:
            break
        # inner loop for the current α
        while time.time() < deadline - 0.2:
            it += 1

            conv = _conv_full(a, n)
            w = _softmax_weights(conv, alpha)

            grad = _smooth_gradient(a, w, n)

            # exponentiated‑gradient step on the simplex
            eta = 0.2 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-30)               # protect from exact zeros
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            else:
                a = np.full_like(a, 1.0 / n)

            # occasional exact evaluation
            if it % 6000 == 0:
                cur_score = _exact_score(a, n)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time()-start_time:.2f}s] smooth α={alpha:.1e} it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – direct max‑subgradient descent
# ----------------------------------------------------------------------
def _max_subgradient_descent(a: np.ndarray, n: int, deadline: float,
                             start_time: float, best_score: float):
    """Exponentiated‑gradient sub‑gradient descent on the true max."""
    best_a = a.copy()
    it = 0
    while time.time() < deadline - 0.2:
        it += 1
        grad, cur_max = _subgradient_at_max(a, n)

        # step size with a slightly slower decay than 1/√t
        eta = 0.5 / (it ** 0.6)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        if it % 5000 == 0:
            cur_score = _exact_score(a, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – targeted hill‑climbing (focused simulated annealing)
# ----------------------------------------------------------------------
def _targeted_hill_climb(a: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score: float):
    """Move mass away from the biggest contributors to the peak."""
    cur = a.copy()
    cur_score = _exact_score(cur, n)

    best_seq = cur.copy()
    best = best_score

    rng = np.random.default_rng()
    it = 0
    start_temp = 5e-2
    end_temp = 1e-12

    while time.time() < deadline:
        it += 1

        # progress‑dependent temperature (geometric decay)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        # compute the index that attains the current maximum
        conv = _conv_full(cur, n)
        k = int(conv.argmax())

        # indices that actually appear in that convolution term
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # contribution of each index to the maximal entry
        contrib = cur[idx] * cur[k - idx] * 2.0
        middle = (idx == k - idx)
        contrib[middle] = cur[idx[middle]] * cur[idx[middle]]   # diagonal counted once

        src = idx[np.argmax(contrib)]

        # choose a destination that is not the source
        dst = rng.integers(n)
        if dst == src:
            continue

        # move a modest fraction of the source mass
        eps = cur[src] * rng.random() * 0.35
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        cand_score = _exact_score(cand, n)
        delta = cand_score - cur_score

        # Metropolis criterion
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score
            if cand_score < best:
                best = cand_score
                best_seq = cand.copy()
                print(f"[{time.time()-start_time:.2f}s] hill‑climb it={it:,} best={best:.6f}")

        # occasional random shake‑up to escape plateaus
        if it % 15000 == 0:
            perturb = rng.normal(scale=0.01, size=n)
            shaken = best_seq + perturb
            shaken = np.maximum(shaken, 0.0)
            shaken_sum = shaken.sum()
            if shaken_sum > 0.0:
                shaken /= shaken_sum
                shaken_score = _exact_score(shaken, n)
                if shaken_score < best:
                    best = shaken_score
                    best_seq = shaken.copy()
                    cur = shaken.copy()
                    cur_score = shaken_score
                    print(f"[{time.time()-start_time:.2f}s] random shake it={it:,} best={best:.6f}")

    return best_seq, best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the allotted time budget.

    The routine works on a reduced length (default 1024) to allow many
    optimisation steps.  It combines smooth‑max mirror descent,
    max‑subgradient descent and a targeted hill‑climbing phase,
    performing several random restarts and returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3      # safety margin

    # ------------------------------------------------------------------
    # Problem size – we use a moderate n (speed vs quality trade‑off)
    # ------------------------------------------------------------------
    n = 1024

    # ------------------------------------------------------------------
    # Initialise a starting point
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise Exception
        # adjust length to n
        if starter.size > n:
            starter = starter[:n]
        elif starter.size < n:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    # normalise onto the probability simplex
    s = starter.sum()
    if s > 0.0:
        starter /= s
    else:
        starter = np.full(n, 1.0 / n)

    best_seq = starter.copy()
    best_score = _exact_score(best_seq, n)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Run several independent restarts (each using the three‑phase pipeline)
    # ------------------------------------------------------------------
    num_restarts = 4
    for rs in range(num_restarts):
        if time.time() >= deadline - 30.0:      # keep a little slack for final polish
            break

        # random Dirichlet start (except the first which re‑uses the uniform starter)
        if rs == 0:
            a0 = starter.copy()
        else:
            a0 = np.random.dirichlet(np.ones(n))
        cur_score = _exact_score(a0, n)

        # Phase 1 – smooth‑max mirror descent
        phase_deadline = min(deadline - 20.0, time.time() + (deadline - time.time()) * 0.3)
        a1, s1 = _smooth_descent(a0, n, phase_deadline, start_time, cur_score)

        # Phase 2 – max‑subgradient descent
        phase_deadline = min(deadline - 10.0, time.time() + (deadline - time.time()) * 0.3)
        a2, s2 = _max_subgradient_descent(a1, n, phase_deadline, start_time, s1)

        # Phase 3 – targeted hill‑climb
        a3, s3 = _targeted_hill_climb(a2, n, deadline, start_time, s2)

        if s3 < best_score:
            best_score = s3
            best_seq = a3.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart {rs+1}/{num_restarts} NEW BEST = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
