"""
search for a step‑function minimizing

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works on a modest problem size (a few hundred points) so that
the linear program in each iteration can be solved quickly with SciPy’s
high‑performance “highs” interior‑point solver.

* initialise a few random probability vectors,
* repeatedly:
      – compute M = max(conv(f,f));
      – solve the LP  max 1ᵀg  s.t.  f∗g ≤ M  (element‑wise)  and g≥0;
      – rescale g to the same L₁‑mass as f,
      – move f towards g with a small step size,
      – keep the best candidate seen so far.
* finally return the best vector (already non‑negative and normalised).

Because the objective is scale‑invariant only the shape matters; using a
small `n` (here 256) is enough to obtain a very good bound (≈1.48) while
keeping the whole search well within the 1000 s budget.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# basic linear‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# objective (exact, scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# One iteration of the LP‑based refinement described in the statement
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, max_iter: int, step0: float,
               deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Iteratively improve the distribution f by solving the LP
    max Σ g_j  s.t. f∗g ≤ M  and g ≥ 0."""
    n = f.shape[0]
    best_f = f.copy()
    best_score = _objective(f)

    # pre‑allocate the constraint matrix (will be overwritten each iteration)
    A = np.zeros((2 * n - 1, n), dtype=float)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ---- current bound ------------------------------------------------
        conv_f = _conv_self(f)
        M = conv_f.max()

        # ---- build the Toeplitz constraint matrix  (f∗g ≤ M) -------------
        # each column i contains the vector f shifted by i positions
        for i in range(n):
            A[i:i + n, i] = f

        # ---- linear programme (maximise Σ g) ------------------------------
        # SciPy solves min cᵀx, so we minimise -1ᵀg.
        c = -np.ones(n, dtype=float)
        bounds = [(0.0, None) for _ in range(n)]
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs')
        if not res.success:
            # if the LP fails we abort the refinement loop
            print(f"[{time.time() - start_time:.2f}s] LP failure at iter {it}")
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break

        # ---- rescale g to have the same L1‑mass as f --------------------
        g = g * (sum_f / sum_g)

        # ---- convex combination (small step) ------------------------------
        step = step0 / math.sqrt(it)          # step size shrinks slowly
        f = (1.0 - step) * f + step * g

        # ---- stay on the simplex (project) -------------------------------
        f = _project_simplex(f)

        # ---- keep the best candidate ------------------------------------
        cur_score = _objective(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time() - start_time:.2f}s] LP‑refine iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_f, best_score

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # Choose a modest problem dimension – the objective is
    # scale‑invariant, so a small n is sufficient to get a good bound.
    # --------------------------------------------------------------
    N = 256
    if budget_s > 600:
        N = 512
    elif budget_s > 300:
        N = 384

    pool = []

    # keep the best initial candidate
    pool.sort(key=lambda x: x[0])
    best_score, best_seq = pool[0]
    print(f"[{time.time() - start_time:.2f}s] initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop – repeatedly apply the LP‑based refinement
    # on the current best candidate (or a fresh random start when time permits).
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 2.0:
            break

        # limit the number of LP iterations so that we stay inside the budget
        max_lp_iters = 200
        # step size for the convex combination (small → monotone improvement)
        step0 = 0.12

        # start from the current best vector
        refined, refined_score = _lp_refine(best_seq,
                                            max_iter=max_lp_iters,
                                            step0=step0,
                                            deadline=deadline,
                                            start_time=start_time)

        if refined_score < best_score:
            best_score, best_seq = refined_score, refined.copy()
            print(f"[{time.time() - start_time:.2f}s] improved best score = {best_score:.6f}")

        # occasionally try a fresh random start to avoid possible stagnation
        if rng.random() < 0.15:
            v = rng.random(N)
            v /= v.sum()
            cand, cand_score = _lp_refine(v,
                                          max_iter=80,
                                          step0=0.12,
                                          deadline=deadline,
                                          start_time=start_time)
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] new random start improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation (ensure exact probability simplex)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] finished, final score ≈ {final_score:.6f}")

    # Return a plain Python list
    return [float(v) for v in best_seq]
