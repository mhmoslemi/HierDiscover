"""
Propose a step function (non‑negative list) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three stages:

1️⃣  **Extended parametric sweep** – geometric (both decreasing and
    increasing), power‑law and two‑parameter beta families, plus a
    random‑sampling stage that explores a broader exponent range.
    All candidates are projected to the probability simplex
    (sum = 1), evaluated quickly via FFT convolution and the exact
    objective, and the best is kept.

2️⃣  **Multiple Adam‑style projected gradient runs** – each run starts
    from a slightly perturbed copy of the current best sequence.
    The sub‑gradient of the max‑convolution is taken as the average of
    the gradients of all active (max‑attaining) shifts.  Adam’s
    moments are reset for each run; a small Gaussian shake‑up is
    performed when progress stalls.

3️⃣  **Stochastic hill‑climb with temperature** – random mass transfers
    between two positions are accepted if they improve the true
    objective; with a linearly decreasing temperature we occasionally
    also accept slight deteriorations to escape tiny plateaus.

The routine respects a total wall‑time budget (default 1000 s),
uses only a single CPU, and returns a plain Python list ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (full length)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq))/(sum(seq)²).  Assumes seq≥0."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    maxb = conv.max()
    s = seq.sum()
    # protection against degenerate sum – not expected here
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    Takes the arithmetic mean of the gradients of all active (max‑attaining) shifts.
    """
    n = a.shape[0]
    max_val = conv.max()
    # active shifts (tiny tolerance)
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    if active.size == 0:
        return np.zeros_like(a)

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution a[k-idx] corresponds to a paired with the shifted copy
        grad[idx] += a[k - idx]
    # each pair appears twice (i ↔ k‑i), hence the factor 2
    grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected optimisation (single run)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex starting from a0.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15                # a bit more conservative than the previous run

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_score = _objective(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # sub‑gradient
        grad = _subgradient_maxconv(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection back onto simplex
        a = _project_simplex(a)

        # if stuck for many iterations, add a tiny perturbation
        if it - last_improve > 4000:
            noise = 5e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with a simple temperature schedule
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that moves a tiny random fraction of mass between two positions.
    Accepts improving moves; with a linearly decreasing temperature it also
    accepts marginally worsening moves to escape tiny plateaus.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    rng = np.random.default_rng()
    total_iters = 0
    while time.time() < deadline:
        total_iters += 1
        # temperature linearly drops from 0.02 to 0
        temp = max(0.0, 0.02 * (deadline - time.time()) / (deadline - start_time))

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # random fraction up to 5 %
        frac = rng.random() * 0.05 + 1e-6
        delta = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        # keep feasibility
        if a_prop[src] < -1e-15:
            continue

        cur_score = _objective(a_prop)

        improve = cur_score < best_score
        if improve or (temp > 0 and rng.random() < np.exp(-(cur_score - best_score) / temp)):
            a = a_prop
            if improve:
                best_a = a.copy()
                best_score = cur_score
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    2·n·max(conv(a,a)) / (sum(a)²).

    The search consists of:
        * an extended parametric sweep (geometric, power, beta, random);
        * several short Adam‑style projected gradient runs from perturbed starts;
        * a final stochastic hill‑climb with temperature.
    The routine respects the given time budget and returns a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline: uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter exists (height_sequence_1) use it
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = _project_simplex(starter)   # force sum=1
            cur = _objective(starter)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Extended parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric decreasing (r close to 1) – already explored, but denser grid
    eps_grid = np.logspace(-6, -0.2, 250)          # r = 1 - eps
    for eps in eps_grid:
        if time.time() >= deadline - 250:          # keep ~250 s for later stages
            break
        r = 1.0 - eps
        geo = r ** idx
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo = geo / geo_sum
        # mixture with uniform – lam in [0,1] coarse grid
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = _project_simplex(seq)
            cur = _objective(seq)
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo dec eps={eps:.1e} lam={lam:.2f} score={cur:.6f}")

    # geometric increasing (reverse of the above)
    for eps in eps_grid:
        if time.time() >= deadline - 250:
            break
        r = 1.0 - eps
        geo = r ** (n - 1 - idx)          # increasing shape
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo = geo / geo_sum
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = _project_simplex(seq)
            cur = _objective(seq)
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo inc eps={eps:.1e} lam={lam:.2f} score={cur:.6f}")

    # power‑law family – broaden the exponent range
    p_grid = np.linspace(-2.0, 6.0, 161)      # step ≈0.05
    for p in p_grid:
        if time.time() >= deadline - 250:
            break
        pow_seq = (idx + 1.0) ** p
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq = pow_seq / pow_sum
        cur = _objective(pow_seq)
        if cur < best_score:
            best_score = cur
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={cur:.6f}")

    # beta‑type  (i+1)^p * (n-i)^q   – explore a wider grid
    exp_vals = np.arange(0.0, 8.0, 0.5)      # up to 7.5
    for p in exp_vals:
        for q in exp_vals:
            if time.time() >= deadline - 250:
                break
            beta_seq = ((idx + 1.0) ** p) * ((n - idx) ** q)
            beta_sum = beta_seq.sum()
            if beta_sum == 0:
                continue
            beta_seq = beta_seq / beta_sum
            cur = _objective(beta_seq)
            if cur < best_score:
                best_score = cur
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={cur:.6f}")

    # random sampling of parametric families (lightweight)
    rng = np.random.default_rng(seed + 999)
    while time.time() < deadline - 250:
        shape_type = rng.choice(['geom_dec', 'geom_inc', 'power', 'beta'])
        if shape_type == 'geom_dec':
            r = rng.uniform(0.90, 0.9999)
            seq = r ** idx
        elif shape_type == 'geom_inc':
            r = rng.uniform(0.90, 0.9999)
            seq = r ** (n - 1 - idx)
        elif shape_type == 'power':
            p = rng.uniform(-2.0, 8.0)
            seq = (idx + 1.0) ** p
        else:   # beta
            p = rng.uniform(0.0, 8.0)
            q = rng.uniform(0.0, 8.0)
            seq = ((idx + 1.0) ** p) * ((n - idx) ** q)

        seq = _project_simplex(seq)
        cur = _objective(seq)
        if cur < best_score:
            best_score = cur
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Random {shape_type} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Multiple short Adam runs (bulk of the remaining budget)
    # ------------------------------------------------------------------
    # Reserve some time for the final hill‑climb (≈150 s)
    hill_reserve = 150.0
    remaining = deadline - time.time()
    adam_budget = max(0.0, remaining - hill_reserve)

    # number of independent Adam runs (at most 4, depending on remaining time)
    n_adam_runs = min(4, int(adam_budget // 60) + 1)   # aim for ≈60 s each
    if n_adam_runs < 1:
        n_adam_runs = 1
    time_per_run = adam_budget / n_adam_runs if n_adam_runs else 0.0

    for run in range(n_adam_runs):
        if time.time() >= deadline - hill_reserve:
            break
        # start from a jittered copy of the current best
        a0 = best_seq + 0.02 * rng.random(n)
        a0 = _project_simplex(np.maximum(a0, 0.0))

        run_deadline = time.time() + time_per_run
        a_opt, sc_opt = _adam_optimize(a0, start_time, run_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam run {run+1}/{n_adam_runs} improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb (with a tiny temperature)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb best score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
