"""
Propose a low‑objective non‑negative step function.

Algorithm
---------
1. **Warm‑starts** – a small library of analytic shapes (tent, cosine,
   half‑circle, Gaussian, power‑tent) and a few random Dirichlet vectors
   are generated. The best of them (according to the exact objective)
   seeds the optimisation.

2. **Phase A – Smooth Adam** – minimise a smooth approximation of the
   ``max`` of the self‑convolution using the log‑sum‑exp (LSE) trick.
   The temperature `α` of the LSE is increased linearly from a small
   value (≈ 5) to a larger one (≈ 80) so that the gradient starts
   smooth and gradually focuses on the true maximum.  Adam with Euclidean
   simplex projection is used.

3. **Phase B – Top‑k Adam** – the same Adam optimiser but with the
   sub‑gradient obtained from the average of the largest 0.5 % of the
   convolution entries (the “top‑k” method from the reference).  This
   refines the solution obtained in Phase A.

4. **Phase C – Simulated‑annealing polish** – a cheap Metropolis walk
   moves a random packet of mass between two indices.  The schedule is
   fast‑cooled and runs on the remaining time budget.

All steps respect the global time limit (≈ 1000 s).  The final vector is
projected onto the probability simplex, guaranteeing non‑negativity and
`∑a = 1`.  Because the evaluation function is homogeneous, this scaling
does not affect the objective value.

The function ``propose_candidate`` returns a plain ``list[float]`` ready
for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)
    return conv_full[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a normalised (∑a = 1) vector.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient computations
# ----------------------------------------------------------------------
def _topk_gradient(a: np.ndarray, top_k: int, L: int):
    """
    Gradient of the *average* of the `top_k` largest convolution values.
    Returns gradient and the convolution array.
    """
    n = a.shape[0]
    conv = _conv_self(a, L)

    if top_k >= conv.size:
        w = np.full(conv.shape, 1.0 / conv.size, dtype=float)
    else:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros_like(conv, dtype=float)
        w[top_idx] = 1.0 / top_k

    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad, conv


def _lse_gradient(a: np.ndarray, alpha: float, L: int):
    """
    Gradient of the log‑sum‑exp approximation to the max of the convolution.
    Returns the gradient and the convolution array.
    """
    n = a.shape[0]
    # 1) self‑convolution
    conv = _conv_self(a, L)

    # 2) soft‑max weights (stable)
    m = conv.max()
    # shift to keep exponentials ≤ 1
    exp_term = np.exp(alpha * (conv - m))
    sum_exp = exp_term.sum()
    if sum_exp == 0.0:               # numerical fallback (should not happen)
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = exp_term / sum_exp

    # 3) gradient via convolution w * a_rev
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser – generic driver (uses a gradient function)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   grad_fn,
                   grad_args: tuple,
                   lr: float = 0.025,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Generic projected Adam optimiser.
    `grad_fn` must have signature ``grad_fn(a, *grad_args)`` and return
    ``(grad, conv)`` where ``conv`` is the full self‑convolution array.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    # initial objective
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    best_score = float(2.0 * n * conv.max())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, conv = grad_fn(a, *grad_args)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # scale to the true objective (2·n·max)
        grad_obj = 2.0 * n * grad

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing polish (lightweight)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float):
    """
    Cheap Metropolis walk that moves a random packet of mass between two
    indices. Returns the best sequence seen and its objective.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # Number of proposals is roughly proportional to remaining time.
    # A single proposal costs an FFT ⇒ we keep it moderate.
    max_steps = 40000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # Simple exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)                 # source proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25      # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

def _make_analytic_starts(n: int):
    """Generate a few analytic shapes (tent, cosine, etc.)."""
    x = np.arange(n, dtype=float) / (n - 1)

    tent = np.minimum(x, 1.0 - x)
    cosine = np.sin(np.pi * x)                 # zero at ends, smooth in between
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normals.append(c / s)
    return normals


def _make_random_starts(n: int, rng, count=5):

        vec /= vec.sum()
