"""
Hybrid hill‑climbing / simulated‑annealing search for a low‑peak
autocorrelation step function.

The algorithm works with a probability vector p (Σp = 1, p_i ≥ 0).  
Score = 2·n·max(conv(p,p)).  Reducing the ℓ∞‑norm of the
self‑convolution directly reduces the score.

Main ideas
----------

* **Mass‑transfer moves** – repeatedly move a small amount ε of mass
  from a “source’’ index that contributes most to the current worst
  convolution shift to a “destination’’ that does not influence this
  shift (or has the smallest probability).  The convolution is
  updated analytically in O(n) time.

* **Simulated annealing** – occasionally accept moves that increase the
  peak, using a temperature that decays linearly.  This helps escape
  plateaus where pure hill‑climbing gets stuck.

* **Gradient‐descent rescue** – after a long period without improvement
  the algorithm computes the exact gradient of the currently maximal
  shift(s) and performs a few projected‑gradient steps (projected onto
  the probability simplex).  This often yields a tiny but decisive
  improvement.

* **Periodic re‑normalisation** – to avoid floating‑point drift the
  convolution is recomputed from scratch every few thousand steps.

The routine respects the 1000 s time budget, prints progress, and
returns the best probability vector found (as a plain list of floats).
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D arrays are supported.")
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_for_shift(p: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of conv[p,p][k] w.r.t. p.
    For k = 0 the gradient is 2·p.
    For k > 0 the gradient is
        g_i = p_{k-i} (if 0 ≤ k-i < n) + p_{i-k} (if 0 ≤ i-k < n).
    """
    n = p.shape[0]
    if k == 0:
        return 2.0 * p

    g = np.zeros_like(p)

    # first term: p_{k-i}
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    g[idx] += p[k - idx]

    # second term: p_{i-k}
    if k < n:
        g[k:] += p[: n - k]

    return g


def _combined_gradient(p: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Sum of gradients for a collection of shift indices.
    Used when several shifts are (almost) maximal.
    """
    grad = np.zeros_like(p)
    for k in shifts:
        grad += _gradient_for_shift(p, int(k))
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The algorithm uses a hybrid hill‑climbing /
    simulated‑annealing strategy with occasional projected‑gradient
    refinements.  It respects the given time budget and returns the best
    vector seen.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector
    # ------------------------------------------------------------------
    try:
        # Use the previously discovered construction if available
        p0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if p0.ndim != 1 or p0.size == 0 or np.all(p0 == 0):
            raise ValueError
        p0 = np.maximum(p0, 0.0)
        p = p0 / p0.sum()
        n = p.shape[0]
        print(f"[{time.time() - start_time:.2f}s] Using saved height_sequence_1 (n={n})")
    except Exception:
        # Fallback: arcsine‑type distribution, n = 4096
        n = 4096
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        alpha = 0.38
        eps = 1e-6
        w = (x * (1.0 - x) + eps) ** (-alpha)
        p = w / w.sum()
        print(f"[{time.time() - start_time:.2f}s] No starter – generated arcsine start (n={n})")

    # ------------------------------------------------------------------
    # 2) Initial convolution and bookkeeping
    # ------------------------------------------------------------------
    conv = _conv_self(p)
    cur_max = conv.max()
    best_max = cur_max
    best_score = 2.0 * n * cur_max        # Σp = 1 ⇒ score = 2·n·max(conv)
    best_p = p.copy()
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    rng = np.random.default_rng()
    iteration = 0
    last_improve_iter = 0

    # Temperature schedule (linear decay)
    T_initial = cur_max * 0.02   # modest initial temperature

    # ------------------------------------------------------------------
    # 3) Main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        remaining = deadline - time.time()

        # ---- locate the currently worst shift ---------------------------------
        k = int(conv.argmax())
        cur_max = conv.max()

        # ---- choose a source index: the pair (i, k‑i) with largest product -----    
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = p[idx] * p[k - idx]
        src = idx[int(np.argmax(contrib))]

        # ---- choose a destination that does NOT affect shift k if possible ----
        # indices with zero contribution to conv[k] are i > k (or i < k-(n-1))
        dst_candidates = np.where((np.arange(n) > k) |
                                  (np.arange(n) < k - (n - 1)))[0]

        if dst_candidates.size > 0:
            # favour small probabilities among the admissible set
            inv = 1.0 - p[dst_candidates]
            prob = inv / inv.sum()
            dst = rng.choice(dst_candidates, p=prob)
        else:
            # fall back to the original “light‑weight’’ rule
            inv = 1.0 - p
            prob = inv / inv.sum()
            dst = rng.choice(n, p=prob)

        if dst == src:
            continue

        # ---- step size --------------------------------------------------------
        # maximal fraction shrinks from 0.3 to 0.01 over the run
        max_frac = max(0.01, 0.30 * (1.0 - elapsed / budget_s))
        epsilon = p[src] * rng.random() * max_frac
        if epsilon <= 0.0:
            continue

        # ---- backup current state (cheap copies) --------------------------------
        p_old = p.copy()
        conv_old = conv.copy()
        max_old = cur_max

        # ---- fast convolution update (linear + quadratic terms) -----------------
        # remove contribution of the transferred mass
        conv[src:src + n] -= 2.0 * epsilon * p
        # add contribution of the incoming mass
        conv[dst:dst + n] += 2.0 * epsilon * p
        # quadratic ε² corrections
        conv[2 * src] += epsilon * epsilon
        conv[2 * dst] += epsilon * epsilon
        conv[src + dst] -= 2.0 * epsilon * epsilon

        # update the probability vector
        p[src] -= epsilon
        p[dst] += epsilon
        if p[src] < 0.0:
            p[src] = 0.0

        new_max = conv.max()

        # ---- acceptance test (simulated annealing) ----------------------------
        T = max(0.0, T_initial * (1.0 - elapsed / budget_s))
        accept = False
        if new_max < cur_max - 1e-16:
            accept = True
        else:
            # probabilistic uphill move
            if T > 0.0:
                prob_accept = math.exp(-(new_max - cur_max) / T)
                if rng.random() < prob_accept:
                    accept = True

        if accept:
            cur_max = new_max
            # store the overall best
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_score = 2.0 * n * new_max
                best_p = p.copy()
                last_improve_iter = iteration
                print(f"[{elapsed:.2f}s] iter={iteration:,}  new best score={best_score:.6f}")
        else:
            # reject – revert
            p = p_old
            conv = conv_old
            cur_max = max_old

        # ---- periodic re‑normalisation (drift control) ------------------------
        if iteration % 5000 == 0:
            s = p.sum()
            if abs(s - 1.0) > 1e-12:
                p /= s
                conv = _conv_self(p)
                cur_max = conv.max()

        # ---- plateau handling -------------------------------------------------
        if iteration - last_improve_iter > 200_000:
            # Identify all shifts that are (almost) maximal
            thr = best_max * (1.0 + 1e-9)    # tiny slack
            max_shifts = np.where(conv >= thr)[0]

            # Compute a combined gradient and try a few projected‑gradient steps
            grad = _combined_gradient(p, max_shifts)
            improved = False
            for step_frac in (0.02, 0.01, 0.005, 0.002):
                p_candidate = p - step_frac * grad
                p_candidate = _project_simplex(p_candidate)
                conv_candidate = _conv_self(p_candidate)
                cand_max = conv_candidate.max()
                if cand_max < best_max - 1e-13:
                    # accept this gradient improvement
                    p = p_candidate
                    conv = conv_candidate
                    cur_max = cand_max
                    best_max = cand_max
                    best_score = 2.0 * n * cand_max
                    best_p = p.copy()
                    last_improve_iter = iteration
                    print(f"[{elapsed:.2f}s] gradient step (α={step_frac}) → best score={best_score:.6f}")
                    improved = True
                    break
            if not improved:
                # No gradient gain – perform a small random jitter to escape
                jitter = 0.005 * p * (rng.random(p.shape) - 0.5)
                p = np.maximum(p + jitter, 0.0)
                p /= p.sum()
                conv = _conv_self(p)
                cur_max = conv.max()
                last_improve_iter = iteration
                print(f"[{elapsed:.2f}s] jittered after plateau (iter {iteration})")

    # ------------------------------------------------------------------
    # 4) Return the best vector (already a probability vector)
    # ------------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0:
        best_p = best_p / total
    else:
        best_p = np.full_like(best_p, 1.0 / n)

    # Convert to plain Python list of floats
    return [float(v) for v in best_p.tolist()]
