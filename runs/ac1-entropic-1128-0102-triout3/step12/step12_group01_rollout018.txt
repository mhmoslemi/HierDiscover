def _adam_softmax_optimize(a0, start_time, deadline, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    # pre-alloc L for conv: L1 for conv, L2 for correlation
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)  # Since correlation length may be up to 3n-2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1

    alpha = 5.0  # initial smoothing parameter
    alpha_inc_step = 5.0
    iterCount = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        iterCount += 1
        conv = _conv_self(a, L_conv)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = iterCount
            print(f"[{time.time() - start_time:.2f}s] SoftAdam it={iterCount:,} score={cur_score:.6f}")

        # compute gradient via softmax
        maxc = conv.max()
        # stable compute
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp

        # compute correlation gradient
        # Use FFT to compute gradient = 2 * correlation of w and a
        # Using real FFT product conj.
        # Compute FFTs with appropriate length L_corr
        # Ensure w and a zero padded to length L_corr
        Wf = np.fft.rfft(w, n=L_corr)
        Af = np.fft.rfft(a, n=L_corr)
        # correlation: ifft( Af * conj(Wf) )
        corr_full = np.fft.irfft(Af * np.conj(Wf), n=L_corr)

        start = n - 1
        grad = 2.0 * corr_full[start:start + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iterCount)
        v_hat = v / (1.0 - beta2 ** iterCount)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # adjust alpha occasionally
        if iterCount % 3000 == 0 and alpha < 50:
            alpha = min(alpha + alpha_inc_step, 50.0)

        # plateau handling: add noise if no improvement for many ops
        if iterCount - last_improve > 5000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = iterCount
    return best_a, best_score

    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = best_score_initial
    # FFT sizes
    L_self = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)
    # Adam hyperparams
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    last_improve = 0
    rng = np.random.default_rng()
    alpha = 5.0
    while time.time() < deadline:
        it += 1
        # compute conv
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] SoftAdam it={it:,} score={cur_score:.6f}")

        maxc = cur_max
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp

        # gradient: correlation of w and a
        # compute conv of w with reversed a
        a_rev = a[::-1]
        # convolution
        # compute FFT
        Wf = np.fft.rfft(w, n=L_corr)
        Af_rev = np.fft.rfft(a_rev, n=L_corr)
        conv_wa = np.fft.irfft(Wf * Af_rev, n=L_corr)[: (len(w) + len(a_rev) - 1)]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # adjust alpha gradually
        if it % 5000 == 0 and alpha < 40.0:
            alpha = min(alpha * 1.5, 40.0)

        # plateau handling
        if it - last_improve > 5000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

"""
A fast multi‑stage optimizer for the non‑negative step‑function problem.

The algorithm works with a longer sequence (n = 8192) which gives more
degrees of freedom than the previous 4096‑element runs.  It consists of:

1️⃣  A quick parametric sweep over several simple families (geometric,
    power, beta) and a few random Dirichlet starts – this provides a good
    initial point.

2️⃣  Several restarts of an Adam‑style projected optimiser, but instead of a
    hard max‑gradient we minimise a *smooth* approximation of the max
    (log‑sum‑exp) which yields a gradient that takes all convolution shifts
    into account.  This helps escape the plateaus that limited the earlier
    implementation.

3️⃣  A final stochastic hill‑climb that moves a tiny random fraction of mass
    between two positions and keeps only improving moves.

All heavy work (convolutions, correlations) is done with FFTs; the loop
structure checks the remaining time at each step and stops before the
1000 s budget expires.  The returned list is normalised to sum to 1 and
contains only non‑negative Python ``float`` objects ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv(v1: np.ndarray, v2: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    L = _next_pow2(v1.shape[0] + v2.shape[0] - 1)
    f1 = np.fft.rfft(v1, n=L)
    f2 = np.fft.rfft(v2, n=L)
    conv = np.fft.irfft(f1 * f2, n=L)[: v1.shape[0] + v2.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the *hard* max‑convolution (used for a fast baseline)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a))."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size > 0:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser using the *smooth* max (log‑sum‑exp) gradient
# ----------------------------------------------------------------------
def _adam_softmax_optimize(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smoothed version of
    2·n·max(conv(a,a)).  The smoothing uses a log‑sum‑exp (soft‑max)
    with a gradually increasing temperature.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    # FFT sizes
    L_self = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)        # for correlation w * a_rev

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    alpha = 5.0            # initial smoothing strength
    while time.time() < deadline:
        it += 1

        # current convolution and score (hard max)
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(
                f"[{time.time() - start_time:.2f}s] Soft‑Adam it={it:,} "
                f"score={cur_score:.6f}"
            )

        # ----- smooth gradient -------------------------------------------------
        maxc = cur_max
        exp_vals = np.exp(alpha * (conv - maxc))       # stable; maxc cancels overflow
        w = exp_vals / exp_vals.sum()                 # soft‑max weights, sum to 1

        # correlation w * a (linear), via convolution with reversed a
        a_rev = a[::-1]
        conv_wa = _conv(w, a_rev)                     # length = 2n‑1 + n‑1
        grad = 2.0 * conv_wa[n - 1 : n - 1 + n]        # slice of length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # slowly increase the smoothing (making the objective sharper)
        if it % 5000 == 0 and alpha < 40.0:
            alpha = min(alpha * 1.5, 40.0)

        # occasional random shake‑up if we are stuck
        if it - last_improve > 5000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser using the *hard* max‑gradient (kept as a fast fallback)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """Standard Adam optimiser on the simplex with sub‑gradient of max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                f"score={cur_score:.6f}"
            )

        grad = _subgradient_maxconv(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        if it - last_improve > 5000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (micro‑mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(
    start_a: np.ndarray,
    start_time: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Very cheap stochastic hill‑climb on the simplex."""
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a tiny random fraction (up to about 1 %)
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(
                f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine respects a 1000 s budget, runs a parametric sweep,
    several Adam‑style optimisations (both hard‑max and smoothed‑max
    gradients) with multiple restarts, and finishes with a stochastic
    hill‑climb.  The returned list is ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # problem size (larger n gives more degrees of freedom)
    # ------------------------------------------------------------------
    n = 8192
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = float(2.0 * n * _conv_self(best_seq, L_self).max())
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # incorporate any externally supplied starter
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                raise ValueError
            # adapt length: pad with zeros or truncate
            if starter.size < n:
                starter = np.concatenate([starter, np.zeros(n - starter.size)])
            elif starter.size > n:
                starter = starter[:n]
            starter = starter / starter.sum()
            cur = float(2.0 * n * _conv_self(starter, L_self).max())
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (geometric, power, beta families)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric family (coarse‑to‑fine grid)
    eps_grid = np.logspace(-5, -1, 80)          # r = 1 - eps
    lam_grid = np.linspace(0.0, 1.0, 11)       # mixture with uniform
    for eps in eps_grid:
        if time.time() >= deadline - 200:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo = geo / geo_sum
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            cur = float(2.0 * n * _conv_self(seq, L_self).max())
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} "
                    f"score={cur:.6f}"
                )

    # power family
    p_grid = np.linspace(-1.5, 2.5, 61)
    for p in p_grid:
        if time.time() >= deadline - 200:
            break
        pow_seq = np.power(idx + 1, p)
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq = pow_seq / pow_sum
        cur = float(2.0 * n * _conv_self(pow_seq, L_self).max())
        if cur < best_score:
            best_score = cur
            best_seq = pow_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={cur:.6f}"
            )

    # beta‑type family (two‑parameter polynomial shape)
    beta_vals = [0.0, 0.5, 1.0, 1.5, 2.0, 3.0]
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 200:
                break
            beta_seq = (np.power(idx + 1, p) * np.power(n - idx, q))
            beta_sum = beta_seq.sum()
            if beta_sum == 0:
                continue
            beta_seq = beta_seq / beta_sum
            cur = float(2.0 * n * _conv_self(beta_seq, L_self).max())
            if cur < best_score:
                best_score = cur
                best_seq = beta_seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Beta p={p} q={q} score={cur:.6f}"
                )

    # a few random Dirichlet‑like starts
    rng = np.random.default_rng()
    for i in range(10):
        if time.time() >= deadline - 200:
            break
        rand_seq = rng.exponential(scale=1.0, size=n)
        rand_seq = rand_seq / rand_seq.sum()
        cur = float(2.0 * n * _conv_self(rand_seq, L_self).max())
        if cur < best_score:
            best_score = cur
            best_seq = rand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Random seed {i} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Multi‑restart Adam optimisation (soft‑max gradient)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    hill_time = 80.0                           # reserve for the final hill‑climb
    if remaining > hill_time + 30.0:
        opt_deadline = deadline - hill_time
        # build a small pool of starts (best so far + a few random ones)
        candidates = [best_seq]
        for _ in range(3):
            cand = rng.exponential(scale=1.0, size=n)
            cand = cand / cand.sum()
            candidates.append(cand)

        # run a soft‑max Adam on each start, sharing the remaining budget
        for start_seq in candidates:
            if time.time() >= opt_deadline:
                break
            # allocate at most 200 s (or the remaining slice) to this run
            time_left = opt_deadline - time.time()
            slice_time = min(200.0, time_left / max(1, len(candidates)))
            slice_deadline = time.time() + slice_time
            a_opt, sc_opt = _adam_softmax_optimize(
                start_seq, start_time, slice_deadline, best_score
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Soft‑Adam improvement "
                    f"score={sc_opt:.6f}"
                )

        # one final short run of the *hard* max‑gradient Adam as a safety net
        if time.time() < opt_deadline:
            a_opt, sc_opt = _adam_optimize(
                best_seq, start_time, opt_deadline, best_score
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Hard‑Adam final improvement "
                    f"score={sc_opt:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Final hill‑climb best score={sc_hc:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
