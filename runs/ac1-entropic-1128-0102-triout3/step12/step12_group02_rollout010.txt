"""
Submission:  propose_candidate
Algorithm Overview
------------------
The objective 2·n·max(conv(a,a))/(sum(a)²) is homogeneous, so we may fix
∑a_i = 1 and minimise 2·n·max(conv(a,a)).  The search proceeds in four
stages within the 1000 s budget:

1️⃣  **Broad parametric sweep** – a rich family of monotone shapes
    (geometric × power, pure geometric, pure power) and mixtures with the
    uniform distribution are sampled on a fine grid.  The best candidate
    is kept.

2️⃣  **Random Dirichlet draws** – vectors drawn from Dirichlet(α) for a
    range of concentration parameters (α∈{0.01,…,10}) are sorted
    decreasingly (to respect the intuition that a decreasing shape is
    favourable) and evaluated.

3️⃣  **Exact sub‑gradient Adam descent** – when ∑a_i = 1 the denominator
    is constant, thus the gradient of the objective is
    `∇f = 2·n·∇max(conv)`.  The gradient of a single convolution entry
    `conv_k` is `2·a_{k−i}` (zero outside the support).  Adam is applied
    to this exact sub‑gradient with projection onto the probability simplex.

4️⃣  **Targeted hill‑climb** – using the current maximal‑lag `k*`,
    mass is moved from a source index whose partner `a_{k*−i}` is large
    to a destination whose partner is small.  The transferred amount
    starts large (≈10 % of the source) and decays over time, allowing
    rapid escape from plateaux followed by fine tuning.

All heavy work uses FFT‑based convolutions (O(n log n)).  The routine
keeps the best sequence found and returns it as a plain list of Python
floats (normalised to sum 1, which is already satisfied throughout the
search).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (∑a≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

# Sub‑gradient Adam descent (exact max‑conv gradient)
# ----------------------------------------------------------------------
def _adam_subgrad_descent(a0: np.ndarray, n: int, deadline: float,
                          start_time: float, best_score_initial: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on the exact sub‑gradient of 2·n·max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12
    t = 0
    L_fft = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        t += 1

        # current convolution and its maximal index
        conv = _conv_self(a, L_fft)
        k = int(conv.argmax())
        maxc = float(conv[k])
        cur_score = 2.0 * n * maxc  # denominator is 1 (∑a=1)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam‑subgrad it={t:,} "
                  f"score={cur_score:.6f}")

        # exact sub‑gradient of the max‑conv term
        grad = np.zeros_like(a)
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        grad[mask] = 2.0 * a[partner[mask]]
        grad = 2.0 * n * grad               # gradient of the full objective

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score

def _targeted_hill_climb(a0: np.ndarray, deadline: float,

    """Greedy mass transfer using the current max‑lag information."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    n = a.shape[0]
    rng = np.random.default_rng()
    # start with a relatively aggressive transfer fraction
    frac = 0.12

    while time.time() < deadline:

        mask = (partner >= 0) & (partner < n)
        partner_vals = np.zeros(n)
        partner_vals[mask] = a[partner[mask]]

        # source: index whose partner value is largest and that still holds mass

        dst = int(np.argmin(partner_vals))
