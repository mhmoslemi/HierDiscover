def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4096

    # Warm start shapes
    warm_starts = _make_warm_starts(n)

    # Add global sequence if available
    try:
        gs = np.asarray(height_sequence_1, dtype=float)
        if gs.ndim == 1 and gs.size>0:
            gs = np.maximum(gs,0.0)
            if gs.sum()>0:
                gs = gs/gs.sum()
                warm_starts.append(gs)
    except Exception:
        pass

    # Generate random seeds:
    rng = np.random.default_rng(seed+12345)
    rand_starts = []
    for _ in range(4): # 4 random
        r = rng.exponential(scale=1.0,size=n)
        r = r/r.sum()
        rand_starts.append(r)
    candidates = warm_starts + rand_starts

    best_a = None
    best_score = float('inf')

    # quick short Adam from each candidate
    for i, cand in enumerate(candidates):
        cur_time = time.time()
        if cur_time >= deadline:
            break
        # allow at most 5 seconds per candidate
        typ_deadline = cur_time + min(5.0, (deadline - cur_time) * 0.1)
        L = next_pow2(3*n)
        a_opt, score_opt = _adam_max_opt(cand, typ_deadline, start_time, lr=0.05)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Candidate {i} improved to {score_opt:.6f}")

    # If no best yet, initialize with uniform
    if best_a is None:
        best_a = np.full(n, 1.0/n)

    # Main Adam for most of remaining time
    remaining = deadline - time.time()
    if remaining > 10.0:
        # allocate 70% to Adam
        adam_deadline = time.time() + remaining * 0.7
        a_opt, score_opt = _adam_max_opt(best_a, adam_deadline, start_time, lr=0.05)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final Adam improved to {score_opt:.6f}")

    # Simulated annealing for remaining time
    if time.time() < deadline:
        a_opt, score_opt = _anneal_opt(best_a, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improved to {score_opt:.6f}")

    # Ensure non-negative and normalized
    best_a = np.maximum(best_a,0.0)
    total = best_a.sum()
    if total>0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0/n)

    return [float(v) for v in best_a]

    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, conv = _max_gradient(a, L)  # returns grad directed to reduce max (i.e., subgradient)
        # gradient for objective: d f = 2 n * grad_max? Wait we need to compute subgradient for f = 2 n max(conv)/sum^2.

        grad_obj = 2.0 * n * grad

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it} score={cur_score:.6f}")

    conv_wr = np.fft.irfft(Wf * Revf, n=L)
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad, conv

"""
Propose a low‑objective non‑negative step function.

The algorithm consists of three stages

1️⃣ Warm‑start – a few analytically simple shapes (tent, cosine, …)
   and the optional global solution `height_sequence_1` are normalised to
   a probability simplex and the best one (according to the exact
   objective) is kept.

2️⃣ Projected Adam on the *true* sub‑gradient of the maximum
   convolution entry.  The gradient is obtained in O(n log n) by an FFT
   based convolution of a sparse weight vector that marks the
   maximal convolution positions.  Adam’s adaptive step size together
   with an Euclidean projection onto the simplex keeps the iterate
   feasible.  Several random simplex starts are tried and the best
   result is refined for the bulk of the time budget.

3️⃣ Light simulated‑annealing polish – a Metropolis walk that moves a
   random packet of mass between two indices.  This helps to escape
   shallow local minima after the deterministic optimisation.

All heavy work uses NumPy’s real FFT which is very fast for powers of
two.  The default length `n = 4096` gives a good trade‑off between
expressiveness and speed, allowing hundreds of thousands of Adam
updates within the 1000 s budget.

The returned list satisfies the scoring function’s constraints
(non‑negative, finite, sum ≥ 0.01) and is ready to be passed to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (full convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    The vector `a` is assumed to be non‑negative.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:                     # safety – should never happen
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient of the *maximum* convolution entry
# ----------------------------------------------------------------------


def _max_gradient(a: np.ndarray, L: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns the sub‑gradient of the maximum entry of conv(a,a)
    together with the full convolution.
    The gradient w.r.t. a_i is 2·average_{t∈Argmax} a_{t‑i}.
    Computed via a single FFT‑based convolution (O(n log n)).
    """
    n = a.shape[0]
    conv = _conv_self(a, L)

    max_val = conv.max()
    # indices where the maximum is attained (could be several)
    max_idx = np.where(conv == max_val)[0]
    k = max_idx.size

    # weight vector w: uniform over the maximal indices
    w = np.zeros_like(conv)
    if k > 0:
        w[max_idx] = 1.0 / k

    a_rev = a[::-1]

    # convolution w * a_rev (full)
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)

    # extract the n entries that correspond to the gradient
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser using the exact max‑gradient
# ----------------------------------------------------------------------


def _adam_max_opt(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex using the sub‑gradient of the
    maximum convolution entry.
    Returns the best iterate encountered and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)  # sufficient size for all FFTs

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)  # 1st moment
    v = np.zeros_like(a)  # 2nd moment

    it = 0
    while time.time() < deadline:
        it += 1

        grad, conv = _max_gradient(a, L)

        # objective for bookkeeping (fast, no extra FFT)
        cur_score = float(2.0 * n * conv.max() / (a.sum() ** 2))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}"
            )

        # Adam update – the true objective is 2·n·max(conv) / sum².
        # Since we keep the vector on the simplex (sum=1) the factor
        # simplifies to 2·n·grad.
        grad_obj = 2.0 * n * grad

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection back onto the probability simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------


def _anneal_opt(seq: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk that moves a random packet of mass between two
    indices.  Used after the deterministic optimisation to escape shallow
    minima.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # total number of attempted moves – we stop early if time runs out
    total_steps = 200_000

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast) – start at 0.02, finish near 1e‑6
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        # pick a source index proportional to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Warm‑starting analytic shapes
# ----------------------------------------------------------------------


def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a small collection of simple analytic candidates."""
    x = np.arange(n, dtype=float) / (n - 1)  # normalised coordinate in [0,1]

    # 1) symmetric linear tent
    tent = np.minimum(x, 1.0 - x)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

    # 5) power‑tent (flatter than linear)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normalised.append(c / s)
    return normalised


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑objective step function within the time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid overshooting the deadline
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 1) Warm‑start – analytic shapes + optional global starter.
    # --------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # incorporate a global solution if the harness provides one
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                warm_starts.append(starter)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2) Random simplex seeds – increase diversity.
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    random_starts = []
    for _ in range(4):
        r = rng.exponential(scale=1.0, size=n)
        r = r / r.sum()
        random_starts.append(r)

    candidates = warm_starts + random_starts

    best_a = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Quick short Adam runs on each candidate (≈ 5 s each or less).
    # --------------------------------------------------------------
    for idx, cand in enumerate(candidates):
        now = time.time()
        if now >= deadline:
            break
        # allocate a tiny slice of the remaining budget
        short_deadline = now + min(5.0, (deadline - now) * 0.08)
        a_opt, sc_opt = _adam_max_opt(cand, short_deadline, start_time, lr=0.05)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Warm‑run {idx} improved to {sc_opt:.6f}")

    # Fallback to uniform if nothing succeeded
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 3) Main deterministic optimisation (major part of the budget).
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        # allocate ~70 % of the rest to Adam, the rest to annealing
        adam_deadline = time.time() + remaining * 0.70
        a_opt, sc_opt = _adam_max_opt(best_a, adam_deadline, start_time, lr=0.07)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Main Adam improved to {sc_opt:.6f}")

    # --------------------------------------------------------------
    # 4) Light annealing polish for the remaining seconds.
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _anneal_opt(best_a, deadline, start_time)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Annealing final improvement to {sc_opt:.6f}"
            )

    # --------------------------------------------------------------
    # 5) Final normalisation & conversion to plain list.
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    return [float(v) for v in best_a]
