assistantfinal**Solution Overview**

The task is to minimise  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

where `a*a` is the *full* (linear) self‑convolution of the non‑negative vector `a`.  
For a probability vector (`∑a_i = 1`) the denominator is constant, therefore we only have to make the
largest entry of the convolution as small as possible.

A good solution is a **probability vector that makes its autocorrelation almost flat** – the average of the
convolution is `1/(2n‑1)`, the theoretical lower bound, and the objective would be `≈1`.  
The optimisation is carried out in three stages:

1. **Warm‑starts** – a small library of analytically‑motivated shapes (uniform, tent,
   edge‑heavy power laws, Gaussians, etc.) together with a few random Dirichlet samples.
   The best of them (according to the exact objective) seeds the search.
2. **Smooth‑max gradient descent** –  
   * a smoothed approximation of the max,  
     \(\operatorname{softmax}_\tau (c)=\tau\log\sum_i e^{c_i/\tau}\)  (τ≈1 % of the current maximum),
   * its gradient can be written as a single convolution, which we evaluate with FFT
     (`O(n log n)`);
   * mirror‑descent on the probability simplex using an *exponential* update
     (`a ← a·exp(-η·∇)`), with a diminishing step‑size `η_t = η₀ / √t`.
   The loop runs until the time budget is exhausted; the best vector seen is kept.
3. **Final annealing** – a cheap Metropolis walk that moves a random packet of mass between two
   positions; it improves the result a little when a few seconds are left.

All heavy work is done with FFT‑based convolutions, so even for `n = 16384` the inner loop is fast
enough to perform many thousands of iterations in the 1000 s budget.

The code below follows exactly this plan and complies with the competition rules.

"""
Fast search for a low‑objective step‑function.

* warm‑starts: several analytically motivated shapes + random Dirichlet draws
* smooth‑max gradient descent on the probability simplex (exponentiated‑gradient /
  mirror‑descent) – the gradient of the soft‑max of the convolution is obtained
  by a single FFT‑based convolution
* light simulated‑annealing polish for the last few seconds

The implementation works with n = 16384 (a power of two, fast FFTs) and always
returns a plain Python list of non‑negative floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full, linear) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (used only for safety)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑start shape generators
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    """Central‑peaked shape: (triangle)^p."""
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape: (1‑triangle)^p."""
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # [0,1]
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random probability vector (Gamma‑(1,1) – Dirichlet)."""
    samp = rng.gamma(1.0, 1.0, size=n)
    return samp / samp.sum()


def _collect_warm_starts(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Assemble a modest library of diverse candidate vectors."""
    cand = []

    # deterministic shapes
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_power_center(n, 1.0))
    cand.append(_shape_power_edge(n, 1.0))

    # edge‑heavy powers (larger p)
    for p in (2.0, 4.0, 8.0, 12.0):
        cand.append(_shape_power_edge(n, p))

    # centre‑heavy powers (for completeness)
    for p in (0.5, 1.0, 2.0):
        cand.append(_shape_power_center(n, p))

    # exponentials
    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    # Gaussians
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # a few random Dirichlet vectors
    for _ in range(12):
        cand.append(_shape_random_dirichlet(n, rng))

    # optional global starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # sanitise
    uniq = []
    for a in cand:
        if not np.any(np.isnan(a)):
            uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (exponentiated‑gradient on simplex)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    eta0: float = 0.6,
                    tau_factor: float = 0.01,
                    min_tau: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex using the smooth (log‑sum‑exp)
    approximation of max(conv).  Returns the best vector seen and its exact
    objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)                     # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # sum(a)=1 throughout

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] descent it={it:,} "
                  f"score={cur_score:.6f}")

        # smooth max parameters
        tau = max(cur_max * tau_factor, min_tau)

        # numerically stable soft‑max weights
        max_c = cur_max
        exp_shifted = np.exp((conv - max_c) / tau)   # max term -> 1
        pw = exp_shifted / exp_shifted.sum()        # length 2n‑1, non‑negative

        # gradient of the smooth max: 2 * (pw  *  rev(a))  (convolution)
        rev_a = a[::-1]
        grad_full = _conv_fft(pw, rev_a)            # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # keep n entries

        # exponentiated‑gradient (mirror descent)
        step = eta0 / math.sqrt(it)
        a = a * np.exp(-step * grad)

        # renormalise to the simplex, protect against under‑flow
        a = np.maximum(a, 1e-15)
        a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Light annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Metropolis walk that swaps a random packet of mass between two indices."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of proposals is fixed; we stop early when time runs out
    total_steps = 60_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule – very quick cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # source proportional to current mass
        dst = rng.integers(n)             # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin so that returning does not hit the hard limit
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size (power of two → fast FFTs).  16384 offers more flexibility.
    # ------------------------------------------------------------------
    n = 16384

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Build a catalogue of warm‑start candidates.
    # ------------------------------------------------------------------
    candidates = _collect_warm_starts(n, rng)

    # ------------------------------------------------------------------
    # 2) Choose the best warm‑start according to the exact objective.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates, 1):
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] warm‑start {idx}/{len(candidates)} "
              f"score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # 3) Smooth‑max gradient descent – we allocate the bulk of the remaining budget.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                    # keep a few seconds for polishing
        sub_deadline = time.time() + remaining * 0.86
        refined_seq, refined_score = _smooth_descent(
            best_seq,
            sub_deadline,
            start_time,
            eta0=0.6,
            tau_factor=0.01
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light annealing polish (if any time left).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation & conversion to plain list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # The harness will run evaluate_sequence on the returned list.
    return [float(v) for v in best_seq]
