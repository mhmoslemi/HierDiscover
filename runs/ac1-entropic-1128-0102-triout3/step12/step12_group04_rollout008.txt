"""
Propose a low‑objective step‑function.

The algorithm is deliberately different from the reference:
  * it uses a *soft‑max* (log‑sum‑exp) surrogate for the max‑convolution
    and gradually sharpens the surrogate (continuation method);
  * the optimiser is Adam on the simplex (Euclidean projection);
  * warm starts consist of analytic shapes *and* a handful of sparse
    Dirichlet draws (low concentration) to explore spiky regimes;
  * after the main optimisation a cheap Metropolis annealing polish is run.

All heavy work is performed with real FFTs (numpy.rfft) on a power‑of‑two
size, which makes each iteration O(n log n) and allows thousands of
iterations within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns the full convolution (length 2·n‑1) as a real array.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)).
    The input is assumed to be non‑negative and normalised (∑a = 1).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform simplex point
        w = np.full_like(v, 1.0 / n)
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Given a weight vector w (same length as the full convolution),
    returns the gradient of Σ_k w_k·conv_k(a,a) with respect to a.
    The formula is   g = 2·(w * a_rev)   (central part of the convolution).
    """
    n = a.shape[0]
    a_rev = a[::-1]

    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)

    # the relevant slice is indices n‑1 … n‑1+n‑1
    grad = 2.0 * conv_wr[n - 1: n - 1 + n]
    return grad


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weight vector w_i = exp(alpha·(conv_i-max))/Z.
    Subtract the max for numerical stability.
    """
    maxc = conv.max()
    # shift before exponentiation
    shifted = alpha * (conv - maxc)
    # avoid overflow – values are now ≤ 0
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    if Z == 0.0:
        # fallback to uniform weights (should not happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / Z


# ----------------------------------------------------------------------
# Warm‑start generation
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Analytic symmetric shapes (tent, cosine, half‑circle, Gaussian, power‑tent)."""
    x = np.arange(n, dtype=float) / (n - 1)               # [0,1]

    tent = np.minimum(x, 1.0 - x)

    cosine = np.sin(np.pi * x)               # 0 at ends, positive interior

    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))

    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


def _random_sparse_starts(n: int, rng: np.random.Generator,
                         count: int = 5, concentration: float = 0.3) -> list[np.ndarray]:
    """
    Produce Dirichlet draws with a low concentration parameter
    (creates sparse, spiky vectors).  The draws are normalised to sum to 1.
    """
    draws = []
    for _ in range(count):
        a = rng.dirichlet(np.full(n, concentration))
        draws.append(a)
    return draws


def _resample_to_length(arr: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation to obtain an array of length target_len."""
    if arr.size == target_len:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, arr.size)
    x_new = np.linspace(0.0, 1.0, target_len)
    return np.interp(x_new, x_old, arr)


# ----------------------------------------------------------------------
# Adam optimiser using the soft‑max surrogate of max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   alpha_start: float = 0.5,
                   alpha_end: float = 8.0,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex using the gradient of a
    log‑sum‑exp surrogate of the maximum convolution entry.
    The smoothing parameter α is increased linearly from
    alpha_start to alpha_end as the allotted time elapses.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)                # safe size for all FFTs

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- compute convolution and soft‑max weights -----
        conv = _conv_self(a, L)

        # linear schedule for α
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        w = _softmax_weights(conv, alpha)

        # ----- gradient of (soft) max -----
        grad = _gradient_from_weights(a, w, L)          # gradient of surrogate

        # ----- Adam bookkeeping -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ----- enforce symmetry (optional but helps in practice) -----
        a = (a + a[::-1]) * 0.5

        # ----- projection onto the simplex -----
        a = _project_simplex(a)

        # ----- keep the best iterate seen -----
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score

# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(a0: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk that transfers a random packet of mass between
    two indices.  The temperature decays exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 80_000          # enough steps but still cheap
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.05 * (1e-4 / 0.05) ** (it / total_steps)

        src = rng.choice(n, p=a)            # source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.3   # move up to 30 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis criterion
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            hs = _resample_to_length(hs, n)
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0:
                candidates.append(hs / s)
    except Exception:
        pass

    # add a few sparse random starts
    rng = np.random.default_rng(seed + 12345)
    candidates.extend(_random_sparse_starts(n, rng, count=5, concentration=0.3))

    # pick the best of the warm starts (exact evaluation)
    best_a = None
    best_score = np.inf
    for a in candidates:
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    a0 = best_a
    print(f"[{time.time() - start:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Main optimisation – Adam on a log‑sum‑exp surrogate of max‑conv.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate the bulk of the remaining budget to Adam
        adam_deadline = time.time() + remaining * 0.80
        a_opt, score_opt = _adam_optimize(a0,
                                          adam_deadline,
                                          start,
                                          alpha_start=0.3,
                                          alpha_end=12.0,
                                          lr=0.025)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start:.2f}s] Adam final best score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Light simulated‑annealing polish (if any time left).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, anneal_score = _anneal(a0, deadline, start)
        if anneal_score < best_score:
            best_score = anneal_score
            a0 = a_final.copy()
            print(f"[{time.time() - start:.2f}s] Post‑anneal best score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list.
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    return [float(v) for v in a0]
```

This implementation follows the outlined strategy, exploits fast FFT‑based convolutions, employs a continuation‑style soft‑max surrogate for the max‑convolution, starts from both smooth analytic shapes and sparse random candidates, and finishes with a lightweight annealing polish. It respects the 1000 s budget and returns a plain list ready for `evaluate_sequence`.
