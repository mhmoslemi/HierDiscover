"""
Optimiser for the “step‑function” challenge.

The objective is
    2 * n * max(conv(a, a)) / (sum(a)**2)

which is scale‑invariant; we therefore normalise the vector to sum to 1
and minimise  2·n·max(conv(a,a)).  The optimiser works in four stages
inside the 1000 s budget:

1. **Broad parametric sweep** – geometric, power and beta families,
   the latter explored on a fine grid (α,β ∈ [0.05,4]).

2. **Random Dirichlet starts** – a handful of sparse / dense draws
   (Gamma concentration ∈ [0.1,5]) to diversify the search.

3. **Smooth‑max continuation** – minimise a differentiable proxy
   Sτ(a)= (1/τ)·log Σ exp(τ·conv) by gradient descent.
   τ is increased geometrically, turning Sτ from a smooth average
   into the true max.  Each step uses the exact gradient
   ∇Sτ = 2·(w * a_rev) on the active window (w = soft‑max
   weights).  After the update the vector is Euclidean‑projected
   onto the probability simplex.

4. **Directed hill‑climb** – repeatedly moves a tiny amount of mass
   from the index with the most positive sub‑gradient to the one with
   the most negative, accepting only improving moves.

All convolutions use FFT (length = next power‑of‑two) for speed,
and the simplex projection is the standard O(n) algorithm.
The routine respects the time budget and finally returns a plain list
of Python floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the non‑smooth max‑convolution (used for hill‑climb)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Objective helper (fast, scale‑invariant)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (geometric, power, beta)
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray,
                    best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    uniform = np.full(n, 1.0 / n)

    # ---- geometric ----------------------------------------------------
    eps_grid = np.logspace(-5, -1, 400)          # finer grid
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.0, 121)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type ------------------------------------------------------
    alphas = np.linspace(0.05, 4.0, 81)   # finer than the old version
    betas = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            # protect against overflow/underflow
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 30
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # random concentration between 0.1 and 5
        c = rng.uniform(0.1, 5.0)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smooth‑max continuation (gradient descent)
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                       start_time: float,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of max‑convolution.
    The temperature τ is increased geometrically; the step size decays
    as 1/√t.  The iterate is always projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    a_rev = a[::-1]                     # will be updated each iteration
    L_fft = _next_pow2(2 * n - 1)       # for self‑convolution
    L_grad = _next_pow2(3 * n - 2)      # for gradient convolution (w * a_rev)

    τ = 500.0                           # start temperature
    τ_factor = 1.25                      # geometric increase
    η0 = 0.04                           # base step size
    it = 0
    while time.time() < deadline:
        it += 1
        # self‑convolution and current score
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} score={cur_score:.6f}")

        # smooth weights (soft‑max over shifts)
        max_c = conv.max()
        # exponentials of (conv - max) are ≤1, safe from overflow
        w = np.exp(τ * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            # numerical degenerate, fall back to hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2 * (w * a_rev) restricted to length n
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)               # length = 3n-2
        grad = 2.0 * conv_w[n - 1: n - 1 + n]       # slice of length n

        # adaptive step (vanilla diminishing step)
        η = η0 / np.sqrt(it)
        step = η * grad

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)               # stay non‑negative
        a_new = _project_simplex(a_new)

        a = a_new

        # increase temperature every few thousand iterations
        if it % 4000 == 0:
            τ *= τ_factor
            # keep τ bounded to avoid extreme under‑flow
            τ = min(τ, 1e9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (mass transfer guided by sub‑gradient)
# ----------------------------------------------------------------------
def _hill_climb_directed(start_a: np.ndarray,
                         start_time: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Fine‑grained stochastic hill‑climb.
    At each iteration we move a tiny fraction of mass from the index with
    the largest positive sub‑gradient to the index with the most
    negative sub‑gradient.  Only improving moves are kept.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # choose source (mass to give away) and destination (mass to receive)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)             # most negative gradient

        if src == dst:
            # fallback to a random move if gradients are degenerate
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        # amount of mass to transfer (up to 0.5 % of source)
        frac = rng.random() * 0.005 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score
