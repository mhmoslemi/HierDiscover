    a0 = a0 / a0.sum()

    # Adam variables
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05   # base learning rate
    it = 0
    best_score = float('inf')
    best_seq = None

    top_k = 8

    while time.time() < deadline:
        it += 1

        a = _softmax(p)  # sum=1
        conv = _conv_self(a)
        maxval = conv.max()
        cur_score = 2.0 * n * maxval  # since sum=1

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} score={cur_score:.6f}")

        # choose top_k indices
        ks = np.argpartition(conv, -top_k)[-top_k:]

        gradA = np.zeros_like(a)
        for k in ks:
            # valid i such that 0<=i<n and 0<=k-i<n
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            inds = np.arange(lo, hi + 1)
            gradA[inds] += a[k - inds]

        gradA = (4.0 * n / len(ks)) * gradA

        # gradient wrt p (softmax)
        a_dot_grad = np.dot(a, gradA)
        gradP = a * (gradA - a_dot_grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * gradP
        v = beta2 * v + (1 - beta2) * (gradP ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)   # schedule
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps)

    K largest convolution entries (K = 8 by default).
* After the Adam phase a very cheap simulated‑annealing polishing step
  moves tiny packets of mass between random positions (Metropolis rule).
* The best vector seen during the whole run is returned as a plain
  Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Softmax utilities (keeps Σa = 1, a ≥ 0)
# ----------------------------------------------------------------------
def _softmax(p: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    shift = p - np.max(p)
    e = np.exp(shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser for the max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(n: int,
                   budget_s: float,
                   start_time: float,
                   top_k: int = 8,
                   lr0: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Run Adam on the softmax parameters.  Returns the best vector seen
    (as a probability distribution) and its score.
    """
    deadline = start_time + budget_s - 0.5  # safety margin

    # ---- start from a gently decreasing exponential shape ----
    r = 0.9995
    idx = np.arange(n, dtype=float)
    a0 = r ** idx
    a0 = a0 / a0.sum()
    p = np.log(a0 + 1e-30)

    m = np.zeros_like(p)
    v = np.zeros_like(p)

    best_a = a0.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        # current probability vector
        a = _softmax(p)

        # convolution and current score
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # Σa = 1 ⇒ denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # ---- gradient of max‑convolution (averaged over top_k entries) ----
        ks = np.argpartition(conv, -top_k)[-top_k:]          # indices of K biggest coeffs
        grad_a = np.zeros_like(a)

        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idxs = np.arange(lo, hi + 1, dtype=int)
            # contribution 2·a_{k-i} for each i in the valid range
            grad_a[idxs] += a[k - idxs]

        # full gradient of 2·n·max(conv) w.r.t. a
        grad_a = (4.0 * n / len(ks)) * grad_a

        # ---- transform gradient to the softmax parameters (Bregman/softmax trick) ----
        # ∂O/∂p = a ⊙ (∂O/∂a – ⟨a,∂O/∂a⟩)
        avg = np.dot(a, grad_a)
        grad_p = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)           # simple 1/√t decay
        p -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny packets of mass between random positions, Metropolis rule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = 2.0 * n * _conv_self(a).max()
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of attempts is limited by the remaining time budget
    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # fast exponential cooling
        T = 0.02 * ((1e-6) / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # pick a source proportionally to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move up to 25 % of the source mass
        eps_move = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        # keep non‑negativity (floating point noise may create a tiny negative)
        if a_prop[src] < -1e-15:
            continue

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search a non‑negative step‑function (probability vector) that yields a
    low value of evaluate_sequence().  Uses Adam optimisation on a
    soft‑max parametrisation followed by a light simulated‑annealing
    polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # reserve a half‑second margin

    # ------------------------------------------------------------------
    # 1) problem size – a power of two for fast FFTs
    # ------------------------------------------------------------------
    n = 4096  # 2¹¹ – enough resolution, cheap enough for many iterations

    # ------------------------------------------------------------------
    # 2) Adam optimisation
    # ------------------------------------------------------------------
    best_seq, best_score = _adam_optimize(
        n=n,
        budget_s=budget_s,
        start_time=start_time,
        top_k=8,
        lr0=0.05,
        beta1=0.9,
        beta2=0.999,
        eps=1e-8,
    )
    print(f"[{time.time() - start_time:.2f}s] End Adam, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Light annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polished_seq, polished_score = _anneal(best_seq,
                                               deadline,
                                               start_time)
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq
            print(f"[{time.time() - start_time:.2f}s] Polished score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
