"""
A smooth‑max gradient descent with simplex projection.

The evaluation function is scale‑invariant:
    score = 2·n·max(conv(a,a)) / (sum(a)²)

Hence we may (and do) keep the vector on the probability simplex
(sum = 1, a_i ≥ 0).  The algorithm minimises the peak of the
self‑convolution by a smoothed approximation of the max (log‑sum‑exp)
and uses a momentum update together with a decreasing step‑size.
After the descent a cheap Metropolis annealing phase is applied to
polish the result.

The code follows a multi‑stage strategy:
  1. generate a small pool of analytic and random starting shapes,
  2. run the smooth‑max descent on the most promising candidate,
  3. finish with a light simulated‑annealing polish,
  4. return a plain Python list for the harness.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility ---------------------------------------------------------------
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution helpers ---------------------------------------------------
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (fast for powers of two)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (scale invariant) -------------------------------------------
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Return the exact value of evaluate_sequence without the safety checks.
    The vector is assumed to be non‑negative.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:                     # should never happen
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Projection onto simplex ------------------------------------------------
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape families ------------------------------------------------
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape built from two power ramps that meet a plateau.
    Parameters
    ----------
    p     – exponent (>0)
    alpha – relative plateau width ∈[0,0.5] (0 → pure tent, 0.5 → uniform)
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                     # rises from 0…1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    total = shape.sum()
    return shape / total if total > 0 else np.full(n, 1.0 / n)


def _make_gaussian_shape(n: int, sigma_frac: float) -> np.ndarray:
    """
    Discrete Gaussian centred at the middle.
    sigma_frac = σ / n (relative width, 0 < σ_frac < 0.5)
    """
    sigma = max(1e-6, min(0.5, sigma_frac)) * n
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    total = shape.sum()
    return shape / total


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Beta‑distribution shape on a discrete grid.
    a,b > 0.  a = b gives a symmetric bell.
    """
    t = np.linspace(0.0, 1.0, n, endpoint=False) + 0.5 / n
    shape = (t ** (a - 1)) * ((1 - t) ** (b - 1))
    total = shape.sum()
    return shape / total if total > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent -----------------------------------------
# ----------------------------------------------------------------------
def _smoothmax_descent(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       lambda0: float = 12.0,
                       lambda_max: float = 200.0,
                       step0: float = 0.5,
                       beta: float = 0.92) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv(a,a)) using a smooth approximation
    (log‑sum‑exp) of the max.
    Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]
    L = 2 * n - 1                     # length of the linear convolution
    Lfft_conv = _next_pow2(L)         # FFT length for self‑convolution
    Lfft_big = _next_pow2(3 * n)      # FFT length for the weighted correlation

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    it = 0
    lam = lambda0

    # pre‑allocate space for reused arrays
    while time.time() < deadline:
        it += 1

        # --------- convolution a*a (needed for max and weights) ----------
        Af = np.fft.rfft(a, n=Lfft_conv)
        conv = np.fft.irfft(Af * Af, n=Lfft_conv)[:L]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # sum(a)=1 by projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Iter={it:,} score={cur_score:.6f}")

        # --------- smooth max weights (softmax) ---------------------------
        # subtract cur_max for numerical stability
        expw = np.exp(lam * (conv - cur_max))
        w = expw / expw.sum()

        # --------- gradient of the smooth max ---------------------------
        w_rev = w[::-1]

        # FFTs for the correlation a ⋆ w  (actually a * w_rev)
        Af_big = np.fft.rfft(a, n=Lfft_big)
        Wf_big = np.fft.rfft(w_rev, n=Lfft_big)
        corr_full = np.fft.irfft(Af_big * Wf_big, n=Lfft_big)[: n + L - 1]

        # extract the required slice: grad_j = 2 * corr_full[L-1-j]
        grad = 2.0 * corr_full[L - 1 - np.arange(n)]

        # --------- momentum + step ---------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # --------- simplex projection ------------------------------------
        a = _project_simplex(a)

        # --------- increase smoothing parameter ---------------------------
        lam = min(lambda_max, lam * 1.001)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light annealing polish ------------------------------------------------
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny mass moves with a Metropolis rule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # more mass → more likely to move
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30   # move up to 30 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
