"""
Propose a low‐score step function.

Algorithm
---------
* Initialise a non‑negative vector `p` (Σp = 1).  Use the previously
  discovered `height_sequence_1` if it exists, otherwise start from an
  arcsine‑type distribution.
* Compute its linear self‑convolution `conv = p * p` (FFT, O(n log n)).
* Repeatedly perform **deterministic mass‑transfer moves** that are
  guaranteed (or very likely) to lower the peak of the convolution:
      1.   Find the shift `k` where `conv` attains its maximum.
      2.   Among the indices that actually contribute to `conv[k]`,
           pick the source `src` with the largest product `p[i]·p[k‑i]`.
      3.   Choose a destination `dst` **outside** the contributing
           interval `[max(0,k‑(n‑1)), min(n‑1,k)]`.  For such a `dst`
           the contribution `p[k‑dst]` is zero, which gives the largest
           possible reduction of `conv[k]`.
      4.   Compute a step size `ε` that would bring `conv[k]` down close
           to the second‑largest value of the convolution (or a fraction
           of the source mass if that is too large).  The step size
           decays linearly from ~0.20 to 0.01 during the run.
      5.   Update `p` and `conv` **exactly** using the linear formulas
           derived from the quadratic nature of the convolution.
      6.   Accept the move only if the new maximal convolution entry does
           not increase (tiny tolerance allowed).  If the move is
           rejected, revert to the previous state.
* Every 5 000 iterations the convolution is recomputed from scratch with
  FFT to eliminate accumulated floating‑point drift and the vector is
  renormalised to Σp = 1.
* If no improvement occurs for a long stretch, the current best vector
  is jittered slightly to escape plateaus.
* The best vector seen so far (always normalised) is returned.  Because
  the evaluation function is scale‑invariant, a probability vector is a
  valid answer.

The method keeps the same cheap O(n) update cost as the previous
implementation but replaces the random destination choice by a
deterministic “outside‑support” destination and chooses the step size so
that the current peak is driven toward the second‑largest peak.  This
typically yields a much faster reduction of the objective and allows us
to push the score below the target 1.503 within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector onto the probability simplex (Σv = 1)."""
    # Simple renormalisation – vectors stay non‑negative by construction.
    s = v.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

# Main entry point required by the verifier
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a step‑function (list of non‑negative floats) that
    minimises `evaluate_sequence`.  Returns the best sequence found
    before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin before the hard limit
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1) Initialise probability vector p (Σp = 1)
    # --------------------------------------------------------------
    try:
        # If a previous best construction is available, use it.
        p0 = np.asarray(height_sequence_1, dtype=float)
        if p0.ndim != 1 or p0.size == 0:
            raise ValueError
        p0 = np.maximum(p0, 0.0)
        if p0.sum() == 0:
            raise ValueError
        p = p0 / p0.sum()
        n = p.shape[0]
        print(f"[{time.time() - start_time:.2f}s] Using height_sequence_1 (n={n})")
    except Exception:

        n = 4096

    while time.time() < deadline:

        # ---- locate current worst shift --------------------------------

        # ---- source index: largest contributor to conv[k] ---------------
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        contrib = p[idx] * p[k - idx]

            epsilon = min(p[src] * max_frac, eps_needed * 0.9)
            # avoid overly tiny moves

        else:

        epsilon = min(epsilon, p[src])
        if epsilon <= 0.0:

        p_old = p.copy()
        conv_old = conv.copy()

        # subtract contribution of the removed mass
        conv[src:src + n] -= 2.0 * epsilon * p

        conv[dst:dst + n] += 2.0 * epsilon * p
        # quadratic correction terms
        conv[2 * src] += epsilon * epsilon
        conv[2 * dst] += epsilon * epsilon
        conv[src + dst] -= 2.0 * epsilon * epsilon

        # update the probability vector
        p[src] -= epsilon
        p[dst] += epsilon
        # numerical safeguard
        if p[src] < 0.0:

        new_max = conv.max()

        if new_max < cur_max - 1e-16:

            if 2.0 * n * cur_max < best_score:

                best_p = p.copy()

                print(
                    f"[{time.time() - start_time:.2f}s] iter={iteration:,} "
                    f"new best score={best_score:.6f}"

            conv = conv_old

        if iteration % reeval_interval == 0:

            conv = _conv_self(p)
            cur_max = conv.max()  # reset current max after recompute

        if iteration - last_improve_iter > plateau_limit:

            cur_max = conv.max()
            last_improve_iter = iteration
            print(
                f"[{time.time() - start_time:.2f}s] plateau – re‑jittered at iter {iteration}"
            )

    # ------------------------------------------------------------------
    # 4) Return the best construction (already a probability vector)
    # ------------------------------------------------------------------
    best_p = _project_simplex(best_p)
    return [float(v) for v in best_p]
