"""
Propose a non‑negative step‑function (list of floats) that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

by

1️⃣  building a large, diverse catalogue of analytic warm‑starts
    (uniform, tent, arcsine, Beta(α,α) with many α,
    power‑law centre/edge, exponential, Gaussian, Hann/Hamming/Blackman
    windows and mixtures of the above);

2️⃣  picking the best warm‑start (exact objective) and refining it with a
    mirror‑descent on the probability simplex (exponentiated‑gradient
    with a momentum term).  The sub‑gradient is the average of the true
    gradients of all convolution entries whose value is within a
    decreasing relative tolerance of the current maximum.

3️⃣  polishing the refined vector with a cheap simulated‑annealing walk
    that repeatedly transfers a random packet of mass between two
    positions and keeps the move when it improves the objective
    (or very rarely accepts a worsening move).

The whole procedure runs for the allocated budget (default 1000 s) and
returns the best vector found, normalised to sum = 1.  All vectors contain
only non‑negative floats, satisfying the defensive checks of
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    # the evaluator protects against very tiny sums, but we keep the same guard
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on n points."""
    i = np.arange(n, dtype=float) + 0.5
    a = (i) ** (alpha - 1) * (n - i) ** (alpha - 1)
    a = np.maximum(a, 0.0)
    a += 1e-16
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # in [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # in [0,1]
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _shape_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _shape_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = (0.42
         - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _mix(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Return lam*a + (1‑lam)*b, re‑normalised to sum 1."""
    mixed = lam * a + (1.0 - lam) * b
    mixed = np.maximum(mixed, 0.0)
    s = mixed.sum()
    if s > 0:
        mixed /= s
    else:                     # degenerate – fall back to uniform
        mixed = np.full_like(mixed, 1.0 / mixed.size)
    return mixed

def _collect_warm_starts(n: int) -> list[np.ndarray]:
    cand = []

    # basic shapes
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))
    cand.append(_shape_hann(n))
    cand.append(_shape_hamming(n))
    cand.append(_shape_blackman(n))

    # Power‑law families
    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    # Exponential and Gaussian families
    for gamma in (0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # Beta(α,α) sweep (α>0.5 avoids the divergent L2‑norm)
    for alpha in np.linspace(0.55, 3.0, 26):
        cand.append(_shape_beta(n, alpha))

    # Mixtures of arcsine with uniform – useful knob for fine‑tuning
    arcs = _shape_arcsine_like(n)
    uni  = _shape_uniform(n)
    for lam in (0.0, 0.1, 0.25, 0.4, 0.55, 0.7, 0.85, 1.0):
        cand.append(_mix(arcs, uni, lam))

    # Random Dirichlet draws with various concentration parameters
    rng = np.random.default_rng()
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(5):
            a = rng.dirichlet(np.full(n, conc))
            cand.append(a)

    # Optional global starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # Remove NaNs / duplicates (rare)
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) on the simplex
# ----------------------------------------------------------------------
def _subgrad_exponential(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rel_tol_start: float = 0.02,
                         rel_tol_end: float = 0.001,
                         beta: float = 0.95,
                         eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the probability simplex.
    The sub‑gradient is the average of the true gradients of all
    convolution entries whose value lies within a (decreasing) relative
    tolerance of the current maximum.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max  # sum(a)=1 by construction

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # linearly decay the relative tolerance
        rel_tol = rel_tol_end + (rel_tol_start - rel_tol_end) * max(
            0.0, 1.0 - it / 2e5)   # shrink to rel_tol_end after ~200k iters
        tol = max(cur_max * rel_tol, 1e-15)
        top_idx = np.where(conv >= cur_max - tol)[0]
        if top_idx.size == 0:
            top_idx = np.array([conv.argmax()])

        # build averaged sub‑gradient
        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(top_idx.size)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = eta0 / math.sqrt(it)

        # exponential (mirror) update on the simplex
        a = a * np.exp(-step * momentum)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis walk that repeatedly moves a random packet
    of mass from a source index (chosen with probability proportional to a)
    to a random destination.  Improves the objective whenever possible.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # Slowly cooling schedule
    while time.time() < deadline:
        # temperature that decays roughly exponentially
        remaining = deadline - time.time()
        frac = remaining / (deadline - start_time)
        T = 0.02 * (0.0005 / 0.02) ** (1 - frac)   # from 0.02 down to 5e‑4

        src = rng.choice(n, p=a)          # source weighted by current mass
        dst = rng.integers(n)             # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass (kept small for stability)
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal improve score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    #   4096 balances resolution and speed.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) Gather a rich catalogue of warm‑start candidates.
    # ------------------------------------------------------------------
    candidates = _collect_warm_starts(n)

    # ------------------------------------------------------------------
    # 2) Pick the best warm‑start (exact objective evaluation).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates, 1):
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] Warm‑start {idx}/{len(candidates)} "
              f"score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # 3) Main optimisation – exponential‑gradient sub‑gradient descent.
    #    Use roughly 80 % of the remaining budget.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + remaining * 0.85
        refined_seq, refined_score = _subgrad_exponential(
            best_seq,
            sub_deadline,
            start_time,
            rel_tol_start=0.02,
            rel_tol_end=0.001,
            beta=0.95,
            eta0=0.25,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light annealing polish (if any time left).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation & conversion to a plain list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
