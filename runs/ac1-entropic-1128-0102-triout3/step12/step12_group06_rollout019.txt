def simplex_projection(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def projected_gradient_descent(a, deadline):
    best_a = a.copy()
    best_score = _exact_score(a)
    m = np.zeros(n); v = np.zeros(n)
    beta1=0.9; beta2=0.999; eta=0.1; eps=1e-8
    t=0
    while time.time() < deadline:
        t+=1
        # compute conv
        Af = np.fft.rfft(a, n=_L_SELF)
        conv = np.fft.irfft(Af * Af, n=_L_SELF)[:2*n-1]
        max_val = conv.max()
        ks = np.where(conv >= max_val - 1e-12)[0] # may be multiple indices
        # compute subgradient as sum of subgradients for each k
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k-(n-1))
            hi = min(k, n-1)
            idx = np.arange(lo, hi+1)
            grad[idx] += 2*a[k-idx]
        # Adam
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # project onto simplex
        a_new = simplex_projection(a_new)
        a = a_new
        # evaluate objective
        if t % 5000 == 0:
            score = _exact_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score

            return np.full_like(v, 1.0 / len(v))

        w = np.maximum(v - theta, 0.0)
        return w

"""
search_optimizer.py

A fast multi‑restart optimizer for the step‑function objective

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm works on the probability simplex (∑a = 1) and uses three
building blocks:

1.  Projected Adam (exponentiated‑gradient) on the exact max‑convolution.
    The sub‑gradient of the current maximal entry is computed, an Adam‑style
    update is applied and the iterate is projected back onto the simplex.

2.  A simulated‑annealing hill‑climber that repeatedly transfers a random
    fraction of mass between two positions.  The temperature decays
    linearly with the elapsed time.

3.  Multiple random restarts (Dirichlet draws with various concentration
    parameters) are interleaved until the time budget expires.  The best
    sequence seen during the whole run is returned.

All heavy work (convolution, gradients) is performed with FFTs, so each
iteration costs O(n log n) with n = 4096.  The code respects the 1000 s
budget and never performs forbidden I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    w = np.sort(v)[::-1]
    cssv = np.cumsum(w) - 1.0
    ind = np.arange(len(v)) + 1
    cond = w - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (fast via FFT)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray, L_self: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) with n = len(a)."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projected Adam optimisation on the true max‑convolution
# ----------------------------------------------------------------------
def _proj_adam(start_a: np.ndarray, deadline: float, L_self: int):
    """Adam on the exact max‑convolution, projected onto the simplex."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _exact_score(a, L_self)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1

        # convolution and maximal index(ices)
        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]
        max_val = conv.max()
        ks = np.where(conv >= max_val - 1e-12)[0]   # all indices tied for max

        # sub‑gradient of max(conv)
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # back to the simplex
        a = _simplex_projection(a)

        # occasional exact evaluation
        if t % 2000 == 0:
            cur = _exact_score(a, L_self)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                print(f"[{time.time():.2f}s] Adam iter={t:,} score={cur:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill climbing
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray, deadline: float, L_self: int, rng):
    """Small random mass transfers with a Metropolis acceptance rule."""
    cur = start_a.copy()
    cur_score = _exact_score(cur, L_self)

    best_seq = cur.copy()
    best_score = cur_score

    it = 0
    start_temp = 5e-2
    end_temp = 1e-12

    n = cur.shape[0]

    while time.time() < deadline:
        it += 1

        # linear temperature decay
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if cur[src] < 1e-15:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move up to 40 % of the source mass
        eps_move = cur[src] * rng.random() * 0.40
        if eps_move <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps_move
        cand[dst] += eps_move
        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            continue
        cand /= cand_sum

        cand_score = _exact_score(cand, L_self)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / max(T, 1e-16)) > rng.random():
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill it={it:,} best={best_score:.6f}")

    return best_seq, best_score

    Return a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` within the given time budget.
    The optimiser performs several random restarts, each consisting of a
    projected Adam phase followed by a simulated‑annealing hill‑climb.
    """
    global height_sequence_1  # may be defined by the evaluation harness

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3    # safety margin

    # --------------------------------------------------------------
    # Determine the problem size n (use the supplied sequence if present)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise Exception
        n = starter.size
        starter = np.maximum(starter, 0.0)
        if starter.sum() == 0.0:
            starter = np.full(n, 1.0 / n)
        else:
            starter = starter / starter.sum()
    except Exception:
        n = 4096
        starter = np.full(n, 1.0 / n)

    # pre‑computed FFT length (next power of two ≥ 2n‑1)
    L_self = 1 << ((2 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Keep the current overall best
    # ------------------------------------------------------------------
    best_seq = starter.copy()
    best_score = _exact_score(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to run a full optimisation from a given start vector
    # ------------------------------------------------------------------
    def _optimise_from(init_a):
        # allocate about 70 % of the remaining budget to Adam, 30 % to hill‑climb
        now = time.time()
        remaining = deadline - now
        if remaining < 5.0:
            return None, None
        adam_deadline = now + remaining * 0.65
        a_adam, sc_adam = _proj_adam(init_a, adam_deadline, L_self)

        # hill‑climb on the result returned by Adam
        if time.time() < deadline - 1.0:
            hc_deadline = deadline - 0.5
            a_hc, sc_hc = _hill_climb(a_adam, hc_deadline, L_self, rng)
        else:
            a_hc, sc_hc = a_adam, sc_adam

        # return the better of the two
        if sc_hc is not None and sc_hc < sc_adam:
            return a_hc, sc_hc
        else:
            return a_adam, sc_adam

    # ------------------------------------------------------------------
    # First try the known construction (if any)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq, sc = _optimise_from(starter)
        if sc is not None and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Improved from starter: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Random restarts – Dirichlet draws with varying concentration
    # ------------------------------------------------------------------
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]   # a few spreads
    while time.time() < deadline - 4.0:
        alpha = float(rng.choice(alphas))
        rand_vec = rng.gamma(alpha, 1.0, size=n)
        rand_vec = np.maximum(rand_vec, 0.0)
        if rand_vec.sum() == 0.0:
            continue
        rand_vec /= rand_vec.sum()
        seq, sc = _optimise_from(rand_vec)
        if sc is not None and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New best (α={alpha}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polish on the globally best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        final_seq, final_sc = _hill_climb(best_seq, deadline, L_self, rng)
        if final_sc is not None and final_sc < best_score:
            best_score = final_sc
            best_seq = final_seq
            print(f"[{time.time() - start_time:.2f}s] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more and convert to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
