"""
propose_candidate
=================

A two‑phase optimizer for the self‑convolution L∞‑objective

    F(a) = 2·n·max(conv(a,a)) / (∑a)²

(the evaluation function `evaluate_sequence` is scale‑invariant, therefore we
work with a normalized probability vector ∑a = 1).

1️⃣  **Warm‑start** – several analytic shapes (tent, cosine, half‑circle,
    Gaussian, power‑tent) and, if present, the globally known best sequence
    `height_sequence_1` are normalised and the best of them (according to the
    exact objective) is taken as the initial point.

2️⃣  **Soft‑max Adam** – we minimise a smooth upper‑bound of the max:
       Lα(a) = (1/α)·log(∑exp(α·conv(a,a))) .
    Its gradient is obtained analytically by a single FFT‑based convolution,
    which makes one Adam update O(n log n).  The smoothing parameter α is
    increased from a small value to a large one, so the optimisation gradually
    focuses on the true maximum.  After each step the iterate is projected onto
    the probability simplex.

3️⃣  **Simulated‑annealing polish** – a cheap Metropolis walk moves a random
    packet of mass between two indices.  The convolution of the candidate is
    recomputed with an FFT (still fast for the chosen size) and the move is
    accepted with the usual Metropolis probability.  The temperature decays
    exponentially and the number of moves is limited only by the remaining
    time budget.

The whole search respects the 1000 s limit and returns the best sequence
encountered as a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution (full length 2·n‑1) using a real FFT.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a normalised (∑a=1) vector.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑start shapes
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a short list of analytic candidate vectors."""
    x = np.arange(n, dtype=float) / (n - 1)               # coordinate in [0,1]

    # 1) symmetric linear tent
    tent = np.minimum(x, 1.0 - x)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))

    # 4) Gaussian centred in the middle
    sigma = 0.20
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

    # 5) power‑tent (flatter than linear)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]
    normalized = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        if c.sum() > 0.0:
            normalized.append(c / c.sum())
    return normalized


# ----------------------------------------------------------------------
# Soft‑max gradient of the convolution max (used in Adam)
# ----------------------------------------------------------------------
def _softmax_grad(a: np.ndarray, alpha: float, L: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Return (gradient, full convolution) of the smooth approximation

        Lα(a) = (1/α)·log Σₖ exp(α·convₖ)

    Gradient formula:
        ∂L/∂a = 2·conv(w, a_rev)
    where wₖ = exp(α·convₖ) / Σ exp(α·convₖ) .
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _conv_self(a, L)

    # 2) soft‑max weights (stable version)
    m = conv.max()
    scaled = alpha * (conv - m)          # values ≤ 0 → safe exponentials
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()

    # 3) gradient via convolution w * a_rev
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)

    # slice the n entries that correspond to ∂/∂a_i
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Projected Adam optimiser with soft‑max smoothing
# ----------------------------------------------------------------------
def _adam_softmax_optimize(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           alpha_start: float = 2.0,
                           alpha_end: float = 200.0,
                           ramp_iters: int = 2000,
                           lr: float = 0.04,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the probability simplex using the gradient of the
    soft‑max approximation of the max‑convolution.
    Returns the best iterate found and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)                # safe size for all FFTs

    best_a = a.copy()
    best_score = _objective(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # linearly increase the smoothing parameter α
        if it <= ramp_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * it / ramp_iters
        else:
            alpha = alpha_end

        # gradient of the smooth surrogate
        grad, conv = _softmax_grad(a, alpha, L)

        # exact objective for logging / best‑track
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"alpha={alpha:.1f} score={cur_score:.6f}")

        # Adam update (the true objective gradient would be 2·n·grad)
        grad_obj = 2.0 * n * grad

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the probability simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (full FFT recomputation)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that moves a random packet of mass between two indices.
    The convolution of each candidate is recomputed with an FFT (still fast
    for the moderate problem sizes we use).  The walk accepts uphill moves
    with a temperature that decays exponentially.
    """
    a = seq.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    total_steps = 120_000               # enough to fill the remaining budget

    # temperature schedule parameters
    T0 = 0.05
    Tf = 5e-4
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        T = T0 * (Tf / T0) ** (it / total_steps)

        # pick source proportional to current mass, destination uniformly
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # compute exact objective of the proposal
        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
