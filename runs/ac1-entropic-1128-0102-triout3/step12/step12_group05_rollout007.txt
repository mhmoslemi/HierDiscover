"""
Propose a low‑objective step‑function.

The search consists of three stages:

1️⃣  **Analytic warm‑starts** – a small library of hand‑crafted shapes
    (uniform, arcsine / Beta(½,½), a grid of symmetric Beta(α,α)
    distributions, a few asymmetric Beta’s and simple linear mixes).
    All candidates are normalised to sum 1 and the best according to the
    exact objective is kept.

2️⃣  **Long simulated‑annealing walk** – the current distribution is
    repeatedly perturbed by moving a tiny packet of mass from one index
    to another.  Temperature and step‑size decay linearly with the
    elapsed fraction of the time budget, allowing large exploratory
    moves early on and fine adjustments later.  The best distribution
    seen during the walk is retained.

3️⃣  **Final normalisation** – the best record is projected back onto the
    probability simplex (negative values are clipped to zero) and returned
    as a plain list of floats.

The whole routine respects the 1000 s budget (a short safety margin is
kept) and never uses filesystem/network I/O.  All helper functions are
top‑level to allow pickling if required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Identical to ``evaluate_sequence`` (without defensive checks).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)


def _shape_arcsine_like(n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine density:
        a_i ∝ 1 / sqrt( (i+½)*(n‑½‑i) )
    """
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _shape_beta(n: int, a: float, b: float) -> np.ndarray:
    """
    Symmetric or asymmetric Beta(α,β) law discretised on mid‑points.
    The points are x_i = (i+0.5)/n ∈ (0,1).
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    pdf += 1e-16                     # avoid exact zeros
    return pdf / pdf.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    """Central‑peaked shape (triangle)^p."""
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)    # scaled to [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape (1‑triangle)^p."""
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


# ----------------------------------------------------------------------
# Warm‑start catalogue
# ----------------------------------------------------------------------
def _collect_warm_starts(n: int) -> list[np.ndarray]:
    """
    Build a modest library of diverse initial vectors.
    All vectors are normalised to sum 1.
    """
    cand = []

    # elementary shapes
    cand.append(_shape_uniform(n))
    cand.append(_shape_arcsine_like(n))

    # grid of symmetric Beta(α,α)
    for a in np.linspace(0.2, 1.0, 17):       # 0.2 … 1.0
        cand.append(_shape_beta(n, a, a))

    # a few asymmetric Beta pairs (edge‑biased)
    for a, b in [(0.3, 0.7), (0.7, 0.3), (0.2, 0.8), (0.8, 0.2)]:
        cand.append(_shape_beta(n, a, b))

    # linear mixes of arcsine with uniform
    arcsine = _shape_arcsine_like(n)
    uniform = _shape_uniform(n)
    for w in np.linspace(0.0, 1.0, 11):
        mix = w * arcsine + (1.0 - w) * uniform
        mix /= mix.sum()
        cand.append(mix)

    # mixes of centre‑ and edge‑heavy power laws
    centre = _shape_power_center(n, 2.0)
    edge   = _shape_power_edge(n, 2.0)
    for w in np.linspace(0.0, 1.0, 11):
        mix = w * centre + (1.0 - w) * edge
        mix /= mix.sum()
        cand.append(mix)

    # optional starter supplied by the surrounding harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # remove any accidental NaNs
    uniq = []
    for a in cand:
        if not np.any(np.isnan(a)):
            uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Simulated annealing
# ----------------------------------------------------------------------
def _anneal(initial: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Long Metropolis walk that transports a tiny packet of mass between two
    indices.  Temperature and step‑size decay linearly with the elapsed
    fraction of the time budget.
    """
    a = initial.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    # initial temperature / max step‑size (as fraction of source mass)
    T0 = 0.006
    eps0 = 0.25
    T_min = 1e-8
    eps_min = 1e-4

    while time.time() < deadline:
        # linear decay based on elapsed fraction
        progress = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1.0 - progress) + T_min
        eps_scale = eps0 * (1.0 - progress) + eps_min

        # pick source proportional to its current mass, destination uniformly
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # amount of mass to move (up to eps_scale of the source)
        eps = a[src] * eps_scale * rng.random()
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal improve "
                      f"score={best_score:.6f}")

        # occasional re‑normalisation to fight round‑off drift
        if progress > 0.0 and int(progress * 1000) % 37 == 0:
            total = a.sum()
            if total > 0:
                a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    ``2·n·max(conv(a,a)) / (sum(a)²)``.
    The routine respects the 1000 s time budget, returns a plain Python
    list that can be fed directly to ``evaluate_sequence``.
    """
    # deterministic pseudo‑randomness
    np.random.seed(seed)

    start_time = time.time()
    # a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 1) Build a catalogue of warm‑starts and pick the best one.
    # --------------------------------------------------------------
    candidates = _collect_warm_starts(n)
    best_seq = None
    best_score = np.inf

    for idx, a in enumerate(candidates, 1):
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] Warm‑start {idx}/{len(candidates)} "
              f"score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # --------------------------------------------------------------
    # 2) Long simulated‑annealing polish (takes the bulk of the budget).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing "
              f"with initial score={best_score:.6f}")
        best_seq, best_score = _anneal(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] Annealing finished "
              f"best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Final sanitisation – clip negatives, renormalise, convert.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
