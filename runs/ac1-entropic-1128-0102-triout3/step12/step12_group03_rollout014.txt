"""
Propose a step‑function (non‑negative heights) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Initial pool** – generate a handful of analytic shapes:
   * power‑ramps (the “tent‑with‑plateau” used before),
   * Beta‑shapes  a_i ∝ x^{α‑1}(1‑x)^{α‑1}  (U‑shaped for α<1),
   * a few random Dirichlet vectors and, if present, the global starter.
   The best few candidates are kept.

2. **Heavy simulated‑annealing** – start from the current best vector and
   repeatedly move a small amount of mass from one index to another.
   Acceptance follows the Metropolis rule with an exponential cooling
   schedule.  The sum of the vector stays equal to 1, therefore the
   evaluation is cheap (a single FFT per move).  The best ever found is
   retained.

3. **Polishing (optional)** – a short Adam‑style smooth‑max descent is
   run for the remaining budget to squeeze a bit more performance.

All stages respect the 1000 s budget; the final list is normalised to sum 1
and contains only non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generator (power‑ramps with optional plateau)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric shape of length n, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Analytic Beta‑shape (U‑shaped for α<1)
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Return a symmetric Beta(α,α) density on {0,…,n‑1},
    normalised to sum 1.  For α<1 the shape is U‑shaped.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    # density ∝ x^{α‑1} (1‑x)^{α‑1}
    # evaluate in log‑space for stability
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Keep the K best candidates (simple linear list)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool (kept sorted)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max surrogate of max)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a smooth surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.05          # base learning rate
    T0 = 0.02             # initial soft‑max temperature
    T_end = 1e-4          # final temperature

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} "
                  f"score={cur_score:.6f}")

        # temperature schedule (linear decay)
        frac = it / (it + 1000.0)        # smooth freeze
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation (w * a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)        # length for convolution w * a_rev
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]  # length 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]    # slice of length n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger N → finer discretisation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # ------------------------------------------------------------------
    # 1) Build an initial pool of candidates
    # ------------------------------------------------------------------
    TOP_K = 5
    top = []   # list of (score, vector)

    # uniform vector (baseline)
    uniform = np.full(N, 1.0 / N)
    _push_top(top, _objective(uniform), uniform, TOP_K)

    # analytic power‑ramp shapes
    p_grid = np.linspace(0.2, 2.0, 9)          # 0.2,0.4,…,2.0
    alpha_grid = np.linspace(0.0, 0.5, 6)      # 0.0,0.1,…,0.5
    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_shape(N, p, alpha)
            _push_top(top, _objective(a), a, TOP_K)

    # Beta‑shapes (U‑shaped for α<1)
    for alpha in np.linspace(0.1, 0.9, 9):
        a = _make_beta_shape(N, alpha)
        _push_top(top, _objective(a), a, TOP_K)

    # random Dirichlet vectors (shape<1 gives sparsity)
    rng = np.random.default_rng(seed + 12345)
    for _ in range(30):
        if time.time() >= deadline:
            break
        a = rng.gamma(0.3, 1.0, size=N)   # Gamma(0.3) ~ sparse Dirichlet
        s = a.sum()
        if s == 0:
            continue
        a /= s
        _push_top(top, _objective(a), a, TOP_K)

    # incorporate the global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _push_top(top, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    # current best
    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Heavy simulated‑annealing refinement
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    best_global_seq = best_seq.copy()
    best_global_score = best_score

    max_move_factor = 0.2          # up to 20 % of source mass per move
    T_start = 0.5
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1

        # temperature schedule (exponential cooling)
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * (T_end / T_start) ** frac

        # pick a source index proportionally to its current weight
        src = rng.choice(N, p=cur_seq)
        if cur_seq[src] < 1e-15:
            continue
        dst = rng.integers(N)
        if src == dst:
            continue

        # propose moving a small chunk of mass
        delta = cur_seq[src] * max_move_factor * rng.random()
        if delta <= 0.0:
            continue
        a_prop = cur_seq.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        delta_score = new_score - cur_score

        if delta_score < 0.0 or rng.random() < math.exp(-delta_score / T):
            cur_seq = a_prop
            cur_score = new_score
            if new_score < best_global_score:
                best_global_score = new_score
                best_global_seq = cur_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} "
                      f"new best={new_score:.6f}")

        # occasional restart from a random top‑candidate (helps escape traps)
        if it % 250000 == 0 and len(top) > 1:
            choice = rng.integers(len(top))
            cur_seq = top[choice][1].copy()
            cur_score = _objective(cur_seq)

        # occasional tiny Gaussian perturbation (helps exploration when T is tiny)
        if it % 100000 == 0:
            sigma = 0.005
            noise = rng.normal(0.0, sigma, size=N)
            a_noisy = cur_seq + noise
            a_noisy = np.maximum(a_noisy, 0.0)
            s = a_noisy.sum()
            if s > 0:
                a_noisy /= s
                cur_seq = a_noisy
                cur_score = _objective(cur_seq)

    # ------------------------------------------------------------------
    # 3) Optional smooth‑max polishing (if any time left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + remaining
        refined_seq, refined_score = _smooth_gradient_descent(best_global_seq,
                                                             sub_deadline,
                                                             start_time)
        if refined_score < best_global_score:
            best_global_score = refined_score
            best_global_seq = refined_seq
            print(f"[{time.time() - start_time:.2f}s] After smooth GD best = {best_global_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_global_seq = np.maximum(best_global_seq, 0.0)
    total = best_global_seq.sum()
    if total > 0:
        best_global_seq /= total
    else:
        best_global_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_global_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_global_seq]
