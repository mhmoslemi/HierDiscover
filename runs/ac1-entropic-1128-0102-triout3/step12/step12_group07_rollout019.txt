"""
Deterministic safe mass‑transfer hill climbing.
Starts from an arcsine‑type probability vector and repeatedly
moves mass from the index that contributes most to the current
worst autocorrelation shift to an index that contributes the least.
The step size ε is computed analytically as the largest
transfer that does not raise any other shift above the current
maximum, guaranteeing monotone improvement.
When progress stalls, a small random jitter is applied.
The procedure runs until the time budget expires and returns the
best sequence encountered.
"""

import time, math, numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_self(p: np.ndarray, L: int) -> np.ndarray:
    # linear self‑convolution using FFT of length L (>=2*n-1)
    A = np.fft.rfft(p, n=L)
    c = np.fft.irfft(A * A, n=L)
    return c[: 2 * p.shape[0] - 1]

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Sequence length – large enough to give a good constant, still fast.
    n = 4096
    conv_len = 2 * n - 1
    L = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 1) Build an arcsine‑type start vector – we also try a few exponents.
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_p = None
    alphas = np.linspace(0.30, 0.45, 16)   # sweep around the known good value
    eps = 1e-6
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    for a in alphas:
        w = (x * (1.0 - x) + eps) ** (-a)
        p = w / w.sum()
        conv = _conv_self(p, L)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_p = p.copy()
            best_alpha = a
    print(f"[{time.time() - start_time:.2f}s] start α={best_alpha:.3f}, score={best_score:.6f}")

    p = best_p.copy()
    conv = _conv_self(p, L)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_score = cur_score
    best_p = p.copy()
    last_improve_iter = 0
    iteration = 0

    rng = np.random.default_rng(seed + 12345)

    # auxiliary arrays used each iteration (re‑used to avoid allocations)
    inc = np.empty(conv_len, dtype=np.float64)
    dec = np.empty(conv_len, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # locate the worst shift
        k = int(conv.argmax())
        cur_max = conv.max()

        # ----- source (largest contributor to shift k) -----
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = p[idx] * p[k - idx]
        src = int(idx[int(np.argmax(contrib))])

        # ----- destination (smallest contribution to shift k) -----
        rev_idx = k - np.arange(n)
        mask = (rev_idx >= 0) & (rev_idx < n)
        dst_contrib = np.empty_like(p)
        dst_contrib.fill(0.0)
        dst_contrib[mask] = p[mask] * p[rev_idx[mask]]
        # avoid picking the source itself
        dst_contrib[src] = np.inf
        dst = int(np.argmin(dst_contrib))

        # ----- compute the largest safe mass transfer ε -----
        # linear effect of decreasing p[src] by ε on every shift
        dec.fill(0.0)
        inc.fill(0.0)

        # dec[ℓ] = 2 * p[ℓ‑src] for ℓ ∈ [src, src+n‑1]; handle the central term correctly
        dec[src:src + n] = 2.0 * p
        if 0 <= 2 * src < conv_len:
            dec[2 * src] = p[src]      # only one product when i=j=src

        # inc[ℓ] = 2 * p[ℓ‑dst] for ℓ ∈ [dst, dst+n‑1]
        inc[dst:dst + n] = 2.0 * p
        if 0 <= 2 * dst < conv_len:
            inc[2 * dst] = p[dst]

        diff = inc - dec                     # net change per unit ε
        # where diff > 0 the shift would increase – keep ε small enough
        mask_pos = diff > 1e-15
        if np.any(mask_pos):
            # bound = (cur_max - conv) / diff   for those shifts
            bound = (cur_max - conv)
            # avoid division by zero already handled by mask
            bound = bound[mask_pos] / diff[mask_pos]
            # numerical safety – negative bounds are set to 0
            bound = np.where(bound < 0, 0.0, bound)
            epsilon_max = min(p[src], bound.min())
        else:
            epsilon_max = p[src]

        # No safe move – apply a tiny jitter and continue
        if epsilon_max <= 0.0:
            # jitter: add a very small random perturbation (≤0.1 % of each entry)
            jitter = 0.001 * p * (rng.random(p.shape) - 0.5)
            p = np.maximum(p + jitter, 0.0)
            p /= p.sum()
            conv = _conv_self(p, L)
            cur_max = conv.max()
            continue

        # use the full safe amount (slightly shrunk to stay strictly below the bound)
        epsilon = epsilon_max * 0.999

        # ----- apply the mass transfer -----
        p[src] -= epsilon
        p[dst] += epsilon

        # renormalise occasionally to combat drift
        if iteration % 1000 == 0:
            total = p.sum()
            if abs(total - 1.0) > 1e-12:
                p /= total

        # ----- evaluate the new state -----
        conv = _conv_self(p, L)
        new_max = conv.max()
        new_score = 2.0 * n * new_max

        # monotone improvement is guaranteed, but we double‑check
        if new_max <= cur_max + 1e-15:
            cur_max = new_max
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_p = p.copy()
                last_improve_iter = iteration
                print(f"[{time.time() - start_time:.2f}s] iter={iteration:,} best={best_score:.6f}")
        else:
            # revert – should rarely happen
            p[src] += epsilon
            p[dst] -= epsilon

        # ----- plateau detection -----
        if iteration - last_improve_iter > 200_000:
            # jitter around the best vector to escape a flat region
            jitter = 0.005 * best_p * (rng.random(best_p.shape) - 0.5)
            p = np.maximum(best_p + jitter, 0.0)
            p /= p.sum()
            conv = _conv_self(p, L)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            last_improve_iter = iteration
            print(f"[{time.time() - start_time:.2f}s] plateau – re‑jittered at iter {iteration}")

    # ------------------------------------------------------------------
    # 3) Return the best vector seen
    # ------------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0:
        best_p = best_p / total
    else:
        best_p = np.full(n, 1.0 / n)

    # The verifier is scale‑invariant; a probability vector is fine.
    return [float(v) for v in best_p]

"""
Deterministic safe‑mass‑transfer hill climbing.

The algorithm starts from an arcsine‑type probability vector and repeatedly
moves probability mass from the index that contributes most to the current
worst autocorrelation shift to an index that contributes the least.
The transfer amount ε is computed analytically as the largest amount that
guarantees no other shift exceeds the present maximum – this makes every
accepted move monotonically improve the objective.
If progress stalls a tiny random jitter is applied to escape plateaus.
The loop runs until the time budget expires (≈1000 s) and the best vector
seen is returned.  The returned list is a non‑negative step sequence; the
verifier treats it as scale‑invariant, so a probability vector (Σ=1) is
sufficient.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two and linear convolution via FFT
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(p: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution p * p (length 2·len(p)‑1) using FFT.
    L must be a power of two ≥ 2·len(p)‑1.
    """
    A = np.fft.rfft(p, n=L)
    c = np.fft.irfft(A * A, n=L)
    return c[: 2 * p.shape[0] - 1]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score within the given time budget.  The function returns
    the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin before the hard deadline
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1) Fixed length and FFT size
    # --------------------------------------------------------------
    n = 4096                          # length of the sequence
    conv_len = 2 * n - 1
    L = _next_pow2(conv_len)          # FFT length (power of two)

    # --------------------------------------------------------------
    # 2) Initialise with a short sweep over arcsine‑type exponents
    # --------------------------------------------------------------
    best_score = float("inf")
    best_p = None
    best_alpha = None
    eps = 1e-6
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    alphas = np.linspace(0.30, 0.45, 16)   # sweep around the known good value
    for a in alphas:
        w = (x * (1.0 - x) + eps) ** (-a)
        p = w / w.sum()
        conv = _conv_self(p, L)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_p = p.copy()
            best_alpha = a
    print(f"[{time.time() - start_time:.2f}s] start α={best_alpha:.3f}, "
          f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Core optimisation loop (safe mass‑transfer moves)
    # --------------------------------------------------------------
    p = best_p.copy()
    conv = _conv_self(p, L)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_score = cur_score
    best_p = p.copy()
    last_improve_iter = 0
    iteration = 0

    rng = np.random.default_rng(seed + 12345)

    # reusable auxiliary arrays (avoid per‑iteration allocations)
    inc = np.empty(conv_len, dtype=np.float64)
    dec = np.empty(conv_len, dtype=np.float64)

    while time.time() < deadline:
        iteration += 1

        # ---- identify the current worst shift ----
        k = int(conv.argmax())
        cur_max = conv.max()

        # ---- source: index that contributes most to shift k ----
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib_src = p[idx] * p[k - idx]
        src = int(idx[int(np.argmax(contrib_src))])

        # ---- destination: index with smallest contribution to shift k ----
        rev_idx = k - np.arange(n)
        mask = (rev_idx >= 0) & (rev_idx < n)
        dst_contrib = np.empty_like(p)
        dst_contrib.fill(0.0)
        dst_contrib[mask] = p[mask] * p[rev_idx[mask]]
        dst_contrib[src] = np.inf                     # avoid picking the source
        dst = int(np.argmin(dst_contrib))

        # ---- compute linear impact of moving ε from src to dst on every shift ----
        dec.fill(0.0)
        inc.fill(0.0)

        # decreasing p[src] by ε reduces each shift ℓ∈[src, src+n‑1] by 2·p[ℓ‑src]
        dec[src:src + n] = 2.0 * p
        if 0 <= 2 * src < conv_len:
            dec[2 * src] = p[src]                     # the central term has only one product

        # increasing p[dst] by ε adds 2·p[ℓ‑dst] to each shift ℓ∈[dst, dst+n‑1]
        inc[dst:dst + n] = 2.0 * p
        if 0 <= 2 * dst < conv_len:
            inc[2 * dst] = p[dst]

        diff = inc - dec                               # net change per unit ε

        # ---- largest ε that does not raise any shift above cur_max ----
        mask_pos = diff > 1e-15                         # shifts that would increase
        if np.any(mask_pos):
            cur_max_minus = cur_max - conv
            bounds = cur_max_minus[mask_pos] / diff[mask_pos]
            bounds = np.where(bounds < 0.0, 0.0, bounds)   # safety
            epsilon_max = min(p[src], bounds.min())
        else:
            epsilon_max = p[src]

        if epsilon_max <= 0.0:
            # No safe move – apply a tiny random jitter and continue
            jitter = 0.001 * p * (rng.random(p.shape) - 0.5)
            p = np.maximum(p + jitter, 0.0)
            p /= p.sum()
            conv = _conv_self(p, L)
            continue

        # use almost the whole safe amount (slightly reduced to stay strictly below)
        epsilon = epsilon_max * 0.999

        # ---- perform the transfer ----
        p[src] -= epsilon
        p[dst] += epsilon

        # occasional renormalisation to counteract floating‑point drift
        if iteration % 1000 == 0:
            total = p.sum()
            if abs(total - 1.0) > 1e-12:
                p /= total

        # ---- evaluate the new state ----
        conv = _conv_self(p, L)
        new_max = conv.max()
        new_score = 2.0 * n * new_max

        # monotone improvement is guaranteed – double‑check for safety
        if new_max <= cur_max + 1e-15:
            cur_max = new_max
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_p = p.copy()
                last_improve_iter = iteration
                print(f"[{time.time() - start_time:.2f}s] iter={iteration:,} "
                      f"best={best_score:.6f}")
                if best_score <= 1.503:
                    print("[*] Target reached – exiting early.")
                    break
        else:
            # Extremely unlikely, but revert just in case
            p[src] += epsilon
            p[dst] -= epsilon
            conv = _conv_self(p, L)
            cur_max = conv.max()

        # ---- plateau detection – inject a small random jitter ----
        if iteration - last_improve_iter > 200_000:
            jitter = 0.005 * best_p * (rng.random(best_p.shape) - 0.5)
            p = np.maximum(best_p + jitter, 0.0)
            p /= p.sum()
            conv = _conv_self(p, L)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            last_improve_iter = iteration
            print(f"[{time.time() - start_time:.2f}s] plateau – jittered at iter {iteration}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence found (normalised to sum = 1)
    # ------------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0:
        best_p = best_p / total
    else:
        best_p = np.full(n, 1.0 / n)

    # The verifier is scale‑invariant; a probability vector is acceptable.
    return [float(v) for v in best_p]
