def project_onto_simplex(v):
    """Project a vector onto the probability simplex (sum=1, >=0)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero or negative, return uniform distribution
        return np.full(n, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑score step function.

Algorithm
---------
1. Initialise a probability vector p (Σp = 1).  If a previously
   discovered construction `height_sequence_1` exists we normalise it,
   otherwise we start from a short sweep over a family
   `p_i ∝ (x_i·(1−x_i)+ε)^{−α}` (arcsine‑type) and keep the best.
2. Keep the linear self‑convolution c = p∗p (length 2·n−1) updated
   exactly after each move (O(n) slice operations, no FFT per step).
3. Repeatedly perform *directed mass‑transfer* moves:

   * Let k be the shift where c is maximal.
   * Among the indices that contribute to c[k] pick a source `src`
     maximising `p[i]·p[k−i]` (largest contribution to the worst shift).
   * For every possible destination i compute the partner value
     `q_i = p[k−i]` (zero if the partner lies outside the support of
     the shift).  Moving mass to i will increase the worst shift by
     `2·ε·q_i`.  Choose a destination `dst` among the indices with the
     smallest partner value (deterministic, no random sampling).  A
     small set of the best candidates is examined and the one that
     yields the largest reduction of the maximal convolution entry is
     taken.
   * Transfer a fraction `ε` of the mass from `src` to `dst`
     (ε = p[src]·rand·max_frac, `max_frac` shrinks from 0.2 to 0.01
     during the run).  The convolution is updated by the analytical
     O(n) formula used in the reference implementation.
   * Accept the move only if the new maximal convolution value is
     strictly smaller; otherwise revert.

4. Periodically renormalise the vector and recompute the convolution
   from scratch to avoid accumulated floating‑point drift.
5. If no improvement has been seen for a long stretch, jitter the
   current best solution (≤0.5 % of each entry) and continue.
6. The best vector seen during the whole run is returned.  The verifier
   is scale‑invariant, therefore a normalised probability vector is a
   valid answer.

The method keeps the cheap O(n) update of the original hill‑climber
while selecting a destination that provably causes the smallest rise
in the dominant autocorrelation peak, leading to faster and deeper
improvements.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _test_move(p: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float):
    """
    Return (new_max, new_conv) obtained by moving `eps` from src to dst.
    The original arrays are left untouched.
    """
    n = p.shape[0]
    conv_new = conv.copy()
    # linear part of the update
    conv_new[src:src + n] -= 2.0 * eps * p
    conv_new[dst:dst + n] += 2.0 * eps * p
    # quadratic correction terms
    conv_new[2 * src] += eps * eps
    conv_new[2 * dst] += eps * eps
    conv_new[src + dst] -= 2.0 * eps * eps
    return conv_new.max(), conv_new


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | Σ x = 1, x_i ≥ 0}.
    """
    if v.ndim != 1:
        raise ValueError
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) whose
    evaluation score is as low as possible, respecting a time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a tiny safety margin before the hard deadline.
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1) Initial probability vector (Σp = 1)
    # --------------------------------------------------------------
    p0 = globals().get("height_sequence_1")
    if isinstance(p0, (list, tuple, np.ndarray)):
        try:
            p = np.asarray(p0, dtype=float)
            if p.ndim != 1 or p.size == 0:
                raise ValueError
            p = np.maximum(p, 0.0)
            if p.sum() == 0.0:
                raise ValueError
            p = p / p.sum()
            n = p.shape[0]
            print(f"[{time.time() - start_time:.2f}s] Using height_sequence_1 (n={n})")
        except Exception:
            p = None
    else:
        p = None

    if p is None:
        # fallback – try a few arcsine‑type exponents and keep the best.
        n = 4096
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        eps = 1e-6
        alphas = [0.33, 0.35, 0.37, 0.38, 0.40]
        best_score = np.inf
        best_p = None
        for alpha in alphas:
            w = (x * (1.0 - x) + eps) ** (-alpha)
            p_tmp = w / w.sum()
            conv_tmp = _conv_self(p_tmp)
            score_tmp = 2.0 * n * conv_tmp.max()
            if score_tmp < best_score:
                best_score = score_tmp
                best_p = p_tmp
        p = best_p
        print(f"[{time.time() - start_time:.2f}s] Generated arcsine start (n={n})")

    # --------------------------------------------------------------
    # 2) Initial convolution and bookkeeping
    # --------------------------------------------------------------
    conv = _conv_self(p)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max   # Σp = 1  →  score = 2·n·max(conv)
    best_p = p.copy()
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    rng = np.random.default_rng()
    iteration = 0
    last_improve_iter = 0
    K_candidates = 5                     # number of destination candidates examined

    # --------------------------------------------------------------
    # 3) Main optimisation loop (directed mass‑transfer)
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # locate the current worst shift
        k = int(conv.argmax())
        cur_max = conv.max()

        # indices that influence conv[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # source: largest product p[i]·p[k−i]
        contrib = p[idx] * p[k - idx]
        src = idx[int(np.argmax(contrib))]

        # partner values for every possible destination
        indices = np.arange(n, dtype=int)
        partner = k - indices
        valid = (partner >= 0) & (partner < n)
        partner_val = np.empty(n, dtype=float)
        partner_val[valid] = p[partner[valid]]
        partner_val[~valid] = 0.0
        # exclude the source index from consideration
        partner_val[src] = np.inf

        # pick a few destinations with the smallest partner values
        if K_candidates < n:
            cand_idxs = np.argpartition(partner_val, K_candidates)[:K_candidates]
        else:
            cand_idxs = np.arange(n)
        # ensure we really have candidates (unlikely to be empty)
        if cand_idxs.size == 0:
            continue

        # step size – a decreasing fraction of the source mass
        elapsed = time.time() - start_time
        max_frac = max(0.01, 0.20 * (1.0 - elapsed / budget_s))
        epsilon = p[src] * rng.random() * max_frac
        if epsilon <= 0.0:
            continue

        # backup current state (cheap copies – O(n))
        p_old = p.copy()
        conv_old = conv.copy()

        # Try the primary candidate; if it fails, try the remaining ones.
        move_accepted = False
        for dst in cand_idxs:
            if dst == src:
                continue
            new_max, conv_test = _test_move(p, conv, src, dst, epsilon)
            if new_max < cur_max - 1e-16:
                # Accept this move.
                conv = conv_test
                p[src] -= epsilon
                p[dst] += epsilon
                # protect against tiny negatives from rounding
                if p[src] < 0.0:
                    p[src] = 0.0
                cur_max = new_max
                move_accepted = True
                break

        if not move_accepted:
            # revert to the previous state
            p = p_old
            conv = conv_old

        # record best solution so far
        if 2.0 * n * cur_max < best_score:
            best_score = 2.0 * n * cur_max
            best_p = p.copy()
            last_improve_iter = iteration
            print(
                f"[{time.time() - start_time:.2f}s] iter={iteration:,} "
                f"new best score={best_score:.6f}"
            )

        # --------------------------------------------------
        # Periodic re‑normalisation and full convolution recompute
        # --------------------------------------------------
        if iteration % 5000 == 0:
            s = p.sum()
            if abs(s - 1.0) > 1e-12:
                p /= s
                conv = _conv_self(p)

        # --------------------------------------------------
        # Plateau handling – small random jitter around the best
        # --------------------------------------------------
        if iteration - last_improve_iter > 200_000:
            p = best_p.copy()
            jitter = 0.005 * p * (rng.random(p.shape) - 0.5)   # ≤0.5 %
            p = np.maximum(p + jitter, 0.0)
            total = p.sum()
            if total > 0.0:
                p /= total
            else:
                p = np.full(n, 1.0 / n)
            conv = _conv_self(p)
            last_improve_iter = iteration
            print(
                f"[{time.time() - start_time:.2f}s] plateau – re‑jittered at iter {iteration}"
            )

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0.0:
        best_p = best_p / total
    else:
        best_p = np.full(n, 1.0 / n)

    # The verifier only cares about the relative shape (scale‑invariant).
    return [float(v) for v in best_p]
