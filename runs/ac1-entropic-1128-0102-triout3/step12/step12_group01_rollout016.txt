def generate_shape(params, n, knot_positions):
    # params length K: y_i = log10 height
    heights = 10**params
    # ensure at least positive
    # linear interpolation
    idx = np.arange(n)
    a = np.interp(idx, knot_positions, heights)
    # Clip negative (should not be negative)
    a = np.maximum(a, 0)
    # Normalize sum
    total = a.sum()
    if total <= 0:
        return np.ones(n)/n
    a = a / total
    return a
```

def compute_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2*n-2).bit_length())
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    maxb = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * maxb / (sum_a * sum_a))
```

"""
Propose a step function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases (all respecting a 1000 s budget)

1️⃣  **Low‑dimensional parametric search** – a cheap global optimiser
    (SciPy's differential evolution) works on a piecewise‑linear shape
    described by a handful (≈12) knot heights.  The heights are optimised
    in log‑scale, the resulting curve is interpolated to the full length
    `n` and normalised to unit sum.  This discovers very diverse shapes,
    in particular “U‑shaped’’ distributions that are difficult to obtain
    with the simple geometric / power families used before.

2️⃣  **Projected Adam optimisation** – the best shape from step 1 is fed
    to an Adam‑style optimiser that works on the probability simplex.
    A sub‑gradient of ``max(conv)`` is obtained by averaging the gradients
    of all active shifts.  A decreasing learning‑rate schedule together
    with occasional Gaussian perturbations helps to push the objective
    further down.

3️⃣  **Stochastic hill‑climb** – the final refinement repeatedly transfers
    a tiny random fraction of mass between two indices and keeps the
    move only if the true objective improves.  The hill‑climb runs for the
    last part of the budget and typically squeezes the last few percent
    off the score.

All heavy work (convolutions) is performed with FFTs; the implementation
uses only a single CPU core and stays well within the 1000 s limit.
The returned list is plain Python floats, ready for ``evaluate_sequence``.
"""
import time
import math
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT‑based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {w | w ≥ 0, sum(w)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution term
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv(a,a)) at the current point.
    If several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size > 0:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex.
    Returns the best sequence encountered and its (approximate) objective
    2·n·max(conv)  (the sum of a is kept ≈1 by projection).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters (schedule with simple decay)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.3

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        # linear decay of learning rate
        lr = lr0 * max(0.1, 1.0 - (time.time() - start_time) / (deadline - start_time))

        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)  # sum(a)≈1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto simplex
        a = _project_simplex(a)

        # plateau handling – add a tiny log‑normal perturbation
        if it - last_improve > 3000:
            noise = 5e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a tiny random fraction of mass
    between two positions and keeps only improving moves.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # random fraction of the source mass, up to 0.5 %
        frac = rng.random() * 0.005 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to compute the exact objective (identical to evaluate_sequence
# except for the security‑checks that are unnecessary here)
# ----------------------------------------------------------------------
def _compute_score(seq: np.ndarray) -> float:
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Low‑dimensional global optimisation with differential evolution
# ----------------------------------------------------------------------
def _de_optimize(n: int, de_deadline: float, rng_seed: int) -> tuple[np.ndarray, float]:
    """
    Optimise a piecewise‑linear shape on the simplex using DE.
    Returns the best sequence and its score.
    """
    # 12 knots give enough flexibility while keeping the search cheap
    num_knots = 12
    knot_pos = np.linspace(0, n - 1, num_knots)

    # search in log10‑space; heights may vary over many orders
    log_bounds = [(-10.0, 2.0) for _ in range(num_knots)]

    def objective(log_heights):
        # Decode heights, build full sequence by linear interpolation
        heights = 10.0 ** np.array(log_heights)
        seq = np.interp(np.arange(n), knot_pos, heights)
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total <= 0.0:
            return np.inf
        seq /= total
        return _compute_score(seq)

    # tiny callback to respect the overall deadline
    def callback(x, convergence):
        if time.time() > de_deadline:
            return True
        return False

    result = differential_evolution(
        objective,
        bounds=log_bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=15,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=rng_seed,
        callback=callback,
        polish=False,
        disp=False,
    )

    # Re‑build the best sequence
    best_heights = 10.0 ** result.x
    best_seq = np.interp(np.arange(n), knot_pos, best_heights)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    best_score = _compute_score(best_seq)
    return best_seq, best_score

            seq_de, score_de = _de_optimize(n, de_deadline, seed)
            if score_de < best_score:
                best_score = score_de
                best_seq = seq_de.copy()
                print(f"[{time.time() - start_time:.2f}s] DE improvement score={score_de:.6f}")
        except Exception as e:
            print(f"[{time.time() - start_time:.2f}s] DE failed: {e}")

    # ------------------------------------------------------------------
    # 2️⃣ Adam‑style projected descent (bulk of the remaining budget)
    # ------------------------------------------------------------------
    hill_time = 150.0                     # seconds reserved for final hill‑climb
    adam_deadline = overall_deadline - hill_time
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        a_opt, sc_opt = _adam_optimize(best_seq, start_time, adam_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final stochastic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < overall_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, overall_deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
