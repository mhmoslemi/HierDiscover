def subgradient_descent(a0, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * len(a) * conv.max() / (a.sum()**2)
    # Get n
    n = a.shape[0]
    t = 0
    eta0 = 0.1  # maybe step size
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # indices where conv is close (abs < 1e-12 relative)
        K = np.where(conv >= max_val - 1e-12)[0]
        grad = np.zeros_like(a)
        # compute gradient contributions
        for k in K:
            if k == 0:
                grad += 2 * a
            else:
                # forward part
                grad[: n - k] += a[k:]
                # backward part
                grad[k:] += a[: n - k]
        # subgradient descent step
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad
        # project onto simplex
        a = _project_simplex(a)
        # evaluate objective
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={t} score={cur_score:.6f}")
        # optional random perturbation occasionally
        if t % 500 == 0:
            # small random jitter
            eps = 1e-4 * np.random.rand(n)
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_a, best_score

"""
Propose a step‑function (list of non‑negative numbers) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three parts:

1️⃣  Several cheap initialisations (uniform, a global starter, a few
    random Dirichlet draws).  The best of these seeds is remembered.

2️⃣  **Exact sub‑gradient descent** on the true objective.
    The objective depends on the *maximum* entry of the self‑convolution.
    For a fixed index k the convolution term is
        c_k(a) = Σ_i a_i a_{i+k}.
    Its gradient is
        ∂c_k/∂a_i = a_{i+k} (if i+k<n) + a_{i‑k} (if i‑k≥0).
    By moving a a little opposite to the sum of those gradients over all
    currently maximal indices we directly reduce the true max‑conv term.
    After each step the vector is projected onto the probability simplex
    (non‑negative and sum = 1).  A diminishing step size η₀/√t gives a
    stable descent while the time budget is respected.

3️⃣  A final **hill‑climb** that repeatedly transfers a tiny amount of mass
    between two random coordinates, keeping only improving moves.
    The transfer size starts at 0.5 % of the source mass and is slowly
    reduced, allowing a fine‑grained polishing of the best candidate.

All heavy work uses FFT‑based convolution (O(n log n)).  The routine stays
within the supplied time budget and returns the best sequence it discovered
as a plain Python list of floats.
"""

# ----------------------------------------------------------------------
# Standard imports
# ----------------------------------------------------------------------
import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·len(a)‑1) using FFT.
    The padding length L must be a power of two ≥ 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv_full


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Returns a vector w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast (exact) objective, assumes sum ≈ 1
# ----------------------------------------------------------------------
def _score_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact value of 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * a.shape[0] * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_score: float,
                         best_vec: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Perform a simple sub‑gradient descent on the exact max‑conv term.
    The iterate is kept on the simplex; the best vector seen so far is
    returned together with its score.
    """
    a = a0.copy()
    n = a.shape[0]
    # step‑size schedule parameters
    eta0 = 0.1
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        # convolution and current maximum
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        # indices where the maximum is attained (usually just one)
        K = np.where(conv >= cur_max - 1e-15)[0]

        # ----- gradient of the *max* term -----
        grad = np.zeros_like(a)
        for k in K:
            if k == 0:
                # derivative of Σ_i a_i^2
                grad += 2.0 * a
            else:
                # forward contribution  a_{i+k}
                grad[: n - k] += a[k:]
                # backward contribution a_{i‑k}
                grad[k:] += a[: n - k]

        # ----- sub‑gradient step (moving opposite to the gradient) -----
        eta = eta0 / np.sqrt(t)          # diminishing step
        a = a - eta * grad
        a = _project_simplex(a)

        # ----- keep the best point -----
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={t:,} "
                  f"score={cur_score:.6f}")

        # occasional tiny random jitter to break plateaus
        if t % 500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_vec, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (microscopic mass transfers)
# ----------------------------------------------------------------------
def _final_hill_climb(a_start: np.ndarray,
                      L_fft: int,
                      deadline: float,
                      start_time: float,
                      best_score: float,
                      best_vec: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Randomly move a small fraction of mass between two coordinates.
    Only moves that improve the exact objective are kept.
    """
    a = a_start.copy()
    n = a.shape[0]
    rng = np.random.default_rng()

    # start with a relatively generous transfer size and shrink it slowly
    eps_factor = 5e-3   # 0.5 % of source mass at the beginning

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * eps_factor
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_vec = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

        # slowly tighten the transfer magnitude
        eps_factor = max(eps_factor * 0.9995, 1e-4)

    return best_vec, best_score
