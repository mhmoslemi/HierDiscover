def smooth_max(conv, epsilon):
    max_val = conv.max()
    exps = np.exp((conv - max_val)/epsilon)
    sum_exps = exps.sum()

            best_obj = obj
            best_seq = a.copy()

    z = np.log(best_seq + eps)
    # Adam params
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.05
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    # epsilon schedule
    epsilon0 = 0.1
    epsilon_min = 0.001
    # Keep best across iterations
    best_a = best_seq.copy()
    best_score = best_obj
    # optimize
    while time.time() < deadline - 2.0:   # reserve time for anneal
        t += 1
        # compute epsilon

        m_hat = m / (1.0 - beta1 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = z - step
        # enforce symmetry
        z = 0.5 * (z + z[::-1])

            # compute a again from z

"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

using a smooth‑max surrogate and Adam optimisation on the
log‑probabilities (soft‑max parametrisation).  The objective
max(conv) is approximated by ε·log∑exp(conv/ε); its gradient can be
computed analytically via two FFT‑based convolutions, which makes each
iteration O(n log n).  Starting from a library of analytically‑motivated
shapes (uniform, tent, arcsine, power‑law, exponential, Gaussian) we
pick the best as a warm‑start, run many Adam steps with a decreasing
ε (coarse → fine smoothing) and keep the best exact objective seen.
A cheap Metropolis‑style annealing polish is applied in the final seconds.
The whole search respects the 1000 s budget and returns a plain list of
non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors via FFT.
    Returns a vector of length len(x)+len(y)-1.
    """
    n = x.shape[0]
    m = y.shape[0]
    L = _next_pow2(n + m - 1)
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)[: n + m - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)          # full self‑convolution
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Shape library (warm‑starts)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)   # [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)   # [0,1]
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _collect_warm_starts(n: int) -> list[np.ndarray]:
    """Collect a modest library of diverse initial vectors."""
    cand = []

    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # mixed centre/edge blends
    for lc in (0.25, 0.5, 0.75):
        centre = _shape_power_center(n, 2.0)
        edge = _shape_power_edge(n, 2.0)
        a = lc * centre + (1 - lc) * edge
        a = a / a.sum()
        cand.append(a)

    # optional global starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # sanitise
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (on logits)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(z: np.ndarray, epsilon: float, n: int):
    """
    Returns
        smooth_obj : ε·log‑sum‑exp of the convolution (smooth max)
        grad_z     : gradient w.r.t. the logits z (soft‑max parametrisation)
        a          : current probability vector (soft‑max of z)
        max_conv   : the true max(conv(a,a)) (used for bookkeeping)
    """
    # soft‑max to obtain a (guaranteed non‑negative, sums to 1)
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / exp_z.sum()

    # full self‑convolution a*a
    conv = _conv_fft(a, a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth max weights
    max_c = max_conv
    # avoid overflow/underflow
    exp_c = np.exp((conv - max_c) / epsilon)
    w = exp_c / exp_c.sum()          # w_i ≥ 0, Σ w_i = 1

    # gradient of max(conv) w.r.t a :
    #   ∇_a max ≈ Σ_i w_i ∇_a conv_i = 2 * (w_rev * a)  (restricted to 0…n‑1)
    w_rev = w[::-1]
    grad_full = _conv_fft(w_rev, a)          # length 3·n‑2
    Lw = len(w)                              # = 2·n‑1
    idx = (Lw - 1) - np.arange(n)            # indices giving the needed slice
    grad_a = 2.0 * grad_full[idx]

    # chain rule through the soft‑max (a = softmax(z))
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    # smooth objective value (log‑sum‑exp style)
    smooth_obj = max_c + epsilon * np.log(exp_c.sum())

    return smooth_obj, grad_z, a, max_conv


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same as in the previous programme)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float):
    """
    Cheap Metropolis walk that swaps a random packet of mass between two
    positions.  Used only in the final few seconds.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # source chosen proportionally to mass
        dst = rng.integers(n)              # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # up‑to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # 4096 is a good trade‑off between resolution and speed.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) Warm‑starts
    # ------------------------------------------------------------------
    warm = _collect_warm_starts(n)

    best_seq = None
    best_score = np.inf
    for i, a in enumerate(warm, 1):
        sc = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] Warm‑start {i}/{len(warm)} "
              f"score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # 2) Adam optimisation on the smooth‑max surrogate
    # ------------------------------------------------------------------
    eps = 1e-16
    z = np.log(best_seq + eps)                # logits (soft‑max parametrisation)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0

    # ε‑schedule for the smooth‑max (starts coarse, ends fine)
    epsilon0 = 0.1
    epsilon_min = 0.001

    # keep the best exact objective seen during optimisation
    best_a = best_seq.copy()
    best_obj = best_score

    # Main optimisation loop – we reserve a few seconds for the anneal step
    while time.time() < deadline - 2.0:
        t += 1
        # decay ε logarithmically (halving every 2000 iterations)
        epsilon = max(epsilon_min, epsilon0 * (0.5 ** ((t - 1) // 2000)))

        smooth_obj, grad_z, a, cur_max = _smooth_max_and_grad(z, epsilon, n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = z - step

        # enforce exact symmetry (helps keep the optimum centred)
        z = 0.5 * (z + z[::-1])

        # occasional exact evaluation (every 50 iterations)
        if t % 50 == 0:
            # recompute a from the (now symmetric) logits
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            a = exp_z / exp_z.sum()
            obj = _objective(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} "
                      f"best_score={obj:.6f}")

    # ------------------------------------------------------------------
    # 3) Light annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_obj = _anneal(best_a, deadline, start_time)

    # ------------------------------------------------------------------
    # 4) Final normalisation & conversion to plain list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    return [float(v) for v in best_a]
