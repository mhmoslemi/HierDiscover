"""
Optimization of a step‑function sequence to minimise
2*n*max(conv(a,a)) / (sum(a)**2).

The algorithm:
1. Initialise with the uniform distribution (or a provided global starter).
2. Run many short “exponentiated‑gradient” (mirror‑descent) runs from random
   Dirichlet starts.  The sub‑gradient of the max‑convolution is taken over
   all shifts whose value is within a small relative tolerance of the current
   maximum.  After each update we renormalise onto the probability simplex.
3. Keep the best sequence seen.
4. Finish with a stochastic hill‑climb that moves a small random fraction of
   mass between two positions only when the true objective improves.
All stages respect the overall 1000 s budget.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _active_subgradient(a: np.ndarray,
                        conv: np.ndarray,
                        rel_tol: float = 0.02) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) that averages over all shifts whose
    value is within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val <= 0:
        return np.zeros_like(a)

    # select active shifts
    thresh = max_val * (1 - rel_tol)
    active = np.where(conv >= thresh)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]

    if active.size > 0:
        grad = 2.0 * grad / active.size
    return grad

def _mirror_descent(a0: np.ndarray,
                    start_time: float,
                    deadline: float,
                    best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) optimisation on the simplex.
    Returns the best point seen and its associated score 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    it = 0
    stagnant = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnant = 0
            print(f"[{time.time() - start_time:.2f}s] Mirror it={it:,} score={cur_score:.6f}")
        else:
            stagnant += 1

        # compute sub‑gradient on a slightly enlarged active set
        grad = _active_subgradient(a, conv, rel_tol=0.02)

        # multiplicative update (entropic mirror map)
        eta = 0.5 / math.sqrt(it)   # simple diminishing stepsize
        a = a * np.exp(-eta * grad)

        # renormalise onto simplex
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        # occasional perturbation to escape plateaus
        if stagnant >= 3000:
            rng = np.random.default_rng()
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            stagnant = 0

    return best_a, best_score

def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb: move a tiny random fraction of mass from a
    source index to a destination index and keep the move only if it improves
    the true objective.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # random fraction up to 3 %
        frac = rng.random() * 0.03 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2).

    The strategy consists of:
        * a quick uniform baseline (and a fallback to any provided global
          starter);
        * many short mirror‑descent runs started from random Dirichlet
          vectors;
        * a final stochastic hill‑climb.
    The routine respects the time budget and returns a plain list of Python
    floats suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more degrees of freedom
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline: uniform (or a supplied global starter)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = float(2.0 * n * _conv_self(best_seq, L_fft).max())
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # If a global starter exists, try it
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # truncate or pad to length n
            if starter.size > n:
                starter = starter[:n]
            elif starter.size < n:
                starter = np.concatenate([starter, np.zeros(n - starter.size)])
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                cur = float(2.0 * n * _conv_self(starter, L_fft).max())
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Mirror‑descent optimisation (bulk of the budget)
    # ------------------------------------------------------------------
    # Reserve ~150 s for the final hill‑climb
    hill_time = 150.0
    opt_deadline = deadline - hill_time
    run_time = 30.0   # each mirror‑descent run gets ~30 s

    while time.time() < opt_deadline:
        # decide how much time we can spend on this run
        this_deadline = min(time.time() + run_time, opt_deadline)

        # initialise a random Dirichlet vector; concentration <1 encourages
        # a few larger entries while still keeping many small ones
        alpha = 0.4
        a0 = np.random.dirichlet(np.full(n, alpha))

        # occasionally seed from the current best to refine it
        if np.random.rand() < 0.1:
            a0 = best_seq.copy()

        a_opt, sc_opt = _mirror_descent(a0, start_time, this_deadline, best_score)
        if sc_opt < best_score:

2️⃣  Run many short “exponentiated‑gradient” (mirror‑descent) optimisations

    relative tolerance of the current maximum; after each multiplicative
    update we renormalise onto the probability simplex.
3️⃣  Keep the best sequence seen.
4️⃣  Finish with a stochastic hill‑climb that moves a tiny random fraction
    of mass from one position to another only when the true objective

import time

import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _active_subgradient(
    a: np.ndarray, conv: np.ndarray, rel_tol: float = 0.02
) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) that averages over all shifts whose
    value is within ``rel_tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val <= 0:
        return np.zeros_like(a)

    thresh = max_val * (1 - rel_tol)
    active = np.where(conv >= thresh)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]

    if active.size > 0:
        grad = 2.0 * grad / active.size
    return grad
