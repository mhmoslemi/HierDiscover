 def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑score step function.

Algorithm
---------
1. Initialise a symmetric “arcsine‑type” probability vector `p` of length `n`
   (default 4096).  A random exponent α∈[0.2,0.5] is used; the vector is
   normalised to sum = 1.
2. Repeatedly perform a *targeted mass‑transfer* move:
      • Find the shift `k` where the self‑convolution `c=p∗p` attains its
        maximum.
      • Among the indices contributing to this shift pick the source `src`
        with the largest product `p[i]·p[k−i]`.  Its partner is `partner=k‑src`.
      • Compute the second‑largest convolution value `second`.
      • The reduction of `c[k]` is linear in the transferred mass ε:
          Δc[k] ≈ –2·p[partner]·ε  (or –2·p[src]·ε if src=partner).
        The largest ε that could bring `c[k]` down to `second` is
          ε_max = (c[k]−second) / (2·p[partner]).
        The actual ε is the smaller of ε_max and a fraction of the source
        mass (the fraction shrinks from 0.2 → 0.01 during the run).
      • Destination `dst` is chosen among indices *outside* the support of
        shift `k` (so the move does not increase `c[k]`).  Low‑mass indices
        are preferred.
      • Perform the move `p[src]–=ε , p[dst]⁺=ε`, recompute the convolution,
        and keep the move only if the peak autocorrelation strictly decreased.
3. If no improvement is observed for many iterations a small random
   jitter is added to the current best vector to escape plateaus.
4. The best vector seen during the whole budget is returned.  Because the
   objective is scale‑invariant, a probability vector (sum = 1) is a valid
   answer.

The method improves on the previous code by:
* choosing ε based on the *gap* to the second‑largest autocorrelation,
  thus taking larger, more purposeful steps;
* restricting destinations to indices that cannot increase the current
  worst shift;
* a simple simulated‑annealing style jitter to escape local minima.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: linear self‑convolution via FFT (fast enough for all n)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Simple projection onto the probability simplex Σx=1, x≥0
# ------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fall back to uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step sequence with low evaluation score.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a small safety margin for the outer harness
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector p (Σp = 1)
    # ------------------------------------------------------------------
    try:
        # If a previous construction exists we use it as a warm start
        p0 = np.asarray(height_sequence_1, dtype=float)
        if p0.ndim != 1 or p0.size == 0:
            raise ValueError
        p0 = np.maximum(p0, 0.0)
        if p0.sum() == 0.0:
            raise ValueError
        p = p0 / p0.sum()
        n = p.shape[0]
        print(f"[{time.time() - start:.2f}s] Using height_sequence_1 (n={n})")
    except Exception:
        # fallback: an arcsine‑type distribution with a random exponent
        n = 4096
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        alpha = 0.2 + 0.3 * rng.random()           # α ∈ [0.2,0.5]
        eps = 1e-6
        w = (x * (1.0 - x) + eps) ** (-alpha)
        p = w / w.sum()
        # small random perturbation to break symmetry a bit
        jitter = 0.02 * p * (rng.random(n) - 0.5)
        p = np.maximum(p + jitter, 0.0)
        p /= p.sum()
        print(f"[{time.time() - start:.2f}s] Generated fresh arcsine start (n={n}, α={alpha:.3f})")

    # ------------------------------------------------------------------
    # 2) Initial convolution and bookkeeping
    # ------------------------------------------------------------------
    conv = _conv_self(p)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max            # because Σp == 1
    best_p = p.copy()
    print(f"[{time.time() - start:.2f}s] initial score = {best_score:.6f}")

    iteration = 0
    last_improve_iter = 0

    # ------------------------------------------------------------------
    # 3) Main optimisation loop (targeted mass‑transfer + jitter)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 3a) locate the current worst shift and its second‑largest value
        # --------------------------------------------------------------
        k = int(conv.argmax())
        # two largest values (might be equal)
        top2 = np.partition(conv, -2)[-2:]          # last two are the largest
        max_val = top2.max()
        second_val = top2.min()
        # if all entries equal we cannot make progress – perform a jitter
        if max_val - second_val < 1e-15:
            # jitter the best candidate and continue
            jitter = 0.005 * best_p * (rng.random(n) - 0.5)
            p = np.maximum(best_p + jitter, 0.0)
            p /= p.sum()
            conv = _conv_self(p)
            cur_max = conv.max()
            continue

        # --------------------------------------------------------------
        # 3b) choose source index (largest product contributing to shift k)
        # --------------------------------------------------------------
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = p[idx] * p[k - idx]
        src = idx[int(np.argmax(contrib))]
        partner = k - src                     # partner index (may equal src)

        # --------------------------------------------------------------
        # 3c) compute a sensible step size ε
        # --------------------------------------------------------------
        # maximal fraction of the source mass we are willing to move
        elapsed = time.time() - start
        max_frac = max(0.01, 0.20 * (1.0 - elapsed / budget_s))
        max_move = p[src] * max_frac

        # reduction needed to bring the peak down to the second value
        partner_mass = p[partner] if 0 <= partner < n else 0.0
        if partner_mass <= 0.0:
            # fallback: just respect the maximal mass‑fraction
            eps_max = max_move
        else:
            eps_gap = (max_val - second_val) / (2.0 * partner_mass + 1e-12)
            eps_max = min(max_move, eps_gap)

        if eps_max <= 0.0:
            # nothing to move – try a random jitter later
            continue
        epsilon = eps_max * rng.random()          # random fraction of eps_max

        # --------------------------------------------------------------
        # 3d) pick a destination outside the support of shift k
        # --------------------------------------------------------------
        # indices that *do not* influence c[k]
        outside = np.concatenate(
            (np.arange(0, lo, dtype=int), np.arange(hi + 1, n, dtype=int))
        )
        if outside.size == 0:
            # degenerate case (should not happen for large n) – pick any other
            candidates = np.delete(np.arange(n, dtype=int), src)
            dst = rng.choice(candidates)
        else:
            # favour indices with small current mass
            w = 1.0 - p[outside]
            if w.sum() <= 0.0:
                w = np.ones_like(w)
            prob = w / w.sum()
            dst = rng.choice(outside, p=prob)

        if dst == src:
            continue

        # --------------------------------------------------------------
        # 3e) perform the move and evaluate
        # --------------------------------------------------------------
        p_new = p.copy()
        p_new[src] -= epsilon
        p_new[dst] += epsilon
        # numerical safety
        if p_new[src] < 0.0:
            p_new[src] = 0.0

        conv_new = _conv_self(p_new)
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # accept only if the peak strictly decreased
        if new_max < cur_max - 1e-16:
            p = p_new
            conv = conv_new
            cur_max = new_max
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                last_improve_iter = iteration
                print(
                    f"[{time.time() - start:.2f}s] iter={iteration:,} "
                    f"new best score={best_score:.6f}"
                )
        # ------------------------------------------------------------------
        # 3f) periodic housekeeping (renormalise to avoid drift)
        # ------------------------------------------------------------------
        if iteration % 5000 == 0:
            s = p.sum()
            if abs(s - 1.0) > 1e-12:
                p /= s
                conv = _conv_self(p)

        # ------------------------------------------------------------------
        # 3g) plateau handling – add a tiny jitter around the best seen
        # ------------------------------------------------------------------
        if iteration - last_improve_iter > 200_000:
            p = best_p.copy()
            jitter = 0.005 * p * (rng.random(n) - 0.5)
            p = np.maximum(p + jitter, 0.0)
            if p.sum() == 0.0:
                p = np.full(n, 1.0 / n)
            else:
                p /= p.sum()
            conv = _conv_self(p)
            cur_max = conv.max()
            last_improve_iter = iteration
            print(
                f"[{time.time() - start:.2f}s] plateau – re‑jittered at iter {iteration}"
            )

    # ------------------------------------------------------------------
    # 4) Return the best candidate found
    # ------------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0.0:
        best_p = best_p / total
    else:
        best_p = np.full(n, 1.0 / n)

    return [float(v) for v in best_p]
