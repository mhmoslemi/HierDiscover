def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # find rho = last index where condition holds
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    """Projects v onto the probability simplex {x : x>=0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All <=0, return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

"""
Gradient‑descent search on a soft‑max approximation of the
max‑self‑convolution.  The algorithm keeps the vector on the
probability simplex (non‑negative, sum = 1) and uses an Adam‑style
step together with a temperature schedule that gradually focuses the
soft‑max on the true maximum.  Plateaus trigger a small random jitter
and a restart, so the search can escape local minima.  The best vector
found within the time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    Implements the O(n log n) algorithm of
    Duchi et al., "Efficient Projections onto the l1‑Ball for
    Learning in High Dimensions".
    """
    if v.ndim != 1:
        raise ValueError("simplex projection requires a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all numbers are <= 0 → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(p: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using a fixed FFT length L."""
    Af = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * p.shape[0] - 1]
    return conv


def _gradient_softmax(p: np.ndarray, softmax_weights: np.ndarray,
                     Lgrad: int) -> np.ndarray:
    """
    Gradient of the smoothed objective
        g(p) = T * logsumexp(p*p / T)
    with respect to p.  The result is
        2 * correlation(softmax_weights, p)
    and is computed via FFT.
    """
    revp = p[::-1]                     # reversed version of p
    Wf = np.fft.rfft(softmax_weights, n=Lgrad)
    Pf = np.fft.rfft(revp, n=Lgrad)
    corr_full = np.fft.irfft(Wf * Pf, n=Lgrad)
    # length of correlation = len(softmax_weights) + len(p) - 1 = 3n-2
    n = p.shape[0]
    start = n - 1
    # slice the part that corresponds to correlation at lags 0..n‑1
    grad = 2.0 * corr_full[start : start + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step sequence within the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start_time = time.time()
    # keep a tiny safety margin before the hard deadline
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1) initialise a probability vector (Σp = 1)
    # --------------------------------------------------------------
    try:
        # try to reuse a previously discovered construction
        p0 = np.asarray(height_sequence_1, dtype=float)
        if p0.ndim != 1 or p0.size == 0:
            raise ValueError
        p0 = np.maximum(p0, 0.0)
        if p0.sum() == 0.0:
            raise ValueError
        p = p0 / p0.sum()
        n = p.shape[0]
        print(f"[{time.time() - start_time:.2f}s] Using height_sequence_1 (n={n})")
    except Exception:
        # fall back to the simple arcsine‑type distribution (n = 4096)
        n = 4096
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        alpha = 0.38
        eps = 1e-6
        w = (x * (1.0 - x) + eps) ** (-alpha)
        p = w / w.sum()
        print(f"[{time.time() - start_time:.2f}s] Generated arcsine start (n={n})")

    # pre‑compute FFT lengths that stay constant throughout the run
    L_self = _next_pow2(2 * n - 1)      # for p * p
    L_grad = _next_pow2(3 * n - 2)      # for correlation(softmax, p)

    # --------------------------------------------------------------
    # 2) bookkeeping
    # --------------------------------------------------------------
    best_p = p.copy()
    conv = _conv_self_fft(p, L_self)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max
    best_iter = 0
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # schedule parameters
    lr0 = 0.20                # initial learning rate
    T0 = 0.03                 # initial temperature
    Tmin = 0.001              # minimum temperature
    plateau_limit = 3000     # iterations without improvement before jitter

    iteration = 0
    last_improve_iter = 0

    # --------------------------------------------------------------
    # 3) main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # --- evaluate current objective ---------------------------------
        conv = _conv_self_fft(p, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            best_iter = iteration
            last_improve_iter = iteration
            print(
                f"[{time.time() - start_time:.2f}s] iter={iteration:,} "
                f"new best score={best_score:.6f}"
            )

        # --- temperature schedule (linear decay) -----------------------
        T = max(Tmin, T0 * (1.0 - iteration / (budget_s * 10.0)))

        # --- soft‑max over the convolution entries ----------------------
        shift = conv - conv.max()          # shift → max becomes 0 (numerical stability)
        exp_shift = np.exp(shift / T)      # all values ≤ 1
        soft_weights = exp_shift / exp_shift.sum()   # length 2n‑1, sums to 1

        # --- gradient of the smoothed objective ------------------------
        grad = _gradient_softmax(p, soft_weights, L_grad)

        # --- Adam update (step = -lr * grad) ---------------------------
        lr = lr0 * (0.99 ** iteration)     # slowly decay learning rate
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        p_new = p - step

        # --- project back onto the simplex -------------------------------
        p = _project_simplex(p_new)

        # --- plateau handling (jitter + restart) -----------------------
        if iteration - last_improve_iter > plateau_limit:
            # small random perturbation around the best seen vector
            jitter_scale = 0.01
            jitter = jitter_scale * (rng.random(p.shape) - 0.5)
            p = best_p + jitter
            p = _project_simplex(p)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = iteration
            print(
                f"[{time.time() - start_time:.2f}s] plateau at iter {iteration} – jittered restart"
            )

    # --------------------------------------------------------------
    # 4) final output
    # --------------------------------------------------------------
    best_p = _project_simplex(best_p)   # final safety projection
    return [float(val) for val in best_p]
