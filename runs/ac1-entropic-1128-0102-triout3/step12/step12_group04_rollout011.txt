def _subgradient_optimize(a0, deadline, start_time, initial_eta=0.05, decay=0.9995, min_eta=1e-5):
    a = a0.copy()
    n = len(a)
    L = _next_pow2(3*n)  # safe
    best_a = a.copy()
    best_score = _objective(a)
    eta = initial_eta

    iter=0
    while time.time() < deadline and eta > min_eta:
        iter+=1
        # compute conv
        conv = _conv_self(a, L)

        # find max; find indices where conv diff <= tolerance
        M = conv.max()
        # Use tolerance relative to M maybe 1e-12 or 1e-10?
        # To avoid missing near-equal values due to rounding, we can treat indices where conv >= M - 1e-12
        tolerance = 1e-12
        max_idx = np.where(conv >= M - tolerance)[0]
        # Build w vector
        w = np.zeros_like(conv)
        w[max_idx] = 1.0 / max_idx.size

        # compute gradient
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L)
        Revf = np.fft.rfft(a_rev, n=L)
        conv_wr = np.fft.irfft(Wf * Revf, n=L)
        grad = 2.0 * conv_wr[n-1:n-1+n]  # gradient of max conv

        # multiply by 2*n for objective gradient
        grad_obj = 2.0 * n * grad

        # update with step size
        a -= eta * grad_obj

        # project onto simplex
        a = _project_simplex(a)

        # Evaluate objective
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Subgrad iter={iter:,} best={cur_score:.6f}")

        # adapt step size: maybe reduce gradually
        eta *= decay

    return best_a, best_score

def _adam_optimize_softmax(a0, deadline, start_time, lam, lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3*n)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0

    while time.time() < deadline:
        it += 1

        grad, conv = _softmax_gradient(a, lam, L)

        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Softmax λ={lam} iter={it:,} best={cur_score:.6f}")

        # gradient of objective
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection
        a = _project_simplex(a)

    return best_a, best_score

"""
Propose a low‑objective step‑function.

The algorithm consists of four stages:

1. **Warm‑start** – a handful of analytic shapes (tent, cosine, half‑circle,
   Gaussian, power‑tent) and a few random Dirichlet samples are generated.
   The one with the smallest exact objective `2·n·max(conv(a,a))` is kept.

2. **Soft‑max Adam** – we minimise a smooth approximation of the
   `max` using a soft‑max over the convolution values.
   A first phase (λ≈5) explores the landscape, a second phase (λ≈20)
   sharpens the approximation.

3. **Exact‑max Adam** – Adam is run again with the true sub‑gradient
   (`top_k=1`).  This refines the solution to the genuine objective.

4. **Light annealing** – a Metropolis walk moves a random packet of mass
   between two indices; it uses the exact objective for acceptance and
   provides a final polishing step.

All updates are projected onto the probability simplex (non‑negative and
summing to 1).  The routine respects the 1000 s budget and returns a plain
list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a normalised (Σa = 1) vector.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform simplex point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑start shapes
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a few analytic candidates and normalise them."""
    x = np.arange(n, dtype=float) / (n - 1)  # coordinate in [0,1]

    tent = np.minimum(x, 1.0 - x)                    # linear tent
    cosine = np.sin(np.pi * x)                       # raised cosine
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1) ** 2, 0.0, 1.0))
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normalised.append(c / s)
    return normalised


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------
def _topk_gradient(a: np.ndarray,
                   top_k: int,
                   L: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Gradient of the *average* of the `top_k` largest convolution entries.
    Returns (gradient, full convolution).
    """
    n = a.shape[0]

    conv = _conv_self(a, L)

    if top_k >= conv.size:
        w = np.full(conv.shape, 1.0 / conv.size, dtype=float)
    else:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros_like(conv, dtype=float)
        w[top_idx] = 1.0 / top_k

    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad, conv


def _softmax_gradient(a: np.ndarray, lam: float, L: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Gradient of a smooth (soft‑max) approximation of `max(conv)`.
    Returns (gradient, full convolution).
    """
    n = a.shape[0]
    conv = _conv_self(a, L)

    # stable softmax
    cmax = conv.max()
    exp_vals = np.exp(lam * (conv - cmax))
    w = exp_vals / exp_vals.sum()

    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)

    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]          # ∂max/∂a (soft)
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser (top‑k version)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   top_k: int = 150,
                   lr: float = 0.025,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex using the gradient of the averaged
    top‑k convolution entries.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, conv = _topk_gradient(a, top_k, L)

        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam topk={top_k} iter={it:,} best={cur_score:.6f}")

        grad_obj = 2.0 * n * grad

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser with soft‑max gradient
# ----------------------------------------------------------------------
def _adam_softmax_optimize(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           lam: float,
                           lr: float = 0.025,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex using a soft‑max approximation of the
    max‑convolution.  The parameter `lam` controls the sharpness of the
    approximation (larger → closer to the true max).
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, conv = _softmax_gradient(a, lam, L)

        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Softmax λ={lam:.1f} iter={it:,} best={cur_score:.6f}")

        grad_obj = 2.0 * n * grad

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Metropolis walk moving a random packet of mass between two indices.
    A larger number of steps than the reference implementation is used,
    but the loop respects the global deadline.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of proposals – large but will stop early if time runs out
    total_steps = 200_000

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # source proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps_move = a[src] * rng.random() * 0.25   # up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a power of two enables fast FFTs.
    # ------------------------------------------------------------------
    n = 16384

    # ------------------------------------------------------------------
    # 1) Warm‑start – analytic shapes + a few random Dirichlet samples.
    # ------------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # Add a couple of random Dirichlet‑like vectors (heavy‑tailed)
    rng = np.random.default_rng(seed + 12345)
    for _ in range(2):
        rnd = rng.gamma(0.5, 1.0, size=n)   # shape <1 gives more U‑shapes
        s = rnd.sum()
        if s > 0:
            warm_starts.append(rnd / s)

    # If a previous best is available in the global namespace, use it
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                warm_starts.append(starter)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Soft‑max Adam phases (exploration → sharpening)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # Phase 1 – gentle soft‑max
        phase_deadline = time.time() + remaining * 0.4
        a_opt, sc = _adam_softmax_optimize(a0, phase_deadline, start_time,
                                           lam=5.0, lr=0.025)
        if sc < best_score:
            best_score = sc
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Softmax λ=5 best={sc:.6f}")

    remaining = deadline - time.time()
    if remaining > 30.0:
        # Phase 2 – sharper soft‑max
        phase_deadline = time.time() + remaining * 0.3
        a_opt, sc = _adam_softmax_optimize(a0, phase_deadline, start_time,
                                           lam=20.0, lr=0.015)
        if sc < best_score:
            best_score = sc
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Softmax λ=20 best={sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max Adam (top‑k = 1) – fine tuning
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        phase_deadline = time.time() + remaining * 0.2
        a_opt, sc = _adam_optimize(a0, phase_deadline, start_time,
                                   top_k=1, lr=0.005)
        if sc < best_score:
            best_score = sc
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam top‑1 best={sc:.6f}")

    # ------------------------------------------------------------------
    # 4) Light annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(a0, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            a0 = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal best={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list.
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective={best_score:.6f}")
    return [float(v) for v in a0]
```

This implementation respects the time budget, explores the search space with several complementary optimisation phases, and returns a valid non‑negative step‑function ready for `evaluate_sequence`.
