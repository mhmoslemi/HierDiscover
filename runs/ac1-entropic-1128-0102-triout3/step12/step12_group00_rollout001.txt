"""
Propose a step‑function (non‑negative probability vector) with a very low
``evaluate_sequence`` score.

Strategy
--------
1. Choose a power‑of‑two length `n = 65536` (fast FFT, fine discretisation).
2. Initialise a smooth “tent‑plus‑plateau’’ shape (the same family used
   previously) and normalise it to sum = 1.
3. Perform **continuation gradient descent** on a smooth approximation of the
   max‑convolution:
       g_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
   The gradient is 2·corr(a, p) where p are the soft‑max weights.
   β is increased from 0.5 to 30, each stage using a diminishing step‑size
   and Euclidean projection onto the probability simplex.
4. Refine the best iterate with the (non‑smooth) sub‑gradient method used
   previously, but with a larger active‑set (`top_k = 10`).
5. Light simulated‑annealing polish to escape possible tiny local minima.
6. Normalise and return the final vector as a plain Python list.

All helper functions are defined at top level so they can be pickled if the
harness runs the solution in a child process.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
#  Convolution utilities (real FFT)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (real FFT). Returns length 2·n‑1 array."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
#  Exact objective (scale invariant, sum(a)=1 → score = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
#  Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Wang‑Carreira‑Perpiñán simplex projection (O(n log n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
#  Analytic shape generator – power ramp + optional flat plateau
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape of length `n`.  The side ramps rise as (i/side_len)^p
    and meet a plateau of height 1 of relative width `alpha∈[0,0.5]`.
    The vector is normalised to sum = 1.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                     # values rise from 0…1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # correct possible off‑by‑one because of rounding
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
#  Smooth approximation of max‑convolution and its gradient
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Returns ∇_a  g_β(a) where
        g_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).
    The gradient is 2·corr(a, p) with p the soft‑max weights.
    """
    n = a.shape[0]

    # self‑convolution
    conv = _conv_self(a)
    max_c = conv.max()

    # soft‑max weights (stable)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a * w  (no reversal) = convolution of a with reversed w
    L = _next_pow2(3 * n - 2)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w[::-1], n=L)           # reversed w
    corr = np.fft.irfft(Af * Wf, n=L)

    start = 2 * n - 2                         # index that corresponds to shift 0
    grad = 2.0 * corr[start:start + n]
    return grad


# ----------------------------------------------------------------------
#  Projected (non‑smooth) sub‑gradient descent (as in the reference code)
# ----------------------------------------------------------------------
def _subgradient_descent_multi(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               top_k: int = 3) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    `top_k` determines how many of the largest convolution entries are used
    to form the sub‑gradient (larger → smoother direction).
    Returns the best iterate seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta_mom = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max  # sum(a)=1 → exact score
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # indices of the K largest convolution entries
        if top_k >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # build sub‑gradient (average over selected ks)
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # momentum + diminishing step‑size
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
#  Light simulated‑annealing polish (as in the reference code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny mass‑packet moves with Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # simple exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
#  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) with a low
    evaluate_sequence score.  The routine follows the plan described in
    the module docstring and respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Problem size – power of two for fast FFTs
    # --------------------------------------------------------------
    n = 65536

    # --------------------------------------------------------------
    # 2) Initialise a reasonable shape
    # --------------------------------------------------------------
    # Try to reuse a globally supplied starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                if starter.size != n:
                    # simple linear interpolation to required length
                    x_old = np.linspace(0, 1, starter.size)
                    x_new = np.linspace(0, 1, n)
                    starter = np.interp(x_new, x_old, starter)
                a0 = starter
                print(f"[{time.time() - start_time:.2f}s] Using global starter")
            else:
                raise ValueError
        else:
            raise ValueError
    except Exception:
        # fallback: a symmetric power‑ramp + plateau
        a0 = _make_shape(n, p=0.8, alpha=0.2)

    # Ensure exact probability simplex
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()

    best_a = a0.copy()
    best_score = _objective(best_a)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Continuation gradient descent on smooth max‑convolution
    # --------------------------------------------------------------
    # schedule of inverse‑temperature parameters
    betas = [0.5, 2.0, 5.0, 10.0, 30.0]
    # allocate roughly 70 % of the remaining time to this phase
    smooth_budget = (deadline - time.time()) * 0.70
    stage_start = time.time()
    for beta in betas:
        # each stage gets an equal share of the smooth budget
        stage_deadline = stage_start + smooth_budget / len(betas)
        a = best_a.copy()
        momentum = np.zeros_like(a)
        it = 0
        while time.time() < stage_deadline:
            it += 1
            grad = _smooth_gradient(a, beta)

            # simple momentum
            momentum = 0.9 * momentum + 0.1 * grad
            step = 0.5 / math.sqrt(it) / max(1.0, beta)   # smaller steps for larger β
            a -= step * momentum
            a = _project_simplex(a)

            # occasional exact evaluation (every 200 iterations)
            if it % 200 == 0:
                cur_score = _objective(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] beta={beta:.1f} "
                          f"iter={it:,} score={cur_score:.6f}")

        # carry the best iterate forward
        best_a = best_a.copy()
        stage_start = time.time()

    print(f"[{time.time() - start_time:.2f}s] After smooth phase score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Non‑smooth sub‑gradient refinement (larger active set)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = time.time() + (deadline - time.time()) * 0.60
        refined_a, refined_score = _subgradient_descent_multi(best_a,
                                                             sub_deadline,
                                                             start_time,
                                                             top_k=10)
        if refined_score < best_score:
            best_score = refined_score
            best_a = refined_a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad refined score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_a, anneal_score = _anneal(best_a, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_a = anneal_a.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
