"""
Propose a low‑objective step‑function.

The objective evaluated by the platform is

    2·n·max(conv(a,a)) / (sum(a)^2)

with non‑negative entries.  The objective is invariant under scaling,
so we fix the sequence to lie on the probability simplex
(Σa_i = 1).  The problem then reduces to minimising
`2·n·max(conv(a,a))`.

This implementation improves on the reference code by
* using the exact sub‑gradient of the max‑convolution
  (instead of a top‑k approximation);
* employing projected Adam on the simplex;
* adding a cheap simulated‑annealing polish;
* trying a richer set of warm‑starts (analytic shapes + random Dirichlet).

The whole search respects a 1000 s budget and returns a plain
`list[float]` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w ≥ 0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector already on the simplex."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, conv_full: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Returns the gradient of the max‑convolution entry and the index where it occurs.
    The gradient is:
        g_i = 2·a_{k-i}    if 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1,
        g_i = 0            otherwise,
    where k = argmax conv.
    """
    n = a.shape[0]
    k = int(conv_full.argmax())
    grad = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)

    if i_min <= i_max:
        idx = np.arange(i_min, i_max + 1)
        grad[idx] = 2.0 * a[k - idx]

    return grad, k


# ----------------------------------------------------------------------
# Projected Adam using the exact sub‑gradient
# ----------------------------------------------------------------------
def _adam_exact_opt(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    lr: float = 0.07,
                    beta1: float = 0.9,
                    beta2: float = 0.999,
                    eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex with the exact sub‑gradient of the
    max‑convolution.  Returns the best iterate encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)          # length sufficient for the FFTs

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)              # first moment
    v = np.zeros_like(a)              # second moment

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- convolution & max index -----
        Af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        grad_max, max_idx = _max_gradient(a, conv_full)

        cur_score = float(2.0 * n * conv_full[max_idx])
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # ----- Adam update (objective = 2·n·max) -----
        grad_obj = 2.0 * n * grad_max               # ∇(2·n·max)
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ---- projection onto simplex ----
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk that transfers a random packet of mass
    between two indices.  Operates on the simplex.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 150_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (very fast)
        T = 0.1 * (1e-5 / 0.1) ** (it / total_steps)

        src = rng.choice(n, p=a)            # source as per current mass distribution
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.20   # move up to 20 % of the source mass
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps

        new_score = _objective(prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Warm‑starting candidate generators
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """
    Returns a list of simple analytic shapes plus a few random Dirichlet
    draws, all normalised to sum‑to‑1.
    """
    x = np.linspace(0.0, 1.0, n, dtype=float)

    # Analytic candidates
    tent = np.minimum(x, 1.0 - x)                           # linear tent
    cosine = np.sin(np.pi * x)                              # raised cosine
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))  # half‑circle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)        # Gaussian
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p                # flatter tent

    candidates = [tent, cosine, circ, gauss, power_tent]

    # Random Dirichlet draws (adds stochastic diversity)
    rng = np.random.default_rng()
    for _ in range(5):
        rand = rng.dirichlet(np.ones(n))
        candidates.append(rand)

    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normals.append(c / s)
    return normals
