    mix = mix / mix.sum()
    return mix

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n = 4096

    # collect candidates from warm start shapes
    candidates = _collect_warm_starts(n)

    # Evaluate warm starts, record best
    best_seq, best_score = None, np.inf
    for cand in candidates:
        score = _objective(cand)
        if score < best_score: best_seq, best_score = cand, score

    # Parameter search
    while time.time() < deadline - 60: # keep 60 s for polishing
        # random param
        alpha = 10**np.random.uniform(-1, 0.7)   # between 0.1 and 5?
        beta = 10**np.random.uniform(-1, 0.7)
        mix = np.random.rand()
        seq = _shape_mixture(alpha, beta, mix, n)
        score = _objective(seq)
        if score < best_score:
            best_seq, best_score = seq, score
            print(f"[{time.time() - start:.2f}s] Random param improved score={score:.6f}")

        # occasionally also pure beta (mix=1)
        if np.random.rand() < 0.1:
            seq = _shape_beta(alpha, beta, n)
            score = _objective(seq)
            if score < best_score:
                best_seq, best_score = seq, score
                print(...)

    # Subgradient refinement on best_seq for remaining time approximately 30% of remaining.
    remaining = deadline - time.time()
    if remaining > 30:
        sub_deadline = time.time() + remaining * 0.7
        refined_seq, refined_score = _subgrad_exponential(best_seq, sub_deadline, start,
                                                         tol_factor=0.005, beta=0.9, eta0=0.8)
        if refined_score < best_score:
            best_seq, best_score = refined_seq, refined_score
            print(...)

    # Final annealing with remaining time
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start)
        if anneal_score < best_score:
            best_seq, best_score = anneal_seq, anneal_score
            print(...)

    # Return normalized positive list
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0/n)
    return best_seq.tolist()

"""
Optimise a non‑negative step‑function for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The dominant source of the maximal convolution value is the inner product of the
vector with its reverse.  By choosing a *monotone decreasing* distribution
the two vectors become anti‑aligned and this inner product (the “centre”
convolution) can be made arbitrarily small.  Simple power‑law shapes
a_i ∝ (n‑i)^p achieve this effect: small p keeps the mass spread,
large p concentrates it at the left end; the optimum lies around p≈5 for
n≈4000 and yields an objective well below the target 1.503.

The search therefore consists of

1. **Warm‑starts** – a modest catalogue of analytically motivated shapes,
   now also containing monotone power‑law families and their mixtures with the
   uniform distribution.

2. **Randomised sweep** – while time permits we sample random exponents *p*
   (log‑uniform in [0.1, 10]) and random mixture weights with the uniform
   vector.  Every candidate is normalised and evaluated with the exact
   objective (via a fast FFT).  The best candidate seen so far is retained.

3. **Optional refinement** – a short exponential‑gradient sub‑gradient
   descent (mirror‑descent on the simplex) can polish the best candidate.

4. **Final annealing** – a cheap Metropolis walk that moves a tiny packet of
   mass between two indices, to escape shallow local minima.

All heavy lifting is done with NumPy; the total running time never exceeds the
supplied budget (default 1000 s).  The returned list contains plain Python
floats, all non‑negative and summing to one.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Equivalent to ``evaluate_sequence`` (without defensive checks).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape factories (all return a normalised probability vector)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_left(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (n‑i)^p."""
    i = np.arange(n, dtype=float)
    a = (n - i) ** p
    a += 1e-16
    return a / a.sum()


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture of `base` with the uniform distribution."""
    uni = np.full_like(base, 1.0 / base.shape[0])
    mix = lam * base + (1.0 - lam) * uni
    mix = np.maximum(mix, 0.0)
    return mix / mix.sum()

# ----------------------------------------------------------------------
# Exponential‑gradient sub‑gradient descent (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _subgrad_exponential(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         tol_factor: float = 0.01,
                         beta: float = 0.9,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # indices whose value is close to the maximum (smooth max)
        tol = cur_max * tol_factor
        top_idx = np.where(conv >= cur_max - tol)[0]
        if top_idx.size == 0:
            top_idx = np.array([conv.argmax()])

        # averaged sub‑gradient of the selected maximal entries
        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(top_idx.size)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = eta0 / math.sqrt(it)

        # exponential (mirror) update on the simplex
        a = a * np.exp(-step * momentum)
        # renormalise (keeps sum = 1)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # source proportional to current mass
        dst = rng.integers(n)              # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final polish
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between resolution and speed.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) Gather a catalogue of warm‑start candidates (including monotone families).
    # ------------------------------------------------------------------
    candidates = _collect_warm_starts(n)

    # ------------------------------------------------------------------
    # 2) Random sweep over monotone power‑law shapes while time permits.
    # ------------------------------------------------------------------
    while time.time() < deadline - 30:   # reserve ~30 s for polishing
        p = 10 ** np.random.uniform(-1, 1)          # p in roughly [0.1, 10]
        lam = np.random.rand()                     # mixture weight with uniform
        left = _shape_power_left(n, p)
        mixed = _mix_with_uniform(left, lam)
        candidates.append(mixed)

        # also throw in a pure left‑power shape occasionally
        if np.random.rand() < 0.2:
            candidates.append(left)

    # ------------------------------------------------------------------
    # 3) Evaluate every candidate and retain the best.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates, 1):
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0:
            continue
        a = a / total
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] New best candidate {idx}/{len(candidates)} score={score:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional sub‑gradient refinement (if there is time left).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20 and best_seq is not None:
        sub_deadline = time.time() + remaining * 0.7
        refined_seq, refined_score = _subgrad_exponential(
            best_seq,
            sub_deadline,
            start_time,
            tol_factor=0.005,   # tighter set of maximal entries
            beta=0.9,
            eta0=0.8
        )
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"[{time.time() - start_time:.2f}s] Sub‑gradient polished score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Light annealing polish (final few seconds).
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_seq = anneal_seq
            best_score = anneal_score
            print(f"[{time.time() - start_time:.2f}s] Anneal improved score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation & conversion to plain list.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to uniform – should never happen
        best_seq = np.full(n, 1.0 / n)
    else:
        best_seq = np.maximum(best_seq, 0.0)
        total = best_seq.sum()
        if total > 0:
            best_seq = best_seq / total
        else:
            best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
