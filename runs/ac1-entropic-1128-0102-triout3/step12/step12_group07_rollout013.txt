"""
Uniform step function (optimal under the given evaluation).

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

For non‑negative a, max(conv(a, a)) is attained at zero shift and equals
∑ a_i².  By the Cauchy–Schwarz inequality
    ∑ a_i² ≥ (∑ a_i)² / n,
with equality iff all entries are equal.  Consequently the score is
    ≥ 2,
and the lower bound is attained by any *uniform* vector.
The routine below therefore returns a uniform sequence of the same length
as a previously known construction (if available) or of length 4096
otherwise.  This guarantees the optimal score of 2.0 and trivially satisfies
the time budget.
"""

import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a uniform non‑negative step function.

    Parameters
    ----------
    seed : int
        Ignored – the construction is deterministic.
    budget_s : int
        Ignored – the algorithm is O(n) and finishes instantly.

    Returns
    -------
    list[float]
        A list `a` with length `n` such that a_i = 1/n for every i.
        This minimizes `evaluate_sequence(a)`; the theoretical optimum
        is 2.0.
    """
    # Try to reuse the length of a previously discovered sequence.
    # If that variable does not exist, fall back to a reasonable size.
    try:
        # `height_sequence_1` is provided by the platform when a prior
        # construction exists.  We only need its length.
        n = int(len(height_sequence_1))
        if n <= 0:
            raise Exception
    except Exception:
        n = 4096  # a comfortable default size

    # Uniform value that makes the sum exactly 1 (any positive scaling
    # would give the same score because the metric is scale‑invariant).
    uniform_val = 1.0 / n

    # Build and return the list.
    return [float(uniform_val) for _ in range(n)]
