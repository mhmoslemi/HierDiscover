"""
Propose a step‑function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣  A fast parametric sweep (inverse‑power shapes, geometric decay,
    and a few random Dirichlet draws) to obtain several promising
    starting points.

2️⃣  A sub‑gradient descent on the exact objective.  The gradient of the
    active convolution entry (the one attaining the current maximum)
    can be written in closed form, which allows a cheap O(n) update.
    After each step the vector is projected onto the probability simplex
    (∑a_i = 1, a_i ≥ 0) using the Euclidean projection.

3️⃣  A targeted hill‑climb.  The index of the maximal convolution term
    is identified; the contribution of each coefficient to this term
    is used as a sampling weight to draw a “source” coordinate.  A
    small fraction of its mass is transferred to a “destination” with
    much lower contribution.  Only improving moves are kept.  The
    perturbation size is adaptive (up to a few percent of the source
    mass) which enables the search to escape plateaus.

All heavy work (convolution, FFT sizing) is performed once and reused.
The routine respects the requested 1000 s budget and returns the best
sequence found (as plain Python floats).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (full length 2·len(a)-1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true objective (sum is kept = 1)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    The step size decays geometrically.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    step = 0.5               # initial step size on the simplex
    decay = 0.9999           # very slow decay → many tiny steps

    # keep the FFT length fixed for speed
    while time.time() < deadline:
        # full convolution and index of the current maximum
        conv = _conv_self(a, L=L_fft)
        k_max = int(conv.argmax())
        cur_score = 2.0 * n * conv[k_max]          # denominator = 1 (sum(a)==1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it score={cur_score:.6f}")

        # sub‑gradient of conv[k_max] w.r.t. a (see analysis)
        grad = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        # each contributing pair (i,j) adds a_j to grad_i and a_i to grad_j
        for i in range(i_low, i_high + 1):
            j = k_max - i
            grad[i] += a[j]
            grad[j] += a[i]

        # the objective contains the factor 2·n (sum(a)=1)
        grad_obj = 2.0 * n * grad

        # simple (negative) gradient step + projection
        a = a - step * grad_obj
        a = _project_simplex(a)

        # decay step size
        step *= decay

        # occasional random jitter to avoid stagnation
        if rng.random() < 0.001:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb using contribution weights
# ----------------------------------------------------------------------
def _targeted_hill_climb(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Random mass transfers guided by the coefficients that dominate the
    current maximal convolution entry.
    Only improving moves are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a, L=L_fft)
        k_max = int(conv.argmax())

        # compute contribution of each coordinate to conv[k_max]
        contrib = np.zeros(n)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        for i in range(i_low, i_high + 1):
            j = k_max - i
            c = a[i] * a[j]
            contrib[i] += c
            contrib[j] += c          # both sides (double count harmless)

        if contrib.sum() == 0:
            src = rng.integers(n)
        else:
            prob = contrib / contrib.sum()
            src = rng.choice(n, p=prob)

        if a[src] < 1e-12:
            continue

        # choose a destination with relatively small contribution
        low_mask = contrib < contrib[src]
        if low_mask.any():
            dst = rng.choice(np.nonzero(low_mask)[0])
        else:
            dst = rng.integers(n)

        # transfer up to a few percent of the source mass
        eps = a[src] * rng.random() * 0.02   # ≤2 % of src
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine respects the 1000 s budget and returns the best
    sequence it discovers (as a plain list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # problem size – a power of two is convenient for the FFTs
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline: uniform (sum = 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣   Parametric sweep + a few random Dirichlet draws
    # ------------------------------------------------------------------
    # geometric / power shapes
    idx = np.arange(1, n + 1, dtype=np.float64)        # i+1
    for p in (0.5, 1.0, 1.5, 2.0):
        shape = np.power(idx, -p)
        shape /= shape.sum()
        sc = _eval_fast(shape, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, shape.copy()
            print(f"[{time.time() - start_time:.2f}s] power p={p:.1f} score={sc:.6f}")

    r_vals = (0.96, 0.98, 0.99)
    for r in r_vals:
        shape = np.power(r, np.arange(n, dtype=np.float64))
        shape /= shape.sum()
        sc = _eval_fast(shape, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, shape.copy()
            print(f"[{time.time() - start_time:.2f}s] geo r={r:.2f} score={sc:.6f}")

    # a handful of random Dirichlet vectors (focus on dense vectors)
    for _ in range(8):
        if time.time() >= deadline - 300:
            break
        rnd = np.random.dirichlet(np.ones(n))
        sc = _eval_fast(rnd, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, rnd.copy()
            print(f"[{time.time() - start_time:.2f}s] random Dirichlet score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣   Sub‑gradient descent (several short restarts)
    # ------------------------------------------------------------------
    remaining = deadline - start_time
    # allocate roughly half the remaining time to descent, split among up to 4 restarts
    descent_budget = min(500.0, remaining * 0.55)
    n_restarts = 3
    per_restart = descent_budget / n_restarts

    for r in range(n_restarts):
        if time.time() >= deadline - 120:
            break
        # perturb the current best a bit to obtain a fresh start
        a0 = best_seq + 1e-3 * np.random.standard_normal(n)
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        cur_deadline = min(deadline - 20, start_time + (r + 1) * per_restart)
        a_opt, sc_opt = _subgradient_descent(a0, n, cur_deadline, start_time, best_score, L_fft)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad restart {r+1} improved to {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣   Targeted hill‑climb (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _targeted_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
