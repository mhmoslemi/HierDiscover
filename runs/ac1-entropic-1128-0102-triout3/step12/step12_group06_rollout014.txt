def proj_simplex(v):
    # returns sorted v for simplex projection (non-negative)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def hill_climb(a, deadline):
    # uses evaluate via exact_score (fast)
    cur = a.copy()
    cur_score = _exact_score(cur)
    best_seq = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    start_time = time.time() # but we need global start_time; we have.

    while time.time() < deadline:
        it += 1
        # compute temperature
        progress = (time.time() - start_time) / (deadline - start_time)
        T = 1e-2 * (1e-8 / 1e-2) ** progress   # start 1e-2 end 1e-8 maybe

        # choose source
        src = rng.integers(n)
        if cur[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = cur[src] * rng.random() * 0.5   # up to 50%
        if eps <= 0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score
        if delta <= 0 or math.exp(-delta / (T + 1e-12)) > rng.random():
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill climb iter={it:,} best={best_score:.6f}")
    return best_seq, best_score

"""
Small‑scale convex‑mirror optimizer for the step‑function bound

Goal
----
Minimise 2·n·max(conv(a,a)) / (sum(a)**2)  with a_i≥0 .
Because the expression is homogeneous we restrict the vector to the probability
simplex (∑a = 1).  The algorithm consists of three stages:

1.  Smooth‑max Adam mirror descent.
    A log‑sum‑exp surrogate of the max is optimised by an Adam‑style
    multiplicative update in log‑space (exponentiated‑gradient with
    momentum).  The smoothing parameter α follows a geometric schedule
    (0.5 → 5000) so the optimisation gradually focuses on the true maximum.

2.  Hard‑max Adam sub‑gradient descent.
    After the surrogate is tight we switch to the exact max and continue
    Adam updates on the sub‑gradient of the active convolution entry.

3.  Simulated‑annealing hill‑climb.
    Random mass transfers are accepted with a Metropolis rule, allowing the
    search to escape shallow local minima and polish the best candidate.

All phases respect the global 1000 s time budget, keep the vector on the
simplex (via a fast O(n log n) projection) and periodically evaluate the exact
objective using FFT‑based convolution.  The best sequence seen is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helper constants (powers of two)
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Small helper to round up to the next power of two."""
    return 1 << ((x - 1).bit_length())


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Based on Duchi et al., ICML 2008.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _exact_score(a: np.ndarray, L_self: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _self_conv(a: np.ndarray, L_self: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_self)
    return np.fft.irfft(Af * Af, n=L_self)[: 2 * a.shape[0] - 1]


def _gradient_softmax(a: np.ndarray, conv: np.ndarray,
                     alpha: float,
                     L_grad: int) -> np.ndarray:
    """
    Gradient of the smoothed max  (log‑sum‑exp) w.r.t. a.
    conv = a * a (full length 2·n‑1).
    """
    n = a.shape[0]
    # soft‑max weights over the convolution entries
    z = alpha * conv
    max_z = z.max()
    w = np.exp(z - max_z)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= w_sum

    # convolution w ⋆ a_rev (the gradient of a⋅a weighted by w)
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L_grad)
    Revf = np.fft.rfft(a_rev, n=L_grad)
    grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad


def _gradient_hard_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max‑convolution entry.
    Only the index k = argmax(conv) contributes.
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad, k


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    previously reported bound 1.508476.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin against the 1000 s hard limit
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # Choose problem size ------------------------------------------------
    # ------------------------------------------------------------------
    # The earlier best solution uses 4096 entries; we keep the same size
    # because the FFT constants are cheap and the comparison is fair.
    n = 4096
    L_self = _next_pow_two(2 * n - 2)
    L_grad = _next_pow_two(3 * n - 3)

    # ------------------------------------------------------------------
    # Initialise from the previously known best or uniform
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is a global created by the evaluation harness
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        if starter.size != n:
            if starter.size > n:
                starter = starter[:n]
            else:
                starter = np.pad(starter, (0, n - starter.size), mode='constant')
    except Exception:
        starter = np.full(n, 1.0 / n)

    starter = np.maximum(starter, 0.0)
    if starter.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = starter / starter.sum()

    best_a = a.copy()
    best_score = _exact_score(best_a, L_self)
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max Adam mirror descent
    # ------------------------------------------------------------------
    # Allocate roughly 30 % of the total time.
    phase1_deadline = start_time + 0.30 * (deadline - start_time)
    if time.time() < phase1_deadline - 1.0:
        alphas = np.geomspace(0.5, 5000.0, num=60)      # smoother → true max
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        while time.time() < phase1_deadline:
            for alpha in alphas:
                if time.time() >= phase1_deadline:
                    break
                # 1) self‑convolution
                conv = _self_conv(a, L_self)

                # 2) smooth‑max gradient
                grad = _gradient_softmax(a, conv, alpha, L_grad)

                # 3) Adam update in log‑space
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                lr = 0.30 / math.sqrt(t)           # decaying learning rate
                loga = np.log(a + 1e-30)
                loga = loga - lr * m_hat / (np.sqrt(v_hat) + eps)
                a = np.exp(loga)
                a = _proj_simplex(a)

                # occasional exact evaluation
                if t % 5000 == 0:
                    cur = _exact_score(a, L_self)
                    if cur < best_score:
                        best_score = cur
                        best_a = a.copy()
                        print(f"[{time.time() - start_time:.2f}s] "
                              f"phase1 α={alpha:.1f} it={t:,} best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max Adam sub‑gradient descent
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.30 * (deadline - start_time) + 0.30 * (deadline - start_time)
    if time.time() < phase2_deadline - 1.0:
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        while time.time() < phase2_deadline:
            # 1) current convolution and active index
            conv = _self_conv(a, L_self)
            grad, _ = _gradient_hard_max(a, conv)

            # 2) Adam update (log‑space)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = 0.40 / math.sqrt(t)
            loga = np.log(a + 1e-30)
            loga = loga - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.exp(loga)
            a = _proj_simplex(a)

            if t % 2000 == 0:
                cur = _exact_score(a, L_self)
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"phase2 it={t:,} best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        cur = best_a.copy()
        cur_score = _exact_score(cur, L_self)
        best_seq = cur.copy()
        best_seq_score = cur_score
        rng = np.random.default_rng()
        it = 0
        # temperature schedule: high → low (geometric)
        T_start = 5e-2
        T_end = 1e-12
        while time.time() < deadline:
            it += 1
            progress = (time.time() - start_time) / (deadline - start_time)
            T = T_start * (T_end / T_start) ** progress

            src = rng.integers(n)
            if cur[src] < 1e-14:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue

            # move up to 50 % of the source mass (large jumps encourage escape)
            eps = cur[src] * rng.random() * 0.5
            if eps <= 0.0:
                continue

            cand = cur.copy()
            cand[src] -= eps
            cand[dst] += eps
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s

            cand_score = _exact_score(cand, L_self)
            delta = cand_score - cur_score

            # Metropolis acceptance
            if delta <= 0.0 or math.exp(-delta / (T + 1e-20)) > rng.random():
                cur = cand
                cur_score = cand_score
                if cand_score < best_seq_score:
                    best_seq_score = cand_score
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} best={best_seq_score:.6f}")

        # use the best sequence found by hill‑climb
        best_a = best_seq
        best_score = best_seq_score

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
