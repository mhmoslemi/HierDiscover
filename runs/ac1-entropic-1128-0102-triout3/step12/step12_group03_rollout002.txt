"""
Propose a non‑negative step‑function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a multi‑start meta‑heuristic:

1. **Problem size** – choose a length N that fits into the remaining
   time budget (65536 gives the best resolution, smaller N speeds up
   each iteration).

2. **Initial pool** – generate a handful of analytic shapes
   (power‑ramp, exponential decay, sine‑raised) together with a
   few random probability vectors.  The best few candidates (by the
   exact objective) are kept.

3. **Local optimisation** – each candidate is refined by three
   stages that respect a per‑run time budget:

   * *Smooth‑gradient descent* on a soft‑max surrogate of the
     max‑convolution (Adam optimiser, projection onto the simplex).
   * *Sub‑gradient descent* on the true max‑convolution (the exact
     sub‑gradient is cheap to compute).
   * *Short simulated‑annealing* to escape shallow local minima.

   The best result of the three stages is returned.

4. **Multi‑start loop** – as long as time permits new runs are started
   from (i) a random probability vector and (ii) a small perturbation
   of the current global best.  The best sequence over all runs is kept.

5. **Final polish** – the global best is refined once more with a long
   smooth‑gradient run followed by a sub‑gradient run and a final
   annealing pass.

All vectors are kept on the probability simplex (sum = 1), thus the
objective is scale‑invariant.  The code uses only NumPy and the standard
library and obeys the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape = shape / shape.sum()
    return shape


def _make_exponential_shape(n: int, decay: float) -> np.ndarray:
    """Monotonically decreasing exponential, normalised to sum 1."""
    x = np.arange(n, dtype=float)
    shape = np.exp(-decay * x)
    shape = shape / shape.sum()
    return shape


def _make_sine_shape(n: int, power: float) -> np.ndarray:
    """Raised sine on (0,π), normalised."""
    x = np.arange(n, dtype=float) / (n - 1)
    shape = np.sin(np.pi * x) ** power
    shape = shape / shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max surrogate)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.10          # larger base step than the reference
    T0 = 0.2              # start temperature
    T_end = 1e-4

    it = 0
    while time.time() < deadline:
        it += 1

        # current exact objective (cheap, we keep it for logging)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} "
                  f"score={cur_score:.6f}")

        # ----- temperature schedule (linear decay) -----
        frac = it / (it + 1000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # ----- convolution and soft‑max weights -----
        conv = _conv_self(a)
        max_conv = conv.max()
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # ----- gradient via correlation (w * a_rev) -----
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent using the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index of max entry and corresponding index range
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        a = _project_simplex(a)

    return best_a, best_score

# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(a0: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Simple SA that moves a random amount of mass between two positions."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of *potential* steps – we break early when out of time.
    max_steps = 200_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)                # pick source proportionally
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25      # move at most 25 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # projection is not needed – the move preserves sum and positivity

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Single optimisation run (smooth → sub‑grad → anneal)
# ----------------------------------------------------------------------
def _optimise_one(a0: np.ndarray,
                  run_deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """Run the three optimisation phases within the given sub‑budget."""
    # allocate 60 % to smooth descent, 30 % to sub‑grad, rest to anneal
    now = time.time()
    smooth_dead = now + 0.60 * (run_deadline - now)
    a_s, sc_s = _smooth_gradient_descent(a0, smooth_dead, start_time)

    now = time.time()
    sub_dead = now + 0.30 * (run_deadline - now)
    a_sub, sc_sub = _subgradient_descent(a_s, sub_dead, start_time)

    # any remaining time go to a short anneal
    now = time.time()
    if now < run_deadline:
        a_final, sc_final = _anneal(a_sub, run_deadline, start_time)
    else:
        a_final, sc_final = a_sub, sc_sub

    # choose the best among the three checkpoints
    candidates = [(sc_s, a_s), (sc_sub, a_sub), (sc_final, a_final)]
    best_sc, best_a = min(candidates, key=lambda x: x[0])
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under a 1000 s budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a working length N (larger N gives finer granularity)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 700:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    print(f"[{time.time() - start_time:.2f}s] Using length N = {N}")

    # ------------------------------------------------------------------
    # 1) Build an initial pool of candidates (analytic + random)
    # ------------------------------------------------------------------
    top_k = 7
    top = []   # list of (score, vector)

    def _maybe_add(vec):
        sc = _objective(vec)
        if len(top) < top_k:
            top.append((sc, vec))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, vec)
            top.sort(key=lambda x: x[0])

    # analytic shapes ----------------------------------------------------
    for p in np.linspace(0.1, 2.5, 9):
        for alpha in np.linspace(0.0, 0.5, 6):
            _maybe_add(_make_power_shape(N, p, alpha))

    for decay in np.linspace(1e-4, 4e-3, 6):
        _maybe_add(_make_exponential_shape(N, decay))

    for pw in np.linspace(0.5, 3.0, 6):
        _maybe_add(_make_sine_shape(N, pw))

    # random draws -------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(80):
        if time.time() >= deadline:
            break
        a = rng.random(N)
        a /= a.sum()
        _maybe_add(a)

    # global starter if present -----------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _maybe_add(starter_up)
    except Exception:
        pass

    # pick current best from the pool
    best_score, best_seq = top[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Multi‑start optimisation loop
    # ------------------------------------------------------------------
    run_id = 0
    while True:
        now = time.time()
        remaining = deadline - now
        if remaining < 8.0:                     # need a tiny chunk for a full run
            break

        # allocate about 6–9 s per run (dynamic)
        run_time = min(9.0, 0.12 * remaining)
        run_deadline = now + run_time

        # initialise: either random or a perturbed copy of the global best
        if run_id % 3 == 0:
            a0 = rng.random(N)
            a0 /= a0.sum()
        else:
            # small Gaussian perturbation of the current best
            perturb = rng.normal(scale=0.02, size=N)
            a0 = best_seq + perturb
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

        a_opt, sc_opt = _optimise_one(a0, run_deadline, start_time)

        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best (run {run_id}) = {best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 3) Final long‑run polish on the global best
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        final_deadline = time.time() + remaining
        best_seq, best_score = _optimise_one(best_seq, final_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] Final polished score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
