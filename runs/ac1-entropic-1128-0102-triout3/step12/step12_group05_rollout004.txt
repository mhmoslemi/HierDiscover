"""
Propose a non‑negative step‑function that (approximately) minimises

        2·n·max(conv(a,a)) / (sum(a)²)

where conv(a,a) is the linear self‑convolution.  The algorithm works in three
stages:

1️⃣  **Warm‑starts** – a small library of analytically motivated shapes
    (uniform, triangular, power‑law, exponential, Gaussian, arcsine‑like)
    is built and the best member (according to the exact objective) is kept.

2️⃣  **Smooth‑max gradient descent** – the ℓ∞‑norm of the convolution is
    approximated by a log‑sum‑exp with a growing temperature (parameter *α*).
    Its gradient is obtained by an extra FFT‑based convolution.  An
    Adam‑style mirror‑descent on the probability simplex (exponential‑gradient
    update) is performed while the remaining time permits.

3️⃣  **Light simulated‑annealing polish** – in the final seconds a cheap
    Metropolis walk moves a random packet of mass between two indices.
    The best solution obtained throughout the run is finally normalised and
    returned as a plain Python list.

All helper functions are top‑level so they can be pickled.  The total running
time never exceeds the supplied budget (default 1000 s).  The returned list
contains non‑negative floats and will be fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    For a normalised sequence (sum=1) this equals 2·n·max(conv).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑start shapes
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    """Symmetric linear tent."""
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    """Central‑peaked shape: (triangle)^p."""
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # in [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape: (1‑triangle)^p."""
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # in [0,1]
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    """Exponential decay from the centre."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred at the middle."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _shape_arcsine_power(n: int, p: float) -> np.ndarray:
    """Generalised arcsine shape with exponent p (p=0.5 reproduces the basic one)."""
    i = np.arange(n, dtype=float)
    a = 1.0 / ((i + 0.5) * (n - 0.5 - i)) ** p
    a += 1e-16
    return a / a.sum()


def _collect_warm_starts(n: int) -> list[np.ndarray]:
    """Build a modest library of diverse initial vectors."""
    cand = []

    # basic shapes
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    # power‑law centre / edge families
    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    # arcsine with different exponents
    for p in (0.3, 0.5, 0.7, 1.0):
        cand.append(_shape_arcsine_power(n, p))

    # exponentials
    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    # Gaussians
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # mixed centre/edge blends
    for lc in (0.25, 0.5, 0.75):
        centre = _shape_power_center(n, 2.0)
        edge = _shape_power_edge(n, 2.0)
        a = lc * centre + (1 - lc) * edge
        cand.append(a / a.sum())

    # If the harness provides a prior solution, add it
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # clean NaNs / duplicates (very cheap)
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Smooth‑max gradient utilities
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed maximum
        M_α(a) = (1/α)·log( Σ_k exp(α·(a*a)_k) )
    where (a*a)_k is the self‑convolution.
    Returns a vector of length n (same shape as a).
    """
    n = a.shape[0]

    # self‑convolution
    conv = _conv_self(a)                      # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights
    w = np.exp(alpha * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2·(w * a_rev)  (central slice)
    a_rev = a[::-1]
    L = _next_pow2(w.size + a_rev.size - 1)
    Wf = np.fft.rfft(w, n=L)
    Af = np.fft.rfft(a_rev, n=L)
    grad_full = np.fft.irfft(Wf * Af, n=L)[: w.size + a_rev.size - 1]
    grad = 2.0 * grad_full[n - 1: n - 1 + n]   # take the middle n entries
    return grad


# ----------------------------------------------------------------------
# Optimisation core (smooth‑max + exponential‑gradient)
# ----------------------------------------------------------------------
def _optimize_smooth_max(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         eta0: float = 1.0,
                         beta: float = 0.9,
                         alpha0: float = 5.0,
                         alpha_max: float = 200.0,
                         alpha_step: int = 500) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a smooth‑max approximation.
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    iteration = 0
    alpha = alpha0

    while time.time() < deadline:
        iteration += 1

        # gradient of the smoothed max
        grad = _smooth_gradient(a, alpha)

        # momentum (exponential‑moving‑average)
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step‑size
        step = eta0 / math.sqrt(iteration)

        # exponential‑gradient update (keeps positivity)
        a = a * np.exp(-step * momentum)
        # re‑normalise to the simplex
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        # evaluate
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Iter={iteration:,} score={cur_score:.6f}")

        # slowly increase smoothing (focus more on the true max)
        if iteration % alpha_step == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.2, alpha_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light annealing polish (same idea as the previous version, but a bit
# more aggressive in the final few seconds)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float,
            steps: int = 80_000) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk that swaps a random packet of mass between two positions."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # a very simple linear cooling schedule
        T = 0.03 * (1e-6 / 0.03) ** (it / steps)

        src = rng.choice(n, p=a)          # source proportional to current mass
        dst = rng.integers(n)             # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30   # move up‑to 30 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats that sum to one.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 1) Gather a palette of warm‑start candidates.
    # --------------------------------------------------------------
    candidates = _collect_warm_starts(n)

    # --------------------------------------------------------------
    # 2) Pick the best warm‑start.
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates, 1):
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] Warm‑start {idx}/{len(candidates)} score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # --------------------------------------------------------------
    # 3) Main optimisation – smooth‑max exponential‑gradient descent.
    #    Allocate the bulk of the remaining budget to this phase.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                               # keep a few seconds for polishing
        sub_deadline = time.time() + remaining * 0.85   # ~85 % of what is left
        refined_seq, refined_score = _optimize_smooth_max(
            best_seq,
            sub_deadline,
            start_time,
            eta0=1.0,
            beta=0.9,
            alpha0=5.0,
            alpha_max=200.0,
            alpha_step=400
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 4) Light annealing polish (if time permits).
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={anneal_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation & conversion to plain list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
