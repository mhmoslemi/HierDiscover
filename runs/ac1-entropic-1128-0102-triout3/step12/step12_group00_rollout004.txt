def _adam_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active set
        active = np.where(conv >= max_val - 1e-12)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k-(n-1))
            hi = min(n-1, k)
            idx = np.arange(lo, hi+1, dtype=int)
            grad[idx] += 2*a[k - idx]
        grad /= float(len(active))

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

        # periodic evaluation
        if t % 200 == 0:
            cur_score = 2.0*n*max_val/(a.sum()**2)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] Adam iter={t:,} score={cur_score:.6f}")
    return best_a, best_score

def _anneal(seq, deadline, start_time):
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 20000
    for it in range(1, max_steps+1):
        if time.time() >= deadline:
            break
        # random move: choose src with probability proportional to a
        src = rng.choice(n, p=a)
        dst = rng.integers(n)

            continue

        eps = a[src] * rng.random() * 0.2

            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

"""
A fast multi‑stage optimiser for the step‑function evaluation problem.

The algorithm works in four stages:

1. **Coarse grid + random draws** – generate a handful of symmetric
   “power‑ramp + plateau’’ candidates on a modest size (n=4096),
   keep the best few.

2. **Projected Adam descent** – starting from each elite candidate,
   perform Adam optimisation on the simplex, where the gradient is the
   sub‑gradient of the *active* convolution entry (the one attaining the
   maximum).  The update is projected back onto the probability simplex.
   This converges much faster than plain sub‑gradient descent.

3. **Hierarchical refinement** – up‑sample the currently best vector to a
   larger resolution (n=16384) by linear interpolation and run a short
   Adam phase again.  The larger discretisation reduces the rounding error
   of the continuous optimum.

4. **Light simulated‑annealing polish** – cheap Metropolis moves that
   shift a small packet of mass between two indices are accepted if they
   improve (or sufficiently almost improve) the objective.  The temperature
   is decayed exponentially.

All helper functions are defined at top level so they can be pickled if the
harness runs the solution in a subprocess.  The routine respects the 1000 s
budget, prints progress, and finally returns a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)²)   (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical safety net
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generator – symmetric power ramps with optional flat plateau
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape of length n built from two power ramps that meet a
    central plateau of height 1. The vector is normalised to sum = 1.
    Parameters
    ----------
    p      – exponent of the side ramps (p>0).
    alpha  – relative plateau width ∈[0,0.5]; 0 → pure tent,
             0.5 → uniform.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                     # values rise from 0…1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # adjust possible off‑by‑one caused by rounding
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (sub‑gradient of the max‑convolution)
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the simplex.
    The gradient is the Clarke sub‑gradient of the active (maximum)
    self‑convolution entry.
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1

        conv = _conv_self(a)
        max_val = conv.max()

        # locate all almost‑active indices (numerical tolerance)
        active = np.where(conv >= max_val - 1e-12)[0]

        # sub‑gradient of the max entry (average over the active set)
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(active))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_simplex(a)

        # occasional progress report
        if t % 200 == 0:
            cur_score = 2.0 * n * max_val / (a.sum() ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} "
                      f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (local mass transfers)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Move tiny packets of mass between random positions; Metropolis acceptance.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)

    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of attempted moves – adapt to remaining time
    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # a simple exponential cooling schedule
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 20 %) of the source mass
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
