"""
Random‑sparse optimizer for the step‑function attack.

The idea:
*   The objective 2·n·max(conv(a,a))/sum(a)² is scale‑invariant, so we
    enforce sum(a)=1 (simplex projection) throughout.
*   A uniformly random sparse vector (few non‑zero entries of equal weight)
    has very few overlapping pairs for any shift.  The expected peak of the
    convolution is ≈ 1/(2·n), which yields an objective close to 1 – well
    below the target 1.503.
*   We therefore spend most of the time generating random sparse candidates
    with different sparsities, keep the best one and finish with a short
    sub‑gradient polish (the same routine used in the reference solution).

The algorithm works within the 1000 s budget, adapts the problem size N
according to the remaining time and finally returns a plain list of
non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq))/sum(seq)²."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:               # safety – should never happen
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Simple sub‑gradient descent (same as reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Sub‑grad refine iter={it:,} "
                  f"score={cur_score:.6f}")

        # index where the max occurs
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑conv term
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Random‑sparse generator
# ----------------------------------------------------------------------
def _random_sparse(N: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a vector of length N with exactly m non‑zero equal entries,
    each equal to 1/m (so the sum is 1).  Positions are chosen uniformly
    without replacement.
    """
    a = np.zeros(N, dtype=np.float64)
    idx = rng.choice(N, size=m, replace=False)
    a[idx] = 1.0 / m
    return a

# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function within the given time budget.
    The strategy is a massive random‑sparse exploration followed by a
    short sub‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # -------------- choose problem size (larger N → finer resolution) -----
    remaining = deadline - time.time()
    if remaining > 500:
        N = 65536
    elif remaining > 200:
        N = 32768
    else:
        N = 16384

    print(f"[{time.time()-start_time:.2f}s] Using length N = {N}")

    # -------------- baseline: uniform vector (objective = 2) -------------
    best_seq = np.full(N, 1.0 / N, dtype=np.float64)
    best_score = _objective(best_seq)
    print(f"[{time.time()-start_time:.2f}s] Baseline (uniform) score = {best_score:.6f}")

    # -------------- random‑sparse exploration ---------------------------
    # fractions of non‑zero entries we will try; the exact number of
    # non‑zeros (m) is drawn from a log‑uniform distribution between 0.5 % and
    # 20 % of N.
    min_frac = 0.005
    max_frac = 0.20
    trial = 0

    while time.time() < deadline:
        trial += 1
        # log‑uniform draw for the sparsity level
        f = 10 ** rng.uniform(np.log10(min_frac), np.log10(max_frac))
        m = max(2, int(N * f))
        if m > N:
            m = N

        a = _random_sparse(N, m, rng)
        sc = _objective(a)

        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] New best (trial {trial}) "
                  f"score={sc:.6f}  sparsity={m}/{N}")

        # Every ~300 trials perform a tiny sub‑gradient polishing step
        # (only if we still have a few seconds left)
        if trial % 300 == 0 and time.time() < deadline - 5.0:
            refined_seq, refined_sc = _subgradient_descent(best_seq,
                                                          deadline - 1.0,
                                                          start_time)
            if refined_sc < best_score:
                best_score, best_seq = refined_sc, refined_seq.copy()
                print(f"[{time.time()-start_time:.2f}s] Polished best score = {best_score:.6f}")

    # -------------- final sanity: make sure the vector sums to 1 ----------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        # fallback – uniform (should never happen)
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time()-start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
