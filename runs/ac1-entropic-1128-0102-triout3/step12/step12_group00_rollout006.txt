assistantfinal**Solution Overview**

The goal is to produce a non‑negative step‑function (a probability vector) that makes  

```
2·n·max(conv(a,a)) / (sum(a)²)
```  

as small as possible.  
The objective is scale‑invariant, therefore the vector can be kept on the
probability simplex `{x≥0, Σx=1}`.

The previous attempt used a simple sub‑gradient on the *K* largest convolution
entries.  Empirically the improvement stalls because the update does not focus
on the true maximising index and the step‑size rapidly vanishes.

The new algorithm contains three ideas:

1. **Exact sub‑gradient of the max‑convolution** – the gradient of the
   *maximum* entry is known analytically.  Using only the active indices gives a
   true descent direction and removes the need for the “top‑K” approximation.

2. **Projected accelerated gradient on the simplex** – we keep a momentum term
   (Nesterov‑style) and project after every step onto the probability simplex.
   The step‑size follows `η₀ / sqrt(t)`, which is large enough to move the
   solution early on but slowly decays to guarantee convergence.

3. **Multi‑phase search** –  
   * Phase 1 builds a small pool of promising candidates from several analytic
     families (power‑ramps, beta‑shapes, exponentials, random Dirichlet).  
   * Phase 2 runs the refined projected gradient on the best candidate for the
     majority of the time budget.  
   * Phase 3 finishes with a cheap simulated‑annealing polish.

The whole routine respects the 1000 s budget, works entirely with `numpy`,
and returns a plain Python list ready for the evaluation harness.

```python
"""
search.py – minimise 2·n·max(conv(a,a)) / (∑a)².

Algorithm
---------
1. Build a pool of candidate shapes (power‑ramp, beta, exponential, random
   Dirichlet).  Keep the best few.
2. Starting from the best candidate run a projected accelerated sub‑gradient
   descent that uses the *exact* gradient of the maximal convolution entry.
   Momentum (β=0.9) and a step‑size η₀/√t are employed.
   If no improvement is seen for a while the current iterate is perturbed
   slightly to escape a plateau.
3. Light simulated‑annealing polish moves tiny packets of mass between two
   positions and accepts worsening moves with a fast exponential cooling schedule.
4. Normalise onto the simplex and return the list of floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based self‑convolution (real FFT, power‑of‑two length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, no safety checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:               # Numerical safeguard
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_power_ramp(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape built from two power ramps that meet a plateau of height 1.
    The vector is normalised to sum = 1.
    p     – exponent of the side ramps (p>0).
    alpha – relative plateau width ∈[0,0.5]; 0 → pure tent, 0.5 → uniform.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # pad/trim to exact length
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution shape on a regular grid, normalised to sum = 1."""
    x = np.linspace(0.0, 1.0, n)
    # avoid exact 0/1 when a or b < 1
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential(n: int, beta: float) -> np.ndarray:
    """Exponential decay (or growth) shape, normalised to sum = 1."""
    idx = np.arange(n, dtype=float) / (n - 1)
    shape = np.exp(-beta * idx)
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _grad_max(conv: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute ∂/∂a max_k (a*a)[k] .
    For each index k attaining the maximum we add the contribution
    g_i += 2·a_{k‑i} (when the index is valid).  The result is averaged
    over all maximising k.
    """
    n = a.shape[0]
    max_val = conv.max()
    ks = np.where(conv == max_val)[0]
    grad = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    grad /= float(len(ks))
    return grad


# ----------------------------------------------------------------------
# Projected accelerated sub‑gradient descent (using the exact max‑grad)
# ----------------------------------------------------------------------
def _descent_max(a0: np.ndarray,
                 deadline: float,
                 start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform accelerated projected sub‑gradient descent until ``deadline``.
    Returns the best iterate seen (and its objective value).
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9               # momentum factor
    step0 = 0.6              # initial step‑size multiplier
    it = 0
    last_improve = 0

    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Descent iter={it:,} "
                  f"score={cur_score:.6f}")

        # exact max‑gradient
        grad = _grad_max(conv, a)

        # momentum + diminishing step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # project onto the simplex
        a = _project_simplex(a)

        # if we have been stuck for a while, inject a tiny random jitter
        if it - last_improve > 20000:
            eps = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            momentum.fill(0.0)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny packets of mass between random positions; Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 250_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # simple exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)                 # more mass → more likely to move
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25       # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
```
