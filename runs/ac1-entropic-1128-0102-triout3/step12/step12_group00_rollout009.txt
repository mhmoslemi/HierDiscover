"""
Gradient‑descent optimizer for the step‑function evaluation problem.

The algorithm works on a fixed length `n` (default 65536).  It starts from a
previously found solution (global variable ``height_sequence_1``) or from a
simple analytic shape, normalises it to a probability vector and then runs an
entropy‑regularised, soft‑max based gradient descent in the Fourier domain.
The soft‑max temperature anneals (τ≈0.02/√iteration) so that the objective
approaches the true ∞‑norm of the self‑convolution.  After the main phase a
light simulated‑annealing polish is applied.

All updates are projected onto the probability simplex (non‑negative,
summing to 1).  The final sequence is returned as a plain Python list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: smallest power of two ≥ x
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum(a)=1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Wang & Carreira‑Perpiñán projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical safeguard
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Simple analytic family (power ramp + flat plateau)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny packets of mass between random positions; Metropolis rule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")
    return best_a, best_score

    lam = 5e-4                          # entropy regularisation weight

    it = 0
    # reserve a few seconds for the polishing phase
    sub_deadline = deadline - 30.0
    while time.time() < sub_deadline:
        it += 1
        # ---------- forward FFT of the current vector ----------
        Af = np.fft.rfft(a, n=L)

        # ---------- self‑convolution (to obtain the objective) ----------
        conv_full = np.fft.irfft(Af * Af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = conv.max()
        cur_score = 2.0 * n * max_c          # sum(a)=1 → denominator =1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} "
                  f"tau≈{tau0 / math.sqrt(it):.2e} score={cur_score:.6f}")

        # ---------- soft‑max weights ----------
        tau = max(tau_min, tau0 / math.sqrt(it))
        # stabilise the exponentials by subtracting the maximum
        exp_vals = np.exp((conv - max_c) / tau)
        w = exp_vals / exp_vals.sum()        # length 2·n‑1, sums to 1

        # ---------- gradient of the smoothed objective ----------
        Wf = np.fft.rfft(w, n=L)
        # correlation a ⋆ w = ifft(conj(Af) * Wf)
        grad_full = np.fft.ifft(np.conj(Af) * Wf).real
        grad = 2.0 * grad_full[:n]           # 2· Σ_j w_k a_{k‑i}

        # entropy regularisation (helps avoid degenerate spikes)
        grad += lam * (np.log(a + 1e-15) + 1.0)

        # ---------- momentum and update ----------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # ---------- projection onto the simplex ----------
        a = _project_simplex(a)

    # --------------------------------------------------------------
    # 4) Light simulated‑annealing polish (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _anneal(best_a, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After anneal best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to a plain list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score≈{_objective(best_a):.6f}")
    return [float(v) for v in best_a]
