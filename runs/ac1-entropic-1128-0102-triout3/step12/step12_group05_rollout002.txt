"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

using a three‑phase strategy:

1️⃣  Warm‑starts – a collection of analytically‑motivated shapes,
    including uniform, tent, power‑law, exponential, Gaussian,
    arcsine‑like and Beta‑distribution families.  The best of these
    (according to the exact objective) is taken as the initial point.

2️⃣  Smooth‑gradient descent – a continuation method that replaces the
    hard ``max`` by a softmax (log‑sum‑exp) with a temperature parameter
    α.  The gradient of the smoothed objective is obtained analytically
    (via a convolution with the softmax weights).  An Adam optimiser
    updates the coefficients while a Euclidean projection onto the
    probability simplex maintains the non‑negativity and unit‑sum
    constraints.  α is increased linearly during the run so that the
    optimisation gradually focuses on the true maximum.

3️⃣  Light simulated‑annealing polish – a cheap Metropolis walk that moves
    a small packet of mass between two indices and accepts worsening
    moves with a temperature schedule.  This helps escape shallow local
    minima.

The whole routine respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence found as a plain Python
list of floats.
"""

# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _conv_with_kernel(a: np.ndarray, k: np.ndarray) -> np.ndarray:
    """Convolution of a (length n) with an arbitrary kernel k (length m) via FFT."""
    n = a.shape[0]
    m = k.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Kf = np.fft.rfft(k, n=L)
    conv = np.fft.irfft(Af * Kf, n=L)[: n + m - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()

def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # in [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()

def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # in [0,1]
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()

def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()

def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()

def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()

def _shape_beta(n: int, a_param: float, b_param: float) -> np.ndarray:
    """Discretised Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    # Use log‑beta to avoid overflow for extreme parameters
    logdens = (a_param - 1) * np.log(x) + (b_param - 1) * np.log(1 - x)
    dens = np.exp(logdens - np.max(logdens))      # normalise for stability
    dens += 1e-16
    return dens / dens.sum()

def _collect_warm_starts(n: int) -> list[np.ndarray]:
    """Assemble a diverse set of analytically‑motivated initial vectors."""
    cand = []

    # basic families
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    # power‑law families
    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    # exponentials
    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    # Gaussians
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # Beta families (covers a wide range of asymmetric shapes)
    for a_param, b_param in ((0.5, 0.5), (0.5, 2.0), (2.0, 0.5),
                              (1.5, 1.5), (0.8, 0.8), (3.0, 3.0)):
        cand.append(_shape_beta(n, a_param, b_param))

    # mixed centre/edge blend (hand‑crafted)
    centre = _shape_power_center(n, 2.0)
    edge   = _shape_power_edge(n, 2.0)
    for lc in (0.25, 0.5, 0.75):
        a = lc * centre + (1 - lc) * edge
        cand.append(a / a.sum())

    # optional global starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)

                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # filter NaNs and duplicates (very cheap)
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq

# Sub‑gradient (exponential‑gradient) descent – retained for quick
# coarse improvement before the smoother Adam phase.
# ----------------------------------------------------------------------
def _subgrad_exponential(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         tol_factor: float = 0.01,
                         beta: float = 0.9,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        tol = cur_max * tol_factor
        top_idx = np.where(conv >= cur_max - tol)[0]
        if top_idx.size == 0:
            top_idx = np.array([conv.argmax()])

        # compute sub‑gradient (average over active indices)
        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(top_idx.size)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = eta0 / math.sqrt(it)

        # exponential (mirror) update on simplex
        a = a * np.exp(-step * momentum)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

# ----------------------------------------------------------------------
# Adam optimiser on a smoothed version of the max (soft‑max)
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 alpha_start: float = 5.0,
                 alpha_end: float = 50.0,
                 lr: float = 0.08,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the objective 2·n·max(conv)/sum², where max is approximated by a
    soft‑max with temperature α that grows linearly from α_start to α_end.
    The iterate stays on the probability simplex via Euclidean projection.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # linear continuation for α
        elapsed = time.time() - start_time
        total_budget = deadline - start_time
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / total_budget)

        # compute exact convolution (needed for soft‑max weights)
        conv = _conv_self(a)
        max_conv = conv.max()

        # soft‑max weights (smooth approximation of the max)
        shifted = conv - max_conv               # for numerical stability
        exp_term = np.exp(alpha * shifted)
        w = exp_term / exp_term.sum()           # shape (2n‑1,)

        # gradient of the smoothed max w.r.t. a:
        # grad_s = 2 * sum_i a_i * w_{i + j}
        w_rev = w[::-1]
        conv_rev = _conv_with_kernel(a, w_rev)   # length 3n‑2
        idx = 2 * n - 2 - np.arange(n)          # positions that give Σ_i a_i w_{i+j}
        grad_s = 2.0 * conv_rev[idx]            # shape (n,)

        # gradient of the (unsmoothed) objective for fixed sum=1:
        grad = 2.0 * n * grad_s                 # scaling by 2·n

        # Adam update (on the raw vector, projection later)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_simplex(a)

        # evaluate true objective (hard max) for bookkeeping
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} α={alpha:.1f} score={cur_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # source proportional to mass
        dst = rng.integers(n)              # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move ≤25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

            sub_deadline,
            start_time,
            tol_factor=0.01,
            beta=0.9,
            eta0=0.5
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Main Adam optimisation on smoothed max (majority of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        adam_deadline = time.time() + remaining * 0.78
        adam_seq, adam_score = _adam_smooth(
            best_seq,
            adam_deadline,
            start_time,
            alpha_start=5.0,
            alpha_end=60.0,
            lr=0.08,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8
        )
        if adam_score < best_score:
            best_score = adam_score
            best_seq = adam_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam refined score={adam_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Light annealing polish (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation & conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
