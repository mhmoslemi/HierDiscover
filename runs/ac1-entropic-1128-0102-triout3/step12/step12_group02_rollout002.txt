assistantfinal**Solution Overview**

The evaluation function is  

\[
F(a)=\frac{2\,n\; \max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the linear self‑convolution.  
Because the objective is homogeneous we may impose the normalisation  

\[
\sum_i a_i = 1,\qquad a_i\ge 0 .
\]

With this normalisation  

\[
F(a)=2n\;\max (a*a) .
\]

Hence the problem is: *choose a non‑negative vector on the simplex that minimises the maximum entry of its self‑convolution*.

The code below searches for such a vector in three stages:

1. **Initialisation** – start from the globally supplied solution (if any), from the uniform vector and from a few random Dirichlet draws.  
2. **Exact‑max sub‑gradient descent** – at every iteration the exact index where the convolution attains its maximum is found, the sub‑gradient  
   \[
   \nabla_k F(a)_i = 2\,a_{k-i}\;(0\le k-i<n)
   \]
   is computed analytically (O(n)), and an Adam‑style update on the simplex is performed.  
   The step size slowly decays; occasional tiny random perturbations help to escape plateaux.
3. **Fine‑grained hill‑climbing** – after the descent finishes a simple stochastic hill‑climber transfers a tiny fraction of mass between two coordinates and keeps only improving moves.

All heavy linear algebra uses FFT‑based convolution (`_conv_self`) – fast enough to allow tens of thousands of iterations within the 1000 s budget.

The algorithm is completely self‑contained, uses only `numpy`, and respects the requirement that the returned list contains non‑negative Python floats.

```python
"""
Propose a step‑function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

1️⃣  Initialise from the global starter, the uniform distribution and a
    few random Dirichlet samples (concentrations < 1 → sparse starts).

2️⃣  Run many *exact‑max* sub‑gradient Adam steps.
    In each iteration the convolution is computed by FFT,
    the (unique) index attaining the maximum is identified,
    and the corresponding sub‑gradient (2 · a shifted) is formed.
    The iterate is projected onto the probability simplex.
    The step size decays with 1/√t and a tiny random jitter is added
    every few thousand steps.

3️⃣  A final hill‑climb moves a small fraction of mass from one coordinate
    to another, accepting only moves that improve the true objective.

The routine respects the 1000 s budget, prints progress and returns the
best sequence found as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient for the exact max‑convolution term
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the sub‑gradient of  max(conv(a,a))  w.r.t. a.
    Returns (grad, conv) where conv = a*a and grad[j] = 2·a[k_max-j]
    for the index k_max where conv attains its maximum.
    """
    n = a.shape[0]
    conv = _conv_self(a)                       # length 2n‑1
    k = int(conv.argmax())                     # index of the maximal entry
    grad = np.zeros_like(a)

    # valid j satisfy 0 ≤ j ≤ n‑1 and 0 ≤ k‑j ≤ n‑1
    j_min = max(0, k - (n - 1))
    j_max = min(n - 1, k)
    if j_min <= j_max:
        js = np.arange(j_min, j_max + 1, dtype=int)
        partners = k - js
        grad[js] = 2.0 * a[partners]

    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected descent using the exact max‑gradient
# ----------------------------------------------------------------------


def _max_adam_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type descent on the exact objective 2·n·max(conv(a,a)).
    The gradient is the exact sub‑gradient at the current maximising index.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # step‑size schedule parameters
    base_step = 0.08
    while time.time() < deadline:
        t += 1
        grad, conv = _max_conv_gradient(a)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] max‑adam it={t:,} "
                f"score={cur_score:.6f}"
            )

        # Adam update (projected onto simplex)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_step / (np.sqrt(v_hat) + eps)    # step decays automatically
        a = a - step * m_hat
        a = _project_simplex(a)

        # occasional tiny jitter to leave flat regions
        if t % 4000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final microscopic hill‑climb (ε≈10⁻³·a_src)
# ----------------------------------------------------------------------


def _final_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers between two coordinates.
    Only improving moves are kept; ε≈10⁻³·a_src.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-18:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-3  # up to 0.1 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
    return best_a, best_score

    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and convert to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
