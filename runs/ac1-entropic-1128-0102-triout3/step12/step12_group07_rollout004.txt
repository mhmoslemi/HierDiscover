"""
Propose a low‑score step function.

The algorithm runs in two phases:

1. **Fast parametric search** – Randomly sample a few simple
   families of symmetric non‑negative vectors (arcsine‑type,
   beta‑type or a mixture with a uniform component).  Each candidate
   is evaluated with a fast FFT‑based convolution, and the best
   candidate so far is kept.

2. **Refinement by simulated‑annealing hill‑climbing** – Starting
   from the best candidate, repeatedly move a small amount of mass
   from a “source” index (which contributes most to the current
   worst autocorrelation entry) to a low‑weight “destination”.
   The convolution is updated analytically in O(n) time, and moves
   are accepted if they improve the peak or with a temperature‑scaled
   probability (simulated annealing) to escape plateaus.  Periodic
   re‑normalisation prevents drift.

The whole search respects the given time budget (≈1000 s) and returns
a list of non‑negative floats (the step heights).  The score is
scale‑invariant, so the returned vector is normalised to sum to 1.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: fast convolution using FFT (returns linear self‑conv)
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(p: np.ndarray) -> np.ndarray:
    """Linear self‑convolution p*p using real FFT (length 2·len(p)‑1)."""
    n = p.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _score(p: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(conv(p,p)) / (sum(p)²).
    The vector is assumed non‑negative.
    """
    n = p.shape[0]
    s = p.sum()
    if s < 0.01:      # protect against division by zero
        return float("inf")
    conv = _conv_self_fft(p)
    return 2.0 * n * conv.max() / (s * s)

    # -----------------------------------------------------------------

        n = int(rng.choice(n_choices))
        x = (np.arange(n, dtype=np.float64) + 0.5) / n      # centre of each bin
        eps = 1e-12

        # randomly pick a family
        family = rng.choice(["arcsine", "beta"])
        if family == "arcsine":
            # w_i ∝ (x·(1‑x)+eps)^(‑α)
            alpha = rng.uniform(0.30, 0.50)   # α≈0.5 is the classical arcsine
            w = (x * (1.0 - x) + eps) ** (-alpha)
        else:  # beta‑type
            a = rng.uniform(-0.55, 0.40)     # negative → heavy tails
            b = rng.uniform(-0.55, 0.40)
            w = (x + eps) ** a * ((1.0 - x) + eps) ** b

        # optional uniform mixing (helps regularisation)
        lam = rng.uniform(0.85, 1.0)          # λ close to 1 keeps mostly the shape
        w = lam * w + (1.0 - lam) * np.ones_like(w)

        # enforce symmetry (the optimal solution is believed to be symmetric)
        w = (w + w[::-1]) * 0.5

        # normalise to a probability vector
        w_sum = w.sum()
        if w_sum == 0:
            continue
        p = w / w_sum

        sc = _score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
            best_n = n
            print(f"[{time.time() - start_time:.2f}s] "
                  f"new param best: score={sc:.6f}, n={n}, family={family}")

    # -----------------------------------------------------------------
    # 3) If param search produced nothing, fall back to a simple start
    # -----------------------------------------------------------------
    if best_p is None:
        n = 4096
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        alpha = 0.38
        eps = 1e-6
        w = (x * (1.0 - x) + eps) ** (-alpha)
        best_p = (w + w[::-1]) * 0.5
        best_p = best_p / best_p.sum()
        best_n = n
        best_score = _score(best_p)
        print(f"[{time.time() - start_time:.2f}s] using generic arcsine fallback, score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 4) Simulated‑annealing hill climbing (remaining time)
    # -----------------------------------------------------------------
    p = best_p.copy()
    n = best_n
    conv = _conv_self_fft(p)                     # current convolution
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max                # because Σp = 1
    best_vec = p.copy()
    iteration = 0
    last_improve_iter = 0

    # schedule for the maximal transferred fraction (starts larger, shrinks)
    def _max_frac(elapsed):
        # linearly decay from 0.20 to 0.01 over the whole budget
        return max(0.01, 0.20 * (1.0 - elapsed / budget_s))

    while time.time() < deadline:
        iteration += 1

        # ---- locate the worst shift and a dominant contributing index ----
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = p[idx] * p[k - idx]
        src = idx[int(np.argmax(contrib))]

        # ---- choose a destination among the smallest entries ----
        # give higher probability to very small entries
        inv = 1.0 - p
        inv_sum = inv.sum()
        if inv_sum <= 0:
            inv = np.ones_like(p)
            inv_sum = inv.sum()
        prob = inv / inv_sum
        dst = rng.choice(n, p=prob)
        if dst == src:
            continue

        # ---- decide transfer magnitude ----
        elapsed = time.time() - start_time
        epsilon = p[src] * rng.random() * _max_frac(elapsed)
        if epsilon <= 0.0:
            continue

        # ---- backup current state (cheap copies) ----
        p_old = p.copy()
        conv_old = conv.copy()
        cur_max_old = cur_max

        # ---- exact convolution update (see derivation above) ----
        conv[src:src + n] -= 2.0 * epsilon * p
        conv[dst:dst + n] += 2.0 * epsilon * p
        conv[2 * src] += epsilon * epsilon
        conv[2 * dst] += epsilon * epsilon
        conv[src + dst] -= 2.0 * epsilon * epsilon

        # update the probability vector
        p[src] -= epsilon
        p[dst] += epsilon
        if p[src] < 0.0:
            p[src] = 0.0

        # ---- evaluate new maximum autocorrelation ----
        new_max = conv.max()

        # ---- simulated‑annealing acceptance test ----
        accept = False
        if new_max < cur_max - 1e-15:          # strict improvement
            accept = True
        else:
            # temperature decays linearly; scale the acceptance probability
            temp = max(1e-6, 0.02 * (1.0 - elapsed / budget_s))
            prob_accept = math.exp(-(new_max - cur_max) / (temp * cur_max))
            if rng.random() < prob_accept:
                accept = True

        if accept:
            cur_max = new_max
            # record global best if improved
            if 2.0 * n * cur_max < best_score:
                best_score = 2.0 * n * cur_max
                best_vec = p.copy()
                last_improve_iter = iteration
                print(f"[{time.time() - start_time:.2f}s] iter={iteration:,} "
                      f"new best score={best_score:.6f}")
        else:
            # revert to previous state
            p = p_old
            conv = conv_old
            cur_max = cur_max_old

        # ---- occasional re‑normalisation to combat floating‑point drift ----
        if iteration % 4000 == 0:
            total = p.sum()
            if abs(total - 1.0) > 1e-12:
                p /= total
                conv = _conv_self_fft(p)
                cur_max = conv.max()

        # ---- simple stagnation detector – re‑jitter if stuck too long ----
        if iteration - last_improve_iter > 150_000:
            # re‑initialise from the best vector with a tiny random jitter
            jitter = 0.004 * best_vec * (rng.random(best_vec.shape) - 0.5)
            p = np.maximum(best_vec + jitter, 0.0)
            p /= p.sum()
            conv = _conv_self_fft(p)
            cur_max = conv.max()
            last_improve_iter = iteration
            print(f"[{time.time() - start_time:.2f}s] stagnation – re‑jittered (iter={iteration})")

    # -----------------------------------------------------------------
    # 5) Final normalisation and conversion to plain Python list
    # -----------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec = best_vec / total
    else:
        best_vec = np.full(n, 1.0 / n)

    final_score = _score(best_vec)
    print(f"[{time.time() - start_time:.2f}s] finished – final score={final_score:.6f}, n={n}")

    return [float(v) for v in best_vec]
