def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    # Determine n from starter or default.
    global height_sequence_1
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        starter = None

    n = 4096
    if starter is None or starter.size == 0:
        a = np.full(n, 1.0/n)
    else:
        # adjust to desired n
        s = np.maximum(starter, 0.0)
        if s.size > n:
            s = s[:n]
        elif s.size < n:
            s = np.pad(s, (0, n - s.size), mode='constant')
        if s.sum() == 0.0:
            a = np.full(n, 1.0/n)
        else:
            a = s / s.sum()
    # Ensure a sum to 1.
    a = np.maximum(a, 0.0)
    a /= a.sum()

    # Initialize logits y = log(a + epsilon)
    epsilon = 1e-30
    y = np.log(a + epsilon)

    # Precompute some FFT parameters.
    L = 1 << ((2*n - 2).bit_length())
    
    # Helper functions
    def conv_full(x):
        X = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(X * X, n=L)[:2*n-1]
        return conv

    # Adam parameters
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    lr0 = 0.05  # base learning rate for logits
    eps_adam = 1e-8

    best_a = a.copy()
    best_score = _exact_score(best_a)  # compute using same exact function
    # print start
    print(f"[{time.time() - start:.2f}s] Starting optimization, initial score {best_score:.6f}")

    it = 0
    # Phase: Adam gradient descent on max subgradient
    while time.time() < deadline - 5.0:  # leave few seconds for final hill climb
        it += 1
        # compute conv and max index
        conv = conv_full(a)
        max_idx = int(conv.argmax())
        # compute gradient w.r.t a
        grad_a = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)

        m_hat = m / (1 - beta1 ** it)

        a = np.exp(y_shift)

"""
Hybrid optimization for the step‑function objective

The algorithm proceeds in three stages:

1. **Random‐parameter seeding** – generate many Beta‑shaped candidates
   (controlled by support length and shape parameters) and keep the
   best one.  This quickly finds a good region of the search space.

2. **Adam sub‑gradient descent on the true max‑convolution** – keep the
   vector on the probability simplex via a soft‑max (log‑its) parametrisation.
   At each iteration we compute the index that attains the current maximum,
   obtain the sub‑gradient of that max‑function and update the logits with
   Adam.  The step‑size decays as 1/√t.

3. **Aggressive simulated‑annealing hill‑climb** – starting from the
   best vector of stage 2 we perform many random mass‑transfer moves,
   accepting them with a Metropolis rule.  The temperature is lowered
   exponentially to a very small value, allowing a final polishing.

All heavy work (convolution, gradient) is performed with FFTs, so the
method stays well within the 1000 s budget on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers – the length is a power‑of‑two ≥ 2·n‑1 (full linear convolution)
# ----------------------------------------------------------------------
def _fft_length(n: int) -> int:
    """Return the smallest power of two >= 2·n‑1."""
    return 1 << ((2 * n - 2).bit_length())


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full (linear) self‑convolution of a (length n) using FFT, returning an
    array of length 2·n‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _exact_score(a: np.ndarray, L: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    The vector a is assumed to be non‑negative.
    """
    conv = _conv_full(a, L)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parameterised Beta‑shaped candidate generator
# ----------------------------------------------------------------------
def _beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a candidate probability vector of length n.
    The shape is a (truncated) Beta distribution on a random support.
    """
    # choose support length (at least 2 entries)
    support_frac = rng.uniform(0.2, 1.0)          # fraction of n used
    L = max(2, int(support_frac * n))

    # Beta shape parameters (avoid values too close to 0)
    alpha = rng.uniform(0.2, 5.0)
    beta = rng.uniform(0.2, 5.0)

    # discretise the Beta pdf on the support
    idx = np.arange(L, dtype=float) + 0.5          # centre of each bin
    x = idx / L
    # safe evaluation (prevent 0**negative)
    eps = 1e-12
    w = np.power(np.maximum(x, eps), alpha - 1.0) * \
        np.power(np.maximum(1.0 - x, eps), beta - 1.0)

    # place the window somewhere (centre is a good default)
    offset = (n - L) // 2
    cand = np.zeros(n, dtype=float)
    cand[offset:offset + L] = w

    # normalise to a probability distribution
    total = cand.sum()
    if total > 0.0:
        cand /= total
    else:
        cand = np.full(n, 1.0 / n)
    return cand


# ----------------------------------------------------------------------
# Adam‑based sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def _adam_subgrad(a0: np.ndarray, L: int,
                  deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform Adam updates on the logits of the soft‑max representation of a.
    Returns the best vector found and its exact score.
    """
    n = a0.shape[0]
    # initialise logits (avoid log(0))
    eps = 1e-30
    y = np.log(np.maximum(a0, eps))

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eta0 = 0.15               # base learning rate for the logits
    eps_adam = 1e-8

    best_a = a0.copy()
    best_score = _exact_score(best_a, L)

    it = 0
    while time.time() < deadline - 2.0:   # keep a few seconds for final hill‑climb
        it += 1

        # ----- compute current convolution and max index -----
        a = np.exp(y - np.max(y))            # stable soft‑max
        a /= a.sum()
        conv = _conv_full(a, L)
        k = int(conv.argmax())                # index giving the current max

        # ----- sub‑gradient of the max‑convolution w.r.t. a -----
        grad_a = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad_a[idx] = 2.0 * a[k - idx]        # g_i = 2·a_{k‑i}

        # ----- convert gradient to logits (soft‑max Jacobian) -----
        dot = np.dot(a, grad_a)               # scalar = a·grad_a
        grad_y = a * (grad_a - dot)           # ∂f/∂y_i = a_i (g_i - a·g)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = eta0 / math.sqrt(it)            # mild decay
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional exact evaluation -----
        if it % 2000 == 0:
            cur_score = _exact_score(np.exp(y - np.max(y)) / np.exp(y - np.max(y)).sum(), L)
            if cur_score < best_score:
                best_score = cur_score
                best_a = np.exp(y - np.max(y))
                best_a /= best_a.sum()
                print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} improved score={cur_score:.6f}")

    # final evaluation of the current iterate (in case it never hit the % check)
    final_a = np.exp(y - np.max(y))
    final_a /= final_a.sum()
    final_score = _exact_score(final_a, L)
    if final_score < best_score:
        best_score = final_score
        best_a = final_a.copy()
        print(f"[{time.time() - start_time:.2f}s] Adam final improvement score={final_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (larger moves than the previous code)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, L: int,
                deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Metropolis hill‑climbing that transfers up to 50 % of a source mass.
    The temperature is exponentially decayed, ending at a very small value.
    """
    cur = a0.copy()
    cur_score = _exact_score(cur, L)

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = cur.shape[0]

    # temperature schedule
    T_start = 2e-2
    T_end = 1e-12

    it = 0
    while time.time() < deadline:
        it += 1
        # linearised progress for temperature decay
        progress = (time.time() - start_time) / (deadline - start_time)
        T = T_start * (T_end / T_start) ** progress

        # ---- propose a mass transfer ----
        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # up to half of the source mass may be moved
        eps = cur[src] * rng.random() * 0.5
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        # re‑normalise (should already sum to 1, but numerical drift may happen)
        cand_sum = cand.sum()
        if cand_sum > 0.0:
            cand /= cand_sum
        else:
            continue

        cand_score = _exact_score(cand, L)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} new best={best_score:.6f}")

    return best_seq, best_score

    best_a = a0.copy()
    best_score = _exact_score(best_a, L)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # allocate a modest slice of the total time for the exploratory phase
    explore_deadline = start_time + 0.10 * (deadline - start_time)
    while time.time() < explore_deadline:
        cand = _beta_candidate(n, rng)
        cand_score = _exact_score(cand, L)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Beta seed improved score={cand_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam sub‑gradient descent on the exact max
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:   # keep a buffer for the final polish
        phase_deadline = deadline - 15.0
        best_a, best_score = _adam_subgrad(best_a, L, phase_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After Adam best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – aggressive hill‑climbing (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _hill_climb(best_a, L, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
