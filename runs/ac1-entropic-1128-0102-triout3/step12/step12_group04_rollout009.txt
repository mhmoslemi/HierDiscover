def max_gradient(a, k):
    n = a.shape[0]
    grad = np.zeros_like(a)
    # for i from max(0, k-(n-1)) to min(n-1, k):
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i_range = np.arange(lo, hi+1)
    j = k - i_range
    # For positions where i != j
    mask = i_range != j
    grad[i_range[mask]] = 2 * a[j[mask]]
    grad[i_range[~mask]] = a[i_range[~mask]]  # i=j case
    # rest remain zero
    return grad
```

"""
A three‑phase optimiser for the step‑function problem.

Phase 1 – Warm‑start: a handful of analytically simple shapes
          (tent, cosine, half‑circle, Gaussian, power‑tent) together
          with a few random Dirichlet samples are normalised and the
          best is kept.

Phase 2 – Sub‑gradient descent on the exact maximum of the
          self‑convolution.  For the current vector a the index k
          where (a*a) attains its maximum is found (via one FFT).
          The sub‑gradient of this max is g_i = 2·a_{k−i} (zero when
          k−i is out of range).  A diminishing step size
          η_t = η₀/√t is used and after each move a is projected onto
          the probability simplex.

Phase 3 – Light simulated‑annealing polish: a Metropolis walk that
          moves a random fraction of the mass from one index to another
          (temperature drops linearly).  The walk uses the fast FFT
          based convolution for objective evaluation.

The routine respects the 1000 s budget and returns a plain list of
non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a normalised (∑a=1) vector.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform simplex point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution entry
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution entry at index k.
    For each i with 0 ≤ k‑i < n we have g_i = 2·a_{k‑i}.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        i_vals = np.arange(lo, hi + 1, dtype=int)
        j_vals = k - i_vals
        grad[i_vals] = 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exact max gradient)
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(conv(a,a)).
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    # initial convolution and best record
    conv = _conv_self(a, L)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        it += 1

        # index of the current maximum entry
        max_idx = int(conv.argmax())

        # sub‑gradient at that index
        grad = _max_gradient(a, max_idx)

        # diminishing step size
        step = step0 / math.sqrt(it)

        # gradient step (moving opposite to the gradient)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        # new convolution & objective
        conv = _conv_self(a_new, L)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter={it:,} "
                  f"score={cur_score:.6f}")

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (more steps than the reference)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that moves a random packet of mass between two indices.
    The temperature drops linearly from init_T to near zero.
    """
    a = seq.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    init_T = 0.5

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = max(1e-8, init_T * (1.0 - elapsed / total))

        src = rng.choice(n, p=a)            # source proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        eps = a[src] * rng.random() * 0.5   # move up to 50 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv.max()
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Warm‑starting shapes
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a small collection of simple analytic candidate vectors."""
    x = np.arange(n, dtype=float) / (n - 1)               # normalised coordinate in [0,1]

    # 1) symmetric linear tent
    tent = np.minimum(x, 1.0 - x)

    # 2) half‑cosine (raised cosine) – zero at the ends, smooth interior
    cosine = np.sin(np.pi * x)

    # 3) half‑circle (Archimedes) – √(1‑(2x‑1)²)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

    # 5) power‑tent (flatter than the linear tent)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)

    return normals

def propose_candidate(seed: int = 0,

                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # problem size – power‑of‑two length gives fast FFTs
    n = 4096

    # ------------------------------------------------------------------
    # Phase 1 – Warm‑start
    # ------------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # a few random Dirichlet samples (helps escape systematic bias)
    for _ in range(12):
        rnd = np.random.rand(n)
        rnd = rnd / rnd.sum()
        warm_starts.append(rnd)

    # include the globally supplied best sequence if present
    try:
        gs = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if gs.ndim == 1 and gs.size > 0:
            gs = np.maximum(gs, 0.0)
            if gs.sum() > 0:
                gs = gs / gs.sum()
                warm_starts.append(gs)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Sub‑gradient descent (exact max gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + remaining * 0.70
        a_sub, sub_score = _subgradient_optimize(a0,
                                                 sub_deadline,
                                                 start_time,
                                                 step0=0.15)
        if sub_score < best_score:
            best_score = sub_score
            a0 = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad best score={sub_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Light annealing polish (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_a, anneal_score = _anneal(a0, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            a0 = anneal_a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final best={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & conversion to plain list
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    return [float(v) for v in a0]
