"""
search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages:

1️⃣  A fairly exhaustive parametric sweep over several families
    (geometric × power, pure power‑law, truncated linear, mixtures
    with the uniform distribution) together with a cheap random
    sampling stage.  This provides a strong initial point.

2️⃣  A smooth‑max Adam descent.  We replace the hard “max” by a
    soft‑max (log‑sum‑exp) with an annealed temperature β,
    perform Adam‑style updates and keep the best iterate.

3️⃣  A direct sub‑gradient descent on the *exact* max‑convolution.
    The sub‑gradient of max(conv) can be written explicitly as a
    shifted copy of the current vector, which makes the step
    ultra‑cheap (one FFT for the convolution, O(n) for the gradient).

4️⃣  A final hill‑climb that moves a tiny amount of mass between
    two coordinates while accepting only improving moves.

All stages respect the 1000 s budget.  The routine returns the best
sequence found (as a plain list of Python floats).  The search works
entirely on the probability simplex (sum = 1) – the evaluation function
`evaluate_sequence` is scale‑invariant, so this does not restrict optimality.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (fast, assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / sum(a)^2."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max Adam descent (stage 2)
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, beta: float, L_conv: int, L_corr: int):
    """Gradient of (1/β)·log Σ exp(β·conv) w.r.t. a."""
    conv = _conv_self(a, L=L_conv)               # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                       # stabilize
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()                # soft‑max weights
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)        # linear convolution
    start = a.shape[0] - 1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


def _smooth_descent(a0: np.ndarray, n: int, deadline: float, start_time: float,
                    best_score_initial: float):
    """Adam‑style descent on the smooth‑max surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0         # initialise with a smooth landscape
    beta_mul = 1.015
    beta_max = 400.0
    base_lr = 0.2

    while time.time() < deadline:
        t += 1
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth‑descent it={t:,} "
                f"β={beta:.2f} score={cur_score:.6f}"
            )

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # occasional tiny jitter to escape tiny plateaus
        if t % 4000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Direct sub‑gradient descent on the exact max (stage 3)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score_initial: float):
    """Projected sub‑gradient descent on f(a)=max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.3
    t = 0
    L_fft = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, L_fft)
        max_idx = int(np.argmax(conv))
        max_b = conv[max_idx]
        cur_score = 2.0 * n * max_b / (a.sum() ** 2)   # sum(a)=1 after projection
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad it={t:,} score={cur_score:.6f}"
            )

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)          # inclusive
        if i_start <= i_end:
            idxs = np.arange(i_start, i_end + 1)
            grad[idxs] = 2.0 * a[max_idx - idxs]

        step = step0 / np.sqrt(t)
        a = a - step * grad
        a = _project_simplex(a)

        if t % 5000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (stage 4)
# ----------------------------------------------------------------------


def _final_hill_climb(start_a: np.ndarray, deadline: float,
                      start_time: float, L_fft: int):
    """Microscopic random mass transfers – only accepting improvements."""
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-3       # ≤0.1 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the given
    evaluation function.  The routine respects the 1000 s time budget
    and returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the host process
    deadline = start_time + budget_s - 0.3

    # problem size – larger n gives a finer search space, but also cost
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ---------- baseline ----------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ---------- optional global starter ----------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:

            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ---------- stage 1 : parametric & random sweep ----------
    sweep_deadline = deadline - 300.0   # reserve ~5 min for later stages
    idx = np.arange(1, n + 1, dtype=np.float64)

    # systematic grid over geometric × power families
    r_grid = np.logspace(np.log10(0.85), np.log10(0.99999), num=50)
    p_grid = np.linspace(-5.0, 3.0, num=81)

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= sweep_deadline:
                break
            shape = geo * np.power(idx, p)
            shape_sum = shape.sum()
            if shape_sum == 0.0:
                continue
            shape /= shape_sum

            # blend with uniform to explore intermediate points
            for lam in (0.0, 0.3, 0.6, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] grid r={r:.5f} p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                    )

    # a few extra families (pure power‑law, truncated linear, etc.)
    extra_families = [
        lambda i: (i + 1) ** -2.0,                     # 1/(i+1)^2
        lambda i: np.maximum(0.0, 1.0 - i / (n - 1)),   # triangular decreasing
        lambda i: np.maximum(0.0, (n - i) ** 2),        # quadratic decreasing

    for fam in extra_families:
        shape = fam(np.arange(n, dtype=np.float64))
        if shape.sum() == 0.0:

        shape = shape / shape.sum()
        sc = _eval_fast(shape, L_fft)
        if sc < best_score:

            print(f"[{time.time() - start_time:.2f}s] extra‑family score={sc:.6f}")

    # random sampling of mixed geometric / power families

        shape = (r ** np.arange(n)) * (idx ** p)

        seq = lam * uniform + (1 - lam) * shape
