def _random_hill_climb(a, time_limit):

Minimum‑max‑convolution step‑function search (≈ 1 500 ms budget).

Algorithm
---------
1. **Warm‑start** – generate a battery of analytic shapes
   (tent, half‑cosine, half‑circle, Gaussian, power‑tent with many
   exponents, exponential decays, and their reversals).  The best
   shape w.r.t. the exact objective 2·n·max(conv) is kept.

2. **Projected Adam on the true sub‑gradient** – at each iteration
   the full linear convolution a∗a is computed by a real FFT.
   The index *k* attaining the maximum is located and the exact
   sub‑gradient of max(conv) is
        g_i = 2·a_{k‑i}   (zero outside the overlap range).
   Adam’s adaptive step‑size together with Euclidean projection onto
   the probability simplex (sum = 1, a_i ≥ 0) drives the optimisation.
   The learning‑rate decays after a few thousand iterations.

3. **Greedy stochastic hill‑climbing** – after the Adam phase a cheap
   Metropolis walk is performed: a random packet of mass is moved
   from one index to another; any strictly improving move is kept.
   This polish frequently pushes the objective a few 10⁻⁴ lower.

All heavy work uses NumPy’s FFT (fast for powers‑of‑two).  The routine
automatically respects the 1000 s time limit – if the budget runs out
the current best sequence is returned.

The returned list is normalised to sum = 1, contains only
non‑negative floats and can be passed directly to ``evaluate_sequence``.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – analytic families
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]

    candidates = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    candidates.append(tent)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)                # zero at the ends
    candidates.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    candidates.append(gauss)

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        candidates.append(exp_fwd)
        candidates.append(exp_rev)

    # 7) a very flat shape – constant (uniform) – useful as a baseline
    candidates.append(np.ones_like(x))

    # Normalise each candidate to sum = 1 and keep only the non‑negative ones.
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution entry
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return the sub‑gradient of max(conv) w.r.t. a.
    conv is the full linear self‑convolution (length 2·n‑1).
    """
    n = a.shape[0]
    k = int(conv.argmax())            # index where the max is attained
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]      # g_i = 2·a_{k‑i} on the overlap
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex using the exact sub‑gradient of the max‑convolution.
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    # compute initial objective once (no extra FFT needed)
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # 1) self‑convolution (full length 2·n‑1)
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # 2) sub‑gradient of the current max entry
        grad = _max_subgradient(a, conv)

        # Scale because the true objective is 2·n·max(conv)
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the probability simplex
        a = _project_simplex(a)

        # current objective (no extra convolution needed)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"score={cur_score:.6f}")

        # optional learning‑rate decay (helps in the long run)
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy Metropolis walk that moves a random packet of mass between
    two indices, accepting only improvements.  Very cheap and works
    well as a final polish.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        # pick a source proportionally to its mass – encourages moves
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move at most 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:                     # greedy acceptance
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function returns a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – power of two for FFT speed.
    # ------------------------------------------------------------------
    n = 4096                     # 4 k works well within the time budget

    # ------------------------------------------------------------------
    # 1) Warm‑start – pick the best analytic shape.
    # ------------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # include a globally supplied construction if present
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            # resample to length n (simple linear interpolation)
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_resampled = np.interp(xq, xs, h1)
            if np.all(h1_resampled >= 0):
                h1_resampled = np.maximum(h1_resampled, 0.0)
                s = h1_resampled.sum()
                if s > 0:
                    warm_starts.append(h1_resampled / s)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Projected Adam optimisation with exact sub‑gradient.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + remaining * 0.70   # 70 % of remaining time
        a_opt, score_opt = _adam_optimize(a0, adam_deadline, start_time, lr=0.02)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam best score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Light hill‑climb polish (as much time as left).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _hill_climb(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation & conversion to plain list.
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, length={n}")
    return [float(v) for v in a0]
