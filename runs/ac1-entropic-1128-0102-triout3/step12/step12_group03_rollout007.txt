Propose a non‑negative step function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The search consists of three phases:

1. **Fast exploration on a modest size (≈8 k).**  
   A diverse set of analytic families (central‑peak ramps,
   edge‑peak “U” ramps, cosine bumps, symmetric Beta pdfs) and many
   random Dirichlet draws are generated; the best few candidates are
   kept.

2. **Refinement on the small size** – Adam‑style gradient descent on a
   smooth soft‑max surrogate of the max‑convolution, followed by a few
   sub‑gradient steps.

3. **Upsampling & final polishing** – the best small‑size vector is
   interpolated to the full budget size (≈65 k) and the same optimisation
   stages are repeated, finishing with a short simulated‑annealing
   polish.

All steps respect the 1000 s wall‑time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT (fast for large n)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a, length 2·len(a)‑1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape families
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “tent” shape, low at the edges, high in the centre.
    Normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_u_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """U‑shaped counterpart of _make_shape (high at edges, low centre)."""
    return _make_shape(n, p, alpha)[::-1]


def _make_cosine_shape(n: int, edge: bool = False) -> np.ndarray:
    """Cosine‑based bump (edge=False) or dip (edge=True); normalised."""
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    # centre‑peak: 0 at edges, 1 in the middle
    centre = (np.cos(np.pi * (x - 0.5) * 2) + 1.0) / 2.0
    if edge:
        shape = 1.0 - centre
    else:
        shape = centre
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, a: float) -> np.ndarray:
    """
    Symmetric beta pdf on a regular grid.
    a<1 gives a U‑shape, a>1 a central bump.
    """
    eps = 1e-12
    x = np.linspace(0.0, 1.0, n, endpoint=False) + eps
    mex = x * (1.0 - x) + eps
    shape = mex ** (a - 1.0)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_sparse(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k equally weighted spikes placed at random positions."""
    idx = rng.choice(n, size=k, replace=False)
    shape = np.zeros(n)
    shape[idx] = 1.0 / k
    return shape


# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max surrogate of max)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a smooth surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.05          # base learning rate
    T0 = 0.02             # initial soft‑max temperature
    T_end = 1e-4          # final temperature

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} score={cur_score:.6f}")

        # ----- temperature schedule (exponential decay) -----
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = max(T_end, T0 * (0.5 ** frac))

        # ----- soft‑max weights -----
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # ----- gradient via correlation (w * a_rev) -----
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]  # length 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]               # slice length n

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (same as reference implementation)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # index of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑conv term
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # pick source proportionally to weight
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25    # move up to 25 % of the source weight
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose problem sizes
    # ------------------------------------------------------------------
    # small size for cheap exploration, large size for final polishing
    N_small = 8192
    N_large = 65536

    # ------------------------------------------------------------------
    # 1) Fast exploration on the small size
    # ------------------------------------------------------------------
    top_k = 8
    top = []   # list of (score, vector)

    def _maybe_add(vec: np.ndarray):
        """Insert vector into the top‑k list if it improves the bound."""
        sc = _objective(vec)
        if len(top) < top_k:
            top.append((sc, vec.copy()))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, vec.copy())
            top.sort(key=lambda x: x[0])

    # uniform baseline
    _maybe_add(np.full(N_small, 1.0 / N_small))

    # symmetric tent ramps (central peak)
    for p in np.linspace(0.2, 2.5, 9):
        for alpha in np.linspace(0.0, 0.5, 6):
            _maybe_add(_make_shape(N_small, p, alpha))

    # U‑shaped ramps (edge peak)
    for p in np.linspace(0.2, 2.5, 7):
        for alpha in np.linspace(0.0, 0.5, 6):
            _maybe_add(_make_u_shape(N_small, p, alpha))

    # cosine‑based families
    _maybe_add(_make_cosine_shape(N_small, edge=False))   # central bump
    _maybe_add(_make_cosine_shape(N_small, edge=True))    # edge bump

    # beta‑pdf families (α<1 gives U, α>1 gives bump)
    for a in [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0]:
        _maybe_add(_make_beta_shape(N_small, a))

    # sparse spikes (helps expose low‑overlap patterns)
    rng = np.random.default_rng()
    for k in [4, 8, 16, 32]:
        for _ in range(8):
            _maybe_add(_make_sparse(N_small, k, rng))

    # random Dirichlet draws
    for _ in range(120):
        a = rng.random(N_small)
        a /= a.sum()
        _maybe_add(a)

    # incorporate the global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N_small:
                if N_small % starter.size == 0:
                    factor = N_small // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N_small, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _maybe_add(starter_up)
    except Exception:
        pass

    # pick the current best from the pool
    best_score, best_seq = top[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best (small N) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Refine the best small‑size candidate (smooth GD + sub‑grad)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30:
        # allocate ~70 % of remaining budget to this phase
        sub_deadline = time.time() + 0.7 * time_left
        best_seq, best_score = _smooth_gradient_descent(best_seq, sub_deadline, start_time)

    time_left = deadline - time.time()
    if time_left > 15:
        sub_deadline = time.time() + 0.6 * time_left
        best_seq, best_score = _subgradient_descent(best_seq, sub_deadline, start_time)

    # ------------------------------------------------------------------
    # 3) Upsample to the full size and polish again
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # linear interpolation to the large size
        if best_seq.shape[0] != N_large:
            x_old = np.linspace(0, 1, best_seq.size, endpoint=False)
            x_new = np.linspace(0, 1, N_large, endpoint=False)
            best_seq = np.interp(x_new, x_old, best_seq)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq /= best_seq.sum()

        # smooth GD on large N
        time_left = deadline - time.time()
        if time_left > 40:
            sub_deadline = time.time() + 0.5 * time_left
            best_seq, best_score = _smooth_gradient_descent(best_seq, sub_deadline, start_time)

        # sub‑gradient refinement
        time_left = deadline - time.time()
        if time_left > 20:
            sub_deadline = time.time() + 0.5 * time_left
            best_seq, best_score = _subgradient_descent(best_seq, sub_deadline, start_time)

        # final annealing polish
        if time.time() < deadline:
            best_seq, best_score = _anneal(best_seq, deadline, start_time)

    # ------------------------------------------------------------------
    # 4) Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
