"""
Propose a step‑function (non‑negative list) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation improves on the reference solution by:

1️⃣  A *very* fine 2‑D sweep of the geometric family
    a_i ∝ r^i mixed with the uniform distribution.
    The sweep uses a dense log‑grid for the decay ε=1‑r
    and a step of 0.01 for the mixing weight λ.

2️⃣  A long projected Adam optimiser **with a sharper sub‑gradient**:
    instead of averaging over all active shifts we take the gradient
    of a single maximising shift (the one with the largest convolution
    value).  This gives a stronger descent direction and usually pushes
    the objective lower than the original averaged sub‑gradient.

3️⃣  A final stochastic hill‑climb that moves tiny fractions of mass
    between two positions and accepts only improving moves (the same
    as the reference code).

All operations respect the 1000 s budget, use FFT‑based convolutions
(O(n log n)) and Euclidean projection onto the probability simplex.
The routine returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex { w ≥ 0, Σ w = 1 }.
    Returns w.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (single active shift)
# ----------------------------------------------------------------------
def _subgrad_one_active(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) obtained from **one** active shift:
    the shift with maximal convolution value (ties are broken arbitrarily).
    """
    n = a.shape[0]
    k = int(conv.argmax())                 # a single maximising shift
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]           # ∂/∂a_i of (a ⋆ a)[k]
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score_initial: float,
                   max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser using the *single‑active* sub‑gradient.
    Returns the best sequence seen and its objective value
    (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2                       # relatively aggressive

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1

        # current convolution and score
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # sum(a)=1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # sub‑gradient from a *single* active shift
        grad = _subgrad_one_active(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection back onto the simplex
        a = _project_simplex(a)

        # plateau handling – tiny Gaussian shake
        if it - last_improve > 5_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (micro‑mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb: transfer a tiny random fraction of mass
    from one index to another; keep the move only if it improves the true
    objective.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # random fraction of source mass, up to 1 %
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = float(2.0 * n * _conv_self(best_seq, L_fft).max())
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # Accept an optional global starter if present
    try:
        global height_sequence_1

            if s > 0:

                cur = float(2.0 * n * _conv_self(starter, L_fft).max())
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:

    # ------------------------------------------------------------------

    eps_grid = np.logspace(-4.5, -2.5, 400)   # ≈ 3·10⁻⁵ … 3·10⁻³

            cur = float(2.0 * n * _conv_self(seq, L_fft).max())
            if cur < best_score:

                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Grid eps={eps:.1e} λ={lam:.2f} score={cur:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    hill_time = 150.0

        a_opt, sc_opt = _adam_optimize(best_seq, start_time, adam_deadline, best_score)

            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb
    # ------------------------------------------------------------------

        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
