    packet of mass between two positions.

All helper functions are top‑level (picklable).  The routine respects a
1000 s budget, keeps a safety margin, and returns a plain Python list of
non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of real vectors using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    L = _next_pow2(len(a) + len(b) - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: len(a) + len(b) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a∗a using FFT (linear, not circular)."""
    return _conv_fft(a, a)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes for warm‑starts
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2.0)  # in [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2.0)
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _shape_power_product(n: int, q: float) -> np.ndarray:
    """a_i ∝ ((i+½)*(n‑½‑i))^(‑q)  – generalises the arcsine law."""
    i = np.arange(n, dtype=float)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = denom ** (-q)
    a += 1e-16
    return a / a.sum()


def _gather_warm_starts(n: int) -> list[np.ndarray]:
    """Return a modest library of diverse starting vectors."""
    cand = []

    # classic families
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    # power‑law centre / edge
    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    # exponential / Gaussian
    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # product‑power (generalised arcsine) – sweep a few exponents
    for q in (0.30, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70):
        cand.append(_shape_power_product(n, q))

    # a few random Dirichlet draws for extra diversity
    rng = np.random.default_rng()
    for _ in range(6):
        a = rng.random(n) + 1e-8
        cand.append(a / a.sum())

    # optional global starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # clean up degenerate entries
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Soft‑max‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Return (grad, conv) where grad is ∇Fτ(a) and conv = a∗a.
    Fτ(a) = (1/τ)·log Σk exp(τ·conv[k]).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2·n‑1
    max_c = conv.max()
    # soft‑max weights (stable by subtracting the max)
    w = np.exp(tau * (conv - max_c))
    w /= w.sum()                             # size 2·n‑1, sums to 1

    # gradient of weighted sum: 2·∑_j a_j·w_{i+j}
    w_rev = w[::-1]
    full = _conv_fft(a, w_rev)               # length 3·n‑2
    grad = 2.0 * full[2 * n - 2 - np.arange(n)]
    return grad, conv


def _run_softmax_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) with momentum on the simplex,
    using the smooth approximation Fτ.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # hyper‑parameters
    beta = 0.9               # momentum coefficient
    eta0 = 0.8               # initial step size (scaled by 1/√t)
    tau = 5.0                # start temperature
    tau_max = 500.0
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, conv = _softmax_gradient(a, tau)

        # momentum update (Adam‑style first moment)
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = eta0 / math.sqrt(it)

        # exponentiated‑gradient (mirror descent on simplex)
        a = a * np.exp(-step * momentum)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)

        # current objective (exact, not the smoothed one)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Soft‑max GD iter={it:,} "
                  f"score={cur_score:.6f} tau={tau:.2f}")

        # slowly raise the temperature so the approximation sharpens
        if tau < tau_max:
            tau *= 1.001   # geometric increase

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap Metropolis walk that transfers a random packet of mass
    between two positions.  Accepts downhill moves always and uphill
    moves with probability exp(‑Δ/ T).  The temperature cools linearly.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # schedule: 30 000 proposals (or fewer if time runs out)
    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling (very low temperatures)
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # pick a source biased by mass
        dst = rng.integers(n)             # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25  # move ≤25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # --------------------------------------------------------------
    n = 4096                     # 2¹² is a good trade‑off for speed/resolution

    # --------------------------------------------------------------
    # Warm‑starts
    # --------------------------------------------------------------
    candidates = _gather_warm_starts(n)

    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates, 1):
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] Warm‑start {idx}/{len(candidates)} "
              f"score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # --------------------------------------------------------------
    # Soft‑max gradient descent (major part of the budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                     # keep a few seconds for polishing
        sub_deadline = time.time() + remaining * 0.85
        refined_seq, refined_score = _run_softmax_descent(
            best_seq,
            sub_deadline,
            start_time,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # Light simulated‑annealing polish (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve "
                  f"score={anneal_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation & conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
