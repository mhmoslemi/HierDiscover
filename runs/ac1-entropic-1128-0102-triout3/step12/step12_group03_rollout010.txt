"""
Propose a step‑function (non‑negative heights) with a low value of

    2·n·max(conv(a,a)) / (sum(a)**2).

The search works in several phases and stays well within the 1000 s budget.

1. **Coarse stage (small problem size).**
   * Build many candidate vectors of length `N_small` (random Dirichlet draws,
     beta‑distribution shapes, a down‑sampled global starter if present).
   * Keep the few best ones.
   * For each of them run a short *alternating linear‑program / mixing*
     routine: given `f` compute `M = max(conv(f,f))` and solve the LP
     `max ∑ g_i`  s.t. `conv(f,g) ≤ M`.  Mix the solution back into `f`
     and repeat a few times.  This quickly drives the convolution towards a
     flatter profile.

2. **Upsampling.**
   The best coarse vector is interpolated to the target length
   `N_large` (default 65536).

3. **Fine optimisation (large problem).**
   * A smooth approximation of the max‑convolution (soft‑max) is minimised
     with Adam‑style adaptive steps.  The temperature of the soft‑max is
     annealed exponentially to focus the optimisation on the true maximum.
   * A sub‑gradient descent (as in the reference implementation) further
     refines the solution.

4. **Polishing.**
   Light simulated‑annealing moves are applied to escape possible stagnation.

All updates are projected onto the probability simplex, guaranteeing
non‑negativity and a unit sum (the objective is scale‑invariant).  The
procedure reports intermediate scores and finally returns a plain list of
floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators (small problem size)
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shaped probability vector (α,β>0)."""
    x = (np.arange(n, dtype=float) + 0.5) / n          # avoid exact 0/1
    shape = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    shape = np.maximum(shape, 1e-12)                  # avoid exact zeros
    shape /= shape.sum()
    return shape


def _ramp_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric “tent” shape, used only for the coarse stage."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max surrogate of max)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a soft‑max approximation of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.08          # base learning rate
    T0 = 0.02             # initial soft‑max temperature
    T_end = 1e-7          # final temperature (very sharp)

    it = 0
    while time.time() < deadline:
        it += 1

        # current exact score (for logging)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} "
                  f"score={cur_score:.6f}")

        # ----- temperature schedule (exponential decay) -----
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T0 * (0.1 ** frac))   # exponential cooling

        # ----- soft‑max over the convolution entries -----
        conv = _conv_self(a)
        max_c = conv.max()
        w_raw = np.exp((conv - max_c) / T)
        w = w_raw / w_raw.sum()

        # ----- gradient via correlation (w * a_rev) -----
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]  # length n

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex -----
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (max‑convolution only)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑conv term
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 250_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # move mass from a big entry
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Alternating linear‑program optimisation (coarse stage)
# ----------------------------------------------------------------------
def _lp_mixing_step(f: np.ndarray, t: float = 0.5) -> np.ndarray:
    """One iteration of the LP‑mixing routine described in the statement."""
    n = f.shape[0]
    # maximal self‑convolution entry (with sum(f)=1 it equals objective/(2n))
    max_conv = _conv_self(f).max()
    M = max_conv

    rows = 2 * n - 1
    # Build A (rows × n) where (A @ g)[k] = (f * g)[k]
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        A[k, j_vals] = f[i_vals]

    # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n)
    b = np.full(rows, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'disp': False})
        g = res.x if res.success else np.zeros_like(f)
    except Exception:
        g = np.zeros_like(f)

    # Normalise g to a probability vector (if possible)
    s = g.sum()
    if s > 0:
        g = g / s
    else:
        g = np.full_like(f, 1.0 / n)

    # Convex combination and projection
    f_new = (1.0 - t) * f + t * g
    f_new = _project_simplex(f_new)
    return f_new
