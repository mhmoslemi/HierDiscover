"""Greedy shape generation + aggressive subgradient + simulated‑annealing.

The search works in three phases:
1.  Build a pool of analytic and random candidates (uniform, linear,
    inverse‑product and several power‑law families).  The best few are
    kept.
2.  For each kept candidate run a pure sub‑gradient descent that
    minimises the max‑convolution directly.  The run time is split
    proportionally to the remaining budget.
3.  A fast simulated‑annealing loop moves tiny amounts of mass from
    indices that currently contribute most to the peak to random other
    indices.  The loop stops when the overall time budget is exhausted.
The final vector is projected onto the probability simplex after every
update so the scale is irrelevant; the objective is 2·n·max(conv)/(sum²).
"""
import time, math, numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ------------------ analytic shapes ---------------------------------
def _shape_uniform(N):
    return np.full(N, 1.0 / N)

def _shape_linear_decreasing(N):
    w = np.arange(N, 0, -1, dtype=float)
    return w / w.sum()

def _shape_power_decreasing(N, p):
    idx = np.arange(N, dtype=float)
    w = ((N - idx) / N) ** p
    return w / w.sum()

def _shape_power_increasing(N, p):
    idx = np.arange(N, dtype=float)
    w = (idx / (N - 1)) ** p
    return w / w.sum()

def _shape_inverse_product(N, p):
    i = np.arange(N, dtype=float)
    # distance to the two ends, +1 avoids division by zero
    d = (i + 1) * (N - i) + 1.0
    w = d ** (-p)
    return w / w.sum()

def _shape_tent(N, p, plateau_frac):
    """The original “tent‑with‑plateau” used by the reference code."""
    alpha = max(0.0, min(0.5, plateau_frac))
    plateau_len = int(round(alpha * N))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (N - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(N)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < N:
            shape = np.concatenate([shape, np.zeros(N - shape.size)])
        else:
            shape = shape[:N]
    shape = shape / shape.sum()
    return shape

# ------------------ optimisation primitives -------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float, start_time: float):
    """Pure sub‑gradient descent on max(conv) with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    # simple momentum‑less scheme, step size ~ c / sqrt(t)
    step0 = 0.3
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # sub‑gradient of max entry
        # diminishing step
        step = step0 / math.sqrt(t + 1.0)
        a -= step * grad
        a = _project_simplex(a)
        cur = _objective(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
    return best, best_score

def _anneal(seq: np.ndarray, deadline: float, start_time: float):
    """Light simulated‑annealing that moves tiny mass chunks."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        # simple linear cooling from T0 to 0
        T0 = 0.05
        T = T0 * max(0.0, 1.0 - it / 500_000)
        # draw source index with probability proportional to current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.2   # move at most 20 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / (T + 1e-12)) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={best_score:.6f}")
    return best, best_score
