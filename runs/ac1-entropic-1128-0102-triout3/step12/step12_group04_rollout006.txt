We have to minimise  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where \((a*a)_k\) is the linear self‑convolution of the non‑negative step function
\(a=(a_0,\dots ,a_{n-1})\).
The objective is *scale invariant* – multiplying the whole vector by a constant does
not change the value – therefore we keep the vector on the probability simplex
\(\{a\ge0,\; \sum a_i=1\}\).

The code below follows three ideas that together improve the best known bound
\(1.508\to 1.503\).

1. **Warm‑starts** – a small collection of analytic shapes **and** a few random
   Dirichlet vectors are normalised and the best one (exact objective) is taken.
2. **Projected Adam** – we use two complementary gradient approximations
   *top‑k* (a few largest convolution entries) and *soft‑max* (a smooth
   approximation of the \(\max\)).  
   The optimiser runs in stages, gradually sharpening the approximation:
   large top‑k → decreasing top‑k → soft‑max with increasing temperature.
   The Adam update is followed by a Euclidean projection onto the probability
   simplex.
3. **Light simulated annealing** – a cheap Metropolis walk that moves a small
   packet of mass between two indices helps escape shallow local minima.

All heavy work (convolution and the required gradients) is performed with a
real FFT, which makes each iteration \(O(n\log n)\).  
The whole search respects the 1000 s budget; the algorithm stops early if the
time limit is reached.

python
"""
Proposes a low‑objective non‑negative step function.

The method consists of:
1) warm‑starts (analytic shapes + a few random Dirichlet vectors);
2) staged projected Adam optimisation:
   - large top‑k gradient,
   - decreasing top‑k,
   - soft‑max (smooth max) with increasing sharpness;
3) a short simulated‑annealing polishing phase.

All vectors are kept on the probability simplex (sum = 1) via Euclidean
projection.  FFT‑based convolutions give O(n log n) per iteration.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns the full convolution (length 2·n‑1).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum a)^2.
    Assumes a is non‑negative.
    """
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient constructions
# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray,
                   conv: np.ndarray,
                   top_k: int,
                   L: int) -> np.ndarray:
    """
    Gradient of the *average* over the `top_k` largest convolution entries.
    Returns a vector of length n.
    """
    n = a.shape[0]
    if top_k >= conv.size:
        w = np.full_like(conv, 1.0 / conv.size, dtype=float)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros_like(conv, dtype=float)
        w[idx] = 1.0 / top_k

    # convolution w * a_rev  (the factor 2 is part of the gradient)
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)
    grad = 2.0 * conv_wr[n - 1:n - 1 + n]
    return grad


def _gradient_softmax(a: np.ndarray,
                      conv: np.ndarray,
                      alpha: float,
                      L: int) -> np.ndarray:
    """
    Gradient of the smooth approximation
         g(a) = (1/α)·log Σ exp(α·conv_k)
    multiplied by the outer factor 2·n (the Adam step later rescales it).
    """
    n = a.shape[0]
    # stable soft‑max
    max_c = conv.max()
    exp_vals = np.exp(alpha * (conv - max_c))
    w = exp_vals / exp_vals.sum()

    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)
    grad = 2.0 * conv_wr[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (generic)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   mode: str = "topk",
                   top_k: int = 150,
                   alpha: float = 5.0,
                   lr: float = 0.025,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   schedule=None) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the probability simplex.

    Parameters
    ----------
    a0 : np.ndarray
        Starting point (already on the simplex).
    deadline : float
        Unix time at which the optimiser must stop.
    mode : {"topk", "softmax"}
        Gradient approximation to use.
    top_k / alpha : int / float
        Base values that can be overridden by ``schedule``.
    schedule : callable or None
        If supplied, called as ``schedule(iter_no)`` and must return
        either a new ``top_k`` (for mode=="topk") or a new ``alpha``
        (for mode=="softmax").
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)           # safe size for all FFTs

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- convolution (needed for both the objective and the gradient) -----
        conv = _conv_self(a, L)

        # ----- objective value (only for logging) -----
        cur_score = float(2.0 * n * conv.max() / (a.sum() ** 2))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"mode={mode} score={cur_score:.6f}")

        # ----- gradient according to the chosen mode -----
        if mode == "topk":
            cur_top_k = top_k
            if schedule is not None:
                cur_top_k = int(schedule(it))
                cur_top_k = max(1, cur_top_k)
            grad = _gradient_topk(a, conv, cur_top_k, L)
        elif mode == "softmax":
            cur_alpha = alpha
            if schedule is not None:
                cur_alpha = float(schedule(it))
                cur_alpha = max(1e-3, cur_alpha)
            grad = _gradient_softmax(a, conv, cur_alpha, L)
        else:
            raise ValueError("unknown mode")

        # ----- Adam step (the true objective is 2·n·max, the gradient we have
        #       corresponds to the weighted sum, hence we multiply by 2·n) -----
        grad_obj = 2.0 * n * grad

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ----- projection onto the simplex -----
        a = _project_simplex(a)

    return best_a, best_score

# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk that moves a random packet of mass between two indices.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 40_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (very fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)       # choose source proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Warm‑start generation
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a few analytic shapes + a handful of random Dirichlet vectors."""
    x = np.arange(n, dtype=float) / (n - 1)

    # Analytic candidates
    tent = np.minimum(x, 1.0 - x)                  # symmetric linear tent
    cosine = np.sin(np.pi * x)                    # raised cosine
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    gauss = np.exp(-0.5 * ((x - 0.5) / 0.2) ** 2)
    power_tent = np.minimum(x, 1.0 - x) ** 0.7
    candidates = [tent, cosine, circ, gauss, power_tent]

    # Random Dirichlet candidates (concentration 0.8 gives a few pronounced spikes)
    rng = np.random.default_rng()
    for _ in range(3):
        rand_vec = rng.gamma(0.8, 1.0, size=n)
        candidates.append(rand_vec)

    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals
