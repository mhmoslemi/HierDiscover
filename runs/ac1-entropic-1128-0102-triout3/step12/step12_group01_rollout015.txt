def _smooth_optimize(a0, start_time, deadline, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = len(a)
    L_fft = _next_pow2(2*n - 1)
    Lc = _next_pow2(3*n - 2) # for gradient conv size
    # Adam hyperparameters
    beta1 = 0.9; beta2 = 0.999; eps=1e-8; lr = 0.2
    m = np.zeros_like(a); v = np.zeros_like(a)
    tau = 0.5
    tau_max = 100.0
    tau_inc_iter = 2000
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv and smooth gradient:
        conv = _conv_self(a, L_fft) # full conv
        # compute smoothed max and gradient
        # subtract max for numerical stability in exp
        maxc = conv.max()
        exp_term = np.exp(tau * (conv - maxc))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp
        # gradient using convolution
        a_rev = a[::-1]
        # FFT sizes
        Lgrad = _next_pow2(len(a_rev) + len(w) - 1)
        A_rev = np.fft.rfft(a_rev, n=Lgrad)
        W = np.fft.rfft(w, n=Lgrad)
        conv_rev_w = np.fft.irfft(A_rev * W, n=Lgrad)
        grad = 2 * conv_rev_w[(len(a_rev)-1):(len(a_rev)-1+len(a))]

        # optional: scale gradient? multipliers? We'll keep as is.

        # Evaluate true score
        cur_score = float(2.0 * n * maxc / (a.sum()**2))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it} tau={tau:.2f} score={cur_score:.6f}")

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto simplex
        a = _project_simplex(a)

        # adjust tau
        if it % tau_inc_iter == 0 and tau < tau_max:
            tau = min(tau * 2.0, tau_max)

    return best_a, best_score

"""
Improved search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages

1️⃣  A very broad parametric sweep (geometric, power‑law, beta‑type and
    random Dirichlet seeds).  Each candidate is normalised to sum = 1
    and evaluated with a fast FFT convolution.

2️⃣  A smooth‑max Adam optimiser on the simplex.
    Instead of the non‑differentiable max we minimise
          φτ(a) = (1/τ)·log Σk exp(τ·convk(a,a))
    whose gradient is a simple weighted cross‑correlation.
    Starting with a small τ we gradually increase it, so the optimiser
    first sees a smooth landscape and later focuses on the true maximum.

3️⃣  A final stochastic hill‑climb that repeatedly transfers a tiny random
    amount of mass between two indices and keeps the move only if it improves
    the exact objective.

All stages respect the 1000 s budget, use FFT‑based convolutions
(O(n log n)) and Euclidean projection onto the probability simplex.
The routine returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.signal import windows  # optional, only for extra starters


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Full linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and Σw = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (same as evaluate_sequence for our scale)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray, n: int) -> float:
    conv = _conv_self(seq, _next_pow2(2 * n - 1))
    max_val = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, tau: float, L_fft: int):
    """
    Returns (smooth_max, gradient, true_max, full_convolution).

    gradient = ∂/∂a ( (1/τ)·log Σ exp(τ·conv)) .
    """
    n = a.shape[0]

    # 1) full convolution a*a
    conv = _conv_self(a, L_fft)                # length 2n‑1
    maxc = conv.max()

    # 2) soft‑max weights (numerically stable)
    exp_term = np.exp(tau * (conv - maxc))      # maxc subtracted → max weight = 1
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp                     # shape (2n‑1), Σ w = 1

    # 3) gradient = 2 * (a_rev ⋆ w)  (cross‑correlation)
    a_rev = a[::-1]
    Lc = _next_pow2(len(a_rev) + len(w) - 1)
    A_rev = np.fft.rfft(a_rev, n=Lc)
    W = np.fft.rfft(w, n=Lc)
    conv_rev_w = np.fft.irfft(A_rev * W, n=Lc)
    start = len(a_rev) - 1
    grad = 2.0 * conv_rev_w[start:start + n]

    # 4) smooth approximation of the max
    smooth_max = maxc + np.log(sum_exp) / tau

    return smooth_max, grad, maxc, conv


# ----------------------------------------------------------------------
# Adam‑style optimisation on the simplex using the smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          start_time: float,
                          deadline: float,
                          best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises the exact objective.
    The surrogate smooth‑max is used for the gradient, while the true
    score (based on the actual max) is recorded for progress.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    tau = 0.5                 # start very smooth
    tau_max = 80.0
    tau_inc_interval = 2500   # double τ every this many iterations

    while time.time() < deadline:
        it += 1

        smooth_max, grad, cur_max, _ = _smooth_obj_and_grad(a, tau, L_fft)

        # true objective for this iterate
        cur_score = float(2.0 * n * cur_max / (a.sum() ** 2))

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2f} score={cur_score:.6f}")

        # Adam descent step (note the minus sign)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex (keeps sum = 1, non‑negative)
        a = _project_simplex(a)

        # gradually sharpen the surrogate
        if it % tau_inc_interval == 0 and tau < tau_max:
            tau = min(tau * 2.0, tau_max)

    return best_a, best_score

def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Randomly moves a tiny fraction of mass from one index to another and
    keeps the move only if the true objective improves.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] <= 0.0:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 2 % of the source mass (larger early, smaller later)
        frac = rng.random() * 0.02 + 1e-6
        delta = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

        if cur < best_score:
            best_score = cur
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet conc={concentration:.2f} score={cur:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam optimisation (majority of the remaining budget)
    # --------------------------------------------------------------
    # reserve ~150 s for the final hill‑climb
    hill_time = 150.0
    adam_deadline = deadline - hill_time
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth Adam optimiser")
        a_opt, sc_opt = _smooth_adam_optimize(best_seq, start_time, adam_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
