The goal is to return a non‑negative step‑function (a list of heights) that minimises  

\[
F(a)=\frac{2\,n\;\max\bigl((a*a)\bigr)}{\bigl(\sum_i a_i\bigr)^2}.
\]

The evaluation routine `evaluate_sequence` already performs all safety checks, so the
search routine only has to supply a list of non‑negative floats.

The algorithm consists of three stages:

1. **Diverse warm‑starts** – a small library of analytically motivated shapes
   (uniform, triangular, power‑law, exponential, Gaussian, arcsine) together
   with a few random Dirichlet draws.  The best warm‑start (according to the
   exact objective) is kept.

2. **Smooth exponential‑gradient descent** – instead of the crude “max‑sub‑gradient”
   used before we approximate the maximum with a *soft‑max* (log‑sum‑exp).  
   For a temperature parameter λ we set  

   \[
   w_i=\frac{e^{\lambda\,c_i}}{\sum_j e^{\lambda\,c_j}},\qquad
   c=(a*a)
   \]

   and obtain the exact gradient of the smooth maximum

   \[
   \nabla_a\,\operatorname{softmax}(c)=
   2\;{\rm corr}(a,w)
   \]

   (correlation can be computed with a single FFT).  
   The gradient is combined with an exponential‑mirror update (mirror‑descent on
   the simplex) and a momentum term (Adam‑style).  λ is increased linearly from
   a small value (λ≈5) to a large value (λ≈400) so that the optimisation starts
   smooth and gradually focuses on the true maximum.

3. **Final polishing** – a short run of the original max‑sub‑gradient
   method followed by a cheap simulated‑annealing walk, both limited to the
   remaining time budget.

All steps keep the vector on the probability simplex (non‑negative, sum = 1)
by normalising after each exponential update.  The whole procedure respects
the 1000 s time limit; a 0.5 s safety margin is kept.

The code below implements the complete search.  It is self‑contained, uses only
`numpy`, `math` and the standard library, and follows the required
`propose_candidate` signature.  Print statements give a short progress log.  



```python
"""
search for a low‑objective step function

1) build a catalogue of diverse initial shapes (analytic families + random Dirichlet)
2) pick the best warm‑start (exact objective)
3) smooth exponential‑gradient descent with an annealing λ‑schedule
   – gradient of a smooth max is obtained by a correlation via FFT
   – mirror‑descent on the simplex (exponential update) with momentum
4) a brief max‑sub‑gradient refinement + tiny simulated‑annealing polish
5) return the best vector (non‑negative floats)
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑start shape generators
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # ∈[0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # ∈[0,1]
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _collect_warm_starts(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a modest list of heterogeneous starting vectors."""
    cand = []

    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # random Dirichlet draws – they often discover unexpected patterns
    for alpha in (0.3, 0.5, 1.0, 2.0):
        for _ in range(2):
            a = rng.dirichlet(np.full(n, alpha))
            cand.append(a)

    # optional global starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # sanitise
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# 2) Smooth exponential‑gradient descent (soft‑max of the convolution)
# ----------------------------------------------------------------------
def _smooth_exponential(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        lam_start: float = 5.0,
                        lam_end: float = 400.0,
                        beta: float = 0.95,
                        eta0: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the gradient of a smoothed max.
    λ is increased linearly from lam_start to lam_end.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    # momentum (Adam‑like)
    m = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- convolution (exact) ---------------------------------
        Lc = _next_pow2(2 * n - 1)
        Af = np.fft.rfft(a, n=Lc)
        conv = np.fft.irfft(Af * Af, n=Lc)[: 2 * n - 1]

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth iter={it:,} "
                  f"score={cur_score:.6f}")

        # ----- temperature λ ---------------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        lam = lam_start + (lam_end - lam_start) * (elapsed / total)

        # soft‑max weights (stable)
        shifted = conv - cur_max          # max becomes 0
        w = np.exp(lam * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # ----- gradient of the smooth max -------------------------
        # gradient = 2 * correlation(a, w)
        Lg = _next_pow2(3 * n)            # enough for len(w)+len(a_rev)-1
        A_rev = np.fft.rfft(a[::-1], n=Lg)
        Wf = np.fft.rfft(w, n=Lg)
        corr_full = np.fft.irfft(A_rev * Wf, n=Lg)
        start = n - 1
        grad = 2.0 * corr_full[start:start + n]

        # ----- momentum / step ------------------------------------
        m = beta * m + (1.0 - beta) * grad
        step = eta0 / math.sqrt(it)

        # ----- exponential (mirror) update -----------------------
        a = a * np.exp(-step * m)
        total_a = a.sum()
        if total_a > 0:
            a = a / total_a
        else:
            a = np.full_like(a, 1.0 / n)

    return best, best_score


# ----------------------------------------------------------------------
# 3) Max‑sub‑gradient descent (as in the reference implementation)
# ----------------------------------------------------------------------
def _subgrad_exponential(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         tol_factor: float = 0.01,
                         beta: float = 0.9,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # indices near the current maximum
        tol = cur_max * tol_factor
        top_idx = np.where(conv >= cur_max - tol)[0]
        if top_idx.size == 0:
            top_idx = np.array([conv.argmax()])

        # averaged sub‑gradient of the max
        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(top_idx.size)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = eta0 / math.sqrt(it)

        a = a * np.exp(-step * momentum)
        total = a.sum()
        a = a / total if total > 0 else np.full_like(a, 1.0 / n)

    return best, best_score


# ----------------------------------------------------------------------
# 4) Light simulated‑annealing polish (same as reference)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # source proportional to mass
        dst = rng.integers(n)              # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – power of two gives fast FFTs
    n = 8192   # a finer discretisation than the reference (still fast)

    # ------------------------------------------------------------------
    # 1) collect warm starts
    # ------------------------------------------------------------------
    candidates = _collect_warm_starts(n, rng)

    # ------------------------------------------------------------------
    # 2) pick the best warm start
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates, 1):
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] Warm‑start {idx}/{len(candidates)} "
              f"score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # 3) smooth exponential‑gradient descent (major part of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:   # keep a few seconds for polishing
        smooth_deadline = time.time() + remaining * 0.80
        refined_seq, refined_score = _smooth_exponential(
            best_seq, smooth_deadline, start_time,
            lam_start=5.0, lam_end=400.0,
            beta=0.95, eta0=0.30
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4) short max‑sub‑gradient refinement (more accurate near the end)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = time.time() + (deadline - time.time()) * 0.50
        refined_seq, refined_score = _subgrad_exponential(
            best_seq, sub_deadline, start_time,
            tol_factor=0.005, beta=0.9, eta0=0.5
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 5) final tiny annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 6) sanitise & return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
```
