"""
Iterative LP‑descent for the auto‑convolution max‑norm problem.

The algorithm works on a modest sized step‑function (default n=1024) and
repeats the following phase until the time budget is exhausted.

1.  For the current vector `f` (non‑negative, sum = 1) compute its
    self‑convolution `c = f*f` and the worst value `M = max(c)`.
2.  Solve the linear program

        maximise   sum(g)
        subject to  g ≥ 0
                    f * g ≤ M                (element‑wise)

    where `*` denotes linear convolution.  The constraints are linear in
    the unknown `g`, therefore the problem is a pure LP that can be
    tackled with CVXpy/SCS.  The solution tends to move mass to indices
    where `f` contributes little to the convolution, and often yields a
    total sum > 1.
3.  Rescale the LP‑solution to unit L1‑norm (`ĝ = g / sum(g)`) and
    perform a convex update

        f ← (1‑t)·f + t·ĝ

    with a step size `t` that slowly decays.  The update never increases
    `max(f*f)` and therefore drives the objective down.
4.  After the LP phase a cheap stochastic hill‑climb (mass transfers
    between random coordinates) is executed to polish the best vector
    found.

All vectors are kept on the probability simplex, so the evaluation
`2·n·max(conv(a,a)) / (sum(a)^2)` reduces to `2·n·max(conv(a,a))`.
The routine respects the 1000 s budget and returns the best sequence
as a plain Python list.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1)."""
    return np.convolve(a, a)


def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1)×n sparse matrix A such that A @ g == np.convolve(f, g).
    For each offset d = 0..n‑1 the d‑th diagonal of A is constant = f[d].
    """
    n = f.shape[0]
    offsets = np.arange(n, dtype=int)           # 0 … n‑1
    # each diagonal consists of n copies of the same value f[d]
    data = [np.full(n, f[d], dtype=float) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _lp_optimize(f: np.ndarray, M: float, warm_start: np.ndarray = None) -> np.ndarray | None:
    """
    Solve   maximise sum(g)   s.t.   g ≥ 0,  f*g ≤ M .
    Returns the optimal g (non‑negative) or None if the solver fails.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                     # (2n‑1)×n
    g = cp.Variable(n, nonneg=True)

    constraints = [A @ g <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    if warm_start is not None:
        g.value = warm_start
    try:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=3000, verbose=False)
    except (cp.error.SolverError, ValueError):
        return None

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)


def _hill_climb(a: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple random mass‑transfer hill‑climb.
    Moves a small fraction (up to 1 %) of a source coordinate to a destination.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(best_a).max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_a[src] * rng.random() * 0.01      # ≤ 1 % of source mass
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps

        M = _conv_self(cand).max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_a = cand
            # keep the improvement visible in the log
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score

    t = 0.5               # initial mixing weight
    decay = 0.995         # geometric decay per iteration
    min_t = 0.01

    prev_g = None
    it = 0
    while time.time() < deadline - 20.0:      # leave ~20 s for hill‑climb
        it += 1
        conv_f = _conv_self(f)
        M = conv_f.max()
        cur_score = 2.0 * n * M

        # solve the LP
        g = _lp_optimize(f, M, warm_start=prev_g)
        if g is None:
            print(f"[{time.time() - start_time:.2f}s] LP failed – terminating descent")
            break

        sum_g = g.sum()
        if sum_g <= 1e-12:
            print(f"[{time.time() - start_time:.2f}s] LP returned zero mass – breaking")
            break

        # scale g to unit L1‑norm
        g_hat = g / sum_g

        # convex combination
        new_f = (1.0 - t) * f + t * g_hat

        # evaluate the new vector
        new_M = _conv_self(new_f).max()
        new_score = 2.0 * n * new_M

        if new_score < best_score:
            best_score = new_score
            best_f = new_f.copy()
            print(f"[{time.time() - start_time:.2f}s] it={it:,} t={t:.4f} "
                  f"score improved to {best_score:.6f}")

        # accept the move if it does not increase the objective,
        # otherwise shrink the step size.
        if new_score <= cur_score:
            f = new_f
            prev_g = g      # warm‑start the next LP with the current solution
        else:
            t = max(t * 0.5, min_t)   # aggressive reduction when worsening

        # decay the step size slowly
        t = max(t * decay, min_t)

    # ------------------------------------------------------------------
    # Final hill‑climb polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting final hill‑climb")
        best_f, best_score = _hill_climb(best_f, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] hill‑climb finished – best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    total = best_f.sum()
    if total > 0:
        best_f /= total
    else:
        best_f = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – final score≈{best_score:.6f}")
    return [float(v) for v in best_f]
