def _subgradient_descent_max(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    momentum = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] subgrad iter {it} score {cur_score:.6f}")

        # find index of max conv
        k = conv.argmax()
        # compute gradient at k
        grad = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        # indices i in [low, high]
        # For each i: corresponding partner j = k-i
        # gradient for a[i] add a[j]; if i != j also add a[j] (double)
        # We'll compute directly:
        for i in range(low, high+1):
            j = k - i
            if i == j:
                grad[i] += a[i]  # derivative from squared term
            else:
                grad[i] += a[j] * 2.0
        # momentum and update
        momentum = beta * momentum + (1-beta) * grad
        step = eta0 / math.sqrt(it)
        a -= step * momentum
        # project onto simplex
        a = _project_simplex(a)
        # check for convergence maybe early stop if score not improving
    return best_a, best_score

"""
Propose a low‑score step function for the given evaluation routine.

The core idea
--------------
* Use a *monotone* (decreasing) shape – this breaks the symmetry that makes the
  uniform vector a local optimum for the max‑convolution objective.
* Generate many diverse initial candidates:
    – power‑law decreasing (a_i ∝ (1‑i/(n‑1))^p)
    – exponential decay (a_i ∝ r^i)
    – Dirichlet draws with different concentration parameters
* Refine the best candidates with a simple projected sub‑gradient method.
  The sub‑gradient is taken only at the *active* convolution entry (the
  current maximal one) – this is an exact sub‑gradient of the objective.
  A momentum term (β≈0.9) and a diminishing step size η_t = η₀/√t are used.
* Finish with a cheap simulated‑annealing “mass‑swap” polish.
* All operations are vectorised, the only O(n log n) step is the FFT used to
  compute the linear convolution (the mask for the gradient is O(n)).
* The search works on n = 4096 (a power of two), which fits comfortably in the
  1000 s budget while still giving a good discretisation of the continuous
  problem.

The algorithm returns a plain Python list of non‑negative numbers whose sum is
positive; the evaluation routine is scale‑invariant, so any normalisation works.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT based self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (scale‑invariant, without the security wrappers)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex (Wang‑Carreira‑Perpiñán)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators – monotone decreasing families
# ----------------------------------------------------------------------
def _make_decreasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((n‑1‑i)/(n‑1))^p  (p>0)."""
    idx = np.arange(n, dtype=float)
    shape = ((n - 1 - idx) / (n - 1)) ** p
    if shape.sum() == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape = shape / shape.sum()
    return shape


def _make_exponential_decay(n: int, base: float) -> np.ndarray:
    """a_i ∝ base^i  (0<base<1)."""
    idx = np.arange(n, dtype=float)
    shape = base ** idx
    if shape.sum() == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape = shape / shape.sum()
    return shape


# ----------------------------------------------------------------------
# Gradient of the *active* convolution entry (the maximal one)
# ----------------------------------------------------------------------
def _grad_at_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the objective at the current maximum convolution
    entry.  Returns a vector of size n.
    """
    n = a.shape[0]
    k = int(conv.argmax())                 # index of the maximal entry
    low = max(0, k - (n - 1))
    high = min(n - 1, k)
    idx = np.arange(low, high + 1, dtype=int)
    partner = k - idx                      # the index that pairs with idx

    grad = np.zeros_like(a)
    # positions where idx == partner (the “diagonal” term)
    diag = idx == partner
    if np.any(diag):
        grad[idx[diag]] = a[idx[diag]]    # derivative of a_i² term
    # off‑diagonal terms: each appears twice in the sum
    off = ~diag
    if np.any(off):
        grad[idx[off]] = 2.0 * a[partner[off]]
    return grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (using only the active max entry)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 10_000) -> tuple[np.ndarray, float]:
    """
    Refine a vector by projected sub‑gradient descent.
    Stops when the time budget expires or after ``max_iters`` iterations.
    Returns the best iterate seen and its (internal) score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # momentum variables
    momentum = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.7
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_self(a)
        cur_score = _objective(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SG iter={it:,} score={cur_score:.6f}")

        # exact sub‑gradient at the current maximal entry
        grad = _grad_at_max(a, conv)

        # momentum + diminishing step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = eta0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)            # stay on the simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Move tiny packets of mass between random positions.
    Simple Metropolis acceptance with an exponential cooling schedule.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # Total number of attempts – adapt to the remaining time
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # cooling (fast exponential)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # source drawn proportionally to its current mass (more likely to move)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------

    np.random.seed(seed)
    start_time = time.time()

    # --------------------------------------------------------------

    n = 4096                       # 2¹¹ – concise enough for many iterations

    # --------------------------------------------------------------

    pool_size = 12

        a = _make_decreasing_power(n, p)
        sc = _objective(a)

            pool.sort(key=lambda x: x[0])

    for base in np.linspace(0.88, 0.999, 20):    # exponential decay
        a = _make_exponential_decay(n, base)
        sc = _objective(a)
        if len(pool) < pool_size:
            pool.append((sc, a.copy()))
            pool.sort(key=lambda x: x[0])

            pool[-1] = (sc, a.copy())

            if len(pool) < pool_size:

                sc = _objective(starter)
