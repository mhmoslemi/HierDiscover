def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # Build candidates
    top_k = 10
    top = [] # list of (score, seq)

    # analytic shapes
    p_vals = np.concatenate([np.linspace(-2.0, -0.2, 9), np.linspace(0.2, 2.0, 9)])
    alpha_vals = np.linspace(0.0, 0.5, 6)
    for p in p_vals:
        for alpha in alpha_vals:
            a = _make_shape(N, p, alpha)
            sc = _objective(a)
            # update top
            if len(top) < top_k:
                top.append((sc, a))
                top.sort(key=lambda x: x[0])
            elif sc < top[-1][0]:
                top[-1] = (sc, a)
                top.sort(key=lambda x: x[0])

    # Beta shapes
    beta_params = np.linspace(0.2, 3.0, 10) # includes <1 for U-shape
    x = (np.arange(N) + 0.5) / N
    for a in beta_params:
        shape = (x**(a-1)) * ((1-x)**(a-1))
        # handle near-zero values
        shape = np.maximum(shape, 1e-14)
        shape /= shape.sum()
        sc = _objective(shape)
        if len(top) < top_k:
            top.append((sc, shape))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, shape)
            top.sort(key=lambda x: x[0])

    # Inverted shapes for each shape in top that we have (maybe just a few)
    # later after gather top we can generate inverses.

    # Random vectors
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline and attempts < 200:
        a = rng.random(N)
        a /= a.sum()
        sc = _objective(a)
        attempts += 1
        if len(top) < top_k:
            top.append((sc, a))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, a)
            top.sort(key=lambda x: x[0])

    # Inverted versions of top candidates (just compute after previous)
    for _, seq in top.copy():
        inv = 1.0/(seq + 1e-14)
        inv /= inv.sum()
        sc = _objective(inv)
        if len(top) < top_k:
            top.append((sc, inv))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, inv)
            top.sort(key=lambda x: x[0])

    # incorporate height_sequence_1
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up = starter_up / starter_up.sum()
                sc = _objective(starter_up)
                if len(top) < top_k:
                    top.append((sc, starter_up))
                    top.sort(key=lambda x: x[0])
                elif sc < top[-1][0]:
                    top[-1] = (sc, starter_up)
                    top.sort(key=lambda x: x[0])
    except Exception:
        pass

    # pick best candidate
    best_score, best_seq = top[0]
    print(f"[{time.time()-start:.2f}s] Initial best score = {best_score:.6f}")

    # Phase 2 smooth gradient descent
    remaining = deadline - time.time()
    if remaining > 20:
        # allocate 70% of remaining to run GD on top candidates (maybe top 3)
        gd_budget = remaining * 0.7
        # split among top few
        num_gd = min(3, len(top))
        per_budget = gd_budget / num_gd
        for idx in range(num_gd):
            if time.time() >= deadline:
                break
            seq = top[idx][1]
            sub_deadline = time.time() + per_budget
            refined_seq, refined_score = _smooth_gradient_descent(seq, sub_deadline, start)
            if refined_score < best_score:
                best_score, best_seq = refined_score, refined_seq
                print(f"[{time.time()-start:.2f}s] GD improved score = {best_score:.6f}")

    # Phase 3 subgradient
    remaining = deadline - time.time()
    if remaining > 10:
        sub_deadline = time.time() + remaining * 0.6
        refined_seq, refined_score = _subgradient_descent(best_seq, sub_deadline, start)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq
            print(f"[{time.time()-start:.2f}s] Subgrad improved score = {best_score:.6f}")

    # Phase 4 annealing
    if time.time() < deadline:
        annealed_seq, annealed_score = _anneal(best_seq, deadline, start)

"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. **Candidate pool** – generate a diverse set of analytic shapes:
   * power‑ramps with *negative* exponents (U‑shaped),
   * symmetric beta‑pdfs (α=β) covering both convex and concave cases,
   * “inverse” shapes (1/shape) of the above,
   * random vectors,
   * random Fourier‑spectra (constant magnitude, random phase) projected onto the simplex.
   The best few (by the exact objective) are kept.

2. **Multi‑start smoothed gradient descent** – an Adam‑style descent on a
   soft‑max surrogate of the max‑convolution, run from each of the top
   candidates for a fraction of the remaining budget.

3. **Sub‑gradient refinement** – a simple sub‑gradient method (the
   reference implementation) to polish the best result.

4. **Light annealing** – a Metropolis‑style move‑pack that swaps a small
   amount of mass between two positions; it helps escape tiny local minima.

All phases respect the 1000 s time budget.  The final sequence is projected
onto the probability simplex (sum = 1) and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Analytic shape generator (power‑ramps with optional plateau)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric shape of length n, normalised to sum 1.
       `p` may be negative – this creates a U‑shaped ramp.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        # power‑ramp; negative p yields decreasing values towards the centre
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# ----------------------------------------------------------------------
# Random beta‑distribution shapes (symmetric, α = β)
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta pdf on a grid, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n          # mid‑point rule
    # avoid exact zeros at the boundaries
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    shape = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
    shape = np.maximum(shape, eps)
    shape /= shape.sum()
    return shape

# ----------------------------------------------------------------------
# Random Fourier spectrum shape (constant magnitude, random phase)
# ----------------------------------------------------------------------
def _random_fourier_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a non‑negative vector from a random unit‑magnitude Fourier spectrum."""
    # build a Hermitian‑symmetric spectrum with unit magnitude
    F = np.zeros(n, dtype=complex)
    # frequencies 1 … n/2‑1
    if n > 2:
        phases = rng.random(n // 2 - 1) * 2.0 * np.pi
        F[1:n // 2] = np.exp(1j * phases)
        if n % 2 == 0:
            # Nyquist frequency is real‑valued
            F[n // 2] = 0.0
        # conjugate symmetry
        F[n // 2 + 1:] = np.conj(F[1:n // 2][::-1])
    # inverse transform, take the real part
    a = np.fft.ifft(F).real
    # make non‑negative and normalise
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max approximation of max)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a smooth surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.05          # base learning rate
    T0 = 0.02             # initial temperature
    T_end = 1e-5          # final temperature – lower than before for a tighter max

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} score={cur_score:.6f}")

        # ----- temperature schedule (geometric decay) -----
        frac = it / (it + 500.0)
        T = max(T_end, T0 * (1.0 - frac))

        # ----- soft‑max weights -----
        w_raw = np.exp((conv - max_conv) / T)     # shift by max for stability
        w = w_raw / w_raw.sum()

        # ----- gradient via correlation (w * a_rev) -----
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]   # length 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]              # slice of length n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (same as reference implementation)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # index of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑conv term
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score

# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:

    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 300_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
