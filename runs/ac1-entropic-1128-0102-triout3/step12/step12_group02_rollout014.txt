import time
import math
import numpy as np
from typing import List

def _next_pow2(x: int) -> int:
    """Return smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective with sum(a)≈1 using FFT."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))

def _smooth_gradient(a: np.ndarray, beta: float, L_conv: int, L_corr: int):
    """
    Gradient of softened max( conv(a,a) ) using soft‑max weights.
    Returns (grad, conv) where grad has shape (n,).
    """
    conv = _conv_self(a, L=L_conv)
    max_c = conv.max()
    shifted = conv - max_c                 # max becomes 0
    exp_vals = np.exp(beta * shifted)      # safe because shifted≤0
    w = exp_vals / exp_vals.sum()
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # FFT lengths (fixed)
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0
    beta_mul = 1.02  # slightly faster growth
    # base learning rate – we start relatively large then decay
    lr0 = 0.5

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        # exact score for tracking
        cur_sc = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t} β={beta:.1f} score={cur_sc:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule: decay with iteration count
        lr = lr0 / (1.0 + 1e-4 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # increase temperature
        beta = min(beta * beta_mul, beta_max)

        # occasional random jitter to hop out of plateaus
        if t % 3000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_sc

def _targeted_hill_climb(a0: np.ndarray, deadline: float, start_time: float, L_fft: int):
    """
    Stochastic hill‑climb that mixes random transfers with moves that
    directly attack the currently dominant convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    # step size parameters
    step_factor = 0.15          # up to 15 % of source mass per move
    min_factor = 1e-5
    decay = 0.995

    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.6:
            # random mass transfer
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # targeted move: take mass from an index that participates in the max conv
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            # choose source as the larger of the two contributors
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            # avoid moving to the other contributor of the same pair
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        # ensure numeric safety
        if a_prop[src] < 0:
            continue

        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            a = a_prop
            best_sc = new_sc
            best_a = a.copy()
            # reset step size after a success
            step_factor = max(step_factor, 0.15)
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_sc:.6f}")
        else:
            # decay step factor to explore finer moves
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_sc

def _parametric_sweep(n: int, uniform: np.ndarray, deadline: float, start_time: float,

    Explore a family a_i ∝ r^i * (i+1)^p optionally mixed with the uniform vector.
    Returns the best vector and score found during the sweep.
    """

    for r in r_grid:
        if time.time() >= deadline - 120:   # keep ~2 min for later phases

                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

1️⃣  Initialise from a global starter (if any) or a uniform vector.
2️⃣  Perform a focused parametric sweep over monotone shapes
    a_i ∝ r^i·(i+1)^p (mixed with the uniform distribution) to obtain a good seed.
3️⃣  Run several short Adam‑style projected‑gradient descents on a smooth‑max
    surrogate of the “max‑conv” term.  The soft‑max temperature β grows up to 800
    and the learning‑rate decays, giving a precise approximation of the true
    gradient of the maximum convolution entry.
4️⃣  Polish the best vector with a stochastic hill‑climb that mixes
    *random* mass‑transfer moves with *targeted* moves that directly reduce the
    currently dominant convolution entry.  The step‑size adapts on‑the‑fly.
5️⃣  Return the best vector (normalised to sum 1) as a plain list of Python floats.

The routine respects the 1000 s budget and prints intermediate scores.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a*a                     (full linear convolution)
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is computed via FFT‑based correlation.
    """
    conv = _conv_self(a, L=L_conv)          # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (stable)
    shifted = conv - max_c                 # max becomes 0
    exp_vals = np.exp(beta * shifted)      # under‑flow for large negative values is harmless
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # correlation w * a_rev → gradient
    a_rev = a[::-1]                        # length n
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)   # linear convolution w * a_rev

    start = a.shape[0] - 1                 # n‑1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected descent on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 800.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type descent on the smooth‑max surrogate.
    β starts small (smooth landscape) and is slowly increased up to beta_max.
    The iterate is always projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0               # start fairly smooth
    beta_mul = 1.02          # geometric increase per iteration

    while time.time() < deadline and t < max_iters:
        t += 1

        # gradient of the smooth surrogate
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        # exact score (sum of a is 1 → denominator = 1)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} score={cur_score:.6f}"
            )

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # base learning‑rate decays slowly with the number of iterations
        lr = 0.5 / (1.0 + 1e-4 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # tiny random jitter every now and then to escape plateaus
        if t % 3000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

# Targeted hill‑climb (random + max‑entry attacks)
# ----------------------------------------------------------------------
def _targeted_hill_climb(

    """
    Random mass transfers (up to ~15 % of the source mass) plus
    targeted moves that remove mass from the pair of indices contributing
    to the current maximum convolution entry.
    Only improving moves are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    # step‑size control
    step_factor = 0.15      # up to 15 % of source mass per move
    min_factor = 1e-5
    decay = 0.995

    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.6:
            # ---- random transfer -------------------------------------------------
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # ---- targeted transfer ------------------------------------------------
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose the move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)  # reset after success
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
        else:
            # shrink step size when a move is rejected
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score

# Coarse‑to‑fine parametric sweep

            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                    )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine respects the 1000 s budget and returns the best
    sequence found (as a plain list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # problem size – 4096 gives a good trade‑off between expressivity and speed
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise: uniform baseline (sum ≈ 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(
        f"[{time.time() - start_time:.2f}s] baseline uniform score={best_score:.6f}"
    )

    # try to use a globally provided starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (coarse → refined)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:   # reserve ~5 min for the sweep
        best_seq, best_score = _parametric_sweep(
            n, uniform, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # 2️⃣  Multiple smooth‑max Adam descent runs (diverse seeds)
    # ------------------------------------------------------------------
    descent_seeds = [seed, seed + 1, seed + 2]
    for ds in descent_seeds:
        if time.time() >= deadline - 120:
            break
        rng = np.random.default_rng(ds)
        a0 = best_seq.copy()
        a0 += 1e-4 * rng.standard_normal(n) * a0   # tiny perturbation
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        max_iters = 20000
        a_opt, sc_opt = _smooth_descent(
            a0,
            max_iters,
            deadline - 30,          # leave a couple of seconds for final polish
            start_time,
            best_score,
            beta_max=800.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth descent improvement score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb polish (random + targeted moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _targeted_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # sanity check with the official evaluator (optional)
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}"
        )
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
