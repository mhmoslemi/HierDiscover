def _optimize_adam(a0, deadline, start_time):
    n = len(a0)
    z = np.log(a0 + 1e-12)   # initial logits
    L_conv = _next_pow2(2*n-1)
    L_full = _next_pow2(3*n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    best_a = np.exp(z) / np.exp(z).sum()  # same as a0
    best_score = _objective(best_a)
    it = 0
    while time.time() < deadline:
        it += 1
        a = _softmax(z)
        # compute conv via FFT length L_conv
        Af = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(Af * Af, n=L_conv)[:2*n-1]
        # softmax over conv with temperature tau
        tau = 0.5 + (5.0 - 0.5) * min(1.0, it / 5000.0)
        x = tau * conv
        max_x = x.max()
        expx = np.exp(x - max_x)
        w = expx / expx.sum()
        # FFT for w for full conv
        Wf = np.fft.rfft(w, n=L_full)
        # conv2: a * w
        Af_full = np.fft.rfft(a, n=L_full)
        conv2_full = np.fft.irfft(Af_full * Wf, n=L_full)
        conv2 = conv2_full[:n]
        # conv1: w * a_rev
        a_rev = a[::-1]
        a_rev_f = np.fft.rfft(a_rev, n=L_full)
        conv1_full = np.fft.irfft(Wf * a_rev_f, n=L_full)
        conv1 = conv1_full[n-1:n-1+n]
        grad_a = 2.0 * n * (conv1 + conv2)
        # gradient wrt logits
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = 0.05 / (np.sqrt(v_hat) + eps)
        z = z - step * m_hat
        if it % 100 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter={it} score={cur_score:.6f}")
    return best_a, best_score

"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

by a three‑phase search:

1️⃣ Warm‑start: a small library of analytic shapes (uniform, tent,
   power‑law, exponential, Gaussian, arcsine).  The best of them
   (according to the exact objective) is chosen.

2️⃣ Smooth‑max Adam optimisation on the probability simplex.
   The variables are logits `z` such that `a = softmax(z)`.  At each
   iteration the maximum of the self‑convolution is approximated by a
   temperature‑scaled soft‑max (log‑sum‑exp).  The gradient of this
   smooth objective can be written in closed form using two FFT‑based
   convolutions, giving an efficient O(n log n) update.  The temperature
   is increased slowly (0.5 → 5.0) so that early steps see a smooth
   landscape and later steps hone in on the true peak.  Adam is used
   for adaptive step sizes and the best exact score seen is retained.

3️⃣ Light simulated‑annealing polish (random mass swaps) to escape any
   remaining shallow minima.

The whole routine respects the 1000 s budget, keeps all values
non‑negative, and returns a plain list of floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    v = np.asarray(v, dtype=float)
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0:
        return np.full_like(v, 1.0 / v.size)
    return e / s

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·n‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

# ----------------------------------------------------------------------
# Warm‑start shapes
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()

def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = tri ** p
    a += 1e-16
    return a / a.sum()

def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()

def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()

def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()

def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()

def _collect_warm_starts(n: int) -> list[np.ndarray]:
    cand = []

    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # Mixed centre/edge blends
    centre = _shape_power_center(n, 2.0)
    edge   = _shape_power_edge(n, 2.0)
    for lc in (0.25, 0.5, 0.75):
        a = lc * centre + (1 - lc) * edge
        cand.append(a / a.sum())

    # Global starter, if supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # remove NaNs (just in case)
    uniq = []
    for a in cand:
        if not np.isnan(a).any():
            uniq.append(a)
    return uniq

# ----------------------------------------------------------------------
# Phase‑2: Adam optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate of the max‑convolution.
    Returns the best exact‑score vector seen during the run.
    """
    n = a0.shape[0]

    # initialise logits (softmax parametrisation)
    z = np.log(a0 + 1e-12)

    # FFT sizes
    L_conv = _next_pow2(2 * n - 1)     # for a*a
    L_full = _next_pow2(3 * n)         # for convolutions with w (length 2n‑1)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # temperature schedule (smooth → true max)
    tau_start, tau_end, tau_ramp = 0.5, 5.0, 5000   # ramp over 5k iterations

    best_a = _softmax(z)
    best_score = _objective(best_a)
    it = 0

    while time.time() < deadline:
        it += 1

        # current probability vector
        a = _softmax(z)

        # 1️⃣  a*a  (self‑convolution)
        Af = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(Af * Af, n=L_conv)[: 2 * n - 1]

        # 2️⃣  smooth‑max weights w_i = softmax(tau*conv_i)
        tau = tau_start + (tau_end - tau_start) * min(1.0, it / tau_ramp)
        x = tau * conv
        mx = x.max()
        expx = np.exp(x - mx)
        w = expx / expx.sum()                     # length 2n‑1

        # 3️⃣  FFT of w for the two required convolutions
        Wf = np.fft.rfft(w, n=L_full)

        # -----   term Σ_i w_i a_{k‑i}   (standard convolution a * w) -----
        Af_full = np.fft.rfft(a, n=L_full)
        conv2_full = np.fft.irfft(Af_full * Wf, n=L_full)
        conv2 = conv2_full[:n]                     # first n entries give g2_k

        # -----   term Σ_i w_i a_{i‑k}   (correlation w * a_rev) -----
        a_rev = a[::-1]
        a_rev_f = np.fft.rfft(a_rev, n=L_full)
        conv1_full = np.fft.irfft(Wf * a_rev_f, n=L_full)
        conv1 = conv1_full[n - 1: n - 1 + n]       # slice gives g1_k

        # gradient w.r.t. a (exact gradient of the smoothed objective)
        grad_a = 2.0 * n * (conv1 + conv2)

        # chain rule through softmax (gradient w.r.t. logits)
        dot = np.dot(grad_a, a)                    # ⟨grad_a, a⟩
        grad_z = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = 0.05 / (np.sqrt(v_hat) + eps)       # base step size 0.05
        z = z - step * m_hat

        # occasional exact evaluation
        if it % 100 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                      f"score={cur_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Phase‑3: Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis walk that moves a random packet of mass between
    two positions.  More steps than the original implementation are
    performed (up to 150 k) while respecting the remaining time budget.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 150_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling schedule
        T = 0.03 * (1e-5 / 0.03) ** (it / total_steps)

        src = rng.choice(n, p=a)          # source weighted by current mass
        dst = rng.integers(n)             # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function within the supplied time
    budget.  Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem size – a power of two gives fast FFTs
    n = 4096

    # --------------------------------------------------------------
    # 1️⃣ Warm‑starts
    # --------------------------------------------------------------
    candidates = _collect_warm_starts(n)

    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates, 1):
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] Warm‑start {idx}/{len(candidates)} "
              f"score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # --------------------------------------------------------------
    # 2️⃣ Adam optimiser on smooth‑max surrogate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0:                     # keep a few seconds for polishing
        opt_deadline = time.time() + remaining * 0.80
        refined_seq, refined_score = _optimize_adam(best_seq, opt_deadline, start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined (Adam) score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Light annealing polish (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={anneal_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Normalise & convert to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
