def _cross_corr(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Return correlation (u ⋆ v)[j] = Σ_i u[i+j] * v[i] for j = 0..len(v)-1.
    Both u and v are non-negative real arrays. u length >= len(v) - 1.
    """
    m = len(u)
    n = len(v)
    L = _next_pow2(m + n - 1)
    # Use real FFT
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    corr = np.fft.irfft(Uf * np.conj(Vf), n=L)
    # Linear correlation indices from 0..n-1
    return corr[:n]

def _cross_corr(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear cross-correlation (u ⋆ v)[k] = Σ_i u[i+k] * v[i] for k=0..len(v)-1."""
    L = _next_pow2(len(u) + len(v) - 1)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)

"""
Propose a low‑score step‑function.

Approach
--------
1. Work internally on a large power‑of‑two grid (n=2¹¹ = 131 072) to
   reduce the discretisation error of the continuous optimum.
2. Initialise the vector from the previously found construction
   (`height_sequence_1`) – up‑sample it – or fall back to a simple
   power‑ramp shape.
3. Optimise the *smooth* Lₚ‑approximation of the infinity‑norm of the
   self‑convolution:

       fₚ(a) = ( Σ (a*a)ᵢ ᵖ )¹ᐟᵖ      (p large)

   The gradient of fₚ can be written as a cross‑correlation and is
   evaluated with FFTs.  An Adam‑style update together with a projection
   onto the probability simplex (Σa=1, a≥0) drives the optimisation.
4. Run the optimiser for several values of p (8,16,32,64) – a continuation
   scheme that sharpens the approximation to the true max‑norm.
5. A short simulated‑annealing polish moves tiny mass packets between
   random positions to escape possible local minima.
6. Down‑sample the final vector to 2¹⁶=65 536 entries (the size used in
   the reference solution) by averaging consecutive pairs, re‑normalise
   and return a plain Python list.

The whole routine respects the 1000 s wall‑time budget and never
produces invalid inputs for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _cross_corr(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation (u ⋆ v)[k] = Σ_i u[i+k]·v[i] for k=0..len(v)-1."""
    L = _next_pow2(len(u) + len(v) - 1)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    corr = np.fft.irfft(Uf * np.conj(Vf), n=L)
    return corr[: len(v)]


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric power‑ramp shape with an optional flat plateau.
    The vector is normalised to sum = 1.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Returns a non‑negative step‑function (as a list of floats) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # ------------------------------------------------------------------
    # 1) Internal problem size (large for accuracy)
    # ------------------------------------------------------------------
    internal_n = 131072          # 2¹¹⁷ – enough to push discretisation error <0.001
    final_n = 65536              # we down‑sample to this size for the final output

    # ------------------------------------------------------------------
    # 2) Initialise – up‑sample a stored construction if available
    # ------------------------------------------------------------------
    a0 = None
    try:
        base = np.asarray(height_sequence_1, dtype=float)
        if base.ndim == 1 and base.size > 0:
            # linear interpolation to the larger grid
            x_old = np.linspace(0.0, 1.0, base.size)
            x_new = np.linspace(0.0, 1.0, internal_n)
            a0 = np.interp(x_new, x_old, base)
    except Exception:
        a0 = None

    if a0 is None:
        # fallback: a simple two‑parameter analytic shape
        a0 = _make_shape(internal_n, p=0.8, alpha=0.2)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s > 0:
        a0 = a0 / s
    else:
        a0 = np.full(internal_n, 1.0 / internal_n)

    # ------------------------------------------------------------------
    # 3) Smooth Lp‑approximation optimisation (Adam + simplex projection)
    # ------------------------------------------------------------------
    best_seq = a0.copy()
    best_score = _objective(best_seq)

    # continuation over increasing p
    p_schedule = [8, 16, 32, 64]

    # allocate time roughly equally among stages
    remaining = deadline - time.time()
    for stage, p in enumerate(p_schedule):
        if time.time() >= deadline:
            break
        # give each remaining stage an equal share of the time left
        stage_deadline = time.time() + remaining / (len(p_schedule) - stage)

        # Adam state
        m = np.zeros_like(best_seq)
        v = np.zeros_like(best_seq)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.05
        it = 0

        while time.time() < stage_deadline:
            it += 1

            conv = _conv_self(best_seq)                     # a*a
            sp = np.sum(conv ** p)                          # Σ (conv)ᵖ
            if sp <= 0:
                break
            # gradient of ( Σ convᵖ )¹ᐟᵖ
            w = conv ** (p - 1)                             # conv^{p‑1}
            factor = 2.0 / (sp ** ((p - 1) / p))            # 2 / ( Σ convᵖ )^{(p‑1)/p}
            grad_raw = _cross_corr(w, best_seq)             # linear cross‑correlation
            grad = factor * grad_raw

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            best_seq = best_seq - step

            # keep on the simplex (non‑negative, sum = 1)
            best_seq = _project_simplex(best_seq)

            cur_score = _objective(best_seq)
            if cur_score < best_score:
                best_score = cur_score
                print(f"[{time.time() - start_time:.2f}s] p={p} iter={it:,} "
                      f"new best = {cur_score:.6f}")

            # occasional break if improvement stalls (optional)
            # if np.linalg.norm(grad) < 1e-12:
            #     break

        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 4) Light simulated‑annealing polish (mass swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = best_score
        rng = np.random.default_rng()
        max_attempts = 200_000

        for it in range(1, max_attempts + 1):
            if time.time() >= deadline:
                break

            # temperature schedule – fast exponential decay
            T = 0.02 * (1e-6 / 0.02) ** (it / max_attempts)

            src = rng.choice(internal_n, p=a)
            dst = rng.integers(internal_n)
            if src == dst or a[src] < 1e-14:
                continue

            # move up to 20 % of the source mass
            delta = a[src] * rng.random() * 0.20
            a_prop = a.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta

            new_score = _objective(a_prop)
            if new_score < cur_score or math.exp(-(new_score - cur_score) / T) > rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} "
                          f"best={new_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Down‑sample to the final length required by the benchmark
    # ------------------------------------------------------------------
    if internal_n != final_n:
        # simple averaging of consecutive entries
        out = (best_seq[0::2] + best_seq[1::2]) / 2.0
        if out.shape[0] > final_n:
            out = out[:final_n]
        elif out.shape[0] < final_n:
            out = np.concatenate([out, np.zeros(final_n - out.shape[0])])
    else:
        out = best_seq

    # final sanity checks
    out = np.maximum(out, 0.0)
    total = out.sum()
    if total > 0.0:
        out = out / total
    else:
        out = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(v) for v in out]
