* The convolution is updated exactly with a linear‑time formula, so each
  iteration costs O(n).  Periodic re‑normalisation avoids drift.
* When the search plateaus for many iterations we jitter the current best
  solution to escape local minima.
* The best probability vector seen during the allotted time is returned.
  Because the evaluation function is scale‑invariant a probability vector
  (Σp = 1) is a perfectly valid answer.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function with a small evaluation score.

    The function returns a list of non‑negative floats (the heights).  It must
    finish before `budget_s` seconds; otherwise the verifier awards −∞.
    """
    # ------------------------------------------------------------------
    # 0) basic set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin before the hard deadline
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) initialise a probability vector p (Σp = 1)
    # ------------------------------------------------------------------
    try:
        # use the previously discovered best construction when available
        p0 = np.asarray(height_sequence_1, dtype=float)
        if p0.ndim != 1 or p0.size == 0:
            raise ValueError
        p0 = np.maximum(p0, 0.0)
        s0 = p0.sum()
        if s0 == 0:
            raise ValueError
        p = p0 / s0
        n = p.shape[0]
        print(f"[{time.time() - start:.2f}s] Using height_sequence_1 (n={n})")
    except Exception:
        # fallback: a simple arcsine‑type distribution, n = 4096
        n = 4096
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        alpha = 0.38
        eps = 1e-6
        w = (x * (1.0 - x) + eps) ** (-alpha)
        p = w / w.sum()
        print(f"[{time.time() - start:.2f}s] Generated arcsine start (n={n})")

    # ------------------------------------------------------------------
    # 2) initial convolution and bookkeeping
    # ------------------------------------------------------------------
    conv = np.convolve(p, p)                     # length 2·n‑1
    cur_max = conv.max()
    best_max = cur_max
    best_score = 2.0 * n * cur_max               # because Σp = 1
    best_p = p.copy()
    print(f"[{time.time() - start:.2f}s] start score = {best_score:.6f}")

    rng = np.random.default_rng()
    iteration = 0
    last_improve = 0

    # temperature for simulated annealing (starts relatively high, decays)
    T0 = max(cur_max * 0.05, 1e-12)
    temperature = T0

    # ------------------------------------------------------------------
    # 3) main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- identify the current worst shift --------------------------------
        k = int(conv.argmax())
        cur_max = conv.max()

        # ----- source index: the pair that contributes most to conv[k] ----------
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = p[idx] * p[k - idx]
        src = idx[int(np.argmax(contrib))]

        # ----- destination candidates (prefer low‑mass entries) -----------------
        # probability proportional to (1‑p) – avoids the source
        inv_w = 1.0 - p
        inv_w[src] = 0.0
        sw = inv_w.sum()
        if sw <= 0:
            prob = np.ones_like(p) / n
        else:
            prob = inv_w / sw
        K_cand = 5                                   # number of dst trials
        # small‑risk sampling without replacement when possible
        if K_cand < n:
            dsts = rng.choice(n, size=K_cand, replace=False, p=prob)
        else:
            dsts = rng.choice(n, size=K_cand, replace=True, p=prob)

        # ----- step size ε (fraction of the source mass) -----------------------
        elapsed = time.time() - start
        max_frac = max(0.02, 0.20 * (1.0 - elapsed / budget_s))  # shrink over time
        epsilon = p[src] * rng.random() * max_frac
        if epsilon <= 0.0:
            continue

        # ----- evaluate every candidate destination ------------------------------
        best_cand_max = cur_max
        best_dst = None
        best_new_conv = None
        best_new_p = None

        for dst in dsts:
            if dst == src:
                continue
            # exact O(n) update of convolution and probability vector
            new_conv = conv.copy()
            new_p = p.copy()

            new_conv[src:src + n] -= 2.0 * epsilon * p
            new_conv[dst:dst + n] += 2.0 * epsilon * p
            new_conv[2 * src] += epsilon * epsilon
            new_conv[2 * dst] += epsilon * epsilon
            new_conv[src + dst] -= 2.0 * epsilon * epsilon

            new_p[src] -= epsilon
            new_p[dst] += epsilon
            if new_p[src] < 0.0:
                new_p[src] = 0.0

            new_max = new_conv.max()
            if new_max < best_cand_max:
                best_cand_max = new_max
                best_dst = dst
                best_new_conv = new_conv
                best_new_p = new_p

        # ----- if no improving move was found, try a random destination with a
        #       simulated‑annealing acceptance rule                             ----
        accept = False
        if best_dst is not None:
            accept = True
        else:
            dst = rng.choice(n, p=prob)
            if dst != src:
                cand_conv = conv.copy()
                cand_p = p.copy()

                cand_conv[src:src + n] -= 2.0 * epsilon * p
                cand_conv[dst:dst + n] += 2.0 * epsilon * p
                cand_conv[2 * src] += epsilon * epsilon
                cand_conv[2 * dst] += epsilon * epsilon
                cand_conv[src + dst] -= 2.0 * epsilon * epsilon

                cand_p[src] -= epsilon
                cand_p[dst] += epsilon
                if cand_p[src] < 0.0:
                    cand_p[src] = 0.0

                cand_max = cand_conv.max()
                delta = cand_max - cur_max
                if delta <= 0 or np.random.rand() < np.exp(-delta / temperature):
                    accept = True
                    best_cand_max = cand_max
                    best_dst = dst
                    best_new_conv = cand_conv
                    best_new_p = cand_p

        # ----- commit the accepted move -----------------------------------------
        if accept:
            p = best_new_p
            conv = best_new_conv
            cur_max = best_cand_max

            if cur_max < best_max:
                best_max = cur_max
                best_score = 2.0 * n * best_max
                best_p = p.copy()
                last_improve = iteration
                print(
                    f"[{time.time() - start:.2f}s] iter={iteration:,} "
                    f"new best score={best_score:.6f}"
                )

        # ----- occasional re‑normalisation (drift guard) ------------------------
        if iteration % 5000 == 0:
            total = p.sum()
            if abs(total - 1.0) > 1e-12:
                p /= total
                conv = np.convolve(p, p)

        # ----- temperature decay ------------------------------------------------
        temperature = T0 * (0.9995 ** iteration)

        # ----- plateau handling – jitter to break stagnation --------------------
        if iteration - last_improve > 200_000:
            p = best_p.copy()
            jitter = 0.005 * p * (rng.random(p.shape) - 0.5)   # ≤0.5 % jitter
            p = np.maximum(p + jitter, 0.0)
            p /= p.sum()
            conv = np.convolve(p, p)
            last_improve = iteration
            print(
                f"[{time.time() - start:.2f}s] plateau – re‑jittered at iter {iteration}"
            )

    # ------------------------------------------------------------------
    # 4) final normalisation and output
    # ------------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0:
        best_p = best_p / total
    else:
        best_p = np.full(n, 1.0 / n)

    # The verifier treats the vector as a step‑function; a probability vector
    # (Σ=1) is perfectly acceptable.
    return [float(v) for v in best_p]
