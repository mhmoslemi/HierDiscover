"""
Hybrid optimizer for the self‑convolution objective

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases

1️⃣  **Rich warm‑starts** – a large catalogue of analytically‑motivated
    shapes (uniform, tent, arcsine‑like, power‑law centre/edge,
    Gaussian, exponential, Beta‑law) together with random Dirichlet
    draws and simple mixtures.  All candidates are normalised to a
    probability simplex and the best few (according to the exact
    objective) are kept.

2️⃣  **Deterministic‑annealing mirror descent** – starting from each
    selected warm‑start we minimise a *smoothed* version of the
    max‑convolution using a soft‑max (log‑sum‑exp) approximation.
    For a temperature γ the objective is

        Φγ(a) = γ·log Σ exp(conv(a,a)/γ) .

    Its gradient is a weighted cross‑correlation which can be obtained
    with one FFT per iteration.  We perform exponentiated‑gradient
    (mirror descent on the simplex) with a momentum term and a step
    size ηt = η0/√t, while slowly decreasing γ (geometric cooling).
    The best iterate seen is recorded.

3️⃣  **Cheap simulated‑annealing polish** – a short Metropolis walk
    (moving a random packet of mass between two indices) is run
    in the last few seconds of the budget to escape shallow minima.

All heavy work is done with FFT‑based convolutions, the total running
time never exceeds the supplied budget (default 1000 s).  The returned
list contains non‑negative floats and will be fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Parametric shapes (all returned already normalised)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # in [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # in [0,1]
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _shape_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) density on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (beta - 1))
    a = np.maximum(a, 0.0)
    a += 1e-16
    return a / a.sum()

# Smooth‑max mirror‑descent
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        gamma_init: float = 0.05,
                        gamma_decay: float = 0.9,
                        step0: float = 0.5,
                        beta: float = 0.9,
                        dec_step: int = 200) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a soft‑max approximation of the
    max‑convolution.  Returns the best iterate encountered.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # current objective
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SDP iter={it:,} score={cur_score:.6f}")

        # temperature schedule (geometric cooling)
        gamma = gamma_init * (gamma_decay ** ((it - 1) // dec_step))
        if gamma < 1e-10:
            gamma = 1e-10

        # smooth‑max weights (soft‑max on conv values)
        exp_vals = np.exp((conv - cur_max) / gamma)   # stability via shift
        w_sum = exp_vals.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_vals / w_sum

        # gradient = 2 * cross‑correlation(w, a)
        a_rev = a[::-1]
        Lc = _next_pow2(len(w) + len(a_rev) - 1)
        w_fft = np.fft.rfft(w, n=Lc)
        a_rev_fft = np.fft.rfft(a_rev, n=Lc)
        conv_wa = np.fft.irfft(w_fft * a_rev_fft, n=Lc)[: len(w) + len(a_rev) - 1]

        # G_i = 2 * conv_wa[n-1 - i]  for i = 0 … n‑1
        G = 2.0 * conv_wa[n - 1::-1][:n]

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * G
        step = step0 / math.sqrt(it)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-step * momentum)
        a = _project_simplex(a)

    return best_a, best_score

# Light simulated‑annealing polish (identical to previous version)

def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # simple exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # pick source proportionally to mass
        dst = rng.integers(n)              # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 5) Light simulated‑annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation & conversion to plain list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
