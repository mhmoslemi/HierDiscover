def _gradient_submax(a, L_conv, L_corr):
    conv = _conv_self(a, L=L_conv)  # length 2n-1
    max_val = conv.max()
    # tolerance for mask:
    eps = max_val * 1e-12  # maybe 0 if max large?
    mask = np.where(conv >= max_val - eps, 1.0, 0.0)
    # correlation:
    a_rev = a[::-1]
    # compute convolution mask * a_rev
    # To compute cross-correlation we need convolution of mask and a_rev:
    # Compute FFTs
    corr_full = np.fft.irfft(np.fft.rfft(mask, n=L_corr) * np.fft.rfft(a_rev, n=L_corr), n=L_corr)
    start = a.shape[0] - 1
    grad = 2.0 * corr_full[start:start + a.shape[0]]
    return grad, max_val

def _subgradient_descent(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    # Adam vars
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # base learning rate
    lr = 0.3  # should be tuned

    while time.time() < deadline:
        t += 1
        grad, max_val = _gradient_submax(a, L_conv, L_corr)
        # Evaluate score
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)  # sum approx 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={t} score={cur_score:.6f}")

        # Adam-like update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # projection to simplex
        a = _project_simplex(a)

        # Add occasional random perturbation to avoid stagnation
        if t % 5000 == 0:
            # random Gaussian noise proportional to a
            noise = 1e-3 * rng.standard_normal(n) * a
            a = np.clip(a + noise, 0.0, None)
            a = _project_simplex(a)

    return best_a, best_score

"""
Propose a non‑negative step function (list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The routine consists of three stages:

1️⃣  A fast parametric sweep over a rich family of monotone shapes
    (geometric × power, optionally mixed with the uniform distribution)
    to obtain a good starting point.

2️⃣  An exact sub‑gradient descent on the true “max‑conv” term.
    The sub‑gradient of   max_k (a*a)[k]   is obtained analytically
    (it is a correlation with the mask of maximal indices) and the
    iterate is projected onto the probability simplex after each step.
    A simple 1/√t step‑size schedule is used; occasional Gaussian jitter
    helps to escape plateaus.

3️⃣  A final hill‑climb that repeatedly transfers a small fraction of the
    mass from a random source coordinate to a random destination.
    The allowed transfer size decays with time, allowing both coarse
    and fine adjustments.  Only strictly improving moves are kept.

The whole search respects the 1000 s budget, prints progress, and returns
the best sequence found (normalised to sum ≈ 1, which is scale‑free for the
objective).  The code is deliberately different from the previous
implementation – it uses the exact max sub‑gradient instead of a smooth‑max
approximation and a simpler annealing hill‑climber.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (the sum is already ≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution term
# ----------------------------------------------------------------------
def _max_mask_gradient(a: np.ndarray, L_conv: int, L_corr: int) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_val) where
        max_val = max_k (a*a)[k]
        grad_i = 2 * sum_{k : (a*a)[k] = max_val} a_{k-i}
    The gradient is obtained via FFT‑based correlation.
    """
    # 1. full self‑convolution
    conv = _conv_self(a, L=L_conv)                     # length 2n‑1
    max_val = conv.max()

    # 2. mask of maximal entries (use a tiny tolerance)
    eps = max_val * 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)  # shape (2n‑1,)

    # 3. correlation mask_rev * a  → gradient
    # mask_rev = mask reversed
    mask_rev = mask[::-1]
    # convolution a * mask_rev (full) using FFT
    A_fft = np.fft.rfft(a, n=L_corr)
    M_fft = np.fft.rfft(mask_rev, n=L_corr)
    conv_full = np.fft.irfft(A_fft * M_fft, n=L_corr)
    # keep only the relevant part (length 3n‑2)
    full_len = a.shape[0] + mask_rev.shape[0] - 1
    conv_full = conv_full[:full_len]

    M = conv.shape[0]               # = 2n‑1
    idx = M - 1 - np.arange(a.shape[0])   # indices M‑1, M‑2, …, M‑n
    grad = 2.0 * conv_full[idx]
    return grad, max_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on   max_k (a*a)[k]   with projection onto the simplex.
    Step size η_t = η0 / sqrt(t).  The best sequence seen so far is returned.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # FFT sizes (fixed)
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    eta0 = 0.5                     # initial step‑size
    it = 0

    while time.time() < deadline:
        it += 1
        grad, max_val = _max_mask_gradient(a, L_conv, L_corr)

        cur_score = 2.0 * n * max_val / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                f"score={cur_score:.6f}"
            )

        # diminishing step size
        eta = eta0 / math.sqrt(it)
        a = a - eta * grad
        a = _project_simplex(a)

        # occasional jitter to escape flat regions
        if it % 2000 == 0:
            noise = 1e-3 * rng.standard_normal(n) * a
            a = np.clip(a + noise, 0.0, None)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (mass transfers)
# ----------------------------------------------------------------------
def _final_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Randomly moves a fraction of mass from a source coordinate to a destination.
    The permissible fraction decays linearly with remaining time, allowing
    coarse early moves and fine final adjustments.
    Only strictly improving moves are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    rng = np.random.default_rng()
    n = a.shape[0]
    total_hc_time = deadline - time.time()
    if total_hc_time <= 0:
        return best_a, best_score

    max_frac_start = 0.02   # up to 2 % of source mass at the beginning
    min_frac_end = 1e-5     # down to 0.001 % near the end

    while time.time() < deadline:
        remaining = deadline - time.time()
        frac = max_frac_start * (remaining / total_hc_time) + min_frac_end * (1 - remaining / total_hc_time)

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb improved score={best_score:.6f}"
            )

    return best_a, best_score

            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:

    # ------------------------------------------------------------------
