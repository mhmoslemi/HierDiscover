"""
Improved search for low‐evaluation step function.

Algorithm
---------

1. Choose a large power‑of‑two length `n` (65536).  The search works on the
   full vector, not on a down‑sampled version, so the discretisation error
   of the continuous optimum is tiny.

2. Initialise a small pool of promising vectors:
   * the globally stored starter `height_sequence_1` (if present);
   * a few analytic shapes generated by the power‑ramp/plateau family;
   * a handful of random Dirichlet draws.
   All vectors are normalised to sum = 1 (the objective is scale‑invariant).

3. For every candidate run a **smooth Adam optimiser**.
   The non‑smooth “max” in the objective is replaced by a soft‑max
   `log‑sum‑exp` with a temperature `τ`.  The gradient of the smooth
   surrogate can be written as a convolution `2·a * w_rev` and is obtained
   with a single FFT pair per iteration.  Adam’s moment estimates give
   an adaptive step‑size and the iterate is projected onto the probability
   simplex after each update.

   The temperature `τ` is increased slowly so the optimiser first makes
   large‑scale moves and later concentrates on the true maximum.

4. Keep the best iterate seen across all restarts.

5. Finish with a cheap simulated‑annealing polish that swaps tiny packets
   of mass between random positions using a Metropolis rule.

The whole procedure respects the 1000 s wall‑time limit; all heavy work is
done with FFTs (O(n log n)) and the code uses only `numpy` and `scipy`‑free
primitives, so it runs on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using a real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    L = _next_pow2(n)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[:n]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # Numerical safeguard
        w = np.full_like(v, 1.0 / n)
    return w


def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape (length n) built from two power ramps that meet a
    plateau of height 1.  The vector is normalised to sum = 1.
    Parameters
    ----------
    p     – exponent of the side ramps (p>0).
    alpha – relative plateau width ∈[0,0.5]; 0 → pure tent,
            0.5 → uniform.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Smooth gradient (soft‑max approximation of the max)
# ----------------------------------------------------------------------


def _conv_self_and_grad(a: np.ndarray, tau: float):
    """
    Compute self‑convolution of a, the smooth max weight vector and the
    gradient of the soft‑max surrogate.
    Returns (conv_self, max_conv, gradient) where
      - conv_self is the full linear convolution a*a (length 2·n‑1),
      - max_conv is its maximum,
      - gradient is ∇_a (soft‑max) .
    """
    n = a.shape[0]
    # Pad length large enough for both convolutions
    L = _next_pow2(3 * n - 2)

    # FFT of a (shared)
    Af = np.fft.rfft(a, n=L)

    # ----- a * a -------------------------------------------------------
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    max_conv = conv_full.max()

    # ----- soft‑max weights --------------------------------------------
    # w_k ∝ exp(tau·(conv_k - max_conv))
    # we keep them normalised to sum 1 (they form a probability vector)
    exp_weights = np.exp(tau * (conv_full - max_conv))
    w = exp_weights / exp_weights.sum()

    # ----- convolution a * w_rev ----------------------------------------
    w_rev = w[::-1]
    Bf = np.fft.rfft(w_rev, n=L)
    conv_aw_rev = np.fft.irfft(Af * Bf, n=L)[: n + len(w_rev) - 1]   # length = 3·n‑2

    offset = len(w_rev) - 1           # = 2·n‑2
    grad = 2.0 * conv_aw_rev[offset - np.arange(n)]

    return conv_full, max_conv, grad


# ----------------------------------------------------------------------
# Adam optimiser (smooth surrogate)
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  tau0: float = 5.0,
                  lr0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the smooth surrogate.
    Returns the best iterate seen and its (exact) objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    t = 0
    tau = tau0
    lr = lr0

    while time.time() < deadline:
        t += 1

        conv_full, max_conv, grad = _conv_self_and_grad(a, tau)

        # Adam update with bias correction
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

        cur_score = 2.0 * n * max_conv          # sum(a)=1 ⇒ denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} score={cur_score:.6f}")

        # slowly increase the temperature so the surrogate tightens around the true max
        if t % 250 == 0:
            tau = min(tau * 1.25, 40.0)

    return best_a, best_score

def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Metropolis moves of tiny mass packets between random coordinates."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of attempts is bound only by the remaining time budget
    it = 0
    while time.time() < deadline:
        it += 1
        # a very fast cooling schedule
        T = 0.02 * (1e-6 / 0.02) ** (it / 200_000)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

"""
Improved search for a low‑evaluation step function.

Algorithm
---------

1. Use a large power‑of‑two length `n = 65536`.  The problem is scale‑invariant,
   so we keep `∑ a_i = 1`.  The factor `2·n` then appears in the objective.

2. Build a small pool of starting vectors:
   * a globally stored starter `height_sequence_1` (if it has the right size);
   * a few analytic shapes from the “power‑ramp + plateau’’ family;
   * several random Dirichlet draws.

3. Optimise each candidate with a **smooth Adam optimiser**.
   The non‑smooth `max` in the objective is replaced by a soft‑max
   (`log‑sum‑exp`) with temperature `τ`.  The gradient of the smooth
   surrogate is `2·a * w_rev` and can be obtained with a single FFT pair
   per iteration.  After the Adam step the iterate is projected onto the
   probability simplex.  `τ` is slowly increased so the optimiser first
   makes coarse moves and later focuses on the true maximum.

4. Keep the best iterate seen across all restarts.

5. Finish with a cheap simulated‑annealing polish that moves tiny mass
   packets between random positions using a Metropolis rule.

All heavy work is performed with FFTs (`O(n log n)`) and only `numpy`
is required, so the whole search respects the 1000 s wall‑time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using a real FFT."""
    L = _next_pow2(a.shape[0] + b.shape[0] - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: a.shape[0] + b.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerical safeguard
        w = np.full_like(v, 1.0 / n)
    return w


def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape built from two power ramps meeting a flat plateau.
    The result is normalised to sum = 1.

    Parameters
    ----------
    p     – exponent of the side ramps (p > 0).
    alpha – relative plateau width ∈[0,0.5]; 0 → pure tent, 0.5 → uniform.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # fix possible off‑by‑one caused by rounding
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0.0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (∑seq)²."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Smooth gradient (soft‑max surrogate of the max)
# ----------------------------------------------------------------------


def _conv_self_and_grad(a: np.ndarray, tau: float):
    """
    Compute:
        * the self‑convolution a*a,
        * its maximum,
        * the gradient of the soft‑max surrogate with temperature τ.

    Returns (conv_self, max_conv, grad) where
        conv_self – length 2·n‑1,
        max_conv  – scalar,
        grad      – length n vector.
    """
    n = a.shape[0]
    L = _next_pow2(3 * n - 2)          # big enough for both convolutions

    # FFT of a (shared)
    Af = np.fft.rfft(a, n=L)

    # ----- a * a -------------------------------------------------------
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    max_conv = conv_full.max()

    # ----- soft‑max weights --------------------------------------------
    exp_weights = np.exp(tau * (conv_full - max_conv))
    w = exp_weights / exp_weights.sum()               # ⟨w⟩ = 1, length 2·n‑1

    # ----- a * w_rev ---------------------------------------------------
    w_rev = w[::-1]
    Bf = np.fft.rfft(w_rev, n=L)
    conv_aw_rev = np.fft.irfft(Af * Bf, n=L)[: n + len(w_rev) - 1]   # length = 3·n‑2

    offset = len(w_rev) - 1          # = 2·n‑2
    grad = 2.0 * conv_aw_rev[offset - np.arange(n)]

    return conv_full, max_conv, grad

# Adam optimiser (smooth surrogate)
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,

    Projected Adam optimisation of the soft‑max surrogate.
    Returns the best iterate (as a numpy array) and its *exact* objective.
    """

                print(f"[{time.time() - start_time:.2f}s] New best after Adam: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Light simulated‑annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing polish")
        annealed, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = annealed.copy()
            print(f"[{time.time() - start_time:.2f}s] New best after anneal: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
