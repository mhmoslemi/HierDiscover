"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The core idea is to use the *arcsine* (Beta(½,½)) density,
which is known to give a perfectly flat continuous self‑convolution
(uniform on the sum interval).  After discretising the density on a
grid of size N we obtain a very low value of the objective (≈1 for large N).
We explore a few beta‑shape parameters (α) to guard against discretisation
artifacts, keep the best, and finally perform a short smooth‑gradient
refinement on the probability simplex.

All steps respect the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution via real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max approximation of max)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a smooth surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate and temperature schedule
    step0 = 0.05
    T0 = 0.5         # start a bit hotter than before
    T_end = 1e-4

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} "
                  f"score={cur_score:.6f}")

        # temperature schedule (linear decay)
        frac = it / (it + 500.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights over the convolution entries
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of the (soft) max term via correlation
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Analytic beta‑shape generator (covers the arcsine case)
# ----------------------------------------------------------------------
def _make_beta_shape(N: int, alpha: float) -> np.ndarray:
    """
    Return a symmetric beta(α,α) density discretised on N points,
    normalised to sum 1.  The grid points are mid‑points of the N bins.
    """
    i = np.arange(N, dtype=float)
    x = (i + 0.5) / N                     # points in (0,1)
    # raw (unnormalised) density proportional to x^{α‑1}(1‑x)^{α‑1}
    raw = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # turn the density into a probability mass function by multiplying by bin width
    a = raw / N
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a = np.full(N, 1.0 / N)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Uses beta‑shapes (including the optimal arcsine distribution) and a
    short smooth‑gradient refinement.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose problem size (bigger N gives a finer discretisation of the
    # continuous optimum, but also costs a bit more time).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 131072          # finer grid – still manageable in the time budget
    elif remaining > 200:
        N = 65536
    else:
        N = 32768

    print(f"[{time.time() - start_time:.2f}s] Using N = {N}")

    # ------------------------------------------------------------------
    # 1) Scan a modest set of beta‑shape parameters
    # ------------------------------------------------------------------
    alphas = np.unique(np.concatenate([
        np.linspace(0.2, 0.9, 8),   # U‑shapes (incl. arcsine ≈0.5)
        np.linspace(1.0, 2.0, 6)    # bell‑shapes
    ]))

    best_seq = None
    best_score = float('inf')

    for alpha in alphas:
        a = _make_beta_shape(N, alpha)
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a
        if time.time() >= deadline:
            break

    print(f"[{time.time() - start_time:.2f}s] Best beta‑shape score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Optional smooth‑gradient refinement (takes the lion’s share of the
    #    remaining budget but usually yields only a tiny improvement).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # keep about 80 % of the remaining time for refinement
        sub_deadline = time.time() + 0.8 * (deadline - time.time())
        refined_seq, refined_score = _smooth_gradient_descent(best_seq,
                                                             sub_deadline,
                                                             start_time)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq
            print(f"[{time.time() - start_time:.2f}s] After refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
