"""
Improved step‑function search.

We combine three ideas:

1️⃣  A broader parametric sweep (geometric, power, beta families) plus a few
    random Dirichlet starts.

2️⃣  Multiple short Adam‑style projected gradient runs.  The Adam update uses
    the sub‑gradient from *the single* most active convolution shift (the
    index where the maximum is achieved).  This gives a clearer descent
    direction than averaging over all near‑maximal shifts.

3️⃣  A final stochastic hill‑climb that mixes tiny mass‑transfer moves,
    occasional full swaps of two positions and a simple simulated‑annealing
    acceptance rule.  The temperature is slowly cooled, allowing the
    algorithm to escape shallow plateaus.

All stages respect the overall 1000 s budget.  The routine returns a plain
list of Python floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ---------------------------------------------------------------
# Basic linear‑algebra utilities
# ---------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ---------------------------------------------------------------
# Sub‑gradient of the *single* maximal convolution entry
# ---------------------------------------------------------------
def _subgradient_maxconv_single(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal entry of a*a.
    Uses only the (first) index where the maximum is attained.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))                     # index of the active shift
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)      # indices that appear in this conv term
    grad = np.zeros_like(a)
    grad[idx] += a[k - idx]                      # derivative w.r.t each a_i
    grad = 2.0 * grad                             # each product contributes twice
    return grad


# ---------------------------------------------------------------
# Adam optimiser on the simplex (single‑active‑shift gradient)
# ---------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   init_score: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimiser that stays on the simplex.
    Returns the best sequence and its raw 2·n·max(conv) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters – a modest learning‑rate schedule
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.3                     # start a little aggressive

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        # evaluate current objective (sum(a) is 1 by projection)
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # gradient of the maximal entry
        grad = _subgradient_maxconv_single(a, conv)

        # Adam update (with a simple time‑based decay of lr)
        lr = lr0 / (1.0 + 1e-5 * it)   # slowly shrink learning‑rate

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection back onto the simplex
        a = _project_simplex(a)

        # give a tiny random nudge if we stagnate for a while
        if it - last_improve > 3000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ---------------------------------------------------------------
# Final hill‑climb with swaps + simulated annealing
# ---------------------------------------------------------------
def _hill_climb_sa(start_a: np.ndarray,
                   start_time: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that mixes
        * tiny mass transfers,
        * full swaps of two entries,
    and accepts occasional worsening moves according to a simple
    temperature schedule.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    cur_score = best_score

    rng = np.random.default_rng()
    # start temperature is a few percent of the current score
    T = best_score * 0.02
    decay = 0.9995

    while time.time() < deadline:
        move_type = rng.random()
        a_prop = a.copy()

        if move_type < 0.7:          # 70 % – tiny mass transfer (as before)
            src = rng.integers(n)
            if a_prop[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            frac = rng.random() * 0.02 + 1e-6          # up to 2 %
            eps = a_prop[src] * frac
            a_prop[src] -= eps
            a_prop[dst] += eps
        else:                         # 30 % – full swap of two entries
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        # keep sum = 1 automatically (both moves preserve the total)
        a_prop = np.maximum(a_prop, 0.0)
        a_prop = _project_simplex(a_prop)   # re‑project just in case numerical drift

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())

        # accept if better, otherwise with probability exp(-Δ/T)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] SA hill‑climb score={best_score:.6f}")

        # cool the temperature
        T *= decay

    return best_a, best_score

    # -----------------------------------------------------------
    # baseline (uniform)
    # -----------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = float(2.0 * n * _conv_self(best_seq, L_fft).max())
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # -----------------------------------------------------------
    # optional global starter (if provided by the framework)
    # -----------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s == 0:
                starter = uniform.copy()
            else:
                starter = starter / s
            cur = float(2.0 * n * _conv_self(starter, L_fft).max())
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1️⃣  Extended parametric sweep
    # -----------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric families (including a mixture with the uniform)
    eps_grid = np.logspace(-5, -1, 250)               # finer grid
    for eps in eps_grid:
        if time.time() >= deadline - 40:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        if geo.sum() == 0:
            continue
        geo = geo / geo.sum()
        for lam in np.linspace(0.0, 1.0, 13):          # mixture weight
            seq = lam * uniform + (1.0 - lam) * geo
            seq = seq / seq.sum()
            cur = float(2.0 * n * _conv_self(seq, L_fft).max())
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={cur:.6f}")

    # power families
    p_grid = np.linspace(-2.0, 3.0, 90)
    for p in p_grid:
        if time.time() >= deadline - 40:
            break
        pow_seq = np.power(idx + 1, p)
        if pow_seq.sum() == 0:
            continue
        pow_seq = pow_seq / pow_seq.sum()
        cur = float(2.0 * n * _conv_self(pow_seq, L_fft).max())
        if cur < best_score:
            best_score = cur
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={cur:.6f}")

    # beta‑type families (two‑parameter polynomial shape)
    beta_vals = [0.0, 0.5, 1.0, 1.5, 2.0, 3.0, 4.0]
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 40:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            if beta_seq.sum() == 0:
                continue
            beta_seq = beta_seq / beta_seq.sum()
            cur = float(2.0 * n * _conv_self(beta_seq, L_fft).max())
            if cur < best_score:
                best_score = cur
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p} q={q} score={cur:.6f}")

    # -----------------------------------------------------------
    # 2️⃣  Random Dirichlet starts + several Adam runs
    # -----------------------------------------------------------
    # allocate a chunk of time for the Adam phase
    hill_time = 150.0                         # seconds reserved for final hill‑climb
    adam_deadline = deadline - hill_time
    adams_performed = 0

    # a pool of initial sequences: best so far, uniform, and a few random Dirichlet draws
    init_pool = [best_seq, uniform]
    for _ in range(12):
        if time.time() >= adam_deadline - 5:
            break
        # draw concentration from a log‑uniform range → diverse sparsity levels
        conc = 10 ** rng.uniform(-1, 1)        # between 0.1 and 10
        rand_seq = rng.gamma(conc, size=n)
        rand_seq = rand_seq / rand_seq.sum()
        init_pool.append(rand_seq)

    # run Adam from each candidate (if time permits)
    for start_seq in init_pool:
        if time.time() >= adam_deadline:
            break
        cur_score = float(2.0 * n * _conv_self(start_seq, L_fft).max())
        # give each run a personal deadline (about 30 s or the remaining time)
        per_run_deadline = min(adam_deadline, time.time() + 30.0)
        a_opt, sc_opt = _adam_optimize(start_seq, start_time, per_run_deadline, cur_score)
        adams_performed += 1
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam #{adams_performed} improvement score={sc_opt:.6f}")

    # -----------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climb (mass transfers + swaps + SA)
    # -----------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final SA hill‑climb")
        a_hc, sc_hc = _hill_climb_sa(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # -----------------------------------------------------------
    # Normalise and return a plain Python list
    # -----------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
