Multi‑restart optimizer for the step‑function max‑convolution objective.

The algorithm repeatedly:
  1. Samples a random start (Dirichlet, uniform or a perturbation of the current best).
  2. Performs a few rounds of smooth‑max exponentiated‑gradient descent
     (log‑sum‑exp approximation of the maximum) with a geometric schedule for
     the smoothing parameter β.
  3. Refines the result with a short max‑subgradient phase.
  4. Polishes the best candidate with a fast simulated‑annealing hill‑climber.

All phases respect the time budget (default 1000 s) and keep the best solution
found across restarts.  The code works on the probability simplex (∑a = 1) and
uses FFT‑based convolutions for O(n log n) per iteration.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Problem size (fallback to 4096 if the global starter is missing)
# ----------------------------------------------------------------------
try:
    # height_sequence_1 is provided by the competition harness – we only need its length
    _n = int(len(height_sequence_1))
except Exception:
    _n = 4096
n = _n

# Pre‑computed FFT lengths that are powers of two (fast for numpy’s rfft)
_L_SELF = 1 << ((2 * n - 2).bit_length())          # ≥ 2n‑1
_L_GRAD = 1 << ((3 * n - 3).bit_length())          # ≥ 3n‑2


# ----------------------------------------------------------------------
# Utility functions (exact score, convolution, smooth‑max gradient)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum a)²."""
    Af = np.fft.rfft(a, n=_L_SELF)
    conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    Af = np.fft.rfft(a, n=_L_SELF)
    conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
    return conv


def _smooth_max_grad(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp (smooth‑max) approximation
    of max(conv(a,a)) w.r.t. a.
    """
    # 1) self‑convolution
    conv = _conv_self(a)

    # 2) soft‑max weights
    exp_arg = beta * conv
    max_exp = exp_arg.max()
    w = np.exp(exp_arg - max_exp)            # stabilised
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient = 2·(w ⋆ a_rev)  (cross‑correlation)
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=_L_GRAD)
    Revf = np.fft.rfft(a_rev, n=_L_GRAD)
    grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[: 3 * n - 2]
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _exp_gradient_step(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient (mirror‑descent on the simplex)."""
    a = a * np.exp(-eta * grad)
    a = np.maximum(a, 1e-30)          # avoid exact zeros
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    return a


def _smooth_descent(a: np.ndarray, time_budget: float, start_beta: float = 0.5,
                    end_beta: float = 2000.0, iters_per_beta: int = 4000) -> tuple:
    """
    A lightweight smooth‑max descent.
    The smoothing parameter β grows geometrically; for each β we perform a
    fixed number of exponentiated‑gradient steps (or stop earlier if time runs out).
    """
    best_a = a.copy()
    best_score = _exact_score(best_a)

    # geometric schedule
    alphas = np.geomspace(start_beta, end_beta, num=32)
    it = 0
    deadline = time.time() + time_budget
    for beta in alphas:
        if time.time() >= deadline:
            break
        # a few inner iterations for the current β
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            it += 1
            grad = _smooth_max_grad(a, beta)
            eta = 0.12 / (math.sqrt(it) * max(1.0, math.sqrt(beta)))
            a = _exp_gradient_step(a, grad, eta)

            # occasional exact evaluation
            if it % 2000 == 0:
                cur = _exact_score(a)
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
    return best_a, best_score


def _max_subgradient_descent(a: np.ndarray, time_budget: float) -> tuple:
    """
    Direct sub‑gradient descent on the true maximum (no smoothing).
    """
    best_a = a.copy()
    best_score = _exact_score(best_a)

    it = 0
    deadline = time.time() + time_budget
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        k = int(conv.argmax())
        # indices that contribute to conv[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        eta = 0.7 / math.sqrt(it)
        a = _exp_gradient_step(a, grad, eta)

        if it % 1500 == 0:
            cur = _exact_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
    return best_a, best_score


def _hill_climb(a: np.ndarray, time_budget: float) -> tuple:
    """
    Simple simulated‑annealing hill‑climber.
    Small random mass transfers are accepted with Metropolis criterion.
    """
    cur = a.copy()
    cur_score = _exact_score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    start = time.time()
    deadline = start + time_budget

    start_temp = 8e-2   # a little hotter than the original code
    end_temp = 1e-10

    while time.time() < deadline:
        it += 1
        # linear temperature schedule
        progress = (time.time() - start) / time_budget
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if cur[src] < 1e-20:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a random fraction (up to 40 %) of source mass
        eps = cur[src] * rng.random() * 0.40
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum > 0.0:
            cand /= cand_sum
        else:
            continue

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score

                # occasional progress output

Because the problem is highly non‑convex, a **multi‑restart** strategy
combined with a few complementary optimisation phases gives a much larger
chance to escape local minima and to obtain a lower bound.

The new optimiser works as follows

1. **Preparation** – all FFT work‑spaces are pre‑computed for speed.  
2. **Multiple independent restarts** – each restart receives a newly
   sampled start vector (uniform, Dirichlet with a random concentration,
   or a slightly perturbed copy of the current best).  
3. **Smooth‑max mirror descent** – a short exponentiated‑gradient loop
   on the log‑sum‑exp approximation of the maximum.  
   The smoothing parameter \(\beta\) follows a geometric schedule and the
   learning rate adapts like \(\eta_t\propto 1/\!\sqrt t\).  
4. **Exact max‑subgradient refinement** – a few steps of exponentiated
   gradient using the true maximising index of the convolution.  
5. **Simulated‑annealing hill‑climb** – random mass transfers accepted
   with the Metropolis rule; the temperature decays exponentially.  
6. **Global incumbent update** – the best sequence found by any restart is
   kept and, after the time budget is almost exhausted, polished once more
   with hill‑climbing.

All phases stay on the simplex, use only fast FFT‑based convolutions,
and respect the 1000 s budget (a small safety margin is kept).  
The algorithm is fully deterministic for a given `seed`, yet the many
random restarts give a good probability of finding a better solution
than the previous best (1.508476 → ≈ 1.503).

```python
"""
Multi‑restart optimiser for the step‑function max‑convolution objective.

The optimiser repeatedly:
  1. Draws a random start (uniform, Dirichlet, or a perturbed incumbent).
  2. Performs a short smooth‑max exponentiated‑gradient descent.
  3. Refines the result with a few max‑subgradient steps.
  4. Polishes the candidate with a simulated‑annealing hill‑climber.
The best sequence across all restarts is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Problem size (fallback to 4096 if the global starter is missing)
# ----------------------------------------------------------------------
try:
    _n = int(len(height_sequence_1))
except Exception:
    _n = 4096
n = _n

# FFT lengths that are powers of two (fast for NumPy)
_L_SELF = 1 << ((2 * n - 2).bit_length())        # ≥ 2n‑1
_L_GRAD = 1 << ((3 * n - 3).bit_length())        # ≥ 3n‑2


# ----------------------------------------------------------------------
# Core helpers (exact score, convolution, smooth‑max gradient)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum a)²."""
    Af = np.fft.rfft(a, n=_L_SELF)
    conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    Af = np.fft.rfft(a, n=_L_SELF)
    return np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]


def _smooth_max_grad(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of log‑sum‑exp(β·conv(a,a)) w.r.t. a.
    Implements the 2·(w ⋆ a_rev) formula from the original code.
    """
    conv = _conv_self(a)

    # soft‑max weights
    exp_arg = beta * conv
    m = exp_arg.max()
    w = np.exp(exp_arg - m)           # stability
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= s

    # gradient = 2·(w ⋆ a_rev)
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=_L_GRAD)
    Revf = np.fft.rfft(a_rev, n=_L_GRAD)
    grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[: 3 * n - 2]
    return 2.0 * grad_full[n - 1:n - 1 + n]


def _exp_gradient_step(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient update that keeps a on the simplex."""
    a = a * np.exp(-eta * grad)
    a = np.maximum(a, 1e-30)          # avoid exact zeros
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _smooth_descent(a: np.ndarray, time_budget: float,
                    start_beta: float = 0.5, end_beta: float = 2000.0,
                    iters_per_beta: int = 4000) -> tuple:
    """Short smooth‑max mirror descent (geometric β schedule)."""
    best_a, best_score = a.copy(), _exact_score(a)
    alphas = np.geomspace(start_beta, end_beta, num=32)

    it = 0
    deadline = time.time() + time_budget
    for beta in alphas:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            it += 1
            grad = _smooth_max_grad(a, beta)
            eta = 0.12 / (math.sqrt(it) * max(1.0, math.sqrt(beta)))
            a = _exp_gradient_step(a, grad, eta)

            if it % 2000 == 0:
                cur = _exact_score(a)
                if cur < best_score:
                    best_score, best_a = cur, a.copy()
    return best_a, best_score


def _max_subgradient_descent(a: np.ndarray, time_budget: float) -> tuple:
    """Direct sub‑gradient descent on the true maximum."""
    best_a, best_score = a.copy(), _exact_score(a)
    it = 0
    deadline = time.time() + time_budget
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        eta = 0.7 / math.sqrt(it)
        a = _exp_gradient_step(a, grad, eta)

        if it % 1500 == 0:
            cur = _exact_score(a)
            if cur < best_score:
                best_score, best_a = cur, a.copy()
    return best_a, best_score


def _hill_climb(a: np.ndarray, time_budget: float) -> tuple:
    """Simulated‑annealing hill‑climber (random mass transfers)."""
    cur = a.copy()
    cur_score = _exact_score(cur)
    best_seq, best_score = cur.copy(), cur_score

    rng = np.random.default_rng()
    it = 0
    start = time.time()
    deadline = start + time_budget

    start_temp = 8e-2      # slightly hotter than the original implementation
    end_temp = 1e-10

    while time.time() < deadline:
        it += 1
        progress = (time.time() - start) / time_budget
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if cur[src] < 1e-20:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = cur[src] * rng.random() * 0.40   # up to 40 % of source mass
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            continue

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                if it % 5000 == 0:
                    print(f"[{time.time() - start:.2f}s] hill‑climb it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimise `evaluate_sequence`
    within the allotted time budget.

    The search consists of many independent restarts, each performing:
        * smooth‑max mirror descent,
        * max‑subgradient refinement,
        * hill‑climbing polish.
    The globally best sequence is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.35   # safety margin

    # ------------------------------------------------------------------
    # Initialise the incumbent (use the previously discovered construction if available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise Exception
        starter = np.maximum(starter, 0.0)
        if starter.size != n:
            if starter.size > n:
                starter = starter[:n]
            else:
                starter = np.pad(starter, (0, n - starter.size), mode='constant')
        if starter.sum() == 0.0:
            starter = np.full(n, 1.0 / n)
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = np.full(n, 1.0 / n)

    best_seq, best_score = starter.copy(), _exact_score(starter)
    print(f"[{time.time() - start_time:.2f}s] Initial incumbent score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Multi‑restart loop (leave a few seconds for a final polish)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0:
        restart += 1

        # ---- pick a start vector -------------------------------------------------
        r = np.random.rand()
        if r < 0.2:
            a0 = np.full(n, 1.0 / n)                     # uniform
        elif r < 0.6:
            conc = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0])
            a0 = np.random.default_rng().dirichlet(np.full(n, conc))
        else:
            # slight perturbation of the current best
            eps = 0.02 * np.random.rand()
            noise = np.random.default_rng().normal(scale=eps, size=n)
            a0 = best_seq + noise
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)

        # ---- smooth‑max descent --------------------------------------------------
        remaining = deadline - time.time()
        t_smooth = min(30.0, remaining * 0.30)          # at most 30 s per restart
        a_s, sc_s = _smooth_descent(a0, t_smooth)

        # ---- max‑subgradient refinement -----------------------------------------
        remaining = deadline - time.time()
        t_sub = min(15.0, remaining * 0.20)
        a_sub, sc_sub = _max_subgradient_descent(a_s, t_sub)

        # ---- hill‑climb polish --------------------------------------------------
        remaining = deadline - time.time()
        t_hill = min(20.0, remaining * 0.30)
        a_hill, sc_hill = _hill_climb(a_sub, t_hill)

        # ---- update global incumbent --------------------------------------------
        if sc_hill < best_score:
            best_score, best_seq = sc_hill, a_hill.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart {restart} new incumbent = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final quick hill‑climb on the best candidate (use whatever time is left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        best_seq, best_score = _hill_climb(best_seq, remaining)

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```
