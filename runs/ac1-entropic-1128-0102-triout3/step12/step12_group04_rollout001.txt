"""
Hybrid search for a low‑objective non‑negative step function.

1️⃣ Warm‑start – generate a collection of simple analytic shapes
   (tent, cosine, semicircle, Gaussian) **and** a dense sweep of a few
   one‑parameter families (power‑tent, sine‑power, beta‑power).  The best
   of these (according to the exact objective) seeds the optimisation.

2️⃣ Smoothed‑max Adam – the true objective is 2·n·max(conv(a,a)).
   We replace the hard max by a smooth log‑sum‑exp:
        Fα(a)=2·n·(1/α)·log Σi exp(α·conv_i)
   Its gradient is a weighted sum of the individual convolution‑gradients
   with softmax weights w_i∝exp(α·conv_i).  An Adam update is performed
   on the probability simplex (Euclidean projection).  The smoothing
   parameter α is increased gradually so that the optimiser first sees
   a gentle landscape and later concentrates on the true maximum.

3️⃣ Light simulated‑annealing polish – a cheap Metropolis walk moves a
   random packet of mass between two indices.  It runs on the remaining
   time budget to escape possible shallow minima.

The whole routine respects a 1000 s budget, works entirely with NumPy
(and a few helper functions), and finally returns a plain list of
non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT size, simplex projection, convolution, objective)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex { w≥0, Σw = 1 }."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·n‑1) using a real FFT.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a already sums to 1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed‑max gradient (softmax weights over convolution entries)
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float, L: int):
    """
    Returns (grad, conv) where grad is the gradient of the smoothed‑max
    objective 2·n·(1/α)·log Σ exp(α·conv_i) and conv is the full linear
    self‑convolution (length 2·n‑1).
    """
    n = a.shape[0]

    # 1) full convolution
    conv = _conv_self(a, L)                     # length 2n‑1

    # 2) softmax weights (numerically stable)
    max_c = conv.max()
    shifted = conv - max_c                     # non‑positive
    exp_w = np.exp(alpha * shifted)            # ≤ 1
    w = exp_w / exp_w.sum()                    # sum to 1

    # 3) gradient = 2 * (a_rev * w)   (restricted to the n entries of a)
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)

    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]    # size n
    return grad, conv


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          lr: float = 0.025,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed‑max surrogate.  Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)          # safety margin for all FFTs

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    alpha = 4.0                     # start smoothing relatively soft
    alpha_max = 500.0
    alpha_inc_every = 500          # iterations

    while time.time() < deadline:
        it += 1

        grad, conv = _smooth_max_gradient(a, alpha, L)

        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f} α={alpha:.2f}")

        # Gradient of the smoothed‑max objective
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # Projection onto simplex
        a = _project_simplex(a)

        # increase smoothing slowly
        if it % alpha_inc_every == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.12, alpha_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk that swaps a random packet of mass."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 60_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)            # source weighted by its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Warm‑start shape generators
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of analytic and parametric shapes."""
    x = np.linspace(0.0, 1.0, n, dtype=float)

    candidates = []

    # 1) Fixed analytic bases
    tent = np.minimum(x, 1.0 - x)
    cosine = np.sin(np.pi * x)                         # zero at ends
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

    for base in (tent, cosine, circ, gauss):
        base = np.maximum(base, 0.0)
        if base.sum() > 0:
            candidates.append(base / base.sum())

    # 2) Parameter sweeps (few dozen shapes – cheap)
    # power‑tent: (min(x,1‑x))**p
    for p in np.linspace(0.2, 1.4, 13):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape / shape.sum())

    # sine‑power: sin(πx)**p
    for p in np.linspace(0.2, 2.0, 19):
        shape = np.sin(np.pi * x) ** p
        candidates.append(shape / shape.sum())

    # beta‑style: (x*(1‑x))**p
    for p in np.linspace(0.2, 2.0, 19):
        shape = (x * (1.0 - x)) ** p
        candidates.append(shape / shape.sum())

    # 3) Random Dirichlet draws (a few)
    rng = np.random.default_rng()
    for _ in range(5):
        rnd = rng.random(n)
        candidates.append(rnd / rnd.sum())

    return candidates


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # Problem size – a power of two for FFT speed.
    n = 16384

    # --------------------------------------------------------------
    # 1) Warm‑start – analytic + parametric candidates
    # --------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # If a previous good construction is available, add it.
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                warm_starts.append(starter)
    except Exception:
        pass

    # Pick the best among the warm starts (exact objective).
    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Smoothed‑max Adam optimisation (most of the remaining budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate ≈85 % of what is left to the optimiser
        adam_deadline = time.time() + remaining * 0.85
        a_opt, score_opt = _smooth_adam_optimize(a0,
                                                 adam_deadline,
                                                 start_time,
                                                 lr=0.025,
                                                 beta1=0.9,
                                                 beta2=0.999,
                                                 eps=1e-8)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam best score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(a0, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            a0 = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal best={anneal_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final sanity checks and conversion to list
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    return [float(v) for v in a0]
