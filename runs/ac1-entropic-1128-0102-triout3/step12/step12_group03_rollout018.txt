"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The implementation uses three ideas that complement each other:

1. **Diverse seeding** – uniform, sinusoidal, power‑ramp shapes and many
   random draws (Dirichlet).  All candidates are normalised to sum 1 and the
   best few are kept.

2. **Fast smoothed descent** – a soft‑max surrogate of the max‑convolution
   is minimised with Adam.  The gradient is obtained by a single FFT‑based
   correlation, and after each step the vector is projected onto the
   probability simplex.

3. **Heavy annealing** – the current best candidate is polished with a
   simulated‑annealing walk that moves a tiny amount of mass from a
   “high‑contributing’’ index to a random destination.  The walk is long
   (up to a few hundred k steps) and uses an exponential cooling schedule.

All stages respect the 1000 s time budget and fall back gracefully if the
budget runs low.  The returned list is a plain Python list of non‑negative
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes (ramps, sinusoids, uniform)
# ----------------------------------------------------------------------
def _make_ramp(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau (normalised to sum 1)."""
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sinusoid(n: int, freq: float, phase: float) -> np.ndarray:
    """Positive sinusoid (shifted so it is non‑negative) normalised to sum 1."""
    x = np.arange(n, dtype=float) / n
    wave = 0.5 + 0.5 * np.sin(2.0 * math.pi * freq * x + phase)
    wave = np.maximum(wave, 0.0)
    if wave.sum() == 0:
        wave = np.full(n, 1.0 / n)
    else:
        wave = wave / wave.sum()
    return wave


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution (sum 1)."""
    return np.full(n, 1.0 / n)


def _random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet(1,…,1))."""
    vec = rng.random(n)
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0:
        vec = np.full(n, 1.0 / n)
    else:
        vec = vec / s
    return vec

# Soft‑max based gradient descent (smooth surrogate of max)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a smoothed max‑convolution surrogate."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.06          # a bit larger than before
    T0, T_end = 0.025, 5e-5

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = _objective(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD "
                  f"iter={it:,}  score={cur_score:.6f}")

        # temperature schedule (linear in iteration count)
        frac = it / (it + 2000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights over the convolution entries
        max_c = conv.max()
        w_raw = np.exp((conv - max_c) / T)
        w = w_raw / w_raw.sum()

        # gradient of the soft‑max surrogate:
        #   grad = 2 * (w ⊛ a_rev)   (⨂ = correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection
        a = _project_simplex(a)

    return best_a, best_score

        lo = max(0, k - (n - 1))

        a = _project_simplex(a)
