def _mirror_descent(a0, gamma, deadline, start_time):

    n = len(a)

        # compute smooth conv and gradient
        # conv = self convolution
        conv = _conv_self_fft(a)  # length 2n-1
        # compute log-sum-exp terms
        max_conv = conv.max()
        # stability
        exp_term = np.exp(gamma * (conv - max_conv))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp

        w_rev = w[::-1]
        # compute conv_w_rev_a
        conv_w_rev_a = _conv_fft(w_rev, a)

        # gradient of objective

                best_a = a.copy()

def propose_candidate(seed=0,budget_s=1000,**kwargs):
    import time, math, numpy as np
    start = time.time()
    deadline = start + budget_s - 0.5

    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # set length n as power of 2 for FFT efficiency:
    n = 131072   # 2**17

    # retrieve starter
    init_seq = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim==1 and starter.size>0:
            # upscale to n
            base_len = starter.size
            target_x = np.linspace(0,1,n)
            base_x = np.linspace(0,1,base_len)

            a = np.maximum(a,0)

                a = a / a.sum()

        pass

    # generate candidate pool
    pool = []   # list of (score, seq)

    # function to add candidate

        if s>0:

        else:

        if len(pool) < 5:

            pool.sort(key=lambda x: x[0])

                pool[-1] = (score, seq.copy())

    # param grids
    p_vals = np.concatenate([np.linspace(0.1, 4.0, 40), np.random.uniform(0.1,4.0,10)])

    for p in p_vals:

    # exponential shape

    for _ in range(200):

        add_candidate(a)

    best_seq = pool[0][1].copy()

    remaining = deadline - time.time()

        seeds = [seq for _, seq in pool[:2]]

    return [float(v) for v in best_seq]

def _conv_fft(x, y):
    n = x.shape[0]
    m = y.shape[0]
    L = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    z = np.fft.irfft(X * Y, n=L)[:n + m - 1]
    return z

def _smooth_gradient_descent(a0, gamma, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    C = 2.0 * n
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute smooth max and gradient wrt smooth max
        conv = _conv_self(a)
        max_conv = conv.max()
        # stable softmax
        exp_term = np.exp(gamma * (conv - max_conv))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp
        smooth_max = max_conv + np.log(sum_exp) / gamma
        # compute grad of smooth max
        w_rev = w[::-1]
        conv_w_rev_a = _conv_fft(w_rev, a)
        slice_start = n - 1
        slice_end = 2 * n - 1
        grad_smooth = 2.0 * conv_w_rev_a[slice_start:slice_end][::-1]

        sum_a = a.sum()
        # gradient of objective
        grad_f = C * (grad_smooth / (sum_a ** 2) - (2.0 * smooth_max / (sum_a ** 3)) * np.ones_like(a))

        # step size schedule
        eta = 0.3 / math.sqrt(it)

        # exponentiated update
        a = a * np.exp(-eta * grad_f)

        # project onto simplex (renorm)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        # occasional scoring
        if it % 200 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Smooth descent it={it:,} score={cur_score:.6f}")

    return best_a, best_score

def _make_exponential_shape(n, beta, alpha):
    # distance from centre normalized to [0,1]
    idx = np.arange(n, dtype=float) / (n-1)

    shape = shape / shape.sum()
    return shape

"""
Optimise a non‑negative step‑function for the supplied
``evaluate_sequence`` metric.

The approach consists of three stages:

1. **Fast coarse search** – on a modest size (n≈2¹⁵) we sweep a few
   analytical families (power ramps, exponential decay, Gaussians) and
   a handful of random Dirichlet draws.  The best few shapes are kept
   and interpolated to the final large size.

2. **Smooth surrogate descent** – the true objective uses ``max`` of the
   self‑convolution, which is non‑smooth.  We replace it by a
   log‑sum‑exp (LSE) surrogate with a large temperature parameter
   ``γ``.  The gradient of the surrogate can be written in closed form
   and is obtained via two FFT‑based convolutions.  An
   exponentiated‑gradient (mirror‑descent) update keeps the iterate on
   the probability simplex.  A diminishing step‑size (ηₜ∝1/√t) is used.
   Each run is time‑boxed; the best result over all seeds is retained.

3. **Light simulated‑annealing polish** – a final Metropolis walk moves
   small packets of mass between random positions to escape possible
   local minima.

The total running time respects the given budget (≈1000 s).  The
returned list contains non‑negative numbers that sum to 1, satisfying
the security checks in ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (real FFT, power‑of‑two length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    L = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    z = np.fft.irfft(X * Y, n=L)[: n + m - 1]
    return z


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full linear convolution)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Objective (scale‑invariant) and simplex projection
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of the metric (no safety checks)."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical fallback
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Analytic families of symmetric shapes (all normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Two power ramps meeting an optional flat plateau."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                      # rises 0 → 1
        shape = np.concatenate([ramp,
                                 np.ones(plateau_len),
                                 ramp[::-1]])
        # guard against off‑by‑one due to rounding
        if shape.size < n:
            shape = np.concatenate([shape,
                                    np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    shape = np.maximum(shape, 0.0)
    tot = shape.sum()
    if tot > 0.0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential_shape(n: int, beta: float, alpha: float = 0.0) -> np.ndarray:
    """Symmetric exponential decay from the centre, optional flat plateau."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    idx = np.arange(n, dtype=float) / (n - 1)
    centre = 0.5
    dist = np.abs(idx - centre) / 0.5               # 0 at centre, 1 at edges
    shape = np.exp(-beta * dist)
    if plateau_len > 0:
        start = (n - plateau_len) // 2
        shape[start:start + plateau_len] = 1.0
    shape = np.maximum(shape, 0.0)
    tot = shape.sum()
    if tot > 0.0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma is a fraction of n."""
    sigma = max(1e-6, min(0.5, sigma))
    idx = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((idx - centre) / (sigma * n)) ** 2)
    shape = np.maximum(shape, 0.0)
    tot = shape.sum()
    if tot > 0.0:
        shape /= tot
    else:

    return shape


# ----------------------------------------------------------------------
# Smooth surrogate optimisation (log‑sum‑exp + exponentiated gradient)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             gamma: float,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the LSE surrogate.
    Returns the best iterate (and its true objective value).
    """
    a = a0.astype(float).copy()
    n = a.shape[0]
    C = 2.0 * n
    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # --- self‑convolution and softmax weights ---
        conv = _conv_self(a)                # length 2n‑1
        maxc = conv.max()
        # stability: work with (conv‑max) to avoid overflow
        exp_term = np.exp(gamma * (conv - maxc))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp                # softmax over the 2n‑1 entries
        smooth_max = maxc + np.log(sum_exp) / gamma

        # --- gradient of the smooth max (via cross‑correlation) ---
        w_rev = w[::-1]
        conv_wr_a = _conv_fft(w_rev, a)       # length 3n‑2
        slice_start = n - 1
        slice_end = 2 * n - 1
        grad_smooth = 2.0 * conv_wr_a[slice_start:slice_end][::-1]

        # --- gradient of the true objective (chain rule) ---
        sum_a = a.sum()
        grad_f = C * (grad_smooth / (sum_a ** 2)
                      - (2.0 * smooth_max / (sum_a ** 3)) * np.ones_like(a))

        # --- mirror (exponentiated) update with diminishing step size ---
        eta = 0.5 / math.sqrt(it)            # step‑size schedule
        a = a * np.exp(-eta * grad_f)        # positivity is preserved
        s = a.sum()
        if s > 0.0:
            a /= s
        else:                               # extremely unlikely
            a = np.full_like(a, 1.0 / n)

        # --- occasional reporting ---
        if it % 200 == 0:
            cur = _objective(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"grad‑descent it={it:,} score={cur:.6f}")

    return best_a, best_score

# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Random packet moves with a Metropolis acceptance rule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 250_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # simple linear cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)           # more mass → more likely to be chosen
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin before the hard time limit
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1) Problem sizes – small for the coarse sweep, large for final
    # --------------------------------------------------------------
    n_small = 32768                     # 2¹¹⁵, fast for many FFTs
    n_big = 131072                      # 2¹⁷, final resolution

    # --------------------------------------------------------------
    # 2) Build a pool of promising shapes on the small grid
    # --------------------------------------------------------------
    pool: list[tuple[float, np.ndarray]] = []      # (score, shape_small)
    top_k_pool = 6

    def _add_to_pool(shape: np.ndarray):
        shape = np.maximum(shape, 0.0)
        if shape.sum() == 0.0:
            shape = np.full_like(shape, 1.0 / shape.size)
        else:
            shape = shape / shape.sum()
        sc = _objective(shape)
        if len(pool) < top_k_pool:
            pool.append((sc, shape.copy()))
            pool.sort(key=lambda x: x[0])
        else:
            if sc < pool[-1][0]:
                pool[-1] = (sc, shape.copy())
                pool.sort(key=lambda x: x[0])

    # ---- power‑ramp family -------------------------------------------------
    p_vals = np.linspace(0.1, 3.5, 22)          # 0.1 … 3.5
    alpha_vals = np.linspace(0.0, 0.5, 11)      # 0 … 0.5
    for p in p_vals:
        for al in alpha_vals:
            a = _make_power_shape(n_small, p, al)
            _add_to_pool(a)
        if time.time() >= deadline:
            break

    # ---- exponential decay -------------------------------------------------
    for beta in np.linspace(0.0005, 0.025, 20):
        a = _make_exponential_shape(n_small, beta, alpha=0.0)
        _add_to_pool(a)
        if time.time() >= deadline:
            break

    # ---- Gaussian family ---------------------------------------------------
    for sigma in np.linspace(0.02, 0.30, 15):
        a = _make_gaussian_shape(n_small, sigma)
        _add_to_pool(a)
        if time.time() >= deadline:
            break

    # ---- random Dirichlet draws --------------------------------------------
    for _ in range(80):
        a = rng.random(n_small)
        a /= a.sum()
        _add_to_pool(a)
        if time.time() >= deadline:
            break

    # ---- optional global starter (height_sequence_1) -----------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # interpolate to the small grid for scoring
            xs = np.linspace(0, 1, starter.size)
            xt = np.linspace(0, 1, n_small)
            a = np.interp(xt, xs, starter)
            a = np.maximum(a, 0.0)
            if a.sum() > 0.0:
                a = a / a.sum()
                _add_to_pool(a)
                print(f"[{time.time() - start_time:.2f}s] "
                      f"global starter included in pool")
    except Exception:
        pass

    # keep the very best small shape (will be up‑sampled later)
    pool.sort(key=lambda x: x[0])
    best_small = pool[0][1].copy()
    best_small_score = pool[0][0]

    print(f"[{time.time() - start_time:.2f}s] "
          f"Coarse search best score (small n) = {best_small_score:.6f}")

    # --------------------------------------------------------------
    # 3) Upsample the top candidates to the large resolution
    # --------------------------------------------------------------
    def _upsample(shape_small: np.ndarray) -> np.ndarray:
        xs = np.linspace(0, 1, shape_small.size)
        xt = np.linspace(0, 1, n_big)
        big = np.interp(xt, xs, shape_small)
        big = np.maximum(big, 0.0)
        s = big.sum()
        if s > 0.0:
            big /= s
        else:
            big = np.full(n_big, 1.0 / n_big)
        return big

    # we will run the smooth gradient descent from the best 3 upsampled shapes
    seeds_big = [_upsample(entry[1]) for entry in pool[:3]]
    # also try the (possibly) supplied global starter directly at big size
    if 'starter' in locals():
        try:
            xs = np.linspace(0, 1, starter.size)
            xt = np.linspace(0, 1, n_big)
            big = np.interp(xt, xs, starter)
            big = np.maximum(big, 0.0)
            if big.sum() > 0.0:
                big /= big.sum()
                seeds_big.append(big)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 4) Smooth surrogate optimisation (exponentiated gradient)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    remaining = deadline - time.time()
    if remaining > 30.0 and seeds_big:
        # allocate roughly 80 % of the remaining budget to the descent,
        # split evenly among the chosen seeds.
        descent_budget = remaining * 0.78
        per_seed = descent_budget / len(seeds_big)
        for i, seed_seq in enumerate(seeds_big):
            sub_dead = time.time() + per_seed
            refined_seq, refined_score = _smooth_gradient_descent(
                seed_seq, gamma=4e5, deadline=sub_dead, start_time=start_time)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"seed {i} refined score = {refined_score:.6f}")

    # If for some reason nothing was produced, fall back to the up‑sampled best.
    if best_seq is None:
        best_seq = _upsample(best_small)
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # 5) Light annealing polish (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        annealed_seq, annealed_score = _anneal(best_seq, deadline, start_time)
        if annealed_score < best_score:
            best_score = annealed_score
            best_seq = annealed_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"anneal improved final score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n_big, 1.0 / n_big)

    print(f"[{time.time() - start_time:.2f}s] "
          f"Finished – final score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
