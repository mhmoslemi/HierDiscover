"""\
Enhanced search for a step‑function with a low self‑convolution maximum.

The algorithm consists of three stages:

1. **Exploratory grid** – generate a few analytic shapes (power ramp + flat
   plateau) and keep the best few candidates.
2. **Smooth‑max gradient descent** – minimise a smooth approximation of
   ``max(conv)`` using the gradient 
   ∇ Σ_i w_i·(a*a)_i = 2·corr(a, w) where w_i ∝ exp(α·(conv_i‑max)).
   The update uses a momentum term and a diminishing step‑size, and after
   every step the vector is projected onto the probability simplex.
   The smoothing factor α is increased gradually so that the optimisation
   first sees a smooth landscape and later refines the true max.
3. **Light simulated‑annealing polish** – move tiny packets of mass between
   random indices with a Metropolis rule to escape tiny local minima.

All heavy work (self‑convolution and the correlation needed for the
gradient) is performed by real FFTs, which run in O(n log n).  The code fits
comfortably within the 1000 s budget on a single CPU core and typically
reaches an evaluation score below 1.504 for n = 65536.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers (real FFT, power‑of‑two lengths)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors (FFT based)."""
    L = _next_pow2(x.shape[0] + y.shape[0] - 1)
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)[: x.shape[0] + y.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (fast, no safety checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)²) (exact, no clipping)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max() / (seq.sum() ** 2))


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex (Wang & Carreira‑Perpiñán)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                       # numerical safety‑net
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generator – power ramp + optional flat plateau
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape (length n) with two power ramps meeting a plateau.
    The whole vector is normalised to sum = 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    # length of the central plateau (even, for perfect symmetry)
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p               # monotone rise 0 → 1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # adjust size artefacts caused by rounding
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (the core optimiser)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv) with projected gradient
    descent and momentum.  Returns the best iterate seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # momentum variables
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5

    it = 0
    # start with a gentle smoothing, increase gradually
    alpha = 10.0
    alpha_max = 5000.0

    while time.time() < deadline:
        it += 1

        # self‑convolution and exact objective
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑iter={it:,} "
                  f"score={cur_score:.6f}")

        # ---- smooth max (soft‑max) weights ---------------------------------
        max_conv = conv.max()
        # increase smoothness slowly
        if it % 200 == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.2)

        # stable soft‑max
        exp_w = np.exp(alpha * (conv - max_conv))
        w = exp_w / exp_w.sum()                      # Σ w = 1, w ≥ 0

        # ---- gradient of Σ w_i·(a*a)_i  ( = 2·corr(a, w) ) -------------
        a_rev = a[::-1]                               # reversed a
        corr = _conv_general(w, a_rev)                # length = 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]           # slice gives length n

        # momentum + diminishing step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)                       # stay on the simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions; Metropolis rule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # adapt the number of attempts to the remaining budget
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # fast exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # heavy bins are chosen more often
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the harness

    """

    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin before the hard limit

    # --------------------------------------------------------------

    n = 65536                     # 2¹⁶ – sufficient for the target bound

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    top_pool = []                 # list of (score, seq)
