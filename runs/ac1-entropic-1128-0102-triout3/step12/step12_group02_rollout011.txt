def _subgradient_descent(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # FFT sizes for convolution and correlation
    L_conv = _next_pow2(2*n - 1)
    L_corr = _next_pow2(3*n - 2)

    while time.time() < deadline:
        t += 1
        # compute conv
        conv = _conv_self(a, L=L_conv)
        M = conv.max()
        # compute mask
        mask = (conv >= M - 1e-12).astype(float)
        # optionally weight mask uniform
        # compute gradient of M using mask
        a_rev = a[::-1]
        # compute conv2 = convolution mask * a_rev
        # Use FFT
        L = _next_pow2(len(mask) + n - 1)
        W = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv2 = np.fft.irfft(W * A_rev, n=L)
        # gradient = 2 * conv2[n-1 : n-1+n]
        grad = 2.0 * conv2[n-1 : n-1 + n]
        # objective = 2*n*M / (a.sum()**2)
        cur_score = 2.0 * n * M / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={t} score={cur_score:.6f}")
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.05 / (np.sqrt(v_hat) + eps)  # smaller step
        a = a - step * m_hat
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score

"""
Propose a non‑negative step‑function (list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three stages:

1️⃣  A fast parametric sweep over several families of shapes
    (geometric·power, “U‑shaped”, sinusoidal) and mixtures with the
    uniform distribution.  The best sequence from this sweep seeds the
    next stage.

2️⃣  A two‑phase descent:
    – a smooth‑max Adam descent (soft‑max approximation of the max)
      with a slowly rising temperature;
    – a sub‑gradient descent that directly minimises the true max
      using the exact convex sub‑gradient (computed by a single FFT‑based
      convolution).  Both phases keep the iterate on the probability
      simplex.

3️⃣  A final stochastic hill‑climb that exchanges a tiny fraction of the
    mass between random coordinates, accepting only improving moves.
    The routine respects the 1000 s time budget and returns the best
    sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    n = x.shape[0] + y.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    return _conv_fft(a, a, L)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    M = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * M / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) gradient
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float, L_conv: int, L_corr: int):
    """
    Returns (grad, conv) where
      conv = a * a (full linear convolution)
      grad ≈ ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    """
    conv = _conv_self(a, L=L_conv)                # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                       # stability
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()                # soft‑max weights, length 2n‑1

    # correlation w * a_rev => gradient
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)         # length L_corr
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad, conv


def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_score_initial: float):
    """
    Adam‑type descent on the smooth‑max surrogate.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 2.0          # start relatively smooth
    beta_mul = 1.015
    beta_max = 600.0    # go fairly close to the true max

    while time.time() < deadline:
        t += 1
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} "
                  f"score={cur_score:.6f}")

        # Adam update with projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.12 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional random jitter
        if t % 4000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max (conv)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float):
    """
    Adam‑type descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n - 2)          # for mask*a_rev

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, L=L_conv)
        M = conv.max()

        cur_score = 2.0 * n * M / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={t:,} "
                  f"score={cur_score:.6f}")

        # mask for all indices attaining (or almost attaining) the max
        tol = 1e-12
        mask = (conv >= M - tol).astype(float)
        cnt = mask.sum()
        if cnt == 0:
            mask = np.ones_like(conv) / conv.size
        else:
            mask /= cnt                     # normalise to keep the step size stable

        # sub‑gradient of max(conv) w.r.t a
        a_rev = a[::-1]
        conv2 = _conv_fft(mask, a_rev, L=L_corr)   # length 3n‑2
        grad = 2.0 * conv2[n - 1:n - 1 + n]

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.07 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # occasional perturbation to get out of a flat region
        if t % 3000 == 0:
            perturb = 5e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + perturb, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (microscopic random mass transfers)
# ----------------------------------------------------------------------
def _final_hill_climb(start_a: np.ndarray,
                      deadline: float,
                      start_time: float,
                      L_fft: int):
    """
    Randomly move a tiny fraction of mass between two coordinates,
    accepting only moves that improve the exact objective.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-18:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to 0.5 % of the source mass (larger than before)
        eps = a[src] * rng.random() * 5e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine respects the 1000 s budget and returns the best
    sequence found (as a plain list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for bookkeeping
    deadline = start_time + budget_s - 0.25

    # problem size – larger n gives a finer search landscape
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise with the uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Use a globally supplied starter if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep over rich families of shapes
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    # --- family A: geometric × power (already used in the baseline) ---
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=25)
    p_grid = np.linspace(-3.0, 2.0, 41)

    for r in r_grid:
        if time.time() >= deadline - 180:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 180:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep A r={r:.4f} p={p:.2f} "
                          f"λ={lam:.2f} score={sc:.6f}")

    # --- family B: U‑shaped (large mass near the ends) ---
    p_grid_u = np.linspace(0.5, 3.5, 27)
    for p in p_grid_u:
        if time.time() >= deadline - 180:
            break
        shape = ((idx) * (n + 1 - idx)) ** (-p)          # ((i+1)*(n-i))^{-p}
        s = shape.sum()
        if s == 0.0:
            continue
        shape /= s
        for lam in (0.0, 0.3, 0.6, 0.9, 1.0):
            seq = lam * uniform + (1.0 - lam) * shape
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep B p={p:.2f} λ={lam:.2f} "
                      f"score={sc:.6f}")

    # --- family C: sinusoidal “U‑shaped” (low in the centre) ---
    sin_shape = 1.0 - np.sin(np.pi * (np.arange(n, dtype=np.float64)) / (n - 1))
    for exponent in (0.5, 1.0, 1.5, 2.0):
        if time.time() >= deadline - 180:
            break
        shape = sin_shape ** exponent
        shape /= shape.sum()
        for lam in (0.0, 0.4, 0.8, 1.0):
            seq = lam * uniform + (1.0 - lam) * shape
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep C exp={exponent:.1f} "
                      f"λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Descent phases (smooth‑max followed by sub‑gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120:
        # smooth‑max descent – allocate roughly 50 % of the remaining budget
        smooth_deadline = time.time() + remaining * 0.5
        print(f"[{time.time() - start_time:.2f}s] starting smooth‑max descent")
        a_smooth, sc_smooth = _smooth_descent(
            best_seq, n, smooth_deadline, start_time, best_score
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth descent improvement "
                  f"score={sc_smooth:.6f}")

    # sub‑gradient descent – use the rest of the time (leaving ~30 s for hill‑climb)
    remaining = deadline - time.time()
    if remaining > 60:
        sub_deadline = time.time() + remaining * 0.7
        print(f"[{time.time() - start_time:.2f}s] starting sub‑gradient descent")
        a_sub, sc_sub = _subgradient_descent(
            best_seq, n, sub_deadline, start_time, best_score
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient improvement "
                  f"score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] final hill‑climb")
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc

            print(f"[{time.time() - start_time:.2f}s] hill‑climb final score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise to sum = 1 and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
