    max_val = conv.max()

def _adam_smooth(a0, deadline, start_time, beta_start=1.0, beta_end=30.0, lr=0.025, ...):

    a = a0.copy()
    n = len(a)
    L = _next_pow2(3*n)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv via FFT
        Af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Af*Af, n=L)[:2*n-1]
        max_conv = conv_full.max()
        # compute smooth weights
        beta = beta_start + (beta_end - beta_start) * min(1.0, (it / 5000.0))  # linear increase up to some iter
        # shift for numerical stability
        shift = max_conv
        exp_vals = np.exp(beta * (conv_full - shift))
        p = exp_vals / exp_vals.sum()
        # gradient via convolution of p and a_rev
        a_rev = a[::-1]
        Pf = np.fft.rfft(p, n=L)
        Arev_f = np.fft.rfft(a_rev, n=L)
        conv_wr = np.fft.irfft(Pf * Arev_f, n=L)
        grad = 2.0 * conv_wr[n-1:n-1+n]  # length n
        # optionally correct diagonal? Not needed as using smooth.
        # compute objective
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} beta={beta:.2f} score={cur_score:.6f}")

        # Adam update
        grad_obj = 2.0 * n * grad   # Wait objective is 2*n*max(conv). Our gradient is relative to max(conv) (approx). Actually we need gradient of objective approximated by smooth max: grad_obj = 2*n * grad
        m = beta1 * m + (1-beta1) * grad_obj
        v = beta2 * v + (1-beta2) * (grad_obj * grad_obj)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

    return best_a, best_score

"""
A fast “smooth‑max” Adam optimiser for the step‑function problem.

The objective (with ∑a = 1) is

    J(a) = 2·n·max_k (a * a)[k]                     (1)

where * is the linear convolution.
We minimise J by projecting the iterates onto the probability simplex.

Instead of the sub‑gradient of the exact max we use a *log‑sum‑exp*
approximation of the max:
    S_β(a) = (1/β)·log Σ_k exp(β·c_k) ,   c = a*a .
Its gradient is
    ∇S_β = 2·(p * a_rev)    with   p_k = exp(β·c_k)/Σ_j exp(β·c_j) .

For small β the weights p are almost uniform (smooth landscape),
for large β they concentrate on the near‑max entries (approach the true
sub‑gradient).  β is increased during the run.

The optimiser uses Adam on the (scaled) gradient
    ∇J ≈ 2·n·∇S_β .
After the Adam phase a cheap Metropolis annealing walk is applied
to escape shallow minima.

The whole search respects the 1000 s budget and returns a plain list
of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                         # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns the *full* result of length 2·n‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv_full


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Warm‑start shapes
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a few simple analytic candidates (already normalised)."""
    x = np.arange(n, dtype=float) / (n - 1)               # [0,1]

    tent = np.minimum(x, 1.0 - x)                         # linear tent
    cosine = np.sin(np.pi * x)                           # half‑sine
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)       # centred Gaussian
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p               # flatter tent

    candidates = [tent, cosine, circ, gauss, power_tent]
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Adam optimiser using a smooth max (log‑sum‑exp) gradient
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 beta_min: float = 0.5,
                 beta_max: float = 30.0,
                 lr: float = 0.025,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex with a smooth approximation of the max.
    Returns the best iterate encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)          # sufficient for all FFTs

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- self‑convolution (full) ----
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # current objective (sum(a) == 1, so denominator = 1)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        # linear schedule for the temperature β
        elapsed = time.time() - start_time
        total_adam = deadline - start_time
        frac = min(1.0, (time.time() - (deadline - total_adam)) / total_adam)
        beta = beta_min + frac * (beta_max - beta_min)

        # stability: subtract the max before exponentiation
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))
        p = exp_vals / exp_vals.sum()               # shape (2n‑1,)

        # ---- gradient of the smooth max ----
        a_rev = a[::-1]
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_wr = np.fft.irfft(Pf * Ar, n=L)
        grad_smooth = 2.0 * conv_wr[n - 1 : n - 1 + n]   # length n

        # scaled gradient for the original objective J ≈ 2·n·grad_smooth
        grad_obj = 2.0 * n * grad_smooth

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ---- projection onto the simplex ----
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same idea as the reference code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk that moves a random packet of mass between two indices."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 80_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)               # source proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25     # move up to 25 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
