def simplex_projection(v):
    # Ensure non-negative and sum = 1
    # based on algorithm by Wang & Carreira-Perpinan
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑score step sequence.

Algorithm Overview
------------------
1. **Initial candidates** – generate a few simple families (arcsine‑type
   power laws) for several lengths.  The one with the smallest
   objective `2·n·max(conv(p,p))` (sum of p is normalised to 1) is kept.

2. **Local search** – starting from the best candidate we run a
   simulated‑annealing hill‑climber.
   * The current worst convolution shift `k = argmax(conv)`.
   * The index `src` that contributes most to that shift is
     `src = argmax_i p[i]·p[k‑i]`.
   * With probability `p_swap` we *swap* two random entries.
   * Otherwise we *transfer* a random fraction of the mass from `src`
     to a low‑mass destination `dst` (chosen with probability proportional
     to `1‑p`).
   * The move is accepted if the score improves, or with a
     temperature‑dependent probability (simulated annealing) to escape
     plateaus.
   * Every `reset_every` iterations we jitter the current best solution
     to restart from a nearby point.

3. **Time management** – the whole routine stops a few tenths of a second
   before the hard deadline and returns the best vector seen.

The score is exactly the value returned by `evaluate_sequence` for a
probability vector (sum = 1), because the denominator collapses to 1.
All numbers are non‑negative and the vector is re‑normalised before being
returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: linear self‑convolution via FFT (fast and numerically stable)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self(p: np.ndarray) -> np.ndarray:
    """Linear self‑convolution p * p (full length 2·len(p)‑1)."""
    n = p.shape[0]
    L = _next_pow2(2 * n - 1)
    F = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(F * F, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    # This is the algorithm of Wang & Carreira‑Perpiñán (2013).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
# ----------------------------------------------------------------------
def score_of(p: np.ndarray) -> float:
    """Return 2·n·max(conv(p,p)) – this equals evaluate_sequence(p)."""
    n = p.shape[0]
    conv = conv_self(p)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Hill‑climbing with occasional swaps and simulated annealing
# ----------------------------------------------------------------------
def hill_climb(p_start: np.ndarray,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a local search starting from ``p_start``.
    Returns the best vector seen and its score.
    """
    p = p_start.copy()
    n = p.shape[0]
    cur_score = score_of(p)
    best_p = p.copy()
    best_score = cur_score

    iteration = 0
    reset_every = 200_000          # re‑jitter after many non‑improving steps
    last_improve = 0
    T0 = 0.02                      # initial temperature for annealing
    p_swap = 0.1                   # probability of performing a random swap

    while time.time() < deadline:
        iteration += 1
        # linear temperature schedule
        t = max(0.0, T0 * (1.0 - (time.time() - deadline + budget_s) / budget_s))

        # occasionally try a pure random swap (preserves sum exactly)
        if rng.random() < p_swap:
            i, j = rng.choice(n, size=2, replace=False)
            p_new = p.copy()
            p_new[i], p_new[j] = p_new[j], p_new[i]
            new_score = score_of(p_new)
        else:
            # compute current worst shift
            conv = conv_self(p)
            k = int(conv.argmax())

            # indices that influence conv[k]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contrib = p[idx] * p[k - idx]
            src = idx[int(contrib.argmax())]

            # destination – prefer a small entry
            inv = 1.0 - p
            prob = inv / inv.sum()
            dst = rng.choice(n, p=prob)
            if dst == src:
                continue

            # move a random fraction of the source mass
            max_frac = min(0.5, p[src])               # never move more than half
            epsilon = max_frac * rng.random()
            if epsilon <= 0.0:
                continue

            p_new = p.copy()
            p_new[src] -= epsilon
            p_new[dst] += epsilon
            # no need to renormalise – sum is preserved and all entries stay ≥0
            new_score = score_of(p_new)

        # acceptance rule (simulated annealing)
        accept = False
        if new_score < cur_score:
            accept = True
        elif t > 0.0 and rng.random() < math.exp((cur_score - new_score) / t):
            accept = True

        if accept:
            p = p_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                last_improve = iteration
                print(f"[{time.time() - start:.2f}s] iter={iteration:,} "
                      f"new best score={best_score:.6f}")

        # restart from the current best with a tiny jitter if we plateau
        if iteration - last_improve > reset_every:
            jitter = 0.005 * best_p * (rng.random(n) - 0.5)
            p = np.maximum(best_p + jitter, 0.0)
            p = simplex_projection(p)
            cur_score = score_of(p)
            last_improve = iteration
            print(f"[{time.time() - start:.2f}s] plateau – re‑jittered at iter {iteration}")

    return best_p, best_score


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score.
    Returns a list of non‑negative floats whose sum is 1.
    """
    global start, budget_s
    start = time.time()
    rng = np.random.default_rng(seed)

    # safety margin before the hard deadline
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) Build a small library of plausible initial distributions
    # ------------------------------------------------------------------
    candidates = []

    # a) Use a previously discovered construction if it exists
    prev = globals().get('height_sequence_1')
    if isinstance(prev, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(prev, dtype=float)
        if prev_arr.ndim == 1 and prev_arr.size > 0:
            prev_arr = np.maximum(prev_arr, 0.0)
            if prev_arr.sum() > 0:
                prev_arr = prev_arr / prev_arr.sum()
                candidates.append(prev_arr)
                print(f"[{time.time() - start:.2f}s] Loaded height_sequence_1 (n={prev_arr.size})")

    # b) Arcsine‑type family with a few exponents and lengths
    lengths = [1024, 2048, 4096]          # keep n modest – larger n hurts the score factor
    alphas = np.linspace(0.20, 0.60, 9)   # sweep a useful range
    for n in lengths:
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        for alpha in alphas:
            eps = 1e-12
            w = (x * (1.0 - x) + eps) ** (-alpha)
            p = w / w.sum()
            candidates.append(p)

    # ------------------------------------------------------------------
    # 2) Pick the best among the cheap candidates
    # ------------------------------------------------------------------
    best_p = None
    best_score = np.inf
    for p in candidates:
        sc = score_of(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
    if best_p is None:
        # fall‑back: a uniform vector
        n = 1024
        best_p = np.full(n, 1.0 / n)
        best_score = score_of(best_p)
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Refine the best candidate with a longer local search
    # ------------------------------------------------------------------
    refined_p, refined_score = hill_climb(best_p, deadline, rng)
    if refined_score < best_score:
        best_p, best_score = refined_p, refined_score
    print(f"[{time.time() - start:.2f}s] Final best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0:
        best_p /= total
    else:
        # extremely unlikely – fall back to uniform
        n = best_p.shape[0]
        best_p = np.full(n, 1.0 / n)

    return [float(v) for v in best_p]
