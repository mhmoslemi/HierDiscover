"""
Propose a step function (probability vector) with a low value of
``evaluate_sequence``.  The optimiser works in three stages:

1. **Exploration on a modest size** (n≈8 k).  A flexible Beta‑type family
   a_i ∝ (i/(n‑1))**p·(1‑i/(n‑1))**q is swept, together with a few random
   Dirichlet draws.  The best few candidates are kept.

2. **Smooth Adam descent** on the small problem.  Instead of the
   non‑differentiable ``max`` a soft‑max approximation
   `S_β(x)=log Σ exp(β·x_i)/β` is used.  Its gradient can be written as a
   convolution, which is evaluated with FFTs.  An Adam‑style adaptive step‑size
   together with simplex projection drives the iterates while β is slowly
   increased (the approximation becomes sharper).

3. **Upsampling & fine‑tuning** on the target size (n=2¹⁶=65536).  The
   best small‑scale solution is linearly interpolated, then refined with the
   same Adam‑softmax routine (more iterations, larger β).  A light
   simulated‑annealing polish finishes the search.

All helper functions are top‑level so the harness can pickle them.  The
algorithm respects the 1000 s budget and returns a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: smallest power of two ≥ x
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)²)"""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical safeguard
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Flexible Beta‑type shape generator
# ----------------------------------------------------------------------
def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Symmetric Beta‑like density on {0,…,n‑1} (p,q≥0)."""
    x = np.arange(n, dtype=float) / (n - 1)
    shape = (x ** p) * ((1.0 - x) ** q)
    # avoid exact zeros that could hinder the optimiser
    shape = np.where(shape > 0, shape, 0.0)
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(conv)
#   f_β(a) = 2·n·S_β(conv(a)) ,   S_β(c)=log Σ exp(β·c_i)/β
# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Returns ∇_a f_β(a) (up to a global constant).  Implemented via FFT.
    """
    n = a.shape[0]
    # 1) self‑convolution
    conv = _conv_self(a)                     # length 2n‑1
    # 2) soft‑max weights w_i = exp(β·(conv_i‑max))
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0.0:                         # extreme β, fall back to arg‑max
        w_norm = np.zeros_like(w)
        w_norm[np.argmax(conv)] = 1.0
    else:
        w_norm = w / w_sum

    # 3) grad = 2· Σ_i w_norm[i]·a_{i‑j}  (for each j)
    #    This is a correlation of w_norm with a; we compute it via FFT.
    a_rev = a[::-1]
    L = _next_pow2(len(a_rev) + len(w_norm) - 1)   # >= 3n‑2
    Af = np.fft.rfft(a_rev, n=L)
    Wf = np.fft.rfft(w_norm, n=L)
    corr_full = np.fft.irfft(Af * Wf, n=L)[: len(a_rev) + len(w_norm) - 1]

    start = n - 1                     # position where shift = 0
    grad = 2.0 * corr_full[start:start + n]  # length n
    # The true gradient of f_β is 4·n·grad, but a constant factor is irrelevant
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    # --------------------------------------------------------------
    # 1. Small‑scale exploration ( n≈8 k )
    # --------------------------------------------------------------
    n_small = 8192
    pool: list[tuple[float, np.ndarray]] = []
    top_k = 8

    # grid over Beta exponents
    p_vals = np.linspace(0.0, 4.0, 9)   # 0 … 4
    q_vals = np.linspace(0.0, 4.0, 9)
    for p in p_vals:
        for q in q_vals:
            a = _beta_shape(n_small, p, q)
            score = _objective(a)
            if len(pool) < top_k:
                pool.append((score, a.copy()))
                pool.sort(key=lambda x: x[0])
            elif score < pool[-1][0]:
                pool[-1] = (score, a.copy())
                pool.sort(key=lambda x: x[0])

    # a few random Dirichlet draws
    for _ in range(200):
        if time.time() >= deadline:
            break
        a = rng.dirichlet(np.ones(n_small))
        score = _objective(a)
        if len(pool) < top_k:
            pool.append((score, a.copy()))
            pool.sort(key=lambda x: x[0])
        elif score < pool[-1][0]:
            pool[-1] = (score, a.copy())
            pool.sort(key=lambda x: x[0])

    # possibly use the globally stored starter
    try:
        starter_arr = np.asarray(height_sequence_1, dtype=float)
        if starter_arr.ndim == 1 and starter_arr.size > 0:
            starter = np.maximum(starter_arr, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                score = _objective(starter)
                if len(pool) < top_k:
                    pool.append((score, starter.copy()))
                    pool.sort(key=lambda x: x[0])
                elif score < pool[-1][0]:
                    pool[-1] = (score, starter.copy())
                    pool.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2. Adam‑style smooth optimisation on the small problem
    # --------------------------------------------------------------
    best_small = None
    best_small_score = float("inf")

    for idx, (init_score, a0) in enumerate(pool):
        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        step0 = 0.2
        beta = 10.0               # initial soft‑max temperature
        it = 0

        while it < 400 and time.time() < deadline:
            it += 1
            # anneal temperature upwards (more faithful max)
            beta = min(200.0, beta * 1.02)

            # gradient of smoothed objective
            grad = _grad_softmax(a, beta)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            step = step0 / (1.0 + 0.001 * it)
            a -= step * m_hat / (np.sqrt(v_hat) + eps)

            # keep feasibility
            a = _project_simplex(a)

            if it % 50 == 0:
                cur_score = _objective(a)
                if cur_score < best_small_score:
                    best_small_score = cur_score
                    best_small = a.copy()
                print(f"[{time.time() - start_time:.2f}s] small#{idx} iter={it} score={cur_score:.6f}")

        # final evaluation of this candidate
        cur_score = _objective(a)
        if cur_score < best_small_score:
            best_small_score = cur_score
            best_small = a.copy()
        print(f"[{time.time() - start_time:.2f}s] small#{idx} finished score={cur_score:.6f}")

    # fallback if nothing survived
    if best_small is None:
        best_small = np.full(n_small, 1.0 / n_small)

    # --------------------------------------------------------------
    # 3. Upsample to the target size ( n = 2¹⁶ )
    # --------------------------------------------------------------
    n_large = 1 << 16          # 65536
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, best_small)
    a_large = np.maximum(a_large, 0.0)
    a_large = _project_simplex(a_large)

    # --------------------------------------------------------------
    # 4. Fine‑grained Adam optimisation on the full problem
    # --------------------------------------------------------------
    a = a_large.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.08
    beta = 20.0                 # start smoother
    it = 0
    best_large = a.copy()
    best_large_score = _objective(a)

    # allocate most of the remaining budget to this phase
    while time.time() < deadline - 5.0:      # keep ~5 s for polishing
        it += 1
        # slowly tighten the soft‑max approximation
        beta = min(500.0, beta * 1.01)

        grad = _grad_softmax(a, beta)

        # normalise gradient to stabilise the step‑size
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            grad = grad / gnorm

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (1.0 + 0.0005 * it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

        if it % 250 == 0:
            cur_score = _objective(a)
            if cur_score < best_large_score:
                best_large_score = cur_score
                best_large = a.copy()
                print(f"[{time.time() - start_time:.2f}s] large iter={it} new best={cur_score:.6f}")

    # --------------------------------------------------------------
    # 5. Light simulated‑annealing polish
    # --------------------------------------------------------------
    cur_seq = best_large.copy()
    cur_score = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    anneal_rng = np.random.default_rng(seed + 999)

    # a relatively large number of tiny mass moves
    max_moves = 200_000
    for move in range(max_moves):
        if time.time() >= deadline:
            break
        # simple linear cooling schedule
        T = 0.05 * (1.0 - move / max_moves) + 1e-6

        src = anneal_rng.choice(n_large, p=cur_seq)
        dst = anneal_rng.integers(n_large)
        if src == dst or cur_seq[src] < 1e-14:
            continue
        # move at most 15 % of the source mass
        delta = cur_seq[src] * anneal_rng.random() * 0.15
        new_seq = cur_seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        new_score = _objective(new_seq)
        delta_obj = new_score - cur_score
        if delta_obj < 0.0 or math.exp(-delta_obj / T) > anneal_rng.random():
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal step={move} best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n_large, 1.0 / n_large)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
