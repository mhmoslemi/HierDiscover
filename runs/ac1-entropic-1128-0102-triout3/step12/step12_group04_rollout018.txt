"""
A fast mixed‑strategy search for a step‑function that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the linear self‑convolution.  The algorithm proceeds in three
phases:

1.  **Deterministic warm‑start** – generate a variety of analytical shapes
    (power‑tents, exponentials, power‑law, etc.), normalise them and keep the
    best according to the exact objective.

2.  **Fast hill‑climbing on the convolution** – keep the current vector and its
    convolution, then repeatedly move a random packet of mass from one index
    to another.  The convolution is updated in O(n) using the exact
    incremental formula, therefore millions of moves can be tried in the
    remaining time budget.  Only improving moves are kept (a simple
    deterministic descent).

3.  **Return the best vector** – converted to a plain Python list.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum≈1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass‑transfer move
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst`.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current coefficient vector (will be modified in‑place).
    conv : np.ndarray, shape (2n‑1,)
        Current convolution a*a (will be updated in‑place).
    src, dst : int
        Indices with src ≠ dst.
    eps : float
        Amount moved from src to dst (eps ≥ 0, eps ≤ a[src]).
    """
    n = a.shape[0]

    # keep a copy of the original a values needed for the update
    # (they are still unchanged at this point)
    # 1) linear term 2·a*Δ  Δ = eps*(e_dst – e_src)
    # shift by src  (subtract)
    conv[src : src + n] -= 2.0 * eps * a
    # shift by dst  (add)
    conv[dst : dst + n] += 2.0 * eps * a

    # 2) quadratic term Δ*Δ
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    conv[idx2src] += eps * eps
    conv[idx2dst] += eps * eps
    conv[idxsum] -= 2.0 * eps * eps

    # finally update the probability vector
    a[src] -= eps
    a[dst] += eps


# ----------------------------------------------------------------------
# Candidate shape generators (all produce a non‑negative vector of length n)
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of analytic shapes (unnormalised)."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Exponential decay from the centre (symmetric)
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]                     # symmetric
    candidates.append(shape)

    # 5) Half‑cosine (already used as warm‑start but keep for completeness)
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, different sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random Dirichlet draws – a few of them
    rng = np.random.default_rng(0)
    for _ in range(8):
        rnd = rng.random(n)
        candidates.append(rnd)

    # normalise all candidates (sum to 1) and keep only those with positive mass
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)

    return normalised

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin (the harness adds a further 0.5 s)
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Determine the problem size.
    # --------------------------------------------------------------
    try:
        # use the length of the previously discovered vector if available
        n = len(height_sequence_1)
    except Exception:
        n = 4096                     # default – fast FFT size and good performance

    # --------------------------------------------------------------
    # 1) Warm‑start – pick the best among many simple analytic shapes.
    # --------------------------------------------------------------
    warm_starts = _make_candidate_shapes(n)

    # include the global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                warm_starts.append(starter / starter.sum())
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()
    a = best_vec.copy()
    conv = _conv_fft(a)
    cur_score = best_score
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Fast hill‑climbing on the convolution.
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 1)

    # pre‑compute a constant denominator (sum stays 1)
    total = a.sum()
    assert total > 0.0
    n_obj = 2.0 * n / (total * total)      # factor such that objective = n_obj * conv.max()

    it = 0
    while time.time() < deadline:
        it += 1

        # current objective (use cached conv.max())
        cur_obj = n_obj * conv.max()
        # probability of picking a source proportional to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:

        # move a random fraction of the source mass (up to 30 % early on,
        # shrinking to 2 % near the end)
        elapsed = time.time() - start_time
        remaining = deadline - time.time()
        frac_max = 0.30 * remaining / (deadline - start_time) + 0.02
        eps = a[src] * rng.random() * frac_max
        if eps <= 0.0:
            continue

        # remember objective before the move
        before_obj = cur_obj

        # apply the move (in‑place update of a and conv)
        _apply_move(a, conv, src, dst, eps)

        # new objective after the move
        new_obj = n_obj * conv.max()

        # keep the move only if it does not increase the objective
        if new_obj <= before_obj:
            # accept – possibly record a new best
            if new_obj < best_score:
                best_score = new_obj
                best_vec = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # reject – undo the move
            _apply_move(a, conv, dst, src, eps)

    # --------------------------------------------------------------
    # 3) Final normalisation and conversion.
    # --------------------------------------------------------------
    if best_vec is None:

    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
