"""
Propose a step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

by a three‑phase search:

1️⃣  A fast randomised parametric sweep over several simple families
    (geometric, mixture of two geometries, power·geometric, beta‑type).
    Each candidate is normalised to sum = 1 and its exact score is obtained
    with an FFT based self‑convolution.

2️⃣  A local refinement using a simulated‑annealing style hill‑climb:
    tiny random mass transfers are performed and moves are accepted if
    they improve the objective or, with a decaying temperature,
    if they are only slightly worse – this helps escaping shallow plateaus.

3️⃣  The best sequence found is returned as a plain Python list.
   The whole routine respects the 1000 s budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)-1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring helper (exact, using FFT)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray, n: int, L_fft: int) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    conv = _conv_self(seq, L_fft)
    return float(2.0 * n * conv.max() / (seq.sum() ** 2))

    # ------------------------------------------------------------------
    # 2️⃣  Simulated‑annealing hill‑climb (local refinement)
    # ------------------------------------------------------------------
    # remaining time for refinement
    while time.time() < deadline:
        # linear temperature schedule from 1e‑2 down to 1e‑6
        elapsed = time.time() - start_time
        T_start, T_end = 1e-2, 1e-6
        fraction = (elapsed) / (deadline - start_time)
        T = T_start * (1 - fraction) + T_end * fraction

        # propose a random mass transfer
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # allow a fairly large fraction (up to 20 %) to escape plateaus
        frac = rng.random() * 0.20 + 1e-6
        delta = best_seq[src] * frac

        candidate = best_seq.copy()
        candidate[src] -= delta
        candidate[dst] += delta

        # it is possible (numerically) to create a tiny negative entry;
        # clamp and re‑normalise to keep the sum exactly 1
        if candidate[src] < 0:
            candidate[src] = 0.0
        total = candidate.sum()
        if total == 0:
            continue
        candidate /= total

        cand_score = _score(candidate, n, L_fft)

        # acceptance rule
        improve = cand_score < best_score
        if improve or rng.random() < np.exp((best_score - cand_score) / max(T, 1e-12)):
            best_seq = candidate
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Hill‑climb improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
