"""
Simulated‑annealing optimiser for the peak‑autocorrelation problem.

The score (for a normalised vector ∑a_i = 1) is 2·n·max(conv(a,a)).
The algorithm starts from a good analytic shape (arcsine‑type or the
previous best `height_sequence_1` if available).  A temperature schedule
and a decreasing step‑size allow occasional uphill moves, which helps
to escape the plateaus observed with pure hill‑climbing.  Periodic
re‑normalisation fixes drift, and a simple jitter restart is performed
when no improvement has been seen for a while.  The best vector seen
throughout the run is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, fast for the initialisation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Initial shape generation
# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a short list of analytic candidate vectors.
    Uses an arcsine‑type family and a Beta‑type family.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-8
    candidates = []

    # arcsine‑type (w = (x(1-x)+eps)^(-α))
    for alpha in np.linspace(0.20, 0.48, 7):
        w = (x * (1.0 - x) + eps) ** (-alpha)
        p = w / w.sum()
        candidates.append(p)

    # beta‑type (p_i ∝ (i+1)^{a‑1} (n‑i)^{b‑1})
    for a, b in [(0.5, 0.5), (0.7, 0.7), (0.9, 0.9), (1.2, 1.2), (1.5, 1.5)]:
        w = ((np.arange(n) + 1) ** (a - 1)) * ((n - np.arange(n)) ** (b - 1))
        w = np.maximum(w, eps)
        p = w / w.sum()
        candidates.append(p)

    return candidates

        conv_mod = conv.copy()
        # linear part
        conv_mod[src:src + n] -= 2.0 * epsilon * p
        conv_mod[dst:dst + n] += 2.0 * epsilon * p
        # quadratic correction
        idx_2src = 2 * src
        idx_2dst = 2 * dst
        idx_srcdst = src + dst
        if idx_2src < conv_mod.size:
            conv_mod[idx_2src] += epsilon * epsilon
        if idx_2dst < conv_mod.size:
            conv_mod[idx_2dst] += epsilon * epsilon
        if idx_srcdst < conv_mod.size:
            conv_mod[idx_srcdst] -= 2.0 * epsilon * epsilon

        new_max = conv_mod.max()
        delta = new_max - cur_max

        # ----- temperature schedule (linear cooling) -----
        T0 = 0.008
        T = max(0.0, T0 * (1.0 - elapsed / budget_s))

        # ----- acceptance rule -----
        accept = False
        if delta < -1e-16:
            accept = True
        elif T > 0.0:
            prob = math.exp(-delta / T) if delta > 0 else 1.0
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move
            conv = conv_mod
            p[src] -= epsilon
            p[dst] += epsilon
            # numerical safety
            if p[src] < 0.0:
                p[src] = 0.0
            cur_max = new_max

            # update best solution
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                last_improve_time = time.time()
                print(
                    f"[{time.time() - start_time:.2f}s] iter={iteration:,} "
                    f"new best score={best_score:.6f}"
                )
        # else: reject – p and conv stay untouched.

        # ----- periodic housekeeping -----
        if iteration % 4000 == 0:
            # re‑normalise to fight floating‑point drift
            total = p.sum()
            if abs(total - 1.0) > 1e-12:
                p /= total
                conv = _conv_self(p)
                cur_max = conv.max()

        # ----- stagnation detection and jitter -----
        if time.time() - last_improve_time > plateau_time:
            # perturb the current best and continue from there
            p = best_p.copy()
            jitter = jitter_factor * p * (rng.random(p.shape) - 0.5)
            p = np.maximum(p + jitter, 0.0)
            total = p.sum()
            if total > 0:
                p /= total
            else:
                p = np.full_like(p, 1.0 / n)
            conv = _conv_self(p)
            cur_max = conv.max()
            last_improve_time = time.time()
            print(
                f"[{time.time() - start_time:.2f}s] plateau – jittered at iter {iteration}"
            )

    # --------------------------------------------------------------
    # 4) Final polishing with strict hill‑climbing (optional)
    # --------------------------------------------------------------
    # A short deterministic pass can shave off the last few 1e‑4.
    p = best_p.copy()
    conv = _conv_self(p)
    cur_max = conv.max()
    for _ in range(200000):
        # locate worst shift
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = p[idx] * p[k - idx]
        src = idx[int(np.argmax(contrib))]
        # pick a low‑mass destination
        inv = 1.0 - p
        inv_sum = inv.sum()
        if inv_sum <= 0.0:
            dst = rng.integers(0, n)
        else:
            dst = rng.choice(n, p=inv / inv_sum)
        if dst == src:
            continue
        eps = p[src] * 0.01 * rng.random()  # tiny deterministic step
        if eps <= 0.0:
            continue

        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * p
        conv_new[dst:dst + n] += 2.0 * eps * p
        idx_2src = 2 * src
        idx_2dst = 2 * dst
        idx_srcdst = src + dst
        if idx_2src < conv_new.size:
            conv_new[idx_2src] += eps * eps
        if idx_2dst < conv_new.size:
            conv_new[idx_2dst] += eps * eps
        if idx_srcdst < conv_new.size:
            conv_new[idx_srcdst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        if new_max < cur_max - 1e-16:
            # accept strictly improving move
            conv = conv_new
            p[src] -= eps
            p[dst] += eps
            cur_max = new_max
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()

    # --------------------------------------------------------------
    # 5) Return the best vector (normalised to sum 1)
    # --------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0:
        best_p = best_p / total
    else:
        best_p = np.full_like(best_p, 1.0 / n)

    # Optional: report final score (does not affect the returned value)
    final_score = 2.0 * n * best_p.dot(_conv_self(best_p).max())
    print(f"[{time.time() - start_time:.2f}s] final best score ≈ {best_score:.6f}")

    return [float(v) for v in best_p]
