def subgradient(a):
    conv = _conv_self(a)
    max_val = conv.max()
    mask = (conv == max_val).astype(float)
    w = mask / mask.sum()
    # correlation as gradient: grad = 2 * (w * a_rev convolution) (same as earlier)
    a_rev = a[::-1]
    L_corr = _next_pow2(3*n-2)
    W = np.fft.rfft(w, n=L_corr)
    A_rev = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A_rev, n=L_corr)
    start = n - 1
    grad = 2.0 * corr[start:start+n]
    return grad, max_val
```

"""
Propose a step‑function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The strategy mixes three ideas that together tend to beat the previous
baseline (≈1.5083):

1️⃣  A fairly dense parametric sweep over monotone decreasing shapes
    (geometric × power) with several mixing ratios with the uniform
    distribution.  This quickly discovers a decent region of the space.

2️⃣  Several independent “restarts”: each start is taken either from the
    best sweep candidate, a noisy perturbation of it or a random Dirichlet
    draw (sparse).  From each start a *hard‑max* sub‑gradient descent
    (projected onto the probability simplex) is performed.  The gradient
    is the exact derivative of the current maximal convolution entry,
    i.e. we minimise the true objective, not a smoothed surrogate.

3️⃣  A short simulated‐annealing‑style hill‑climb that transfers a small
    fraction of mass between random coordinates, accepting only improving
    moves.  This fine‑tunes the solution to the exact ``evaluate_sequence``
    objective.

The whole pipeline respects a 1000 s time budget and returns the best
sequence discovered (scaled to sum = 1, which makes the objective
scale‑free).  The code is deliberately written with only NumPy and the
standard library so it runs on the evaluation platform.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution objective
# ----------------------------------------------------------------------
def _hardmax_subgrad(a: np.ndarray,
                     L_conv: int,
                     L_corr: int) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_val) where
        grad = ∂/∂a max_k (a*a)[k]  (sub‑gradient),
        max_val = max_k (a*a)[k].
    """
    conv = _conv_self(a, L=L_conv)            # length 2n‑1
    max_val = conv.max()
    # weight = uniform over all maximising indices (sub‑gradient set)
    mask = (conv == max_val).astype(np.float64)
    if mask.sum() == 0:
        w = mask
    else:
        w = mask / mask.sum()                # shape (2n‑1,)
    # correlation w * a_rev to obtain gradient
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A_rev = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A_rev, n=L_corr)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Simple sub‑gradient descent with projection onto the simplex
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         L_conv: int,
                         L_corr: int,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Uses a diminishing step size α_t = α0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    alpha0 = 0.8                     # initial step‑size factor
    t = 0

    while time.time() < deadline:
        t += 1
        grad, max_val = _hardmax_subgrad(a, L_conv, L_corr)

        # current exact score (sum of a is 1 → denominator = 1)
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"subgrad it={t:,} score={cur_score:.6f}")

        # diminishing step size
        step = alpha0 / math.sqrt(t)

        a_next = a - step * grad
        a = _project_simplex(a_next)

        # occasional random jitter to escape tiny plateaus
        if t % 2000 == 0:
            eps = 1e-4 * rng.standard_normal(n) * a
            a = _project_simplex(a + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb (simulated‑annealing‑style, accepting only improvements)
# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """
    Random mass transfers between two coordinates.
    Only moves that improve the exact objective are kept.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to a few percent of the source mass
        eps = a[src] * rng.random() * 0.02   # ≤2 % of src
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score

            deadline=deadline - 5,          # leave a couple seconds for hill‑climb
            start_time=start_time,
            L_conv=L_conv,
            L_corr=L_corr,
            best_score_initial=best_score,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improvement "
                  f"score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
