assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\; \max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

Because the evaluation function normalises the sequence only by the denominator,
we are free to work on the probability simplex  

\[
\sum_i a_i = 1,\qquad a_i\ge0,
\]

in which case  

\[
F(a)=2n\;\max (a*a).
\]

Hence we only have to shrink the largest entry of the self‑convolution while
staying on the simplex.

The search consists of three stages

1. **Warm‑starts** – a small library of analytically‑motivated shapes
   (uniform, triangular, power‑law, exponential, Gaussian, arcsine, …).
   The best of them (according to the exact objective) is taken as the seed.

2. **Adam on the simplex** – the variables are re‑parameterised by logits  
   \(\theta_i\) with \(a_i=\operatorname{softmax}(\theta)_i\).
   For the current \(a\) we identify the indices where the convolution is
   within a tiny tolerance of the current maximum, build the (average)
   sub‑gradient of the maximum, and perform an Adam update on the logits.
   The schedule for the tolerance factor is gradually tightened, the step‑size
   is slowly reduced and the logits are kept in a safe range
   \([-30,30]\).

3. **Light polishing** – when the time budget is almost exhausted we run a
   cheap Metropolis‑Monte‑Carlo walk that moves a random packet of mass
   between two positions; any improving move is kept.

All three phases are timed, and the best sequence discovered at any moment is
returned.  The implementation is self‑contained, uses only NumPy and the
provided `evaluate_sequence`, and respects the one‑CPU, 1000 s budget.

"""
Fast optimiser for 2·n·max(conv(a,a))/(sum a)² on the probability simplex.

* Warm‑start from a small library of analytic shapes.
* Optimise the logits of a soft‑max representation with Adam.
  The sub‑gradient of max(conv) is obtained from all convolution entries
  that are within a shrinking tolerance of the current maximum.
* Final cheap simulated‑annealing polish.

The code runs completely inside the given time budget (default 1000 s) and
returns a plain list of non‑negative floats suitable for evaluate_sequence().
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).  Mirrors
    evaluate_sequence() but without the defensive checks.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑start shapes
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # [0,1]
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _collect_warm_starts(n: int) -> list[np.ndarray]:
    """Create a modest library of diverse initial probability vectors."""
    cand = []

    # basic shapes
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    # power‑law families
    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    # exponentials
    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    # Gaussians
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # mixtures of centre / edge
    for lc in (0.25, 0.5, 0.75):
        centre = _shape_power_center(n, 2.0)
        edge = _shape_power_edge(n, 2.0)
        a = lc * centre + (1 - lc) * edge
        a = a / a.sum()
        cand.append(a)

    # optional global starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # weed out NaNs that could appear in pathological cases
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Adam optimisation on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _softmax(theta: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(theta)
    e = np.exp(theta - m)
    return e / e.sum()


def _grad_max_conv(a: np.ndarray, conv: np.ndarray,
                   top_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) with respect to a.
    Averaged over all indices that are active (top_idx).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in top_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idxs = np.arange(lo, hi + 1, dtype=int)
        grad[idxs] += 2.0 * a[k - idxs]
    grad /= float(top_idx.size)
    return grad


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam on the logits of a soft‑max representation.
    Returns the best probability vector seen and its objective value.
    """
    n = a0.shape[0]

    # initialise logits safely (log of a plus a tiny offset)
    eps = 1e-30
    theta = np.log(np.maximum(a0, eps))

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(theta)
    v = np.zeros_like(theta)

    best_a = a0.copy()
    best_score = _objective(a0)

    it = 0
    while time.time() < deadline:
        it += 1

        # probability vector from logits
        a = _softmax(theta)

        # convolution and its maximum
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # sum(a)==1 by construction

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,}  "
                  f"best={cur_score:.6f}")

        # tolerance that shrinks as we progress
        if it < 200:
            tol_factor = 0.02
        elif it < 1000:
            tol_factor = 0.01
        elif it < 5000:
            tol_factor = 0.005
        else:
            tol_factor = 0.001
        tol = cur_max * tol_factor
        top_idx = np.where(conv >= cur_max - tol)[0]
        if top_idx.size == 0:
            top_idx = np.array([conv.argmax()])

        # sub‑gradient of max(conv) w.r.t a
        grad_a = _grad_max_conv(a, conv, top_idx)

        # gradient of the full objective (2·n·max(conv))
        grad_f = 2.0 * n * grad_a

        # chain rule for the soft‑max parametrisation
        dot = np.dot(a, grad_f)
        g_theta = a * (grad_f - dot)

        # Adam state update
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (large at the beginning, then slowly decays)
        step = 0.3 / (1.0 + 0.0002 * it)

        theta -= step * m_hat / (np.sqrt(v_hat) + eps_adam)

        # keep logits in a safe interval to avoid overflow / underflow
        theta = np.clip(theta, -30.0, 30.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (identical to the previous script)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # source proportional to current mass
        dst = rng.integers(n)             # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the bound.
    The function respects the 1000 s time limit and returns a plain list of
    non‑negative floats (already normalised to sum = 1, which is sufficient).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # 4096 offers a good trade‑off between resolution and speed.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) Gather a palette of warm‑starts and pick the best one.
    # ------------------------------------------------------------------
    warm = _collect_warm_starts(n)
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(warm, 1):
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] Warm‑start {idx}/{len(warm)} "
              f"score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # 2) Adam optimisation on the simplex (major part of the budget).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                     # keep a few seconds for polishing
        adam_deadline = time.time() + remaining * 0.85
        refined_seq, refined_score = _adam_optimize(
            best_seq,
            adam_deadline,
            start_time
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam finished "
                  f"score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Light annealing polish (if any time left).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve "
                  f"score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation & conversion to plain list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # The returned list is a plain Python list of floats.
    return [float(v) for v in best_seq]
