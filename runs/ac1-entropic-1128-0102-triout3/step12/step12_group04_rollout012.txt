def _max_subgradient(a, L):
    n = a.shape[0]
    conv = _conv_self(a, L)
    max_val = conv.max()
    # find indices approx equal within tolerance
    mask = conv >= max_val - 1e-16
    count = mask.sum()
    if count == 0:
        w = np.zeros_like(conv)
    else:
        w = np.where(mask, 1.0 / count, 0.0)
    # compute gradient via convolution with reversed a
    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
    grad = grad[n-1:n-1+n]   # slice n entries
    return grad, conv
```

def _adam_optimize_exact(a0, deadline, start_time, lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it+=1
        grad, conv = _max_subgradient(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamExact iter={it:,} score={cur_score:.6f}")
        # Adam update with gradient of objective (the conv's max)
        # gradient of objective = 2 * n * grad (since objective = 2n * max_conv)
        grad_obj = 2.0 * n * grad
        m = beta1 * m + (1 - beta1) * grad_obj
        v = beta2 * v + (1 - beta2) * (grad_obj ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)
    return best_a, best_score

"""
A fast two‑phase optimiser for the self‑convolution max‑norm.
Phase 1 – projected Adam with the *exact* sub‑gradient of the
maximum convolution entry (instead of a top‑k average).  This gives
precise descent directions and reaches lower values than the
previous smoothed approach.

Phase 2 – aggressive simulated‑annealing / hill‑climbing that
repeatedly moves a random packet of mass between two indices.
An exponential cooling schedule and a large number of steps (as
many as time permits) are used to escape shallow minima and polish
the solution.

The code works on a power‑of‑two length (4096) which yields fast FFTs,
keeps the whole search comfortably inside the 1000 s budget and
produces a list ready for ``evaluate_sequence``.  All numbers are
non‑negative and the vector is projected onto the probability simplex
after every update.
"""
import time
import math
import numpy as np

# --------------------------------------------------------------
# Basic numerical utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT.
    Returns the full convolution of length 2·n‑1."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective used during the search.
    a is assumed to be on the simplex (∑a = 1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform simplex point
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Exact sub‑gradient of the maximum convolution entry
# --------------------------------------------------------------

def _max_subgradient(a: np.ndarray, L: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns the sub‑gradient of max(a*a) together with the convolution.
    The gradient is obtained by convolving the (averaged) indicator of the
    active maximal indices with the reversed a vector.
    """
    n = a.shape[0]

    # full convolution
    conv = _conv_self(a, L)

    # mask of indices attaining the maximum (with a tiny tolerance)
    max_val = conv.max()
    eps = max_val * 1e-14
    mask = conv >= max_val - eps
    count = np.count_nonzero(mask)

    # weight vector w = 1/count on the active set, 0 elsewhere
    w = np.where(mask, 1.0 / count if count > 0 else 0.0, 0.0)

    # gradient = 2 * (w * a_rev)   (convolution)
    a_rev = a[::-1]
    grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
    # slice the n entries that correspond to the gradient components
    grad = 2.0 * grad[n - 1: n - 1 + n]
    return grad, conv


# --------------------------------------------------------------
# Projected Adam optimiser using the exact sub‑gradient
# --------------------------------------------------------------

def _adam_optimize_exact(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         lr: float = 0.02,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex using the *exact* sub‑gradient of
    max(a*a).  Returns the best iterate encountered and its objective.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)          # first‑moment estimate
    v = np.zeros_like(a)          # second‑moment estimate
    it = 0

    while time.time() < deadline:
        it += 1

        # sub‑gradient of max(conv) and the full convolution
        grad, conv = _max_subgradient(a, L)

        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamExact iter={it:,} "
                  f"score={cur_score:.6f}")

        # gradient of the *objective* 2·n·max(conv)
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# --------------------------------------------------------------
# Light simulated‑annealing / hill‑climbing polish
# --------------------------------------------------------------

def _anneal_refine(seq: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that repeatedly transfers a random packet of mass
    between two indices.  The schedule is aggressive (many steps) and
    the temperature decays linearly to a tiny value.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # approximate number of steps we can afford (empirically ~200 µs per step)
    max_steps = 500_000
    step = 0
    while time.time() < deadline and step < max_steps:
        step += 1

        # linear cooling from T0 to T_end
        progress = step / max_steps
        T0 = 0.05
        T_end = 1e-5
        T = T0 * (1.0 - progress) + T_end * progress

        # choose source proportionally to its current mass (more likely to move)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move up to 30 % of the source mass
        delta = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score < 0.0 or math.exp(-delta_score / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal step={step:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# --------------------------------------------------------------
# Warm‑starting shapes (analytic + random)
# --------------------------------------------------------------

def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a collection of simple analytic candidates plus a few
    random Dirichlet samples."""
    x = np.arange(n, dtype=float) / (n - 1)               # normalised coordinate [0,1]

    # analytic shapes
    tent = np.minimum(x, 1.0 - x)
    cosine = np.sin(np.pi * x)                           # sin(π·x) vanishes at ends
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]

    # normalise each candidate to sum‑to‑one and keep non‑negative entries
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normals.append(c / s)

    # a few random Dirichlet draws for diversity
    rng = np.random.default_rng()
    for _ in range(8):
        rand = rng.dirichlet(np.ones(n))
        normals.append(rand)

    return normals


# --------------------------------------------------------------
# Main entry point required by the harness
# --------------------------------------------------------------

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step function.
    Returns a list of non‑negative floats (the vector is already normalised).
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # --------------------------------------------------------------
    n = 4096  # 2^12, matches the length used in the reference run

    # --------------------------------------------------------------
    # 1) Warm‑start – pick the best among analytic + random candidates.
    # --------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # incorporate the globally supplied sequence if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                warm_starts.append(starter)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Main optimisation – projected Adam with exact sub‑gradient.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0:                      # make sure we have enough time left
        # allocate about 70 % of the remaining budget to Adam
        adam_deadline = time.time() + remaining * 0.70
        a_opt, score_opt = _adam_optimize_exact(a0,
                                                adam_deadline,
                                                start_time,
                                                lr=0.02)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamExact best={score_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing polish (as many steps as time permits).
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal_refine(a0, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            a0 = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal best={anneal_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation & conversion to plain list.
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    # return a plain Python list
    return [float(v) for v in a0]
