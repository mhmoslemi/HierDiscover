"""
Hybrid Simulated Annealing + simple gradient refinement for the
auto‑convolution objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The routine works in three stages:

1. **Initial pool** – a few analytic “power‑ramp” shapes, a handful of
   random vectors and (if present) the global starter `height_sequence_1`.
   The best few candidates are kept.

2. **Exploratory simulated‑annealing** – repeatedly moves a tiny amount of
   probability mass from a source index to a destination index.  The
   temperature and the step size decay linearly with the remaining time,
   allowing large exploratory jumps early on and fine adjustments later.
   The best vector found during this phase is retained.

3. **Simple smooth‑gradient polishing** – a short Adam‑style descent on a
   soft‑max approximation of the max‑convolution, projected onto the
   probability simplex.  This cleans up the remaining irregularities.

All steps respect the supplied time budget (default 1000 s).  The returned
list contains non‑negative numbers whose sum is 1, ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # fallback – uniform vector
        w = np.full_like(v, 1.0 / n)
    return w

def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “power‑ramp” shape, normalised to sum = 1.
    `p` controls the ramp curvature, `alpha` (0‑0.5) the relative plateau width.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0.0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _interpolate_to_length(src: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation / repeat to reach `target_len`."""
    src_len = src.shape[0]
    if src_len == target_len:
        return src.copy()
    if target_len % src_len == 0:
        factor = target_len // src_len
        return np.repeat(src, factor)
    # otherwise use simple linear interpolation on the unit interval
    x_old = np.linspace(0.0, 1.0, src_len, endpoint=False)
    x_new = np.linspace(0.0, 1.0, target_len, endpoint=False)
    return np.interp(x_new, x_old, src)

# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max surrogate of max)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Few‑iteration Adam‑style descent on a soft‑max approximation."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.04                     # base learning rate
    T0, T_end = 0.02, 1e-5           # soft‑max temperature schedule

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"SmoothGD it={it:,} score={cur_score:.6f}")

        # temperature (linear decay)
        frac = (deadline - time.time()) / (deadline - start_time)
        T = max(T_end, T0 * frac)

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation (w ⊛ a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex (keeps sum = 1 and non‑negativity)
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Simulated‑annealing exploration
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Move small probability parcels between coordinates.
    Temperature and step size decay linearly with remaining time.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    # parameters for the cooling schedules
    T_start, T_end = 0.08, 5e-6
    step_start, step_min = 0.5, 5e-5

    while time.time() < deadline:
        it += 1
        # linear decay based on remaining fraction
        frac = (deadline - time.time()) / (deadline - start_time)
        T = T_end + (T_start - T_end) * frac
        step_factor = step_min + (step_start - step_min) * frac

        # pick source with probability proportional to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # amount to move (up to step_factor of src's mass)
        eps = a[src] * step_factor * rng.random()
        if eps <= 0.0:
            continue

        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps

        new_score = _objective(prop)
        delta = new_score - cur_score

        # Metropolis acceptance rule
        if delta < 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            a = prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Anneal it={it:,} best={new_score:.6f}")

        # occasional renormalisation to combat accumulated rounding error
        if it % 2000 == 0:
            s = a.sum()
            if abs(s - 1.0) > 1e-12:
                a /= s

    return best_a, best_score

                top.append((sc, a))
                top.sort(key=lambda x: x[0])
            elif sc < top[-1][0]:
                top[-1] = (sc, a)
                top.sort(key=lambda x: x[0])

    # Random vectors (simple Dirichlet draws)
    for _ in range(80):
        if time.time() >= deadline:
            break
        a = rng.random(N)
        a /= a.sum()
        sc = _objective(a)
        if len(top) < top_k:
            top.append((sc, a))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, a)
            top.sort(key=lambda x: x[0])

    # Incorporate the global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter_up = _interpolate_to_length(starter, N)
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                sc = _objective(starter_up)
                if len(top) < top_k:
                    top.append((sc, starter_up))
                    top.sort(key=lambda x: x[0])
                elif sc < top[-1][0]:
                    top[-1] = (sc, starter_up)
                    top.sort(key=lambda x: x[0])
    except Exception:
        pass

    # Select the current best candidate
    best_score, best_seq = top[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Exploratory Simulated Annealing (major part of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate roughly 70 % of remaining time to SA
        sa_deadline = time.time() + 0.7 * remaining
        cand_seq, cand_score = _simulated_annealing(best_seq, sa_deadline, start_time)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq
            print(f"[{time.time() - start_time:.2f}s] After SA best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Smooth gradient‑descent polishing (uses a soft‑max surrogate)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        gd_deadline = time.time() + remaining   # use whatever is left
        cand_seq, cand_score = _smooth_gradient_descent(best_seq, gd_deadline, start_time)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq
            print(f"[{time.time() - start_time:.2f}s] After GD best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
