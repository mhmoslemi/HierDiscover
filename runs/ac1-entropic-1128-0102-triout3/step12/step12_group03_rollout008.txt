        if fine_score < best_score:

            best_seq = fine_a

    # Final normalization
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:

    print(f"[{time.time() - start_time:.2f}s] Final score {final_score:.6f}")

    return [float(v) for v in best_seq]

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The method is a multi‑scale, multi‑restart stochastic optimisation:

1. **Coarse search** – on a small problem (N≈16 k) we run many short
   Adam‑style gradient descents from diverse initialisations
   (Dirichlet draws, analytic windows, Gaussians, Beta‑pdfs, sine windows).
   The best coarse vector is kept.

2. **Upsample** – the coarse best is interpolated to the final size
   (N≈65 k).  This gives a good starting point at low computational cost.

3. **Fine optimisation** – on the full size we run a longer soft‑max
   gradient descent, a sub‑gradient descent, and a light simulated‑annealing
   polish.  All steps respect the probability simplex (non‑negative, sum = 1).

The whole routine respects the 1000 s budget, prints intermediate scores,
and returns a plain Python list ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau (normalised)."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        shape = shape[:n]                     # safety trim
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian(n: int, rel_sigma: float) -> np.ndarray:
    """Discrete Gaussian (σ = rel_sigma·n) normalised to sum = 1."""
    sigma = max(1.0, rel_sigma * n)
    x = np.arange(n) - (n - 1) / 2.0
    shape = np.exp(-0.5 * (x / sigma) ** 2)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution shape on a uniform grid."""
    x = np.linspace(0.0, 1.0, n)
    shape = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    shape[np.isnan(shape) | np.isinf(shape)] = 0.0
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sine(n: int, cycles: float = 0.5) -> np.ndarray:
    """Half‑sine window (positive) with given number of half‑cycles."""
    i = np.arange(n)
    shape = np.sin(np.pi * cycles * (i + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a soft‑max surrogate of the max‑conv."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.07               # a bit larger than the reference code
    T0, T_end = 0.02, 1e-4

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} "
                  f"score={cur_score:.6f}")

        # temperature schedule – exponential decay is slightly smoother
        frac = it / (it + 800.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights over the convolution entries
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient = 2 * (a * w) convolved (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto simplex
        a = _project_simplex(a)

    return best_a, best_score


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient method on the true non‑smooth objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.6

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

    return best_a, best_score


def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Very light simulated‑annealing polish."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 500_000               # larger upper bound – will stop at deadline
    it = 0
    while it < max_steps and time.time() < deadline:
        it += 1
        # exponential cooling
        T = 0.015 * (1e-6 / 0.015) ** (it / max_steps)

        # pick a source proportional to its mass (biased move)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.2          # move up to 20 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Stochastic multi‑restart search for a low‑score step‑function.
    Returns a list of non‑negative floats (the harness will pass it to
    ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # 1. Problem size decisions
    # --------------------------------------------------------------
    # coarse stage (fast, many restarts)
    coarse_N = 16384
    # fine stage (final precision)
    remaining = deadline - time.time()
    if remaining > 600:
        fine_N = 65536
    elif remaining > 300:
        fine_N = 32768
    else:
        fine_N = coarse_N

    # --------------------------------------------------------------
    # 2. Build a diverse initial pool (coarse resolution)
    # --------------------------------------------------------------
    def evaluate_and_update(vec, score, best):
        if score < best[0]:
            best[0] = score
            best[1] = vec.copy()
            print(f"[{time.time() - start_time:.2f}s] New coarse best {score:.6f}")
        return best

    best_coarse = [np.inf, None]               # [score, vector]

    # analytic families
    for p in np.linspace(0.2, 2.0, 7):
        for alpha in np.linspace(0.0, 0.5, 5):
            a = _make_shape(coarse_N, p, alpha)
            sc = _objective(a)
            best_coarse = evaluate_and_update(a, sc, best_coarse)

    for sigma in np.linspace(0.02, 0.30, 5):
        a = _make_gaussian(coarse_N, sigma)
        sc = _objective(a)
        best_coarse = evaluate_and_update(a, sc, best_coarse)

    for a_par, b_par in [(0.5, 0.5), (0.8, 0.8), (2.0, 2.0), (0.3, 5.0), (5.0, 0.3)]:
        a = _make_beta(coarse_N, a_par, b_par)
        sc = _objective(a)
        best_coarse = evaluate_and_update(a, sc, best_coarse)

    for cycles in [0.3, 0.5, 0.7, 1.0]:
        a = _make_sine(coarse_N, cycles)
        sc = _objective(a)
        best_coarse = evaluate_and_update(a, sc, best_coarse)

    # random Dirichlet draws – varying concentration
    for _ in range(30):
        conc = rng.uniform(0.05, 5.0)
        a = rng.gamma(conc, 1.0, size=coarse_N)
        a /= a.sum()
        sc = _objective(a)
        best_coarse = evaluate_and_update(a, sc, best_coarse)

    # ------------------------------------------------------------------
    # 2b. Short smooth‑gradient refinements on the coarse vectors
    # ------------------------------------------------------------------
    coarse_budget = max(0.0, deadline - time.time() - 30.0)   # keep 30 s for fine phase
    n_restart = 12
    per_restart = max(0.1, coarse_budget / (n_restart + 1e-9))

    for i in range(n_restart):
        if time.time() >= deadline - 30.0:
            break
        a0 = best_coarse[1].copy()
        # add a slight random perturbation before each restart
        a0 += rng.normal(scale=0.005, size=coarse_N)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        sub_dead = time.time() + per_restart
        a_opt, sc_opt = _smooth_gradient_descent(a0, sub_dead, start_time)
        best_coarse = evaluate_and_update(a_opt, sc_opt, best_coarse)

    # ------------------------------------------------------------------
    # 3. Upscale the best coarse candidate to the final resolution
    # ------------------------------------------------------------------
    if best_coarse[1] is None:
        coarse_vec = np.full(coarse_N, 1.0 / coarse_N)
    else:
        coarse_vec = best_coarse[1]

    if fine_N != coarse_N:
        x_old = np.linspace(0, 1, coarse_N, endpoint=False)
        x_new = np.linspace(0, 1, fine_N, endpoint=False)
        fine_vec = np.interp(x_new, x_old, coarse_vec)
    else:
        fine_vec = coarse_vec.copy()

    fine_vec = np.maximum(fine_vec, 0.0)
    if fine_vec.sum() > 0:
        fine_vec /= fine_vec.sum()
    else:
        fine_vec = np.full(fine_N, 1.0 / fine_N)

    best_score = _objective(fine_vec)
    best_seq = fine_vec.copy()
    print(f"[{time.time() - start_time:.2f}s] Upscaled start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Fine‑grained smooth gradient descent (major optimisation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40:
        sub_dead = time.time() + 0.78 * remaining
        fine_vec, fine_score = _smooth_gradient_descent(best_seq, sub_dead, start_time)
        if fine_score < best_score:
            best_score, best_seq = fine_score, fine_vec
            print(f"[{time.time() - start_time:.2f}s] After SmoothGD best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Sub‑gradient refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20:
        sub_dead = time.time() + 0.6 * remaining
        fine_vec, fine_score = _subgradient_descent(best_seq, sub_dead, start_time)
        if fine_score < best_score:
            best_score, best_seq = fine_score, fine_vec
            print(f"[{time.time() - start_time:.2f}s] After Sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Light annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_vec, fine_score = _anneal(best_seq, deadline, start_time)
        if fine_score < best_score:
            best_score, best_seq = fine_score, fine_vec
            print(f"[{time.time() - start_time:.2f}s] After anneal best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
