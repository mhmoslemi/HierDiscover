"""
Propose a low‑score step function.

The optimiser consists of several fast, gradient‑based phases that are
repeated from many random restarts:

1.  Adam‑accelerated smooth‑max mirror descent (log‑sum‑exp of the
    convolution).  α (the smoothness) is increased geometrically up to
    2·10⁴ so the surrogate becomes very close to the true max.
2.  Direct max‑subgradient descent on the real maximum (exponentiated‑
    gradient on the simplex).
3.  A targeted stochastic hill‑climb that always moves probability mass
    away from the pair of indices that contribute the most to the current
    peak of the self‑convolution.

The whole budget (≈1000 s) is split into many short restarts (≈10 s each);
the best sequence found is polished with a final hill‑climb.
All operations stay on the probability simplex (∑a = 1) and respect the
non‑negativity constraint required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ----------------------------------------------------
# ----------------------------------------------------------------------


def _fft_lengths(n: int):
    """FFT lengths needed for exact convolution and for the gradient."""
    # smallest power of two >= 2*n‑1 (full self‑convolution)
    L_self = 1 << ((2 * n - 2).bit_length())
    # smallest power of two >= 3*n‑2 (gradient convolution)
    L_grad = 1 << ((3 * n - 3).bit_length())
    return L_self, L_grad


def _exact_score(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(conv(a,a)) / (sum(a)**2)
    assuming a is already non‑negative.  The function works for any
    length n and uses FFT for speed.
    """
    n = a.shape[0]
    L_self, _ = _fft_lengths(n)
    Af = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _smooth_adam_descent(a: np.ndarray, deadline: float, start_time: float):
    """
    Adam‑accelerated mirror descent on the smooth log‑sum‑exp
    approximation of the maximum of the self‑convolution.
    Returns the best sequence seen during the run.
    """
    n = a.shape[0]
    L_self, L_grad = _fft_lengths(n)

    best_a = a.copy()
    best_score = _exact_score(best_a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # geometric schedule of the smoothing parameter α
    alphas = np.geomspace(1.0, 2e4, num=45)

    it = 0
    for alpha in alphas:
        if time.time() >= deadline - 0.3:
            break

        # inner loop – run until the phase deadline is reached
        while time.time() < deadline - 0.3:
            it += 1

            # 1) self‑convolution a*a
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights (softmax of α·conv)
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum

            # 3) gradient of the smooth surrogate
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam update on the simplex (exponentiated‑gradient step)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            # bias‑corrected moments
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # learning‑rate that shrinks with α and iteration number
            eta = 0.5 / (np.sqrt(alpha) * math.sqrt(it))
            step = eta * m_hat / (np.sqrt(v_hat) + eps)

            # exponentiated‑gradient move and renormalisation
            a = a * np.exp(-step)
            a = np.maximum(a, 1e-30)
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            else:
                a = np.full_like(a, 1.0 / n)

            # occasional exact evaluation
            if it % 5000 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] "
                        f"smooth‑adam α={alpha:.1f} it={it:,} score={cur_score:.6f}"
                    )
    return best_a, best_score


def _max_subgradient_descent(a: np.ndarray, deadline: float, start_time: float):
    """
    Simple exponentiated‑gradient sub‑gradient descent on the true
    maximum of the self‑convolution.
    """
    n = a.shape[0]
    L_self, _ = _fft_lengths(n)

    best_a = a.copy()
    best_score = _exact_score(best_a)

    it = 0
    while time.time() < deadline - 0.3:
        it += 1

        # current self‑convolution and its maximiser
        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        # contribution to the active maximal entry
        grad[idx] = 2.0 * a[k - idx]

        # exponentiated‑gradient step (step size ~ 1/√t)
        eta = 0.5 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        if it % 4000 == 0:
            cur_score = _exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}"
                )
    return best_a, best_score

    peak of the self‑convolution.
    """
    n = a.shape[0]
    L_self, _ = _fft_lengths(n)

    cur = a.copy()

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    start_temp = 5e-2
    end_temp = 1e-12

    while time.time() < deadline:
        it += 1

        # linear temperature schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        # current convolution & its maximal index
        Af = np.fft.rfft(cur, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)

        # product a_i * a_{k-i} tells how much each pair contributes
        contrib = cur[idx] * cur[k - idx]
        src = idx[np.argmax(contrib)]

        # pick a destination far from the current peak (distance‑weighted)
        distances = np.abs(np.arange(n) - k) + 1e-9
        probs = distances / distances.sum()
        dst = rng.choice(n, p=probs)
        if src == dst:
            continue

        eps = cur[src] * rng.random() * 0.20  # move up to 20 % of src mass
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} best={best_score:.6f}"
                )
    return best_seq, best_score


def _optimize_once(start_a: np.ndarray, deadline: float, start_time: float):
    """
    Run the three optimisation phases on a single starting point.
    The function respects the given deadline and returns the best sequence
    discovered in this run.
    """
    # Phase 1 – smooth‑max Adam descent
    if time.time() < deadline - 1.5:
        phase_deadline = deadline - 1.0
        a, _ = _smooth_adam_descent(start_a, phase_deadline, start_time)

    # Phase 2 – max‑subgradient descent
    if time.time() < deadline - 0.8:
        phase_deadline = deadline - 0.5
        a, _ = _max_subgradient_descent(a, phase_deadline, start_time)

    # Phase 3 – targeted hill‑climb (final polish)
    if time.time() < deadline:
        a, _ = _targeted_hill_climb(a, deadline, start_time)

    return a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.

    The optimiser repeatedly restarts a fast gradient‑based search from
    random initial points, keeps the best sequence seen and finally
    applies a focused hill‑climb for polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – we must finish before the external watchdog kills us
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # Choose problem size n and an initial candidate
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise Exception
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = None

    if starter is None:
        n = 4096
        best_seq = np.full(n, 1.0 / n)
    else:
        n = starter.size
        # enforce the correct length (pad or truncate)
        if n != 4096:
            # the competition does not demand a fixed length, but we keep it
            # to stay comparable to the reference implementation.
            if n > 4096:
                starter = starter[:4096]
                n = 4096
            else:
                starter = np.pad(starter, (0, 4096 - n), mode="constant")
                n = 4096
        best_seq = starter.copy()
        best_seq = np.maximum(best_seq, 0.0)
        s = best_seq.sum()
        if s > 0.0:
            best_seq /= s
        else:
            best_seq = np.full(n, 1.0 / n)

    best_score = _exact_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main loop – many short independent optimisation runs
    # ------------------------------------------------------------------
    # allocate most of the budget to random restarts; keep ~30 s for
    # a final polish on the current best.
    min_remaining = 30.0
    run_id = 0
    while True:
        time_left = deadline - time.time()
        if time_left <= min_remaining:
            break

        run_id += 1
        # a fresh random starting point (Dirichlet with concentration 0.5)
        gamma = np.random.gamma(0.5, 1.0, size=n)
        a0 = gamma / gamma.sum()

        # give each run at most 12 s (or whatever fits)
        run_deadline = time.time() + min(12.0, time_left - min_remaining)
        cand = _optimize_once(a0, run_deadline, start_time)
        cand_score = _exact_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Run #{run_id} NEW BEST = {best_score:.6f}"
            )
        else:
            # occasional logging for progress
            if run_id % 5 == 0:
                print(
                    f"[{time.time() - start_time:.2f}s] Run #{run_id} done – best so far = {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Final polishing on the best sequence found
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, _ = _targeted_hill_climb(best_seq, deadline, start_time)

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    final_score = _exact_score(best_seq)
    print(
        f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {final_score:.6f}"
    )
    return [float(v) for v in best_seq]
