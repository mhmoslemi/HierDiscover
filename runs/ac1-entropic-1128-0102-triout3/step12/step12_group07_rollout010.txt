def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # init p
    try:
        p_initial = np.asarray(height_sequence_1, dtype=float)
        p_initial = np.maximum(p_initial, 0.0)
        if p_initial.size == 0 or p_initial.sum()==0:
            raise ValueError
        p = p_initial / p_initial.sum()
        n = p.shape[0]
    except Exception:
        n = 4096
        x = (np.arange(n)+0.5)/n
        alpha = 0.38
        eps = 1e-6
        w = (x*(1-x)+eps)**(-alpha)
        p = w / w.sum()
        # p shape (n,)
    # utilities
    def compute_score(vec):
        s = vec.sum()
        if s < 1e-12:
            return np.inf
        conv = _conv_self(vec)
        return float(2.0 * len(vec) * conv.max() / (s**2))
    # initial score
    best_p = p.copy()
    best_score = compute_score(p)
    print(f"[{time.time() - start:.2f}s] initial score={best_score:.6f}, n={n}")
    # Phase 1: LP improvement
    while time.time() < deadline - 5.0:  # keep 5s for hill climbing
        # compute M
        conv = _conv_self(p)
        M = conv.max()
        # Build A_ub
        # Use diags to build trivially
        offsets = np.arange(n)
        data = [p[:n-d] for d in range(n)]
        from scipy.sparse import diags
        A = diags(data, offsets, shape=(2*n-1, n), format='csr')
        b = np.full(2*n-1, M, dtype=float)
        # objective: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=float)
        # solve LP
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            print(f"[{time.time() - start:.2f}s] LP failed: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        # improvement threshold
        if sum_g <= 1.0 + 1e-4:
            # no significant increase, stop LP phase
            print(f"[{time.time() - start:.2f}s] LP sum_g≈{sum_g:.6f} not enough, stop LP")
            break
        alpha = 1.0 / sum_g
        # new candidate
        p_new = alpha * g
        # re-normalize to sum approx 1
        p_new = p_new / p_new.sum()
        new_score = compute_score(p_new)
        if new_score < best_score:
            best_score = new_score
            best_p = p_new.copy()
            p = p_new.copy()
            print(f"[{time.time() - start:.2f}s] LP improved: score={best_score:.6f}, sum_g={sum_g:.6f}, alpha={alpha:.6f}")
        else:
            # fallback: mixture
            t = 0.5
            p_mix = (1-t) * p + t * (alpha * g)
            p_mix = p_mix / p_mix.sum()
            mix_score = compute_score(p_mix)
            if mix_score < best_score:
                best_score = mix_score
                best_p = p_mix.copy()
                p = p_mix.copy()
                print(f"[{time.time() - start:.2f}s] LP mixture improved: score={best_score:.6f}")
            else:
                print(f"[{time.time() - start:.2f}s] LP no improvement (score={new_score:.6f})")
                # Optionally break
                break
        # Check time
        if time.time() >= deadline - 30:
            break

    # Phase 2: hill-climbing (mass-transfer) with simulated annealing
    # Use same code as earlier but with acceptance probability
    p = best_p.copy()
    conv = _conv_self(p)
    cur_max = conv.max()
    n = p.shape[0]
    rng = np.random.default_rng()
    iter_count = 0
    last_improve = 0
    # Temperature schedule: start T0 as fraction of max conv
    T0 = (cur_max) * 0.1  # 10% of max conv as initial temperature
    max_iter = 100_000_00  # large upper bound
    while time.time() < deadline:
        iter_count += 1
        # choose shift with max conv (maybe random among top few)
        # Let's choose top 5 indices randomly
        max_indices = np.argwhere(conv >= cur_max - 1e-12).flatten()
        if max_indices.size > 1:
            k = rng.choice(max_indices)
        else:
            k = int(conv.argmax())
        # compute contributors
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        contrib = p[idx] * p[k - idx]
        src = idx[int(np.argmax(contrib))]
        # destination: choose among low probability entries
        # Weighted by inverse p, but also random
        low_weight = 1.0 - p
        prob_dst = low_weight / low_weight.sum()
        dst = rng.choice(n, p=prob_dst)
        if dst == src:
            continue
        # step size schedule: epsilon = a fraction of src mass decreasing over time
        elapsed = time.time() - start
        remaining = deadline - time.time()
        # dynamic max fraction: start at 0.2, down to 0.001 near end
        max_frac = max(0.001, 0.2 * (remaining / budget_s))
        epsilon = p[src] * rng.random() * max_frac
        if epsilon <= 0:
            continue
        # Save copies
        p_old = p.copy()
        conv_old = conv.copy()
        # Efficient update using convolution formula (similar as earlier)
        # We'll compute contributions using vectorized update:
        # conv[k'] for k' in src..src+n-1 subtract 2*epsilon*p
        # conv[k'] for dst..dst+n-1 add 2*epsilon*p
        # quadratic corrections
        conv[src:src+n] -= 2.0 * epsilon * p
        conv[dst:dst+n] += 2.0 * epsilon * p
        # corrections
        conv[2*src] += epsilon**2
        conv[2*dst] += epsilon**2
        conv[src+dst] -= 2.0 * epsilon**2
        # update p
        p[src] -= epsilon
        p[dst] += epsilon
        if p[src] < 0:
            p[src] = 0.0
        # compute new max conv
        new_max = conv.max()
        # Accept based on annealing
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # Temperature declines linearly
            progress = (elapsed / budget_s)
            T = T0 * (1 - progress)  # linear decay
            if T > 0:
                delta = new_max - cur_max
                prob = math.exp(-delta / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_max = new_max
            if 2.0 * n * cur_max < best_score:
                best_score = 2.0 * n * cur_max
                best_p = p.copy()
                last_improve = iter_count
                print(f"[{time.time() - start:.2f}s] iter={iter_count} new best={best_score:.6f}")
        else:
            # revert
            p = p_old
            conv = conv_old
        # occasional renorm & recompute full convolution to avoid drift
        if iter_count % 5000 == 0:
            sum_p = p.sum()
            if abs(sum_p - 1.0) > 1e-12:
                p /= sum_p
                conv = _conv_self(p)
                cur_max = conv.max()
        # early plateau handling
        if iter_count - last_improve > 200_000:
            # jitter
            p = best_p.copy()
            jitter = 0.005 * p * (rng.random(p.shape) - 0.5)
            p = np.maximum(p + jitter, 0.0)
            p /= p.sum()
            conv = _conv_self(p)
            cur_max = conv.max()
            last_improve = iter_count
            print(f"[{time.time() - start:.2f}s] plateau -> jitter at iter {iter_count}")
    # final output
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0:
        best_p /= total
    else:
        best_p = np.full(n, 1.0 / n)
    print(f"[{time.time() - start:.2f}s] final best score={best_score:.6f}")
    return [float(v) for v in best_p]

"""
Propose a low‑score step function.

Algorithm
---------
1. **Initialisation** – use the previously‑found best construction
   (`height_sequence_1`) if available, otherwise generate an arcsine‑type
   distribution.  The vector is normalised to sum 1.

2. **Global improvement (LP‑step)** – repeatedly solve a linear program:
      maximise  Σ g_i
      subject to g_i ≥ 0 and  (p * g)(k) ≤ M   for all shifts k,
   where M is the current maximal autocorrelation of p.
   The solution g typically has a larger total mass; scaling it to sum 1
   (factor α = 1/Σg) provably reduces the maximal autocorrelation.
   Accept the scaled solution if it improves the score, otherwise try a
   simple convex combination with the old vector.  Continue while time
   permits (a few LP iterations are cheap).

3. **Local refinement** – a simulated‑annealing hill‑climber that moves a
   small amount of mass from the pair contributing most to the worst
   shift to a location with little mass.  Moves that increase the maximal
   autocorrelation are occasionally accepted with a temperature that
   decays linearly, allowing the search to escape plateaux.

4. **Output** – return the best vector seen (normalised to sum 1).  All
   entries are non‑negative floats, suitable for the verifier.

The combination of a global LP‑driven jump and a fine‑grained annealed
hill‑climber typically beats pure greedy hill‑climbing and can reach the
target score under the 1000 s budget.
"""

import time
import math
import numpy as np

# optional imports – if SciPy is unavailable we fall back to pure hill‑climb
try:
    from scipy.sparse import diags
    from scipy.optimize import linprog
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, works for any length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _score_of(vec: np.ndarray) -> float:
    """Exact objective value for a vector (no clamping)."""
    s = vec.sum()
    if s < 1e-12:
        return float('inf')
    conv = _conv_self(vec)
    return float(2.0 * len(vec) * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Build the linear‑program matrix for the constraint (p * g) ≤ M
# ----------------------------------------------------------------------
def _lp_matrix(p: np.ndarray) -> "scipy.sparse.csr_matrix":
    """
    Returns a CSR matrix A of shape (2·n‑1, n) such that
    (A @ g)[k] = (p * g)[k] for all k.
    """
    n = p.shape[0]
    # diagonal offset d contains p[:n-d] on the d‑th super‑diagonal
    offsets = np.arange(n, dtype=int)               # 0 … n‑1
    data = [p[: n - d] for d in range(n)]          # decreasing lengths
    return diags(data, offsets, shape=(2 * n - 1, n), format='csr')


# ----------------------------------------------------------------------
# Main entry point required by the verifier
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a short safety margin before the hard deadline
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1) Initialise a probability vector p (Σp = 1)
    # --------------------------------------------------------------
    try:
        p0 = np.asarray(height_sequence_1, dtype=float)   # may be undefined
        if p0.ndim != 1 or p0.size == 0:
            raise ValueError
        p0 = np.maximum(p0, 0.0)
        if p0.sum() == 0.0:
            raise ValueError
        p = p0 / p0.sum()
        n = p.shape[0]
        print(f"[{time.time() - start_time:.2f}s] Using height_sequence_1 (n={n})")
    except Exception:
        # fallback: simple arcsine‑type start (n = 4096)
        n = 4096
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        alpha = 0.38
        eps = 1e-6
        w = (x * (1.0 - x) + eps) ** (-alpha)
        p = w / w.sum()
        print(f"[{time.time() - start_time:.2f}s] No starter – generated arcsine start (n={n})")

    # --------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------
    best_p = p.copy()
    best_score = _score_of(p)
    print(f"[{time.time() - start_time:.2f}s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Global improvement via LP (if SciPy is available)
    # --------------------------------------------------------------
    if _HAS_SCIPY:
        # give the LP phase roughly the first half of the remaining budget
        lp_deadline = start_time + (budget_s * 0.5)
        iteration_lp = 0
        while time.time() < lp_deadline:
            iteration_lp += 1
            conv = _conv_self(p)
            M = conv.max()
            # build constraint matrix A (sparse)
            A = _lp_matrix(p)
            b = np.full(2 * n - 1, M, dtype=float)

            # linear programme: maximise Σ g_i  → minimise -Σ g_i
            c = -np.ones(n, dtype=float)

            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True}
            )
            if not res.success:
                print(f"[{time.time() - start_time:.2f}s] LP #{iteration_lp} failed: {res.message}")
                break

            g = np.asarray(res.x, dtype=float)
            sum_g = g.sum()
            if sum_g <= 1.0 + 1e-4:           # not enough slack – stop LP phase
                print(f"[{time.time() - start_time:.2f}s] LP #{iteration_lp} sum_g≈{sum_g:.6f} – stop")
                break

            # scale g back to a probability vector
            alpha_scale = 1.0 / sum_g
            p_candidate = alpha_scale * g
            p_candidate = p_candidate / p_candidate.sum()   # numerical safety

            cand_score = _score_of(p_candidate)
            if cand_score < best_score:
                best_score = cand_score
                best_p = p_candidate.copy()
                p = p_candidate.copy()
                print(f"[{time.time() - start_time:.2f}s] LP #{iteration_lp} improved → score={best_score:.6f} (α={alpha_scale:.6f})")
                continue  # go to next LP iteration with the improved p

            # If the pure scaling did not improve, try a convex blend
            t = 0.5
            mixed = (1.0 - t) * p + t * p_candidate
            mixed /= mixed.sum()
            mixed_score = _score_of(mixed)
            if mixed_score < best_score:
                best_score = mixed_score
                best_p = mixed.copy()
                p = mixed.copy()
                print(f"[{time.time() - start_time:.2f}s] LP #{iteration_lp} blend improved → score={best_score:.6f}")
                continue

            # No improvement – stop the LP loop
            print(f"[{time.time() - start_time:.2f}s] LP #{iteration_lp} no improvement (score={cand_score:.6f})")
            break
    else:
        print(f"[{time.time() - start_time:.2f}s] SciPy not available – skipping LP phase")

    # --------------------------------------------------------------
    # 3) Fine‑grained hill‑climbing with simulated annealing
    # --------------------------------------------------------------
    p = best_p.copy()
    conv = _conv_self(p)
    cur_max = conv.max()
    n = p.shape[0]
    rng = np.random.default_rng()
    ite = 0
    last_improve = 0

    # initial temperature (10 % of the worst autocorrelation)
    T0 = cur_max * 0.10

    while time.time() < deadline:
        ite += 1

        # pick a shift attaining the current maximum (break ties randomly)
        max_shifts = np.flatnonzero(conv >= cur_max - 1e-15)
        if max_shifts.size > 1:
            k = rng.choice(max_shifts)
        else:
            k = int(conv.argmax())

        # indices that participate in conv[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # source = index that contributes most to the worst shift
        contrib = p[idx] * p[k - idx]
        src = idx[int(np.argmax(contrib))]

        # destination = an index with little mass (inverse‑weighted sampling)
        inv = 1.0 - p
        inv_sum = inv.sum()
        if inv_sum <= 0.0:
            prob = np.full(n, 1.0 / n)
        else:
            prob = inv / inv_sum
        dst = rng.choice(n, p=prob)
        if dst == src:
            continue

        # step size – decreasing fraction of the source mass
        remaining = deadline - time.time()
        max_frac = max(0.001, 0.20 * (remaining / budget_s))
        epsilon = p[src] * rng.random() * max_frac
        if epsilon <= 0.0:
            continue

        # ---- backup current state (cheap copies) ----
        p_old = p.copy()
        conv_old = conv.copy()

        # ---- exact convolution update (see the analysis) ----
        conv[src:src + n] -= 2.0 * epsilon * p
        conv[dst:dst + n] += 2.0 * epsilon * p
        conv[2 * src] += epsilon * epsilon
        conv[2 * dst] += epsilon * epsilon
        conv[src + dst] -= 2.0 * epsilon * epsilon

        p[src] -= epsilon
        p[dst] += epsilon
        if p[src] < 0.0:
            p[src] = 0.0

        new_max = conv.max()

        # ---- simulated‑annealing acceptance ----
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # linear temperature schedule
            elapsed = time.time() - start_time
            T = max(0.0, T0 * (1.0 - elapsed / budget_s))
            if T > 0.0:
                delta = new_max - cur_max
                prob_acc = math.exp(-delta / T)
                if rng.random() < prob_acc:
                    accept = True

        if accept:
            cur_max = new_max
            cur_score = 2.0 * n * cur_max  # sum(p) ≈ 1
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                last_improve = ite
                print(f"[{time.time() - start_time:.2f}s] iter={ite:,} new best={best_score:.6f}")
        else:
            # revert
            p = p_old
            conv = conv_old

        # ---- periodic re‑normalisation to suppress drift ----
        if ite % 5000 == 0:
            s = p.sum()
            if abs(s - 1.0) > 1e-12:
                p /= s
                conv = _conv_self(p)
                cur_max = conv.max()

        # ---- plateau handling (large jitter) ----
        if ite - last_improve > 200_000:
            p = best_p.copy()
            jitter = 0.005 * p * (rng.random(p.shape) - 0.5)
            p = np.maximum(p + jitter, 0.0)
            p /= p.sum()
            conv = _conv_self(p)
            cur_max = conv.max()
            last_improve = ite
            print(f"[{time.time() - start_time:.2f}s] plateau – jittered at iter {ite}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0.0:
        best_p = best_p / total
    else:
        best_p = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] final best score = {best_score:.6f}")
    return [float(v) for v in best_p]
