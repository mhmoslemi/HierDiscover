    decays with time.

All stages respect the supplied time budget (default 1000 s).  The final
sequence is normalised to sum to one and returned as a plain Python list
ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a non‑negative vector with sum 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    The average of the gradients of all active shifts (those attaining the maximum)
    is returned.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size > 0:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# 1️⃣  Parametric families
# ----------------------------------------------------------------------
def _parametric_sweep(n: int, deadline: float, start_time: float):
    """Return the best sequence found by a quick sweep over geometric / power / beta families."""
    best_seq = np.full(n, 1.0 / n)
    best_score = _objective(best_seq)
    idx = np.arange(n, dtype=np.float64)

    # geometric
    for eps in np.logspace(-5, -1, 120):
        if time.time() >= deadline - 40:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        cur = _objective(geo)
        if cur < best_score:
            best_score, best_seq = cur, geo.copy()

    # power
    for p in np.linspace(-1.0, 2.0, 71):
        if time.time() >= deadline - 40:
            break
        pow_seq = (idx + 1) ** p
        pow_seq /= pow_seq.sum()
        cur = _objective(pow_seq)
        if cur < best_score:
            best_score, best_seq = cur, pow_seq.copy()

    # beta‑type (two‑parameter polynomial)
    beta_vals = [0.0, 0.5, 1.0, 1.5, 2.0, 3.0]
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 40:
                break
            beta_seq = (idx + 1) ** p * (n - idx) ** q
            beta_seq /= beta_seq.sum()
            cur = _objective(beta_seq)
            if cur < best_score:
                best_score, best_seq = cur, beta_seq.copy()

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Linear‑programming refinement (requires SciPy)
# ----------------------------------------------------------------------
def _lp_refinement(a: np.ndarray, t: float, deadline: float):
    """One LP refinement step: minimise M such that a * g ≤ M, Σg = 1, g≥0."""
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except Exception:
        return a, None  # SciPy unavailable

    n = a.shape[0]
    # Build sparse matrix for constraints Σ_i a_i g_{k-i} - M ≤ 0
    # Row k corresponds to shift k (0 … 2n‑2)
    rows = []
    cols = []
    data = []
    for i in range(n):
        row_idx = i + np.arange(n, dtype=np.int32)          # k = i + j
        col_idx = np.arange(n, dtype=np.int32)             # j
        rows.append(row_idx)
        cols.append(col_idx)
        data.append(np.full(n, a[i], dtype=np.float64))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A_g = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    M_col = sparse.csr_matrix((-np.ones((2 * n - 1, 1))), shape=(2 * n - 1, 1))
    A_ub = sparse.hstack([A_g, M_col], format='csr')

    c = np.zeros(n + 1)
    c[-1] = 1.0                         # minimise M
    b_ub = np.zeros(2 * n - 1)          # RHS of ≤‑constraints
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0                    # Σg = 1
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    # Time‑limited solve
    options = {"presolve": True, "time_limit": max(0.0, deadline - time.time())}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options=options)

    if not res.success:
        return a, None

    g = res.x[:n]
    # Mix and renormalise
    new_a = (1.0 - t) * a + t * g
    new_a = np.maximum(new_a, 0.0)
    new_a_sum = new_a.sum()
    if new_a_sum > 0:
        new_a /= new_a_sum
    return new_a, res.x[-1]   # also return the M value (optional)


# ----------------------------------------------------------------------
# 3️⃣  Simple projected sub‑gradient descent with Nesterov momentum
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray, deadline: float):
    """Run a few hundred iterations of projected sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Nesterov parameters
    lr0 = 0.5
    it = 0
    momentum = np.zeros_like(a)
    beta = 0.9

    while time.time() < deadline - 5:
        it += 1
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1 - beta) * grad
        lr = lr0 / np.sqrt(it)   # diminishing step size
        a_new = a - lr * momentum
        a = _project_simplex(a_new)

    return a


# ----------------------------------------------------------------------
# 4️⃣  Stochastic hill‑climb (global polishing)
# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray, deadline: float):
    """Aggressive hill‑climb that transfers a random fraction of mass."""
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(best_a)
    rng = np.random.default_rng()
    max_frac = 0.3               # start with fairly large transfers
    decay_start = time.time()
    while time.time() < deadline:
        # gradually shrink allowed fraction
        elapsed = time.time() - decay_start
        max_frac = max(0.02, max_frac * (0.9995 ** (elapsed * 10)))

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        frac = rng.random() * max_frac
        eps = a[src] * frac
        if eps <= 0:
            continue

        proposal = a.copy()
        proposal[src] -= eps
        proposal[dst] += eps
        # keep numerical sanity
        if proposal[src] < -1e-14:
            continue
        prop_score = _objective(proposal)
        if prop_score < best_score:
            best_score = prop_score
            best_a = proposal.copy()
            a = proposal
            # occasional progress report
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2).

    The routine respects the given time budget and returns a
    plain list of Python floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 1024                     # size that balances LP tractability and quality
    print(f"[{time.time() - start_time:.2f}s] using n={n}")

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep
    # ------------------------------------------------------------------
    best_seq, best_score = _parametric_sweep(n, deadline, start_time)
    print(f"[{time.time() - start_time:.2f}s] after sweep best≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  LP refinement loop (few iterations)
    # ------------------------------------------------------------------
    t = 0.5                      # initial mixing weight
    lp_iter = 0
    while time.time() < deadline - 150 and lp_iter < 8:
        new_seq, M_val = _lp_refinement(best_seq, t, deadline)
        if new_seq is None:
            break                     # SciPy not available or LP failure
        cur_score = _objective(new_seq)
        if cur_score < best_score:
            best_score, best_seq = cur_score, new_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter} improved score={best_score:.6f}")
        else:
            print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter} no improvement")
        t = max(t * 0.7, 0.05)          # shrink step‑size
        lp_iter += 1

    # ------------------------------------------------------------------
    # 3️⃣  Short sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient polishing")
        polished = _subgrad_descent(best_seq, deadline)
        polished_score = _objective(polished)
        if polished_score < best_score:
            best_score, best_seq = polished_score, polished.copy()
            print(f"[{time.time() - start_time:.2f}s] Polishing improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb (majority of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        final_seq, final_score = _hill_climb(best_seq, deadline)
        if final_score < best_score:
            best_score, best_seq = final_score, final_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq.tolist()]
