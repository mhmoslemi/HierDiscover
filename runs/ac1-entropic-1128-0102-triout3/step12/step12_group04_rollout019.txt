def _gradient_from_mask(a, mask, L):
    # mask length should be 2*n-1
    # compute gradient g_i = 2 * sum_t mask[t] * a[t - i]
    rev = a[::-1]
    Wf = np.fft.rfft(mask, n=L)
    Revf = np.fft.rfft(rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)
    n = a.shape[0]
    grad = 2.0 * conv_wr[n-1:n-1+n]
    return grad
```

def _generate_candidates(n):
    x = np.linspace(0, 1, n)
    candidates = []
    # basic shapes
    candidates.append(np.minimum(x, 1-x)) # linear tent
    candidates.append(np.sin(np.pi * x))  # sine
    candidates.append(np.sqrt(np.clip(1 - (2*x - 1)**2, 0, 1))) # half circle
    # parameter sweeps
    # power tent
    for p in [1.2, 1.5, 2, 3, 4]:
        cand = np.minimum(x, 1-x) ** p
        candidates.append(cand)
    # power sine
    for p in [2, 3, 4]:
        cand = np.sin(np.pi * x) ** p
        candidates.append(cand)
    # gaussian sigma
    for sigma in [0.1, 0.15, 0.2, 0.25, 0.3]:
        cand = np.exp(-0.5 * ((x-0.5)/sigma)**2)
        candidates.append(cand)
    # combination maybe product of tent and sine
    cand = np.minimum(x, 1-x) * np.sin(np.pi * x)
    candidates.append(cand)
    # Normalize each
    normalized = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normalized.append(c / s)
    return normalized
```

def _adam_optimize_exact(a0, deadline, start_time,
                         lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3*n)
    best_a = a.copy()
    best_score = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        max_conv = conv.max()
        # identify active indices
        tol = max_conv * 1e-12 + 1e-15  # tolerance
        active = np.where(conv >= max_conv - tol)[0]
        # weight mask
        mask = np.zeros_like(conv)
        if active.size > 0:
            mask[active] = 1.0 / active.size
        else:
            mask[conv.argmax()] = 1.0
        grad = _gradient_from_mask(a, mask, L)
        grad_obj = 2.0 * n * grad  # subgradient of full objective
        m = beta1*m + (1-beta1)*grad_obj
        v = beta2*v + (1-beta2)*(grad_obj**2)
        m_hat = m / (1-beta1**it)
        v_hat = v / (1-beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        # projection
        a = _project_simplex(a)
        # symmetrize
        a = (a + a[::-1]) / 2.0
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam iter={it:,} best={cur_score:.6f}")
    return best_a, best_score

"""
Hybrid optimizer for the max‑convolution objective.

1.  Warm‑start  – generate a diverse set of analytic shapes
    (tent, half‑circle, Gaussian, power‑tents, etc.) and a
    fine‑grained parameter sweep.  The best shape (according to the
    exact objective 2·n·max(conv)) is taken as the seed.

2.  Exact‑subgradient Adam – at each iteration the true sub‑gradient
    of the max‑convolution is obtained by a single FFT‑based
    convolution with a mask that selects the active (max) index.
    Adam’s adaptive step‑size together with Euclidean projection onto
    the probability simplex keeps the vector feasible.
    The iterate is symmetrised after each step to exploit the
    apparent near‑symmetry of good solutions.

3.  Simulated‑annealing polish – a cheap Metropolis walk moves a
    random packet of mass between two positions; a fast exponential
    cooling schedule allows the walk to escape shallow minima.
    The best iterate seen is returned.

The whole search respects the 1000 s budget and returns a plain
list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient from an arbitrary mask (active set) – exact sub‑gradient
# ----------------------------------------------------------------------
def _gradient_from_mask(a: np.ndarray, mask: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the sub‑gradient of the (averaged) convolution over the
    entries selected by ``mask``.  ``mask`` has length 2·n‑1 and
    sums to 1 (or 0 if empty).  The returned vector has length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Wf = np.fft.rfft(mask, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Exact‑subgradient Adam optimiser
# ----------------------------------------------------------------------
def _adam_optimize_exact(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         lr: float = 0.02,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex using the exact sub‑gradient
    of the max‑convolution (mask contains the active indices).
    Returns the best iterate encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)                # FFT size large enough for all convolutions

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L)
        max_val = conv.max()
        # active set – indices where the maximum is attained (numerical tolerance)
        tol = max_val * 1e-12 + 1e-15
        active = np.where(conv >= max_val - tol)[0]

        # build mask (average over the active entries)
        mask = np.zeros_like(conv)
        if active.size > 0:
            mask[active] = 1.0 / active.size
        else:
            mask[conv.argmax()] = 1.0

        # exact sub‑gradient of the objective
        grad = _gradient_from_mask(a, mask, L)          # gradient of max(conv)
        grad_obj = 2.0 * n * grad                       # gradient of 2·n·max(conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the simplex
        a = _project_simplex(a)

        # enforce (optional) symmetry – many good solutions are symmetric
        a = (a + a[::-1]) * 0.5

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (more steps than the reference)
# ----------------------------------------------------------------------
def _anneal_refine(seq: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that moves a random packet of mass between two indices.
    Returns the best sequence seen before the deadline.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 300_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling – fast initial exploration, slow final convergence
        T = 0.02 * (1e-5 / 0.02) ** (it / total_steps)

        # pick a source proportionally to its mass (helps move mass from peaks)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Warm‑starting shapes (analytic families + parameter sweeps)
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a collection of analytically simple candidate vectors."""
    x = np.arange(n, dtype=float) / (n - 1)               # normalised coordinate in [0,1]

    # Basic families
    tent = np.minimum(x, 1.0 - x)                         # symmetric linear tent
    cosine = np.sin(np.pi * x)                            # raised cosine (zero at ends)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))   # half‑circle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)       # Gaussian

    candidates = [tent, cosine, circ, gauss]

    # Parameter sweeps -------------------------------------------------
    # Power‑tent (flatter than linear)
    for p in (1.2, 1.5, 2.0, 3.0, 4.0):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # Power‑cosine
    for p in (2.0, 3.0, 4.0):
        candidates.append(np.sin(np.pi * x) ** p)

    # Half‑circle raised to a power (p < 0.5 gives flatter tops)
    for p in (0.3, 0.4, 0.45):
        candidates.append(np.power(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0), p))

    # Gaussian with different sigmas
    for sigma in (0.07, 0.09, 0.12, 0.15, 0.2, 0.25):
        candidates.append(np.exp(-0.5 * ((x - 0.5) / sigma) ** 2))

    # Normalise each to sum = 1 (simple probability simplex)
    normalized = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normalized.append(c / s)
    return normalized


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ----------------------- problem size -----------------------------
    n = 16384                     # power‑of‑two gives fast FFTs

    # ---------------------- 1) Warm‑start ---------------------------
    warm_starts = _make_warm_starts(n)

    # include the global construction if it exists (might be a different size)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # up‑sample / down‑sample to the current length if necessary
            if starter.size != n:
                idx = np.linspace(0, starter.size - 1, n)
                starter = np.interp(idx, np.arange(starter.size), starter)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                warm_starts.append(starter)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # -------------------- 2) Adam optimisation ----------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + remaining * 0.80   # 80 % of the remaining budget
        a_opt, score_opt = _adam_optimize_exact(best_a,
                                                adam_deadline,
                                                start_time,
                                                lr=0.02)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam final best={score_opt:.6f}")

    # ------------------- 3) Simulated‑annealing polish -------------
    if time.time() < deadline:
        a_ref, score_ref = _anneal_refine(best_a, deadline, start_time)
        if score_ref < best_score:
            best_score = score_ref
            best_a = a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final best={score_ref:.6f}")

    # ------------------- 4) Final normalisation --------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    # return as plain Python list
    return [float(v) for v in best_a]
