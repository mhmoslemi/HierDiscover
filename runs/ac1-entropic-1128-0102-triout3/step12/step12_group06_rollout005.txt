"""
Local‐search optimizer for the step‑function objective.

The algorithm works on the probability simplex (∑a = 1) and keeps an
exact self‑convolution `conv = a*a`.  Each move transfers a small amount
ε of mass from a source index i to a destination j and updates the
convolution analytically (O(n) slice updates).  A simulated‑annealing
schedule allows occasional uphill moves; the best sequence seen so far
is returned.  The implementation is self‑contained, uses only NumPy,
and runs within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _next_pow2(m: int) -> int:
    """Smallest power of two >= m."""
    return 1 << (m - 1).bit_length()


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using FFT of length L (≥2·len(a)-1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.size - 1]
    return conv.real  # drop tiny imag part


def _objective_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Return 2·n·max(conv)/(sum_a**2).  sum_a is assumed > 0."""
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _initial_sequence(n: int) -> np.ndarray:
    """Start from the known best, a triangular shape or uniform."""
    # Try to use the global `height_sequence_1` if it exists
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim != 1 or seq.size == 0:
            raise Exception
        # truncate / pad to required length
        if seq.size > n:
            seq = seq[:n]
        elif seq.size < n:
            seq = np.pad(seq, (0, n - seq.size), mode="constant")
    except Exception:
        # fall back to a symmetric triangular shape
        idx = np.arange(n, dtype=float)
        mid = (n - 1) / 2.0
        seq = np.minimum(idx, n - 1 - idx) + 1.0  # 1,2,...,mid+1,...,2,1
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def _local_search(a: np.ndarray,
                  deadline: float,
                  start_time: float,
                  seed: int = 0) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb with exact convolution updates.
    Returns the best sequence found and its objective value.
    """
    rng = np.random.default_rng(seed)

    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    # current state
    conv = _conv_fft(a, L)
    sum_a = a.sum()                     # should be 1.0
    cur_score = _objective_from_conv(conv, n, sum_a)

    best_a = a.copy()
    best_score = cur_score

    # annealing schedule
    T_start = 5e-2
    T_end = 1e-10

    # max‑transfer factor schedule (large early, tiny later)
    f_start = 0.5
    f_end = 0.01

    # how often we recompute convolution from scratch (avoids drift)
    RECOMPUTE_EVERY = 30000

    it = 0
    while time.time() < deadline - 0.05:
        it += 1

        # progress and schedules
        prog = (time.time() - start_time) / (deadline - start_time)
        T = T_start * (T_end / T_start) ** prog
        max_frac = f_start * (f_end / f_start) ** prog

        # pick source with non‑zero mass
        i = rng.integers(n)
        if a[i] < 1e-14:
            continue
        # pick a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # random transfer amount (up to max_frac of source mass)
        eps = a[i] * rng.random() * max_frac
        if eps <= 0.0:
            continue

        # ---- tentative update of convolution -------------------------
        # we work on a copy to avoid destroying the current state
        new_conv = conv.copy()
        # affect all positions i+t and j+t (t = 0..n-1)
        # subtraction from source
        new_conv[i:i + n] -= 2.0 * eps * a
        # addition to destination
        new_conv[j:j + n] += 2.0 * eps * a

        # new objective (sum stays 1)
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max / (sum_a * sum_a)

        # Metropolis acceptance
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            if T > 0.0 and math.exp(-(new_score - cur_score) / T) > rng.random():
                accept = True

        if accept:
            # commit the move
            a[i] -= eps
            a[j] += eps
            conv = new_conv
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"iter={it:,} new best={best_score:.6f}")

        # periodic full recomputation to keep numerical errors bounded
        if it % RECOMPUTE_EVERY == 0:
            conv = _conv_fft(a, L)
            cur_score = _objective_from_conv(conv, n, sum_a)

        # occasional progress report
        if it % 5000 == 0:
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} "
                  f"cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.

    The routine:
        1. builds an initial probability vector (triangular / known best),
        2. runs a fast simulated‑annealing hill‑climb that updates the
           self‑convolution analytically (O(n) per move),
        3. returns the best sequence encountered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Initialise sequence
    # ------------------------------------------------------------------
    n = 4096                               # default length
    a = _initial_sequence(n)
    n = a.shape[0]                         # in case the starter had a different size

    # ------------------------------------------------------------------
    # Quick sanity check of the initial score
    # ------------------------------------------------------------------
    L = _next_pow2(2 * n - 1)
    init_conv = _conv_fft(a, L)
    init_score = _objective_from_conv(init_conv, n, a.sum())
    print(f"[{time.time() - start_time:.2f}s] initial score = {init_score:.6f}")

    # ------------------------------------------------------------------
    # Phase: local search with analytic convolution updates
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_a, best_score = _local_search(a, deadline, start_time, seed=seed + 1)
        print(f"[{time.time() - start_time:.2f}s] final best = {best_score:.6f}")
    else:
        best_a = a
        best_score = init_score

    # ------------------------------------------------------------------
    # Normalise for the final output (the evaluator will normalise anyway)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    # Optional sanity check using the provided evaluate_sequence (if available)
    try:
        from __main__ import evaluate_sequence  # the evaluator lives in the global namespace
        test_val = evaluate_sequence(best_a.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluator check = {test_val:.6f}")
    except Exception:
        pass

    return [float(v) for v in best_a]
