"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The method consists of three phases:

1️⃣  A quick parametric sweep (geometric·power, exponential, random Dirichlet)
    to obtain a decent starting distribution.

2️⃣  A pure sub‑gradient descent on the *exact* max‑convolution term.
    The sub‑gradient of M = max(conv(a,a)) is computed by
    convolving a uniform weight on the active (max‑tight) indices with the
    reversed vector – all via FFT.  After each step the iterate is projected
    onto the probability simplex (sum = 1, a_i ≥ 0).  A diminishing step‑size
    (η_k = η0/√k) together with occasional tiny noise keeps the dynamics
    from stalling.

3️⃣  A final hill‑climb that transfers a small fraction of mass between
    random coordinates, keeping only improving moves.  The step size is
    gradually reduced.

The whole routine respects the 1000 s time budget, works with up to a few
thousand coefficients (default n=8192) and returns the best list of floats
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / projection utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient descent on M = max(conv(a,a))
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the exact objective.
    The iterate always stays on the probability simplex (sum = 1).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = best_score
    rng = np.random.default_rng()

    # FFT sizes (fixed for the whole run)
    L_conv = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n - 2)          # for correlation w * a_rev

    it = 0
    eta0 = 0.5  # initial step size (will be divided by sqrt(it))

    while time.time() < deadline:
        it += 1

        # ----- compute convolution and current score -----
        conv = _conv_self(a, L=L_conv)
        M = conv.max()
        cur_score = 2.0 * n * M / (a.sum() ** 2)   # a.sum()≈1 because of projection
        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] subgrad it={it:,} score={cur_score:.6f}"
            )

        # ----- build sub‑gradient of M -----
        # active set (indices attaining the maximum, within a tiny tolerance)
        eps = 1e-12
        active = np.flatnonzero(conv >= M - eps)
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[active] = 1.0 / active.size

        # correlation w * a_rev  -> gradient of M
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)  # length L_corr
        start = n - 1
        grad_M = 2.0 * corr[start:start + n]   # shape (n,)

        # ----- projected sub‑gradient step -----
        step = eta0 / math.sqrt(it)
        a_tmp = a - step * grad_M
        a = _project_simplex(a_tmp)

        # occasional tiny random jitter to escape flat plateaus
        if it % 3000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = _project_simplex(a + noise)

    return best_a, best_val


# ----------------------------------------------------------------------
# Final microscopic hill‑climb (expects sum≈1)
# ----------------------------------------------------------------------
def _final_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers between two coordinates.
    Only improving moves are kept; ε≈10⁻³–10⁻²·a_src.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # eps scales with the source mass; allow up to 1 % transfer
        eps = a[src] * rng.random() * 1e-2
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score

                # adapt size to n (pad with zeros or truncate)
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                elif starter.size > n:
                    starter = starter[:n]
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (geometric·power, exponential, random Dirichlet)
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 300.0          # reserve ~5 min for later phases
    idx = np.arange(1, n + 1, dtype=np.float64)

    # geometric × power family
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=35)
    p_grid = np.linspace(-3.0, 2.0, 41)

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        geo = r ** np.arange(n, dtype=np.float64)
        for p in p_grid:
            if time.time() >= sweep_deadline:
                break
            shape = geo * (idx ** p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} "
                        f"λ={lam:.2f} score={sc:.6f}"
                    )

    # exponential decay family
    for alpha in np.linspace(0.0002, 0.012, 45):
        if time.time() >= sweep_deadline:
            break
        shape = np.exp(-alpha * np.arange(n, dtype=np.float64))
        shape /= shape.sum()
        for lam in (0.0, 0.5, 1.0):
            seq = lam * uniform + (1.0 - lam) * shape
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep exp α={alpha:.5f} λ={lam:.2f} "
                    f"score={sc:.6f}"
                )

    # a few random Dirichlet draws (small concentration → heavy tails)
    rng = np.random.default_rng(seed + 12345)
    for _ in range(30):
        if time.time() >= sweep_deadline:
            break
        conc = rng.uniform(0.05, 0.5)
        shape = rng.gamma(conc, 1.0, size=n)
        shape /= shape.sum()
        sc = _eval_fast(shape, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = shape.copy()
            print(
                f"[{time.time() - start_time:.2f}s] random Dirichlet conc={conc:.3f} "
                f"score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent on the exact max‑convolution term
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 120.0:                     # need at least a couple of minutes
        descent_deadline = time.time() + min(70.0, time_left / 2)
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        a_sgd, sc_sgd = _subgradient_descent(
            best_seq, n, descent_deadline, start_time, best_score
        )
        if sc_sgd < best_score:
            best_score = sc_sgd
            best_seq = a_sgd.copy()
            print(f"[{time.time() - start_time:.2f}s] descent improvement score={sc_sgd:.6f}")

        # a few cheap restarts to diversify the search
        for restart_idx in range(4):
            if time.time() >= deadline - 30.0:
                break
            start_a = rng.random(n)
            start_a /= start_a.sum()
            a_tmp, sc_tmp = _subgradient_descent(
                start_a,
                n,
                time.time() + 30.0,
                start_time,
                best_score,
            )
            if sc_tmp < best_score:
                best_score = sc_tmp
                best_seq = a_tmp.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] restart {restart_idx} improvement "
                    f"score={sc_tmp:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improved score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise to sum = 1 and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
