def _smooth_optimize_z(a0, deadline, start_time):
    a = a0.copy()
    # add epsilon to avoid log zero
    eps = 1e-12
    z = np.log(a + eps)
    n = len(a)
    best_a = a.copy()
    best_score = _objective(a)
    # Adam hyperparameters
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9; beta2 = 0.999; eps_adam = 1e-8
    it = 0
    while time.time() < deadline:
        it += 1
        # compute a from z (softmax)
        # using stable softmax
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()
        # compute conv
        conv = _conv_self(a)
        # compute weight w = exp(conv / T)
        # temperature schedule linear decay
        T = max(1e-4, 0.1 * (1 - it / (it + 5000.0))) # adjust
        w_raw = np.exp(conv / T)
        w = w_raw / w_raw.sum()
        # compute grad_a = 2 * correlation of w with reversed a
        a_rev = a[::-1]
        L = _next_pow2(3*n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[:3*n - 2]
        grad_a = 2.0 * corr[n-1:n-1+n]   # shape n
        # project onto simplex gradient: compute grad_z = a * (grad_a - <a, grad_a>)
        inner = np.dot(a, grad_a)
        grad_z = a * (grad_a - inner)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = 0.1 / np.sqrt(it)   # base step
        z -= step * m_hat / (np.sqrt(v_hat) + eps_adam)
        # evaluate objective
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] CGD iter={it} score={cur_score:.6f}")
    return best_a, best_score

    n = a.shape[0]

    eps = 1e-12

    # Adam params
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    best_a = a.copy()
    best_score = _objective(a)
    it = 0

        it += 1
        # compute a from z using softmax
        max_z = np.max(z)
        e = np.exp(z - max_z)
        a = e / e.sum()
        # conv
        conv = _conv_self(a)
        # compute temperature schedule
        # maybe T = max(1e-4, 0.1 * (1 - it/ (it + 5000.0))) or linear decay from 0.5 to 1e-5 across iterations

        T = max(1e-5, 0.5 * (1 - frac))

"""
minimize_selfconv.py

Find a non‑negative step‑function a = (a0,…,a_{n‑1}) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The code works inside a 1000 s budget.  It builds a modest set of
analytic and random initial vectors, then improves the best one with a
soft‑max (log‑sum‑exp) surrogate of the max‑convolution using a
softmax‑parameterisation (the variables are the log‑probabilities of a)
and Adam optimisation.  After that a light simulated‑annealing phase
performs fine‑grained moves.  All vectors are kept on the probability
simplex, so the objective reduces to 2·n·max(conv(a,a)).

The implementation is self‑contained, uses only NumPy and the standard
library, and respects the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generator (power‑ramps with optional plateau)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric shape of length n, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Helper to keep a sorted list of the best candidates
# ----------------------------------------------------------------------
def _update_top(pool, cand, top_k=5):
    """Insert candidate (score, vector) into a size‑limited sorted pool."""
    if len(pool) < top_k:
        pool.append(cand)
        pool.sort(key=lambda x: x[0])
    else:
        if cand[0] < pool[-1][0]:
            pool[-1] = cand
            pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) optimiser on the probability simplex
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate   T·log Σ exp(conv/T)
    using the soft‑max (log‑sum‑exp) gradient.
    The variables are log‑probabilities (z) so the simplex constraint is
    satisfied automatically.
    Returns the best vector seen and its exact objective.
    """
    n = a0.shape[0]
    # initialise log‑probabilities (softmax parametrisation)
    eps = 1e-12
    z = np.log(a0 + eps) - np.log(np.sum(a0 + eps))

    # pre‑compute FFT length for correlation (w * a_rev)
    L = _next_pow2(3 * n - 2)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_a = None
    best_score = np.inf

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- a = softmax(z) -----
        max_z = np.max(z)
        e = np.exp(z - max_z)
        a = e / e.sum()

        # ----- convolution -----
        conv = _conv_self(a)

        # ----- surrogate temperature (linear decay) -----
        fraction = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T_start, T_end = 0.2, 1e-6
        T = max(T_end, T_start * (1.0 - fraction))

        # ----- log‑sum‑exp surrogate weights -----
        maxc = conv.max()
        w_raw = np.exp((conv - maxc) / T)   # shift for stability
        w = w_raw / w_raw.sum()            # Σ w = 1

        # ----- gradient wrt a : 2 * correlation(w, a_rev) -----
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad_a = 2.0 * corr[n - 1 : n - 1 + n]

        # ----- transform to gradient wrt z (softmax Jacobian) -----
        inner = np.dot(a, grad_a)
        grad_z = a * (grad_a - inner)

        # ----- Adam update on z -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.05 / math.sqrt(it)  # decreasing base step
        z -= step * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- exact objective (fast, because sum(a)=1) -----
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Small‑scale simulated‑annealing on the simplex."""
    a = seq.copy()
    n = a.shape[0]
    # exact objective for fast checks
    cur_score = 2.0 * n * _conv_self(a).max()
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 250_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # take mass from a heavily‑weighted index
        dst = rng.integers(n)             # random destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move at most 25 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
