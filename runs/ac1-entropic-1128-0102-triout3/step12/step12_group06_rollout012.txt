"""
Probabilistic step‑function optimizer

The algorithm consists of three stages:

1. **Active‑set linear‑programming descent** –  
   Given a current probability vector `a`, we compute the current
   maximum of the self‑convolution `M = max(a*a)`.  Only the constraints
   that are (almost) active (`conv >= (1‑tol)·M`) are kept and a LP
   `max sum(g)`  subject to `a * g <= M` (element‑wise convolution) is
   solved.  The LP solution `g` is renormalised to a probability vector
   and mixed with the current iterate (`a ← (1‑t)·a + t·g`).  This
   systematic step reduces the peak of the convolution quickly.

2. **Smooth‑max mirror descent** – a continuation of the previously
   published exponentiated‑gradient method, but with a longer
   schedule and momentum (Adam) to fine‑tune the distribution.

3. **Stochastic hill‑climbing** – a final annealed Metropolis walk that
   polishes the result.

The whole procedure respects the 1000 s budget, keeps all coefficients
non‑negative and normalised (∑a = 1) and returns the best sequence found.
"""

# Global problem size (can be changed if desired)

n = 4096                     # length of the step function

# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2)."""
    Af = np.fft.rfft(a, n=_L_SELF)
    conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – active‑set linear‑programming descent
# ----------------------------------------------------------------------
def _lp_descent(a: np.ndarray, deadline: float, start_time: float):
    """
    Reduce the peak of a*a by repeatedly solving a small LP.
    The LP keeps only constraints that are within `tol` of the current max.
    """
    best_a = a.copy()
    best_score = _exact_score(best_a)

    # mixing parameter (convex combination weight)
    t = 0.25
    # tolerance for picking active constraints (relative to current max)
    tol = 0.02        # 2 % band around the current maximum

    it = 0
    while time.time() < deadline - 0.5:
        it += 1

        # current self‑convolution and its maximum
        Af = np.fft.rfft(a, n=_L_SELF)
        conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
        M = conv.max()

        # indices where the convolution is close to the maximum
        active = np.where(conv >= M * (1.0 - tol))[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # build sparse constraint matrix A_ub (|active| × n)
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1, dtype=int)
            coeff = a[k - js]                # f_{k‑j}
            rows.append(np.full_like(js, row_idx))
            cols.append(js)
            data.append(coeff)

        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        data = np.concatenate(data)
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))

        b_ub = np.full(len(active), M, dtype=float)
        c = -np.ones(n)                     # maximize sum(g) → minimize -sum(g)

        # solve the LP (highs‑dll is fast and sparse‑aware)
        res = scipy.optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': max(0.0, deadline - time.time() - 0.1)}
        )
        if not res.success:
            # fallback: stop the LP phase
            print(f"[{time.time() - start_time:.2f}s] LP failed at iter {it}, msg={res.message}")
            break

        g = res.x
        if g.sum() == 0.0:
            break
        g = g / g.sum()                     # renormalise to a probability vector

        # convex combination update
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        a /= a.sum()

        cur_score = _exact_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {it:,} improved score={cur_score:.6f}")

        # slowly tighten the tolerance and mixing weight when progress stalls
        if it % 8 == 0:
            tol = max(tol * 0.9, 0.001)
            t = max(t * 0.95, 0.05)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max mirror descent (exponentiated‑gradient with Adam)
# ----------------------------------------------------------------------
def _smooth_descent(a: np.ndarray, deadline: float, start_time: float):
    """Exponentiated‑gradient descent on a log‑sum‑exp (smooth‑max) objective."""
    best_a = a.copy()
    best_score = _exact_score(best_a)

    # schedule of smoothing parameters α
    alphas = np.geomspace(0.5, 3000.0, num=64)

    # Adam‑style accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    for alpha in alphas:
        if time.time() >= deadline - 0.3:
            break

        while time.time() < deadline - 0.3:
            it += 1

            # 1) self‑convolution a*a
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]

            # 2) smooth‑max weights w = softmax(α·conv)
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_SELF)
            Revf = np.fft.rfft(a_rev, n=_L_SELF)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_SELF)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam update on the log‑domain (mirrors exponentiated gradient)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            eta = 0.05 / (np.sqrt(v_hat) + eps)    # step size
            a = a * np.exp(-eta * m_hat)

            a = np.maximum(a, 1e-30)
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            else:
                a = np.full_like(a, 1.0 / n)

            # occasional exact evaluation
            if it % 5000 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] smooth α={alpha:.1f} it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – direct max‑subgradient descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _max_subgradient_descent(a: np.ndarray, deadline: float, start_time: float):
    """Iteratively push mass away from the index that attains the current max."""
    best_a = a.copy()
    best_score = _exact_score(best_a)

    it = 0
    while time.time() < deadline - 0.2:
        it += 1

        Af = np.fft.rfft(a, n=_L_SELF)
        conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]      # g_i = 2·a_{k‑i}

        eta = 0.5 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        if it % 4000 == 0:
            cur_score = _exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – stochastic hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, start_time: float):
    """Small random mass transfers with Metropolis acceptance."""
    cur = a.copy()
    cur_score = _exact_score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    start_temp = 8e-2
    end_temp = 5e-12

    while time.time() < deadline:
        it += 1
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if cur[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = cur[src] * rng.random() * 0.45   # up to 45 % of source mass
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()                     # renormalise (sum stays 1)

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score

        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.

    The optimiser consists of:
        1. Active‑set linear‑programming descent (rapid peak reduction)
        2. Smooth‑max mirror descent (Adam‑accelerated EG)
        3. Direct max‑subgradient descent (EG)
        4. Stochastic hill‑climbing (simulated annealing)
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3      # safety margin

    # ------------------------------------------------------------------
    # Initialise: try to re‑use a previously known construction, otherwise uniform
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise Exception
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    # adjust length
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _exact_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – active‑set LP descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 250.0:          # reserve ample time for later phases
        phase_deadline = deadline - 150.0
        best_seq, best_score = _lp_descent(best_seq, phase_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After LP descent best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max mirror descent (Adam)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        phase_deadline = deadline - 80.0
        best_seq, best_score = _smooth_descent(best_seq, phase_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – direct max‑subgradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        phase_deadline = deadline - 15.0
        best_seq, best_score = _max_subgradient_descent(best_seq, phase_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After sub‑gradient best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – stochastic hill‑climb (annealing) for final polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
