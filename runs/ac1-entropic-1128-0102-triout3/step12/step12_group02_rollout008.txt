```
def annealing_search(a, cur_score):
    a = a.copy()
    cur_score = cur_score
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    L_fft = _next_pow2(2*n - 1)
    rng = np.random.default_rng(seed+12345) # maybe new seed

    # schedule parameters:
    T0 = (best_score - 1.0) * 0.2 + 1e-3 maybe; use 0.01.
    T_end = 1e-7
    max_iter = int(1e7)   # but break on deadline

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # choose source and destination
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1  # ensure diff
        # step scale relative to a[src]
        scale = max(0.001, (1.0 - iteration / 1e6))  # optional
        eps = a[src] * rng.random() * scale  # up to scale fraction of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score
        # temperature schedule:
        temp = T0 * (T_end / T0) ** (iteration / 1e6)  # maybe use 1e6 iteration scale.

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start:.2f}s] anneal iter={iteration} best={best_score:.6f}")

        # occasional resampling for shaking:
        if iteration % 10000 == 0:
            # jitter
            jitter = rng.normal(scale=0.0001, size=n) * a
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
```

assistantfinal**Solution Overview**

The programme below searches for a non‑negative step‑function that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the linear (zero‑padded) convolution of `a` with itself.  
The search consists of three phases

1. **Parametric sweep** – evaluates a rich family of analytic shapes  
   (power‑law, geometric‑power and symmetric “U’’ shapes).  
   The best shape from this cheap sweep is used as the start point.

2. **Simulated‑Annealing optimisation** – repeatedly moves a small amount of
   mass from a source index to a destination index.  Moves that improve the
   objective are always accepted, worsening moves are accepted with a
   temperature‑dependent probability.  The temperature decays exponentially,
   allowing the algorithm to escape local minima early and to settle fine‑grained
   improvements later.

3. **Deterministic hill‑climb** – after the annealing phase a final
   greedy hill‑climb (only improving moves) is performed to polish the solution.

All heavy lifting (convolution, max‑extraction) is done with FFTs
(`O(n log n)`) and the code respects the 1000 s time budget.
The returned sequence is normalised to sum = 1 (the objective is scale‑free).

```python
"""
Simulated‑annealing optimiser for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser works in three stages:

1️⃣  A fast parametric sweep over several analytic families:
    – pure power law        a_i ∝ (i+α)^p
    – geometric × power   a_i ∝ r^i · (i+α)^p
    – symmetric “U’’     a_i ∝ ((i+α)(N‑1‑i+β))^p   (p<0 gives high ends, low centre)

   The best shape from the sweep seeds the next phase.

2️⃣  Simulated annealing on the exact objective.
    A move consists of transferring a random fraction of the mass
    from one index to another.  Improvements are always kept;
    otherwise the move is accepted with probability exp(‑Δ/T).
    The temperature T decays exponentially.

3️⃣  A final greedy hill‑climb (only improving moves) that
    fine‑tunes the best candidate found.

The routine runs for the full 1000 s budget, prints progress
and finally returns the best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1.
    Returns np.inf for degenerate inputs.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self(a, L=L_fft)
    maxb = conv.max()
    n = a.shape[0]
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Parametric sweep
# ----------------------------------------------------------------------
def _parametric_sweep(n: int, L_fft: int, deadline: float,
                      cur_best_a: np.ndarray, cur_best_score: float):
    """
    Explore several analytic families and keep the best candidate.
    Returns (best_a, best_score).
    """
    rng = np.random.default_rng()
    best_a = cur_best_a.copy()
    best_score = cur_best_score

    idx = np.arange(n, dtype=np.float64)          # 0 … n‑1
    uniform = np.full(n, 1.0 / n)

    # ----- power‑law (monotone) ---------------------------------------
    p_grid = np.linspace(-2.0, 0.0, 41)           # p ≤ 0
    alpha_grid = np.logspace(-2, 2, 9)           # shift to avoid singularity

    for p in p_grid:
        for α in alpha_grid:
            if time.time() >= deadline - 150:
                return best_a, best_score
            shape = (idx + α) ** p
            shape_sum = shape.sum()
            if shape_sum == 0.0:
                continue
            shape /= shape_sum
            # mix with uniform (λ = 0 … 1)
            for lam in (0.0, 0.3, 0.6, 0.9, 1.0):
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()
                    print(f"[{time.time():.2f}s] sweep power p={p:.3f} α={α:.3f} λ={lam:.2f} → {sc:.6f}")

    # ----- geometric × power -------------------------------------------
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), 14)
    for r in r_grid:
        for p in p_grid:
            for α in (0.0, 1.0):
                if time.time() >= deadline - 150:
                    return best_a, best_score
                geo = r ** idx
                shape = geo * (idx + α + 1) ** p
                shape_sum = shape.sum()
                if shape_sum == 0.0:
                    continue
                shape /= shape_sum
                for lam in (0.0, 0.4, 0.8, 1.0):
                    cand = lam * uniform + (1.0 - lam) * shape
                    cand /= cand.sum()
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_score, best_a = sc, cand.copy()
                        print(f"[{time.time():.2f}s] sweep geom r={r:.4f} p={p:.3f} α={α:.1f} λ={lam:.2f} → {sc:.6f}")

    # ----- symmetric “U’’ shapes (product of left–right factors) -------
    p_grid_u = np.linspace(-2.0, -0.1, 31)       # negative exponent → high ends
    for p in p_grid_u:
        for α in (0.01, 0.1, 1.0):
            for β in (0.01, 0.1, 1.0):
                if time.time() >= deadline - 150:
                    return best_a, best_score
                left = idx + α
                right = (n - 1 - idx) + β
                shape = (left * right) ** p
                shape_sum = shape.sum()
                if shape_sum == 0.0:
                    continue
                shape /= shape_sum
                for lam in (0.0, 0.5, 1.0):
                    cand = lam * uniform + (1.0 - lam) * shape
                    cand /= cand.sum()
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_score, best_a = sc, cand.copy()
                        print(f"[{time.time():.2f}s] sweep U p={p:.3f} α={α:.3f} β={β:.3f} λ={lam:.2f} → {sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Simulated annealing on the exact objective
# ----------------------------------------------------------------------
def _annealing_search(a0: np.ndarray, L_fft: int, deadline: float):
    """
    Simulated‑annealing optimiser.
    Returns (best_a, best_score).
    """
    a = a0.copy()
    cur_score = _eval_fast(a, L_fft)

    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]

    # Temperature schedule
    T0 = max(1e-3, (cur_score - 1.0) * 0.02)   # start a few percent of the gap
    T_end = 1e-8
    decay_rate = 0.99995                         # per iteration exponential decay

    T = T0
    it = 0
    print(f"[{time.time():.2f}s] annealing start, init score={cur_score:.6f}")

    while time.time() < deadline:
        it += 1
        # ---- propose a move ------------------------------------------------
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move size adapts with temperature (larger early, tiny later)
        max_frac = 0.3 * (T / T0) + 0.001      # between ~0.001 and 0.3
        eps = a[src] * rng.random() * max_frac

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # ---- evaluate ------------------------------------------------------
        new_score = _eval_fast(a_new, L_fft)
        delta = new_score - cur_score

        # ---- accept / reject ------------------------------------------------
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[{time.time():.2f}s] anneal it={it:,} best={best_score:.6f}")

        # ---- temperature update --------------------------------------------
        T = max(T_end, T * decay_rate)

        # ---- occasional random jitter to escape plateaus --------------------
        if it % 20000 == 0:
            jitter = rng.standard_normal(n) * a * 1e-4
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            cur_score = _eval_fast(a, L_fft)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final greedy hill‑climb (only improving moves)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray, L_fft: int, deadline: float):
    """
    Fine‑grained hill‑climb that only accepts strictly improving mass transfers.
    Returns (best_a, best_score).
    """
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    best_a = a.copy()
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        eps = a[src] * rng.random() * 5e-4      # up to 0.05 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score
```
