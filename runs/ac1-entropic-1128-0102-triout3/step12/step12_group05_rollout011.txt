assistantfinal**Solution Overview**

The evaluation function  

\[
E(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum_i a_i\bigr)^2}
\]

is scale‑invariant – only the *shape* of the non‑negative vector matters.
The best known shape (a mixture of a centre‑heavy and an edge‑heavy power law)
gives a value of about **1.5084**.  

To push this bound lower we:

1. **Search a rich family of analytic shapes.**  
   *Beta* densities cover uniform, centre‑heavy, edge‑heavy and everything in‑between:
   \[
   a_i \propto \Bigl(\tfrac{i+½}{n}\Bigr)^{\alpha-1}
                \Bigl(1-\tfrac{i+½}{n}\Bigr)^{\beta-1}.
   \]
   We evaluate a dense grid of \((\alpha,\beta)\) and also convex mixtures of
   the power‑law centre/edge shapes for many exponents and mixing weights.

2. **Random Dirichlet draws** – to explore the high‑dimensional simplex
   without bias.

3. **Intensive local optimisation** – a simulated‑annealing / Metropolis walk
   that repeatedly moves a small fraction of mass from one index to another.
   The move is accepted if it improves the objective, otherwise with a
   temperature‑controlled probability.
   The walk runs until the time budget is exhausted.

All heavy work (the self‑convolution) is performed with a real FFT
(`np.fft.rfft`), giving an \(O(n\log n)\) cost per evaluation.  
With \(n=4096\) we can comfortably evaluate many thousands of candidates
inside the 1000 s limit.

The final vector is projected onto the non‑negative simplex, returned as a
plain Python list and can be fed directly to `evaluate_sequence`.

```python
"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

using a rich parametric search (Beta‑densities, power‑law mixtures,
random Dirichlet draws) followed by a long simulated‑annealing polish.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (identical to the reference code)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT.  Returns length 2·n‑1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    """Symmetric triangle raised to the power p (p≥0)."""
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # ∈[0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape: (1‑tri)^p."""
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # ∈[0,1]
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑density on the grid (i+0.5)/n, normalised to sum‑1."""
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a += 1e-16
    return a / a.sum()


def _shape_cosine_power(n: int, p: float) -> np.ndarray:
    """
    Cosine hill: ( (cos(π·(2i‑(n‑1))/(n‑1)) + 1)/2 )^p,
    zero at the ends, 1 at the centre.
    """
    i = np.arange(n, dtype=float)
    arg = np.pi * (2.0 * i - (n - 1)) / (n - 1)
    a = (np.cos(arg) + 1.0) / 2.0
    a = np.power(a, p)
    a += 1e-16
    return a / a.sum()

def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """Metropolis walk that transfers a random packet of mass between two sites."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # schedule parameters – chosen to give a slow cooling over the remaining time
    T0 = 0.05
    T_min = 1e-6
    # target number of iterations – we adapt it to the remaining time
    max_iters = 400_000

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = T0 * (T_min / T0) ** (it / max_iters)

        src = rng.choice(n, p=a)               # source chosen proportionally to mass
        dst = rng.integers(n)                  # destination uniform
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 20 % of the source mass
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # accept downhill always, uphill with Metropolis probability
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={best_score:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – leave a little time for the final return
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two yields fast FFTs.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) Analytic candidates (fast, exhaustive sweep)
    # ------------------------------------------------------------------
    candidates = _collect_candidates(n)

    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates, 1):
        if time.time() >= deadline:
            break
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] Candidate {idx}/{len(candidates)} "
              f"score={score:.6f}")

    if best_seq is None:
        # fallback – uniform (should never happen)
        best_seq = _shape_uniform(n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 2) Random Dirichlet sampling (adds diversity)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    while time.time() < deadline - 1.0:   # keep a second for polishing
        conc = rng.choice([0.15, 0.3, 0.6, 1.2])
        raw = rng.gamma(conc, 1.0, size=n)
        a = raw / raw.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Random Dirichlet improved to "
                  f"{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing polish (the heavy part)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polished = _anneal(best_seq, deadline, start_time)
        polished_score = _objective(polished)
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished.copy()
            print(f"[{time.time() - start_time:.2f}s] Polishing improved to "
                  f"{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation & conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # sanity check – the sequence must be a list of floats
    return [float(v) for v in best_seq]
```
