def gradient_descent_softmax(a, deadline, start_time, step0=0.5, T0=0.02, T_end=1e-5):
    n = len(a)
    it = 0
    best = a.copy()
    best_score = _objective(a)
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        # compute softmax weights
        T = max(T_end, T0 * (1 - it / (it + 1000)))  # just decays
        w = np.exp((conv - max_conv) / T)
        w /= w.sum()
        # gradient
        a_rev = a[::-1]
        L = _next_pow2(3*n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[:3*n-2]
        grad = 2.0 * corr[n-1:n-1+n]
        # step size
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)
        # eval
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time()-start_time:.2f}s] SoftmaxGD iter={it}, score={cur_score:.6f}")
    return best, best_score

def targeted_anneal(a, deadline, start_time, init_temp=0.01):
    n = len(a)
    rng = np.random.default_rng()
    cur = a.copy()
    cur_score = _objective(cur)
    best = cur.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline:
        it += 1
        T = init_temp * (0.9 ** (it // 1000))  # exponential cooling
        conv = _conv_self(cur)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi+1, dtype=int)
        # contributions
        contrib = cur[idx] * cur[k - idx]
        total_contrib = contrib.sum()
        if total_contrib == 0:
            src = rng.choice(idx)
        else:
            prob = contrib / total_contrib
            src = rng.choice(idx, p=prob)
        # destination choose index with small a
        # compute probs inversely proportional to a (plus epsilon)
        inv = 1.0 / (cur + 1e-12)
        inv_prob = inv / inv.sum()
        dst = rng.choice(n, p=inv_prob)
        if src == dst or cur[src] < 1e-14:
            continue
        eps = cur[src] * rng.random() * 0.2
        prop = cur.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _objective(prop)
        delta = new_score - cur_score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            cur = prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                print(f"[{time.time()-start_time:.2f}s] TargetedAnneal it={it} best={new_score:.6f}")
    return best, best_score
```

"""
Multi‑scale search for a low‑peak self‑convolution step‑function.

1.  Work first on a modest size (N≈8‑16 k) where many cheap
    gradient steps are feasible.  A soft‑max surrogate of the
    max‑convolution is minimised with projected gradient descent
    (simple step‑size schedule, projection onto the probability
    simplex).

2.  The best small‑scale shape is up‑sampled to the target size
    (N=65536) and refined again with the same surrogate optimiser,
    followed by a true sub‑gradient descent on the exact
    max‑convolution.

3.  A short targeted simulated‑annealing phase moves mass away
    from the indices that actually realise the current maximal
    convolution entry.

All steps respect the 1000 s time budget; the algorithm prints
progress and finally returns a plain list of non‑negative numbers.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “power‑ramp with plateau’’ shape of length n,
    normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0.0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _softmax_gradient_step(a: np.ndarray,
                           T: float,
                           step: float) -> np.ndarray:
    """
    One projected gradient step for the soft‑max surrogate of max(conv).
    Returns the updated vector.
    """
    n = a.shape[0]

    # convolution and max
    conv = _conv_self(a)
    m = conv.max()

    # soft‑max weights
    w = np.exp((conv - m) / T)
    w /= w.sum()

    # gradient = 2 * (w * a_rev)  (correlation)
    a_rev = a[::-1]
    L = _next_pow2(3 * n - 2)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
    grad = 2.0 * corr[n - 1 : n - 1 + n]

    # explicit gradient descent step
    a_new = a - step * grad
    a_new = _project_simplex(a_new)
    return a_new


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.2,
                             T0: float = 0.02,
                             T_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate.
    Uses a simple 1/√t step‑size decay and a linear temperature schedule.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # temperature schedule (linear decay towards T_end)
        frac = it / (it + 1000.0)
        T = max(T_end, T0 * (1.0 - frac))

        step = step0 / math.sqrt(it)
        a = _softmax_gradient_step(a, T, step)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Soft‑GD iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective.
    Uses a diminishing 1/√t step‑size and simplex projection.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        k = int(conv.argmax())                # index of current max entry
        lo = max(0, k - (a.shape[0] - 1))
        hi = min(a.shape[0] - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max‑conv term
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        step = step0 / math.sqrt(it)
        a -= step * grad
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


def _targeted_anneal(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     init_temp: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing that explicitly targets the
    indices contributing to the current maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        # exponential cooling
        T = init_temp * (0.7 ** (it // 500))

        conv = _conv_self(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # contribution of each index to the maximal entry
        contrib = a[idx] * a[k - idx]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
            src = rng.choice(idx, p=prob_src)
        else:
            src = rng.choice(idx)

        # pick a destination that currently carries little mass
        inv = 1.0 / (a + 1e-12)
        prob_dst = inv / inv.sum()
        dst = rng.choice(n, p=prob_dst)

        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.2
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] TargetedAnneal it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
