"""
Propose a step‑function (probability vector) with a low evaluation score.

Algorithm
---------
1.  Use a large power‑of‑two length `n = 65536`.  The convolution
    operations are performed with real FFTs (O(n log n)).

2.  Initialise the vector `a` from the global starter `height_sequence_1`
    (if present) by linear interpolation, otherwise use a uniform
    distribution.

3.  Run a projected Adam optimiser on the *smooth* surrogate of the
    objective  
         fβ(a) = (1/β)·log Σk exp(β·(a*a)[k]) .  
    The gradient of `fβ` is the weighted sum of the usual sub‑gradient
    of each convolution entry, where the weights are the soft‑max of the
    self‑convolution.  The soft‑max temperature β is increased from 1 to
    30 during the run, turning the smooth surrogate gradually into the
    true max.

4.  After the Adam phase (≈ 85 % of the time budget) a cheap simulated‑
    annealing polish moves tiny packets of mass between random positions
    with a Metropolis acceptance rule.

5.  The final vector is projected onto the probability simplex
    (non‑negative, sum = 1) and returned as a plain Python list.

The whole procedure respects the 1000 s wall‑time limit; if time runs
out the best sequence seen so far is returned (the harness will then
call `evaluate_sequence` on it).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: smallest power‑of‑two ≥ x
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two not smaller than ``x``."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution a*a (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length L (L ≥ 2·len(a)-1)."""
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
# Correlation a ★ w  (a convolved with w, no reversal, length n)
# ----------------------------------------------------------------------
def _correlate(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation (a★w)[i] = Σ_j a[j]·w[j+i]  for i = 0…n‑1.
    Implemented via FFT of length L ≥ n+len(w)-1.
    """
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(Af * np.conj(Wf), n=L)
    n = a.shape[0]
    return corr_full[:n]


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx=1, xⱼ≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex using the algorithm of
    Wang & Carreira‑Perpiñán (2013)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:   # numerical fallback
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact (scale‑invariant) objective  f(a) = 2·n·max(a*a)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, L_self: int) -> float:
    """Return 2·n·max(conv(a,a)) (the quantity minimized by `evaluate_sequence`)."""
    n = a.shape[0]
    conv = _conv_self(a, L_self)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Soft‑max weights for a smooth approximation of the max
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return softmax(conv·β) as a probability vector."""
    # shift for numerical stability
    m = conv.max()
    shifted = conv - m
    exps = np.exp(beta * shifted)
    w = exps / exps.sum()
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) with a low
    value of `evaluate_sequence`.  Returns the sequence as a plain Python list.
    """
    # --------------------------------------------------------------
    # 0) basic set‑up
    # --------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # keep a safety margin

    # --------------------------------------------------------------
    # 1) problem size – a power of two for fast FFTs
    # --------------------------------------------------------------
    n = 65536                       # 2¹⁶, gives a fine discretisation
    L_self = _next_pow2(2 * n - 1)   # length for self‑convolution FFT
    L_corr = _next_pow2(3 * n - 2)   # length for correlation (a★w) FFT

    # --------------------------------------------------------------
    # 2) initialise a – try to use the globally stored starter
    # --------------------------------------------------------------
    a = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # linear interpolation to the required length
            if starter.size != n:
                x_src = np.linspace(0.0, 1.0, starter.size)
                x_tgt = np.linspace(0.0, 1.0, n)
                a = np.interp(x_tgt, x_src, starter)
            else:
                a = starter.copy()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
    except Exception:
        pass

    if a is None:
        a = np.full(n, 1.0 / n)          # uniform start

    # --------------------------------------------------------------
    # 3) Adam optimiser on a smooth surrogate of the max‑convolution
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                         # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # soft‑max temperature schedule
    beta_start = 1.0
    beta_end = 30.0
    beta_ramp_iters = 20000

    best_a = a.copy()
    best_score = _objective(a, L_self)

    it = 0
    # leave a slice of time for the final annealing polish
    remaining = deadline - time.time()
    sub_deadline = time.time() + 0.85 * remaining

    while time.time() < sub_deadline:
        it += 1

        # ---------- convolution ----------
        conv = _conv_self(a, L_self)                 # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max                # sum(a)=1 => objective = 2 n max_B

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.1f}s] iter={it:,}  score={cur_score:.6f}")

        # ---------- soft‑max weights ----------
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp_iters)
        w = _softmax_weights(conv, beta)            # size 2n‑1, sums to 1

        # ---------- gradient (smooth max) ----------
        # g_i = 2 * Σ_k w_k * a_{k-i}   = 2 * (a ★ w)[i]
        grad = 2.0 * _correlate(a, w, L_corr)       # length n

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)                     # mild decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # ---------- projection onto simplex ----------
        a = _project_simplex(a_new)

        # occasional progress report (every ~30 s)
        if (time.time() - start_time) % 30.0 < 0.2:
            print(f"[{time.time() - start_time:.1f}s] iter={it:,}  beta={beta:.2f}")

    # --------------------------------------------------------------
    # 4) Light simulated‑annealing polish (if there is time left)
    # --------------------------------------------------------------
    def _anneal(seq: np.ndarray, deadline_inner: float) -> tuple[np.ndarray, float]:
        """Move small packets of mass between random positions; Metropolis rule."""
        a_loc = seq.copy()
        cur_score = _objective(a_loc, L_self)
        best_loc = a_loc.copy()
        best_sc = cur_score
        rng = np.random.default_rng()
        max_steps = 300_000

        for i in range(1, max_steps + 1):
            if time.time() >= deadline_inner:
                break
            # exponential cooling (fast)
            T = 0.02 * (1e-6 / 0.02) ** (i / max_steps)

            src = rng.choice(n, p=a_loc)          # more mass → more likely to move
            dst = rng.integers(n)
            if src == dst or a_loc[src] < 1e-15:
                continue

            eps_move = a_loc[src] * rng.random() * 0.25
            a_cand = a_loc.copy()
            a_cand[src] -= eps_move
            a_cand[dst] += eps_move

            new_sc = _objective(a_cand, L_self)
            delta = new_sc - cur_score

            if delta < 0.0 or math.exp(-delta / T) > rng.random():
                a_loc = a_cand
                cur_score = new_sc
                if new_sc < best_sc:
                    best_sc = new_sc
                    best_loc = a_loc.copy()
                    print(f"[{time.time() - start_time:.1f}s] anneal step={i:,}  best={new_sc:.6f}")

        return best_loc, best_sc

    if time.time() < deadline:
        best_a, best_score = _anneal(best_a, deadline)

    # --------------------------------------------------------------
    # 5) final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 0.0:
        best_a /= s
    else:
        best_a = np.full(n, 1.0 / n)

    final_eval = _objective(best_a, L_self)
    print(f"[{time.time() - start_time:.1f}s] finished – final score ≈ {final_eval:.6f}")

    return [float(v) for v in best_a]
