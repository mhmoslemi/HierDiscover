Pseudo:

def simplex_projection(v):
    # v may have negative; returns projection onto {x>=0, sum=1}
    # Use sorting algorithm.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    # find rho = max i where cond true
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def proj_simplex(v):
    """Project v onto the probability simplex (nonnegative, sum = 1)."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # Find rho
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def pgd_optimize(p_init, n, deadline):
    p = p_init.copy()
    conv = _conv_self(p)
    max_c = conv.max()
    best_p = p.copy()
    best_score = 2 * n * max_c

    step = 0.5   # initial step size
    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1
        k = int(conv.argmax())
        max_c = conv.max()

        # compute gradient
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g = np.zeros(n, dtype=float)
        # for each i in idx, gradient = 2 * p[k - i]
        g[idx] = 2.0 * p[k - idx]

        # line search
        success = False
        for _ in range(10):
            # propose new p
            p_new = proj_simplex(p - step * g)
            # evaluate conv for new p
            conv_new = _conv_self(p_new)
            max_new = conv_new.max()
            if max_new < max_c - 1e-16:
                success = True
                break
            else:
                step *= 0.5  # reduce step
                if step < 1e-12:
                    break

        if not success:
            # fallback: random mass transfer (like earlier algorithm)
            # simple move: pick src idx with large values, dst low values
            # keep within time
            src = np.argmax(p)
            # choose destination among smallest
            inv = 1.0 - p
            prob = inv / inv.sum()
            dst = np.random.choice(n, p=prob)
            if dst != src:
                epsilon = p[src] * 0.1 * np.random.rand()
                if epsilon > 0:
                    p_temp = p.copy()
                    p_temp[src] -= epsilon
                    p_temp[dst] += epsilon
                    p_temp = proj_simplex(p_temp)  # ensure sum=1
                    conv_temp = _conv_self(p_temp)
                    max_temp = conv_temp.max()
                    if max_temp < max_c:
                        p_new = p_temp
                        conv_new = conv_temp
                        max_new = max_temp
                        success = True

        if success:
            p = p_new
            conv = conv_new
            max_c = max_new
            score = 2.0 * n * max_c
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                last_improve_iter = iteration
                print(f"[{time.time() - start_time:.2f}s] iter={iteration} new best={best_score:.6f}")

            p = np.maximum(p + jitter, 0)

            conv = _conv_self(p)

"""
Projected‑gradient optimisation of a step function.

The evaluation function is scale‑invariant, therefore we work with a
probability vector p (p_i ≥ 0, Σ p_i = 1).  For a fixed length n the score is

        score(p) = 2·n·max_k (p * p)[k] ,

where * denotes linear convolution.  The sub‑gradient of the active shift
k* (the one attaining the maximum) is very simple:

        ∂/∂p_i  (p * p)[k*] = 2·p_{k*‑i}   (if the partner index lies inside 0…n‑1,
                                          otherwise 0).

Hence a cheap projected‑gradient step can be performed:
    p ← Π_S ( p – η·g ),
where Π_S projects onto the simplex S={x≥0, Σx=1} and g is the sub‑gradient.
A line‑search adapts the step size η; after a long plateau a small random
jitter is added to escape local minima.  Several initial shapes (arcsine‑type
with different exponents) and a few random Dirichlet starts are tried; each
run is given a short time slice, the best vector over the whole budget is
returned.

The algorithm runs within the 1000 s limit, uses only numpy (FFT based
convolution) and no external optimisation libraries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (fast for the sizes we use)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    # real FFT – the routine returns the non‑negative frequencies only
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)          # 1‑based index
    # find the largest ρ such that u_ρ > (cssv_ρ)/ρ
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is all zeros or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Sub‑gradient for the active (maximal) shift
# ----------------------------------------------------------------------
def _gradient_max_shift(p: np.ndarray, k: int) -> np.ndarray:
    """Return the sub‑gradient of max_k (p*p)[k] at the shift k."""
    n = p.shape[0]
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    g = np.zeros_like(p)
    # for every i that participates in conv[k] we have ∂/∂p_i = 2·p_{k‑i}
    g[idx] = 2.0 * p[k - idx]
    return g

# ----------------------------------------------------------------------
# Projected‑gradient refinement (runs until *deadline* seconds have passed)
# ----------------------------------------------------------------------
def _pgd_refine(p_start: np.ndarray,
                n: int,
                deadline: float,
                start_time: float):
    """
    Perform a projected‑gradient descent on the simplex starting from p_start.
    Returns the best vector found and its corresponding score.
    """
    p = p_start.copy()
    conv = _conv_self(p)
    cur_max = conv.max()
    best_p = p.copy()
    best_score = 2.0 * n * cur_max

    # step‑size parameters
    step = 0.4          # initial step
    min_step = 1e-12
    max_step = 0.5
    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # active shift
        k = int(conv.argmax())
        cur_max = conv.max()

        # sub‑gradient for this shift
        g = _gradient_max_shift(p, k)

        # ---- line search -------------------------------------------------
        success = False
        trial_step = step
        for _ in range(10):
            p_candidate = _proj_simplex(p - trial_step * g)
            conv_candidate = _conv_self(p_candidate)
            new_max = conv_candidate.max()
            if new_max < cur_max - 1e-18:          # strict improvement
                success = True
                break
            trial_step *= 0.5
            if trial_step < min_step:
                break

        if success:
            # accept the move
            p = p_candidate
            conv = conv_candidate
            cur_max = new_max
            cur_score = 2.0 * n * cur_max
            step = min(trial_step * 1.1, max_step)   # slightly enlarge after success
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                last_improve = it
                print(f"[{time.time() - start_time:.2f}s] it={it:,} new best={best_score:.6f}")
        else:
            # no improvement – shrink the step size
            step = max(step * 0.5, min_step)

        # ---- occasional random jitter to escape plateaus ------------------
        if it - last_improve > 200_000:
            jitter = 0.005 * p * (np.random.rand(n) - 0.5)
            p = _proj_simplex(p + jitter)
            conv = _conv_self(p)
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] plateau – jittered at it={it}")

    return best_p, best_score

# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step function within the time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Try to reuse a previously discovered construction, if available
    # ------------------------------------------------------------------
    try:
        p0 = np.asarray(height_sequence_1, dtype=float)   # may be defined globally
        if p0.ndim != 1 or p0.size == 0:
            raise ValueError
        p0 = np.maximum(p0, 0.0)
        if p0.sum() == 0.0:
            raise ValueError
        p0 = p0 / p0.sum()
        print(f"[{time.time() - start_time:.2f}s] Using existing height_sequence_1 (n={p0.size})")
    except Exception:
        p0 = None

    # ------------------------------------------------------------------
    # 2) Settings for the search
    # ------------------------------------------------------------------
    n_candidates = [4096, 8192]               # two lengths – larger may give lower constant
    alphas = [0.20, 0.30, 0.38, 0.45, 0.55]   # shape parameters for the arcsine family
    best_score = float("inf")
    best_vec = None

    # ------------------------------------------------------------------
    # 3) Try a handful of deterministic starts (arcsine family)
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time.time() >= deadline:
            break

        # prepare a base vector (either reuse p0 or generate a fresh arcsine)
        base_vec = None
        if p0 is not None and p0.size == n:
            base_vec = p0.copy()
        elif p0 is not None and p0.size < n:
            # pad the old vector with a tiny uniform tail and renormalise
            pad = np.full(n - p0.size, 1e-12, dtype=float)
            base_vec = np.concatenate([p0, pad])
            base_vec = base_vec / base_vec.sum()
        else:
            base_vec = None        # will be generated inside the α‑loop

        for alpha in alphas:
            if time.time() >= deadline:
                break

            if base_vec is None:
                # arcsine‑type seed: w_i = (x_i·(1‑x_i)+eps)^(‑α)
                x = (np.arange(n, dtype=float) + 0.5) / n
                eps = 1e-6
                w = (x * (1.0 - x) + eps) ** (-alpha)
                p = w / w.sum()
            else:
                p = base_vec.copy()

            # quick evaluation
            cur_max = _conv_self(p).max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_vec = p.copy()
                print(f"[{time.time() - start_time:.2f}s] init n={n} α={alpha:.2f} score={cur_score:.6f}")

            # ------------------------------------------------------------------
            # 4) Refine the seed with projected‑gradient descent
            # ------------------------------------------------------------------
            remaining = deadline - time.time()
            # give each refinement up to 30 s (or whatever remains)
            run_time = min(30.0, remaining)
            if run_time < 5.0:
                continue
            sub_deadline = time.time() + run_time
            p_opt, sc_opt = _pgd_refine(p, n, sub_deadline, start_time)
            if sc_opt < best_score:
                best_score = sc_opt
                best_vec = p_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] PGD improved n={n} α={alpha:.2f} score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 5) A few random Dirichlet starts for extra diversity
    # ------------------------------------------------------------------
    for _ in range(3):
        if time.time() >= deadline:
            break
        n = 4096
        p = np.random.dirichlet(np.ones(n))
        cur_score = 2.0 * n * _conv_self(p).max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = p.copy()
            print(f"[{time.time() - start_time:.2f}s] random start score={cur_score:.6f}")

        remaining = deadline - time.time()
        run_time = min(20.0, remaining)
        if run_time < 5.0:
            continue
        sub_deadline = time.time() + run_time
        p_opt, sc_opt = _pgd_refine(p, n, sub_deadline, start_time)
        if sc_opt < best_score:
            best_score = sc_opt
            best_vec = p_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] random PGD improved score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 6) Final clean‑up and return
    # ------------------------------------------------------------------
    if best_vec is None:
        # as a last resort fall back to a uniform distribution
        n = 4096
        best_vec = np.full(n, 1.0 / n, dtype=float)

    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec = best_vec / total
    else:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.size)

    # The evaluator treats any positive scaling equally; returning a probability
    # vector keeps the numbers small and well‑behaved.
    return [float(v) for v in best_vec]
