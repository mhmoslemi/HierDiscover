"""
Propose a non‑negative step function that (approximately) solves

    minimise   2·n·max(conv(a,a)) / (sum(a)**2)

by a combination of:

1.   Several cheap parametric seeds (uniform, geometric, power, sinusoid).
2.   A multi‑start Adam optimiser on the *log‑softmax* (latent) variables,
     i.e. we optimise y where a = softmax(y) and keep a on the probability
     simplex without an explicit projection.  The gradient on y is obtained
     from the sub‑gradient of the max‑convolution.
3.   A simulated‑annealing hill‑climb that transfers a random fraction of mass
     between positions (with a slowly decreasing temperature) and accepts
     uphill moves with a Boltzmann probability.  Symmetry a_i = a_{n‑1‑i}
     is enforced after each step, which reduces the effective dimension and
     respects the inherent symmetry of the problem.

The routine respects a total time budget (default 1000 s) and returns a plain
list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    If several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # active shifts (tiny tolerance)
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size > 0:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Softmax utilities (log‑space representation)
# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n‑1‑i} and renormalise."""
    a = 0.5 * (a + a[::-1])
    total = a.sum()
    if total > 0:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax variables
# ----------------------------------------------------------------------
def _adam_optimize_y(
    y0: np.ndarray,
    start_time: float,
    deadline: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on the latent logits y (a = softmax(y)).
    Returns the best a encountered and its 2·n·max(conv) score.
    """
    y = y0.copy()
    n = y.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = None
    best_score = best_score_initial
    it = 0
    last_improve = 0

    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        # ---- a = softmax(y) ----
        a = _softmax(y)
        a = _symmetrize(a)

        # ---- evaluate ----
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # ---- gradient w.r.t. a (sub‑gradient) ----
        grad_a = _subgradient_maxconv(a, conv)

        # ---- convert to gradient w.r.t. y (softmax Jacobian) ----
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # ---- Adam update on y ----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y -= step

        # ---- plateau handling ----
        if it - last_improve > 4000:
            # Small Gaussian perturbation to escape flat regions
            noise = 1e-4 * rng.standard_normal(n)
            y += noise
            last_improve = it

    if best_a is None:
        # Fallback: use the final y
        a = _softmax(y)
        a = _symmetrize(a)
        best_a = a
        conv = _conv_self(best_a, L_fft)
        best_score = float(2.0 * n * conv.max())

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (mass transfers)
# ----------------------------------------------------------------------
def _hill_climb_simulated(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with temperature schedule.
    Moves a tiny random fraction of mass from one index to another;
    uphill moves are occasionally accepted (Boltzmann rule).
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # current state
    cur_a = a.copy()
    conv = _conv_self(cur_a, L_fft)
    cur_score = float(2.0 * n * conv.max())

    # best ever
    best_a = cur_a.copy()
    best_score = cur_score

    rng = np.random.default_rng()

    # temperature schedule (linear cooling)
    T_start = 0.02
    T_end = 1e-4
    total_time = deadline - start_time

    while time.time() < deadline:
        elapsed = time.time() - start_time
        T = T_start + (T_end - T_start) * (elapsed / total_time)

        src = rng.integers(n)
        if cur_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # fraction of source mass to move (up to ~5 %)
        frac = rng.random() * 0.05 + 1e-6
        delta = cur_a[src] * frac

        prop_a = cur_a.copy()
        prop_a[src] -= delta
        prop_a[dst] += delta

        # keep symmetry after the move
        prop_a = _symmetrize(prop_a)

        conv_prop = _conv_self(prop_a, L_fft)
        prop_score = float(2.0 * n * conv_prop.max())

        if prop_score < cur_score:
            # always accept improvement
            cur_a = prop_a
            cur_score = prop_score
            if prop_score < best_score:
                best_score = prop_score
                best_a = prop_a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb new best={best_score:.6f}")
        else:
            # uphill acceptance
            if rng.random() < math.exp(-(prop_score - cur_score) / max(T, 1e-12)):
                cur_a = prop_a
                cur_score = prop_score
        # loop continues

    return best_a, best_score

                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  cheap parametric seeds (to diversify the starting points)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    seeds = []

    # geometric (r close to 1)
    r = 0.9992
    geo = np.power(r, idx)
    geo /= geo.sum()
    seeds.append(geo)

    # power law
    p = -0.6
    pow_seq = np.power(idx + 1, p)
    pow_seq /= pow_seq.sum()
    seeds.append(pow_seq)

    # sinusoidal shape (half‑period)
    sin_seq = np.sin(np.pi * idx / (n - 1))
    sin_seq = np.maximum(sin_seq, 0.0)
    sin_seq /= sin_seq.sum()
    seeds.append(sin_seq)

    # add a few random Dirichlet samples
    rng = np.random.default_rng()
    for _ in range(3):
        rand = rng.dirichlet(alpha=np.ones(n))
        seeds.append(rand)

    # evaluate seeds and keep the best as a first Adam start
    for s in seeds:
        cur = float(2.0 * n * _conv_self(s, L_fft).max())
        if cur < best_score:
            best_score = cur
            best_seq = s.copy()
            print(f"[{time.time() - start_time:.2f}s] Seed improvement score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  multi‑start Adam optimisation (log‑softmax representation)
    # ------------------------------------------------------------------
    # reserve ~150 s for the final hill‑climb
    hill_time = 150.0
    adam_deadline = deadline - hill_time

    # we will perform a few Adam runs, each limited by the remaining time
    run_counter = 0
    while time.time() < adam_deadline - 5.0:          # keep a small safety margin
        run_counter += 1

        # choose an initial point
        if run_counter == 1:
            init_a = best_seq.copy()
        else:
            # random Dirichlet for diversity
            init_a = rng.dirichlet(alpha=np.ones(n))

        # convert to logits (log‑softmax variables)
        y0 = np.log(init_a + 1e-12)          # avoid log(0)

        # allocate up to 180 s for a single run (or whatever is left)
        run_deadline = min(adam_deadline, time.time() + 180.0)

        a_opt, sc_opt = _adam_optimize_y(y0, start_time, run_deadline, best_score)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam run {run_counter} improved to {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  final stochastic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb_simulated(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final best={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # final normalisation and safety clipping
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
