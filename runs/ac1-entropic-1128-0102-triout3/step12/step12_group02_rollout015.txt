"""
Efficient search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines several ideas:

1️⃣  A broad parametric sweep over many classic window families
    (geometric‑power, Hann, Hamming, Blackman, Kaiser) as well as
    simple mixtures with the uniform distribution.

2️⃣  A soft‑max Adam optimiser that follows a smooth surrogate of the
    “max‑conv’’ term and gradually sharpens it (β grows to a large value).

3️⃣  A true sub‑gradient descent on the exact ‑max‑conv objective.
    The sub‑gradient for max(conv) is known analytically, so we can
    take fast projected steps on the probability simplex.

4️⃣  A final random hill‑climb that transfers a tiny fraction of mass
    between random coordinates, accepting only improving moves.

All stages respect the global 1000 s time budget and keep the best
sequence found.  The implementation is fully self‑contained, uses only
NumPy (and SciPy if available for the Kaiser window) and works for any
sequence length – the default is n=4096 (the same size as the reference
solution).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (no closures, top‑level)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with `sum≈1`."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # the denominator is (sum(a))**2 – the known invariance makes the ratio
    # independent of the absolute scale, so we can keep the raw max.
    return float(2.0 * n * max_b / (s * s))

# 1️⃣  Parametric sweep (geometric‑power + classic windows)
# ----------------------------------------------------------------------
def _geometric_power_sweep(n, uniform, L_fft, deadline, start_time):
    """Return best sequence from the geometric‑power family."""
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)

    idx = np.arange(1, n + 1, dtype=np.float64)  # (i+1)
    # geometric bases (log‑spaced – very slow decay)
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    # power exponents ranging from strongly decreasing to increasing
    p_grid = np.linspace(-3.0, 2.0, 41)

    for r in r_grid:
        if time.time() >= deadline - 180:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 180:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} "
                          f"λ={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _window_sweep(n, uniform, L_fft, deadline, start_time):
    """Explore classic spectral windows (Hann, Hamming, Blackman, Kaiser)."""
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)

    # Hann window: w[i] = 0.5 * (1 - cos(2π i/(n-1)))
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))

    # Hamming window: w[i] = 0.54 - 0.46*cos(2π i/(n-1))
    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))

    # Blackman window: w[i] = 0.42 - 0.5*cos(2π i/(n-1)) + 0.08*cos(4π i/(n-1))
    blackman = (
        0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )

    windows = {
        "hann": hann,
        "hamming": hamming,
        "blackman": blackman,
    }

    # Try the three basic windows plus mixtures with uniform
    for name, w in windows.items():
        if time.time() >= deadline - 180:
            break
        w = np.maximum(w, 0.0)
        w /= w.sum()
        for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
            seq = lam * uniform + (1.0 - lam) * w
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] {name} λ={lam:.2f} score={sc:.6f}")

    # Optional Kaiser window (requires scipy).  If SciPy is missing we skip it.
    try:
        from scipy.special import i0  # type: ignore
    except Exception:
        i0 = None

    if i0 is not None:
        beta_vals = np.linspace(0.0, 15.0, 31)   # typical range for Kaiser
        # symmetric index centred at (n-1)/2
        centre = (n - 1) / 2.0
        idx = np.arange(n, dtype=np.float64) - centre
        for beta in beta_vals:
            if time.time() >= deadline - 180:
                break
            # Kaiser formula
            arg = beta * np.sqrt(1.0 - (idx / centre) ** 2)
            kaiser = i0(arg) / i0(beta)
            kaiser = np.maximum(kaiser, 0.0)
            kaiser /= kaiser.sum()
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = lam * uniform + (1.0 - lam) * kaiser
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] kaiser β={beta:.2f} λ={lam:.2f} score={sc:.6f}")
    return best_seq, best_score

# 2️⃣  Smooth‑max Adam descent (soft‑max surrogate)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float, L_conv: int, L_corr: int):
    """Gradient of (1/β)·log Σ exp(β·conv) using FFT‑based correlation."""
    conv = _conv_self(a, L=L_conv)               # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                       # stable shift
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()
    # correlation of w with reversed a
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)          # linear convolution
    start = a.shape[0] - 1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


def _smooth_descent(a0: np.ndarray, n: int, deadline: float,
                    start_time: float, best_score_initial: float):
    """Adam‑type descent on the smooth‑max surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # FFT sizes (fixed)
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # temperature schedule
    beta = 5.0
    beta_mul = 1.015
    beta_max = 2000.0   # much larger than the previous code

    while time.time() < deadline:
        t += 1
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        # exact score (sum of a is 1 → denominator = 1)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} "
                  f"score={cur_score:.6f}")

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)          # a bit larger step
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional tiny random perturbation to avoid plateaus
        if t % 5000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Exact sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score_initial: float):
    """Projected sub‑gradient descent directly on max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    t = 0
    # step‑size schedule parameters
    eta0 = 0.5
    while time.time() < deadline:
        t += 1
        # compute convolution and its maximum
        conv = _conv_self(a)
        max_val = conv.max()
        ks = np.where(conv == max_val)[0]

        # sub‑gradient of max(conv) (average over all active k)
        grad = np.zeros_like(a)
        for k in ks:
            # i ranges where both i and k-i are inside [0, n‑1]
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1)
            j_idx = k - i_idx
            grad[i_idx] += 2.0 * a[j_idx]
        grad /= max(1, len(ks))

        # diminishing step size
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad
        a = _project_simplex(a)

        # evaluate exact objective after projection
        cur_score = _eval_fast(a, _next_pow2(2 * n - 1))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={t:,} "
                  f"score={cur_score:.6f}")

        # occasional random shake‑up
        if t % 2000 == 0:
            eps = 1e-3 * rng.random() * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb (tiny random mass transfers)
# ----------------------------------------------------------------------
def _final_hill_climb(start_a: np.ndarray, deadline: float,
                      start_time: float, L_fft: int):
    """Random pairwise mass transfers, keep only improving moves."""
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to 0.3 % of src mass (larger than the previous 0.1 %)
        eps = a[src] * rng.random() * 3e-3
        anew = a.copy()
        anew[src] -= eps
        anew[dst] += eps
        cur = _eval_fast(anew, L_fft)
        if cur < best_score:
            a = anew
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point – called by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the provided
    evaluation function.  The routine respects a 1000 s budget and
    returns the best sequence (plain Python list) discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # problem dimension (feel free to tune – larger n gives more freedom,
    #   smaller n gives faster iterations)
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline – uniform distribution (already normalised, sum≈1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # try to seed with the globally supplied starter, if any
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  rich parametric sweep (geometric‑power + classic windows)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:   # reserve time for later phases
        seq, sc = _geometric_power_sweep(n, uniform, L_fft, deadline, start_time)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

        seq, sc = _window_sweep(n, uniform, L_fft, deadline, start_time)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # 2️⃣  smooth‑max Adam descent (the heavy lifting)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        print(f"[{time.time() - start_time:.2f}s] start smooth‑max descent")
        # allow a few random restarts – each gets a slice of the remaining budget
        remaining = deadline - time.time()
        restarts = 3
        for r in range(restarts):
            portion = remaining / (restarts - r)
            # small random jitter around the current best
            jitter = 1e-3 * np.random.random(n) * best_seq
            a0 = np.maximum(best_seq + jitter, 0.0)
            a0 = _project_simplex(a0)
            a_opt, sc_opt = _smooth_descent(a0, n, time.time() + portion,
                                            start_time, best_score)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] smooth‑descent restart {r+1} "
                      f"improved score={sc_opt:.6f}")
            remaining = deadline - time.time()
            if remaining < 30:
                break

    # ------------------------------------------------------------------
    # 3️⃣  exact sub‑gradient descent on the true max‑conv objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        print(f"[{time.time() - start_time:.2f}s] start sub‑gradient descent")
        a_opt, sc_opt = _subgradient_descent(best_seq, n,
                                             deadline - 5, start_time,
                                             best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient improved score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  final hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _final_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # final normalisation (sum → 1) and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
