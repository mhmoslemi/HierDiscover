"""
Propose a non‑negative step function (probability vector) with a low
``evaluate_sequence`` score.

Algorithm
---------
1. **Problem size** – use a power‑of‑two length ``n = 65536`` (FFT‑friendly).
2. **Start point** – try the globally supplied ``height_sequence_1``,
   otherwise fall back to a random Dirichlet vector.
3. **Exponentiated‑gradient descent** – minimise the scale‑invariant
   objective  ``2·n·max(conv(a,a))/ (sum(a)**2)``.
   * The self‑convolution is obtained by a real FFT (zero‑padded to a
     length that also accommodates the gradient convolution).
   * A soft‑max over the convolution entries (temperature ``β``) yields a
     smooth approximation of the sub‑gradient of the ``max``.
   * The gradient is ``2·(w * a_rev)`` where ``w`` are the soft‑max
     weights and ``a_rev`` is ``a`` reversed.  This convolution is also
     performed by FFT.
   * Update rule:  ``a ← a·exp(-η·grad)`` (multiplicative, keeps the vector
     on the simplex).  Step size ``η = η₀ / sqrt(t)`` and the temperature
     ``β`` is increased slowly to focus on the true maximum.
4. **Light simulated‑annealing polish** – after the gradient phase move
   small packets of mass between random positions, accepting moves with a
   Metropolis rule.
5. **Finalisation** – clip negatives, renormalise to sum 1 and return a
   plain Python list.

The whole search respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Fast linear self‑convolution using a real FFT
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using a real FFT of length L.
    ``L`` must be ≥ 2·len(a)-1.
    """
    A = np.fft.rfft(a, n=L)
    full = np.fft.irfft(A * A, n=L)
    return full[: 2 * a.shape[0] - 1]

# ----------------------------------------------------------------------
# Scale‑invariant objective for a probability vector (sum may deviate slightly)
# ----------------------------------------------------------------------
def _objective_norm(a: np.ndarray, n: int, L: int) -> float:
    """
    Return 2·n·max(conv(a,a)) / (sum(a)**2).
    ``L`` is the FFT length used for the convolution (must be ≥ 2·n‑1).
    """
    conv = _conv_self_fft(a, L)
    maxc = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * maxc / (s * s))

# ----------------------------------------------------------------------
# Gradient from soft‑max weights
# ----------------------------------------------------------------------
def _grad_from_weights(w: np.ndarray, a_rev: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Compute 2·(w * a_rev) where * denotes linear convolution.
    ``w`` has length 2·n‑1, ``a_rev`` length n, result length n.
    """
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A_rev, n=L)          # length L
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad

# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (same as the previous script)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float,
            n: int, L: int) -> tuple[np.ndarray, float]:
    """Move tiny packets of mass between random positions; Metropolis rule."""
    a = seq.copy()
    cur_score = _objective_norm(a, n, L)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # number of attempts is limited by remaining time
    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # pick source proportionally to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective_norm(a_prop, n, L)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    n = 65536                         # 2¹⁶

    # --------------------------------------------------------------
    best_seq = None

    # try the globally supplied starter (if present)
    try:

                init = init[:n]

            s = init.sum()
            if s > 0.0:

    # --------------------------------------------------------------
