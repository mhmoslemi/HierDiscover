def _hard_descent(a0, n, deadline, start_time, best_score):
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score
    # Adam hyperparams
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1=0.9; b2=0.999; eps=1e-8
    t=0
    L_conv = _next_pow2(2*n -1)
    L_corr = _next_pow2(3*n -2)
    while time.time() < deadline:
        t+=1
        # compute conv and active weights
        conv = _conv_self(a, L_conv)
        max_c = conv.max()
        # tolerance for active set
        tol = 1e-12 * max_c if max_c>0 else 1e-12
        mask = (conv >= max_c - tol).astype(float)
        w = mask / mask.sum()
        # correlation as gradient
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)
        start = n-1
        grad = 2.0 * corr[start:start+n]
        # evaluate exact
        cur_score = 2.0 * n * max_c / (a.sum()**2)  # sum a =1, but may deviate due to rounding.
        if cur_score < best_sc:
            best_sc = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hard-descent it={t} score={cur_score:.6f}")
        # Adam update
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*(grad**2)
        m_hat = m/(1-b1**t)
        v_hat = v/(1-b2**t)
        step = 0.05/(np.sqrt(v_hat)+eps)
        a = a - step*m_hat
        a = _project_simplex(a)
    return best_a, best_sc

def _annealed_hill_climb(a_start, deadline, start_time, L_fft):
    a = a_start.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]
    temperature = 1e-3  # initial relative scale
    while time.time() < deadline:
        # systematic schedule: temperature decays
        elapsed = time.time() - start_time
        # simple linear decay: temp = max(1e-6, init_temp * (1 - elapsed/(deadline-start_time)))
        temp = max(1e-6, 1e-3 * (1 - (elapsed/(deadline - start_time))))
        src = rng.integers(n)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # propose moving fraction epsilon of src mass to dst
        frac = rng.random() * 0.01  # up to 1%
        eps = a[src] * frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # project onto simplex - but since we preserved sum positivity, it's fine.
        # Do not need to reproject.
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - best_score

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] annealed climb score={best_score:.6f}")

"""
Hybrid search for a non‑negative step function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines three ideas that were missing from the previous
implementation:

1️⃣  A richer parametric sweep (geometric × power, pure power, pure
    exponential and discrete Gaussian families) to get a strong start.

2️⃣  A high‑temperature smooth‑max Adam descent where the temperature β
    is allowed to grow up to 5 000 (≈ “hard‑max”) and is followed by a
    true sub‑gradient descent that uses the exact active set of the
    maximal convolution coefficients.

3️⃣  An annealed hill‑climb that can accept occasional worsening moves.
    This helps escape the plateaus that often trap a pure greedy hill‑climb.

All stages respect the 1000 s budget, keep the best sequence found
throughout the run, and finally return a plain ``list[float]`` suitable
for ``evaluate_sequence``.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max Adam descent (soft‑max approximation of max)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_score_initial: float,
                    beta_max: float = 5000.0) -> tuple[np.ndarray, float]:
    """
    Adam‑type descent on the soft‑max surrogate.
    β starts small (very smooth) and is increased geometrically up to
    ``beta_max`` (≈ hard‑max).  The iterate is always projected onto the
    probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0
    beta_mul = 1.015

    while time.time() < deadline:
        t += 1
        # --- smooth gradient ------------------------------------------------
        conv = _conv_self(a, L=L_conv)
        max_c = conv.max()
        shifted = conv - max_c                 # stability shift
        exp_vals = np.exp(beta * shifted)      # softmax numerator
        w = exp_vals / exp_vals.sum()          # (2n‑1,) probabilities

        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        # --- exact score ----------------------------------------------------
        cur_score = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑descent it={t:,} β={beta:.2f} score={cur_score:.6f}")

        # --- Adam update on the simplex ------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.1 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional tiny random jitter (helps escape plateaus)
        if t % 4000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hard‑max (sub‑gradient) refinement
# ----------------------------------------------------------------------
def _hard_descent(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that uses the exact active set of the maximal
    convolution coefficient (the “hard‑max”).  Adam‑style updates are kept
    for robustness.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, L=L_conv)
        max_c = conv.max()

        # active set: all indices whose value is within a tiny tolerance
        tol = 1e-12 * max_c if max_c > 0 else 1e-12
        mask = (conv >= max_c - tol).astype(float)
        w = mask / mask.sum()  # (2n‑1,) weights, sum to 1

        # gradient = 2 * correlation(w, a_rev)
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        cur_score = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hard‑descent it={t:,} score={cur_score:.6f}")

        # Adam‐style projected update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.05 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def _annealed_hill_climb(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Random mass transfers between two coordinates.
    Moves that increase the objective are accepted with probability
    exp(−Δ/temperature).  The temperature decays linearly to zero.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    init_temp = 1e-3  # relative scale of accepted uphill moves

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linear cooling
        temp = max(1e-9, init_temp * (1.0 - elapsed / (deadline - start_time)))

        src = rng.integers(n)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer at most 1 % of the source mass
        eps = a[src] * rng.random() * 0.01
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - best_score
        if delta < 0 or rng.random() < math.exp(-delta / temp):
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] annealed‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: generate a candidate from a base shape using a random
# transformation (exponentiation, Gaussian smoothing, mixture, …)
# ----------------------------------------------------------------------
def _transform_shape(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a random nonlinear transform and re‑normalise to sum 1."""
    a = base.copy()

    # random power
    p = rng.uniform(0.3, 3.0)
    a = np.power(a, p)

    # optional Gaussian smoothing
    if rng.random() < 0.4:
        sigma = rng.uniform(5.0, 300.0)
        size = a.shape[0]
        x = np.arange(size)
        kernel = np.exp(-0.5 * ((x - size // 2) / sigma) ** 2)
        kernel /= kernel.sum()
        # convolution via FFT (circular, but we zero‑pad to keep shape)
        L = _next_pow2(2 * size - 1)
        A = np.fft.rfft(a, n=L)
        K = np.fft.rfft(kernel, n=L)
        a = np.fft.irfft(A * K, n=L)[:size]

    # mixture with uniform
    lam = rng.random()
    a = lam * (np.full_like(a, 1.0 / a.shape[0])) + (1.0 - lam) * a

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / a.shape[0])
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    evaluation function.  Up to 1000 s are used; the best sequence found
    (as a plain list of floats) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the evaluator
    deadline = start_time + budget_s - 0.3

    # problem size – we keep the size used in the reference solution
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise – uniform baseline
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optionally accept a global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Expanded parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-3.0, 2.5, 46)
    tau_grid = np.linspace(50.0, 2000.0, 12)  # for pure exponential
    sigma_grid = np.linspace(10.0, 300.0, 12)  # for Gaussian width

    # geometric × power family (already used, but we sweep finer)
    for r in r_grid:
        if time.time() >= deadline - 180:
            break
        geom = r ** np.arange(n, dtype=np.float64)
        for p in p_grid:
            if time.time() >= deadline - 180:
                break
            shape = geom * (idx ** p)
            s = shape.sum()
            if s == 0:
                continue
            shape /= s
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # pure power family (r = 1)
    for p in p_grid:
        if time.time() >= deadline - 180:
            break
        shape = idx ** p
        s = shape.sum()
        if s == 0:
            continue
        shape /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * uniform + (1.0 - lam) * shape
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] pure‑power p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # pure exponential family
    for tau in tau_grid:
        if time.time() >= deadline - 180:
            break
        shape = np.exp(-np.arange(n, dtype=np.float64) / tau)
        shape /= shape.sum()
        for lam in (0.0, 0.5, 1.0):
            seq = lam * uniform + (1.0 - lam) * shape
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] expo τ={tau:.1f} λ={lam:.2f} score={sc:.6f}")

    # discrete Gaussian family
    centre = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    for sigma in sigma_grid:
        if time.time() >= deadline - 180:
            break
        shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        shape /= shape.sum()
        for lam in (0.0, 0.5, 1.0):
            seq = lam * uniform + (1.0 - lam) * shape
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] gauss σ={sigma:.1f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam descent (harder temperature schedule)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:  # reserve time for later phases
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent (β up to 5000)")
        a_smooth, sc_smooth = _smooth_descent(best_seq, n,
                                               deadline - 60, start_time,
                                               best_score,
                                               beta_max=5000.0)
        if sc_smooth < best_score:
            best_score, best_seq = sc_smooth, a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑descent improvement score={sc_smooth:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Sub‑gradient (hard‑max) refinement – possibly several restarts
    # ------------------------------------------------------------------
    if time.time() < deadline - 60:
        print(f"[{time.time() - start_time:.2f}s] Starting hard‑max sub‑gradient refinements")
        # we may try a few brief restarts starting from perturbed versions
        n_restarts = 3
        for r in range(n_restarts):
            if time.time() >= deadline - 30:
                break
            # perturb best_seq a little before each restart
            pert = _transform_shape(best_seq, rng)
            a_hard, sc_hard = _hard_descent(pert, n,
                                            deadline - 30, start_time,
                                            best_score)
            if sc_hard < best_score:
                best_score, best_seq = sc_hard, a_hard.copy()
                print(f"[{time.time() - start_time:.2f}s] hard‑descent restart {r+1} improved score={sc_hard:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final annealed hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final annealed hill‑climb")
        a_final, sc_final = _annealed_hill_climb(best_seq, deadline,
                                                 start_time, L_fft)
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb improved score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
