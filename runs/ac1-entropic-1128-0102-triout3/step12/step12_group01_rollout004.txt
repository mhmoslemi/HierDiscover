import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self(a, L=None):
    n = a.shape[0]
    if L is None:
        L=_next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    return conv

def _objective(a, n):
    # assume a non-negative after clipping
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    conv = _conv_self(a)
    max_val = conv.max()
    return float(2.0 * n * max_val / (sum_a*sum_a))

"""
Improved search for a step‑function with a small 2·n·max(conv(a,a))/sum(a)².

The algorithm consists of three stages:

1️⃣  A richer parametric sweep (geometric, power, beta‑type,
    Gaussian and sinusoidal families) to obtain a decent starter.

2️⃣  An Adam‑style projected gradient descent on the probability simplex.
    The learning‑rate schedule and the perturbation on plateaus are
    a bit more aggressive than the reference implementation.

3️⃣  A simulated‑annealing / hill‑climb hybrid that repeatedly moves a
    random fraction of mass between two coordinates.  Worse moves are
    occasionally accepted with a temperature that decays to zero,
    allowing us to escape local minima and to improve the best
    objective found in the previous phases.

All operations respect the 1000 s budget, use FFT‑based convolutions
and keep the sequence normalised (sum ≈ 1).  The routine finally returns
a plain Python list suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (needed for Adam)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the current point.
    If several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size > 0:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Adam‑style optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score_initial: float,
                   n: int,
                   Lfft: int) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with projection onto the simplex.
    Returns the best sequence encountered and its (approximate) objective
    2·n·max(conv(a,a)) (the denominator sum(a)≈1 because of projection).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.4          # aggressive start
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        # learning‑rate decay (linear)
        lr = lr0 * (1.0 - it / 2_000_000.0)

        # current convolution / score
        conv = _conv_self(a, Lfft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # sum(a)≈1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # sub‑gradient
        grad = _subgradient_maxconv(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection back onto the simplex
        a = _project_simplex(a)

        # plateau handling – add a stronger Gaussian jitter
        if it - last_improve > 4000:
            sigma = 5e-4 * rng.standard_normal(n) * a
            a = np.clip(a + sigma, 0.0, None)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climb hybrid
# ----------------------------------------------------------------------
def _anneal_hillclimb(start_a: np.ndarray,
                      start_time: float,
                      deadline: float,
                      n: int,
                      Lfft: int) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that moves a random fraction of mass
    from one coordinate to another.  Worse moves are accepted with
    a temperature that linearly decays to zero.  The routine also
    behaves as a hill‑climb in the final stages.
    """
    a = start_a.copy()
    cur_score = float(2.0 * n * _conv_self(a, Lfft).max())
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    total_time = deadline - start_time
    # temperature schedule: start relatively high, ends at zero
    T0 = 0.008 * cur_score          # 0.8 % of the current score as initial T
    # max fraction of source mass that can be moved initially
    f0 = 0.12
    # minimal values when we approach the end of the annealing window
    f_min = 5e-4
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1
        progress = (time.time() - start_time) / total_time
        T = T0 * (1.0 - progress)                     # linear cooling
        max_frac = f0 * (1.0 - progress) + f_min      # shrink move size

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * max_frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # keep numerics safe
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, Lfft).max())

        # acceptance test
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            if T > 0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={iter_cnt:,} "
                      f"score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine respects the budget and returns a plain list of Python
    floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.4

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # uniform baseline
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = float(2.0 * n * _conv_self(best_seq, L_fft).max())
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # starter from a previously found construction (if available)
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s == 0:
                starter = uniform.copy()
            else:
                starter = starter / s
            cur = float(2.0 * n * _conv_self(starter, L_fft).max())
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Extended parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric family (very fine grid)
    eps_grid = np.logspace(-6, -0.5, 250)        # r = 1 - eps
    for eps in eps_grid:
        if time.time() >= deadline - 180:          # keep ~3 min for later phases
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo = geo / geo_sum
        # also try a few mixtures with uniform
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = seq / seq.sum()
            cur = float(2.0 * n * _conv_self(seq, L_fft).max())
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} "
                      f"score={cur:.6f}")

    # power family
    p_grid = np.linspace(-1.8, 2.8, 101)
    for p in p_grid:
        if time.time() >= deadline - 180:
            break
        pow_seq = np.power(idx + 1, p)
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq = pow_seq / pow_sum
        cur = float(2.0 * n * _conv_self(pow_seq, L_fft).max())
        if cur < best_score:
            best_score = cur
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={cur:.6f}")

    # beta‑type (polynomial) family – fractional exponents
    beta_vals = np.arange(0.0, 5.5 + 0.5, 0.5)   # 0,0.5,…,5.0
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 180:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            beta_sum = beta_seq.sum()
            if beta_sum == 0:
                continue
            beta_seq = beta_seq / beta_sum
            cur = float(2.0 * n * _conv_self(beta_seq, L_fft).max())
            if cur < best_score:
                best_score = cur
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} "
                      f"score={cur:.6f}")

    # Gaussian family (centered)
    sigma_factors = np.linspace(0.08, 0.30, 9)   # relative to n
    mu = (n - 1) / 2.0
    for sf in sigma_factors:
        if time.time() >= deadline - 180:
            break
        sigma = sf * n
        gauss = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        gauss_sum = gauss.sum()
        if gauss_sum == 0:
            continue
        gauss = gauss / gauss_sum
        cur = float(2.0 * n * _conv_self(gauss, L_fft).max())
        if cur < best_score:
            best_score = cur
            best_seq = gauss.copy()
            print(f"[{time.time() - start_time:.2f}s] Gaussian sigma={sigma:.0f} "
                  f"score={cur:.6f}")

    # Half‑sine family (positive bump)
    for p in (0.5, 1.0, 2.0, 4.0):
        if time.time() >= deadline - 180:
            break
        sine = np.sin(np.pi * (idx + 0.5) / n) ** p
        sine_sum = sine.sum()
        if sine_sum == 0:
            continue
        sine = sine / sine_sum
        cur = float(2.0 * n * _conv_self(sine, L_fft).max())
        if cur < best_score:
            best_score = cur
            best_seq = sine.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine p={p:.1f} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimiser (most of the remaining budget)
    # ------------------------------------------------------------------
    hill_time = 150.0                     # seconds reserved for final hill‑climb
    anneal_time = 200.0                   # seconds reserved for simulated annealing
    adam_deadline = deadline - (hill_time + anneal_time)
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        a_opt, sc_opt = _adam_optimize(best_seq, start_time,
                                       adam_deadline, best_score,
                                       n, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated annealing / hill‑climb hybrid
    # ------------------------------------------------------------------
    if time.time() < deadline - hill_time:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing / hill‑climb")
        a_hc, sc_hc = _anneal_hillclimb(best_seq, start_time,
                                         deadline - hill_time,
                                         n, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Annealing best score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final short hill‑climb (pure improvements)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final pure hill‑climb")
        # reuse the same routine but with temperature set to zero
        a_final, sc_final = _anneal_hillclimb(best_seq, start_time, deadline,
                                              n, L_fft)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list (sum normalised to 1)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
