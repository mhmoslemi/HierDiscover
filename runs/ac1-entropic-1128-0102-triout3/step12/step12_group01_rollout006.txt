    # compute conv (linear conv of a with itself) length 2n-1
    L = _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    # compute softmax probabilities
    # shift for numerical stability
    max_c = conv.max()
    # scaling parameter = alpha
    exp_scaled = np.exp(alpha * (conv - max_c))
    sum_exp = exp_scaled.sum()
    p = exp_scaled / sum_exp
    # compute loss (approx objective)
    loss = 2 * n * (max_c + np.log(sum_exp) / alpha)  # Actually max_c + log sum exp / alpha - but we used shift subtract max_c; need reconstruct.
    # Below: log(sum_i exp(alpha * conv_i)) = alpha * max_c + log(sum_exp)
    # So loss = 2n * (logsumexp(alpha*conv)/alpha) = 2n * (max_c + log(sum_exp)/alpha).
    loss = 2.0 * n * (max_c + np.log(sum_exp) / alpha)
    # compute gradient:
    # p already normalized probabilities (softmax of alpha*conv)
    # compute conv_rev = np.convolve(p, a[::-1]), taking first n entries.
    # Use FFT for speed.
    a_rev = a[::-1]
    L_grad = _next_pow2(len(p) + n - 1)   # len(p)==2n-1
    # using rfft for real signals
    P_f = np.fft.rfft(p, n=L_grad)
    A_rev_f = np.fft.rfft(a_rev, n=L_grad)
    conv_rev = np.fft.irfft(P_f * A_rev_f, n=L_grad)[: n + len(p) - 1]  # length 3n-2
    # slice first n entries reversed
    g = conv_rev[:n][::-1]  # shape (n,)
    grad = 4.0 * n * g
    return loss, grad
```

def _adam_subgrad_opt(a0, start_time, deadline, best_score):
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2*n-1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.08
    it = 0
    last_improve_it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] Adam step={it} score={cur_score:.6f}")
        # compute subgradient
        grad = _subgradient_maxconv(a, conv)  # gradient = subgradient of max conv term (value)
        # Note: objective = 2n * max(conv). So gradient of objective w.r.t a is 2n * grad (since grad from max conv). Indeed _subgradient returns gradient of max(conv) w.r.t a (the subgradient) (should be appropriate). We need to multiply by 2n.
        # But _subgradient returns gradient of max(conv) (maybe included factor 2?). Let's inspect original: It computed grad += a[k-idx]; then after loop grad = 2.0 * grad / active.size. Indeed, they added a[k - idx] computed for each active shift, then multiplied by 2. So gradient approximates derivative of conv? For each shift they added a[k - idx], which corresponds to gradient contributions of each term (since derivative of conv_i w.r.t a_j is a_{i-j}, and they are summing these). They then multiply by 2. So final grad is approximate subgradient of max(conv). Since objective is 2 * n * max(conv), gradient w.r.t a = 2 * n * subgrad. Thus we need to multiply grad by 2 * n.
        grad_obj = grad * (2.0 * n)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj * grad_obj)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_simplex(a)
        # plateau handling
        if it - last_improve_it > 8000:
            # reduce learning rate and add noise

def _hill_climb_sa(a_start, start_time, deadline, best_score):
    a = a_start.copy()
    best_a = a.copy()
    best_score = best_score
    n = len(a)
    L_fft = _next_pow2(2*n-1)
    rng = np.random.default_rng()
    # parameters
    init_temp = 0.001  # small since objective differences are small
    while time.time() < deadline:
        remaining = deadline - time.time()
        temp = init_temp * (remaining / (deadline - start_time))
        # propose move
        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1
        # choose fraction from src
        frac = rng.random() * 0.2 + 1e-6
        eps = a[src] * frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # ensure nonnegativity
        if a_prop[src] < -1e-12:
            continue

"""
Improved search for a non‑negative step function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines:

1️⃣  An extensive parametric sweep over several families (geometric,
    power, beta, Gaussian, exponential tails, cosine windows).

2️⃣  A long projected Adam optimiser that works directly with the
    sub‑gradient of the max‑convolution (exact, not a smooth surrogate).
    Learning‑rate is halved whenever progress stalls, with a tiny Gaussian
    perturbation to escape plateaus.

3️⃣  A final stochastic hill‑climb with simulated‑annealing acceptance.
    Random mass transfers of up to a few percent are tried; worsening
    moves are occasionally accepted with a temperature that cools to zero.

All steps respect the global 1000 s budget, use FFT‑based convolutions,
and maintain the sequence on the probability simplex (non‑negative,
unit sum).  The routine returns a plain Python list ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # small negative rounding errors are harmless for the max
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the current point.
    If several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size > 0:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Scoring utilities (fast, assuming sum(a)=1)
# ----------------------------------------------------------------------
def _score_a(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) – the denominator sum(a)² is 1."""
    n = a.shape[0]
    conv = _conv_self(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (exact sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser using the exact sub‑gradient of max‑convolution.
    Returns the best sequence seen and its true score.
    """
    a = a0.copy()
    best_a = a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                     # start a bit more gentle than before
    plateau_iters = 4000
    rng = np.random.default_rng()

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        # current score
        cur = _score_a(a, L_fft)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur:.6f}")

        # sub‑gradient of max‑convolution; objective = 2·n·max(conv)
        conv = _conv_self(a, L_fft)
        subgrad = _subgradient_maxconv(a, conv)       # ∂ max(conv) / ∂ a
        grad = subgrad * (2.0 * n)                    # ∂ objective / ∂ a

        # Adam update on the (unconstrained) vector a
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection back onto the simplex
        a = _project_simplex(a)

        # plateau handling – shrink learning rate & shake the iterate
        if it - last_improve > plateau_iters:
            lr = max(lr * 0.5, 1e-4)
            noise = 1e-3 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb with a tiny simulated‑annealing scheme
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float,
                best_score: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb.  Randomly transfers a small fraction of mass
    between positions and accepts improvements; occasional worsening
    moves are accepted with a temperature that cools to zero.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    rng = np.random.default_rng()

    init_temp = 2e-4            # tiny temperature – only rarely accept worsening moves
    while time.time() < deadline:
        # linear cooling schedule
        remaining = deadline - time.time()
        total = deadline - start_time
        temp = init_temp * (remaining / total)

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move a random fraction (up to 5 %) of src's mass
        frac = rng.random() * 0.05 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur = _score_a(a_prop, L_fft)

        if cur < best_score:
            a = a_prop
            best_a = a_prop.copy()
            best_score = cur
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb new best={cur:.6f}")
        else:
            # occasional acceptance of worse moves (simulated annealing)
            if temp > 0 and rng.random() < math.exp(-(cur - best_score) / temp):
                a = a_prop

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step function (list of non‑negative floats) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2).  The routine respects the supplied
    time budget and returns a plain Python list suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # problem size – 4096 provides a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _score_a(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # use a global starter if one exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                cur = _score_a(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Extensive parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # ---- geometric family -------------------------------------------------
    eps_grid = np.logspace(-5, -1, 200)        # r = 1 - eps
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo = geo / geo_sum
        # mix with uniform – 11 mixing ratios
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            cur = _score_a(seq, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={cur:.6f}")

    # ---- power family ----------------------------------------------------
    p_grid = np.linspace(-1.5, 2.5, 81)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p)
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq = pow_seq / pow_sum
        cur = _score_a(pow_seq, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={cur:.6f}")

    # ---- beta‑type family -------------------------------------------------
    beta_vals = [0.0, 0.5, 1.0, 1.5, 2.0, 3.0]
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            beta_sum = beta_seq.sum()
            if beta_sum == 0:
                continue
            beta_seq = beta_seq / beta_sum
            cur = _score_a(beta_seq, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p} q={q} score={cur:.6f}")

    # ---- Gaussian bump ----------------------------------------------------
    center = (n - 1) / 2.0
    for sigma_fac in np.linspace(0.05, 0.5, 12):
        sigma = sigma_fac * n
        gauss = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
        gauss_sum = gauss.sum()
        if gauss_sum == 0:
            continue
        gauss = gauss / gauss_sum
        cur = _score_a(gauss, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = gauss.copy()
            print(f"[{time.time() - start_time:.2f}s] Gaussian sigma={sigma_fac:.2f} score={cur:.6f}")

    # ---- double‑exponential tails ----------------------------------------
    for scale_fac in np.linspace(0.05, 0.5, 12):
        scale = scale_fac * n
        exp_tail = np.exp(-idx / scale) + np.exp(-(n - 1 - idx) / scale)
        exp_sum = exp_tail.sum()
        if exp_sum == 0:
            continue
        exp_tail = exp_tail / exp_sum
        cur = _score_a(exp_tail, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = exp_tail.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpTail scale={scale_fac:.2f} score={cur:.6f}")

    # ---- cosine window (Hann‑like) ----------------------------------------
    for width_fac in np.linspace(0.1, 0.9, 9):
        width = int(width_fac * n)
        if width < 2:
            continue
        win = np.zeros(n)
        start = int(center) - width // 2
        for i in range(width):
            pos = start + i
            if 0 <= pos < n:
                win[pos] = 0.5 * (1 + np.cos(np.pi * (i - width / 2) / (width / 2)))
        win_sum = win.sum()
        if win_sum == 0:
            continue
        win = win / win_sum
        cur = _score_a(win, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = win.copy()
            print(f"[{time.time() - start_time:.2f}s] Cosine width={width_fac:.2f} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Projected Adam optimisation (major part of the budget)
    # ------------------------------------------------------------------
    hill_time = 150.0                       # seconds reserved for the final hill‑climb
    adam_deadline = deadline - hill_time
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        a_opt, sc_opt = _adam_optimize(best_seq, start_time, adam_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline, best_score)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
