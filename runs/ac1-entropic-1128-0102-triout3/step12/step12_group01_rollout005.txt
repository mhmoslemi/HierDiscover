"""
Propose a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

by:

1️⃣  Parametrising the vector on the simplex with a soft‑max (log‑variables).
2️⃣  Optimising a *smooth* approximation of the max‑convolution using
    Adam on the log‑variables.  The smooth max is implemented with a
    temperature‑controlled log‑sum‑exp; the temperature is increased
    linearly during the run (coarse → accurate).
3️⃣  A final stochastic hill‑climb that moves a small random amount of
    mass between two coordinates and keeps only improving moves.

All operations use FFT‑based convolutions (O(n log n)).  The routine
respects the supplied time budget (default 1000 s) and returns a plain
list of Python floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_j a_j * w_{i+j}   (i = 0 … n‑1).
    This is the (non‑circular) cross‑correlation of w with a.
    Implemented via FFT: g = conv(w, a_rev)[n‑1 : n‑1+n].
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(len(w) + n - 1)
    Wf = np.fft.rfft(w, n=L)
    Af = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(Wf * Af, n=L)[: len(w) + n - 1]
    # slice n‑1 … n‑1+n‑1  → length n
    return conv[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Smoothed max‑convolution optimisation (soft‑max + Adam)
# ----------------------------------------------------------------------
def _adam_softmax_opt(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_score: float,
    best_seq: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex using a soft‑max parametrisation.
    Returns the best sequence (sum≈1) and its true objective score.
    """
    n = a0.shape[0]

    # initialise log‑variables (soft‑max input)
    x = np.log(np.maximum(a0, 1e-20))

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.4                                   # relatively aggressive

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    # smoothing schedule
    beta_start = 1.0
    beta_end = 80.0

    while time.time() < deadline:
        it += 1
        # --------------------------------------------------------------
        # a = softmax(x)
        # --------------------------------------------------------------
        mx = np.max(x)
        ex = np.exp(x - mx)
        a = ex / ex.sum()                       # ∑a = 1, a ≥ 0

        # --------------------------------------------------------------
        # convolution of a with itself
        # --------------------------------------------------------------
        L_conv = _next_pow2(2 * n - 1)
        conv = _conv_full(a, L_conv)            # length 2n‑1
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)    # true objective (sum≈1)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = it
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                f"score={cur_score:.6f}"
            )

        # --------------------------------------------------------------
        # smooth max (log‑sum‑exp) and its gradient w.r.t. a
        # --------------------------------------------------------------
        progress = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + (beta_end - beta_start) * progress

        # avoid overflow by subtracting the maximum value before exp
        conv_shift = conv - conv.max()
        w_raw = np.exp(beta * conv_shift)       # w_raw ≥ 0
        w = w_raw / w_raw.sum()                 # probabilities, Σw = 1

        # gradient of the smoothed max w.r.t. a (g_i = 2 Σ_j a_j w_{i+j})
        g_a = 2.0 * _cross_corr(w, a)           # length n

        # --------------------------------------------------------------
        # back‑prop to log‑variables (soft‑max chain rule)
        # --------------------------------------------------------------
        dot_ag = np.dot(a, g_a)                 # scalar ⟨a,g⟩
        g_x = a * (g_a - dot_ag)                # ∂f/∂x

        # --------------------------------------------------------------
        # Adam update on x (gradient descent)
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # --------------------------------------------------------------
        # plateau handling – occasional jitter
        # --------------------------------------------------------------
        if it - last_improve > 8000:
            # add a tiny Gaussian perturbation to the log‑variables
            jitter = 1e-4 * rng.standard_normal(n)
            x = x + jitter
            last_improve = it

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(
    a_start: np.ndarray, start_time: float, deadline: float, best_score: float
) -> tuple[np.ndarray, float]:
    """
    Random local search that transfers a small random fraction of mass
    from one index to another, keeping only improving moves.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    rng = np.random.default_rng()

    # pre‑compute convolution length
    L_full = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # random fraction (up to a few per cent) of the source mass
        frac = rng.random() * 0.03 + 1e-6          # ≤ 3 %
        delta = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        # tiny numeric safety
        if a_prop[src] < -1e-15:
            continue
        a_prop = np.maximum(a_prop, 0.0)
        a_prop /= a_prop.sum()                    # re‑normalise to sum=1

        conv_prop = _conv_full(a_prop, L_full)
        cur_score = float(2.0 * n * conv_prop.max())

        if cur_score < best_score:
            a = a_prop
            best_a = a_prop.copy()
            best_score = cur_score
            print(
                f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}"
            )

    return best_a, best_score
