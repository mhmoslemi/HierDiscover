"""
Improved step‑function search.

The algorithm consists of three phases:

1️⃣  **Warm‑starts** – a set of analytically‑motivated shapes (uniform,
    triangular, power‑law, exponential, Gaussian, arcsine) together with a
    handful of random Dirichlet samples.  The best of these are taken as
    seeds.

2️⃣  **Smoothed exponential‑gradient descent** – we minimise the exact
    objective 2·n·max(conv(a,a)) (with ∑a=1) by replacing the hard max with
    a temperature‑controlled soft‑max (log‑sum‑exp).  The gradient of the
    smooth objective is obtained analytically (a convolution of the weight
    vector with the current a).  An exponentiated‑gradient (mirror descent)
    update on the probability simplex is performed with a diminishing step
    size and a geometric schedule for the soft‑max temperature.  Several
    restarts are carried out while time permits.

3️⃣  **Heavy simulated‑annealing polish** – a cheap Metropolis walk that
    transfers a random packet of mass between two coordinates for the
    remaining time budget.

All helper functions are top‑level and stateless so that the harness can
pickle them.  The final sequence is normalised to sum 1 (the objective is
scale‑invariant) and returned as a plain Python list of non‑negative
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length a power of two.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Equivalent to ``evaluate_sequence`` (without defensive checks).
    The sequence is assumed to be non‑negative; the sum is forced to be >0.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Warm‑start shapes
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # in [0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # in [0,1]
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _collect_warm_starts(n: int) -> list[np.ndarray]:
    """Build a modest library of diverse initial vectors."""
    cand = []

    # deterministic families
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # blend of centre/edge
    centre = _shape_power_center(n, 2.0)
    edge = _shape_power_edge(n, 2.0)
    for lc in (0.25, 0.5, 0.75):
        a = lc * centre + (1 - lc) * edge
        a = a / a.sum()
        cand.append(a)

    # possible global starter supplied by the harness
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # remove NaNs just in case
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max (i.e. ∑_k p_k conv_k) w.r.t. the probability
    vector a.  Formula:   g_i = 2 * Σ_j a_j * p_{i+j}
    implemented via FFT‑based convolution.
    """
    n = a.shape[0]
    # reverse p because convolution with reversed p implements the required sum
    p_rev = p[::-1]
    L = _next_pow2(len(p_rev) + n - 1)
    A = np.fft.rfft(a, n=L)
    P = np.fft.rfft(p_rev, n=L)
    conv_full = np.fft.irfft(A * P, n=L)
    offset = len(p_rev) - 1          # = 2*n-2
    grad = 2.0 * conv_full[offset: offset + n]
    # numerical noise may produce a tiny imaginary part
    if np.iscomplexobj(grad):
        grad = np.real(grad)
    return grad


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent (single run)
# ----------------------------------------------------------------------
def _optimize_from(start_a: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iter: int = 3000,
                   eta0: float = 0.4,
                   alpha0: float = 2.0,
                   alpha_final: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a soft‑max approximation of the max.
    Returns the best sequence found and its exact objective.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _objective(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current self‑convolution
        conv = _conv_self(a)

        # geometric schedule for the temperature (soft‑max sharpness)
        alpha = alpha0 * (alpha_final / alpha0) ** (it / max_iter)

        # stable soft‑max weights
        maxc = conv.max()
        scaled = np.exp(alpha * (conv - maxc))
        Z = scaled.sum()
        p = scaled / Z                     # shape (2n‑1,)

        # gradient of the smooth objective (2·n·smooth_max)
        grad_s = _grad_smooth_max(a, p)    # ∂ smooth_max / ∂a
        grad_obj = 2.0 * n * grad_s        # ∂ objective / ∂a

        # remove the component along the all‑ones direction (does not affect
        # the simplex after the multiplicative update)
        grad_adj = grad_obj - grad_obj.mean()

        # diminishing learning rate
        step = eta0 / math.sqrt(it)

        # multiplicative (exponential) update on the simplex
        exp_arg = -step * grad_adj
        # clamp to avoid overflow/underflow in the exponential
        exp_arg = np.clip(exp_arg, -20.0, 20.0)
        a = a * np.exp(exp_arg)

        # renormalise to stay on the probability simplex
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)

        # occasional jitter to escape flat plateaux
        if it % 500 == 0:
            jitter = (np.random.rand(n) - 0.5) * 2e-3
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Iter {it:,}  score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (large number of cheap moves)
# ----------------------------------------------------------------------
def _anneal_many(seq: np.ndarray,
                 deadline: float,
                 start_time: float,
                 steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Random mass transfer walk with a simple Metropolis acceptance rule.
    Designed to consume the last few seconds of the time budget.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # linear temperature decay
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = max(0.001, 0.2 * (1 - it / steps))

        src = rng.choice(n, p=a)          # source proportional to current mass
        dst = rng.integers(n)              # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        accept = delta < 0.0 or math.exp(-delta / T) > rng.random()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal {it:,}  best={new_score:.6f}")

    return best_a, best_score

    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Best final score: {best_score:.6f}")
    return [float(v) for v in best_seq]
