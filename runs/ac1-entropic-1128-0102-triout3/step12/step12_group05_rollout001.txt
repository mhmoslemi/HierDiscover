"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

using a three‑phase scheme

1️⃣  Warm‑starts – a small library of analytically‑motivated shapes
    (uniform, triangular, power‑law, exponential, Gaussian, arcsine).
    The best of them (according to the exact objective) seeds the optimisation.

2️⃣  Smooth‑max Adam – we re‑parameterise the vector as a soft‑max of free
    variables *b* (ensuring a≥0, Σa=1).  The non‑smooth max is replaced by a
    log‑sum‑exp with a slowly increasing temperature *α*; the resulting
    differentiable surrogate is minimised by the Adam optimiser (mirrored‑gradient
    on the simplex).  Convolutions and the gradient are evaluated with FFTs
    (O(n log n)) so that many thousands of iterations fit comfortably into the
    1000 s budget.

3️⃣  Light simulated‑annealing polish – a cheap Metropolis walk that moves a
    random packet of mass between two indices, run for the final few seconds.

The implementation is self‑contained, uses only ``numpy`` and the provided
``evaluate_sequence`` for final validation, and respects the time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


def _gradient_wrt_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Given weights w (length 2·n‑1, sum to 1) and a∈ℝⁿ,
    compute ∂/∂a  Σ_k w_k (a*a)_k = 2·(w * a_rev) taken on the central n entries.
    """
    n = a.shape[0]
    len_w = w.shape[0]                         # = 2·n‑1
    L = _next_pow2(len_w + n - 1)               # size for FFT convolution
    # reverse a for the convolution
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Af = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(Wf * Af, n=L)[: len_w + n - 1]
    # central slice corresponds to indices n‑1 … n‑1+n‑1
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Warm‑start shapes (identical to the previous baseline)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    if a.sum() == 0:
        return _shape_uniform(n)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)          # ∈[0,1]
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _collect_warm_starts(n: int) -> list[np.ndarray]:
    """Return a modest library of diverse initial vectors."""
    cand = []

    # Basic shapes
    cand.append(_shape_uniform(n))
    cand.append(_shape_tent(n))
    cand.append(_shape_arcsine_like(n))

    # Power‑law families
    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(_shape_power_center(n, p))
        cand.append(_shape_power_edge(n, p))

    # Exponential decays
    for gamma in (0.0005, 0.001, 0.0025, 0.005):
        cand.append(_shape_exponential(n, gamma))

    # Gaussians
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        cand.append(_shape_gaussian(n, sigma))

    # Blends between centre‑heavy and edge‑heavy
    centre = _shape_power_center(n, 2.0)
    edge = _shape_power_edge(n, 2.0)
    for lc in (0.25, 0.5, 0.75):
        a = lc * centre + (1 - lc) * edge
        cand.append(a / a.sum())

    # Optional external starter
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cand.append(starter)
    except Exception:
        pass

    # Remove NaNs (defensive)
    uniq = []
    for a in cand:
        if np.any(np.isnan(a)):
            continue
        uniq.append(a)
    return uniq


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     alpha_start: float = 5.0,
                     alpha_end: float = 500.0,
                     lr: float = 0.08,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     eval_every: int = 100) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max parametrisation of ``a``.
    The non‑smooth ``max`` is replaced by a log‑sum‑exp with a temperature α
    that is increased linearly from ``alpha_start`` to ``alpha_end``.
    Returns the best vector encountered and its exact objective value.
    """
    n = a0.shape[0]
    # initialise free parameters b = log(a) (add tiny epsilon to avoid log(0))
    b = np.log(a0 + 1e-12)

    m = np.zeros_like(b)
    v = np.zeros_like(b)

    best_a = a0.copy()
    best_score = _objective(best_a)

    # schedule for α – linear increase
    total_iters_est = 200_000  # generous upper bound
    alpha_inc = (alpha_end - alpha_start) / total_iters_est

    it = 0
    while time.time() < deadline:
        it += 1

        # soft‑max to obtain a (numerically stable)
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        # self‑convolution
        conv = _conv_self(a)

        # current smoothing parameter
        alpha = min(alpha_end, alpha_start + it * alpha_inc)

        # log‑sum‑exp (smooth max)
        c_max = conv.max()
        exp_shifted = np.exp(alpha * (conv - c_max))
        sum_exp = exp_shifted.sum()
        M = c_max + np.log(sum_exp) / alpha

        # weights w = ∂M/∂conv (they form a probability vector)
        w = exp_shifted / sum_exp

        # gradient of the smooth objective w.r.t. a
        grad_a = _gradient_wrt_a(w, a)

        # gradient w.r.t. b (soft‑max Jacobian)
        dot = np.dot(grad_a, a)
        grad_b = a * (grad_a - dot)

        # Adam update with bias correction
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * (grad_b ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        b = b - step

        # occasional exact evaluation
        if it % eval_every == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                      f"score={cur_score:.6f}")

    return best_a, best_score

# Light simulated‑annealing polish (same as in the baseline)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk that swaps a random packet of mass between two positions."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # source proportional to mass
        dst = rng.integers(n)              # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up‑to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid stepping over the hard deadline
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # 4096 is a good compromise between expressiveness and speed.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑starts
    # ------------------------------------------------------------------
    candidates = _collect_warm_starts(n)

    # also add a few random Dirichlet starts to increase diversity
    for _ in range(5):
        candidates.append(np.random.dirichlet(np.ones(n)))

    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(candidates, 1):
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] Warm‑start {idx}/{len(candidates)} "
              f"score={score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimisation with smooth‑max surrogate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:               # keep a few seconds for polishing
        adam_deadline = time.time() + remaining * 0.80
        refined_seq, refined_score = _adam_smooth_max(
            best_seq,
            adam_deadline,
            start_time,
            alpha_start=5.0,
            alpha_end=500.0,
            lr=0.08,
            eval_every=150   # less frequent exact evaluations → faster
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Light simulated‑annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Normalise & return as plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq.tolist()]
