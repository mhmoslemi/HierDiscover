def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # set lengths
    N_target = 65536
    N_small = 4096
    # piecewise linear shape using K knots
    K = 30

    # DE objective function
    def de_obj(params):
        # params length K
        heights = np.exp(params)  # ensure positive
        # normalize to sum 1 after interpolation
        # interpolate to N_small
        x_knots = np.linspace(0, 1, K)
        x = np.linspace(0, 1, N_small)
        seq = np.interp(x, x_knots, heights)
        # Ensure nonnegative
        seq = np.maximum(seq, 0)
        total = seq.sum()
        if total <= 0:
            return np.inf
        seq = seq / total
        # objective
        obj = _objective(seq)
        return obj

    # bounds for params: let's set -10 to 10
    bounds = [(-10, 10)] * K

    # run DE
    result = differential_evolution(de_obj, bounds,
                                    maxiter=30,
                                    popsize=15,
                                    tol=1e-6,
                                    seed=seed,
                                    polish=False,
                                    updating='deferred',
                                    disp=False,
                                    workers=1)
    best_small_seq = np.interp(np.linspace(0,1,N_small), np.linspace(0,1,len(result.x)), np.exp(result.x))

"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The method:

1. **Low‑dimensional parametric search** – a piecewise linear shape with a
   modest number of knots (K≈30) is optimised on a small problem size
   (N≈4096) by SciPy's differential evolution.  The knot heights are forced
   to be positive (exponential), the shape is interpolated, normalised to
   a unit L1‑norm and the exact objective is evaluated via an FFT‑based
   convolution.

2. **Upsampling** – the best small‑scale shape is interpolated to the target
   length (65536).  The upsampled vector is again normalised.

3. **Local refinement** – a cheap sub‑gradient descent (the same scheme used
   in the reference implementation) is run for the remaining time budget
   to polish the candidate.

All steps respect the 1000 s budget.  The returned list consists of
non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Helper: next power of two (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT (fast and exact)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑normalised to 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent (same as reference, but usable as a final polish)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient descent on the exact objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑conv term
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score
