def adam_step(a, grad, m, v, t, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1*m + (1-beta1)*grad
    v = beta2*v + (1-beta2)*(grad**2)
    m_hat = m/(1-beta1**t)
    v_hat = v/(1-beta2**t)
    a_new = a * np.exp(-lr * m_hat / (np.sqrt(v_hat)+eps))
    a_new = np.maximum(a_new, 1e-30)
    a_new /= a_new.sum()
    return a_new, m, v

def _smooth_descent(a, deadline, start_time):
    best_a = a.copy()
    best_score = _exact_score(best_a)

    alphas = np.geomspace(0.05, 5000.0, num=80)

    it = 0
    for alpha in alphas:
        if time.time() >= deadline - 0.2:
            break
        while time.time() < deadline - 0.2:
            it += 1
            # compute conv and w
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[:2*n-1]

            max_exp = (alpha * conv).max()
            w = np.exp(alpha * conv - max_exp)
            w_sum = w.sum()
            if w_sum == 0:
                w.fill(1.0 / w.shape[0])
            else:
                w /= w_sum

            # gradient
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_GRAD)
            Revf = np.fft.rfft(a_rev, n=_L_GRAD)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[:3*n-2]
            grad = 2.0 * grad_full[n-1:n-1+n]

            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-30)
            a /= a.sum()

            if it % 5000 == 0:

                if cur_score < best_score:

                    best_a = a.copy()

def _hill_climb(a, deadline, start_time):

    cur_score = _exact_score(cur)
    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0

    end_temp = 1e-9

    while time.time() < deadline:
        it += 1
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if cur[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a random fraction up to 80% of source mass
        eps = cur[src] * rng.random() * 0.80
        if eps <= 0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        # ensure nonnegative, renormalize
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill-climb it={it:,} best={best_score:.6f}")

    return best_seq, best_score

"""
Multi‑restart exponentiated‑gradient optimizer for the step‑function objective

The objective to minimise is
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works entirely on the probability simplex (∑a = 1) because the
objective is scale‑invariant.  For each start vector it runs three phases:

1.  Smooth‑max mirror descent – a softmax approximation of the max is used.
    The smoothing parameter α is increased from 0.05 to 5000 on a geometric
    schedule while an exponentiated‑gradient update (mirrored on the simplex)
    is performed.

2.  Direct max‑subgradient descent – the exact index attaining the current
    maximum of the convolution is identified and a sub‑gradient step (again
    exponentiated‑gradient) is taken.

3.  Stochastic hill‑climbing (simulated annealing) – random mass transfers
    between two positions are accepted with a Metropolis rule.  The temperature
    schedule starts higher (0.2) than in the reference code, allowing larger
    uphill moves.

The whole optimisation is run several times from different initialisations:
uniform, the previous best sequence (if available), its reversed version,
a random permutation of it and a fresh Dirichlet‑sampled vector.  The best
result obtained before the time budget expires is returned as a plain python
list.  All heavy work uses FFTs; the code respects the 1000 s budget and works
with a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global problem size (can be changed, but 4096 works well within the budget)
# ----------------------------------------------------------------------
n = 4096
_L_SELF = 1 << ((2 * n - 2).bit_length())          # power of two ≥ 2n‑1
_L_GRAD = 1 << ((3 * n - 3).bit_length())          # power of two ≥ 3n‑2


# ----------------------------------------------------------------------
# Fast objective (uses FFT, works for any non‑negative a with sum≈1)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2).  a is assumed non‑negative."""
    Af = np.fft.rfft(a, n=_L_SELF)
    conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max mirror descent
# ----------------------------------------------------------------------
def _smooth_descent(a: np.ndarray, deadline: float, start_time: float):
    """
    Exponentiated‑gradient on the log‑sum‑exp approximation of max.
    The smoothing parameter α follows a geometric schedule up to 5 000.
    """
    best_a = a.copy()
    best_score = _exact_score(best_a)

    # geometric schedule: start very smooth, finish very sharp
    alphas = np.geomspace(0.05, 5_000.0, num=80)

    it = 0
    for alpha in alphas:
        if time.time() >= deadline - 0.2:
            break
        while time.time() < deadline - 0.2:
            it += 1

            # self‑convolution a*a
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]

            # smooth‑max weights w = softmax(α·conv)
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.shape[0])
            else:
                w /= w_sum

            # gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_GRAD)
            Revf = np.fft.rfft(a_rev, n=_L_GRAD)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # exponentiated‑gradient step (step size mildly decreasing with α)
            eta = 0.20 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-30)          # avoid exact zeros
            a /= a.sum()

            # occasional exact evaluation
            if it % 5000 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] smooth α={alpha:.1f} "
                          f"it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – direct max‑subgradient (exponentiated) descent
# ----------------------------------------------------------------------
def _max_subgradient_descent(a: np.ndarray, deadline: float, start_time: float):
    """Push mass away from the index that currently attains the maximum."""
    best_a = a.copy()
    best_score = _exact_score(best_a)

    it = 0
    while time.time() < deadline - 0.2:
        it += 1

        # current self‑convolution and its maximum index
        Af = np.fft.rfft(a, n=_L_SELF)
        conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
        k = int(conv.argmax())

        # sub‑gradient: only entries that contribute to (a*a)[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # g_i = 2·a_{k‑i}

        # larger learning rate than the reference implementation
        eta = 0.70 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        # periodic exact evaluation
        if it % 2000 == 0:
            cur_score = _exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} "
                      f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – stochastic hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, start_time: float):
    """Small random mass transfers with a Metropolis acceptance rule."""
    cur = a.copy()
    cur_score = _exact_score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    start_temp = 2e-1            # higher start temperature for broader exploration
    end_temp = 1e-9

    while time.time() < deadline:
        it += 1

        # linear progress for temperature decay
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if cur[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 80 % of the source mass
        eps = cur[src] * rng.random() * 0.80
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} "
                      f"best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper that runs the three phases within a given time window
# ----------------------------------------------------------------------
def _run_optimisation(start_a: np.ndarray, deadline: float, global_start: float):
    """Run the three optimisation phases on a copy of start_a."""
    a = start_a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= s

    now = time.time()
    # Phase 1 – 50 % of the remaining time
    phase1_deadline = now + 0.50 * (deadline - now)
    a, _ = _smooth_descent(a, phase1_deadline, global_start)

    # Phase 2 – 30 % of the remaining time
    now = time.time()
    phase2_deadline = now + 0.30 * (deadline - now)
    a, _ = _max_subgradient_descent(a, phase2_deadline, global_start)

    # Phase 3 – use whatever time is left for hill climbing
    if time.time() < deadline:
        a, _ = _hill_climb(a, deadline, global_start)

    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.

    The optimiser tries several diverse initialisations (uniform, the last
    known best, its reverse, a random permutation, and a fresh Dirichlet sample)
    and keeps the best result found before the budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # Build a list of initial sequences to try
    # ------------------------------------------------------------------
    initial_seqs = []

    # 1. uniform simplex
    uniform = np.full(n, 1.0 / n)
    initial_seqs.append(uniform)

    # 2. previous best if the global variable exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.size != n:
                if prev.size > n:
                    prev = prev[:n]
                else:
                    prev = np.pad(prev, (0, n - prev.size), mode='constant')
            prev /= prev.sum()
            initial_seqs.append(prev)                # as‑is
            initial_seqs.append(prev[::-1])          # reversed order
            rng = np.random.default_rng()
            perm = rng.permutation(n)
            initial_seqs.append(prev[perm])          # random permutation
    except Exception:
        pass

    # 3. a random Dirichlet point (moderately sparse)
    rng = np.random.default_rng()
    dirichlet = rng.dirichlet(np.full(n, 0.3))
    initial_seqs.append(dirichlet)

    # ------------------------------------------------------------------
    # Run optimisation from each start (time sharing)
    # ------------------------------------------------------------------
    best_seq = uniform.copy()
    best_score = _exact_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Starter uniform score = {best_score:.6f}")

    remaining_starts = len(initial_seqs)
    for start_vec in initial_seqs:
        if time.time() >= deadline - 5.0:
            break
        # allocate an equal share of the remaining budget to this start
        now = time.time()
        time_left = deadline - now
        # keep a small reserve for the next starts
        alloc = time_left / (remaining_starts + 1)
        end_deadline = now + alloc
        candidate = _run_optimisation(start_vec, end_deadline, start_time)
        cand_score = _exact_score(candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            print(f"[{time.time() - start_time:.2f}s] New overall best = {best_score:.6f}")
        remaining_starts -= 1

    # ------------------------------------------------------------------
    # If we still have time, throw a few extra random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        now = time.time()
        if deadline - now < 30.0:
            break
        extra_start = rng.dirichlet(np.full(n, 0.3))
        candidate = _run_optimisation(extra_start, deadline, start_time)
        cand_score = _exact_score(candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            print(f"[{time.time() - start_time:.2f}s] New overall best (extra) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
