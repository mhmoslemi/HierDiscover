def subgradient(seq, ks):
    n = len(seq)

    for k in ks:

        idx = np.arange(lo, hi + 1)

def _grad_smooth_max(a, beta):
    n = a.shape[0]
    conv = _conv_self(a)
    maxc = conv.max()
    # shift conv to avoid overflow
    shifted = conv - maxc  # negative or zero.
    # compute weight
    # to avoid underflow, scale
    w = np.exp(beta * shifted)
    sumw = w.sum()
    if sumw == 0.0 or np.isnan(sumw):
        # fallback
        # compute top_k.
        ks = np.argpartition(conv, -3)[-3:]
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi+1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(ks)
        return grad
    p = w / sumw
    # compute gradient via FFT
    rev_a = a[::-1]  # shape (n,)
    L = _next_pow2(3 * n - 2)
    # compute FFT using real rfft for efficiency
    Fp = np.fft.rfft(p, n=L)
    Fr = np.fft.rfft(rev_a, n=L)
    h = np.fft.irfft(Fp * Fr, n=L)
    # extract segment (n-1) to (2n-2)
    grad = 2.0 * h[(n - 1):(n - 1 + n)]
    return grad

def _adam_optimize(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5
    beta_smooth = 5.0      # initial smoothing beta
    beta_max = 200.0
    beta_inc = (beta_max - beta_smooth) / 100000   # schedule
    it = 0
    while time.time() < deadline:
        it += 1
        # compute grad
        beta_cur = min(beta_max, beta_smooth + beta_inc * it)
        grad = _grad_smooth_max(a, beta_cur)
        # adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)  # diminishing step
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_simplex(a)
        # evaluate objective
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")
        # optional early stop if stagnates < threshold
        if it % 10000 == 0:
            # if best hasn't improved enough, maybe increase smoothing?
            pass
    return best, best_score

"""
Propose a step‑function with a low evaluation score.

Algorithm (≈1000 s budget)
---------------------------
1. **Problem size** – choose a power‑of‑two length `n`.  
   For a generous budget we use `n = 131072` (2¹⁷); this gives a
   finer discretisation than the previous 2¹⁶ and still allows many
   FFTs within the time limit.

2. **Initial pool** – sweep a few analytic families
   (power‑ramp, symmetric Beta‑shapes and powered sine windows)
   on a coarse grid, evaluate them with the fast objective
   `_objective` and keep the best 8 candidates.

3. **Projected Adam descent** – for each candidate we run a
   *scale‑invariant* optimisation.
   * The loss we minimise is the maximum entry of the self‑convolution
     (the `evaluate_sequence` score is `2·n·max(conv)` when the sum is
     normalised to 1).  
   * The gradient is obtained from a *smooth* approximation of the
     maximum:  
     `p_k ∝ exp(β·(conv_k−max(conv)))`.  
     For β≈5 the gradient is a weighted sum over many convolution
     entries, which gives a much smoother search direction than the
     previous “top‑K” sub‑gradient.  
   * The gradient is computed by a second FFT (convolution of `p`
     with the reversed vector) – the whole iteration costs only a
     few FFTs.  
   * An Adam optimiser (first‑ and second‑moment adaptation) with a
     diminishing learning‑rate (`η_t = η₀/√t`) is used, followed by a
     projection onto the probability simplex `{x≥0, Σx=1}`.

   The β‑parameter is increased slowly up to `β_max = 200`, turning the
   smoothed maximum into the true maximum as the optimisation proceeds.

4. **Light simulated‑annealing polish** – after the Adam phase a few
   hundred‑thousand tiny mass‑transfer moves are attempted with a
   Metropolis rule; this often removes the last few spikes.

5. **Result normalisation** – clip small negatives, re‑normalise to a
   probability vector and return a plain Python list.

All helper functions are defined at top level so they can be pickled
if the harness runs the solution in a child process.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Small utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear convolution a*a via real FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (scale‑invariant):
#   2·n·max(conv)  when sum(a)=1
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq))  (sum(seq) is assumed = 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families for the initial pool
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric power‑ramp with a possible flat plateau.
    Parameters
    ----------
    p     – exponent of the side ramps (p>0).
    alpha – relative plateau width ∈[0,0.5]; 0 → pure tent,
            0.5 → uniform.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0.0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, exp: float) -> np.ndarray:
    """Symmetric Beta‑type shape: (x(1−x))**exp, normalised."""
    x = (np.arange(n) + 0.5) / n
    shape = (x ** exp) * ((1.0 - x) ** exp)
    total = shape.sum()
    if total > 0.0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _sin_shape(n: int, exp: float) -> np.ndarray:
    """Powered sine window: sin(πx)**exp."""
    x = (np.arange(n) + 0.5) / n
    shape = np.sin(np.pi * x) ** exp
    total = shape.sum()
    if total > 0.0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Gradient of a smoothed maximum of the self‑convolution
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, beta: float, fallback_top_k: int = 3) -> np.ndarray:
    """
    Compute a sub‑gradient of the smooth approximation of max(conv).
    The smoothing weight is  p_k ∝ exp(β·(conv_k−max_conv)).
    If the weights under‑flow, fall back to a top‑K sub‑gradient.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    maxc = conv.max()
    shifted = conv - maxc                     # ≤0
    # avoid overflow/under‑flow
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        # fallback – use a few largest entries (the previous approach)
        ks = np.argpartition(conv, -fallback_top_k)[-fallback_top_k:]
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))
        return grad

    p = w / w_sum

    # gradient = 2 * (p  ⋆  a_rev)   (⋆ = linear convolution)
    a_rev = a[::-1]
    L = _next_pow2(3 * n - 2)                # length needed for (2n‑1)+(n‑1)‑1
    Fp = np.fft.rfft(p, n=L)
    Fr = np.fft.rfft(a_rev, n=L)
    h = np.fft.irfft(Fp * Fr, n=L)           # length L
    grad = 2.0 * h[(n - 1):(n - 1 + n)]
    return grad


# ----------------------------------------------------------------------
# Projected Adam descent (smooth‑max gradient)
# ----------------------------------------------------------------------
def _adam_descent(start_seq: np.ndarray,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with simplex projection.
    Returns the best iterate seen and its (scale‑invariant) score.
    """
    a = start_seq.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.6

    # smoothing schedule for the soft‑max
    beta_cur = 5.0
    beta_max = 200.0
    beta_inc = (beta_max - beta_cur) / 120000   # roughly reach max after 1.2e5 iters

    it = 0
    while time.time() < deadline:
        it += 1

        # --- gradient of smooth max ---
        grad = _smooth_max_grad(a, beta_cur, fallback_top_k=3)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)                 # keep on the simplex

        # --- evaluate ---
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # increase smoothing parameter
        beta_cur = min(beta_max, beta_cur + beta_inc)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,

    """Tiny mass transfers with Metropolis acceptance."""

    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of attempts – scaled to remaining time
    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # simple exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)        # more mass ⇒ more likely to be a source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move at most 25 % of the source mass
        eps_move = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
