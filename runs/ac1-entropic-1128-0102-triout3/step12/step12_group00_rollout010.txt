"""
Propose a low‑score step function (non‑negative sequence).

Algorithm
---------
* **Problem size** – use `n = 2**16 = 65536`.  The FFT based convolution
  (`np.fft.rfft`) runs in a few milliseconds, so many iterations fit into the
  1000 s budget.

* **Rich initial pool** – generate a few hundred candidate shapes from several
  families (symmetric power ramps, monotone power ramps, exponential decay,
  double‑Gaussian bumps).  All candidates are normalised to a probability
  vector (sum = 1) and the best few are kept.

* **Adam‑based projected optimisation** – for each of the top candidates we
  run a projected Adam optimiser on the simplex.
  The sub‑gradient for `max(conv(a,a))` is obtained from the `top_k` largest
  convolution entries (default `top_k = 5`).  After each Adam step the vector
  is projected back onto the probability simplex with the O(n) algorithm of
  Wang & Carreira‑Perpiñán.

* **Greedy annealing polish** – after the Adam phase the remaining time is
  spent on a simple Metropolis‑style mass‑transfer annealing that moves a tiny
  fraction of the mass from a source index to a random destination.  Only
  moves that improve (or are accepted according to a fast cooling schedule)
  are kept.

* **Return** – the final vector is clipped to non‑negative, re‑normalised to
  sum = 1 and converted to a plain Python list (exactly what the harness
  expects).

The whole search respects the 1000 s time limit, prints progress reports and
aims to beat the previous best bound `≈ 1.508047`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    The exact objective used by ``evaluate_sequence`` (no safety clamp).
    For a probability vector (sum=1) this equals 2·n·max(conv)/1².
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:               # should not happen for a normalized vector
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Implementation of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:               # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w

# Shape factories – diverse families for the initial pool

def _make_symmetric_power_ramp(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape: two power ramps meeting a plateau of height 1.
    ``p`` > 0 controls the curvature, ``alpha`` ∈ [0,0.5] the relative
    plateau width (0 → pure tent, 0.5 → uniform).
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                       # goes from 0 … 1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # fix possible size mismatch caused by rounding
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_monotone_power(n: int, p: float, reverse: bool = False) -> np.ndarray:
    """
    Strictly monotone power shape a_i ∝ ((i+1)/n) ** p.
    If ``reverse`` is True the curve is flipped (large values on the left).
    """
    t = (np.arange(n, dtype=float) + 1) / n
    shape = t ** p
    if reverse:
        shape = shape[::-1]
    total = shape.sum()
    shape /= total if total > 0 else 1.0
    return shape


def _make_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    shape = np.exp(-lam * i)
    if reverse:
        shape = shape[::-1]
    total = shape.sum()
    shape /= total if total > 0 else 1.0
    return shape


def _make_two_gaussian(n: int,
                       mu1: float, sigma1: float,
                       mu2: float, sigma2: float,
                       amp1: float = 1.0, amp2: float = 1.0) -> np.ndarray:
    """
    Sum of two Gaussian bumps (not normalised).  All parameters are in
    index units (0 … n‑1).  The result is projected onto the simplex.
    """
    i = np.arange(n, dtype=float)
    g1 = amp1 * np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
    g2 = amp2 * np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
    shape = g1 + g2
    total = shape.sum()
    shape = shape / total if total > 0 else np.full(n, 1.0 / n)
    return shape

# Optimisation – Adam on the simplex
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of  max(conv(a,a)).
    Returns the best vector encountered and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    it = 0
    while time.time() < deadline:
        it += 1

        # current objective
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # indices that currently bind the max (top‑k largest entries)
        if top_k >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient (average over selected ks)
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size (similar to 0.5 / sqrt(t) used before)
        lr = 0.5 / math.sqrt(it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex
        a = _project_simplex(a_new)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light annealing – greedy mass transfer
# ----------------------------------------------------------------------


def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Greedy Metropolis annealing that swaps tiny packets of mass."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 300_000   # enough to fill the remaining time in most runs
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # simple exponential cooling (fast decay)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # pick source proportionally to its current mass (more mass → larger chance)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 20 %) of the source mass
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

    start_time = time.time()
    # keep a small safety margin before the hard limit
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two for a fast FFT
    # --------------------------------------------------------------
    n = 65536  # 2**16 – good trade‑off between quality and speed

    # --------------------------------------------------------------
    # 1) Build a diverse initial pool of candidates
    # --------------------------------------------------------------
    top_pool_size = 8
    pool = []  # list of (score, vector)

    def _push_candidate(vec: np.ndarray):
        """Insert vec into the pool if it improves the top‑k."""
        sc = _objective(vec)
        if len(pool) < top_pool_size:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        else:
            if sc < pool[-1][0]:
                pool[-1] = (sc, vec.copy())
                pool.sort(key=lambda x: x[0])

    # a) Global starter, if the harness provided one
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                _push_candidate(starter)
    except Exception:
        pass

    rng = np.random.default_rng()

    # b) Symmetric power‑ramp families (already proved useful)
    for _ in range(120):
        p = rng.uniform(0.2, 3.0)
        alpha = rng.uniform(0.0, 0.5)
        a = _make_symmetric_power_ramp(n, p, alpha)
        _push_candidate(a)

    # c) Monotone power ramps (both directions)
    for _ in range(80):
        p = rng.uniform(0.2, 4.0)
        a_fwd = _make_monotone_power(n, p, reverse=False)
        a_rev = _make_monotone_power(n, p, reverse=True)
        _push_candidate(a_fwd)
        _push_candidate(a_rev)

    # d) Exponential decays (both sides)
    for _ in range(80):
        lam = rng.uniform(0.0005, 0.05)
        a_fwd = _make_exponential(n, lam, reverse=False)
        a_rev = _make_exponential(n, lam, reverse=True)
        _push_candidate(a_fwd)
        _push_candidate(a_rev)

    # e) Two‑Gaussian mixtures – random centres & widths
    for _ in range(80):
        mu1 = rng.uniform(0.0, n * 0.4)
        mu2 = rng.uniform(n * 0.6, n - 1)
        sigma1 = rng.uniform(0.02, 0.12) * n
        sigma2 = rng.uniform(0.02, 0.12) * n
        amp1 = rng.uniform(0.5, 1.5)
        amp2 = rng.uniform(0.5, 1.5)
        a = _make_two_gaussian(n, mu1, sigma1, mu2, sigma2, amp1, amp2)
        _push_candidate(a)

    # keep only the best few for the intensive phase
    pool.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in pool[:3]]
    best_score = pool[0][0] if pool else float("inf")
    best_seq = pool[0][1].copy() if pool else np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Projected Adam optimisation on the best few candidates
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # allocate most of the remaining budget to Adam, keep a few seconds for polish
        adam_budget = remaining * 0.9
        per_candidate = adam_budget / len(top_candidates)
        for idx, cand in enumerate(top_candidates, 1):
            sub_deadline = time.time() + per_candidate
            refined, refined_score = _adam_optimize(cand, sub_deadline, start_time, top_k=5)
            print(f"[{time.time() - start_time:.2f}s] Adam finish cand{idx} score = {refined_score:.6f}")
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined.copy()

    # --------------------------------------------------------------
    # 3) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        annealed, annealed_score = _anneal(best_seq, deadline, start_time)
        if annealed_score < best_score:
            best_score = annealed_score
            best_seq = annealed.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")

    # Return as a plain list of Python floats
    return [float(v) for v in best_seq]
