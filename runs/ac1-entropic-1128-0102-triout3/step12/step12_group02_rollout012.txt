"""
Algorithm to minimize 2 * n * max(conv(a,a)) / (sum(a)^2).

The approach combines:
1. A parametric sweep over monotone families (geometric * power + uniform).
2. Several random restarts followed by smooth‑max Adam descent.
3. A final subgradient descent that directly reduces the true max‑convolution.
4. A microscopic hill‑climb for final polishing.

All steps respect the 1000 s budget.
"""
import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    return conv

def _project_simplex(v):
    # Euclidean projection onto probability simplex.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho+1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _eval_fast(a, L_fft):
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    n = a.shape[0]
    return float(2.0 * n * max_b / (s*s))

def _smooth_descent(a0, n, deadline, start_time, best_score):
    """
    Adam‑style descent on the soft‑max surrogate.
    """
    a = a0.copy()
    best_a = a.copy()
    best_local = best_score
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2*n-1)
    L_corr = _next_pow2(3*n-2)

    beta = 5.0
    beta_mul = 1.02
    beta_max = 500.0   # higher than before

    while time.time() < deadline:
        t += 1
        # compute soft‑max gradient
        # convolution
        conv = _conv_self_fft(a, L_conv)
        max_c = conv.max()
        shifted = conv - max_c
        # avoid overflow/underflow using clipping
        # large negative values cause exp close to zero; safe
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        # correlation
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W*A, n=L_corr)
        start = n-1
        grad = 2.0 * corr[start:start+n]

        # exact score for monitoring
        cur_score = 2.0 * n * conv.max() / (a.sum()**2)
        if cur_score < best_local:
            best_local = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] smooth it={t} β={beta:.1f} score={cur_score:.6f}")

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)
        step = 0.2 / (np.sqrt(v_hat) + eps)   # a bit larger step
        a = a - step*m_hat
        # ensure non‑negativity by projection onto simplex
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # occasional random jitter
        if t % 4000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_local

def _max_subgradient(a):
    """
    Compute a subgradient of max(conv(a,a)).
    Returns gradient vector g (same shape as a).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)  # direct O(n^2) might be heavy for n=4k, but ok occasionally
    max_val = conv.max()
    # obtain index of a maximal entry (first)
    k = int(np.argmax(conv))
    # compute subgradient for shift k
    g = np.zeros_like(a)
    # i ranges where j = k - i lies in [0, n-1]
    i_start = max(0, k - (n-1))
    i_end = min(n-1, k)
    # vectorized loop
    i_vals = np.arange(i_start, i_end+1)
    j_vals = k - i_vals
    # add a[j] to grad[i]
    g[i_vals] += a[j_vals]
    # For i != j, also add a[i] to grad[j]
    mask = i_vals != j_vals
    if np.any(mask):
        g[j_vals[mask]] += a[i_vals[mask]]
    # For i==j term contributes only a[i], already added above since i_vals==j_vals
    # Multiply by 2 for i != j case? Wait we already added both contributions separately.
    return g, max_val

def _subgradient_descent(a0, n, deadline, start_time, best_score_initial):
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    # step size schedule
    step0 = 0.1
    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # compute subgradient
        g, max_val = _max_subgradient(a)
        # step size decays
        step = step0 / np.sqrt(iter_cnt)
        # gradient step (we want to *decrease* max, so move opposite direction)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)
        # evaluate
        cur_score = _eval_fast(a_candidate, _next_pow2(2*n-1))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time()-start_time:.2f}s] subgrad it={iter_cnt} step={step:.5f} score={cur_score:.6f}")
        else:
            # occasionally accept a worse move to escape (simulated annealing)
            # probability decays with iteration
            prob = np.exp(-(cur_score - best_score) * 50.0 / (iter_cnt+1))
            if rng.random() < prob:
                a = a_candidate
        # occasional jitter
        if iter_cnt % 2000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

def _final_hill_climb(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2*n-1)
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] hill‑climb score={best_score:.6f}")
    return best_a, best_score

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step function of length n that minimizes the
    evaluation function.  The routine mixes a parametric sweep,
    several random restarts with Adam‑style smooth‑max descent,
    a dedicated sub‑gradient refinement on the true max, and a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin
    # problem size: we keep n moderate to keep per‑iteration cheap,
    # but large enough to allow fine shaping.
    n = 4096
    L_fft = _next_pow2(2*n-1)

    # ------------------------------------------------------------------
    # baseline
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0/n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time()-start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # if a global starter exists, use it
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time()-start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ parametric sweep (fast)
    # ------------------------------------------------------------------
    idx = np.arange(1, n+1, dtype=np.float64)
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-3.0, 2.0, 41)
    for r in r_grid:
        if time.time() >= deadline - 200:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 200:
                break
            shape = geo * np.power(idx, p)
            total = shape.sum()
            if total == 0:
                continue
            shape /= total
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                cand = lam * uniform + (1-lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time()-start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ several random restarts with smooth‑max descent
    # ------------------------------------------------------------------
    # allocate ~500 s for this stage
    time_for_smooth = min(deadline - time.time() - 250, 500)  # keep 250 s for later phases
    n_restarts = 3
    per_restart = time_for_smooth / n_restarts
    for i in range(n_restarts):
        if time.time() >= deadline - 250:
            break
        # initialise from a noisy version of best_seq
        rng = np.random.default_rng(seed + i + 100)
        a0 = best_seq.copy()
        noise = rng.normal(scale=0.02, size=n)   # 2 % relative noise
        a0 = a0 + noise
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        sub_deadline = time.time() + per_restart
        a_smooth, sc_smooth = _smooth_descent(a0, n, sub_deadline, start_time, best_score)
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time()-start_time:.2f}s] restart {i+1}/{n_restarts} improvement score={sc_smooth:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ sub‑gradient refinement on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        sub_deadline = deadline - 120
        a_sub, sc_sub = _subgradient_descent(best_seq, n, sub_deadline, start_time, best_score)

            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time()-start_time:.2f}s] sub‑gradient final score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ final hill‑climb polish (light)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time()-start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0/n)

    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The strategy mixes several complementary phases:

1️⃣  A fast parametric sweep over geometric‑power families (mixed with the
    uniform distribution) to obtain a strong initial candidate.

2️⃣  Several random restarts followed by an Adam‑style projected descent on a
    smooth‑max surrogate of the ``max(conv)`` term.  The temperature β is
    increased slowly so the optimisation first sees a smooth landscape and
    later focuses on the true objective.

3️⃣  A dedicated sub‑gradient descent that works on the exact max‑convolution.
    The gradient is the sub‑gradient of the current maximal convolution entry,
    computed in O(n) time after an O(n log n) FFT to locate the maximum.

4️⃣  A final hill‑climb that transfers tiny amounts of mass between random
    coordinates, accepting only improving moves (microscopic polishing).

All phases respect the 1000 s budget, keep the vector on the probability
simplex (non‑negative and sum = 1) and finally return the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w≥0, sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑2: smooth‑max Adam descent
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray, n: int, deadline: float,
                    start_time: float, best_score_initial: float):
    """
    Adam‑style projected descent on the soft‑max surrogate of max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0                # start temperature
    beta_mul = 1.02
    beta_max = 500.0

    while time.time() < deadline:
        t += 1

        # --- soft‑max gradient -------------------------------------------------
        conv = _conv_self_fft(a, L_conv)            # linear conv, length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c                      # stabilise exponentials
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()               # soft‑max weights, shape (2n‑1,)

        # correlation w * a_rev  →  gradient of soft‑max
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)        # linear convolution
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        # exact objective for monitoring
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} "
                  f"score={cur_score:.6f}")

        # --- Adam update on the simplex ---------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)          # a bit larger than before
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional jitter to get out of plateaus
        if t % 4000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score_initial: float):
    """
    Projected sub‑gradient descent directly on the true max‑convolution
    objective.  The sub‑gradient is obtained from a single maximal shift.
    """
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.12   # initial step magnitude
    it = 0

    while time.time() < deadline:
        it += 1

        # locate the (first) maximal convolution entry using FFT
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))            # shift achieving the max

        # --- compute sub‑gradient for shift k (O(n)) -------------------------
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        # contribution a_j to ∂/∂a_i
        g[i_vals] += a[j_vals]
        # for i≠j also add a_i to ∂/∂a_j (the other half of the pair)
        mask = i_vals != j_vals
        if np.any(mask):
            g[j_vals[mask]] += a[i_vals[mask]]

        # --- step with diminishing size ------------------------------------
        step = step0 / np.sqrt(it)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional probabilistic acceptance (simulated‑annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 15.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter every few thousand iterations
        if it % 2000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4: final microscopic hill‑climb
# ----------------------------------------------------------------------


def _final_hill_climb(a0: np.ndarray, deadline: float, start_time: float):
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-3      # up to 0.1 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score
