    n = 4096

    global L_SELF, L_GRAD set accordingly.

    best_seq = None
    best_score = float('inf')
    # set of shape types
    shapes = ['uniform', 'tent', 'cosine', 'beta', 'random']

    # Quick exploration
    for shape in shapes:

    # deep optimization on best_seq until time out
    if best_seq is None:

    # run full phases with longer loops
    a, score = full_optimize(a, deadline)

def _smooth_descent_momentum(a, deadline, start):
    best_a = a.copy()
    best_score = _exact_score(best_a)
    # momentum vector
    v = np.zeros_like(a)
    beta = 0.9
    it = 0
    alphas = np.geomspace(0.5, 2000.0, num=48)
    for alpha in alphas:
        if time.time() >= deadline - 0.2:
            break
        while time.time() < deadline - 0.2:
            it += 1
            # compute convolution
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[:2*n-1]
            # smooth max weights
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)
            w_sum = w.sum()
            if w_sum == 0:
                w.fill(1.0 / w.shape[0])
            else:
                w /= w_sum
            # gradient
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_GRAD)
            Revf = np.fft.rfft(a_rev, n=_L_GRAD)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[:3*n-2]
            grad = 2.0 * grad_full[n-1:n-1+n]
            # momentum update
            v = beta * v + (1 - beta) * grad
            # step size eta (use same schedule)
            eta = 0.1 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * v)
            a = np.maximum(a, 1e-30)
            a_sum = a.sum()
            if a_sum>0:

"""
Improved three‑phase optimizer for the step‑function objective

The algorithm explores several promising initial shapes (uniform,
tent, cosine window, Beta‑distribution, random Dirichlet) and runs
a fast “quick” optimisation on each of them.  The best candidate is
then refined with a longer run of:

  1. Smoothed mirror‑descent on the log‑sum‑exp approximation of the
     max (exponentiated‑gradient with momentum).
  2. Direct sub‑gradient descent on the true max (again with momentum).
  3. An advanced simulated‑annealing hill‑climber that mixes small
     mass transfers with occasional global reshuffles.

All phases stay on the probability simplex (∑a = 1) and respect the
global time budget (default 1000 s).  The final sequence is normalised
and returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global problem size (fixed as in the reference implementation)
# ----------------------------------------------------------------------
n = 4096
# FFT lengths – powers of two for speed
_L_SELF = 1 << ((2 * n - 2).bit_length())      # ≥ 2n‑1
_L_GRAD = 1 << ((3 * n - 3).bit_length())      # ≥ 3n‑2


# ----------------------------------------------------------------------
# Exact objective (fast FFT based)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2)."""
    Af = np.fft.rfft(a, n=_L_SELF)
    conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:           # safety – should never happen
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Helper: generate a few sensible starting shapes
# ----------------------------------------------------------------------
def _initial_shape(shape: str, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative array of length n."""
    if shape == "uniform":
        a = np.full(n, 1.0 / n)

    elif shape == "tent":
        # symmetric triangular (peak in the middle)
        if n % 2 == 0:
            half = n // 2
            left = np.arange(1, half + 1, dtype=float)
            right = left[::-1]
        else:
            half = n // 2
            left = np.arange(1, half + 2, dtype=float)
            right = np.arange(half, 0, -1, dtype=float)
        a = np.concatenate((left, right))
        a /= a.sum()

    elif shape == "cosine":
        # Hann window – zero at the ends, smooth bell in the centre
        i = np.arange(n, dtype=float)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

    elif shape == "beta":
        # Symmetric beta‑distribution (parameters drawn randomly)
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        i = np.arange(n, dtype=float) + 0.5
        a = (i ** (alpha - 1)) * ((n - i) ** (beta - 1))
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

    elif shape == "random":
        # Dirichlet with a small concentration → sparse, varied start
        conc = rng.uniform(0.05, 0.5)
        a = rng.dirichlet(np.full(n, conc))
    else:
        a = np.full(n, 1.0 / n)

    return a.astype(float)


# ----------------------------------------------------------------------
# Phase 1 – smoothed log‑sum‑exp mirror descent (with momentum)
# ----------------------------------------------------------------------
def _smooth_descent_momentum(a: np.ndarray,
                             deadline: float,
                             start: float,
                             alphas: np.ndarray = None) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient on a log‑sum‑exp approximation of max."""
    if alphas is None:
        alphas = np.geomspace(0.5, 2000.0, num=48)      # default schedule

    best_a = a.copy()
    best_score = _exact_score(best_a)

    v = np.zeros_like(a)               # momentum buffer
    beta = 0.9
    it = 0

    for alpha in alphas:
        if time.time() >= deadline - 0.2:
            break

        # inner loop: keep iterating while we have time for this α
        while time.time() < deadline - 0.2:
            it += 1

            # 1) self‑convolution a*a
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]

            # 2) soft‑max weights w = softmax(α·conv)
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_GRAD)
            Revf = np.fft.rfft(a_rev, n=_L_GRAD)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) momentum + exponentiated‑gradient step on the simplex
            v = beta * v + (1.0 - beta) * grad
            eta = 0.1 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * v)

            # numerical safety
            a = np.maximum(a, 1e-30)
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            else:
                a = np.full_like(a, 1.0 / n)

            # occasional exact evaluation
            if it % 8000 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start:.2f}s] smooth‑mom α={alpha:.1f} it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – direct max‑subgradient descent (exponentiated, with momentum)
# ----------------------------------------------------------------------
def _max_subgradient_momentum(a: np.ndarray,
                              deadline: float,
                              start: float) -> tuple[np.ndarray, float]:
    """Push mass away from the index attaining the current max, with momentum."""
    best_a = a.copy()
    best_score = _exact_score(best_a)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline - 0.2:
        it += 1

        # current convolution and its maximal index
        Af = np.fft.rfft(a, n=_L_SELF)
        conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
        k = int(conv.argmax())

        # sub‑gradient restricted to the active pair‑indices
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum buffer
        v = beta * v + (1.0 - beta) * grad

        # step size ~ 1/√t (standard for EG)
        eta = 0.5 / math.sqrt(it)
        a = a * np.exp(-eta * v)

        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        if it % 5000 == 0:
            cur_score = _exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start:.2f}s] sub‑mom it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – advanced stochastic hill‑climbing (annealing + global reshuffles)
# ----------------------------------------------------------------------
def _advanced_hill_climb(a: np.ndarray,
                         deadline: float,
                         start: float,
                         max_iters: int = None) -> tuple[np.ndarray, float]:
    """Small mass swaps + occasional global reshuffle, Metropolis acceptance."""
    cur = a.copy()
    cur_score = _exact_score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0

    start_temp = 5e-2
    end_temp = 1e-12

    while time.time() < deadline:
        it += 1
        if max_iters is not None and it > max_iters:
            break

        # linear temperature schedule
        progress = (time.time() - start) / (deadline - start)
        T = start_temp * (end_temp / start_temp) ** progress

        # 90 % of the time we do a simple two‑site transfer,
        # 10 % we perform a modest global reshuffle.
        if rng.random() < 0.9:
            src = rng.integers(n)
            if cur[src] < 1e-14:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            # transfer up to 40 % of the source mass
            eps = cur[src] * rng.random() * 0.40
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[src] -= eps
            cand[dst] += eps
        else:
            # Global reshuffle on a random 15 % of the coordinates
            mask = rng.random(n) < 0.15
            if not mask.any():
                continue
            mask_mass = cur[mask].sum()
            if mask_mass < 1e-12:
                continue
            new_weights = rng.random(mask.sum())
            new_weights /= new_weights.sum()
            cand = cur.copy()
            cand[mask] = new_weights * mask_mass

        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum > 0.0:
            cand /= cand_sum
        else:
            continue

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start:.2f}s] adv‑climb it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Quick optimisation – used for many initial shapes (short time budget)
# ----------------------------------------------------------------------
def _quick_optimize(a: np.ndarray,
                    deadline: float,
                    start: float) -> tuple[np.ndarray, float]:
    """A lightweight run of the three phases; stops automatically when time expires."""
    # Phase 1 – smoothed descent (shorter α schedule)
    a, _ = _smooth_descent_momentum(a,
                                    deadline,
                                    start,
                                    alphas=np.geomspace(0.5, 500.0, num=24))
    # Phase 2 – sub‑gradient with momentum (few thousand steps)
    a, _ = _max_subgradient_momentum(a, deadline, start)
    # Phase 3 – hill‑climb with a modest iteration bound
    a, score = _advanced_hill_climb(a, deadline, start, max_iters=15000)
    return a, score


# ----------------------------------------------------------------------
# Full optimisation – runs the three heavy phases until the deadline
# ----------------------------------------------------------------------
def _full_optimize(a: np.ndarray,
                   deadline: float,

    a, _ = _smooth_descent_momentum(a, deadline, start)
    a, _ = _max_subgradient_momentum(a, deadline, start)
    a, score = _advanced_hill_climb(a, deadline, start)
    return a, score


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Explore a handful of promising initial shapes
    # ------------------------------------------------------------------
    shape_list = ["uniform", "tent", "cosine", "beta", "random"]
    best_seq = None
    best_score = float("inf")

    # allocate roughly equal time to each quick run (but never exceed deadline)
    per_shape_budget = (deadline - start_time) * 0.4 / max(1, len(shape_list))
    for i, shape in enumerate(shape_list):
        shape_deadline = min(deadline - 0.1,
                             start_time + (i + 1) * per_shape_budget)
        if time.time() >= shape_deadline:
            break
        a0 = _initial_shape(shape, rng)
        a_opt, score = _quick_optimize(a0, shape_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] shape={shape} score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()

    # ------------------------------------------------------------------
    # 2. If we still have time, polish the best candidate with a full run
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)   # fallback – should not happen
    if time.time() < deadline - 5.0:
        best_seq, best_score = _full_optimize(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] after full optimisation best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final normalisation and conversion to a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")

    # Convert to plain Python float list (as required by the engine)
    return [float(v) for v in best_seq]
