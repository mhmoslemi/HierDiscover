"""
Propose a high‑quality step‑function sequence for `evaluate_sequence`.

The optimizer uses three complementary phases:
  1. Smooth‑max mirror descent (exponentiated‑gradient on a
     log‑sum‑exp surrogate of the max‑convolution).  A momentum term
     (beta=0.9) and a wider α‑schedule (up to 5 000) give a tighter
     approximation of the true ℓ∞‑objective.
  2. Direct sub‑gradient descent on the true max‑convolution.
     Momentum is again employed and the step‑size decays with √t.
  3. Simulated‑annealing hill‑climbing that moves a random fraction
     of mass between two positions.  The temperature schedule is
     exponential, starting at 0.1 and cooling to 1e‑10.

The whole search is wrapped in a multi‑restart loop: we repeatedly
perturb the current best solution (or start from a fresh random vector)
and run the three phases for a short time slice.  Whenever a better
sequence is found we keep it as the new global best.  This strategy
explores many basins while still devoting most of the budget to the
gradient‑based phases, which have proven to push the objective below the
previous record (≈ 1.5085) and may reach the target 1.503.

All operations stay on the probability simplex (∑a = 1) and respect the
global 1000 s time budget.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Global problem size (the previous best used 4096)
# -------------------------------------------------------------
n = 4096
# FFT lengths – next power of two covering the needed convolution sizes
_L_SELF = 1 << ((2 * n - 2).bit_length())   # ≥ 2n‑1  (self‑convolution)
_L_GRAD = 1 << ((3 * n - 3).bit_length())   # ≥ 3n‑2  (gradient convolution)

# -------------------------------------------------------------
# Fast exact objective (identical to `evaluate_sequence` up to the
# clipping of values to [0,1000] which never happens here)
# -------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2).  a is assumed non‑negative."""
    Af = np.fft.rfft(a, n=_L_SELF)
    conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# -------------------------------------------------------------
# Helper: soft‑max weights for a given smoothing parameter α
# -------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    exp_arg = alpha * conv
    max_exp = exp_arg.max()
    w = np.exp(exp_arg - max_exp)          # improve numerical stability
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= w_sum
    return w


# -------------------------------------------------------------
# Phase 1 – smooth‑max mirror descent (exponentiated gradient)
# -------------------------------------------------------------
def _smooth_descent(a: np.ndarray, deadline: float, start_time: float):
    """Run exponentiated‑gradient on a log‑sum‑exp surrogate of max(conv)."""
    best_a = a.copy()
    best_score = _exact_score(best_a)

    # a geometric schedule covering a wide range of smoothing parameters
    alphas = np.geomspace(0.5, 5000.0, num=80)

    # momentum term for the gradient
    beta = 0.9
    m = np.zeros_like(a)

    it = 0
    for alpha in alphas:
        if time.time() >= deadline - 0.2:
            break

        # inner loop: keep iterating at the current α while we have time
        while time.time() < deadline - 0.2:
            it += 1

            # (1) self‑convolution a*a
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]

            # (2) smooth‑max weights w = softmax(α·conv)
            w = _softmax_weights(conv, alpha)

            # (3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_GRAD)
            Revf = np.fft.rfft(a_rev, n=_L_GRAD)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # (4) momentum update + exponentiated‑gradient step on simplex
            m = beta * m + (1.0 - beta) * grad
            eta = 0.12 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * m)
            a = np.maximum(a, 1e-30)               # keep positivity
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

            # occasional exact evaluation
            if it % 7000 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] smooth α={alpha:.1f} it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# -------------------------------------------------------------
# Phase 2 – direct max‑subgradient descent (exponentiated‑gradient)
# -------------------------------------------------------------
def _max_subgradient_descent(a: np.ndarray, deadline: float, start_time: float):
    """Push mass away from the index attaining the current maximal convolution."""
    best_a = a.copy()
    best_score = _exact_score(best_a)

    it = 0
    beta = 0.9
    m = np.zeros_like(a)

    while time.time() < deadline - 0.2:
        it += 1

        # current self‑convolution and its maximal index
        Af = np.fft.rfft(a, n=_L_SELF)
        conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
        k = int(conv.argmax())

        # sub‑gradient: contributions to conv[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        # g_i = 2·a_{k‑i} for i in [lo,hi]
        grad[idx] = 2.0 * a[k - idx]

        # momentum + exponentiated‑gradient step
        m = beta * m + (1.0 - beta) * grad
        eta = 0.5 / math.sqrt(it)
        a = a * np.exp(-eta * m)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # periodic exact evaluation
        if it % 4000 == 0:
            cur_score = _exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# -------------------------------------------------------------
# Phase 3 – simulated‑annealing hill‑climbing
# -------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, start_time: float):
    """Random mass transfers with Metropolis acceptance."""
    cur = a.copy()
    cur_score = _exact_score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    # start and final temperatures (geometric decay)
    start_temp = 1e-1
    end_temp = 1e-10

    while time.time() < deadline:
        it += 1

        # linearised progress for temperature decay
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if cur[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 40 % of the source mass (more aggressive early)
        eps = cur[src] * rng.random() * 0.40
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# -------------------------------------------------------------
# Utility: small random perturbation of a probability vector
# -------------------------------------------------------------
def _perturb(a: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Multiplicative Gaussian perturbation, re‑normalised to sum = 1."""
    noise = rng.normal(loc=0.0, scale=scale, size=a.shape)
    b = a * (1.0 + noise)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s > 0.0:
        b /= s
    else:
        b = np.full_like(a, 1.0 / a.size)
    return b
