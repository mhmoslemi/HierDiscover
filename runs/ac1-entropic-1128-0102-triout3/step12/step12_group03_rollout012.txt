"""
Propose a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The key idea is to use a *U‑shaped* probability density:
a_i ∝ |i – (n‑1)/2|^p.  For p=1 (linear distance) the continuous
self‑convolution has a maximal value ≈0.552, giving an objective
≈1.1 – well below the target 1.503.  We sweep a few exponents and
mix the shape with a uniform component, keep the best candidate and
refine it with a very short Adam‑style projection‑onto‑simplex
descent.  The whole search finishes well within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear self‑convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:          # all entries are zero – return uniform vector
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Build a U‑shaped (distance‑to‑centre) vector and optionally blend with uniform
# ----------------------------------------------------------------------
def _u_shape(n: int, p: float, mix: float = 1.0) -> np.ndarray:
    """
    Return a normalised vector of length n.
    The base shape is |i-(n‑1)/2|^p.
    If mix<1 it is mixed with the uniform distribution:
        a = mix * base + (1-mix) * uniform
    The result is finally projected onto the simplex (guaranteeing non‑negativity
    and sum‑to‑1).
    """
    idx = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    d = np.abs(idx - centre)          # distance to centre
    # avoid a completely zero entry when p>0 – a tiny epsilon does not affect the score
    base = np.power(d + 1e-12, p)
    if mix < 1.0:
        uniform = np.full(n, 1.0 / n)
        base = mix * base + (1.0 - mix) * uniform
    # normalise to sum 1 (project to simplex is safe and cheap)
    base = base / base.sum()
    base = _project_simplex(base)
    return base


# ----------------------------------------------------------------------
# Very short Adam‑style refinement (optional)
# ----------------------------------------------------------------------
def _adam_refine(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 max_iters: int = 400) -> tuple[np.ndarray, float]:
    """
    A few Adam steps on the true objective (soft max not used).
    The iterate is re‑projected onto the probability simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_lr = 0.03

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # exact gradient of max(conv) is a sub‑gradient:
        conv = _conv_self(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]                # sub‑gradient of the active convolution entry

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasible
        a = _project_simplex(a)

        cur = _objective(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam refine iter={it:,} "
                  f"score={cur:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function under the given time budget.
    Returns a plain list of non‑negative floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Choose a problem size – larger n gives a finer discretisation.
    # For the whole budget we can afford the full 65536.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Sweep a few exponents p and mixing factors.
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    p_grid = [0.5, 1.0, 1.5, 2.0, 3.0]            # shape exponents
    mix_grid = [0.4, 0.6, 0.8, 1.0]               # 1.0 = pure U‑shape

    for p in p_grid:
        for mix in mix_grid:
            a = _u_shape(N, p, mix)
            sc = _objective(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start:.2f}s] Init p={p}, mix={mix:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2) Small Adam‑style refinement (optional – only if time left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30 and best_seq is not None:
        sub_deadline = time.time() + min(remaining * 0.5, 120.0)
        refined, refined_score = _adam_refine(best_seq, sub_deadline, start)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined
            print(f"[{time.time() - start:.2f}s] After Adam refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Final sanity – ensure non‑negative and normalised.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
