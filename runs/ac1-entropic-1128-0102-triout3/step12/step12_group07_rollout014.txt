def conv_self_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n - 2).bit_length())
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2*n-1]
    return c

propose_candidate
-----------------
A smooth‑gradient descent on the “max‑autocorrelation” objective.

The score of a probability vector p (Σp = 1) is
        score(p) = 2·n·max(p * p)
where “*” is linear convolution.  The problem is scale‑invariant, so we
search over the probability simplex.

Our algorithm iterates
  1. compute the convolution c = p * p,
  2. replace the non‑smooth max by a soft‑max  w_i ∝ exp(c_i/λ),
     (λ is decreased slowly so the approximation becomes tighter),
  3. the gradient of the smooth objective is
        g = 2·conv(p_rev , w)[n‑1 : n‑1+n]          (p_rev = reverse(p)),
  4. take a multiplicative (exponentiated‑gradient) step
        p ← p·exp(‑η·g)   and renormalise to the simplex,
  5. accept the step only if the true hard‑max score improves;
     otherwise shrink η and retry, or perform a tiny random jitter to
     escape plateaus.

All operations are performed with FFT‑based convolutions (O(n log n)),
allowing thousands of iterations within the 1000 s budget.  The method

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    L = _next_pow2(len(x) + len(y) - 1)

    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv

    total = best_p.sum()
    if total > 0:
        best_p = best_p / total
    else:
        best_p = np.full(n, 1.0 / n)

    return [float(v) for v in best_p]
