"""Step‑function optimiser.

The optimiser works in three stages:

1️⃣  A very broad parametric sweep over many simple families
    (geometric, power, beta‑type, triangular, cosine‑window,
    Gaussian) using fine grids and optional mixing with the uniform
    distribution.  The best shape found is kept.

2️⃣  One or more runs of a projected Adam optimiser on the probability
    simplex.  Each run starts from the current best shape or from a
    random Dirichlet point, runs until the allotted “Adam” time expires,
    and keeps the best iterate (sub‑gradient of the max‑convolution is
    used).

3️⃣  A stochastic hill‑climb that repeatedly transfers a random
    fraction of mass between two indices.  The transfer size is large
    at the beginning and shrinks linearly with the remaining budget,
    allowing both coarse and fine adjustments.

All convolutions are computed by FFT (O(n log n)) and the objective
value is `2·n·max(conv(a,a))` because the sequence is always projected
onto the simplex (∑a=1).  The whole search respects a user‑supplied
budget (default 1000 s).  The function returns a plain list of Python
floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, …)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full mode) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = a.shape[0]
    max_val = conv.max()
    # active shifts (tiny tolerance)
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _score_normalized(seq: np.ndarray, n: int, L_fft: int) -> float:
    """Return 2·n·max(conv(seq,seq)) for a sequence normalised to sum = 1."""
    total = seq.sum()
    if total <= 0:
        return np.inf
    seq = seq / total
    conv = _conv_self(seq, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser (projected on the simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score_initial: float,
                   n: int,
                   L_fft: int) -> tuple[np.ndarray, float]:
    """Projected Adam on the simplex, returns best encountered point."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam iteration {it:,} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # If no improvement for a while, shake the iterate a little.
        if it - last_improve > 3000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float,
                n: int,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Hill‑climb that moves a random fraction of mass between indices."""
    a = start_a.copy()
    best_a = a.copy()
    best_score = _score_normalized(a, n, L_fft)
    rng = np.random.default_rng()
    init_time = start_time

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # transfer size shrinks as time runs out
        remaining = deadline - time.time()
        total = deadline - init_time
        max_frac = 0.1 * (remaining / total) + 0.001
        frac = rng.random() * max_frac
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        cand_score = _score_normalized(a_prop, n, L_fft)
        if cand_score < best_score:
            a = a_prop
            best_a = a_prop.copy()
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The search consists of a dense parametric sweep, several Adam
    optimisation runs, and a final hill‑climb.  The routine respects the
    given time budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so we always finish before the external timeout
    deadline = start_time + budget_s - 0.15

    # Problem size
    # --------------------------------------------------------------
    n = 4096                     # fixed length that works well in practice
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline (uniform) and possible global starter
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _score_normalized(best_seq, n, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _score_normalized(starter, n, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter improved score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Fast but dense parametric sweep
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    for r in np.logspace(np.log10(0.5), np.log10(0.9999), 150):
        if time.time() >= deadline - 150:
            break
        geo = r ** idx
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo = geo / geo_sum
        for lam in lam_vals:
            seq = lam * uniform + (1.0 - lam) * geo
            seq = seq / seq.sum()
            cur = _score_normalized(seq, n, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo r={r:.6f} lam={lam:.2f} score={cur:.6f}")

    # ---- Power law -------------------------------------------------------
    for p in np.linspace(-3.0, 4.0, 141):
        if time.time() >= deadline - 150:
            break
        pow_seq = (idx + 1) ** p
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq = pow_seq / pow_sum
        cur = _score_normalized(pow_seq, n, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={cur:.6f}")

    # ---- Beta‑type (continuous p,q) --------------------------------------
    p_vals = np.linspace(0.0, 5.0, 51)
    q_vals = np.linspace(0.0, 5.0, 51)
    for p in p_vals:
        if time.time() >= deadline - 150:
            break
        for q in q_vals:
            beta_seq = (idx + 1) ** p * (n - idx) ** q
            beta_sum = beta_seq.sum()
            if beta_sum == 0:
                continue
            beta_seq = beta_seq / beta_sum
            cur = _score_normalized(beta_seq, n, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.2f} q={q:.2f} score={cur:.6f}")

    # ---- Triangular (min(i, n‑i)) ----------------------------------------
    for p in np.linspace(0.0, 4.0, 41):
        tri_seq = np.minimum(idx + 1, n - idx) ** p
        tri_seq = tri_seq / tri_seq.sum()
        cur = _score_normalized(tri_seq, n, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = tri_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Triangular p={p:.2f} score={cur:.6f}")

    # ---- Cosine (Hann‑type) ------------------------------------------------
    for p in np.linspace(0.0, 4.0, 41):
        cos_seq = (1.0 - np.cos(np.pi * idx / (n - 1))) ** p
        cos_seq = cos_seq / cos_seq.sum()
        cur = _score_normalized(cos_seq, n, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = cos_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Cosine p={p:.2f} score={cur:.6f}")

    # ---- Gaussian centred at the centre -----------------------------------
    for sigma_factor in np.linspace(0.05, 0.5, 30):
        sigma = sigma_factor * n
        gauss_seq = np.exp(-0.5 * ((idx - (n - 1) / 2) / sigma) ** 2)
        gauss_seq = gauss_seq / gauss_seq.sum()
        cur = _score_normalized(gauss_seq, n, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = gauss_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Gaussian σ={sigma_factor:.3f} score={cur:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Projected Adam optimisation (several short runs)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # Reserve a chunk for the final hill‑climb (≈15 % of the remaining)
    hill_time = min(150.0, remaining * 0.15)
    adam_deadline = deadline - hill_time

    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        runs = 0
        while time.time() < adam_deadline and runs < 5:
            # First run starts from the current best; later runs from a random Dirichlet point.
            if runs == 0:
                init_seq = best_seq.copy()
            else:
                init_seq = np.random.dirichlet(np.ones(n))
                # Blend a bit with the incumbent best to keep the search in a promising region.
                init_seq = 0.6 * init_seq + 0.4 * best_seq
                init_seq = _project_simplex(init_seq)

            a_opt, sc_opt = _adam_optimize(
                init_seq, start_time, adam_deadline, best_score, n, L_fft
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam run {runs} improved to {sc_opt:.6f}")

            runs += 1

    # --------------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline, n, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improvement to {sc_hc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best approx. score = {best_score:.6f}")
    return [float(x) for x in best_seq]
