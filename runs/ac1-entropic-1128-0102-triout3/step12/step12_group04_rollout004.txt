    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)

"""
Improved optimizer for the max‑convolution objective.

The algorithm consists of four stages:

1. **Warm‑start** – generate a collection of analytic shapes (tent,
   cosine, half‑circle, Gaussian, power‑tent) together with a few
   randomly sampled Dirichlet vectors.  The best of them (by the exact
   objective) seeds the main optimisation.

2. **Projected Adam with a smoothed max** – at each iteration the
   convolution `c = a*a` is computed.  A temperature‑controlled
   soft‑max turns `c` into a smooth approximation `w` of the
   ``max``‑operator.  The gradient of the weighted sum `w·c` is obtained
   analytically by a single FFT‑based convolution.  Adam’s adaptive
   step‑size together with a Euclidean projection onto the probability
   simplex keeps the vector non‑negative and normalised.  The temperature
   is exponentially cooled, so the optimisation progressively follows
   the true sub‑gradient of the max‑convolution.

3. **Light simulated‑annealing polish** – a cheap Metropolis walk moves
   a random packet of mass between two indices; it helps escape shallow
   local minima that the gradient step may miss.

4. **Final normalisation** – the best sequence seen is returned as a
   plain ``list[float]`` ready for ``evaluate_sequence``.

The whole search respects the 1000 s budget (a small safety margin is
kept).  The implementation works for any power‑of‑two length; here we
use ``n = 16384`` which gives a good trade‑off between expressive power
and runtime.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a normalised (∑a=1) vector.
    The function works for any non‑negative a (the sum is not re‑scaled).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient utilities – soft‑max weighted gradient
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Given a convolution vector ``conv`` and temperature ``tau``,
    return a probability vector that approximates the arg‑max:
    w_i ∝ exp((conv_i - max(conv))/tau).
    """
    if tau <= 0.0:
        tau = 1e-12
    max_val = conv.max()
    # subtract max for numerical stability
    exp_part = np.exp((conv - max_val) / tau)
    total = exp_part.sum()
    if total == 0.0:
        # fall back to a uniform distribution over the max entries
        w = np.zeros_like(conv)
        w[conv == max_val] = 1.0 / np.count_nonzero(conv == max_val)
        return w
    return exp_part / total


def _weighted_gradient(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_i w_i·conv_i (where conv = a*a)
    with respect to `a`.  The result is an array of length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)
    # The relevant slice corresponds to indices n‑1 … n‑1+n‑1
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser with temperature‑controlled soft‑max
# ----------------------------------------------------------------------


def _adam_optimize_softmax(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    init_lr: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    init_tau_factor: float = 0.2,
    tau_decay: float = 0.9995,
    min_tau_factor: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex that follows a soft‑max approximation
    of the true max‑convolution objective.
    Returns the best iterate encountered and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    # one FFT size works for both the convolution and the weighted gradient
    L = _next_pow2(3 * n)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- convolution and soft‑max weights -----
        conv = _conv_self(a, L)
        max_val = conv.max()

        # temperature schedule
        tau = max_val * max(min_tau_factor, init_tau_factor * (tau_decay ** it))

        w = _softmax_weights(conv, tau)

        # ----- gradient of the weighted sum -----
        grad = _weighted_gradient(a, w, L)

        # Scale the gradient to the true objective (2·n·max)
        grad_obj = 2.0 * n * grad

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ----- projection onto the simplex -----
        a = _project_simplex(a)

        # ----- bookkeeping -----
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                f"score={cur_score:.6f}  tau={tau:.2e}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same idea as the reference)
# ----------------------------------------------------------------------


def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk that moves a random packet of mass between two
    indices.  Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of proposals – adapt to remaining time
    total_steps = 80_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (very fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)            # source proportional to current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                    f"best={new_score:.6f}"
                )

    return best_a, best_score

# Warm‑starting shape generation

def _make_analytic_shapes(n: int) -> list[np.ndarray]:
    """Generate a small collection of analytic candidates (with a few
    parameter sweeps) normalised to sum = 1."""
    x = np.linspace(0.0, 1.0, n, dtype=float)

    candidates = []

    # 1) symmetric linear tent, optionally raised to a power
    for p in [0.5, 0.7, 1.0, 1.2, 1.5]:
        tent = np.minimum(x, 1.0 - x) ** p
        candidates.append(tent)

    # 2) raised cosine (sin) with optional power
    for p in [0.5, 1.0, 1.5, 2.0]:
        cosine = np.sin(np.pi * x) ** p
        candidates.append(cosine)

    # 3) half‑circle (Archimedes) with optional power
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    for p in [0.7, 1.0, 1.3]:
        candidates.append(circ ** p)

    # 4) Gaussian centred in the middle with several sigmas
    for sigma in [0.10, 0.15, 0.20, 0.25]:
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 5) mixture of two Gaussians (symmetric about the centre)
    for sigma in [0.12, 0.18]:
        g1 = np.exp(-0.5 * ((x - 0.33) / sigma) ** 2)
        g2 = np.exp(-0.5 * ((x - 0.66) / sigma) ** 2)
        mixture = 0.5 * g1 + 0.5 * g2
        candidates.append(mixture)

    # Normalise and keep only non‑negative vectors
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normals.append(c / s)
    return normals


def _make_random_candidates(n: int, rng: np.random.Generator, count: int = 12) -> list[np.ndarray]:
    """
    Produce a few random Dirichlet vectors with different concentration
    parameters.  This adds diversity beyond the analytic family.
    """
    candidates = []
    for i in range(count):
        # concentration <1 gives spiky vectors, >1 gives smoother ones
        alpha = rng.uniform(0.2, 2.0)
        vec = rng.gamma(alpha, 1.0, size=n)
        if vec.sum() > 0.0:
            candidates.append(vec / vec.sum())
    return candidates
