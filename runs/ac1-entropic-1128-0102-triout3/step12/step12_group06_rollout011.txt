"""
Non‑negative step‑function optimizer.

The algorithm starts from the best known sequence (if available) or from a
uniform distribution, rescales it so that its sum equals sqrt(2 n) –
under this scaling the objective 2·n·max(conv)/sum² reduces to simply
max(conv).  An efficient hill‑climbing loop then repeatedly moves a
fraction of mass from a large entry to a small entry while updating the
self‑convolution *incrementally* (no FFT per move).  The move is accepted
if it lowers the current maximum or with a Metropolis probability that
gradually cools.  Periodically the full convolution is recomputed from
scratch to eliminate numerical drift.  The best sequence found is finally
renormalised to sum = 1 and returned as a plain Python list.

The incremental update reduces the per‑move cost from O(n log n) (FFT)
to O(n), allowing many more iterations within the 1000 s budget and
typically improving the bound below the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray) -> np.ndarray:
    """Exact self‑convolution a*a using an FFT of suitable length."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2·n‑1
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _incremental_update(conv: np.ndarray, a: np.ndarray,
                        src: int, dst: int, eps: float) -> np.ndarray:
    """
    Return a copy of `conv` that would result from moving `eps` mass from
    index `src` to `dst` in `a`.  The update is performed analytically,
    i.e. without recomputing the whole convolution.
    """
    n = a.shape[0]
    new_conv = conv.copy()

    # first‑order contributions (2·a*δ)
    new_conv[src:src + n] -= 2.0 * eps * a
    new_conv[dst:dst + n] += 2.0 * eps * a

    # second‑order term δ*δ (only three positions are non‑zero)
    new_conv[2 * src] += eps * eps
    new_conv[2 * dst] += eps * eps
    idx_sum = src + dst
    if idx_sum != 2 * src and idx_sum != 2 * dst:
        new_conv[idx_sum] -= 2.0 * eps * eps

    return new_conv


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that minimises
    `evaluate_sequence` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # small safety margin

    # ------------------------------------------------------------------
    # Problem size and normalisation
    # ------------------------------------------------------------------
    n = 4096
    target_sum = math.sqrt(2.0 * n)                # scaling where objective = max(conv)

    # ------------------------------------------------------------------
    # Initialise from the best known construction if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        starter = None

    if starter is None or starter.ndim != 1 or starter.size == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = np.maximum(starter, 0.0)
        if a.size != n:
            if a.size > n:
                a = a[:n]
            else:
                a = np.pad(a, (0, n - a.size), mode='constant')
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        a = a / a.sum() * target_sum                # rescale to sum = sqrt(2·n)

    # ------------------------------------------------------------------
    # Initial exact convolution and objective
    # ------------------------------------------------------------------
    conv = _full_conv(a)
    cur_max = float(conv.max())
    best_a = a.copy()
    best_max = cur_max

    # ------------------------------------------------------------------
    # Hill‑climbing parameters
    # ------------------------------------------------------------------
    move_fraction = 0.25            # max fraction of source mass to move
    refft_interval = 8000           # recompute full convolution every N accepted moves
    start_temp = cur_max * 5e-2     # start temperature (a few percent of the peak)
    end_temp = cur_max * 1e-10
    accepted_moves = 0

    # Helper: fast max index
    def _argmax(x):
        return int(np.argmax(x))

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Choose source (large entry) and destination (small entry)
        # ------------------------------------------------------------------
        if rng.random() < 0.7:
            src = _argmax(a)                         # deterministic large entry
        else:
            src = rng.integers(n)                    # occasional random source

        if a[src] < 1e-15:                           # nothing to move
            continue

        # destination – usually the smallest entry, sometimes random
        if rng.random() < 0.6:
            dst = int(np.argmin(a))
        else:
            dst = rng.integers(n)

        if dst == src:
            continue

        # ------------------------------------------------------------------
        # Propose a move: transfer a random fraction of the source mass
        # ------------------------------------------------------------------

        if eps <= 0.0:

                break

        # ------------------------------------------------------------------
        # Periodic exact recomputation to avoid accumulating rounding error
        # ------------------------------------------------------------------
        if accepted_moves % refft_interval == 0:
            conv = _full_conv(a)
            cur_max = float(conv.max())
            if cur_max < best_max:

                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Re‑computed best max={best_max:.8f}")

    # ------------------------------------------------------------------
    # Final normalisation to sum = 1 (the evaluate_sequence routine expects it)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:

    final_score = 2.0 * n * np.max(np.convolve(best_a, best_a)) / (best_a.sum() ** 2)
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {final_score:.6f}")

    # Return plain Python list as required by the framework
    return [float(v) for v in best_a.tolist()]
