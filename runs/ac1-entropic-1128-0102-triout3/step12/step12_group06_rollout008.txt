def smooth_descent_adam(a, deadline):
    best_a = a.copy()
    best_score = _exact_score(best_a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0
    alphas = np.geomspace(0.1, 10000, num=80) # fine schedule
    for alpha in alphas:
        while time.time() < deadline - 0.2:
            t += 1
            # compute conv
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2*n-1]
            # softmax
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)
            w /= w.sum()
            # gradient
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_GRAD)
            Revf = np.fft.rfft(a_rev, n=_L_GRAD)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[: 3*n-2]
            grad = 2.0 * grad_full[n-1:n-1+n]
            # Adam moments
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            # adaptive learning rate
            lr = 0.05 / (np.sqrt(alpha) + 1e-12)  # decays with alpha
            a = a * np.exp(- lr * m_hat / (np.sqrt(v_hat) + epsilon))
            # avoid zeros
            a = np.maximum(a, 1e-30)
            # renorm
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            # occasional evaluation
            if t % 2000 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[...]")
    return best_a, best_score

def _max_subgradient_adam(a, deadline, start_time):
    # similar; but gradient only from max index.

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_score = _exact_score(a)

def _smooth_descent_adam(a, deadline, start_time):
    best_a = a.copy()
    best_score = _exact_score(best_a)
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    alphas = np.geomspace(0.5, 5000.0, num=80)  # larger schedule
    for alpha in alphas:
        if time.time() >= deadline - 0.2:
            break
        while time.time() < deadline - 0.2:
            t += 1
            # conv
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[:2*n-1]
            # softmax weights
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # gradient via FFT
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_GRAD)
            Revf = np.fft.rfft(a_rev, n=_L_GRAD)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[:3*n-2]
            grad = 2.0 * grad_full[n-1:n-1+n]
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            # learning rate schedule

"""
Multi‑restart entropic mirror descent + Adam‑accelerated sub‑gradient +
simulated‑annealing polishing.

The optimiser works on a simplex (∑a = 1) and respects the
non‑negativity constraint by means of exponentiated‑gradient (multiplicative)
updates.  Three phases are executed for each restart:

  1.  Smooth‑max (log‑sum‑exp) mirror descent with Adam momentum.
      The smoothing parameter α is increased geometrically (0.5 → 5 000)
      so that the surrogate gradually approaches the true max.
  2.  Exact max‑sub‑gradient descent, again with Adam, which pushes mass
      away from the indices that realise the current maximal convolution
      entry.
  3.  A stochastic hill‑climb (simulated annealing) that moves a fraction
      of the mass between randomly chosen source and destination positions.
      The temperature is exponentially decayed.

The whole procedure is wrapped in a time‑budget loop that repeatedly
re‑initialises the vector (either from the known best solution or from a
random (dense or sparse) simplex point).  The best sequence over all
restarts is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global problem size – will be (re)initialised in propose_candidate
# ----------------------------------------------------------------------
n = None            # length of the step function
_L_SELF = None      # FFT length for a*a (≥2n‑1)
_L_GRAD = None      # FFT length for smooth‑max gradient (≥3n‑2)


# ----------------------------------------------------------------------
# Exact objective (fast FFT based)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2).  a is assumed non‑negative."""
    Af = np.fft.rfft(a, n=_L_SELF)
    conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max mirror descent with Adam
# ----------------------------------------------------------------------
def _smooth_descent_adam(a: np.ndarray, deadline: float, start_time: float):
    """Log‑sum‑exp mirror descent with Adam‑style adaptive steps."""
    best_a = a.copy()
    best_score = _exact_score(best_a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    t = 0
    # geometric schedule of the smoothing parameter α
    alphas = np.geomspace(0.5, 5_000.0, num=80)

    for alpha in alphas:
        if time.time() >= deadline - 0.2:
            break
        while time.time() < deadline - 0.2:
            t += 1
            beta1_pow *= beta1
            beta2_pow *= beta2

            # 1) self‑convolution a*a
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]

            # 2) smooth‑max weights w = softmax(α·conv)
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)          # stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)  (FFT‑based convolution)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_GRAD)
            Revf = np.fft.rfft(a_rev, n=_L_GRAD)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam‑accelerated multiplicative update on the simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            # learning‑rate schedule (larger α → smaller steps)
            lr = 0.2 / (max(1.0, np.sqrt(alpha)) * np.sqrt(t))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a * np.exp(-step)
            a = np.maximum(a, 1e-30)               # avoid exact zeros
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum

            # occasional exact evaluation
            if t % 4000 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] smooth‑adam α={alpha:.1f} "
                          f"it={t:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – exact max‑subgradient descent with Adam
# ----------------------------------------------------------------------
def _max_subgradient_adam(a: np.ndarray, deadline: float, start_time: float):
    """Push mass away from the index that attains the current max convolution."""
    best_a = a.copy()
    best_score = _exact_score(best_a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    t = 0
    while time.time() < deadline - 0.2:
        # current self‑convolution and its maximal index
        Af = np.fft.rfft(a, n=_L_SELF)
        conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
        k = int(conv.argmax())

        # sub‑gradient (only entries that contribute to the max entry)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]               # g_i = 2·a_{k‑i}

        # Adam update
        t += 1
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.2 / np.sqrt(t)                     # slowly decaying step size
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a * np.exp(-step)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        if t % 3000 == 0:
            cur_score = _exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] sub‑grad‑adam it={t:,} "
                      f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – stochastic hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, start_time: float):
    """Small random mass transfers with Metropolis acceptance."""
    cur = a.copy()
    cur_score = _exact_score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    start_temp = 5e-2
    end_temp = 1e-10

    while time.time() < deadline:
        it += 1

        # linearised progress for temperature decay
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if cur[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 30 % of the source mass
        eps = cur[src] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)

        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            continue
        cand /= cand_sum

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: random initialisation (dense or sparse)
# ----------------------------------------------------------------------
def _random_initialization(rng: np.random.Generator, n: int, mode: str = "uniform"):
    """Return a random point on the simplex (mode = 'uniform' or 'sparse')."""
    if mode == "sparse":
        # pick a random number of active entries (5‑30 % of n)
        k = rng.integers(int(0.05 * n), int(0.30 * n) + 1)
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.random(k) + 1e-12
        a = np.zeros(n)
        a[idx] = vals
    else:  # dense uniform Dirichlet‑like
        a = rng.random(n) + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.

    The optimiser repeatedly restarts from either the known best construction
    or a random simplex point, runs:
        1. Smooth‑max mirror descent (Adam‑accelerated)
        2. Exact max‑subgradient descent (Adam‑accelerated)
        3. Simulated‑annealing hill‑climb
    and retains the globally best sequence.
    """
    # ------------------------------------------------------------------
    # Global initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # a small safety margin prevents accidental overruns
    deadline = start_time + budget_s - 0.3

    # problem size – 4096 works well and matches the previous best
    global n, _L_SELF, _L_GRAD
    n = 4096
    _L_SELF = 1 << ((2 * n - 2).bit_length())          # power of two ≥ 2n‑1
    _L_GRAD = 1 << ((3 * n - 3).bit_length())          # power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # Initialise from the previously known best (if any) or uniform
    # ------------------------------------------------------------------
    starter = None
    if 'height_sequence_1' in globals():
        try:
            seq = np.asarray(globals()['height_sequence_1'], dtype=float)
            if seq.ndim == 1 and seq.size > 0:
                starter = seq
        except Exception:
            starter = None

    if starter is None:
        starter = np.full(n, 1.0 / n)
    else:
        if starter.size != n:
            if starter.size > n:
                starter = starter[:n]
            else:
                starter = np.pad(starter, (0, n - starter.size), mode='constant')
        starter = np.maximum(starter, 0.0)
        total = starter.sum()
        if total > 0.0:
            starter = starter / total
        else:
            starter = np.full(n, 1.0 / n)

    best_seq = starter.copy()
    best_score = _exact_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Multi‑restart loop (each restart gets a fresh time slice)
    # ------------------------------------------------------------------
    restart = 0
    RESTART_BUDGET = 150.0            # seconds per restart (adjustable)

    while True:
        now = time.time()
        remaining = deadline - now
        if remaining < 30.0:
            break                     # not enough time for another full restart

        cur_budget = min(RESTART_BUDGET, remaining - 20.0)  # keep a margin
        cur_deadline = now + cur_budget

        # --------------------------------------------------------------
        # Random initialisation for this restart
        # --------------------------------------------------------------
        if restart == 0:
            a0 = starter.copy()
        else:
            mode = "sparse" if rng.random() < 0.5 else "uniform"
            a0 = _random_initialization(rng, n, mode)

        # --------------------------------------------------------------
        # Phase 1 – smooth‑max mirror descent (Adam)
        # --------------------------------------------------------------
        a1, sc1 = _smooth_descent_adam(a0, cur_deadline, start_time)
        if sc1 < best_score:
            best_seq, best_score = a1.copy(), sc1
            print(f"[{time.time() - start_time:.2f}s] Restart {restart} after smooth best={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – max‑subgradient descent (Adam)
        # --------------------------------------------------------------
        a2, sc2 = _max_subgradient_adam(a1, cur_deadline, start_time)
        if sc2 < best_score:
            best_seq, best_score = a2.copy(), sc2
            print(f"[{time.time() - start_time:.2f}s] Restart {restart} after sub‑grad best={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – hill‑climb polishing
        # --------------------------------------------------------------
        if time.time() < cur_deadline:
            a3, sc3 = _hill_climb(a2, cur_deadline, start_time)
            if sc3 < best_score:
                best_seq, best_score = a3.copy(), sc3
                print(f"[{time.time() - start_time:.2f}s] Restart {restart} after hill‑climb best={best_score:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Final polishing on the globally best sequence with any time left
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
