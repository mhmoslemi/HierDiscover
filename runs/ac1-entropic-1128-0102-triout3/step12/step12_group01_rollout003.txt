"""
propose_candidate – multi‑scale, multi‑restart optimiser for
  2·n·max(conv(a,a))/(sum(a)^2)

The search proceeds in three stages.

1️⃣  A diverse parametric sweep (geometric, power, beta, two‑ended
    exponentials and a few random Dirichlet draws) supplies a strong
    initial point on the simplex.

2️⃣  A long projected Adam optimiser on the probability simplex.
    The gradient is the (averaged) sub‑gradient of the active
    convolution shifts (those attaining the current maximum).  The
    learning‑rate decays slowly and, when progress stalls,
    a small Gaussian perturbation is added to avoid plateaus.

3️⃣  A simulated‑annealing hill‑climb.  Random mass transfers between
    two indices are proposed; the move is accepted if it improves the
    true objective, otherwise it may be accepted with probability
    exp(−Δ/T).  The temperature T is decreased exponentially.
    The climb runs until the time budget expires.

The routine works for any budget ≤ 1000 s, automatically adapts the
sequence length (4096 or 8192) to the remaining time and finally
returns a plain Python list of non‑negative floats ready for
``evaluate_sequence``.
"""

import time
import math

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (same as the reference implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    Uses all shifts attaining the maximum (within a tiny tolerance).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # each pair (i, k-i) contributes a_{k-i} to d/da_i and vice‑versa,
        # i.e. 2 * a_{k-i}
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Objective (exactly the one used by evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a))/(sum(a)^2).  a is a raw numpy array."""
    n = a.shape[0]
    conv = _conv_self(a, L_fft)
    max_val = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_val / (s * s))


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score: float) -> tuple[np.ndarray, float]:
    """Projected Adam on the simplex with a decaying learning‑rate."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.3                     # initial aggressive step
    dec = 1e-6                    # linear decay factor

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()
    best_a = a.copy()
    best_obj = best_score

    while time.time() < deadline:
        it += 1
        # current objective (sum(a) ≈ 1 because of projection)
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_obj = float(2.0 * n * cur_max)   # denominator ≈1
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"score={cur_obj:.6f}")

        # sub‑gradient of the max‑convolution term
        grad = _subgradient_maxconv(a, conv)

        # Adam update with a tiny linear decay of the learning‑rate
        lr = lr0 / (1.0 + dec * it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # when stuck for many iterations, inject a tiny random shake
        if it - last_improve > 30000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_obj


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climb with occasional uphill moves (annealing)."""
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    cur_obj = float(2.0 * n * _conv_self(a, L_fft).max())
    best_a, best_obj = a.copy(), cur_obj

    rng = np.random.default_rng()
    T0 = 1e-3 * cur_obj          # initial temperature (≈0.1 % of the score)
    T = T0
    decay = 0.99995               # exponential cooling per iteration

    # maximum fraction of source mass that may be moved in one step
    max_frac = 0.05

    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        frac = rng.random() * max_frac + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # (projected onto simplex is unnecessary – we keep the total mass)
        prop_obj = float(2.0 * n * _conv_self(a_prop, L_fft).max())

        Δ = prop_obj - cur_obj
        accept = False
        if Δ < 0:
            accept = True
        else:
            # metropolis acceptance
            if T > 0 and rng.random() < math.exp(-Δ / T):
                accept = True

        if accept:
            a = a_prop
            cur_obj = prop_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} "
                      f"score={best_obj:.6f}")

        # cool down
        T *= decay
        if T < 1e-12:
            T = 0.0

    return best_a, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The function respects the supplied time budget, uses a two‑stage
    sequence length (4096 for very short budgets, otherwise 8192),
    combines a large parametric sweep, a long projected Adam optimisation
    and a final simulated‑annealing hill‑climb, and finally returns the
    best sequence found as a list of Python floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin for the outer runner
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a richer search space)
    # --------------------------------------------------------------
    # If the budget is very tight, stay with n=4096; otherwise use 8192.
    if budget_s < 300:
        n = 4096
    else:
        n = 8192
    L_fft = _next_pow2(2 * n - 1)

    print(f"[{time.time() - start_time:.2f}s] Using n = {n}")

    # --------------------------------------------------------------
    # Baseline: uniform distribution
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1) – if present, use it
    # --------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                cur = _objective(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (geometric, power, beta, two‑ended)
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # --- geometric (decreasing) -------------------------------------------------
    eps_grid = np.logspace(-5, -1, 120)          # r = 1‑eps
    for eps in eps_grid:
        if time.time() >= deadline - 60:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        # mixture with uniform (lam ∈ [0,1])
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            cur = _objective(seq, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={cur:.6f}")

    # --- power family ----------------------------------------------------------
    p_grid = np.linspace(-1.7, 2.7, 81)
    for p in p_grid:
        if time.time() >= deadline - 60:
            break
        pow_seq = (idx + 1) ** p
        if pow_seq.sum() == 0:
            continue
        pow_seq /= pow_seq.sum()
        cur = _objective(pow_seq, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={cur:.6f}")

    # --- beta‑type (polynomial) -------------------------------------------------
    beta_vals = [0.0, 0.5, 1.0, 1.5, 2.0, 3.0, 4.0]
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 60:
                break
            beta_seq = ( (idx + 1) ** p ) * ( (n - idx) ** q )
            if beta_seq.sum() == 0:
                continue
            beta_seq /= beta_seq.sum()
            cur = _objective(beta_seq, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p} q={q} score={cur:.6f}")

    # --- two‑ended exponential (mass at both sides) ---------------------------
    for _ in range(40):
        if time.time() >= deadline - 60:
            break
        r1 = rng.uniform(0.90, 0.9995)
        r2 = rng.uniform(0.90, 0.9995)
        w = rng.random()
        left = r1 ** idx
        right = r2 ** (n - 1 - idx)
        seq = w * left + (1.0 - w) * right
        if seq.sum() == 0:
            continue
        seq /= seq.sum()
        cur = _objective(seq, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Two‑ended r1={r1:.4f} r2={r2:.4f} w={w:.2f} score={cur:.6f}")

    # --- a few random Dirichlet draws (spiky candidates) -----------------------
    for _ in range(6):
        if time.time() >= deadline - 60:
            break
        # concentration < 1 yields spikier vectors
        conc = rng.uniform(0.2, 0.8)
        dir_seq = rng.dirichlet(np.full(n, conc))
        cur = _objective(dir_seq, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = dir_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet conc={conc:.2f} score={cur:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Bulk Adam optimisation (major part of the remaining budget)
    # --------------------------------------------------------------
    # Reserve ~150 s for the final hill‑climb
    hill_time = 150.0
    adam_deadline = deadline - hill_time
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        a_opt, sc_opt = _adam_optimize(best_seq, start_time, adam_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score = {sc_opt:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final hill‑climb (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score = {sc_hc:.6f}")

    # --------------------------------------------------------------
    # Normalise and emit the result
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
