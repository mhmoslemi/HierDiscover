def _resize_sequence(seq, new_n):
    seq = np.asarray(seq, dtype=float)
    if seq.ndim != 1:
        return seq
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    # use linear interpolation over indices scaled to [0,1]
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0:
        new_seq = new_seq / s
    else:
        new_seq = np.full(new_n, 1.0 / new_n)
    return new_seq
```

def _adam_softmax_optimize(a0, deadline, start_time,
                           lr=0.15, beta1=0.9, beta2=0.999, eps=1e-8,
                           alpha_start=500.0, alpha_max=20000.0,
                           alpha_factor=1.001):
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    alpha = alpha_start

    while time.time() < deadline:
        it += 1
        # forward FFT of a
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # compute softmax weights
        shifted = conv - max_conv
        # compute exponent with current alpha
        # to avoid overflow, we may clip alpha * shifted > 700 etc.
        exp_arg = alpha * shifted
        # Clip to avoid overflow warnings maybe
        exp_arg = np.clip(exp_arg, -700, 0)  # max 0
        w_exp = np.exp(exp_arg)
        sum_exp = w_exp.sum()
        if sum_exp == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_exp / sum_exp

        # gradient via convolution of w with reversed a
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # scale gradient to match objective scaling
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto simplex
        a = _project_simplex(a)

        # increase alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)
    return best_a, best_score
```

"""
Propose a low‑objective non‑negative step function.

Algorithm
---------
1. **Warm‑start** – generate a few analytic shapes (tent, cosine,
   half‑circle, Gaussian, power‑tent) and optionally resample the
   globally stored `height_sequence_1`.  The best of them (according to
   the exact objective) is the start point.

2. **Soft‑max Adam optimisation** – minimise a smooth surrogate of the
   true objective 2·n·max(conv(a,a)).  The surrogate is
        f(a) = (1/α)·log( Σ_i exp(α·(a*a)[i]) )
   whose gradient is
        ∇f = 2·conv(w , a_rev)                     (1)
   where  w_i = exp(α·c_i) / Σ_j exp(α·c_j)  and  c = a*a.
   The gradient is obtained by a single FFT‑based convolution.
   We start with a small α (smooth surrogate) and increase it
   continuously (continuation method) while using Adam on the simplex.

3. **Light simulated‑annealing polish** – a Metropolis walk that moves a
   random packet of mass between two indices.  It runs on the remaining
   time budget to escape possible shallow minima.

All heavy work uses NumPy’s FFT, which is fast for powers‑of‑two.
The whole search respects a 1000 s budget and returns a plain
``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised (Σa=1) vector."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform simplex point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a few analytically simple candidate vectors."""
    x = np.arange(n, dtype=float) / (n - 1)               # normalised coordinate in [0,1]

    # symmetric linear tent
    tent = np.minimum(x, 1.0 - x)

    # raised cosine (half‑sine)
    cosine = np.sin(np.pi * x)

    # half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))

    # centred Gaussian
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

    # power‑tent (flatter than linear)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]
    warm = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            warm.append(c / s)
    return warm


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a 1‑D sequence to a new length.
    The result is normalised to sum to 1.
    """
    seq = np.asarray(seq, dtype=float)
    if seq.ndim != 1:
        raise ValueError("sequence must be 1‑dimensional")
    old_n = seq.size
    if old_n == new_n:
        out = seq.copy()
    else:
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, new_n)
        out = np.interp(new_x, old_x, seq)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s > 0.0:
        out = out / s
    else:
        out = np.full(new_n, 1.0 / new_n)
    return out


# ----------------------------------------------------------------------
# Soft‑max Adam optimiser
# ----------------------------------------------------------------------
def _adam_softmax_optimize(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           lr: float = 0.15,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8,
                           alpha_start: float = 500.0,
                           alpha_max: float = 20000.0,
                           alpha_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex using the gradient of the
    soft‑max surrogate of the maximum convolution entry.
    Returns the best iterate and its true objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)                # large enough for all convolutions

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    alpha = alpha_start

    while time.time() < deadline:
        it += 1

        # ----- self‑convolution & current objective -----
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # ----- soft‑max weights over convolution entries -----
        shifted = conv - max_conv                     # ≤ 0
        # protect against overflow / underflow
        exp_arg = alpha * shifted
        # clipping prevents NaNs for very large negative arguments
        exp_arg = np.clip(exp_arg, -700, 0)           # exp(-700) ≈ 5e-305
        w_exp = np.exp(exp_arg)
        sum_exp = w_exp.sum()
        if sum_exp == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_exp / sum_exp

        # ----- gradient of the surrogate -----
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        # extract the n entries that correspond to the gradient
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Scale to the true objective (2·n·max) – same scaling used in the reference code
        grad_obj = 2.0 * n * grad

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ----- projection onto the simplex -----
        a = _project_simplex(a)

        # ----- continuation on α -----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk that moves a random packet of mass between two
    indices.  Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 80_000            # more steps than the reference code
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling – very fast
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        # choose a source proportionally to its mass (helps move large piles)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 25 %) of the source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – power of two gives fast FFTs.
    # --------------------------------------------------------------
    n = 16384

    # --------------------------------------------------------------
    # 1) Warm‑start – evaluate a few analytic shapes.
    # --------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # include the global starter if it exists
    try:
        global height_sequence_1
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            hs = _resize_sequence(hs, n)
            warm_starts.append(hs)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Main optimisation – soft‑max Adam on the simplex.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                     # keep a few seconds for polishing
        adam_deadline = time.time() + remaining * 0.85
        a_opt, score_opt = _adam_softmax_optimize(a0, adam_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam best={score_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing polish (if time permits).
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(a0, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            a0 = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal best={anneal_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation & conversion to plain list.
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 = a0 / total
    else:
        a0 = np.full(n, 1.0 / n)

    return [float(v) for v in a0]
