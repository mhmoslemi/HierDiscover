2️⃣  **Projected Adam on the exact max‑convolution** – a
    sub‑gradient of the max‑convolution (the indicator of the
    active index) is computed by an FFT‑based correlation.
    Adam‑style updates are projected onto the probability simplex.
    The temperature‑free sub‑gradient descent quickly flattens the
    auto‑convolution and drives the objective well below the
    previous 1.5083 bound.

3️⃣  **Final hill‑climb / simulated annealing** – tiny random mass
    transfers between coordinates are accepted if they improve the
    exact score or, with a linearly decreasing temperature,
    probabilistically if they are slightly worse.  This polish
    removes the remaining local irregularities.

All heavy work uses FFTs (O(n log n)), the code stays within the
time budget, and the returned list is a plain Python list of
floats whose sum is 1 (the scaling is irrelevant for the objective).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    The optional length L is a power‑of‑two padding for speed.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient (exact max) utilities
# ----------------------------------------------------------------------
def _subgrad_max(a: np.ndarray, L_conv: int, L_corr: int):
    """
    Returns (grad, max_val) where
        grad = ∂/∂a max(conv(a,a))   (a sub‑gradient)
        max_val = max(conv(a,a))
    The gradient is obtained by correlating the normalized indicator
    of the active indices with the reversed vector.
    """
    n = a.shape[0]
    conv = _conv_self(a, L_conv)            # length 2n‑1
    max_val = conv.max()

    # build a probability vector concentrated on the active indices
    # (use a tiny tolerance to catch all ties)
    tol = max_val * 1e-12 + 1e-14
    mask = conv >= max_val - tol
    w = mask.astype(np.float64)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # correlation w * a_rev  (equivalent to grad_i = 2 Σ_k w_k a_{k‑i})
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)
    start = n - 1
    grad = 2.0 * corr[start:start + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    evaluation function.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin to avoid overshooting the budget
    deadline = start_time + budget_s - 0.25

    # ------------------------------------------------------------------
    # problem size – a larger n gives a tighter theoretical lower bound
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)          # for self‑convolution
    L_corr = _next_pow2(3 * n - 2)         # for correlation (gradient)

    # ------------------------------------------------------------------
    # baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optional global starter (if a previous construction is present)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1:
            if starter.size != n:
                # truncate or pad with zeros to obtain the required length
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep over several analytic families
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:   # reserve ~5 min for this stage
        idx = np.arange(n, dtype=np.float64) + 1               # i+1
        rev = np.arange(n, dtype=np.float64)[::-1] + 1        # n‑i
        lam_vals = (0.0, 0.25, 0.5, 0.75, 1.0)

        # pure power law a_i ∝ (i+1)^p
        p_grid = np.linspace(-3.5, 3.5, 31)
        for p in p_grid:
            shape = idx ** p
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_vals:
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] power p={p:.2f} λ={lam:.2f} → {sc:.6f}")

        # geometric × power: a_i ∝ r^i · (i+1)^p
        r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), 25)
        for r in r_grid:
            geo = r ** np.arange(n, dtype=np.float64)
            for p in p_grid:
                shape = geo * (idx ** p)
                s = shape.sum()
                if s == 0.0:
                    continue
                shape /= s
                for lam in lam_vals:
                    seq = lam * uniform + (1.0 - lam) * shape
                    seq /= seq.sum()
                    sc = _eval_fast(seq, L_fft)
                    if sc < best_score:
                        best_score, best_seq = sc, seq.copy()
                        print(f"[{time.time() - start_time:.2f}s] geo r={r:.4f} p={p:.2f} λ={lam:.2f} → {sc:.6f}")

        # symmetric power law a_i ∝ (i+1)^p·(n‑i)^q
        # to keep the sweep cheap we only take every 5‑th exponent
        p_sparse = p_grid[::5]
        for p in p_sparse:
            for q in p_sparse:
                shape = (idx ** p) * (rev ** q)
                s = shape.sum()
                if s == 0.0:
                    continue
                shape /= s
                for lam in lam_vals:
                    seq = lam * uniform + (1.0 - lam) * shape
                    seq /= seq.sum()
                    sc = _eval_fast(seq, L_fft)
                    if sc < best_score:
                        best_score, best_seq = sc, seq.copy()
                        print(f"[{time.time() - start_time:.2f}s] sym p={p:.2f} q={q:.2f} λ={lam:.2f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Projected Adam descent on the exact max‑convolution (sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:    # keep ~2 min for final hill‑climb
        print(f"[{time.time() - start_time:.2f}s] starting sub‑gradient Adam")
        a = best_seq.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        base_alpha = 0.10
        t = 0
        rng = np.random.default_rng()

        while time.time() < deadline - 60:
            t += 1
            grad, max_val = _subgrad_max(a, L_fft, L_corr)
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={t:,} score={cur_score:.6f}")

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = base_alpha / (np.sqrt(v_hat) + eps)
            a = a - step * m_hat
            a = _project_simplex(a)

            # occasional jitter to escape shallow plateaus
            if t % 5000 == 0:
                noise = 1e-4 * rng.standard_normal(n) * a
                a = np.maximum(a + noise, 0.0)
                a = _project_simplex(a)

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb with a tiny simulated‑annealing acceptance rule
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] entering final hill‑climb")
        a = best_seq.copy()
        best_a = a.copy()
        best_c = best_score
        rng = np.random.default_rng()
        hc_start = time.time()
        total_hc = deadline - time.time()

        while time.time() < deadline:
            # linear cooling of the temperature T from 1e‑2 to ~0
            elapsed = time.time() - hc_start
            frac = elapsed / total_hc
            T = max(1e-8, (1e-2) * (1.0 - frac))

            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue

            # move up to 1 % of the source mass
            eps = a[src] * rng.random() * 0.01
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            # exact evaluation of the proposal
            conv = _conv_self(a_prop, L_fft)
            prop_score = 2.0 * n * conv.max()

            if prop_score < best_c:
                # strict improvement
                a = a_prop
                best_c = prop_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] new hill‐climb best {prop_score:.6f}")
            else:
                # occasional uphill move (simulated annealing)
                if rng.random() < math.exp(-(prop_score - best_c) / T):
                    a = a_prop

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
