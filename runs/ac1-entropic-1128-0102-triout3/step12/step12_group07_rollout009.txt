"""
A fast heuristic for the autocorrelation‐peak minimisation problem.

The algorithm proceeds in three stages:

1. **Parameter sweep** – we generate a few hundred simple families of
   step functions (arcsine‑type and asymmetric Beta‑type) and keep the
   best normalised vector seen.  This gives a good starting shape.

2. **Hill‑climbing with mass transfers** – for the current vector `p`
   (∑p = 1) we locate the shift `k` with the largest autocorrelation.
   The index that contributes most to this shift is taken as the *source*.
   A *destination* index with low probability *and* low contribution to
   the same shift is chosen, and a small amount of mass `ε` is moved from
   source to destination.  The self‑convolution is updated analytically
   (O(n) work) and the move is accepted if the peak decreases, otherwise
   it is rejected.  This is essentially the strategy from the previous
   state‑of‑the‑art, but the destination choice is refined and simulated
   annealing is added to escape shallow plateaus.

3. **Plateau handling** – if no improvement is seen for many
   iterations we jitter the best vector slightly and continue.

The whole search runs for the allotted time (≈1000 s) and returns the
best probability vector found; the verifier is scale‑invariant so a
normalised vector is sufficient.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT based convolution (used for occasional full recomputation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).

    The search combines a cheap parametric sweep, a refined hill‑climbing
    mass‑transfer loop and simulated‑annealing acceptance to improve on
    the previously reported bound (≈1.5086 → target < 1.503).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safe margin

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1) Pick a base length.  4096 works well with the time budget.
    # ------------------------------------------------------------------
    base_n = 4096

    # ------------------------------------------------------------------
    # 2) Gather a first candidate – either a known construction or a
    #    parameter sweep over simple families.
    # ------------------------------------------------------------------
    best_p = None
    best_score = float("inf")
    n = base_n

    # try to use the previously found best construction if available
    try:
        cand = np.asarray(height_sequence_1, dtype=float)
        if cand.ndim == 1 and cand.size > 0:
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand = cand / s
                n = cand.size
                conv = _conv_self(cand)
                score = 2.0 * n * conv.max()
                if score < best_score:
                    best_score = score
                    best_p = cand.copy()
                    print(f"[{time.time()-start_time:.2f}s] "
                          f"Seeded from height_sequence_1 (n={n}) "
                          f"score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) Parameter sweep (≈5 % of the total budget, capped at 30 s)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + min(30.0, budget_s * 0.05)
    while time.time() < sweep_deadline:
        shape_type = rng.integers(0, 2)   # 0 – arcsine‑type, 1 – beta‑type
        x = (np.arange(base_n, dtype=np.float64) + 0.5) / base_n

        if shape_type == 0:
            # arcsine‑type: w = (x(1‑x)+eps)^(-alpha)
            alpha = 0.1 + rng.random() * 0.6           # 0.1 – 0.7
            eps = 1e-9
            w = (x * (1.0 - x) + eps) ** (-alpha)
        else:
            # asymmetric beta‑type: w = x^{a‑1} (1‑x)^{b‑1}
            a = 0.1 + rng.random() * 0.9              # 0.1 – 1.0
            b = 0.1 + rng.random() * 0.9
            eps = 1e-12
            w = (np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0) + eps)

        p = w / w.sum()
        conv = _conv_self(p)
        score = 2.0 * base_n * conv.max()
        if score < best_score:
            best_score = score
            best_p = p.copy()
            print(f"[{time.time()-start_time:.2f}s] sweep new best "
                  f"score={score:.6f} shape={shape_type}")

    # If everything failed, fall back to a simple arcsine start.
    if best_p is None:
        x = (np.arange(base_n, dtype=np.float64) + 0.5) / base_n
        alpha = 0.38
        eps = 1e-9
        w = (x * (1.0 - x) + eps) ** (-alpha)
        best_p = w / w.sum()
        n = base_n
        best_score = 2.0 * n * _conv_self(best_p).max()
        print(f"[{time.time()-start_time:.2f}s] fallback arcsine start "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Initialise data structures for the hill‑climbing loop.
    # ------------------------------------------------------------------
    p = best_p.copy()
    n = p.shape[0]                     # honour the length we actually use
    conv = _conv_self(p)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max
    best_seq = p.copy()
    iteration = 0
    last_improve_iter = 0

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- locate the worst shift -----
        k = int(conv.argmax())
        cur_max = conv[k]

        # ----- indices that influence shift k -----
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = p[idx] * p[k - idx]

        # ----- source : biggest contributor to the dominant shift -----
        src = idx[int(np.argmax(contrib))]

        # ----- destination : low probability & low contribution -----
        # full contribution array (zero outside the active range)
        full_contrib = np.zeros(n, dtype=np.float64)
        full_contrib[idx] = contrib
        max_c = full_contrib.max()
        # weight = (1‑p) * (max_contrib – contribution)   (prefer low p, low contribution)
        dst_weights = (1.0 - p) * (max_c - full_contrib + 1e-12)
        total_w = dst_weights.sum()
        if total_w <= 0 or np.isnan(total_w):
            dst_weights = np.ones_like(p)
            total_w = dst_weights.sum()
        prob = dst_weights / total_w
        dst = rng.choice(n, p=prob)
        if dst == src:
            continue

        # ----- step size (fraction of the source mass) -----
        elapsed = time.time() - start_time
        max_frac = max(0.01, 0.20 * (1.0 - elapsed / budget_s))
        epsilon = p[src] * rng.random() * max_frac
        if epsilon <= 0.0:
            continue

        # ----- backup current state (cheap copies) -----
        p_old = p.copy()
        conv_old = conv.copy()

        # ----- apply the exact convolution update -----
        conv[src:src + n] -= 2.0 * epsilon * p
        conv[dst:dst + n] += 2.0 * epsilon * p
        conv[2 * src] += epsilon * epsilon
        conv[2 * dst] += epsilon * epsilon
        conv[src + dst] -= 2.0 * epsilon * epsilon

        p[src] -= epsilon
        p[dst] += epsilon
        if p[src] < 0.0:
            p[src] = 0.0

        new_max = conv.max()

        # ----- simulated‑annealing acceptance -----
        accept = False
        if new_max < cur_max - 1e-16:
            accept = True
        else:
            # temperature proportional to the current peak, linearly cooling
            T = max(1e-8, cur_max * 0.5 * (1.0 - elapsed / budget_s))
            if T > 0.0:
                prob_accept = math.exp(-(new_max - cur_max) / T)
                if rng.random() < prob_accept:
                    accept = True

        if accept:
            cur_max = new_max
            if 2.0 * n * cur_max < best_score - 1e-12:
                best_score = 2.0 * n * cur_max
                best_seq = p.copy()
                last_improve_iter = iteration
                print(f"[{time.time()-start_time:.2f}s] iter={iteration:,} "
                      f"new best score={best_score:.6f}")
        else:
            # revert
            p = p_old
            conv = conv_old

        # ----- occasional full recomputation to curb drift -----
        if iteration % 5000 == 0:
            s = p.sum()
            if abs(s - 1.0) > 1e-12:
                p /= s
            conv = _conv_self(p)

        # ----- plateau handling -----
        if iteration - last_improve_iter > 200_000:
            # jitter the best sequence and restart from there
            p = best_seq.copy()
            jitter = 0.005 * p * (rng.random(p.shape) - 0.5)
            p = np.maximum(p + jitter, 0.0)
            p /= p.sum()
            conv = _conv_self(p)
            last_improve_iter = iteration
            print(f"[{time.time()-start_time:.2f}s] plateau – re‑jittered at iter {iteration}")

    # ------------------------------------------------------------------
    # 6) Return the best normalised vector encountered.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
