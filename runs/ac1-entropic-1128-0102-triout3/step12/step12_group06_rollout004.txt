def _smooth_descent_adam(a, deadline, start_time):
    best_a = a.copy()
    best_score = _exact_score(a)
    it = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.1
    # schedule of alphas
    alphas = np.geomspace(0.5, 5000.0, num=60)
    for alpha in alphas:
        if time.time() >= deadline - 0.5:
            break
        while time.time() < deadline - 0.5:
            it += 1
            # compute conv
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2*n-1]
            # compute softmax weights
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # gradient (size n)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=_L_GRAD)
            Revf = np.fft.rfft(a_rev, n=_L_GRAD)
            grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[: 3*n-2]
            grad = 2.0 * grad_full[n-1:n-1+n]   # size n
            # scale grad for objective L = 2*n*f
            grad = 2.0 * n * grad
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            step = base_lr / (np.sqrt(v_hat) + eps)
            a = a - step * m_hat
            # project onto simplex
            a = _project_simplex(a)
            # periodic evaluation
            if it % 3000 == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.1f} it={it:,} score={cur_score:.6f}")
    return best_a, best_score

```
def _project_simplex(v):
    # ensure nonnegative by projection onto simplex sum=1
    # Implementation of algorithm from Wang & Carreira-Perpiñán.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Multi‑restart Adam optimizer with final simulated‑annealing polish.
The algorithm works on the log‑sum‑exp smooth approximation of the
max‑convolution, uses Adam on the simplex (via projection) and
periodically restarts from random Dirichlet draws to escape local
minima.  After the smooth phase a short max‑subgradient run and a
stochastic hill‑climb finish the search.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global placeholders – will be set in propose_candidate()
# ----------------------------------------------------------------------
n = None               # length of the step function
_L_SELF = None         # FFT length for self‑convolution  (≥2n‑1)
_L_GRAD = None         # FFT length for gradient convolution (≥3n‑2)

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _set_globals(N: int):
    """Initialise the global FFT sizes for a given problem dimension."""
    global n, _L_SELF, _L_GRAD
    n = N
    _L_SELF = 1 << ((2 * n - 2).bit_length())          # next power of two ≥2n‑1
    _L_GRAD = 1 << ((3 * n - 3).bit_length())          # next power of two ≥3n‑2


def _exact_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective 2·n·max(conv(a,a))/sum(a)^2."""
    # linear (full) convolution via FFT
    Af = np.fft.rfft(a, n=_L_SELF)
    conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Log‑sum‑exp (smooth max) weights for a given smoothing parameter."""
    exp_arg = alpha * conv
    max_exp = exp_arg.max()
    w = np.exp(exp_arg - max_exp)          # numerical stability
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    return w


def _gradient_from_weights(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of  (1/α)·log∑exp(α·conv)  w.r.t. a.
    Returns a vector of length n.
    """
    # convolution w ⋆ a_rev (full length 3n‑2)
    Wf = np.fft.rfft(w, n=_L_GRAD)
    Revf = np.fft.rfft(a_rev, n=_L_GRAD)
    grad_full = np.fft.irfft(Wf * Revf, n=_L_GRAD)[: 3 * n - 2]
    # extract the central n elements (indices n‑1 … 2n‑2)
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Phase 0 – quick random‑restart max‑subgradient (a few thousand iters)
# ----------------------------------------------------------------------
def _quick_subgrad(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent for a short budget."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _exact_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # self‑convolution and arg‑max
        Af = np.fft.rfft(a, n=_L_SELF)
        conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]                # g_i = 2·a_{k‑i}

        eta = 0.5 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        if it % 2000 == 0:
            cur = _exact_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max Adam descent (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_adam(a0: np.ndarray, deadline: float, start_time: float):
    """Adam on the smooth‑max approximation with a geometric α‑schedule."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _exact_score(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    base_lr = 0.08                     # base learning‑rate for Adam

    # geometric schedule of smoothing parameters
    alphas = np.geomspace(0.5, 4000.0, num=56)

    for alpha in alphas:
        if time.time() >= deadline - 0.3:
            break
        # inner loop: as many Adam steps as time permits for this α
        while time.time() < deadline - 0.3:
            it += 1

            # 1) self‑convolution
            Af = np.fft.rfft(a, n=_L_SELF)
            conv = np.fft.irfft(Af * Af, n=_L_SELF)[: 2 * n - 1]

            # 2) soft‑max weights (smooth max)
            w = _softmax_weights(conv, alpha)

            # 3) gradient of smooth‑max term
            grad = _gradient_from_weights(w, a[::-1])                # size n
            # overall objective includes factor 2·n
            grad = 2.0 * n * grad

            # 4) Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = base_lr / (np.sqrt(v_hat) + eps)
            a = a - step * m_hat

            # 5) projection back onto the simplex
            a = _project_simplex(a)

            # occasional exact evaluation
            if it % 4000 == 0:        # about once per few seconds
                cur = _exact_score(a)
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.1f} it={it:,} score={cur:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – final simulated‑annealing hill climb
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, deadline: float, start_time: float):
    """Metropolis hill‑climb with a slowly cooling temperature."""
    cur = a0.copy()
    cur_score = _exact_score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    start_temp = 5e-2
    end_temp = 1e-10

    while time.time() < deadline:
        it += 1
        # linear temperature decay
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a random fraction (up to 30 %) of the source mass
        eps = cur[src] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum > 0.0:
            cand /= cand_sum
        else:
            continue

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} best={best_score:.6f}")

    return best_seq, best_score

    # Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
