def _gradient_smooth(a, w):
    # a shape (n,)
    # w shape (2n-1,)
    # Compute grad as 2 * cross-correlation of a and w.
    # Equivalent to conv of a and reversed w.
    rev_w = w[::-1]
    # Convolution length: 3n-2
    L_fft = _next_pow2(len(a) + len(rev_w) - 1)
    Af = np.fft.rfft(a, n=L_fft)
    Wf = np.fft.rfft(rev_w, n=L_fft)
    conv = np.fft.irfft(Af * Wf, n=L_fft)
    # Take slice: indices (len(w)-1 - np.arange(n)) gives grad array.
    idx_start = len(w) - 1 - (n - 1)  # i.e., len(w)-n
    grad = 2.0 * conv[idx_start: idx_start + n]
    return grad

def _smooth_grad(a, w):
    n = len(a)
    rev_w = w[::-1]
    L = _next_pow2(len(a) + len(rev_w) - 1)
    Af = np.fft.rfft(a, n=L)

    start = len(w) - 1 - (n - 1)  # = len(w)-n

def _adam_optimize_smooth(a0, start_time, deadline, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = len(a)
    L_fft = _next_pow2(2 * n - 1)  # for conv of a with a

    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it=0
    last_improve = 0
    rng = np.random.default_rng()
    beta = 10.0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Smooth Adam it={it} score={cur_score:.6f} beta={beta:.1f}")

        # compute smooth weights
        # to avoid overflow
        maxc = conv.max()  # used for shifting
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()

        # gradient
        grad = _smooth_grad(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_simplex(a)

        # increase beta gradually
        if it % 2000 == 0:
            beta = min(beta * 1.5, 2000)  # cap

        # plateau heuristic
        if it - last_improve > 10000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            last_improve = it
            # reset beta maybe lower
            beta = max(beta * 0.5, 10.0)

    return best_a, best_score

"""
Propose a step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

by

1️⃣  a fast random search over two‑term exponential mixtures
    (much richer than the simple geometric family);
2️⃣  a smooth‑max Adam optimiser: we minimise the
    (1/β)·log‑sum‑exp(β·conv) which is a differentiable
    approximation of the hard maximum.  The temperature β is
    increased gradually, the gradient is obtained by a single
    FFT‑based cross‑correlation, and we always project onto the
    probability simplex.
3️⃣  a final stochastic hill‑climb that transfers a modest random
    fraction of mass between two positions and accepts only
    improving moves.

The total running time is limited to the given budget (default
1000 s).  The routine returns a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) of the convolution
# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   f(a)=∑_k w_k * (a*a)[k]   where w sums to 1.
    It equals 2 * cross‑correlation of a with w.
    """
    n = a.shape[0]
    # reverse w for a cross‑correlation
    rev_w = w[::-1]
    L = _next_pow2(len(a) + len(rev_w) - 1)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(rev_w, n=L)
    conv = np.fft.irfft(Af * Wf, n=L)          # length = len(a)+len(rev_w)-1 = 3n-2
    # slice the central n entries (indices n‑1 … 2n‑2) and reverse
    slice_center = conv[n - 1 : n - 1 + n][::-1]
    return 2.0 * slice_center


# ----------------------------------------------------------------------
# Adam optimiser that minimises a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_optimize_smooth(a0: np.ndarray,
                          start_time: float,
                          deadline: float,
                          best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Adam on the soft‑max objective.
    Returns the best sequence seen and its true (hard‑max) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    beta = 5.0                # temperature for the soft‑max
    beta_inc = 1.10           # multiplicative increase every few thousand iters
    beta_max = 200.0

    while time.time() < deadline:
        it += 1

        # true objective (hard max)
        conv = _conv_self(a, L_conv)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam (smooth) it={it:,} "
                  f"score={cur_score:.6f} β={beta:.1f}")

        # --- smooth‑max gradient ------------------------------------------------
        # soft‑max weights (shifted for numerical stability)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:               # should not happen
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = _smooth_grad(a, w)

        # --- Adam update --------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # keep on the simplex (sum = 1, non‑negative)
        a = _project_simplex(a)

        # anneal β (make soft‑max sharper)
        if it % 3000 == 0:
            beta = min(beta * beta_inc, beta_max)

        # restart if stuck for a long time
        if it - last_improve > 12000:
            # tiny random perturbation to escape a flat region
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            last_improve = it
            beta = max(beta * 0.5, 5.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (micro‑mass transfers, slightly larger steps)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a random fraction (up to 5 %) of mass
    from one index to another and keeps only improving moves.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    L_fft = _next_pow2(2 * n - 1)
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer up to 5 % of the source mass
        frac = rng.random() * 0.05 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2).

    The search consists of:
        * a quick random parametric sweep over two‑term exponential mixtures;
        * a smooth‑max Adam optimiser with a temperature schedule;
        * a final hill‑climb.
    The routine respects the given time budget and returns a list of
    Python floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.2

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # baseline: uniform
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = float(2.0 * n * _conv_self(best_seq, L_fft).max())
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # initial deterministic guess from analytic epsilon ≈ score·e/(2n)
    # --------------------------------------------------------------
    eps_analytic = best_score * np.e / (2.0 * n)          # ε = score·e/(2n)
    r = np.exp(-eps_analytic)                           # r ≈ 1‑ε for small ε
    idx = np.arange(n, dtype=np.float64)
    geo = np.power(r, idx)
    geo_sum = geo.sum()
    if geo_sum > 0:
        geo = geo / geo_sum
        cur = float(2.0 * n * _conv_self(geo, L_fft).max())
        if cur < best_score:
            best_score = cur
            best_seq = geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Analytic geo eps={eps_analytic:.2e} score={cur:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Random sweep over two‑term exponential mixtures
    # --------------------------------------------------------------
    num_trials = 2500
    for _ in range(num_trials):
        if time.time() >= deadline - 200:               # reserve time for optimisation phases
            break
        # sample epsilons log‑uniformly in [1e-6, 1e-2]
        eps1 = 10 ** (-np.random.uniform(5, 1))          # 1e-5 … 1e-1, but we shift to smaller later
        eps2 = 10 ** (-np.random.uniform(5, 1))
        r1 = np.exp(-eps1)
        r2 = np.exp(-eps2)
        # enforce r1 ≥ r2 (slow decay component first)
        if r1 < r2:
            r1, r2 = r2, r1
        lam = np.random.rand()
        # build the two‑term mixture (each component already normalised)
        comp1 = (1 - r1) * np.power(r1, idx) / (1 - r1 ** n)
        comp2 = (1 - r2) * np.power(r2, idx) / (1 - r2 ** n)
        seq = lam * comp1 + (1.0 - lam) * comp2
        # numerical safety
        seq = np.maximum(seq, 0.0)
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq = seq / seq_sum
        cur = float(2.0 * n * _conv_self(seq, L_fft).max())
        if cur < best_score:
            best_score = cur
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r1={r1:.5f} r2={r2:.5f} λ={lam:.3f} score={cur:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam optimisation (bulk of the budget)
    # --------------------------------------------------------------
    hill_time = 150.0                                 # seconds we keep for final hill‑climb
    adam_deadline = deadline - hill_time
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max Adam optimiser")
        a_opt, sc_opt = _adam_optimize_smooth(best_seq, start_time, adam_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
