def propose_candidate(seed=0, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4096

    # generate basis shapes
    bases = []
    bases.append(_shape_uniform(n))
    bases.append(_shape_tent(n))
    # arcsine
    bases.append(_shape_arcsine_like(n))
    # power centre for p=2,4
    for p in (1.0, 2.0, 4.0):
        bases.append(_shape_power_center(n, p))
    for p in (1.0, 2.0, 4.0):
        bases.append(_shape_power_edge(n, p))
    # exponential
    bases.append(_shape_exponential(n, 0.001))
    bases.append(_shape_gaussian(n, n/8.0))

    # also precompute beta-like shapes for set of exponent pairs
    exponents = np.linspace(-0.5, 0.0, 11) # 11 values
    for aexp in exponents:
        # symmetric
        shape = ((np.arange(n, dtype=float)+0.5) * (n-0.5 - np.arange(n, dtype=float)))**aexp
        shape = np.maximum(shape, 0)
        shape = shape / shape.sum()
        bases.append(shape)
    # asymmetric
    for aexp in exponents:
        for bexp in exponents:
            shape = ((np.arange(n, dtype=float)+0.5)**aexp) * ((n-0.5 - np.arange(n, dtype=float))**bexp)
            shape = np.maximum(shape, 0)
            # ensure non-zero sum
            if shape.sum() > 0:
                shape = shape / shape.sum()
                bases.append(shape)

    # Evaluate each basis, keep best
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(bases):
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # optional print progress

        if time.time() >= deadline:
            break

    # Random mixture search
    m = len(bases)
    # Precompute basis array as (m, n)
    B = np.stack(bases, axis=0)

    # random mixture loops
    # Determine number of iterations based on time
    while time.time() < deadline:
        # generate random weights using Dirichlet
        w = rng.dirichlet(np.ones(m))
        a = np.dot(w, B)  # shape (n,)
        a = a / a.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time()-start:.2f}s] Mix iter score={score:.6f}")

        # maybe sometimes also generate random beta shape directly
        # could generate additional random exponents
        if time.time() >= deadline:
            break

    # Hill climbing / annealing
    a = best_seq.copy()
    cur_score = best_score
    it = 0
    while time.time() < deadline:
        it += 1
        # choose src weighted by a
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue
        # step size schedule: proportion of src mass to move
        # Use step factor decreasing over time
        remaining = deadline - time.time()
        # we can compute a factor that goes from 0.3 to 0.001
        max_factor = 0.4
        min_factor = 0.001
        factor = max_factor * (remaining / (budget_s)) + min_factor
        eps = a[src] * rng.random() * factor
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # optional smoothing: enforce non-neg and small eps?
        new_score = _objective(a_prop)
        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time()-start:.2f}s] Hill iter {it} improve={new_score:.6f}")

    # final normalization
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]

"""
Heuristic optimizer for the self‑convolution “max‑ratio’’ problem.

The algorithm works in three stages:

1. **Parametric warm‑starts** – a library of analytic shapes is built.
   In addition we explore the 2‑parameter beta family
        a_i ∝ (i+½)^{α}·(n‑½‑i)^{β} ,   α,β∈(‑0.9,‑0.05)
   (the continuous arcsine law corresponds to α=β=‑0.5).  All candidates are
   normalised to sum 1 and the best one is kept.

2. **Random convex mixtures** – the library from (1) is mixed with random
   Dirichlet weights.  Each mixture is evaluated; the best mixture replaces
   the current best solution.  This quickly discovers useful linear
   combinations of the analytic shapes.

3. **Local mass‑transfer hill‑climbing** – while time permits we repeatedly
   move a small random packet of mass from one index to another (the packet
   size shrinks over time).  If the move reduces the objective we keep it,
   otherwise we reject it.  This fine‑tunes the distribution obtained in
   steps 1‑2.

All steps respect the time budget (default 1000 s).  The returned list
contains non‑negative floats that sum to 1 and can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Mirrors ``evaluate_sequence`` but without defensive checks.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic shape generators (all return a probability vector)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i, n - 1 - i)
    return a / a.sum()


def _shape_power_center(n: int, p: float) -> np.ndarray:
    """(triangle)^{p} – centre‑heavy."""
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = tri ** p
    a += 1e-16
    return a / a.sum()


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    """(1‑triangle)^{p} – edge‑heavy."""
    i = np.arange(n, dtype=float)
    tri = np.minimum(i, n - 1 - i) / ((n - 1) / 2)
    a = (1.0 - tri) ** p
    a += 1e-16
    return a / a.sum()


def _shape_exponential(n: int, gamma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-gamma * np.abs(i - centre))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _shape_arcsine_like(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a += 1e-16
    return a / a.sum()


def _shape_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta family:
        a_i ∝ (i+½)^{α}·(n‑½‑i)^{β}
    The parameters may be negative (allowing edge‑heavy shapes) as long as
    they stay >‑1 to keep the integral finite.
    """
    i = np.arange(n, dtype=float)
    left = (i + 0.5) ** alpha
    right = (n - 0.5 - i) ** beta
    a = left * right
    a = np.maximum(a, 1e-16)          # guard against exact zero
    return a / a.sum()


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem size (power of two ⇒ fast FFTs)
    n = 4096

    # ------------------------------------------------------------------
    # 1️⃣  Build a modest library of analytic shapes (the “warm‑starts”).
    # ------------------------------------------------------------------
    basis = [
        _shape_uniform(n),
        _shape_tent(n),
        _shape_arcsine_like(n),
        _shape_power_center(n, 2.0),
        _shape_power_center(n, 4.0),
        _shape_power_edge(n, 2.0),
        _shape_power_edge(n, 4.0),
        _shape_exponential(n, 0.001),
        _shape_gaussian(n, n / 8.0),
    ]

    # Sample the 2‑parameter beta family densely around the arcsine point.
    # α,β ∈ (‑0.9,‑0.05) – the interval is safe (integrable) and contains the
    # optimal continuous law (α=β=‑0.5).  A modest grid is inspected first.
    alphas = np.linspace(-0.9, -0.05, 9)          # 9 values → 81 pairs
    for a in alphas:
        for b in alphas:
            basis.append(_shape_beta(n, a, b))

    # ------------------------------------------------------------------
    # 2️⃣  Evaluate all warm‑starts and keep the best.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(basis, 1):
        if time.time() >= deadline:
            break
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # occasional progress report
        if idx % 20 == 0:
            print(f"[{time.time() - start_time:.2f}s] Warm‑start {idx}/{len(basis)} "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Random convex mixtures of the basis functions.
    #    Each mixture is a Dirichlet‑weighted sum of the pre‑computed bases.
    # ------------------------------------------------------------------
    B = np.stack(basis, axis=0)            # shape (m, n)
    m = B.shape[0]
    while time.time() < deadline:
        # Dirichlet(1,…,1) gives a uniform random point on the simplex of weights.
        w = rng.dirichlet(np.ones(m))
        a = np.dot(w, B)
        a = a / a.sum()                     # re‑normalise (guard against numerical drift)
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix improve score={score:.6f}")

        # Occasionally try a completely random Beta shape (not a mixture)
        if rng.random() < 0.05:
            alpha = rng.uniform(-0.9, -0.05)
            beta  = rng.uniform(-0.9, -0.05)
            a_rand = _shape_beta(n, alpha, beta)
            score = _objective(a_rand)
            if score < best_score:
                best_score = score
                best_seq = a_rand.copy()
                print(f"[{time.time() - start_time:.2f}s] Random β‑shape score={score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Local hill‑climbing (mass‑transfer) polish.
    #    Move a small packet of mass from a source index to a destination
    #    index; accept the move only if it improves the objective.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    it = 0
    while time.time() < deadline:
        it += 1
        # Choose a source index proportionally to the current mass.
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # The packet size shrinks linearly with the remaining time.
        # Early on we allow large swings, later we fine‑tune.
        time_left = deadline - time.time()
        total_time = budget_s - 0.5
        # linearly decrease factor from 0.3 → 0.001
        max_factor = 0.3
        min_factor = 0.001
        factor = min_factor + (max_factor - min_factor) * (time_left / total_time)

        eps = a[src] * rng.random() * factor
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill climb iter={it} "
                      f"new best={new_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final tidy‑up and conversion to plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # sanity check (optional)
    print(f"[{time.time() - start_time:.2f}s] Final score = {best_score:.6f}")

    return [float(v) for v in best_seq]
