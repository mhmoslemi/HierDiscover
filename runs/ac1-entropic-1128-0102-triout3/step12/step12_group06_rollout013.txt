def evaluate_sequence(sequence: list[float]) -> float:

    n = len(sequence)

def propose_candidate(seed: int=0, budget_s: int = 1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # problem size
    n = 4096
    L_self = 1 << ((2 * n - 2).bit_length())
    L_grad = 1 << ((3 * n - 3).bit_length())

    # initialization
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise Exception
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()
    # logits
    eps = 1e-12
    x = np.log(starter + eps)  # initial logits

    # best tracking
    best_a = starter.copy()
    best_score = _exact_score(best_a)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    t_iter = 0

    # alpha schedule
    max_alpha = 4000.0
    alpha = 1.0
    alpha_inc = 1.03  # multiply each 2000 it

    # loop for Adam
    while time.time() < deadline - 200.0:   # leave time for hill climb
        t_iter += 1
        # compute a
        a = np.exp(x - np.max(x))
        a_sum = a.sum()
        a = a / a_sum

        # FFT self-convolution
        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

        # softmax over alpha*conv
        c = alpha * conv
        c_max = np.max(c)
        exp_c = np.exp(c - c_max)
        p = exp_c / exp_c.sum()

        # gradient w.r.t a
        a_rev = a[::-1]
        Pf = np.fft.rfft(p, n=L_grad)
        Af_rev = np.fft.rfft(a_rev, n=L_grad)
        grad_full = np.fft.irfft(Pf * Af_rev, n=L_grad)[: 3 * n - 2]
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # gradient w.r.t logits
        scalar = np.dot(a, grad_a)
        grad_x = a * (grad_a - scalar)

        # Adam step size
        lr = 0.1 / (np.sqrt(t_iter) * max(1.0, np.sqrt(alpha)))
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # update alpha every 2000 it
        if t_iter % 2000 == 0:
            alpha = min(alpha * alpha_inc, max_alpha)
            print(f"[{time.time() - start_time:.2f}s] iter={t_iter} alpha={alpha:.1f}")

        # periodic evaluation
        if t_iter % 5000 == 0:
            # compute exact objective
            cur_score = _exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] new best {best_score:.6f}")

    cur = best_a.copy()
    cur_score = best_score
    best_seq = cur.copy()
    best_seq_score = cur_score
    rng = np.random.default_rng()
    it_hill = 0
    start_temp = 0.2
    end_temp = 1e-9

        progress = (time.time() - start_time) / (deadline - start_time)

        src = rng.choice(n, p=cur)
        if cur[src] < 1e-14:

        # choose dst far from src maybe random
        dst = rng.integers(n-1)
        # ensure not same
        if dst == src:
            continue

        eps_mass = cur[src] * rng.random() * 0.2  # up to 20% move
        if eps_mass <= 0:
            continue

        cand = cur.copy()
        cand[src] -= eps_mass
        cand[dst] += eps_mass
        # keep non-negative
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():

            cur_score = cand_score
            if cand_score < best_seq_score:
                best_seq_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill best={best_seq_score:.6f}")

    # final normalisation
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] final best score ≈ {best_seq_score:.6f}")
    return [float(v) for v in best_seq]

"""
Hybrid Adam‐softmax optimizer + targeted simulated annealing.

* Represent the step heights as a probability vector a (∑a = 1) via a softmax
  of unconstrained logits x.  The objective is
      F(a) = 2·n·max_k (a∗a)[k]
  (exactly the value returned by `evaluate_sequence` after normalisation).

* Approximate the `max` with a smooth soft‑max:
      f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)[k])
  whose gradient w.r.t. a is
      ∇_a f_α = 2·(p⋆a_rev) ,  p_k = softmax(α·(a∗a))[k] .
  Using the chain rule for a = softmax(x) we obtain a gradient
  w.r.t. the logits x, which we optimise with the Adam optimiser.
  The smoothing parameter α is increased geometrically during the run
  (start≈1, end≈4000) – this first “smooth‑max” phase drives the
  solution towards a low maximal convolution.

* After the Adam phase we polish the result with a fast simulated‑annealing
  hill‑climber that moves a small fraction of mass from a high‑weight
  index to a low‑weight one.  The temperature schedule is slow (≈0.2 → 1e‑9)
  so that many local improvements are accepted.

The whole routine respects the 1000 s budget, prints intermediate
scores and finally returns a list of non‑negative heights that
minimise `evaluate_sequence`.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: fast exact evaluation (the same as in the reference code)
# --------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# --------------------------------------------------------------
# Main entry point required by the competition framework
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.

    The optimisation consists of:
      1. Adam on the soft‑max parametrisation with a gradually
         increasing smooth‑max temperature α.
      2. A final simulated‑annealing hill‑climber for polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    # A tiny safety margin – the framework will kill us if we overrun.
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # Problem size (the classic 4096‑point setting)
    # ------------------------------------------------------------------
    n = 4096
    L_self = 1 << ((2 * n - 2).bit_length())      # ≥ 2n‑1
    L_grad = 1 << ((3 * n - 3).bit_length())      # ≥ 3n‑2

    # ------------------------------------------------------------------
    # Initialisation – use the previously found best if available,
    # otherwise fall back to the uniform distribution.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise Exception
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    # adjust length
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    # logits for the soft‑max parametrisation
    eps = 1e-12
    x = np.log(starter + eps)      # ensures a = softmax(x) == starter

    # --------------------------------------------------------------
    # Adam optimiser on the smooth‑max surrogate
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # smoothing parameter schedule
    alpha = 1.0
    max_alpha = 4000.0
    alpha_mul = 1.03          # geometric increase every few thousand it

    best_a = starter.copy()
    best_score = _exact_score(best_a)
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    it = 0
    # keep ~70 % of time for Adam, the rest for hill‑climb
    adam_deadline = deadline - 200.0

    while time.time() < adam_deadline:
        it += 1

        # ---- soft‑max to obtain a on the simplex ----
        # subtract max for numerical stability
        x_shift = x - np.max(x)
        a = np.exp(x_shift)
        a /= a.sum()

        # ---- self‑convolution a*a (full length 2n‑1) ----
        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

        # ---- smooth‑max weights p = softmax(α·conv) ----
        c = alpha * conv
        c_max = np.max(c)
        exp_c = np.exp(c - c_max)
        p = exp_c / exp_c.sum()          # shape (2n‑1,)

        # ---- gradient w.r.t. a : 2·(p ⋆ a_rev) ----
        a_rev = a[::-1]
        Pf = np.fft.rfft(p, n=L_grad)
        Af_rev = np.fft.rfft(a_rev, n=L_grad)
        grad_full = np.fft.irfft(Pf * Af_rev, n=L_grad)[: 3 * n - 2]
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # ---- chain rule: gradient w.r.t. logits x ----
        scalar = np.dot(a, grad_a)               # a·∇_a f
        grad_x = a * (grad_a - scalar)           # ∂f/∂x_j = a_j(∂f/∂a_j - a·∂f/∂a)

        # ---- Adam step ----
        lr = 0.1 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasionally increase α and report progress ----
        if it % 2000 == 0:
            alpha = min(alpha * alpha_mul, max_alpha)
            print(f"[{time.time() - start_time:.2f}s] it={it:,} α={alpha:.1f}")

        # ---- periodic exact evaluation ----
        if it % 5000 == 0:
            cur_score = _exact_score(a)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] new best {best_score:.6f}")

    # final evaluation of the Adam result
    a = np.exp(x - np.max(x))
    a /= a.sum()
    cur_score = _exact_score(a)
    if cur_score < best_score:
        best_score, best_a = cur_score, a.copy()
        print(f"[{time.time() - start_time:.2f}s] post‑Adam best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    cur = best_a.copy()
    cur_score = best_score
    best_seq = cur.copy()
    best_seq_score = cur_score

    rng = np.random.default_rng()
    it_hill = 0
    start_temp = 2e-1                # a bit hotter than the old version
    end_temp = 1e-9

    while time.time() < deadline:
        it_hill += 1

        # linear progress for temperature decay (exponential schedule)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        # biased source: pick an index proportional to its weight (focus on the heavy spots)
        src = rng.choice(n, p=cur)
        if cur[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if dst == src:
            continue

        # move up to 20 % of the source mass (larger moves early on are useful)
        eps_move = cur[src] * rng.random() * 0.20
        if eps_move <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= eps_move
        cand[dst] += eps_move
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()                      # stay on the simplex

        cand_score = _exact_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            cur = cand
            cur_score = cand_score
            if cand_score < best_seq_score:
                best_seq_score, best_seq = cand_score, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑best {best_seq_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise final answer and convert to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_seq_score:.6f}")

    return [float(v) for v in best_seq]
