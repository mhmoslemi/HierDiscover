def _descent(a0, n, deadline, start_time, best_score):
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    L_conv = _next_pow2(2*n-1)
    L_corr = _next_pow2(3*n-2)
    beta = 5.0
    beta_mul = 1.02
    beta_max = 5000.0
    while time.time() < deadline:
        t += 1
        # compute gradient
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_sc:
            best_sc = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] descent it={t} β={beta:.2f} score={cur_score:.6f}")
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.05 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)
        beta = min(beta * beta_mul, beta_max)
        # occasional random perturb
        if t % 2000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
    return best_a, best_sc

2️⃣  An Adam‑style projected descent on the smooth‑max surrogate of the
    objective.  The soft‑max temperature β starts very small and is
    increased to a large value (β≈5 000) so that the optimisation finally
    follows the true max‑convolution landscape.  The learning‑rate is
    scaled by the Adam second‑moment estimate and a modest base step
    (0.05) is used to stay stable near the optimum.  Several random
    restarts are performed if time permits.

3️⃣  A final hill‑climb that moves a tiny fraction of mass between random
    coordinates (ε≈10⁻³·a_src) and accepts only improving moves.  This
    fine‑tunes the exact objective used by ``evaluate_sequence``.

The routine respects the 1000 s budget, prints progress, and returns the
best sequence found (normalised to unit sum so that the evaluation is
scale‑free).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a*a                     (full linear convolution)
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is computed via FFT‑based correlation.
    """
    # 1. convolution a*a
    conv = _conv_self(a, L=L_conv)          # length 2n‑1
    max_c = conv.max()

    # 2. soft‑max weights (stable)
    shifted = conv - max_c                 # max becomes 0
    exp_vals = np.exp(beta * shifted)      # may under‑flow for large negative values
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # 3. correlation w * a (a reversed) → gradient
    a_rev = a[::-1]                        # length n
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)   # linear convolution w * a_rev

    # Extract the slice that equals Σ_k w_k a_{k-i}
    start = a.shape[0] - 1                 # n‑1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected descent on the smooth‑max objective
# ----------------------------------------------------------------------
def _descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type descent on the smooth‑max surrogate with an aggressive
    temperature schedule (β up to several thousand).  Returns the best
    sequence encountered and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # FFT sizes (fixed throughout the run)
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # soft‑max temperature schedule
    beta = 5.0               # start smooth
    beta_mul = 1.02          # geometric increase per iteration
    beta_max = 5000.0        # very sharp at the end

    # base learning‑rate (scaled by Adam's second moment)
    base_lr = 0.05

    while time.time() < deadline:
        t += 1

        # gradient of the smooth surrogate (soft‑max)
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        # exact score (sum of a is ~1 → denominator = 1)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] descent it={t:,} "
                f"β={beta:.1f} score={cur_score:.6f}"
            )

        # Adam‑style update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional tiny random perturbation to escape plateaus
        if t % 3000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final microscopic hill‑climb (larger ε than the previous code)
# ----------------------------------------------------------------------
def _final_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers between two coordinates.  Only improving
    moves are kept; ε≈10⁻³·a_src.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-3       # up to 0.1 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine respects the 1000 s budget and returns the best
    sequence found (as a plain list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin to avoid overshooting the time limit
    deadline = start_time + budget_s - 0.3

    # problem size – a power of two is nicer for FFT, 4096 works well
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise: uniform baseline (sum ≈ 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # try to use the globally provided starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  High‑resolution parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    # geometric bases – denser grid than before
    r_grid = np.logspace(np.log10(0.90), np.log10(0.99999), num=80)
    # power exponents – wider range and finer steps
    p_grid = np.linspace(-5.0, 4.0, 81)

    for r in r_grid:
        # leave ~3 min for later phases
        if time.time() >= deadline - 180:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 180:
                break
            shape = geo * np.power(idx, p)                      # element‑wise product
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s

            # mix with uniform – a few mixing ratios
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} p={p:.2f} "
                        f"λ={lam:.2f} score={sc:.6f}"
                    )

    # ------------------------------------------------------------------
    # 1b️⃣  Random decreasing patterns (to diversify the initial pool)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    while time.time() < deadline - 120:   # reserve ~2 min for descent & climb
        # generate a random vector, sort descending and normalise
        rnd = rng.random(n)
        rnd.sort()
        rnd = rnd[::-1]          # monotone decreasing
        rnd /= rnd.sum()
        sc = _eval_fast(rnd, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = rnd.copy()
            print(
                f"[{time.time() - start_time:.2f}s] random‑decrease score={sc:.6f}"
            )
        # only a few hundred samples are cheap; break if time is low
        if rng.random() < 0.001:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Aggressive smooth‑max Adam descent (the heavy part)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100:   # keep ~1 min for final hill‑climb
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        a_smooth, sc_smooth = _descent(
            best_seq, n, deadline - 30, start_time, best_score
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.copy()
            print(
                f"[{time.time() - start_time:.2f}s] descent improvement score={sc_smooth:.6f}"
            )

        # optional second short restart if time permits
        if time.time() < deadline - 50:
            alt_start = np.maximum(best_seq + 0.001 * rng.standard_normal(n), 0.0)
            alt_start = _project_simplex(alt_start)
            a_restart, sc_restart = _descent(
                alt_start, n, deadline - 30, start_time, best_score
            )
            if sc_restart < best_score:
                best_score = sc_restart
                best_seq = a_restart.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] second‑restart improvement score={sc_restart:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb final score={sc_hc:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
