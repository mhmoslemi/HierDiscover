def _project_simplex(v):
    # v can have negative entries
    # algorithm: sort v descending, compute cumulative sum,
    # find rho such that v_sorted[i] - (cumsum[i] - 1) / (i+1) > 0
    # then theta = (cumsum[rho] - 1) / (rho+1)
    # return np.maximum(v - theta, 0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u - (cssv - 1) / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑score step function.

Algorithm (two‑phase hybrid optimisation)
------------------------------------------------
Phase 1 – smooth gradient descent on a soft‑max of the autocorrelation.
  • Maintain a probability vector p (Σp = 1, p_i ≥ 0).
  • For the current p compute the linear self‑convolution c = p∗p.
  • Form a soft‑max weight w_k ∝ exp(τ·(c_k – max(c))) (τ grows from 1 to ~100).
  • The gradient of the smoothed objective is
        g = 2·(w ∗ reverse(p))[n‑1 : n‑1+n] .
    (convolution performed with FFT, O(n log n)).
  • Take a projected gradient step:
        p ← p – η·g,   p ← max(p,0),   p ← p / Σp .
    η decays slowly.
  • Random jitter is injected every few thousand iterations to avoid
    plateaus; a restart from the best‑so‑far is performed if no
    improvement for a long stretch.

Phase 2 – fine‑grained hill‑climbing “mass‑transfer” moves (the same
  moves as in the reference code).  Starting from the best vector of
  Phase 1, repeatedly move a small fraction of mass from the pair of
  indices that contributes most to the current worst autocorrelation
  to a location that currently carries little mass, accepting only
  strict improvements.  Periodic re‑normalisation and occasional
  jitter keep the search robust.

Both phases keep the best vector seen and return it after the time
budget (≈1000 s).  The method works with any length; the default is
4096, but it automatically uses the length of a supplied
`height_sequence_1` if present.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two ≥ x
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution (fast via FFT)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# General linear convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    L = _next_pow2(a.shape[0] + b.shape[0] - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: a.shape[0] + b.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Simple non‑negative simplex projection (zero out negatives then renorm)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v = np.full_like(v, 1.0 / v.size)
    return v


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the evaluation wrapper
    deadline = start_time + budget_s - 0.5
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Determine length and initialise a probability vector p
    # --------------------------------------------------------------
    try:
        # Use the previously discovered best construction if it exists
        p0 = np.asarray(height_sequence_1, dtype=float)
        if p0.ndim != 1 or p0.size == 0:
            raise ValueError
        p0 = np.maximum(p0, 0.0)
        if p0.sum() == 0.0:
            raise ValueError
        p = p0 / p0.sum()
        n = p.shape[0]
        print(f"[{time.time() - start_time:.2f}s] Using height_sequence_1 (n={n})")
    except Exception:
        # Fall back to a Beta‑type (U‑shaped) distribution
        n = 4096
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        beta = 0.6               # shape parameter in (0,1); beta=1 → uniform
        w = (x * (1.0 - x) + 1e-6) ** (beta - 1.0)   # w ∝ [x(1‑x)]^{beta‑1}
        p = w / w.sum()
        print(f"[{time.time() - start_time:.2f}s] Generated beta={beta} start (n={n})")

    # --------------------------------------------------------------
    # 1) Initial bookkeeping
    # --------------------------------------------------------------
    conv = _conv_self(p)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max            # Σp = 1  →  score = 2·n·max(conv)
    best_p = p.copy()
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth gradient descent on a soft‑max of the autocorrelation
    # --------------------------------------------------------------
    iteration = 0
    last_improve_iter = 0
    eta = 5.0                     # initial step size
    eta_decay = 0.9995
    tau = 1.0                     # temperature of the soft‑max
    # Phase 1 will consume roughly 40 % of the total budget
    phase1_deadline = start_time + 0.40 * budget_s

    while time.time() < deadline and time.time() < phase1_deadline:
        iteration += 1

        # current convolution & score
        conv = _conv_self(p)
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            last_improve_iter = iteration
            print(f"[{time.time() - start_time:.2f}s] GD iter {iteration:,} new best = {best_score:.6f}")

        # soft‑max weights (focus on the current worst shift)
        # we shift by the max to keep the exponent ≤ 0, avoiding overflow
        shift = max_conv
        w = np.exp(tau * (conv - shift))
        w /= w.sum()

        # gradient = 2·(w ∗ reverse(p)) restricted to the n relevant entries
        p_rev = p[::-1]
        grad_full = _conv_fft(w, p_rev)               # length = 3·n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]        # slice of length n

        # projected gradient step (simple non‑negative projection)
        eta = max(0.01, eta * eta_decay)
        p = p - eta * grad
        p = _project_simplex(p)

        # slowly raise the temperature (τ) – higher τ makes the objective
        # more like the true max.  The schedule is linear in elapsed time.
        elapsed = time.time() - start_time
        tau = min(200.0, 1.0 + 199.0 * (elapsed / (0.40 * budget_s)))

        # occasional jitter to shake out of tiny plateaus
        if iteration % 2000 == 0:
            jitter = 0.001 * p * (rng.random(n) - 0.5)
            p = _project_simplex(p + jitter)

        # long plateau handling – re‑jitter from the best‑so‑far
        if iteration - last_improve_iter > 40000:
            p = best_p.copy()
            jitter = 0.005 * p * (rng.random(n) - 0.5)
            p = _project_simplex(p + jitter)
            last_improve_iter = iteration
            print(f"[{time.time() - start_time:.2f}s] GD plateau – re‑jittered at iter {iteration}")

    # --------------------------------------------------------------
    # Phase 2 – fine‑grained hill‑climbing (mass‑transfer moves)
    # --------------------------------------------------------------
    p = best_p.copy()
    conv = _conv_self(p)
    cur_max = conv.max()
    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1

        # identify the current worst shift
        k = int(conv.argmax())
        cur_max = conv.max()

        # indices that actually contribute to conv[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # source = index of the largest term p[i]·p[k‑i] in this shift
        contrib = p[idx] * p[k - idx]
        src = idx[int(np.argmax(contrib))]

        # destination = an index that currently holds little mass
        inv = 1.0 - p
        inv_sum = inv.sum()
        if inv_sum <= 0.0:
            inv = np.ones_like(p)
            inv_sum = inv.sum()
        prob = inv / inv_sum
        dst = rng.choice(n, p=prob)
        if dst == src:
            continue

        # fraction of source mass to move (decreases as time passes)
        elapsed = time.time() - start_time
        max_frac = max(0.01, 0.20 * (1.0 - elapsed / budget_s))
        epsilon = p[src] * rng.random() * max_frac
        if epsilon <= 0.0:
            continue

        # ----- backup current state (cheap copies) -----
        p_old = p.copy()
        conv_old = conv.copy()

        # ----- exact update of the convolution (derived analytically) -----
        # remove source contribution
        conv[src:src + n] -= 2.0 * epsilon * p
        # add destination contribution
        conv[dst:dst + n] += 2.0 * epsilon * p
        # quadratic correction terms
        if 2 * src < conv.size:
            conv[2 * src] += epsilon * epsilon
        if 2 * dst < conv.size:
            conv[2 * dst] += epsilon * epsilon
        if src + dst < conv.size:
            conv[src + dst] -= 2.0 * epsilon * epsilon

        # ----- update the probability vector -----
        p[src] -= epsilon
        p[dst] += epsilon
        if p[src] < 0.0:
            p[src] = 0.0

        # ----- accept or reject the move -----
        new_max = conv.max()
        if new_max < cur_max - 1e-16:
            cur_max = new_max
            if 2.0 * n * cur_max < best_score:
                best_score = 2.0 * n * cur_max
                best_p = p.copy()
                last_improve_iter = iteration
                print(
                    f"[{time.time() - start_time:.2f}s] MC iter {iteration:,} new best = {best_score:.6f}"
                )
        else:
            p = p_old
            conv = conv_old

        # periodic re‑normalisation to counteract floating‑point drift
        if iteration % 5000 == 0:
            s = p.sum()
            if abs(s - 1.0) > 1e-12:
                p /= s
                conv = _conv_self(p)

        # long‑run plateau handling – restart from best with a small jitter
        if iteration - last_improve_iter > 200_000:
            p = best_p.copy()
            jitter = 0.005 * p * (rng.random(n) - 0.5)
            p = _project_simplex(p + jitter)
            conv = _conv_self(p)
            last_improve_iter = iteration
            print(f"[{time.time() - start_time:.2f}s] MC plateau – re‑jittered at iter {iteration}")

    # --------------------------------------------------------------
    # 3) Return the best vector (normalised to a probability vector)
    # --------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0.0:
        best_p /= total
    else:
        best_p = np.full(n, 1.0 / n)

    final_score = 2.0 * n * (_conv_self(best_p).max())
    print(f"[{time.time() - start_time:.2f}s] finished – best score {final_score:.6f}")

    return [float(v) for v in best_p]
