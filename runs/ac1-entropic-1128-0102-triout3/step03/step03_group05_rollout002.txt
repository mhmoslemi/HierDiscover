"""
Propose a low‑value step‑function using a hybrid of
* fast projected gradient descent with momentum on the full
  simplex (∑a_i = 1, a_i ≥ 0);
* a tiny line‑search on the step‑size;
* occasional random “mass‑transfer’’ moves for exploration;
* a low‑dimensional evolutionary refinement (block‑wise mutation)
  that occasionally perturbs the current best solution;
* a diverse pool of analytically motivated initialisers
  (uniform, power‑law, linear, cosine, Hann, DPSS, Gaussian).

The algorithm keeps the best sequence seen so far and restarts from a
noisy version of it if no improvement is observed for a while.
It stops after the allotted time budget (≈ 1000 s) and returns the
best list of floats.
"""

import numpy as np
import time
from scipy import signal  # for DPSS & other windows


# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence (no safety checks).  a must be
    a non‑negative 1‑D array.  The function returns the same value as
    evaluate_sequence().
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # next power of two for zero‑padding (FFT is faster on powers of two)
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def gradient_multi_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient for the max‑convolution approximated by an
    aggregation over the selected active indices.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i = np.arange(low, high + 1)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset+1)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n‑i) (i=0..n‑1) normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


def build_cosine_window(n: int) -> np.ndarray:
    """Cosine (raised‑cosine) window, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def build_hann_window(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    a = np.hanning(n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def build_gaussian_window(n: int, sigma: float = None) -> np.ndarray:
    """Discrete Gaussian window centred in the middle."""
    if sigma is None:
        sigma = n / 6.0
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def build_dpss_window(n: int, NW: float = 4.0) -> np.ndarray:
    """
    Discrete prolate spheroidal sequence (DPSS).  The window is
    non‑negative for the chosen NW and normalised to sum 1.
    """
    # signal.windows.dpss returns shape (Kmax, n) – we ask for a single sequence.
    w = signal.windows.dpss(n, NW, Kmax=1, sym=False)[0]
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation value.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # tiny safety margin

    # --------------------------------------------------------------
    # 1. Problem size – keep the same size as the previous best.
    n = 3000

    # --------------------------------------------------------------
    # 2. Build a diverse collection of analytically motivated seeds.
    candidates = []

    # power‑law family
    for p in [0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60]:
        for off in [0.0, 0.1, 0.5, 1.0, 2.0]:
            candidates.append(build_power_law(n, p, off))

    # linear decreasing
    candidates.append(build_linear_descending(n))

    # cosine / Hann / Gaussian / DPSS
    candidates.append(build_cosine_window(n))
    candidates.append(build_hann_window(n))
    candidates.append(build_gaussian_window(n))
    candidates.append(build_dpss_window(n, NW=4.0))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # use the previous global best if it exists
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                seq = np.zeros(n, dtype=np.float64)
                seq[:prev.size] = prev
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0:
                seq /= seq.sum()
                candidates.append(seq)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Pick the best seed (according to the fast score).
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best seed value: {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation – projected gradient descent with momentum,
    #    occasional line‑search and random mass‑transfer moves.
    a = best_seq.copy()                       # current point (on simplex)
    best_seq = a.copy()
    best_val = fast_score(a)

    # hyper‑parameters
    eta0 = 0.8                                 # base step size
    beta = 0.9                                 # momentum coefficient
    momentum = np.zeros_like(a)
    max_active = 30                            # number of active indices used
    random_move_prob = 0.025
    random_move_frac = 0.20
    uphill_prob = 0.001
    stagnation_time = 30.0                     # seconds without improvement → reseed
    last_improve = time.time()
    it = 0

    # pre‑compute FFT size for convolution
    conv_size = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # ----------------------------------------------------------
        # a) compute self‑convolution and its maximum
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_conv = conv.max()

        # b) active set – the K largest entries (no delta‑filter to keep it cheap)
        if conv.size > max_active:
            idxs = np.argpartition(-conv, max_active)[:max_active]
        else:
            idxs = np.arange(conv.size)
        active_idxs = idxs

        # c) aggregated sub‑gradient
        grad = gradient_multi_active(a, active_idxs)

        # d) momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # e) step‑size (inverse‑square‑root decay)
        eta = eta0 / np.sqrt(it + 1.0)

        # f) tiny line‑search over a few multipliers
        cand_vals = []
        cand_seqs = []
        for fac in (1.0, 0.5, 2.0):
            step = eta * fac
            cand = proj_simplex(a - step * momentum)
            val = fast_score(cand)
            cand_vals.append(val)
            cand_seqs.append(cand)
        cur_idx = int(np.argmin(cand_vals))
        a_new = cand_seqs[cur_idx]
        cur_val = cand_vals[cur_idx]

        # g) accept improvement or occasional uphill move
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
        else:
            if np.random.rand() < uphill_prob:
                a = a_new

        # ----------------------------------------------------------
        # h) random mass‑transfer move (exploration)
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = proj_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < uphill_prob:
                    a = cand

        # ----------------------------------------------------------
        # i) low‑dimensional evolutionary perturbation every few thousand iterations
        if it % 5000 == 0 and it > 0:
            # build a block‑wise representation (B blocks)
            B = 30
            block_len = n // B
            # aggregate mass per block
            block_mass = np.add.reduceat(a, np.arange(0, n, block_len))
            # mutation on block masses
            sigma = 0.05
            mutated = block_mass + sigma * np.random.randn(B)
            mutated = np.maximum(mutated, 0.0)
            mutated_sum = mutated.sum()
            if mutated_sum > 0:
                mutated /= mutated_sum
                # expand back to full length
                a_block = np.repeat(mutated, block_len)[:n]
                a_block = proj_simplex(a_block)  # ensure feasibility
                val_block = fast_score(a_block)
                if val_block < best_val:
                    best_val = val_block
                    best_seq = a_block.copy()
                    a = a_block
                    last_improve = time.time()
                elif np.random.rand() < uphill_prob:
                    a = a_block

        # ----------------------------------------------------------
        # reporting
        if it % 2000 == 0:
            print(f"[iter {it:7d}] current {cur_val:.6f}  best {best_val:.6f}")

        # ----------------------------------------------------------
        # j) stagnation handling – reseed from a noisy version of the best
        if time.time() - last_improve > stagnation_time:
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(0.9 * best_seq + 0.1 * noise)
            momentum[:] = 0.0
            last_improve = time.time()
            print(f"[restart] after {it} iters, re‑seeded")

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # The external checker will re‑normalise the sequence, therefore we
    # simply return the raw best sequence.
    return best_seq.tolist()
