def _run_exp_gradient(a0, n, deadline, start_time, eta0=1.0):
    a = a0.copy()
    best_score = float('inf')
    best_a = a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG improve: n={n}, iter={it}, score={cur_score:.6f}")

        # compute subgradient
        grad = _subgradient_full(a, conv)

        # exponentiated update
        eta = eta0 / math.sqrt(it)
        # to avoid overflow, subtract max(grad) from grad?
        # Actually exp(-eta*grad) may be large/small; we can use np.exp(-eta * grad - max_shift)
        shift = np.max(-eta * grad)  # shift to keep exponentials stable
        w = a * np.exp(-eta * grad - shift)
        # renormalize
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(a, 1.0 / n)
        else:
            w = w / w_sum

        a = w
    return best_a, best_score

def _run_adam_gradient(a0, n, deadline, start_time, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_score = inf
    best_a = a.copy()
    it = 0
    while time.time() < deadline:
        it+=1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it}, score={cur_score:.6f}")
        grad = _subgradient_full(a, conv)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step*m_hat
        # projection
        a = _project_simplex(a)
    return best_a, best_score

"""
Step‑function optimizer for `evaluate_sequence`.

The algorithm proceeds in three growing‑resolution stages.
Each stage runs a short *exponential‑gradient* (EG) phase to
quickly shape the distribution, followed by a longer *Adam‑style*
projected‑gradient descent on the probability simplex.
The best vector is continually up‑sampled to a larger length.
When the main budget is exhausted a final simulated‑annealing
refinement with relatively large mass‑transfer moves is performed.
All operations respect the non‑negative simplex constraint, and
the routine returns the best sequence found within the 1000 s
time limit.

The objective for a normalized vector (sum = 1) is
    score = 2·n·max_k (a∗a)[k] .
Both the EG and Adam phases work directly with this quantity
using the exact sub‑gradient of the maximal convolution entry.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a*a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    # numerical noise may produce tiny negative values; clip to zero
    conv = np.maximum(conv, 0.0)
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‐convolution
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Average sub‑gradient of the max‑convolution.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val <= 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    act = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if act.size:
        g /= float(act.size)
    return g


# ----------------------------------------------------------------------
# Exponential‑gradient ascent on the simplex
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray, n: int,
                      deadline: float, start_time: float,
                      eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Projected exponential‑gradient descent (maximises -score)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG improve: n={n}, iter={it}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        # decreasing step size  η_t = η0 / sqrt(t)
        eta = eta0 / math.sqrt(it)

        # exponentiated update, stabilised by a common shift
        shift = np.max(-eta * grad)          # prevents overflow
        w = a * np.exp(-eta * grad - shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(a, 1.0 / n)
        else:
            w = w / w_sum

        a = w
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent
# ----------------------------------------------------------------------
def _run_adam_gradient(a0: np.ndarray, n: int,
                       deadline: float, start_time: float,
                       lr: float = 0.5,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Projected Adam optimizer on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = beta1
    beta2_pow = beta2

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # step
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection back onto the simplex
        a = _project_simplex(a)

        # update the bias‑correction factors
        beta1_pow *= beta1
        beta2_pow *= beta2
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (large moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """
    Cheap simulated‑annealing walk that moves a random fraction
    of the mass from one index to another (up to 30 % of the source).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 30 % of a[i]
        eps = a[i] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis criterion
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear interpolation between two discretisations
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Rescale `seq` from its current length to `new_n` by linear interpolation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
