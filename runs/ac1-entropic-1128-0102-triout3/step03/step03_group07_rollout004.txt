"""
Propose a candidate sequence that minimizes the auto‑convolution
peak  2·n·max(a∗a) / (∑a)² .
The algorithm combines:

* an Adam‑optimised gradient descent on the probability simplex,
  using a soft‑max (log‑sum‑exp) smoothing of the max that is
  slowly sharpened (β → large);
* occasional linear‑programming updates that maximise the total
  mass of a partner sequence b under the current convolution bound.
  The partner b is normalised and mixed into the current iterate,
  providing a strong “jump’’ direction;
* random restarts and small Gaussian perturbations to avoid stagnation.

All work is done on the simplex (∑a = 1, a ≥ 0), so the evaluation
function reduces to 2·n·max(conv).  The code respects the overall
time budget (≈ 1 000 s) and returns the best sequence found.
"""

import time
import numpy as np
from typing import List
from scipy import sparse, optimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^‑½  (a_i = C(2i,i)/4^i) normalised
    to sum to 1 (probability simplex).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}. Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def convolution_constraint_matrix(a: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix C such that C @ b = a ∗ b (linear convolution).
    Each column i contains the vector a placed on rows i … i+n‑1.
    """
    n = a.shape[0]
    L = 2 * n - 1
    # LIL format is convenient for incremental construction.
    mat = sparse.lil_matrix((L, n), dtype=np.float64)
    for i in range(n):
        mat[i:i + n, i] = a  # slice assignment copies the whole a
    return mat.tocsr()


def solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve    max 1ᵀb    s.t.  a ∗ b ≤ M ,  b ≥ 0 .
    Returns the optimal b (may be None on failure).
    """
    n = a.shape[0]
    L = 2 * n - 1
    C = convolution_constraint_matrix(a)                   # shape (L,n)
    b_ub = np.full(L, M, dtype=np.float64)                # RHS vector
    c = -np.ones(n, dtype=np.float64)                     # minimise -sum b  ≡ maximise sum b
    bounds = [(0, None) for _ in range(n)]

    # SciPy's HiGHS solver handles sparse matrices efficiently.
    res = optimize.linprog(
        c,
        A_ub=C,
        b_ub=b_ub,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 30.0},
    )
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Core optimisation routine

def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises the evaluation
    function.  The routine respects the supplied time budget and
    returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # -------------------- problem size --------------------
    # A moderate size makes FFT cheap and LP still tractable.
    n = 1500                                 # length of the sequence
    L = 2 * n - 1                            # length of a∗a
    Nfft = next_pow_two(3 * n - 2)           # FFT size for all convolutions

    # -------------------- initial point --------------------
    a = half_binomial_series(n)              # already on the simplex
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # -------------------- optimiser hyper‑parameters --------------------
    # Adam parameters
    lr = 0.20                                 # base learning rate
    beta1, beta2 = 0.9, 0.999                 # exponential decay rates
    eps = 1e-8
    m = np.zeros_like(a)                     # first moment
    v = np.zeros_like(a)                     # second moment

    # Soft‑max smoothing of the max
    beta = 1.0
    beta_target = 2000.0
    beta_mul = 1.003

    # Mixing factor for LP‑generated partner
    t_mix = 0.30                               # 0 < t_mix ≤ 1

    # Frequency of LP calls
    lp_interval = 2000                         # iterations between LP solves

    # Restart logic
    last_improve = 0
    restart_gap = 500_000                      # iterations without improvement

    # Random perturbation probabilities
    noise_prob = 0.004
    noise_scale = 2e-4

    # -------------------- timing --------------------
    deadline = time.time() + budget_s - 5.0     # safety margin
    it = 0

    # -------------------- main optimisation loop --------------------
    while time.time() < deadline:

        A_fft = np.fft.rfft(a, n=Nfft)

            grad = 2.0 * corr_full[start:start + n]

        else:

            k = int(conv_full.argmax())
            grad = np.zeros_like(a)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = k - idx
            grad[idx] = 2.0 * a[jdx]

        # ---- 4) Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Prevent steps that would drive entries negative too drastically
        step = np.minimum(step, a * 0.95)
        a_trial = a - step
        a = project_onto_simplex(a_trial)

        # ---- 5) occasional small random perturbation ----
        if rng.random() < noise_prob:
            noise = (rng.random(n) - 0.5) * noise_scale
            a = project_onto_simplex(a + noise)

        # ---- 6) LP jump (every lp_interval iterations) ----
        if it > 0 and it % lp_interval == 0:
            # Build partner sequence b that respects the current bound M = max_conv
            b = solve_lp_max_sum(a, max_conv)
            if b is not None:
                sum_b = b.sum()
                # Only accept if we actually gain mass (sum_b > 1)
                if sum_b > 1.0 + 1e-6:
                    g = b / sum_b               # normalise to the simplex
                    # convex mixture (still on the simplex)
                    a = (1.0 - t_mix) * a + t_mix * g
                    # tiny numerical drift may accumulate – re‑project
                    a = project_onto_simplex(a)
                    print(f"[iter {it:09d}] LP step performed, sum(b)={sum_b:.6f}")

        if it - last_improve > restart_gap:
            a = rng.dirichlet(np.full(n, 0.6))
            a = project_onto_simplex(a)  # just in case
            last_improve = it
            print(f"[iter {it:09d}] restart triggered – random start")

        it += 1

    # -------------------- final evaluation --------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
