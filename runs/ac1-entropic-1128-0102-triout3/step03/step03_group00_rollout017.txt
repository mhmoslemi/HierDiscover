def _conv_and_grad(a, gamma):
    n = a.shape[0]
    L = 1 << ((3*n).bit_length()) # at least 3n
    # compute FFTs
    FA = np.fft.rfft(a, n=L)
    # conv
    conv = np.fft.irfft(FA * FA, n=L)[:2*n-1]
    # compute weights
    max_val = conv.max()
    w = np.exp(gamma * (conv - max_val))

    grad = 2 * corr[n-1:n-1+n]    # slice length n
    return conv, grad

"""
Proposes a step‑function that (empirically) improves the upper bound on
`evaluate_sequence`.  The algorithm has three phases:

1. **Coarse parametric sweep** – a one‑parameter family
   a_i ∝ (i·(n‑1‑i)+ε)^α is sampled (α ∈ [‑0.4, 2]) for a short time.
   The best shape from this sweep gives a good initial guess.

2. **Smoothed gradient descent on the simplex** – we minimise a smooth
   approximation of the max‑convolution
   f(a)=2·n·max(a∗a) by using a softmax with an annealing parameter γ.
   The gradient of the smooth objective is computed with FFTs,
   an Adam‑style update is performed, and each iterate is projected onto
   the probability simplex.

3. **Local mass‑transfer annealing** – a cheap simulated‑annealing walk
   (ε‑mass moves) fine‑tunes the final distribution.

The whole routine respects the 1000 s budget and always returns a
non‑negative list whose sum is 1 (the optimiser works on the simplex,
but a final renormalisation is performed for safety).
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a∗a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Smoothed gradient of 2·n·softmax_γ(a∗a)
# ------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the convolution a∗a, the softmax weights, and the gradient of
    the smoothed objective 2·n·softmax_γ(a∗a).
    Returns (conv, grad) where conv has length 2·n‑1 and grad has length n.
    """
    n = a.shape[0]

    # ----- convolution a ∗ a -------------------------------------------------
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len_conv)
    conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]

    # ----- softmax weights ---------------------------------------------------
    max_val = conv.max()
    # subtract max for numerical stability
    w_raw = np.exp(gamma * (conv - max_val))
    w_sum = w_raw.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w_raw / w_sum

    # ----- gradient of the smoothed objective -------------------------------
    #   grad_i = 4·n· Σ_k w_k·a_{k‑i}   (cross‑correlation of w with a)
    #   compute cross‑correlation via FFT (w * rev(a))
    fft_len_corr = 1 << ((3 * n - 2).bit_length())
    Wf = np.fft.rfft(w, n=fft_len_corr)
    Afr = np.fft.rfft(a[::-1], n=fft_len_corr)   # reversed a
    corr = np.fft.irfft(Wf * Afr, n=fft_len_corr)[: 3 * n - 2]
    # slice the central n elements (indices n‑1 … n‑1+n‑1)
    cross = corr[n - 1:n - 1 + n]
    grad = 4.0 * n * cross
    return conv, grad


# ------------------------------------------------------------
# Simple Adam optimiser on the simplex (projected after each step)
# ------------------------------------------------------------
def _run_smooth_adam(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     gamma_start: float = 5.0,
                     gamma_end: float = 200.0,
                     max_iter: int = 10 ** 7,
                     lr: float = 0.02,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser using the smoothed gradient.
    Returns the best sequence seen and its (non‑smoothed) score
    2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # schedule the softmax temperature
        gamma = gamma_start + (gamma_end - gamma_start) * (it / max_iter)

        conv, grad = _smooth_gradient(a, gamma)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: iter={it}, γ={gamma:.1f}, score={cur_score:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the probability simplex
        a = _project_simplex(a)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing local refinement (mass‑transfer moves)
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_seq: np.ndarray,
                   cur_best_score: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 80000) -> tuple[np.ndarray, float]:
    """
    Tiny “move ε from i to j’’ simulated‑annealing walk.
    Returns possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up.fill(1.0 / new_n)
    return up

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the given time budget,
    performs a cheap parametric sweep, a smoothed Adam optimisation,
    and a final annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------
    # Choose the final length (larger n gives a better asymptotic bound)
    # --------------------------------------------------------
    n_big = 3500                     # a compromise between quality and speed

    # --------------------------------------------------------
    # Initialise from a previously found sequence if possible
    # --------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            a0 = _upsample(starter, n_big)
        else:
            raise ValueError
    except Exception:
        # fallback: uniform start
        a0 = np.full(n_big, 1.0 / n_big)

    best_seq = a0.copy()
    best_score = float("inf")

    # --------------------------------------------------------
    # Stage 1 – cheap parametric sweep over a Beta‑like family
    # --------------------------------------------------------
    sweep_deadline = time.time() + max(20.0, 0.10 * (deadline - time.time()))
    while time.time() < sweep_deadline:
        # α in a range that spans U‑shapes, uniform, and bell‑shapes
        alpha = np.random.uniform(-0.4, 2.0)
        idx = np.arange(n_big, dtype=float) + 0.5
        scaled = idx / n_big
        prod = scaled * (1.0 - scaled) + 1e-12          # avoid exact zero
        shape = prod ** alpha
        shape_sum = shape.sum()
        if shape_sum == 0.0:
            continue
        shape /= shape_sum

        conv = _conv_self(shape)
        score = 2.0 * n_big * conv.max()
        if score < best_score:
            best_score = score
            best_seq = shape.copy()
            print(f"[{time.time() - start_time:6.2f}s] sweep improve: α={alpha:.3f}, score={score:.6f}")

    # --------------------------------------------------------
    # Stage 2 – smoothed Adam gradient descent (bulk of the budget)
    # --------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        grad_time = max(10.0, 0.75 * remaining)          # allocate ~75 % to gradient descent
        grad_deadline = time.time() + grad_time

        a_opt, score_opt = _run_smooth_adam(
            best_seq, n_big, grad_deadline, start_time,
            gamma_start=5.0,
            gamma_end=200.0,
            max_iter=10 ** 7,
            lr=0.02,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: final score={score_opt:.6f}")

    # --------------------------------------------------------
    # Stage 3 – local stochastic refinement (annealing)
    # --------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq, n_big, deadline, start_time,
            best_seq, best_score,
            start_temp=5e-2,
            end_temp=1e-5,
            steps=80000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # --------------------------------------------------------
    # Final safety checks and conversion to plain Python list
    # --------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
