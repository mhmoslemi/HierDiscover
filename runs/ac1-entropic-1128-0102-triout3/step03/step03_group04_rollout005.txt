def weighted_gradient(a, conv, beta):
    # conv is array length m = 2n-1
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / len(w))
    else:
        w = w / w_sum
    # compute conv of a and w
    n = a.shape[0]
    L = n + len(w) - 1  # 3n-2
    # using rfft
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    grad_full = np.fft.irfft(A * W, n=L)
    grad = 2.0 * grad_full[n-1:n-1+n]
    return grad

        – obtain the gradient of the weighted sum   g = 2·a★w   (★ = convolution),
        – update the probabilities a by an exponentiated‑gradient step
          a ← a·exp(−η·g) and renormalise to the simplex,
        – slowly increase β (focus on the true maximum) and decay η.
    The gradient is obtained with a single FFT‑based convolution, so
    each iteration is cheap (≈ 0.3 ms for n≈2500).  Random jitter and an
    occasional restart from a Dirichlet draw help escape plateaus.

3️⃣ **Return the best sequence** – the final candidate is checked with the
    provided ``evaluate_sequence`` before being returned.

The routine respects a configurable time budget (default ≈ 950 s) and
prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)

def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full discrete self‑convolution of a real vector using FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def normalize(a: np.ndarray) -> np.ndarray:
    """Scale a to have unit L1 norm (sum==1); if sum==0 use uniform."""
    total = a.sum()
    if total <= 0.0:
        return np.full_like(a, 1.0 / a.shape[0])
    return a / total


# ----------------------------------------------------------------------
# Families of candidate sequences

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i = (i+offset)^(-p)  (i=0…n‑1) normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    return normalize(a)


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)  normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return normalize(a)


def linear_decay_sequence(n: int, slope: float) -> np.ndarray:
    """a_i = max(0, 1 - slope·i)  normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i)
    return normalize(a)


def gaussian_sequence(n: int, sigma: float, centre: float = None) -> np.ndarray:
    """a_i = exp(- (i−centre)^2 / (2·σ^2) )   (σ>0) normalised."""
    if centre is None:
        centre = 0.0                      # monotone decay from the left
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- ((i - centre) ** 2) / (2.0 * sigma ** 2))
    return normalize(a)


def half_binomial_sequence(n: int) -> np.ndarray:
    """Cumulative product generating the classic half‑binomial seed."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return normalize(a)


# ----------------------------------------------------------------------
# Gradient of a soft‑max weighted sum of convolution entries

def weighted_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k·conv[k]  where w are soft‑max weights
    w_k ∝ exp(β·conv[k]).
    Returns a vector of length len(a).
    """
    # soft‑max over convolution values, stabilised by subtracting the max
    maxc = conv.max()
    shifted = conv - maxc
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution a ★ w  (full length = len(a)+len(w)-1 = 3n-2)
    n = a.shape[0]
    L = n + w.size - 1                # = 3·n‑2
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    full = np.fft.irfft(A * W, n=L)   # real result
    # gradient = 2·a★w evaluated on the central n entries (indices n‑1 … 2n‑2)
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the time budget and returns the best list found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep
    n = 2500                                     # length of the step function
    best_a = None
    best_score = np.inf

    print("[search] phase 1 – parametric sweep (n={})".format(n))

    # -------- power‑law -------------------------------------------------
    p_grid = np.arange(0.35, 0.71, 0.015)
    offsets = [0.2, 0.5, 1.0, 2.0]
    for offset in offsets:
        for p in p_grid:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=offset)
            cur = 2.0 * n * conv_fft(a).max()
            if cur < best_score:
                best_score, best_a = cur, a.copy()
                print(f"[grid] new best: power‑law p={p:.4f}, off={offset}, score={cur:.6f}")

    # -------- exponential -----------------------------------------------
    lam_grid = np.linspace(1e-4, 5e-3, 25)
    for lam in lam_grid:
        if time.time() > deadline:
            break
        a = exponential_sequence(n, lam)
        cur = 2.0 * n * conv_fft(a).max()
        if cur < best_score:
            best_score, best_a = cur, a.copy()
            print(f"[grid] new best: exponential λ={lam:.6f}, score={cur:.6f}")

    # -------- linear decay -----------------------------------------------
    slope_grid = np.linspace(1.0 / n, 5.0 / n, 20)   # support length from n to n/5
    for slope in slope_grid:
        if time.time() > deadline:
            break
        a = linear_decay_sequence(n, slope)
        cur = 2.0 * n * conv_fft(a).max()
        if cur < best_score:
            best_score, best_a = cur, a.copy()
            print(f"[grid] new best: linear slope={slope:.6e}, score={cur:.6f}")

    # -------- discrete Gaussian -------------------------------------------
    sigma_grid = np.linspace(n / 40, n / 4, 20)      # from narrow to wide
    for sigma in sigma_grid:
        if time.time() > deadline:
            break
        a = gaussian_sequence(n, sigma, centre=0.0)  # monotone decay
        cur = 2.0 * n * conv_fft(a).max()
        if cur < best_score:
            best_score, best_a = cur, a.copy()
            print(f"[grid] new best: gaussian σ={sigma:.2f}, score={cur:.6f}")

    # -------- half‑binomial (fallback) -----------------------------------
    if best_a is None:
        a = half_binomial_sequence(n)
        best_a = a.copy()
        best_score = 2.0 * n * conv_fft(a).max()
        print("[fallback] using half‑binomial seed")

    print(f"[phase 1] best score after sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Soft‑max weighted gradient descent (exponentiated update)
    global_best_a = best_a.copy()
    global_best_score = best_score

    # hyper‑parameters (will be adapted during the run)
    eta = 0.12               # initial learning‑rate for exp‑gradient
    beta = 0.5               # initial soft‑max temperature
    beta_max = 25.0
    eta_min = 1e-5
    decay_eta = 0.9995       # multiplicative decay per iteration
    beta_inc = (beta_max - beta) / 300_000.0   # linear increase

    it = 0
    no_improve_iters = 0
    max_no_improve = 30_000    # after this we perform a cheap random restart
    print("[grad] phase 2 – exponentiated‑gradient descent")

    while time.time() < deadline:
        # --- evaluate current point ---
        conv = conv_fft(global_best_a)          # full self‑convolution
        cur_score = 2.0 * n * conv.max()

        # --- weighted gradient ---
        grad = weighted_gradient(global_best_a, conv, beta)

        # --- exponentiated update (always stays non‑negative) ---
        # Clip the gradient to avoid overflow in exp()
        grad_clip = np.clip(grad, -50.0, 50.0)
        factor = np.exp(-eta * grad_clip)
        a_new = global_best_a * factor
        a_new_sum = a_new.sum()
        if a_new_sum == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= a_new_sum

        # occasional tiny jitter to break symmetry
        if rng.random() < 0.002:
            jitter = (rng.random(n) - 0.5) * 1e-7
            a_new += jitter
            a_new = project_onto_simplex(a_new)   # keep on simplex

        # evaluate new candidate
        new_score = 2.0 * n * conv_fft(a_new).max()

        # record improvement
        if new_score < global_best_score:
            global_best_score = new_score
            global_best_a = a_new.copy()
            no_improve_iters = 0
            print(f"[iter {it:07d}] new global best = {global_best_score:.6f}")
        else:
            no_improve_iters += 1

        # accept the move with a tiny probability if it degrades (helps escape)
        if new_score <= cur_score or rng.random() < 0.001:
            global_best_a = a_new
            cur_score = new_score

        # update schedules
        eta = max(eta * decay_eta, eta_min)
        beta = min(beta + beta_inc, beta_max)

        it += 1

        # cheap random restart if we have stalled for a while
        if no_improve_iters >= max_no_improve:
            # reset to a fresh Dirichlet (α=0.7 gives sparse starts)
            a_restart = rng.gamma(0.7, 1.0, size=n)
            a_restart = normalize(a_restart)
            global_best_a = a_restart
            cur_score = 2.0 * n * conv_fft(global_best_a).max()
            no_improve_iters = 0
            print(f"[restart] performed random restart at iter {it}")

    # ------------------------------------------------------------------
    # 3️⃣  Final evaluation and return
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")
    print(f"[info] best sweep parameters yielded score ≈ {best_score:.6f}")

    return global_best_a.tolist()
```
