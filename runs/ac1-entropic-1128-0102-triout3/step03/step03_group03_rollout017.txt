def project_simplex(v):
    # v may be any real array
    # ensure non-negative after projection
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    smooth‑max surrogate, exponentiated‑gradient updates and a final stochastic
    local‑search refinement."""
    import time, math, numpy as np

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # initialise from known sequence if available, else random
    init_seq = globals().get('height_sequence_1')

        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:

        # allocate more iterations for smaller size
        base_iters = 20000

"""
Propose a step‑function (non‑negative heights) that minimises the
`evaluate_sequence` objective.

Approach
--------
* Start from the previously discovered best shape (`height_sequence_1`) if
  it exists, otherwise from a random simplex point.
* The sequence is always kept on the probability simplex
  (non‑negative, sum = 1) by clipping and renormalising.
* For a fixed length `n` we minimise a smooth surrogate of the
  `max(conv(a,a))` term:
        w_k ∝ exp(α·(conv_k / max(conv) – 1))      (soft‑max)
  The gradient of the smoothed maximum is
        g = 2·corr(a, w)                           (full correlation)
  and we perform an exponentiated‑gradient (mirror‑descent) update:
        a ← normalize( a·exp(−η·g) )
  where the step size η decays like 1/√t.
* The smoothing parameter α is slowly increased, turning the smooth
  surrogate into the true max.
* After a few thousand iterations we up‑sample the current best shape
  (linear interpolation, factor ≈1.45) and repeat the optimisation at
  the larger size (up to `MAX_LEN` ≈ 8000).  Larger sequences give a better
  constant in the objective while staying fast enough to search.
* Finally a cheap stochastic local‑search (random mass transfers) tries
  to polish any remaining small improvements.

The whole routine stops when the allotted time (`budget_s`) is exhausted
and returns the best sequence found (as plain Python floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _full_fft_len(n):
    """Return a power‑of‑two length large enough for all FFTs with a vector of length n."""
    # we need at least 3*n‑2 points for the correlation step
    return 1 << ((3 * n - 2).bit_length())


def _conv_and_max(a, fft_len):
    """
    Full auto‑convolution of a with itself using one FFT of length `fft_len`.
    Returns the convolution (length 2*n‑1) and its maximum value.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(A * A, n=fft_len)
    n = a.shape[0]
    conv = conv_full[: 2 * n - 1]
    return conv, conv.max()


def _smooth_weights(conv, max_val, alpha):
    """Soft‑max weights that approximate the max‑operator."""
    if max_val <= 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    rel = conv / max_val                 # max → 1, others ≤ 1
    # shift so the maximum gets exponent 0
    exp_vals = np.exp(alpha * (rel - 1.0))
    s = exp_vals.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def _gradient_corr(w, a, fft_len):
    """
    Gradient of the smoothed max, i.e. 2·corr(a, w).
    Correlation is computed via FFT using the same length as the FFT for the convolution.
    """
    n = a.shape[0]
    A = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w, n=fft_len)
    # correlation a ⋆ w (linear, zero‑lag at index len(w)-1)
    corr_full = np.fft.irfft(np.conj(W) * A, n=fft_len)
    # keep only the linear part
    corr = corr_full[: len(w) + n - 1]
    start = len(w) - 1                     # zero lag
    grad = 2.0 * corr[start:start + n]     # length n
    return grad


def _normalize_simplex(v):
    """Project onto the probability simplex by clipping negatives and renormalising."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / v.shape[0]
    else:
        v /= s
    return v


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Return a list of non‑negative heights with a low `evaluate_sequence` score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0               # small safety margin

    # ------------------------------------------------------------------
    # 1. Initial shape (use the best known construction if present)
    # ------------------------------------------------------------------
    init_seq = globals().get('height_sequence_1')
    if init_seq is not None:
        a = np.array(init_seq, dtype=float)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.random.rand(1500)
    else:
        a = np.random.rand(1500)

    a = _normalize_simplex(a)
    cur_n = a.shape[0]

    # ------------------------------------------------------------------
    # 2. Progressive optimisation at increasing resolutions
    # ------------------------------------------------------------------
    MAX_LEN = 8000           # we will not go beyond this – still fast to search
    SIZE_FACTOR = 1.45       # up‑sampling factor per stage

    best_seq = a.copy()
    fft_len = _full_fft_len(cur_n)
    conv, max_val = _conv_and_max(a, fft_len)
    best_score = 2.0 * cur_n * max_val
    print(f"[{time.time() - start_time:6.2f}s] start: n={cur_n}, score={best_score:.6f}")

    while time.time() < deadline and cur_n <= MAX_LEN:
        # iteration budget for the current size – more iterations for smaller n
        base_iters = 20000
        max_iter = int(base_iters * 2000 / cur_n) + 5000
        max_iter = max(max_iter, 15000)

        # hyper‑parameters (will be mildly adapted during the loop)
        eta0 = 5.0           # initial mirror‑descent step size
        alpha = 5.0          # smoothing for the soft‑max
        alpha_inc = 1.02     # how fast we tighten the smoothing

        it = 0
        while time.time() < deadline and it < max_iter:
            it += 1

            # ---------- convolution, score ----------
            fft_len = _full_fft_len(cur_n)          # length may change after up‑sampling
            conv, max_val = _conv_and_max(a, fft_len)
            cur_score = 2.0 * cur_n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] new best: n={cur_n}, iter={it}, score={cur_score:.6f}")

            # ---------- smooth‑max weights ----------
            w = _smooth_weights(conv, max_val, alpha)

            # ---------- gradient (full correlation) ----------
            grad = _gradient_corr(w, a, fft_len)

            # ---------- exponentiated‑gradient update ----------
            eta = eta0 / math.sqrt(it)
            a = np.maximum(a, 1e-15)                # avoid log(0)
            log_a = np.log(a) - eta * grad
            log_a -= np.max(log_a)                  # numerical stabilisation
            a = np.exp(log_a)
            a = _normalize_simplex(a)

            # slowly increase the concentration of the soft‑max
            if it % 3000 == 0:
                alpha = min(alpha * alpha_inc, 200.0)

        # ------------------------------------------------------------------
        # 3. Upsample to a larger size (if we have not hit the maximum yet)
        # ------------------------------------------------------------------
        if cur_n >= MAX_LEN:
            break

        next_n = int(cur_n * SIZE_FACTOR)
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n <= cur_n:
            break

        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)

        # Interpolate the *best* shape found so far at this resolution
        a = np.interp(x_new, x_old, best_seq)
        a = _normalize_simplex(a)
        cur_n = next_n

        # Refresh best‑score information for the new length
        fft_len = _full_fft_len(cur_n)
        conv, max_val = _conv_and_max(a, fft_len)
        cur_score = 2.0 * cur_n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] after up‑sample: n={cur_n}, score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final cheap stochastic local search (random mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        it_loc = 0
        max_loc = 20000
        while time.time() < deadline and it_loc < max_loc:
            it_loc += 1
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps = np.random.rand() * 0.04          # up to 4 % transfer
            delta = eps * a[i]
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            if a_new[i] < 0.0:
                continue
            fft_len = _full_fft_len(n)
            conv, max_val = _conv_and_max(a_new, fft_len)
            new_score = 2.0 * n * max_val
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                a = a_new
                print(f"[{time.time() - start_time:6.2f}s] local improve: n={n}, iter={it_loc}, score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
