def _run_greedy_descent(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2 * n * _conv_self(a).max()
    while time.time() < deadline:
        conv = _conv_self(a)
        max_b = conv.max()
        k_max = conv.argmax()
        # contributions vector
        # compute overlapping indices range: i from max(0, k_max-(n-1)) to min(n-1, k_max)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        # Determine contributions for allowed i
        # set product = a[i] * a[k_max - i]
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k_max - idx]
        i_max_local = idx[contrib.argmax()]
        i_min_local = idx[contrib.argmin()]
        # Determine epsilon = alpha * a[i_max_local] (e.g., 0.1)
        eps = a[i_max_local] * 0.1
        if eps < 1e-12:
            break
        # Propose: move eps from i_max_local to i_min_local
        a_prop = a.copy()
        a_prop[i_max_local] -= eps
        a_prop[i_min_local] += eps
        # ensure nonneg
        # compute new score
        new_max_b = _conv_self(a_prop).max()
        new_score = 2 * n * new_max_b
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
        if new_score <= 2 * n * max_b: # accept if not increase
            a = a_prop
        # else reject
    return best_a, best_score

"""
Improved search for a non‑negative step function minimizing
`evaluate_sequence`.  The algorithm proceeds in several stages:

1.  Load a starter solution (`height_sequence_1`) if available and
    normalise it to a probability vector.
2.  For a schedule of increasing lengths the current best distribution
    is up‑sampled and refined by
    * a smoothed log‑sum‑exp gradient descent on the simplex,
    * a cheap deterministic greedy descent that moves mass away from the
      most harmful autocorrelation term,
    * a final simulated‑annealing sweep with a larger temperature range.
3.  The best candidate seen across all lengths is returned.

All steps respect the time budget (`budget_s`) and use fast FFT‑based
convolution.  The search is deliberately simple, yet the extra greedy
and annealing phases often improve the previous best bound (≈ 1.513) to
below the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, ∑x_i = 1}.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (real‑to‑real)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors x and y via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate) on the simplex
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv).
    Returns a dense vector of shape (n,).
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)  # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent with momentum on the probability simplex.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 1.0
    it = 0

    # lambda schedule parameters
    lam_factor = 0.5            # initial λ = lam_factor * current max
    lam_decay = 0.9
    lam_decay_interval = 5_000   # halve every …  (actually multiplies by lam_decay)

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, it={it}, score={cur_score:.6f}")

        # update λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # momentum + diminishing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy descent – moves mass away from the dominant lag
# ----------------------------------------------------------------------


def _run_greedy_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple greedy local optimisation.
    While time permits, repeatedly move a small ε from the index that
    contributes most to the current worst autocorrelation term to the
    index that contributes the least.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    max_b = conv.max()
    best_score = 2.0 * n * max_b
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        k_max = int(conv.argmax())

        # range of i that participates in conv[k_max]
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)

        # contribution of each i to this lag
        contrib = a[idx] * a[k_max - idx]

        # indices of largest / smallest contribution
        i_max = idx[int(contrib.argmax())]
        i_min = idx[int(contrib.argmin())]

        if i_max == i_min:
            # nothing to move – stop
            break

        # transfer a modest fraction of the source mass
        eps = min(a[i_max] * 0.2, 0.002)  # 20 % or a tiny absolute bound
        if eps <= 0.0:
            break

        a_prop = a.copy()
        a_prop[i_max] -= eps
        a_prop[i_min] += eps

        new_max_b = _conv_self(a_prop).max()
        new_score = 2.0 * n * new_max_b

        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: n={n}, it={it}, score={new_score:.6f}")

        # Accept the move if it does not increase the objective
        if new_score <= 2.0 * n * max_b:
            a = a_prop
        # otherwise keep the old a (no acceptance of worsening moves)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (more aggressive than the original)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 1e-1,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated annealing.
    At each step pick two distinct indices i (source) and j (target) and
    move ε∈[0,0.3·a_i] from i to j.
    Acceptance follows the Metropolis rule on the true objective.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.3   # up to 30 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score

    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate) on the simplex
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv).
    Returns a dense vector of shape (n,).
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)  # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent with momentum on the probability simplex.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 1.0
    it = 0

    # lambda schedule parameters
    lam_factor = 0.5            # initial λ = lam_factor * current max
    lam_decay = 0.9
    lam_decay_interval = 5_000

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, it={it}, score={cur_score:.6f}")

        # update λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # momentum + diminishing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy descent – moves mass away from the dominant lag
# ----------------------------------------------------------------------


def _run_greedy_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple greedy local optimisation.
    While time permits, repeatedly move a small ε from the index that
    contributes most to the current worst autocorrelation term to the
    index that contributes the least.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    max_b = conv.max()
    best_score = 2.0 * n * max_b
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        k_max = int(conv.argmax())

        # range of i that participates in conv[k_max]
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)

        # contribution of each i to this lag
        contrib = a[idx] * a[k_max - idx]

        i_max = idx[int(contrib.argmax())]
        i_min = idx[int(contrib.argmin())]

        if i_max == i_min:
            break

        # transfer a modest fraction of the source mass
        eps = min(a[i_max] * 0.2, 0.002)  # 20 % or absolute cap
        if eps <= 0.0:
            break

        a_prop = a.copy()
        a_prop[i_max] -= eps
        a_prop[i_min] += eps

        new_max_b = _conv_self(a_prop).max()
        new_score = 2.0 * n * new_max_b

        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: n={n}, it={it}, score={new_score:.6f}")

        # accept the move if it does not worsen the objective
        if new_score <= 2.0 * n * max_b:
            a = a_prop

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (more aggressive than the original)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 1e-1,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated annealing.
    At each step pick two distinct indices i (source) and j (target) and
    move ε∈[0,0.3·a_i] from i to j.
    Acceptance follows the Metropolis rule on the true objective.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.3   # up to 30 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Finds a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm runs for at most `budget_s` seconds
    and combines multi‑scale smoothed gradient descent, a greedy local
    improvement, and a final simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter (if the runner provided a previously found solution)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # length schedule – modest growth, shuffled to avoid bias if stopped early
    lengths = [2400, 3200, 4800, 6400, 8000]
    np.random.shuffle(lengths)

    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise a0
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # split slice: gradient ≈30 %, greedy ≈20 %, annealing the rest
        grad_time = max(4.0, 0.30 * remaining)
        greedy_time = max(2.0, 0.20 * remaining)
        anneal_time = max(4.0, remaining - grad_time - greedy_time)

        # ------------------------------------------------------------------
        # smoothed gradient descent
        # ------------------------------------------------------------------
        grad_deadline = time.time() + grad_time
        a_opt, score_opt = _run_smooth_descent(a0, n, grad_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after gradient: n={n}, score={score_opt:.6f}")

        # ------------------------------------------------------------------
        # greedy deterministic refinement
        # ------------------------------------------------------------------
        greedy_deadline = time.time() + greedy_time
        a_greedy, score_greedy = _run_greedy_descent(a_opt, n, greedy_deadline, start_time)

        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after greedy: n={n}, score={score_greedy:.6f}")

        # ------------------------------------------------------------------
        # final annealing refinement (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + anneal_time
            # estimate number of steps from allocated time (≈200 steps/s)
            steps_est = int(anneal_time * 200)
            steps_est = max(10_000, min(500_000, steps_est))
            a_ref, score_ref = _run_annealing(
                a_greedy,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=1e-1,
                end_temp=1e-6,
                steps=steps_est,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n}, score={score_ref:.6f}")

        # remember the best distribution for the next upscale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_default = 2000
        best_seq = np.full(n_default, 1.0 / n_default)

    result = [float(x) for x in best_seq]

    # optional debugging output of the final score
    try:
        final_score = evaluate_sequence(result)
        print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")
    except Exception:
        pass

    return result
