assistantfinal**Solution Overview**

The task is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a` is a non‑negative step‑function (a list of numbers).  
Because the score is *scale‑invariant* we may normalise the vector to unit
\(L_1\)‑norm (`sum(a)=1`).  The problem then reduces to  

\[
\min_{a\ge 0,\;\sum a=1}\;2n\; \max (a*a) .
\]

The search consists of three stages  

1. **A fast power‑law sweep** to obtain a good starting vector.  
2. **Projected Adam optimisation** on the *soft‑max* approximation of the
   maximum (`\operatorname{smax}_\beta (a*a)`).  
   The gradient of the soft‑max is a weighted sum of the gradients of the
   individual convolution entries; the weighting is obtained from a soft‑max
   distribution whose temperature \(\beta\) is increased during the run.  
   After each Adam‑step we project back onto the probability simplex
   (non‑negative, unit sum).  
3. **Final evaluation** with the supplied `evaluate_sequence` and return the
   best vector found.

All heavy work is performed with real FFTs (`np.fft.rfft`) – both the
self‑convolution and the gradient can be computed in \(O(n\log n)\) time.
The whole algorithm respects the 1000 s budget (a 5 s safety margin is left).

Stage 3 – final validation with evaluate_sequence().

# Helper utilities (top‑level, no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+offset)^(−p)  (i = 0 … n‑1), normalised to unit L¹‑norm."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full self‑convolution x*x using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w≥0, Σw=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def isotonic_decreasing(v: np.ndarray) -> np.ndarray:
    """PAVA – project onto the set of non‑increasing vectors."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # blocks: (sum, count)
    sums = []
    counts = []
    for x in v:
        sums.append(x)
        counts.append(1)
        # merge while monotonicity is violated
        while len(sums) >= 2 and sums[-2] / counts[-2] < sums[-1] / counts[-1]:
            s = sums[-2] + sums[-1]
            c = counts[-2] + counts[-1]
            sums[-2] = s
            counts[-2] = c
            sums.pop()
            counts.pop()
    # expand blocks
    out = np.empty_like(v)
    pos = 0
    for s, c in zip(sums, counts):
        avg = s / c
        out[pos:pos + c] = avg
        pos += c
    return out

# Main optimisation routine

def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative step‑function yielding a low evaluation score.
    Returns the best list of heights found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1️⃣  Initialise – use a previously stored sequence if available,
    #                otherwise fall back to a short power‑law sweep.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a.sum()
        n = a.shape[0]
        print("[init] using supplied height_sequence_1 (n = {})".format(n))
    else:
        n = 2500
        print("[init] no external seed – performing a brief power‑law sweep")
        best_score = np.inf
        best_a = None
        # coarse grid – cheap but already points to a good region
        for offset in (0.3, 0.5, 1.0, 2.0):
            for p in np.arange(0.40, 0.71, 0.02):
                cand = power_law_sequence(n, p, offset)
                cur = 2.0 * n * conv_fft(cand).max()
                if cur < best_score:
                    best_score = cur
                    best_a = cand
        a = best_a if best_a is not None else np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 2️⃣  Projected Adam optimiser on the smooth maximum.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0          # safety margin
    max_iters = 200_000                             # will be cut by the timer

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5

    # soft‑max temperature (starts loose, becomes sharp)
    beta = 0.5
    beta_inc = 0.0005

    # state for Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # current best
    cur_conv = conv_fft(a)
    cur_score = 2.0 * n * cur_conv.max()
    best_score = cur_score
    best_vec = a.copy()

    # pre‑compute FFT size for the gradient step (weights length = 2n‑1)
    len_w = 2 * n - 1
    fft_len = 1 << ((len_w + n - 1).bit_length())   # next power of two

    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- convolution and soft‑max weights -----
        conv = conv_fft(a)
        max_c = conv.max()
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth max (via FFT) -----
        a_rev = a[::-1]
        W_fft = np.fft.rfft(w, n=fft_len)
        A_fft = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(W_fft * A_fft, n=fft_len)
        # slice that corresponds to the n gradient components
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # occasional tiny jitter to escape plateaus
        if rng.random() < 0.004:
            a_new += (rng.random(n) - 0.5) * 1e-6

        # ---- projection onto the feasible set ----
        a_new = project_onto_simplex(a_new)

        # (optional) enforce a non‑increasing shape – usually beneficial
        a_new = isotonic_decreasing(a_new)
        a_new /= a_new.sum()                         # re‑normalise after isotonic

        # ----- evaluate candidate -----
        new_conv = conv_fft(a_new)
        new_score = 2.0 * n * new_conv.max()

        # keep the globally best vector
        if new_score < best_score:
            best_score = new_score
            best_vec = a_new.copy()
            print(f"[iter {it:06d}] new best score = {best_score:.6f}")

        # simple acceptance rule (always keep improvements)
        if new_score <= cur_score:
            a = a_new
            cur_score = new_score

        # increase temperature → focus more on the true max
        beta += beta_inc

        it += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"[done] final reported score = {final_score:.6f}")
    print(f"[info] optimisation finished after {it:,} iterations "
          f"with n = {n}")

    return best_vec.tolist()
