assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 ,
\]

where `a*a` denotes the (linear) auto‑convolution of the vector `a` and `n=len(a)`.  
Because the objective is *scale‑invariant* we can keep the coefficients on the
probability simplex (`∑a = 1`).  Then the problem reduces to

\[
\min_{a\ge0,\; \mathbf 1^\top a = 1}\; 2n\;\max (a*a) .
\]

The previous code approximated the max with a soft‑max and used a
projected gradient descent.  Here we improve the optimisation in two ways

* **Exact sub‑gradient** – instead of a smooth approximation we compute a
  genuine sub‑gradient of the `max` using the set of indices where the
  convolution attains its maximum.
* **Adam‑style adaptive steps** – the sub‑gradient is fed to an Adam optimiser
  (first‑ and second‑moment estimates) and after each update we project back
  onto the simplex.
* **Multi‑scale search** – we start from several diverse initial shapes,
  refine each of them, then repeatedly up‑sample the current best shape
  (linear interpolation) and refine the larger problem.  
  The up‑sampling factor is ≈ 1.5; the total number of iterations is kept
  proportional to the cost of one iteration so that the whole routine stays
  well inside the 1000 s budget.

All heavy work (convolution, sub‑gradient) is performed with FFTs,
which makes one iteration `O(n log n)` even for several thousand coefficients.

The function **`propose_candidate`** follows the required signature and
returns a list of non‑negative heights (Python floats).  
During the run it prints intermediate scores and timing information.

```python
"""
Optimisation of a non‑negative step function for the evaluate_sequence
objective.

Algorithm
---------
1.  Initialise a pool of candidate shapes (uniform, triangular, Hann window,
    previous best and a random Dirichlet).  All candidates are normalised to
    lie on the probability simplex (∑a = 1).

2.  For each candidate run a projected Adam optimiser that uses the *exact*
    sub‑gradient of the infinity norm of the auto‑convolution.
    After each step the iterate is projected onto the simplex (Duchi et al.).
    The optimisation stops either after a fixed number of iterations or when
    the global time budget expires.

3.  Keep the best candidate seen so far.  Re‑scale it to a larger length by
    linear interpolation (≈ 1.5× bigger, up to a hard cap) and repeat step 2.
    Larger problems receive fewer iterations so that the total runtime stays
    within the 1000 s limit.

4.  Finally perform a long Adam run on the best length discovered.

The routine prints progress messages and returns the best sequence as a
plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_autocorr(a: np.ndarray) -> np.ndarray:
    """Linear auto‑convolution a*a using FFT (full length 2n‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _score_simplex(a: np.ndarray) -> float:
    """Score 2·n·max(conv) for a vector already on the simplex."""
    conv = _conv_autocorr(a)
    return 2.0 * a.shape[0] * conv.max()


def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of  max(a*a)  at the point a (a lies on the simplex).

    Returns g (same shape as a) such that
        max(a*a + ε·g) ≈ max(a*a) + ε·⟨g,∇max⟩
    for small ε>0.
    """
    n = a.shape[0]

    # 1) convolution and identification of maximal indices
    conv = _conv_autocorr(a)
    max_val = conv.max()
    # treat indices that are numerically equal to the maximum as active
    tol = max_val * 1e-12 + 1e-14
    mask = np.abs(conv - max_val) <= tol
    active = mask.astype(float)
    m = active.sum()
    if m == 0:
        # fallback – uniform weighting (should not happen)
        active[:] = 1.0 / active.size
    else:
        active /= m

    # 2) gradient of weighted sum wᵀ·conv (w = active)
    #    g = 2 * (w * a_rev)_{n-1 : n-1+n}
    a_rev = a[::-1]
    # length needed for convolution w (len 2n‑1) with a_rev (len n)
    fft_len = 1 << ((3 * n - 2).bit_length())
    W = np.fft.rfft(active, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len)[: len(active) + n - 1]
    g = 2.0 * conv2[n - 1 : n - 1 + n]
    return g


def _adam_projected_subgrad(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimiser with simplex projection for the objective
    f(a) = 2·n·max(a*a).  The input a0 must already be on the simplex.
    Returns (final_a, best_seen_a, best_score).
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)     # first moment
    v = np.zeros_like(a)     # second moment

    best_a = a.copy()
    best_score = _score_simplex(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # sub‑gradient of the max‑convolution
        g = _subgradient_max_conv(a)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + eps)

        a = a - step
        # must stay feasible → project onto simplex
        a = _project_simplex(a)

        # evaluate
        cur_score = _score_simplex(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random shake‑up to escape flat plateaus
        if t % 5000 == 0:
            # add tiny isotropic noise then re‑project
            noise = np.random.randn(n) * 1e-4
            a = _project_simplex(a + noise)

    return a, best_a, best_score

        print(f"[{time.time() - start_time:6.2f}s] Refining candidate {ci}/{len(cand_pool)} "
              f"n={n0}, iters={iters}")
        _, best_local, best_local_score = _adam_projected_subgrad(
            cand,
            max_iters=iters,
            deadline=deadline,
            lr=0.06,
            beta1=0.9,
            beta2=0.999,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"   >>> New global best: score={best_score:.6f} (n={n0})")

    # --------------------------------------------------------------
    # 3) progressive up‑sampling
    # --------------------------------------------------------------
    if best_seq is None:                     # extremely unlikely
        best_seq = np.full(n0, 1.0 / n0)

    current = best_seq.copy()
    cur_n = current.shape[0]
    MAX_LEN = 8000

    while True:
        # decide next size (≈1.5× bigger, capped)
        next_n = int(cur_n * 1.5)
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n <= cur_n:
            break

        # linear interpolation for up‑sampling
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a = np.interp(x_new, x_old, current)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(next_n, 1.0 / next_n)
        else:
            a = a / s

        # iteration budget (larger n → fewer iterations)
        base_iters = 40000
        iters = max(15000, int(base_iters * n0 / next_n) + 15000)

        print(f"[{time.time() - start_time:6.2f}s] Upsampling to n={next_n}, "
              f"allocating {iters} iterations")
        _, best_local, best_local_score = _adam_projected_subgrad(
            a,
            max_iters=iters,
            deadline=deadline,
            lr=0.055,
            beta1=0.9,
            beta2=0.999,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"   >>> Updated global best after up‑sampling: n={next_n}, "
                  f"score={best_score:.6f}")

        # prepare for next level
        current = a
        cur_n = next_n

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 4) final long refinement on the best length discovered
    # --------------------------------------------------------------
    if time.time() < deadline:
        n_final = best_seq.shape[0]
        iters_final = 80000
        print(f"[{time.time() - start_time:6.2f}s] Final refinement on n={n_final}, "
              f"{iters_final} iterations")
        _, best_local, best_local_score = _adam_projected_subgrad(
            best_seq,
            max_iters=iters_final,
            deadline=deadline,
            lr=0.045,
            beta1=0.9,
            beta2=0.999,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"   >>> Final improvement: score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) tidy up and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total

    print(f"[{time.time() - start_time:6.2f}s] Finished. Best score (2·n·max(conv)) = "
          f"{best_score:.6f}")
    return [float(x) for x in best_seq]
```
