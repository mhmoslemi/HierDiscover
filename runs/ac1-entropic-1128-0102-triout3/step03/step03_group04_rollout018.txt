def weighted_gradient(a, conv, beta):
    max_conv = conv.max()
    # Smoothed weights
    exp_term = np.exp(beta * (conv - max_conv))
    w = exp_term / exp_term.sum()
    # Compute term1: correlation = conv of a with reversed w (w_rev)
    w_rev = w[::-1]
    n = a.shape[0]
    # length for FFT: m = n + len(w) - 1 = n + (2n-1) -1 = 3n-2
    m = 3*n - 2
    # compute convolution via FFT
    # We'll use rfft for real
    # pad a to length m, w_rev also.
    # Use np.fft.rfft
    A = np.fft.rfft(a, n=m)
    W_rev = np.fft.rfft(w_rev, n=m)
    conv_corr = np.fft.irfft(A * W_rev, n=m)
    # term1 slice: indices offset = len(w)-1 = 2n-2
    offset = 2*n - 2
    term1 = conv_corr[offset:offset+n]  # shape n
    # term2: convolution of a and w
    W = np.fft.rfft(w, n=m)
    conv_conv = np.fft.irfft(A * W, n=m)
    term2 = conv_conv[:n]  # first n entries
    grad = term1 + term2
    # gradient of log-sum-exp = beta * grad
    return beta * grad

def mirror_descent(a_init, deadline, max_iter=20000):
    a = a_init.copy()
    n = a.shape[0]
    Lc = 2 * n - 1
    m = 3 * n - 2
    best_a = a.copy()
    best_score = np.inf
    step = 0.5
    beta = 0.5
    beta_max = 12.0
    beta_inc = (beta_max - beta) / max_iter   # linear increase
    step_decay = 0.95
    decay_every = 500
    jitter_prob = 0.002

    for it in range(max_iter):
        if time.time() > deadline:
            break
        # compute FFT of a (size m)
        A = np.fft.rfft(a, n=m)
        conv_full = np.fft.irfft(A * A, n=m)
        conv = conv_full[:Lc]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[md] iter {it:07d} new best {best_score:.6f}")

        exp_term = np.exp(beta * (conv - max_conv))
        w = exp_term / exp_term.sum()

        # compute gradient via FFT
        grad = weighted_grad_via_fft(a, w) * beta  # includes factor 2

        # exponentiated update
        a = a * np.exp(-step * grad)
        # ensure numeric stability: clip extremely small values?
        # renormalize
        sum_a = a.sum()
        if sum_a == 0 or np.isnan(sum_a):
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

import time
import numpy as np

def power_law_sequence(n, p, offset=1.0):
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a

def conv_self_fft(a):
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv

def weighted_grad_from_fft(a, w, A):
    """Compute 2 * correlation of a with w (w length 2n-1) using FFT.
    Returns gradient vector of length len(a)."""
    n = a.shape[0]
    Lw = w.shape[0]  # should be 2*n-1
    m = n + Lw - 1  # = 3*n-2
    # A = np.fft.rfft(a, n=m)   # passed as argument
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=m)
    corr_full = np.fft.irfft(A * W_rev, n=m)
    offset = Lw - 1
    term = corr_full[offset:offset + n]
    return 2.0 * term

def mirror_descent(a0, deadline, rng, max_iter=20000,
                  step_start=0.5, step_decay=0.95,
                  beta_start=0.5, beta_target=10.0,
                  decay_every=500, jitter_prob=0.002):
    """Exponentiated (mirror) descent on the simplex.
    Minimises  max(conv(a,a))  while keeping sum(a)=1.
    Returns the best sequence seen and its score."""
    a = a0.copy()
    n = a.shape[0]
    Lc = 2 * n - 1          # length of self‑convolution
    m = 3 * n - 2           # length needed for correlation FFTs
    step = step_start
    beta = beta_start
    beta_inc = (beta_target - beta_start) / max_iter
    best_a = a.copy()
    # compute initial score
    conv = conv_self_fft(a)
    best_score = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_iter:
        # FFT of a (padded to length m)
        A = np.fft.rfft(a, n=m)

        # self‑convolution
        conv_full = np.fft.irfft(A * A, n=m)
        conv = conv_full[:Lc]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[md] iter {it:07d}  best score = {best_score:.6f}")

        # soft‑max weights over all convolution entries
        exp_term = np.exp(beta * (conv - max_conv))        # each ≤ 1
        w = exp_term / exp_term.sum()                     # probability simplex

        # weighted gradient (log‑sum‑exp)  = β·2·corr(a,w)
        grad = weighted_grad_from_fft(a, w, A) * beta

        # safe clipping for exponentiation
        grad_clipped = np.clip(grad, -50.0, 50.0)

        # exponentiated update (mirror descent on KL simplex)
        a = a * np.exp(-step * grad_clipped)

        # optional jitter to escape plateaus
        if rng.random() < jitter_prob:
            a += (rng.random(n) - 0.5) * 1e-7

"""
Search for a non‑negative step‑function a that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm
---------
1. Coarse power‑law sweep (a_i ∝ (i+offset)^(‑p)) to obtain a decent
   starting shape.
2. Multi‑restart exponentiated‑gradient (mirror descent) optimisation.
   The objective max(conv) is smoothed with a full soft‑max
   (log‑sum‑exp) over the convolution entries; its gradient is
   computed efficiently by a single FFT‑based correlation.
   The update a ← a·exp(‑η·∇) keeps the vector positive, after which it
   is re‑normalised onto the probability simplex.
   Step size η decays, β (soft‑max temperature) increases, and a tiny
   jitter is occasionally added to escape plateaus.
3. The best sequence found is evaluated with the provided
   ``evaluate_sequence`` before being returned.

The whole routine respects a ~950 s internal budget (safety margin
of 5 s) and is designed to run on a single CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i = (i+offset)^(‑p) normalised to unit L1‑norm."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


def weighted_grad_from_fft(a: np.ndarray, w: np.ndarray, A: np.ndarray) -> np.ndarray:
    """
    Compute 2 · corr(a,w) where w is a probability vector of length 2·n‑1.
    Returns the gradient ∇_a Σ_k w_k·(a*a)_k .
    """
    n = a.shape[0]
    Lw = w.shape[0]            # should be 2·n‑1
    m = n + Lw - 1           # = 3·n‑2, length needed for linear correlation
    # FFT of reversed weight vector (zero‑padded to length m)
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=m)
    # correlation a ⋆ w (real part is enough)
    corr_full = np.fft.irfft(A * W_rev, n=m)
    offset = Lw - 1          # = 2·n‑2
    term = corr_full[offset:offset + n]     # length n
    return 2.0 * term        # factor 2 = ∂(a*a)_k/∂a_i


# ----------------------------------------------------------------------
# Mirror‑descent optimiser
# ----------------------------------------------------------------------


def mirror_descent(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator,
                  max_iter: int = 20000,
                  step_start: float = 0.5,
                  step_decay: float = 0.95,
                  beta_start: float = 0.5,
                  beta_target: float = 10.0,
                  decay_every: int = 500,
                  jitter_prob: float = 0.002) -> tuple[np.ndarray, float]:
    """
    Exponentiated (mirror) descent on the simplex.
    Minimises  max(conv(a,a))  while keeping  sum(a)=1.
    Returns the best sequence encountered and its (approximate) score
    2·n·max(conv).
    """
    a = a0.copy()
    n = a.shape[0]
    Lc = 2 * n - 1                # length of a*a
    m = 3 * n - 2                 # length for FFT‑based correlation
    step = step_start
    beta = beta_start
    beta_inc = (beta_target - beta_start) / max_iter

    # initial score
    conv = conv_self_fft(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iter:
        # FFT of current a (padded to length m)
        A = np.fft.rfft(a, n=m)

        # self‑convolution a*a
        conv_full = np.fft.irfft(A * A, n=m)
        conv = conv_full[:Lc]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # keep global best seen by this run
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[md] iter {it:07d}  best score = {best_score:.6f}")

        # soft‑max weights over all convolution entries
        exp_term = np.exp(beta * (conv - max_conv))          # ≤ 1
        w = exp_term / exp_term.sum()                       # probability simplex

        # gradient of the log‑sum‑exp (β·2·corr(a,w))
        grad = weighted_grad_from_fft(a, w, A) * beta

        # clip to avoid overflow in the exponentials
        grad = np.clip(grad, -50.0, 50.0)

        # exponentiated update (mirror descent on KL‑divergence)
        a = a * np.exp(-step * grad)

        # occasional tiny jitter to help escape flat regions
        if rng.random() < jitter_prob:
            a += (rng.random(n) - 0.5) * 1e-7
            a = np.maximum(a, 0.0)

        # project back onto the probability simplex
        a_sum = a.sum()
        if a_sum <= 0.0 or np.isnan(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # schedule step‑size decay and β increase
        if (it + 1) % decay_every == 0:
            step *= step_decay
            beta = min(beta + beta_inc * decay_every, beta_target)

        it += 1

    return best_a, best_score
