def _run_eg_smoothing(a0, n, deadline, start):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    lr0 = 0.5  # base

        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start:6.2f}s] EG improve: n={n}, iter={it}, score={score:.6f}")

        # compute gradient
        # we can compute w using alpha til maybe update alpha slowly
        # Increase alpha gradually to focus on max
        if alpha < alpha_max:

        # compute gradient via smooth weighting
        g = _smooth_gradient_from_conv(a, conv, alpha)

        # learning rate schedule
        lr = lr0 / math.sqrt(it)

        # exponentiated update
        # clamp g to avoid overflow
        # safe: g may be large; multiply by -lr may be small; but compute exp(-lr * g)
        exp_factor = np.exp(-lr * g)

"""
Propose a step‑function (non‑negative list) that minimizes
`evaluate_sequence`.  The algorithm works in three phases:

1. **Upsample the best known sequence** (or fall back to a uniform start)
   to a larger length (default 4000).  The vector is kept on the simplex
   (non‑negative, sum = 1).

2. **Exponentiated‑gradient descent with a soft‑max smoothing of the
   max‑convolution**.  The objective  f(a)=2·n·max(a*a)  is convex.
   Instead of using the sub‑gradient from the active peak only, we
   replace the max by a smooth weighted sum
   wₖ ∝ exp(α·convₖ).  As α grows the weights concentrate on the true
   maximum, giving a gradually sharpening direction.  The update
   aᵢ ← aᵢ·exp(−η·gᵢ) (followed by renormalisation) stays on the
   simplex and avoids projection overhead.

3. **Short simulated‑annealing refinement** (tiny mass‑transfer moves)
   to escape any remaining local minima.

All work is done inside the allotted 1000 s budget; intermediate
progress and scores are printed for debugging.  The final sequence is
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolutions (linear, non‑circular) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


def _corr_w_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the (valid) correlation of w (length 2n‑1) with a (length n):
    g_i = Σ_j w_{i+j} a_j  ,  i = 0..n‑1.
    Returned array has length n.
    """
    n = a.shape[0]
    m = w.shape[0]            # = 2n‑1
    L = _next_pow2(n + m - 1)  # length needed for exact linear correlation
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(np.conj(W) * A, n=L)
    corr_full = corr_full[: n + m - 1]  # keep only the linear part
    start = n - 1
    return corr_full[start:start + n]


# ----------------------------------------------------------------------
# Smoothed gradient (soft‑max over convolution entries)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return a smooth approximation of the sub‑gradient of
    f(a)=2·n·max(conv) using a soft‑max weighting with temperature 1/α.
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights, shifted to avoid overflow
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient = 2 * correlation(w, a)
    g = 2.0 * _corr_w_a(w, a)
    return g


# ----------------------------------------------------------------------
# Exponential‑gradient descent with annealing α
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray, n: int,
                      deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Optimise the sequence using a soft‑max smoothed gradient.
    Returns the best sequence encountered and its score (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    lr0 = 0.5           # base learning rate
    alpha = 0.5         # start temperature (softness)
    alpha_max = 200.0   # maximum softness (approaches true max)

    # main loop
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # gradually sharpen the soft‑max
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.001)

        # smooth gradient
        g = _smooth_gradient(a, conv, alpha)

        # learning‑rate schedule (1/√t)
        lr = lr0 / math.sqrt(it)

        # multiplicative update (exponentiated gradient)
        # Clip gradient to avoid overflow in the exponent
        np.clip(g, -1e6, 1e6, out=g)
        factor = np.exp(-lr * g)
        a = a * factor

        # renormalise onto the simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny ε‑mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 80000) -> tuple[np.ndarray, float]:
    """
    Local refinement by moving a random ε from one index to another.
    Accepts worsening moves with Metropolis probability.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a mass transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) to a new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        out = seq.copy()
    else:
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, new_n)
        out = np.interp(new_x, old_x, seq)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.full_like(out, 1.0 / new_n)
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The method follows the three‑phase plan
    described in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Load a starter sequence if the harness provides one
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0:
            starter /= s
        else:
            raise ValueError
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose a target length (larger lengths give more flexibility)
    # ------------------------------------------------------------------
    n_target = 4000

    if starter is not None:
        a0 = _upsample(starter, n_target)
    else:
        a0 = np.random.rand(n_target)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Exponential‑gradient descent with soft‑max smoothing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly 80 % of the remaining budget to gradient descent
        grad_time = max(10.0, 0.80 * remaining)
        grad_deadline = time.time() + grad_time

        seq_grad, score_grad = _run_exp_gradient(a0, n_target, grad_deadline, start_time)
        if score_grad < best_score:
            best_score = score_grad
            best_seq = seq_grad.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase‑1 improve: score={score_grad:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Local stochastic refinement (annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _run_annealing(
            best_seq, n_target,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5,
            steps=120000   # plenty of steps, but will stop early on timeout
        )
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety normalisation and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
