"""
Propose a candidate step‑function sequence that (approximately) minimises

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three stages:

1. **Fine parametric sweep** – explores power‑law and exponential families
   with a very fine grid (p, offset, λ) and keeps the best sequence found.
2. **Adam‑based projected gradient descent** – uses the exact sub‑gradient
   of the active convolution entry (the current maximum) and adapts the
   step‑size with an Adam‐style optimiser.  After each update the vector is
   projected onto the probability simplex to keep it feasible.
3. **Simulated‑annealing local moves** – repeatedly transfers a tiny amount
   of mass between two random coordinates and accepts the move either if it
   improves the score or with a temperature‑dependent probability.  This
   helps escape shallow local minima.

The whole routine respects a user‑provided time budget (default ≈950 s) and
returns the best sequence encountered.  All helper functions are defined at
module level (no closures) and only standard library / NumPy are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities -------------------------------------------------------
# ----------------------------------------------------------------------


def power_law_sequence(n: int, p: float, offset: float = 0.5) -> np.ndarray:
    """Return a non‑negative vector a_i = (i+offset)^(-p) (i = 0 … n‑1)
    normalised to have L1‑norm 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Return a vector a_i ∝ exp(-lam·i) normalised to L1‑norm 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full self‑convolution x*x using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w ≥ 0, Σw = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Exact sub‑gradient of the convolution entry conv[peak_idx] w.r.t. a.
    The gradient is g_i = 2·a_{peak_idx-i} for all i where the index is valid."""
    n = a.shape[0]
    grad = np.zeros_like(a)

    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


def fast_score(a: np.ndarray, n: int) -> float:
    """Score assuming sum(a) == 1 : 2·n·max(conv(a,a))."""
    conv = conv_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the (budget_s) time budget and returns the best
    list found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------- config
    n = 2500                       # length of the candidate vector
    safety_margin = 5.0
    deadline = time.time() + max(0.0, budget_s - safety_margin)

    # ------------------------------------------------------------------ init
    # try to use the previously discovered sequence if available
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if init_seq.shape[0] != n:
            # truncate / pad to the required length and renormalise
            if init_seq.shape[0] > n:
                init_seq = init_seq[:n]
            else:
                pad = np.zeros(n - init_seq.shape[0], dtype=np.float64)
                init_seq = np.concatenate([init_seq, pad])
            init_seq = init_seq / init_seq.sum()
        best_a = init_seq.copy()
        best_score = fast_score(best_a, n)
        print(f"[init] using height_sequence_1, score = {best_score:.6f}")
    else:
        best_a = power_law_sequence(n, p=0.5, offset=0.5)
        best_score = fast_score(best_a, n)
        print(f"[init] default power‑law seed, score = {best_score:.6f}")

    # ------------------------------------------------------------------ phase 1
    print("[phase 1] fine parametric sweep")
    # ----- power‑law grid (very fine around the known good region)
    p_vals = np.arange(0.470, 0.540, 0.001)          # ~70 values
    off_vals = np.arange(0.40, 0.61, 0.01)           # 21 values
    for off in off_vals:
        for p in p_vals:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=off)
            sc = fast_score(a, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[grid] new best p={p:.3f} off={off:.2f} → {sc:.6f}")

    # ----- exponential family
    lam_vals = np.logspace(-4, -0.1, num=30)          # from 1e‑4 to ~0.79
    for lam in lam_vals:
        if time.time() > deadline:
            break
        a = exponential_sequence(n, lam)
        sc = fast_score(a, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[exp] new best λ={lam:.5f} → {sc:.6f}")

    # ------------------------------------------------------------------ phase 2
    print("[phase 2] Adam‑based projected gradient descent")
    max_iter = 250_000
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 0.12               # initial learning rate (will decay)
    eta = eta0
    m = np.zeros_like(best_a)
    v = np.zeros_like(best_a)

    a = best_a.copy()
    cur_score = fast_score(a, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        # compute convolution and the index of the active peak
        conv = conv_fft(a)
        peak_idx = int(conv.argmax())
        grad = gradient_of_peak(a, peak_idx)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_onto_simplex(a_new)

        new_score = fast_score(a_new, n)

        # record global improvement
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[adam] iter {it:07d} global best = {best_score:.6f}")

        # accept if improvement; otherwise keep the old a
        if new_score <= cur_score:
            a = a_new
            cur_score = new_score
        else:
            # optional tiny probabilistic acceptance (simulated‑annealing style)
            temp = 0.01 * np.exp(-it / 150_000.0)
            if rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                a = a_new
                cur_score = new_score

        # occasional jitter to avoid stagnation
        if rng.random() < 0.003:
            jitter = (rng.random(n) - 0.5) * 1e-7
            a = project_onto_simplex(a + jitter)

        # learning‑rate schedule
        if (it + 1) % 5000 == 0:
            eta = max(eta * 0.85, 1e-6)

        it += 1

    print(f"[phase 2] finished – local best score = {cur_score:.6f}")

    # ------------------------------------------------------------------ phase 3
    print("[phase 3] simulated‑annealing local refinement")
    a = best_a.copy()
    cur_score = fast_score(a, n)

    moves = 0
    max_moves = 80_000
    while time.time() < deadline and moves < max_moves:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # mass to transfer (a small fraction of source mass)
        delta = min(a[i], 1e-4 * rng.random())
        if delta == 0.0:
            moves += 1
            continue
        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        # sum stays 1, non‑negativity guaranteed
        trial_score = fast_score(a_trial, n)

        if trial_score < cur_score or rng.random() < np.exp(-(trial_score - cur_score) / max(0.005 * (1.0 - moves / max_moves), 1e-12)):
            a = a_trial
            cur_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_a = a_trial.copy()
                print(f"[anneal] move {moves:07d} new global best = {best_score:.6f}")

        moves += 1

    print(f"[phase 3] finished – best score = {best_score:.6f}")

    # --------------------------------------------------------------- final check
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] reported score by evaluator = {final_score:.6f}")

    return best_a.tolist()
