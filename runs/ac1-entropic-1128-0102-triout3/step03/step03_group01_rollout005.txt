"""
Propose a step‑function (non‑negative list of heights) that minimizes
`evaluate_sequence`.  The algorithm is a multi‑scale optimisation:

* Start from a warm‑start (previous best if available) and successively
  upscale the sequence (800 → 1 600 → 3 200 → 6 400 → 9 600).
* On each scale run a smoothed ℓ∞‑surrogate descent (log‑sum‑exp
  approximation) with Nesterov‑style momentum and Euclidean projection
  onto the probability simplex.
* After the gradient phase a cheap simulated‑annealing sweep (mass
  transfers between two coordinates) is applied.
* When the time budget is almost exhausted a second phase runs a
  mirror‑descent optimiser (exponential updates on the simplex) together
  with a short annealing polish.  The mirror step is well‑suited for the
  simplex and often yields a finer reduction of the max‑convolution.
* All stages constantly keep the best sequence seen and finally return it.

The implementation stays inside the 1000 s time budget, works with a
single CPU and uses only NumPy (no external solvers).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x ≥ 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of x and y using a real FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed ℓ∞ surrogate (log‑sum‑exp) utilities
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a) =
        λ·log Σ exp(conv/λ) .
    Returns a dense gradient of shape (n,).
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights (stable)
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]        # d/d a_i
    return grad


# ----------------------------------------------------------------------
# Euclidean smoothed gradient descent with momentum (Nesterov‑like)
# ----------------------------------------------------------------------


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Smoothed gradient descent on the simplex (Euclidean projection)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 0.8
    lam_factor = 0.25        # initial λ = lam_factor * max(conv)
    lam_decay = 0.5
    decay_interval = 2_000

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # sum(a)=1 ⇒ objective = 2·n·max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, it={it}, score={cur_score:.6f}")

        # λ schedule
        if it % decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # momentum & diminishing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑descent optimiser on the simplex (exponential updates)
# ----------------------------------------------------------------------


def _run_mirror_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iter: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Mirror descent with KL (exponential) updates on the simplex.
    Uses the same log‑sum‑exp surrogate for the gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # schedule parameters (similar to the Euclidean version)
    lam_factor = 0.25
    lam_decay = 0.5
    decay_interval = 2_000

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] mirror improve: n={n}, it={it}, score={cur_score:.6f}")

        # λ update
        if it % decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # step size – diminishing as 1/√t
        eta = 0.5 / math.sqrt(it)

        # exponential (mirror) update, clipped to avoid overflow
        grad_clipped = np.clip(grad, -80.0, 80.0)
        a = a * np.exp(-eta * grad_clipped)

        # renormalise to the simplex
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing: move a small ε from i to j.
    Returns possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The algorithm combines multi‑scale log‑sum‑exp gradient descent,
    mirror‑descent fine tuning and short simulated‑annealing sweeps.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from the previous run (global variable)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter /= starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – a few geometric steps, shuffled to avoid bias.
    lengths = [800, 1600, 3200, 6400, 9600]
    np.random.shuffle(lengths)

    global_a = starter

    # ------------------------------------------------------------------
    # Coarse multi‑scale optimisation
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise a0
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # allocate a time slice for this resolution
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_deadline = time.time() + max(8.0, 0.13 * remaining)

        # smoothed gradient descent (Euclidean)
        a_opt, score_opt = _run_smooth_descent(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after gradient: n={n}, score={score_opt:.6f}")

        # short annealing refinement (if we still have time)
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, 0.04 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=30_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n}, score={score_ref:.6f}")

        # keep the current best as warm‑start for the next scale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fine‑tuning phase (mirror descent + final annealing)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # possibly upscale a little (but never exceed ~20000)
        n_cur = best_seq.shape[0]
        if np.random.rand() < 0.2 and n_cur < 20000:
            new_n = min(20000, int(n_cur * 1.5))
            a0 = _upsample(best_seq, new_n)
            n = new_n
        else:
            # small random perturbation of the current best
            a0 = best_seq.copy()
            eps = 0.02
            noise = (np.random.rand(n_cur) - 0.5) * eps
            a0 = np.maximum(a0 + noise, 0.0)
            a0 /= a0.sum()
            n = n_cur

        # allocate a short slice (up to 30 s) for the mirror optimisation
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(30.0, 0.5 * remaining)

        a_opt, score_opt = _run_mirror_descent(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after mirror: n={n}, score={score_opt:.6f}")

        # one final quick annealing sweep
        if time.time() < deadline:
            anneal_deadline = time.time() + min(10.0, deadline - time.time())
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=1e-2,
                end_temp=1e-6,
                steps=10_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after final annealing: n={n}, score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final sanity check (optional, uncomment if you wish a log line)
    # print(f"Final best score (internal eval) = {best_score:.6f}")

    # return plain Python list – the outer harness will feed it to evaluate_sequence
    return [float(x) for x in best_seq]
