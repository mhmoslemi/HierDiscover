```
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    lam_factor = 0.2
    lam_decay = 0.5
    lam_interval = 2000
    min_lam = 1e-12

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b / (np.sum(a)**2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, it={it}, score={cur_score:.6f}")

        # update lam
        if it % lam_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, min_lam)

        # project onto simplex
        a = _project_simplex(a)
```

```
                   start_temp=5e-2, end_temp=1e-5, steps=40000):

    best_score = cur_best_score

            break

        # propose either mass transfer or swap
        if np.random.rand() < 0.8:

            a_prop[i] -= eps
```

assistantfinal**Solution Overview**

The task is to return a non‑negative step‑function (a list of heights) that minimises  

\[
\text{score}(a)=\frac{2\,n\,\max (a*a)}{(\sum_i a_i)^2},
\qquad n=\text{len}(a)\; .
\]

The score is *scale‑invariant*; we can therefore keep the coefficients on the
probability simplex (\(\sum_i a_i = 1\)).  
With this normalisation the score reduces to  

\[
\text{score}(a)=2\,n\; \max (a*a) .
\]

So the optimisation problem is:  
*find a probability vector whose self‑convolution has the smallest possible
\(\ell_\infty\)‑norm*.

The previous solution used a log‑sum‑exp surrogate together with a plain
momentum scheme.  The new approach keeps the same multi‑scale schedule but
replaces the optimiser by **Adam** (adaptive moments) which converges faster
on the smooth surrogate.  After the Adam phase we run a more aggressive
simulated‑annealing phase that mixes *mass‑transfer* moves with *pair swaps*
to explore the ordering of the coefficients.

Key ideas
---------

* **Simplex projection** – all iterates are projected onto the probability
  simplex after every update (Duchi et al. 2008).  
* **Smoothed objective** – the non‑smooth \(\max\) is approximated by the
  log‑sum‑exp

  \[
  L_\lambda(a)=\lambda\log\!\Bigl(\sum_{k}\exp\bigl((a*a)_k/\lambda\bigr)\Bigr)
  \]

  with a decreasing smoothing parameter \(\lambda\).
* **Adam optimizer** – adaptive learning rates accelerate convergence on the
  smooth surrogate.  The gradient of \(L_\lambda\) is obtained by a weighted
  cross‑correlation that can be evaluated with two FFTs.
* **Multi‑scale warm‑starts** – we solve the problem for an increasing list of
  lengths \([2000,4000,8000,12000,16000]\).  The best solution of a smaller
  scale is linearly interpolated (upsampled) and used as the starting point
  for the next scale.
* **Final annealing** – a cheap simulated‑annealing sweep (mass transfer +
  swaps) finishes the run and can escape the remaining local minima.

All code respects the 1000 s budget, uses only a single CPU and only the
standard scientific Python stack.

```python
"""
Multi‑scale Adam optimisation + aggressive simulated‑annealing
to minimise 2·n·max(a*a)  (equivalently evaluate_sequence).

The algorithm:
  * works on the probability simplex (∑a_i = 1);
  * smoothes the ℓ∞‑norm with a log‑sum‑exp surrogate;
  * computes its gradient via FFT‑based cross‑correlation;
  * optimises the surrogate with Adam (adaptive moments);
  * upsamples the best solution to larger problem sizes;
  * finishes with a simulated‑annealing sweep that mixes mass‑transfer
    and pair‑swap moves.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (real‑to‑real) using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to Σ=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up[:] = 1.0 / new_n
    return up


# ----------------------------------------------------------------------
# Smoothed surrogate and its gradient
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ.
    Returned shape = a.shape.
    """
    n = a.shape[0]
    max_val = conv.max()
    # numerically stable soft‑max over the convolution entries
    w = np.exp((conv - max_val) / lam)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # cross‑correlation w * a_rev  (length 3n‑2)
    full = _conv_fft(w, a[::-1])
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the log‑sum‑exp surrogate.
    Returns the best vector seen (on the simplex) and its exact score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                     # base learning rate for Adam

    # λ‑schedule parameters
    lam_factor = 0.2             # λ = lam_factor * max(conv)
    lam_decay = 0.5
    lam_interval = 2000          # halve λ every 2000 iterations
    min_lam = 1e-12

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        # exact evaluation (sum(a) is 1 after every projection)
        cur_score = 2.0 * n * max_b / (np.sum(a) ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, it={it}, "
                  f"score={cur_score:.6f}")

        # update λ
        if it % lam_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, min_lam)

        # gradient of the surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam update (biased first & second moments)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_simplex(a)      # keep a on the probability simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Final simulated‑annealing refinement
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 40_000) -> tuple[np.ndarray, float]:
    """
    Simple annealing that mixes:
      * mass transfer (ε from i → j)
      * pair swap (exchange a_i and a_j)
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # decide move type
        if np.random.rand() < 0.8:          # mass transfer (most frequent)
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.3   # up to 30 % of source mass
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                # pure swap
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            a_prop = a.copy()
            a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        # evaluate the proposal
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, "
                  f"it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Returns a list of heights that (empirically) minimises `evaluate_sequence`.
    The routine respects the time budget, uses multiple length scales,
    Adam optimisation on a smooth surrogate and a final annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from a previous run (if the global variable exists)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – we go up to ~16000 items (still cheap enough for FFT)
    lengths = [2000, 4000, 8000, 12000, 16000]
    np.random.shuffle(lengths)          # avoid systematic bias if we stop early

    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # initialise a0 (probability vector of length n)
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # allocate a slice of the remaining budget for this length
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # each scale gets at most ~15 % of the remaining time, but never less
        # than a few seconds (to allow at least one Adam pass).
        slice_deadline = time.time() + max(6.0, 0.15 * remaining)

        # ------------------------------------------------------------------
        # Adam (smooth surrogate) optimisation
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_adam_descent(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after Adam: "
                  f"n={n}, score={score_opt:.6f}")

        # ------------------------------------------------------------------
        # Final annealing refinement (if time remains)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, 0.05 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=40_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: "
                      f"n={n}, score={score_ref:.6f}")

        # keep the best vector to up‑sample for the next (larger) scale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final conversion to ordinary Python list of floats
    return [float(x) for x in best_seq]
```
