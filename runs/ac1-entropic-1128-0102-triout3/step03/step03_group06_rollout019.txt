```
def fast_score(a):
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:2*n-1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s*s))
```

```
def hill_climb_set(S, n, max_iter=2000):
    S = np.array(S, dtype=int)
    k = S.size
    # precompute current counts
    counts = compute_counts(S, n)
    max_c = int(counts.max())
    # maintain set for O(1) membership testing
    S_set = set(S.tolist())
    for _ in range(max_iter):
        if max_c <= target_allowed: break
        # find indices causing max count
        # pick a random sum where count == max_c
        sums_max = np.where(counts == max_c)[0]
        sum_idx = int(np.random.choice(sums_max))
        # Determine participants: elements i in S such that sum_idx-i in S
        participants = [i for i in S if 0 <= sum_idx - i < n and (sum_idx - i) in S_set]
        if not participants:
            # fallback to random element
            i_remove = int(np.random.choice(S))
        else:
            i_remove = int(np.random.choice(participants))
        # propose new element not in S
        for trial in range(20):
            new = np.random.randint(0, n)
            if new in S_set:
                continue
            # test new set
            S_new = S.copy()
            S_new[S_new == i_remove] = new
            cnt_new = compute_counts(S_new, n)
            new_max = int(cnt_new.max())
            # Accept if new_max < max_c
            if new_max < max_c:
                S = S_new
                S_set.remove(i_remove)
                S_set.add(new)
                counts = cnt_new
                max_c = new_max
                break
    return S, max_c
```

```
def search_sparse_set(n, k, target_score, time_limit):
    allowed_max = int(math.floor(target_score * k * k / (2 * n)))
    if allowed_max < 1: allowed_max = 1
    deadline = time.time() + time_limit
    best_S, best_L = None, 10**9
    attempts = 0
    while time.time() < deadline and attempts < 5000:
        attempts += 1
        S = np.random.choice(n, size=k, replace=False)
        L = compute_counts(S, n).max()
        # quick prune: if L <= allowed_max: success
        if L <= allowed_max:
            return S, L
        # else attempt hill climb to reduce L
        S2, L2 = hill_climb_set(S, n, max_iter=1000)
        if L2 < best_L:
            best_S, best_L = S2, L2
        if L2 <= allowed_max:
            return S2, L2
    return best_S, best_L
```

def subgradient_optimize(a_init, time_limit):
    a = a_init.copy()
    n = a.size
    # precompute FFT size
    conv_sz = 1 << ((2*n-1).bit_length())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = fast_score(a)
    deadline = time.time() + time_limit - 0.1
    while time.time() < deadline:
        t += 1
        # compute convolution
        A = np.fft.rfft(a, conv_sz)
        conv = np.fft.irfft(A * A, conv_sz)[:2*n-1]
        k_max = int(np.argmax(conv))
        # subgradient
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 / (np.sqrt(v_hat) + eps)   # base lr 0.1
        a = a - step * m_hat
        # projection onto simplex
        a = simplex_projection(a)
        # occasional evaluation
        if t % 1000 == 0:
            cur = fast_score(a)
            if cur < best_val:
                best_val, best_a = cur, a.copy()
    # final best
    return best_a

def simplex_projection(v):
    # Project onto simplex {x >=0, sum = 1}
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - 1
    rho = np.where(v_sorted - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Proposes a low‑score step function.

The method consists of three stages inside the 1000 s budget:

1. **Random sparse support generation** – pick a random subset S
   (size roughly 80‑150) and put uniform mass 1/|S| on it.
   The score of such a vector is cheap to evaluate.

2. **Local hill‑climbing on the support** – repeatedly try to
   replace an element that participates in the current maximal
   convolution count by a random new index.  The counts are updated
   by recomputing the pair‑sum histogram; this quickly reduces the
   maximal count and often brings the score below the target
   (1.503).

3. **Projected Adam refinement** – starting from the best vector
   found, perform a few seconds of ADAM‑type sub‑gradient descent
   on the objective  f↦max (f*f).  After each step the vector is
   projected onto the probability simplex (non‑negative, sum = 1).

If the early stages already achieve a score ≤ 1.503 the function
returns immediately; otherwise the final Adam optimisation is used
as a fallback.  All helper functions are defined at top level and
no I/O is performed.
"""

import numpy as np
import math
import time
import random

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of the provided evaluate_sequence (no security checks).
    a must be non‑negative; the function normalises nothing.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # length for a power‑of‑two FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """Ordered‑pair histogram of a support set S (indices in [0,n))."""
    sums = np.add.outer(S, S)               # all i+j
    return np.bincount(sums.ravel(), minlength=2 * n - 1)


# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def adam_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Projected Adam optimiser for the max‑convolution objective.
    Stops when the allocated time expires and returns the best vector seen.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    fft_size = 1 << ((2 * n - 1).bit_length())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12
    it = 0
    deadline = time.time() + time_limit - 0.2

    while time.time() < deadline:
        it += 1
        # convolution and index of the maximal entry
        A = np.fft.rfft(a, fft_size)
        conv = np.fft.irfft(A * A, fft_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active index
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex (sum = 1, non‑negative)
        a = simplex_projection(a)

        # occasional bookkeeping
        if it % 800 == 0:
            cur = fast_score(a)
            if cur < best_val:
                best_val, best_a = cur, a.copy()

    # final check
    final = fast_score(best_a)
    if final < best_val:
        best_a = best_a
    return best_a


# ----------------------------------------------------------------------
def hillclimb_support(S: np.ndarray, n: int, max_iter: int = 1500):
    """
    Local search on a support set S.
    Repeatedly swaps an element that participates in the current
    maximal convolution count with a random new index, accepting
    the swap if the maximal count does not increase.
    Returns the improved set and its maximal ordered‑pair count.
    """
    S = np.array(S, dtype=int)
    k = S.size
    S_set = set(S.tolist())
    counts = compute_counts(S, n)
    max_c = int(counts.max())

    for _ in range(max_iter):
        if max_c <= 1:
            break
        # pick a sum that attains the current maximum
        sums_max = np.where(counts == max_c)[0]
        sum_idx = int(np.random.choice(sums_max))

        # elements from S that participate in this sum
        participants = [i for i in S if 0 <= sum_idx - i < n and (sum_idx - i) in S_set]
        if participants:
            i_remove = int(np.random.choice(participants))
        else:
            i_remove = int(np.random.choice(S))

        # try a few random replacements
        improved = False
        for _ in range(7):
            i_new = np.random.randint(0, n)
            if i_new in S_set:
                continue
            S_new = S.copy()
            S_new[S_new == i_remove] = i_new
            cnt_new = compute_counts(S_new, n)
            new_max = int(cnt_new.max())
            if new_max <= max_c:
                S = S_new
                S_set.remove(i_remove)
                S_set.add(i_new)
                counts = cnt_new
                max_c = new_max
                improved = True
                break
        if not improved:
            # no improvement – random restart of the whole set occasionally
            if random.random() < 0.02:
                S = np.random.choice(n, size=k, replace=False)
                S_set = set(S.tolist())
                counts = compute_counts(S, n)
                max_c = int(counts.max())
    return S, max_c

                if refined_val < best_val:
                    best_val, best_seq = refined_val, refined.copy()
                    print(f"[refine] improved to {refined_val:.6f}")
                    if refined_val <= target_score:
                        return best_seq.tolist()

    # ------------------------------------------------------------------
    # Stage 2 – final Adam optimisation on the best candidate found
    remaining = deadline - time.time()
    if remaining > 4.0 and best_seq is not None:
        best_seq = adam_optimize(best_seq, remaining)
        best_val = fast_score(best_seq)

    # ------------------------------------------------------------------
    # safety net – uniform distribution (score = 2) if nothing else succeeded
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_val = fast_score(best_seq)

    print(f"[final] best score returned = {best_val:.6f}")
    return best_seq.tolist()
