"""
Propose a low‑score step‑function using a two‑phase projected gradient descent.
The first phase uses a soft‑max approximation of the max‑convolution to obtain a
smooth descent; the second phase switches to the exact sub‑gradient (the true
max‑convolution) for fine‑tuning.  The algorithm works on the probability
simplex (∑a_i = 1, a_i ≥ 0) so the evaluation function is scale‑invariant.
Multiple diverse starts are tried, the best shape is progressively up‑sampled,
and a final long refinement is performed.  The implementation stays well
within the 1000 s time budget and prints intermediate improvements.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (length 2*len(a)-1) using FFT.
    Returned array is of dtype float64.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _score_simplex(a: np.ndarray) -> float:
    """
    Score when a already lies on the simplex (∑a_i = 1, a_i ≥ 0):
        score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_full(a)
    return 2.0 * n * conv.max()


def _grad_softmax(a: np.ndarray, conv: np.ndarray, max_val: float,
                  eps_rel: float) -> np.ndarray:
    """
    Gradient of the smooth (soft‑max) approximation of max(conv).
    Implements the same formula as the reference implementation.
    """
    # temperature proportional to current max
    tau = max(eps_rel * max_val, 1e-14)
    w = np.exp((conv - max_val) / tau)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w.fill(1.0 / w.shape[0])

    # gradient = 2 * (w * a_rev) convolved (see reference)
    n = a.shape[0]
    a_rev = a[::-1]
    fft_len = 1 << ((len(w) + n - 2).bit_length())
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len)[: len(w) + n - 1]
    grad = 2.0 * conv2[n - 1 : n - 1 + n]   # slice exactly n entries
    return grad


def _grad_exact_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact gradient of max(conv) – pick one index attaining the maximum
    (the first one) and return ∂conv_k/∂a.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))                     # index of current max
    idx = k - np.arange(n)                       # k - i
    mask = (0 <= idx) & (idx < n)
    grad = np.zeros_like(a)
    # term a_{k-i}
    grad[mask] += a[idx[mask]]
    # term a_{i-k}
    idx2 = np.arange(n) - k
    mask2 = (0 <= idx2) & (idx2 < n)
    grad[mask2] += a[idx2[mask2]]
    return 2.0 * grad


def _refine_sequence(a: np.ndarray,
                     max_iters: int,
                     deadline: float,
                     step0: float = 0.6,
                     eps_start: float = 0.05,
                     eps_end: float = 0.001,
                     beta: float = 0.9,
                     soft_frac: float = 0.5) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Two‑phase projected gradient descent on a fixed length.

    Parameters
    ----------
    a : np.ndarray
        Starting point (already on the simplex).
    max_iters : int
        Upper bound on the number of iterations.
    deadline : float
        Unix timestamp – stop early if the time budget is exhausted.
    step0 : float
        Base step‑size (scaled by 1/√t).
    eps_start / eps_end : float
        Temperature schedule for the soft‑max phase.
    beta : float
        Momentum coefficient (Nesterov‑style).
    soft_frac : float
        Fraction of the iterations that use the soft‑max approximation.
        Remaining iterations use the exact sub‑gradient of the max.

    Returns
    -------
    final_a : np.ndarray
        The final iterate (projected on the simplex).
    best_a : np.ndarray
        The best point seen during the run.
    best_score : float
        Score of best_a (2·n·max(conv)).
    """
    n = a.shape[0]
    v = np.zeros_like(a)          # momentum buffer
    best_a = a.copy()
    best_score = _score_simplex(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- compute convolution and current score -----
        conv = _conv_full(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- decide which gradient to use -----
        if it <= int(max_iters * soft_frac):
            # soft‑max gradient (smooth approximation)
            eps_rel = eps_start + (eps_end - eps_start) * (it / max_iters)
            grad = _grad_softmax(a, conv, max_val, eps_rel)
        else:
            # exact sub‑gradient of the max‑convolution
            grad = _grad_exact_max(a, conv)

        # ----- Nesterov‑style momentum & projection -----
        step = step0 / math.sqrt(it)
        v = beta * v - step * grad
        a = _project_simplex(a + v)

    return a, best_a, best_score


# ----------------------------------------------------------------------
def _upscale(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation up‑sampling of a simplex‑vector to length new_n."""
    old_n = a.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0:
        a_new = np.full(new_n, 1.0 / new_n)
    else:
        a_new = a_new / s
    return a_new

            if s > 0:
                prev = prev / s
                candidates.append(prev)
        except Exception:
            pass

    # b) uniform
    candidates.append(np.full(n0, 1.0 / n0))

    # c) tent (triangular) shape
    idx = np.arange(n0)
    tent = np.minimum(idx + 1, n0 - idx)
    candidates.append(tent / tent.sum())

    # d) Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n0 - 1)))
    candidates.append(hann / hann.sum())

    # e) a random Dirichlet draw
    candidates.append(np.random.dirichlet(np.ones(n0)))

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Refine each starter (moderate number of iterations)
    # ------------------------------------------------------------------
    for i, cand in enumerate(candidates, 1):
        if time.time() >= deadline:
            break
        iters = 30000
        print(f"[{time.time() - start:6.2f}s] Refining starter {i}/{len(candidates)} "
              f"(n={n0}, iters={iters})")
        _, local_best, local_score = _refine_sequence(
            cand,
            max_iters=iters,
            deadline=deadline,
            step0=0.6,
            eps_start=0.06,
            eps_end=0.001,
            beta=0.9,
            soft_frac=0.55,
        )
        if local_score < best_score:
            best_score = local_score
            best_seq = local_best.copy()
            print(f"   New global best from starter {i}: score={local_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Progressive up‑sampling of the current best shape
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)

    cur_seq = best_seq.copy()
    cur_n = cur_seq.shape[0]
    MAX_LEN = 12000               # generous upper bound still fitting budget

    while True:
        # pick a larger size (≈1.5×) but do not exceed MAX_LEN
        nxt_n = int(cur_n * 1.5)
        if nxt_n > MAX_LEN:
            nxt_n = MAX_LEN
        if nxt_n <= cur_n:
            break

        # ----- upsample -----
        cur_seq = _upscale(cur_seq, nxt_n)

        # ----- allocate iterations (larger n ⇒ fewer iterations) -----
        # keep total computation roughly balanced
        base_iters = 50000
        iters = int(base_iters * n0 / nxt_n) + 20000
        iters = max(iters, 20000)

        print(f"[{time.time() - start:6.2f}s] Upsampling to n={nxt_n}, "
              f"iters={iters}")

        _, local_best, local_score = _refine_sequence(
            cur_seq,
            max_iters=iters,
            deadline=deadline,
            step0=0.55,
            eps_start=0.04,
            eps_end=0.001,
            beta=0.9,
            soft_frac=0.5,
        )

        # keep the best shape observed at this resolution
        if local_score < best_score:
            best_score = local_score
            best_seq = local_best.copy()
            print(f"   Updated global best after up‑sampling: n={nxt_n}, "
                  f"score={local_score:.6f}")

        cur_seq = local_best   # continue optimisation from the refined point
        cur_n = nxt_n

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final intensive refinement on the best length found
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_n = best_seq.shape[0]
        final_iters = 120000
        print(f"[{time.time() - start:6.2f}s] Final refinement on n={final_n}, "
              f"{final_iters} iters")
        _, local_best, local_score = _refine_sequence(
            best_seq,
            max_iters=final_iters,
            deadline=deadline,
            step0=0.45,
            eps_start=0.02,
            eps_end=0.0005,
            beta=0.9,
            soft_frac=0.3,
        )
        if local_score < best_score:
            best_score = local_score
            best_seq = local_best.copy()
            print(f"   Final improvement: score={local_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Clean‑up and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    print(f"[{time.time() - start:6.2f}s] Finished. Best score (2·n·max(conv)) = "
          f"{best_score:.6f}")

    return [float(v) for v in best_seq]
