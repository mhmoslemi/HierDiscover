def solve_lp(f, M):
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    # Build constraints
    constraints = []
    # For each k
    for k in range(2*n - 1):
        # Compute indices i start such that 0 <= i < n and 0 <= k-i < n => i in [max(0, k-n+1), min(k, n-1)]
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # build a linear expression sum_{i=i_min}^{i_max} f[i] * b[k-i]
        # use cp.sum(cp.multiply(f[i_min:i_max+1], b[k-i_max:k-i_min+1][::-1])) maybe easier with loops
        expr = 0
        for i in range(i_min, i_max+1):
            expr += f[i] * b[k - i]
        constraints.append(expr <= M)
    # objective
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=20000) # maybe set options
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None
    return b.value
```

def weighted_gradient(a, weights):
    n = a.shape[0]
    grad = np.zeros_like(a)
    # iterate over k where weights > 0
    nonzero = np.where(weights > 0)[0]
    for k in nonzero:
        w = weights[k]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi+1)
        jdx = k - idx
        grad[idx] += 2.0 * w * a[jdx]
    return grad

"""
Search for a non‑negative step‑function that minimizes

    score = 2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Power‑law sweep** – a quick grid and a fine sweep over
   a_i ∝ (i+offset)⁻ᵖ to obtain a good initial simplex point.

2. **Projected soft‑max gradient descent** – we minimise a smoothed
   version of the infinity‑norm of the self‑convolution.
   The gradient of `soft‑max(conv)` is obtained by a second FFT‑based
   convolution, then a multiplicative (exponentiated) step is taken
   and the iterate is projected back onto the probability simplex.
   Momentum, step‑size decay and a slowly increasing soft‑max temperature
   help to escape plateaus.

3. **Stochastic perturbation** – a short hill‑climbing phase
   adds small Gaussian jitter to the best vector found so far,
   keeping only improvements.

All heavy work is done with NumPy/FFT, no external solvers are required.
The routine respects the supplied time budget (default 950 s) and
returns a list of floats ready for `evaluate_sequence`.
"""

# Helper functions (top‑level, no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Return a non‑negative vector a_i = (i+offset)⁻ᵖ (i=0…n‑1) normalised
    to unit L1‑norm."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:          # defensive fallback
        a[:] = 1.0 / n
    else:
        a /= total
    return a

    """Full self‑convolution x*x using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv

    """Full convolution of two real vectors using a power‑of‑two FFT."""
    n = len(x) + len(y) - 1

    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:n]
    return conv

    """Euclidean projection of v onto the probability simplex {w ≥ 0, Σw = 1}."""

    mval : float
        Current maximum of conv(a,a) (useful for score evaluation).
    """

    conv = conv_fft(a)
    mval = conv.max()
