def solve_lp_max_sum(f, M):
    n = len(f)
    from scipy.optimize import linprog
    # Build A matrix as described
    from scipy.sparse import lil_matrix, csr_matrix
    # Build once per iteration:
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for i in range(n):
        # fill rows i to i+n-1 inclusive (size n)
        A[i:i+n, i] = f[i]
    A = A.tocsr()
    c = -np.ones(n)  # maximize sum(b)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    # solve
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if res.success:
        b = np.maximum(res.x, 0.0)
        return b
    else:
        # fallback: return uniform
        return np.full(n, 1/n)

"""
Hybrid optimizer for the step‑function attack.

The algorithm combines three ideas that are not used together in the
reference implementation:

1.  A fast projected gradient descent on the probability simplex that
    uses a soft‑max smoothing of the max‑convolution and optionally
    enforces symmetry after each projection (the optimum is known to be
    near‑symmetric).

2.  An alternating linear‑programming refinement (the classic “LP‑step”
    from the paper description).  For a fixed shape f we maximise the
    ℓ₁‑norm of a vector g under the linear constraints ‖f∗g‖∞≤M.
    After normalising g we take a small convex combination
    f←(1‑t)f+t g.  This step flattens the auto‑convolution and
    systematically reduces the score.

3.  Progressive up‑sampling: the best shape for a moderate length is
    interpolated to a larger size and refined again.  Larger lengths
    give a finer discretisation of the (continuous) optimal kernel and
    usually improve the bound.

All work is performed on the simplex (∑aᵢ=1, aᵢ≥0); the evaluation
function is scale‑invariant, so the returned vector can be used directly.
The routine respects the 1000 s time limit, prints intermediate scores
and finally returns the best sequence found.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al.)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – full auto‑convolution)
def _auto_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max smoothed max‑convolution
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Returns ∇  Σ_k w_k (a∗a)_k   where w_k ∝ exp((conv_k - max)/τ).
    """
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp((conv - max_val) / tau)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w.fill(1.0 / w.size)

    # gradient = 2 * (w ∗ a_rev) restricted to the n centre entries
    a_rev = a[::-1]
    fft_len = 1 << ((w.size + n - 2).bit_length())
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len)[: w.size + n - 1]
    g = 2.0 * conv2[n - 1 : n - 1 + n]
    return g


# ----------------------------------------------------------------------
# Projected Adam (simple version) with optional symmetry enforcement
def _adam_step(
    a: np.ndarray,
    g: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    it: int,
    step0: float,
    beta1: float,
    beta2: float,
    eps: float,
    symmetric: bool,
) -> np.ndarray:
    """One Adam update followed by projection onto the simplex."""
    # bias‑corrected moments
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)

    m_hat = m / (1.0 - beta1 ** it)
    v_hat = v / (1.0 - beta2 ** it)

    step = step0 / math.sqrt(it)
    a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

    # optional symmetry enforcement (averaging with reverse)
    if symmetric:
        a_new = (a_new + a_new[::-1]) * 0.5

    a_proj = _project_simplex(a_new)
    return a_proj, m, v


# ----------------------------------------------------------------------
# Refine a fixed‑length sequence with Adam + soft‑max.
def _refine_sequence(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.6,
    eps_start: float = 0.05,
    eps_end: float = 0.001,
    beta1: float = 0.9,
    beta2: float = 0.999,
    symmetric: bool = True,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (final_a, best_a, best_score) after at most max_iters updates.
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _auto_convolution(a).max()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution & score
        conv = _auto_convolution(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule (linear taper)
        tau = max(eps_start + (eps_end - eps_start) * (it / max_iters), 1e-14) * conv.max()

        # gradient
        g = _softmax_gradient(a, conv, tau)

        # Adam‑style update + projection (+ optional symmetry)
        a, m, v = _adam_step(
            a,
            g,
            m,
            v,
            it,
            step0,
            beta1,
            beta2,
            eps=1e-12,
            symmetric=symmetric,
        )
    return a, best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (requires SciPy)
def _lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   max Σ b_j   s.t.   b ≥ 0  and  (f∗b)_k ≤ M  (for all k).
    Returns the optimal b (may be all zeros if infeasible).
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import lil_matrix, csr_matrix
    except Exception as exc:
        raise RuntimeError("SciPy required for the LP step") from exc

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the (rows × n) constraint matrix:
    # each column i contributes f[i] on rows i … i+n‑1
    A = lil_matrix((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]          # contiguous block
    A = csr_matrix(A)

    c = -np.ones(n)                     # maximise sum → minimise -sum
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method="highs", options={"presolve": True})
    if not res.success:
        # fallback – uniform distribution (cannot be better than the current)
        return np.full(n, 1.0 / n)
    b = np.maximum(res.x, 0.0)
    return b


# ----------------------------------------------------------------------
# Alternating LP + convex combination update
def _alternating_lp_update(
    a: np.ndarray,
    t: float,
    max_outer: int,
    deadline: float,
    symmetric: bool = True,
) -> np.ndarray:
    """
    Perform a few outer iterations of the LP‑step described in the
    statement (maximise ℓ₁‑norm under the current sup‑convolution bound).
    """
    cur = a.copy()
    n = cur.shape[0]

    for outer in range(1, max_outer + 1):
        if time.time() >= deadline:
            break

        conv = _auto_convolution(cur)
        M = conv.max()
        try:
            b = _lp_max_sum(cur, M)
        except Exception:
            # if LP unavailable, abort the alternating phase
            break

        if b.sum() == 0.0:
            break
        g = b / b.sum()                     # normalise to the simplex

        # convex combination
        cur = (1.0 - t) * cur + t * g

        # keep all coordinates strictly positive (needed for the next LP)
        cur = np.maximum(cur, 1e-12)
        cur = cur / cur.sum()

        # optional symmetry enforcement after the LP step
        if symmetric:
            cur = (cur + cur[::-1]) * 0.5

        # a short inner refinement to exploit the new shape
        cur, _, _ = _refine_sequence(
            cur,
            max_iters=4000,
            deadline=deadline,
            step0=0.5,
            eps_start=0.04,
            eps_end=0.001,
            symmetric=symmetric,
        )
    return cur
