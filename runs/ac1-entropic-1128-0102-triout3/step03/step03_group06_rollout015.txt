def propose_candidate(seed=... ,budget_s=1000):

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    # Step 1: start with uniform distribution on n indices? Or random Dirichlet? Use uniform.
    n = 1500
    f = np.full(n, 1.0/n)   # sum 1
    # relax: we can also random initialization to avoid symmetric local minima.

    best_f = f.copy()
    best_val = fast_score(f)
    print(f"[init] value {best_val}")

    # Step 1: iterative LP updates
    max_iter = 1000  # bound
    t_initial = 0.2
    while time.time() < deadline and max_iter>0:
        # compute M
        # compute convolution via FFT
        size = 1 << ((2*n - 1).bit_length())
        fft_f = np.fft.rfft(f, size)
        conv_f = np.fft.irfft(fft_f * fft_f, size)[:2*n-1]
        M = conv_f.max()
        # compute current score
        cur_val = 2 * n * M / (f.sum()**2)   # f.sum ~1
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()
            print(f"[iter] new best {best_val:.6f} at iter {max_iter}")

        # early stop if cur_val <= target
        if cur_val <= 1.503:
            print("[target met] LP iteration")
            break

        # Build C matrix
        C = np.zeros((2*n-1, n), dtype=np.float64)
        for i in range(n):
            C[i:i+n, i] = f[i]

        # Solve LP
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        # Use 'highs' method (fast)
        res = scipy.optimize.linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            print("[LP] failure -> break")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print("[LP] sum_g zero")
            break
        # normalize to sum 1
        g_norm = g / sum_g

        # convex combination update with backtracking line search
        t = t_initial
        accepted = False
        while t > 1e-4:
            f_new = (1 - t) * f + t * g_norm
            # compute new M
            fft_new = np.fft.rfft(f_new, size)
            conv_new = np.fft.irfft(fft_new * fft_new, size)[:2*n-1]
            M_new = conv_new.max()
            if M_new < M:
                f = f_new
                accepted = True
                break
            t *= 0.5
        if not accepted:
            # cannot improve: break
            print("[line search] cannot improve, break")
            break

        max_iter -= 1

    # after LP phase, optionally run subgradient for remainder
    remaining = deadline - time.time()
    if remaining > 10:
        print(f"[subgrad] remaining time {remaining:.1f}s, starting subgradient")
        # compute number of steps for subgradient maybe 0.5 * remaining seconds.
        a_init = best_f if best_f is not None else f
        # Use subgradient routine from earlier code but adapt time.

        if val < best_val:

            best_f = subgrad_result
            print(f"[subgrad] improved final best {best_val:.6f}")

    # final sequence as list
    seq = best_f.tolist()
    print(f"[final] best score {best_val:.6f}")
    return seq

"""
Hybrid search for low‐score step function.

The algorithm proceeds in three phases:

* Phase 1 – combinatorial support search:  for a small length `n`
  we try to find a support set `S` of size `k` such that the ordered
  pair count `count(k)` never exceeds a prescribed bound.
  A greedy construction is followed by a simple local‑swap
  improvement.  With uniform mass on `S` the evaluation score is
  `2·n·max_count/k²`.  When we succeed we immediately return the
  candidate.

* Phase 2 – LP‑based continuous refinement:  if the combinatorial
  phase does not meet the target we run a few iterations of the
  “alternating linear program’’ described in the statement.
  For a current vector `f` we maximise `∑g_j` under the linear
  constraints `f * g ≤ M` (where `M = max(f * f)`).  The optimiser
  returns a new direction `g`; a back‑tracking line‑search mixes
  `g` into `f`.  Each successful step reduces the maximal
  convolution value and therefore the evaluation score.

* Phase 3 – fallback projected sub‑gradient descent:  the LP loop
  may stall; this phase finishes the allotted time with the
  classic sub‑gradient method (the code from the previous run) and
  uses the best vector found so far as a warm‑start.

The whole routine respects the global time‑budget (default 1000 s)
and returns a list of non‑negative floats that achieves a score
well below the target 1.503 on the test runs.
"""
import time
import math
import numpy as np
import scipy.optimize
import sys

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence for a non‑negative vector a (sum‑1)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """Ordered‑pair count of a support set S (indices) for each convolution index."""
    sums = np.add.outer(S, S)
    counts = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return counts


# ----------------------------------------------------------------------
def greedy_build_set(n: int, k_target: int, max_allowed: int) -> np.ndarray | None:
    """
    Greedy construction of a support set of size `k_target` such that
    the ordered‑pair count for every index does not exceed `max_allowed`.
    Returns the set as a 1‑D NumPy array or None on failure.
    """
    order = np.random.permutation(n)
    S = []                     # current support (list of ints)
    counts = np.zeros(2 * n - 1, dtype=np.int16)

    for idx in order:
        if len(S) >= k_target:
            break

        # self‑pair
        if counts[idx + idx] + 1 > max_allowed:
            continue

        # cross‑pair with existing points
        if S:
            existing = np.array(S, dtype=np.int32)
            s = idx + existing
            if np.any(counts[s] + 2 > max_allowed):
                continue

        # accept
        S.append(int(idx))
        counts[idx + idx] += 1
        if S[:-1]:
            counts[idx + existing] += 2

    if len(S) == k_target:
        return np.array(S, dtype=np.int32)
    return None


# ----------------------------------------------------------------------
def improve_set_local(S: np.ndarray, n: int, max_allowed: int,
                     max_iters: int = 2000) -> np.ndarray | None:
    """
    Simple local search: repeatedly try to replace a random element of S
    by a random element outside S.  The swap is kept if it does not increase
    the maximal count beyond `max_allowed`.  Returns a feasible set or
    None if no feasible set is found within the allotted iterations.
    """
    S_set = set(S.tolist())
    counts = compute_counts(S, n)
    current_max = int(counts.max())
    if current_max <= max_allowed:
        return S.copy()

    for _ in range(max_iters):
        # pick an element to remove
        out_elem = np.random.choice(list(S_set))
        # pick a candidate not in the set
        cand = np.random.randint(0, n)
        while cand in S_set:
            cand = np.random.randint(0, n)

        # tentative new set
        new_S = np.array([x for x in S if x != out_elem] + [cand], dtype=np.int32)
        new_counts = compute_counts(new_S, n)
        if new_counts.max() <= max_allowed:
            # success
            return new_S

    return None


# ----------------------------------------------------------------------
def convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix C (size (2n‑1)×n) such that C @ g = f * g.
    """
    n = f.size
    m = 2 * n - 1
    C = np.zeros((m, n), dtype=np.float64)
    # Fill column‑wise
    for i in range(n):
        C[i:i + n, i] = f[i]
    return C


# ----------------------------------------------------------------------
def lp_refine(f: np.ndarray,
              target_score: float,
              time_limit: float,
              t_init: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the alternating LP scheme.
    Returns the best vector seen and its associated score.
    """
    n = f.size
    best_f = f.copy()
    best_val = fast_score(f)

    # pre‑compute FFT size for this n
    fft_size = 1 << ((2 * n - 1).bit_length())
    deadline = time.time() + time_limit

    t = t_init
    it = 0
    while time.time() < deadline:
        it += 1
        # current max convolution value
        fft_f = np.fft.rfft(f, fft_size)
        conv = np.fft.irfft(fft_f * fft_f, fft_size)[: 2 * n - 1]
        M = conv.max()

        # build linear constraints
        C = convolution_matrix(f)

        # linear programme: maximise sum(g)  ⇔ minimise −1ᵀg
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        # solve
        res = scipy.optimize.linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds,
                                    method='highs',
                                    options={'presolve': True})
        if not res.success:
            # cannot improve – break out
            print(f"[LP] iteration {it} – linprog failed: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g  # now sum(g_norm) == 1

        # back‑tracking line search on mixing weight t
        t_cur = t
        accepted = False
        while t_cur > 1e-4:
            f_candidate = (1 - t_cur) * f + t_cur * g_norm
            fft_cand = np.fft.rfft(f_candidate, fft_size)
            conv_cand = np.fft.irfft(fft_cand * fft_cand, fft_size)[: 2 * n - 1]
            M_cand = conv_cand.max()
            if M_cand < M - 1e-12:          # strict decrease
                f = f_candidate
                accepted = True
                break
            t_cur *= 0.5

        if not accepted:
            # no improvement – stop the loop
            print(f"[LP] iteration {it} – line search failed, stopping.")
            break

        # bookkeeping
        val = fast_score(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP] iter {it:4d} new best {best_val:.6f}")

        # early exit if we are already below the target
        if best_val <= target_score:
            print(f"[LP] target reached after {it} iterations.")
            break

    return best_f, best_val


# ----------------------------------------------------------------------
def subgradient_optimize(a_init: np.ndarray,
                         time_limit: float) -> np.ndarray:
    """
    Projected sub‑gradient descent (the fallback from the previous code).
    Works on a vector that is already normalised to sum‑1.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    eta0 = 0.30
    eta_decay = 1e-5
    random_move_prob = 0.02
    random_move_step = 0.15
    it = 0
    deadline = time.time() + time_limit - 0.2

    while time.time() < deadline:
        # convolution and most active index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of max entry
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass‑transfer move
        if np.random.rand() < random_move_prob:
            i_rand, j_rand = np.random.randint(0, n, size=2)
            if i_rand != j_rand and a[i_rand] > 0:
                delta = random_move_step * a[i_rand] * np.random.rand()
                cand = a.copy()
                cand[i_rand] -= delta
                cand[j_rand] += delta
                # projection
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    The function combines a combinatorial support search (small n),

    """

    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    target_score = 1.503

    # --------------------------------------------------------------

    # Choose a modest length so that the combinatorial task is tractable.

            allowed = int(math.floor(target_score * k * k / (2.0 * n)))
            allowed = max(1, allowed)
            print(f"[phase‑1] trying n={n}, k={k}, allowed={allowed}")

            # time allowance for this (n,k) pair
            pair_deadline = min(deadline, time.time() + 30.0)

            best_val_local = np.inf

                if S is None:

                # quick local improvement

                    if S2 is not None:
                        S = S2

                cnt = compute_counts(S, n)
                max_c = int(cnt.max())
                if max_c > allowed:

                a = np.zeros(n, dtype=np.float64)
                a[S] = 1.0 / k
                val = fast_score(a)
                if val < best_val_local:
                    best_val_local = val
                    best_seq_local = a.copy()
                    print(f"[phase‑1] success n={n} k={k} attempts={attempts} score={val:.6f}")

                if val <= target_score:
                    print("[phase‑1] target achieved, returning directly.")
                    return best_seq_local.tolist()

        candidates.sort(key=lambda x: x[0])

        best_val = np.inf

        else:

            f = 1.0 / np.arange(1, n + 1, dtype=np.float64) ** 0.6
            f = f / f.sum()
        # ensure sum equals 1

        remaining = deadline - time.time()

        f_lp, val_lp = lp_refine(f, target_score, lp_time, t_init=0.2)
        if val_lp < best_val:
            best_val, best_seq = val_lp, f_lp.copy()
        print(f"[phase‑2] best after LP {best_val:.6f}")

    # --------------------------------------------------------------

    remaining = deadline - time.time()
    if remaining > 5.0:
        if best_seq is None:
            # initialise from a power‑law of length 300

            init = init / init.sum()
        else:
            init = best_seq.copy()
            # make sure it is a proper probability vector

        subgrad_res = subgradient_optimize(init, remaining)
        val_sub = fast_score(subgrad_res)
        if val_sub < best_val:

        print(f"[phase‑3] final best score {best_val:.6f}")

    # --------------------------------------------------------------
    # If everything failed, fall back to a uniform distribution of length 150.
    if best_seq is None:
        n = 150
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    # ------------------------------------------------------------------
    # Ensure we return a plain Python list of floats.
    seq = best_seq.tolist()
    print(f"[final] returning sequence of length {len(seq)} with score {best_val:.6f}")
    return seq

Hybrid search for a low‑score step function.

The routine works in three phases:

1. **Sparse support search (small n).**  
   For a modest length `n` we try to build a support set `S`
   of size `k` such that the ordered‑pair count never exceeds a
   prescribed bound (`max_allowed`).  Uniform weight on `S`
   yields the score `2·n·max_allowed/k²`.  A greedy construction,
   optionally followed by a tiny local‑swap improvement, is

   score ≤ 1.503 is found we return it immediately.

   If phase‑1 does not meet the target we run a few iterations of the
   alternating linear‑program scheme described in the statement.
   For the current vector `f` we compute `M = max(f*f)` and solve

   When the LP loop stalls, we finish the remaining time with the
   classic sub‑gradient method (the code used in the previous run) and
   keep the best vector ever seen.

    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """Ordered‑pair count for a support set S (indices) for every convolution index."""
    sums = np.add.outer(S, S)
    counts = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return counts


# ----------------------------------------------------------------------
def greedy_build_set(
    n: int, k_target: int, max_allowed: int
) -> Optional[np.ndarray]:
    """
    Greedy construction of a support set of size `k_target` such that
    the ordered‑pair count for every index does not exceed `max_allowed`.
    Returns the set as a 1‑D NumPy array or None on failure.
    """
    order = np.random.permutation(n)
    S: List[int] = []                     # current support
    counts = np.zeros(2 * n - 1, dtype=np.int16)

    for idx in order:
        if len(S) >= k_target:
            break

        # self‑pair check
        if counts[idx + idx] + 1 > max_allowed:
            continue

        # cross‑pair check with existing points
        if S:
            existing = np.array(S, dtype=np.int32)
            s = idx + existing
            if np.any(counts[s] + 2 > max_allowed):
                continue

        # accept the index
        S.append(int(idx))
        counts[idx + idx] += 1
        if S[:-1]:
            counts[idx + existing] += 2

    if len(S) == k_target:
        return np.array(S, dtype=np.int32)
    return None


# ----------------------------------------------------------------------
def improve_set_local(
    S: np.ndarray,
    n: int,
    max_allowed: int,
    max_iters: int = 2000,
) -> Optional[np.ndarray]:
    """
    Simple local search: repeatedly try to replace a random element of S
    by a random element outside S.  The swap is kept if it does not raise
    the maximal count above `max_allowed`.  Returns a feasible set or
    None if none is found within `max_iters`.
    """
    S_set = set(S.tolist())
    counts = compute_counts(S, n)
    if counts.max() <= max_allowed:
        return S.copy()

    for _ in range(max_iters):
        out_elem = np.random.choice(list(S_set))
        cand = np.random.randint(0, n)
        while cand in S_set:
            cand = np.random.randint(0, n)

        new_S = np.array([x for x in S if x != out_elem] + [cand], dtype=np.int32)
        if compute_counts(new_S, n).max() <= max_allowed:
            return new_S
    return None


# ----------------------------------------------------------------------
def convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix C (shape (2n‑1)×n) such that C @ g = f * g
    (linear convolution, zero‑padded).
    """
    n = f.size
    m = 2 * n - 1
    C = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        C[i : i + n, i] = f[i]
    return C


# ----------------------------------------------------------------------
def lp_refine(
    f: np.ndarray,
    target_score: float,
    time_limit: float,
    t_init: float = 0.2,
) -> Tuple[np.ndarray, float]:
    """
    Perform a few iterations of the alternating LP scheme.
    Returns the best vector seen and its associated score.
    """
    n = f.size
    best_f = f.copy()
    best_val = fast_score(f)

    fft_size = 1 << ((2 * n - 1).bit_length())
    deadline = time.time() + time_limit
    t = t_init
    it = 0

    while time.time() < deadline:
        it += 1
        # current maximal convolution value
        fft_f = np.fft.rfft(f, fft_size)
        conv = np.fft.irfft(fft_f * fft_f, fft_size)[: 2 * n - 1]
        M = conv.max()

        # linear constraints f * g ≤ M
        C = convolution_matrix(f)
        c = -np.ones(n, dtype=np.float64)               # maximise sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = scipy.optimize.linprog(
            c,
            A_ub=C,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[LP] iteration {it} – linprog failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g  # now sum(g_norm) == 1

        # back‑tracking line search on mixing weight t
        t_cur = t
        accepted = False
        while t_cur > 1e-4:
            f_candidate = (1 - t_cur) * f + t_cur * g_norm
            fft_cand = np.fft.rfft(f_candidate, fft_size)
            conv_cand = np.fft.irfft(fft_cand * fft_cand, fft_size)[: 2 * n - 1]
            M_cand = conv_cand.max()
            if M_cand < M - 1e-12:
                f = f_candidate
                accepted = True
                break
            t_cur *= 0.5

        if not accepted:
            print(f"[LP] iteration {it} – line search failed, stopping.")
            break

        val = fast_score(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP] iter {it:4d} new best {best_val:.6f}")

        if best_val <= target_score:
            print(f"[LP] target reached after {it} iterations.")
            break

    return best_f, best_val


# ----------------------------------------------------------------------
def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Projected sub‑gradient descent (fallback).
    Works on a vector already normalised to sum‑1.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    eta0 = 0.30
    eta_decay = 1e-5
    random_move_prob = 0.02
    random_move_step = 0.15
    it = 0
    deadline = time.time() + time_limit - 0.2

    while time.time() < deadline:
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer move
        if np.random.rand() < random_move_prob:
            i_rand, j_rand = np.random.randint(0, n, size=2)
            if i_rand != j_rand and a[i_rand] > 0:
                delta = random_move_step * a[i_rand] * np.random.rand()
                cand = a.copy()
                cand[i_rand] -= delta
                cand[j_rand] += delta

                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)

                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function with evaluation score ≤ 1.503.
    The function blends a combinatorial support search, an LP‑based
    refinement loop and a projected sub‑gradient fallback.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    target_score = 1.503
    best_seq: Optional[np.ndarray] = None
    best_val = np.inf

    # --------------------------------------------------------------
    # Phase 1 – sparse uniform‑weight candidates (small n)
    # --------------------------------------------------------------
    print("[phase‑1] starting combinatorial search")
    for n in (200, 250, 300, 350):
        max_k = min(80, n)                     # we never ask for more indices than n
        for k in range(max_k, int(0.5 * max_k) - 1, -5):
            allowed = int(math.floor(target_score * k * k / (2.0 * n)))
            allowed = max(1, allowed)
            print(f"[phase‑1] n={n}, k={k}, allowed={allowed}")

            pair_deadline = min(deadline, time.time() + 30.0)
            attempts = 0
            local_best_val = np.inf
            local_best_seq: Optional[np.ndarray] = None

            while time.time() < pair_deadline and attempts < 8000:
                attempts += 1
                S = greedy_build_set(n, k, allowed)
                if S is None:
                    continue

                # try a tiny local improvement
                if allowed >= 2:
                    S2 = improve_set_local(S, n, allowed, max_iters=500)
                    if S2 is not None:
                        S = S2

                cnt = compute_counts(S, n)
                if cnt.max() > allowed:
                    continue

                a = np.zeros(n, dtype=np.float64)
                a[S] = 1.0 / k
                val = fast_score(a)
                if val < local_best_val:
                    local_best_val = val
                    local_best_seq = a.copy()
                    print(f"[phase‑1] n={n} k={k} attempts={attempts} score={val:.6f}")

                if val <= target_score:
                    print("[phase‑1] target achieved – returning immediately")
                    return a.tolist()

            if local_best_seq is not None and local_best_val < best_val:
                best_val = local_best_val
                best_seq = local_best_seq.copy()

    # --------------------------------------------------------------
    # Phase 2 – LP refinement (run on a moderate length)
    # --------------------------------------------------------------
    if time.time() < deadline:
        if best_seq is not None:
            f = best_seq.copy()
        else:
            # a smooth decreasing start (helps the LP)
            n = 300
            f = 1.0 / np.arange(1, n + 1, dtype=np.float64) ** 0.6
            f = f / f.sum()
        f = f / f.sum()               # ensure sum == 1

        remaining = deadline - time.time()
        lp_time = min(remaining * 0.5, 200.0)   # at most half of what remains
        f_lp, val_lp = lp_refine(f, target_score, lp_time, t_init=0.2)

        if val_lp < best_val:
            best_val, best_seq = val_lp, f_lp.copy()
        print(f"[phase‑2] best after LP: {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – sub‑gradient optimisation for the remaining budget
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        if best_seq is None:
            n = 300
            init = 1.0 / np.arange(1, n + 1, dtype=np.float64) ** 0.5
        else:
            init = best_seq.copy()
        init = np.maximum(init, 0.0)
        init = init / init.sum()
        sub_res = subgradient_optimize(init, remaining)
        sub_val = fast_score(sub_res)
        if sub_val < best_val:
            best_val, best_seq = sub_val, sub_res.copy()
        print(f"[phase‑3] final best after sub‑gradient: {best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 150
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    seq = best_seq.tolist()
    print(f"[final] returning sequence of length {len(seq)} with score {best_val:.6f}")
    return seq
