def _conv_and_max(seq: np.ndarray):

    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(seq, n=fft_len)

def _active_indices(conv, max_val, rel_eps=1e-3, max_add=12):
    # same logic: indices where conv >= max_val - rel_eps * max_val
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        # Add top max_add indices
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K

def _gradient_active(a, K):
    n = a.shape[0]
    # allocate gradient zeros
    g = np.zeros_like(a)
    for k in K:
        # contributions: for i in [max(0, k-(n-1)), min(n-1, k)]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # i indices
        i_idx = np.arange(lo, hi + 1)
        j_idx = k - i_idx
        # accumulate 2*a[j_idx] into g[i_idx]
        g[i_idx] += 2.0 * a[j_idx]
    return g

"""
Propose a candidate step‑function using a long mirror‑descent (exponentiated
gradient) run on the simplex combined with occasional random perturbations
and a final stochastic hill‑climbing phase.  The method starts from the
previously discovered construction (global variable ``height_sequence_1`` if
present), normalises it to a probability vector, then runs many
iterations of a sub‑gradient descent on the objective
    2·n·max(conv(a,a))
with the vector kept on the simplex via multiplicative updates.  After the
gradient phase a cheap random mass‑transfer local search is performed until
the time budget is exhausted.  The best sequence encountered is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Return the full auto‑convolution of ``a`` and its maximum entry."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 12):
    """
    Return a small set of indices that are “active” for the sub‑gradient.
    All indices within ``rel_eps`` relative distance of the maximum are kept.
    If fewer than ``max_add`` indices are found, the globally largest ones are
    added so that the set always contains at most ``max_add`` entries.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.shape[0])
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Compute a sub‑gradient of ``max(conv(a,a))`` using the active set ``K``.
    For each active convolution index k,
        (∂/∂a_i) conv_k = a_{k-i} + a_{i-k}
    but because the convolution is symmetric the contribution simplifies to
    ``2·a_{k-i}`` for every i that participates in index k.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g
