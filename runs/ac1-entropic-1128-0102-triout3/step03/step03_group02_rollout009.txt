```
def _sgd_optimize(a0, max_iters, deadline, step0=0.5, beta=0.9, eps_start=0.1, eps_end=0.001):
    a = a0.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_of_seq(a)
    n = len(a)
    fft_len_conv = 1 << ((2*n - 1).bit_length())
    fft_len_grad = 1 << ((3*n - 2).bit_length())
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[:2*n-1]
        max_val = conv.max()
        score = 2 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
        eps = eps_start * ((eps_end/eps_start) ** (it / max_iters))   # geometric
        tau = max(eps * max_val, 1e-14)
        w = np.exp((conv - max_val)/tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)
        # gradient
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a[::-1], n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        g = 2 * conv2[n-1:n-1+n]
        # Nesterov momentum
        step = step0 / np.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)

    return best_a, best_score
```

```
def _adam_optimize(a0, max_iters, deadline, lr=0.05, beta1=0.9, beta2=0.999, eps_start=0.1, eps_end=0.001):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_of_seq(a)
    n = len(a)
    fft_len_conv = 1 << ((2*n - 1).bit_length())
    fft_len_grad = 1 << ((3*n - 2).bit_length())

    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute conv
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af*Af, n=fft_len_conv)[:2*n-1]
        max_val = conv.max()
        score = 2 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # softmax schedule
        eps = eps_start * ((eps_end/eps_start) ** (it / max_iters))
        tau = max(eps * max_val, 1e-14)
        w = np.exp((conv - max_val)/tau)
        w_sum = w.sum()
        if w_sum>0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)
        # gradient
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a[::-1], n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n -1]
        g = 2 * conv2[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step
        # projection onto simplex
        a = _project_simplex(a)
    return best_a, best_score
```

```
def _eg_optimize(a0, max_iters, deadline, eta0=0.5, eps_start=0.1, eps_end=0.001):

    best_a = a.copy()

    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[:2*n-1]
        max_val = conv.max()
        cur_score = 2 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        eps = eps_start * ((eps_end/eps_start) ** (it / max_iters))
        tau = max(eps * max_val, 1e-14)
        w = np.exp((conv - max_val)/tau)
        w_sum = w.sum()
        if w_sum>0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0/w.size)
        # gradient
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a[::-1], n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[:len(w) + n -1]
        g = 2 * conv2[n-1:n-1 + n]   # gradient of softmax approximate
        # EG update: a = a * exp(-eta * g); eta schedule decaying with sqrt(it)
        eta = eta0 / np.sqrt(it)
        # shift g to keep exponents stable
        shift = np.max(eta * g)  # maybe large positive
        mul = np.exp(-eta * g - shift)   # subtract shift
        a = a * mul
        # renormalize
        sum_a = a.sum()
        if sum_a > 0:
            a = a / sum_a
        else:
            a = np.full_like(a, 1.0/n)
    return best_a, best_score
```

"""
Propose a candidate step‑function (non‑negative heights) with a low
evaluate_sequence score.

The algorithm works in three stages

1. **Initial pool** – a few diverse starting vectors (uniform, triangular,
   Hann window, the previous best if available and a random Dirichlet) are
   normalised onto the probability simplex (sum = 1).  Each of them is
   refined for a modest number of Adam steps.

2. **Upsampling** – the current best shape is linearly interpolated to a
   larger size (≈1.5×) and refined again.  This yields a richer family of
   functions while preserving the good properties already discovered.

3. **Long final refinement** – the best vector found so far is optimised for
   the remaining time with a large number of Adam‑steps on the simplex.
   The objective that Adam minimises is a smooth approximation of
   ``max(conv(a,a))`` obtained via a soft‑max (temperature τ) over the
   convolution entries.  Gradients are computed by FFT‑based convolutions,
   and after each Adam update the vector is projected back onto the simplex
   (Duchi et al., “Efficient Projections onto the l1‑Ball for Learning in
   High Dimensions”, ICML 2008).  Because the evaluate_sequence function is
   scale‑invariant we keep the sum fixed at 1, so the score equals
   ``2·n·max(conv(a,a))``.

building the pool, then spends the majority of the time on the final

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Score for a sequence already normalised to sum = 1:
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Soft‑max (with temperature τ = eps·max(conv)) over the convolution."""
    max_val = conv.max()
    tau = max(eps * max_val, 1e-14)
    w = np.exp((conv - max_val) / tau)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(w, 1.0 / len(w))
    return w


def _gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]

    # ----- convolution a*a -----
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len_conv)
    conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]

    # ----- soft‑max weights -----
    w = _softmax_weights(conv, eps)

    # ----- gradient = 2 * (w * a_rev) convolution -----
    fft_len_grad = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a[::-1], n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
    g = 2.0 * conv2[n - 1 : n - 1 + n]   # length‑n gradient
    return g, conv.max()


def _adam_optimize(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    *,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_start: float = 0.12,
    eps_end: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the probability simplex.

    Returns the best vector seen (not the final iterate) and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_of_seq(a)

    # pre‑compute FFT lengths (they do not change during optimisation)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current score (for bookkeeping) -----
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- temperature schedule (geometric decay) -----
        eps = eps_start * ((eps_end / eps_start) ** (it / max_iters))

        # ----- gradient of the smooth surrogate -----
        g, _ = _gradient(a, eps)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step

        # ----- projection onto the simplex -----
        a = _project_simplex(a)

    # final check (maybe the loop exited before reaching max_iters)
    final_score = _score_of_seq(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def _interpolate(old: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from the old grid to a new grid of size ``new_len``."""
    old_len = old.shape[0]
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new, 0.0)
    if new.sum() == 0.0:
        new = np.full(new_len, 1.0 / new_len)
    else:
        new = new / new.sum()
    return new

    tri = np.minimum(idx + 1, base_n - idx)                               # triangular
    init_vectors.append(tri / tri.sum())
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (base_n - 1)))             # Hann
    init_vectors.append(hann / hann.sum())
    # a random Dirichlet draw
    init_vectors.append(rng.dirichlet(np.ones(base_n)))

    # ------------------------------------------------------------------
    # 2. Quick refinement of each starter (short Adam runs)
    # ------------------------------------------------------------------
    short_iters = 30000                     # enough to get a decent start
    for i, vec in enumerate(init_vectors, 1):
        if time.time() >= deadline:
            break
        n = vec.shape[0]
        print(f"[{time.time() - start_time:6.2f}s] Short Adam start {i}/{len(init_vectors)} (n={n})")
        best_local, score_local = _adam_optimize(
            vec,
            max_iters=short_iters,
            deadline=deadline,
            lr=0.06,
            eps_start=0.12,
            eps_end=0.004,
        )
        if score_local < best_score:
            best_score = score_local
            best_seq = best_local.copy()
            print(f"   >>> New global best after short run: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Progressive up‑sampling of the current best shape
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen, but keep a safe fallback
        best_seq = np.full(base_n, 1.0 / base_n)

    cur_seq = best_seq.copy()
    cur_n = cur_seq.shape[0]
    MAX_LEN = 8000

    while True:
        # decide next size (≈1.5× larger, cap at MAX_LEN)
        next_n = int(cur_n * 1.5)
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n <= cur_n:
            break

        # ----- up‑sample -----
        up_seq = _interpolate(cur_seq, next_n)

        # allocate a budget that roughly scales with n (larger n → fewer iters)
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # rough estimate: 1 s ≈ 3 000 iterations for n≈2000, scale inversely with n
        iters = int(min(80000, max(20000, (remaining * 3000 * base_n) / next_n)))
        iters = max(iters, 15000)

        print(f"[{time.time() - start_time:6.2f}s] Upsampling to n={next_n}, "
              f"allocating {iters} Adam‑iterations")
        local_best, local_score = _adam_optimize(
            up_seq,
            max_iters=iters,
            deadline=deadline,
            lr=0.05,
            eps_start=0.10,
            eps_end=0.002,
        )
        if local_score < best_score:
            best_score = local_score
            best_seq = local_best.copy()
            print(f"   >>> Updated global best (n={next_n}): score={best_score:.6f}")

        # Prepare for possible further up‑sampling
        cur_seq = up_seq
        cur_n = next_n
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Long final Adam‑refinement on the best length found
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        # Use the remaining time to perform a very long run
        remaining = deadline - time.time()
        # one iteration costs roughly O(n log n) ≈ a few ms; be generous
        max_iters = int(remaining * 1500)          # about 1500 iters per second
        max_iters = max(max_iters, 50000)          # never less than 50 k
        print(f"[{time.time() - start_time:6.2f}s] Final long Adam run (n={n}), "
              f"max_iters={max_iters}")
        final_best, final_score = _adam_optimize(
            best_seq,
            max_iters=max_iters,
            deadline=deadline,
            lr=0.04,
            eps_start=0.08,
            eps_end=0.0005,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_best.copy()
            print(f"   >>> Final improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence (plain Python floats)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s    # normalise again for safety

    print(f"[{time.time() - start_time:6.2f}s] Finished. Best score (2·n·max(conv)) = {best_score:.6f}")
    return [float(x) for x in best_seq]
