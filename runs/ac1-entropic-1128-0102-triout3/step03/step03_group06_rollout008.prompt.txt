def fast_score(a: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative vector a (sum‑1)."""
    n = a.size
    s = a.sum()

    size = 1 << ((2 * n - 1).bit_length())

    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]

def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """
    For a support set S (array of indices) compute the ordered‑pair count
    for each possible convolution index (0 … 2·n‑2).
    The result is a length‑(2·n‑1) integer array.

    # outer sum produces a (k,k) matrix whose entries are i+j.
    sums = np.add.outer(S, S)               # includes both (i,j) and (j,i)
    counts = np.bincount(sums.ravel(), minlength=2 * n - 1)

def greedy_build_set(n: int, k_target: int, max_allowed: int) -> np.ndarray | None:
    """

    convolution count for every index does not exceed `max_allowed`.
    Returns the set as a 1‑D numpy array of integers, or None on failure.
    """

    counts = np.zeros(2 * n - 1, dtype=np.int16)

            break

        # 1) self‑pair check
        if counts[idx + idx] + 1 > max_allowed:
            continue

            sums = idx + existing                     # vector of new sums
            # each new unordered pair adds 2 to the count
