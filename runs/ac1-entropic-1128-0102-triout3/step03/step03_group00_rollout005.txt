"""
Improved multi‑scale optimisation for the step‑function problem.

The algorithm proceeds in three phases:

1.  **Coarse optimisation** – start with a modest size (n≈800) and run many
    projected‑gradient iterations (Nesterov‑style momentum, step ∝1/√t)
    on the probability simplex.

2.  **Progressive up‑sampling** – repeatedly linearly up‑sample the current
    best sequence to a larger size (≈2400 → 4800 → optionally 9600) and
    continue the same optimiser.  Larger n gives the same normalisation
    (∑a=1) but usually a smaller peak of the self‑convolution, thus a lower
    evaluation score.

3.  **Local stochastic refinement** – a cheap simulated‑annealing walk that
    moves a small amount of mass between two positions and accepts
    downhill or occasional uphill moves.  This finishes the search within
    the remaining time budget.

All heavy work respects the supplied time budget (default 1000 s).  The
best sequence found is returned as a plain Python list of non‑negative
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1} (O(n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient from all active convolution entries
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if act.size:
        g /= float(act.size)
    return g


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex (Nesterov‑style)
# ----------------------------------------------------------------------
def _run_gradient_descent_full(a0: np.ndarray, n: int,
                              deadline: float, start_time: float,
                              beta: float = 0.9,
                              step0: float = 0.6,
                              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent.
    Returns the best point found and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, iter={it}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """Tiny “move ε from i to j’’ annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the given time budget,
    alternates between coarse optimisation, progressive up‑sampling and
    a final annealing phase, and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter: use the sequence produced by a previous run if
    # the global variable `height_sequence_1` is available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage definitions (length, fraction of remaining time)
    # ------------------------------------------------------------------
    # We will work with three up‑sampling levels.
    stage_lengths = [800, 2400, 4800]          # increase length progressively
    # Reserve about 15 % of the total budget for the final annealing phase.
    anneal_budget = max(30.0, 0.15 * (deadline - start_time))
    grad_budget = (deadline - start_time) - anneal_budget
    # Split the gradient budget proportionally (more time for larger n)
    total_weight = sum(stage_lengths)
    # Avoid division by zero if anything goes wrong
    total_weight = max(total_weight, 1)

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate time for this stage
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_budget * stage_frac)   # at least a few seconds
        stage_deadline = time.time() + stage_time

        # initialise a0
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # run gradient descent on the current size
        a_opt, score_opt = _run_gradient_descent_full(
            a0, n, stage_deadline, start_time,
            beta=0.9, step0=0.6, max_iter=10**7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – uses whatever length we ended with.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, steps=500000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # convert to plain Python list for the harness
    return [float(x) for x in best_seq]
