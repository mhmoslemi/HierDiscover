```

```

The algorithm works in three stages:

   The beta‑family a_i ∝ t_i^{α‑1}(1‑t_i)^{β‑1},

   temperature schedule to polish the result.

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto {x | x_i≥0, ∑x_i=1}. """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (fast for large n)."""
    n = a.shape[0]
    size = 2 * n - 1
    fft_len = 1 << ((size - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:size]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of x and y via FFT."""
    size = len(x) + len(y) - 1
    fft_len = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    c = np.fft.irfft(X * Y, n=fft_len)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up.fill(1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient descent for the max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iter: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on   f(a) = 2·n·max(conv(a,a))
    with projection onto the probability simplex.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)

    beta = 0.5                # momentum coefficient
    step0 = 1.0               # initial step size
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[{time.time() - start_time:6.2f}s] subgrad it={it:,} "
                      f"score={cur_score:.6f}")

        # active index (could be several – we pick the first)
        k = int(conv.argmax())

        idx = np.arange(n)
        # gradient of conv_k :  g_j = a_j + a_{k-j} (if the second term exists)
        g = a.copy()
        mask = (k - idx >= 0) & (k - idx < n)
        g[mask] += a[k - idx[mask]]

        # momentum + diminishing step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing polishing
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """Random mass‑transfer annealing (very cheap, used only at the end)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose move i → j (transfer a small ε)
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[{time.time() - start_time:6.2f}s] anneal it={it:,} "
                      f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Strategy:
        * generate a handful of beta‑shaped candidates at several lengths,
        * keep the best one,
        * up‑sample it to a larger length,
        * run a focussed sub‑gradient descent,
        * finish with a cheap annealing sweep.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 1) coarse global scan (few lengths, cheap beta families)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]                # modest sizes for quick scan
    np.random.shuffle(lengths)                       # avoid systematic bias

    # symmetric beta exponents to explore (α=β)
    beta_exponents = [0.5, 0.7, 0.9, 1.1, 1.3]

    for n in lengths:
        if time.time() >= deadline:
            break

        # --- generate candidates -------------------------------------------------
        cand_list = []

        # uniform
        cand_list.append(np.full(n, 1.0 / n))

        # beta shapes
        t = (np.arange(n) + 0.5) / n          # points in (0,1)
        for alpha in beta_exponents:
            beta = alpha
            # avoid exact zero/one to keep numbers finite
            eps = 1e-12
            p = np.maximum(t, eps) ** (alpha - 1)
            q = np.maximum(1.0 - t, eps) ** (beta - 1)
            w = p * q
            w /= w.sum()
            cand_list.append(w)

        # include an up‑sampled version of the currently best vector (if any)
        if best_seq is not None:
            cand_list.append(_upsample(np.asarray(best_seq), n))

        # --- evaluate each candidate --------------------------------------------
        for a0 in cand_list:
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s == 0.0:
                continue
            a0 = a0 / s                      # enforce simplex

            # cheap exact objective (sum=1 ⇒ score = 2·n·max(conv))
            conv = _conv_self(a0)
            cur_score = 2.0 * n * conv.max()

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a0.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"new global start n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2) up‑sample to a larger length and run sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is None:                     # safety fallback
        best_seq = np.full(200, 1.0 / 200)

    # target length – larger gives a lower theoretical bound (≈1)
    N_target = max(24000, len(best_seq) * 2)   # at least 24000 points
    a_up = _upsample(best_seq, N_target)

    # allocate most of the remaining budget for the descent
    remaining = deadline - time.time()
    if remaining > 10.0:
        descent_deadline = time.time() + remaining * 0.85
        a_opt, score_opt = _subgradient_descent(
            a_up, N_target, descent_deadline, start_time
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad final improve "
                  f"n={N_target} score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) final simulated‑annealing polish (cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = time.time() + min(8.0, deadline - time.time())
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-5,
            steps=30_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list (the harness expects that)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
