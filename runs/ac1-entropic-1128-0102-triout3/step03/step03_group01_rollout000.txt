def _run_adam(a0, n, slice_deadline, start_time, max_iter=200_000):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    best_a = a.copy()
    best_score = compute_score(a)
    it = 0
    lam_factor = 0.5
    lam_decay = 0.96  # reduce slowly each iteration
    while time.time() < slice_deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        # compute score and update best
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, it={it}, score={cur_score:.6f}")

        # lambda
        lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)  # shape (n,)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score

"""
Hybrid multi‑scale Adam descent + simplex projection + stochastic local search.

* Works on the probability simplex (∑a_i = 1).  The evaluation
  2·n·max(conv) is scale‑invariant, therefore we keep the sum exactly 1.
* For a given length `n` we run an Adam optimiser on a smooth
  log‑sum‑exp surrogate of the ℓ∞‑norm of the auto‑convolution.
  The surrogate parameter λ is exponentially decayed.
* After the Adam phase we perform a cheap stochastic hill‑climbing
  (mass‑transfer) that accepts only improving moves (with a tiny
  probability of uphill steps) – this usually polishes the solution.
* The procedure is applied on a schedule of increasing lengths.
  The best distribution from one level is linearly up‑sampled and used as
  warm‑start for the next level.
* The code respects the 1 000 s budget, prints progress, and finally
  returns a list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear auto‑convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length `len(seq)` to `new_n`, re‑normalised."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Objective and surrogate gradient
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a) ≈ max(conv).
    Returns a dense vector of size n.
    """
    n = a.shape[0]
    # numerically stable softmax over convolution entries
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of the weight vector with the reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]        # 2·∂conv/∂a weighted by softmax
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------


def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed surrogate.
    Returns the best vector seen (projected onto the simplex) and its exact score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5

    best_a = a.copy()
    best_score = _objective(a)

    lam_factor = 0.5               # initial λ = lam_factor * max(conv)
    lam_decay = 0.96               # exponential decay per iteration
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, it={it}, score={cur_score:.6f}")

        # λ schedule
        lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        # surrogate gradient
        grad = _gradient_from_conv(a, conv, lam)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic local improvement (mass transfer)
# ----------------------------------------------------------------------


def _run_local_search(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      cur_best_score: float,
                      cur_best_seq: np.ndarray,
                      start_temp: float = 5e-3,
                      end_temp: float = 1e-7,
                      max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climbing with a tiny amount of simulated annealing.
    Returns possibly improved sequence and its exact score.
    """
    a = a0.copy()
    score = _objective(a)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    it = 0
    while time.time() < deadline and it < max_steps:
        it += 1

        # linear temperature schedule based on elapsed fraction
        frac = it / max_steps
        T = start_temp * (1.0 - frac) + end_temp * frac

        # propose a mass transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 1e-14:
            continue
        eps = a[i] * np.random.rand() * 0.2      # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        # accept if improvement, else probabilistic
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Local improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The algorithm runs for at most ``budget_s`` seconds and returns a list of
    heights (a probability vector).  It combines a multi‑scale Adam descent

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Optional starter from the previous run
    # ------------------------------------------------------------------
    try:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    best_score = float("inf")
    global_a = starter

        # --------------------------------------------------------------
        # Allocate a slice of the remaining budget for this length
        # --------------------------------------------------------------

        if remaining <= 0.0:

        # --------------------------------------------------------------
        # Adam optimisation on the simplex
        # --------------------------------------------------------------
        a_opt, score_opt = _run_adam(a0, n, slice_time, start_time)

        if score_opt < best_score:

            best_seq = a_opt.copy()

        # --------------------------------------------------------------
        # Local stochastic refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:

        # keep this distribution as warm‑start for the next scale

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------

        best_seq = np.full(200, 1.0 / 200.0)

    # Return as plain Python list (evaluate_sequence will handle the rest)
    return [float(x) for x in best_seq]
