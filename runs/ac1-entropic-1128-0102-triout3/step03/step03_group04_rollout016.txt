def weighted_gradient(a, beta):
    conv = conv_fft(a)  # length 2n-1
    # compute softmax weights of conv entries: w_i = exp(beta * (conv_i - max_conv)) normalized.
    max_c = conv.max()
    # subtract max to avoid overflow.
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    # compute first term: correlation w with a
    # We can compute using FFT:
    # N = next pow2 of len(w) + len(a) -1 = 3n-2
    L = conv.shape[0] + a.shape[0] - 1
    N = 1 << (L-1).bit_length()
    # FFT of w and a, compute corr1 as ifft(fft(w,padded)*conj(fft(a,padded))) maybe.
    # usage: rfft returns length N//2+1.
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a, n=N)
    corr1 = np.fft.irfft(W * np.conj(A), n=N)  # shape N
    # same for w_rev
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=N)
    corr2 = np.fft.irfft(W_rev * np.conj(A), n=N)
    # extract first part
    offset = a.shape[0] - 1  # n-1
    first = corr1[offset : offset + a.shape[0]]
    # second: use indices (2*offset + conv_len - 1?) Wait we computed earlier.
    # conv_len = len(w) = 2n-1 = 2*a.shape[0] -1.
    # L = len(w)+len(a)-1 = 3n-2.
    # index for second term = offset_rev + (len(w)-1 - k)
    # offset_rev = a.shape[0] -1 = offset, len(w)-1 = 2*n-2 = 2*offset
    # So index = offset + (2*offset - k) = 3*offset - k.
    # Since offset = n-1, 3*offset = 3n-3.
    second = corr2[3*offset - np.arange(a.shape[0])]
    grad = 2.0 * (first + second)
    return grad

def weighted_gradient_fft(a: np.ndarray, w: np.ndarray, n: int) -> np.ndarray:
    # a length n, w length m=2n-1
    m = w.shape[0]
    L = m + n - 1  # 3n-2
    N = 1 << (L - 1).bit_length()
    # FFT of a and reversed a
    A = np.fft.rfft(a, n=N)
    A_rev = np.fft.rfft(a[::-1], n=N)
    # FFT of w and w_rev
    W = np.fft.rfft(w, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    # correlation first term
    corr1 = np.fft.irfft(W * np.conj(A), n=N)
    # correlation second term (using w_rev and a)
    corr2 = np.fft.irfft(W_rev * np.conj(A), n=N)
    offset = n - 1
    first = corr1[offset : offset + n]
    # second term indices reversed
    second = corr2[3*offset - np.arange(n)]
    grad = 2.0 * (first + second)
    return grad

"""
Propose a sequence of non‑negative heights that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

using a three‑stage search:

1️⃣  Power‑law sweep (coarse + fine) to obtain a good deterministic seed.
2️⃣  Adam optimiser on the softmax‑parameterisation of the vector.
    The loss is a smooth approximation of the max‑convolution via a
    soft‑max (temperature β).  The gradient of the weighted sum of
    convolution entries is obtained analytically with O(N log N) work
    using FFT‑based correlations.
3️⃣  Return the best vector found (checked with ``evaluate_sequence``).

The optimiser works directly on logits (unconstrained variables) and
projects onto the probability simplex by the softmax, guaranteeing
non‑negativity and unit sum (the objective is scale‑invariant).  The
temperature β is increased gradually so that the optimiser first
flattens the whole autocorrelation and then focuses on the true peak.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Return a non‑negative vector a_i = (i+offset)^(-p) (i=0…n‑1) normalised
    to unit L1‑norm."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:          # defensive fallback
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Full discrete self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    # rfft is sufficient because the signal is real
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


def weighted_gradient_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  g(a) = Σ_i w_i·conv_i(a)  w.r.t. a.
    Both a (length n) and w (length 2n‑1) are real non‑negative vectors.
    The formula is
        ∂g/∂a_k = Σ_i w_i ( a_{i‑k} + a_{k‑i} )
    which can be obtained with two FFT‑based correlations.
    """
    n = a.shape[0]          # length of a
    m = w.shape[0]          # = 2n‑1
    # length of the full correlation (a * w) is m + n - 1 = 3n - 2
    L = m + n - 1
    # next power of two for efficient FFT
    N = 1 << (L - 1).bit_length()

    A = np.fft.rfft(a, n=N)                # FFT of a
    W = np.fft.rfft(w, n=N)                # FFT of w
    W_rev = np.fft.rfft(w[::-1], n=N)      # FFT of reversed w

    # correlation w ⋆ a  (first term)
    corr1 = np.fft.irfft(W * np.conj(A), n=N)

    # correlation w_rev ⋆ a  (second term)
    corr2 = np.fft.irfft(W_rev * np.conj(A), n=N)

    offset = n - 1                         # centre of the correlation
    first = corr1[offset : offset + n]     # indices 0 … n‑1
    # second term needs the reversed slice: indices 3*offset‑k
    second = corr2[3 * offset - np.arange(n)]

    grad = 2.0 * (first + second)
    return grad

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score.  The routine respects the time budget (≈950 s) and returns the
    best list found.
    """
    rng = np.random.default_rng(seed)

    # ----- configuration -------------------------------------------------
    n = 2500                     # length of the candidate vector
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – coarse & fine power‑law sweep (deterministic seed)
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    best_p = None
    best_offset = None

    p_grid = np.arange(0.40, 0.71, 0.015)   # coarse grid
    offsets = [0.3, 0.5, 1.0, 2.0]

    print("[search] phase 1 – coarse power‑law sweep")
    for offset in offsets:
        for p in p_grid:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=offset)
            conv = conv_fft(a)
            cur = 2.0 * n * conv.max()      # sum(a)=1 ⇒ denominator = 1
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                best_p = p
                best_offset = offset
                print(f"[grid] new best: p={p:.4f}, off={offset}, score={cur:.6f}")

    # fine sweep around the best grid point
    if best_a is not None:
        p_fine = np.arange(max(0.01, best_p - 0.02),
                           min(0.99, best_p + 0.02) + 1e-5,
                           0.001)
        print("[search] phase 2 – fine power‑law sweep")
        for offset in offsets:
            for p in p_fine:
                if time.time() > deadline:
                    break
                a = power_law_sequence(n, p, offset=offset)
                conv = conv_fft(a)
                cur = 2.0 * n * conv.max()
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    best_p = p
                    best_offset = offset
                    print(f"[fine] new best: p={p:.4f}, off={offset}, score={cur:.6f}")

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * conv_fft(best_a).max()
        print("[fallback] using uniform vector")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation on soft‑max logits
    # ------------------------------------------------------------------
    print("[opt] phase 3 – Adam optimiser (soft‑max gradient)")

    # Initialise logits (unconstrained variables) from the best seed
    eps = 1e-12
    x = np.log(best_a + eps)               # shape (n,)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.12                               # initial learning rate

    # soft‑max temperature (controls how close we are to the true max)
    beta_soft = 5.0
    beta_inc = 0.25                         # increase per schedule step

    decay_every = 5000                      # learning‑rate decay interval

    global_best_a = best_a.copy()
    global_best_score = best_score

    it = 0
    while time.time() < deadline and it < 500_000:
        # ----- probability simplex via soft‑max -----
        a = np.exp(x - np.max(x))           # stabilised exponentiation
        a /= a.sum()

        # ----- current objective -----
        conv = conv_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < global_best_score:
            global_best_score = cur_score
            global_best_a = a.copy()
            print(f"[adam {it:07d}] new global best = {cur_score:.6f}")

        # ----- soft‑max weights over all convolution entries -----
        # w_i ∝ exp(beta·(conv_i - max)), ensuring numerical stability
        w = np.exp(beta_soft * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----- gradient of Σ_i w_i·conv_i w.r.t. a -----
        grad_a = weighted_gradient_fft(a, w)

        # ----- chain rule: gradient w.r.t. logits x (soft‑max) -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam parameter update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Clip extreme steps (helps stability)
        step = np.clip(step, -0.5, 0.5)

        x -= step

        # occasional random jitter to escape shallow plateaus
        if rng.random() < 0.001:
            x += rng.normal(scale=0.01, size=n)

        # learning‑rate decay & temperature increase schedule
        if (it + 1) % decay_every == 0:
            lr *= 0.7
            beta_soft = min(beta_soft + beta_inc, 200.0)

        it += 1

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")
    print(f"[info] best power‑law parameters: p={best_p:.5f}, offset={best_offset}")

    return global_best_a.tolist()
