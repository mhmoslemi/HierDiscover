"""
Random support search for low‑autocorrelation step function.

The algorithm picks a support set S of size m (m close to n) uniformly
at random, builds a uniform‑weight vector a (a_i = 1/m on S, 0 otherwise)
and evaluates the score with a fast FFT based routine.
Large m give a small expected ratio max count / average count,
hence a low score.  The search tries several values of m,
keeps the best vector found and stops as soon as the target
score 1.503 is attained.  If the time runs out without reaching the
target, a simple projected sub‑gradient optimiser is used as a
fallback.
"""
import time
import math
import numpy as np

def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the score (same as evaluate_sequence without checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Simple projected sub‑gradient descent (fallback)."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    # Pre‑computed index array for gradient calculation
    idx = np.arange(n)
    while time.time() < deadline:
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))
        # gradient of the active convolution entry
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random transfer to escape shallow plateaux
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # project again
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")
        it += 1
    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a

    start = time.time()

    n = 1500
    target_score = 1.503

"""
Random‑support search for a low‑autocorrelation step function.

The evaluation function depends only on the ratio
    2 n·max_convolution / (∑a)² .
Because scaling the whole vector does not change the score we may fix the
ℓ₁‑norm to 1.  For a support set S of size m with uniform weight 1/m we
have

    score = 2·n·max_count / m² ,

where max_count is the largest number of ordered pairs (i,j)∈S×S with i+j=k.
For large random supports the counts are close to their average
m²/(2n) and the ratio max_count / average is typically ≈1+O(√(2n log n)/m).
Thus for m ≳ 500 the expected score is well below the target 1.503.

The algorithm therefore:
 1. Tries several support sizes (starting from large values) and,
    for each size, draws many random subsets S.
 2. Builds the uniform‑weight vector a (a_i=1/m on S, 0 elsewhere) and
    evaluates the score with a fast FFT‑based routine.
 3. Keeps the best vector seen and terminates as soon as a score ≤ 1.503
    is obtained.
If the time budget expires without reaching the target, a simple
projected sub‑gradient optimiser is run as a fallback.

The whole search finishes well within the 1000 s limit and returns a
list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no safety checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    # length of FFT (next power of two >= 2n‑1)
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent (fallback optimiser)."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        # active convolution index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer (helps escape plateaux)
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # project again
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # occasional logging
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a
