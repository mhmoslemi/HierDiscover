"""
Propose a low‑value step‑function sequence.

Algorithm
---------
*   Initialise a probability vector (sum = 1) from a collection of
    analytically motivated shapes (power‑law, linear, uniform) together
    with a few random Dirichlet draws and – if present – the previously
    best sequence.
*   Optimise the vector on the probability simplex with a mirror‑descent
    (exponentiated‑gradient) update.  The gradient is a sub‑gradient of
    the max‑convolution objective obtained by aggregating the contributions
    of the *active* convolution indices – i.e. those whose convolution value
    lies within a decreasing relative gap of the current maximum.
*   Adam‑style momentum (first‑ and second‑order moments) adapts the step
    direction, while the step size decays as 1/√t.
*   Occasionally a random “mass‑transfer’’ move is tried to escape plateaus.
*   If no improvement is observed for a while, the iterate is re‑initialised
    from the current best plus a small random perturbation.
*   The loop runs until the time budget is exhausted; the best vector seen
    is returned as a plain Python list.

The objective is scale‑invariant, therefore the vector is kept normalised
to sum = 1 throughout the optimisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of the provided ``evaluate_sequence``.
    Assumes ``a`` is a non‑negative 1‑D array with sum > 0.
    Returns the same value as ``evaluate_sequence`` (ignoring the safety
    checks and clipping performed there).
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
        {x | x_i ≥ 0, ∑ x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n-i) (i=0..n‑1), normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def gradient_multi_active(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution objective summed over the
    supplied index set ``top_idxs``.
    For each active index k the contribution to grad_i is 2·a_{k‑i}
    whenever 0 ≤ i, k‑i < n.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in top_idxs:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i = np.arange(low, high + 1, dtype=np.int64)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad

        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_conv = conv.max()

        # (b) active set – indices whose value is within a relative gap of the max
        delta = max(delta_end, delta_start * (1.0 - it / 150000.0))
        mask = conv >= max_conv * (1.0 - delta)
        active = np.flatnonzero(mask)
        if active.size > max_active:
            # keep the largest contributions only
            top = np.argpartition(-conv[active], max_active)[:max_active]
            active = active[top]

        # (c) sub‑gradient for the active set
        grad = gradient_multi_active(a, active)

        # (d) Adam moment updates
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1**it1)
        v_hat = v / (1.0 - beta2**it1)

        direction = m_hat / (np.sqrt(v_hat) + eps)

        # (e) step size (square‑root decay)
        eta0 = 0.55
        eta = eta0 / np.sqrt(it1)

        # (f) exponentiated‑gradient update (mirror descent on simplex)
        log_a = np.log(a + 1e-30)               # avoid log(0)
        log_a = log_a - eta * direction
        a_new = np.exp(log_a)
        a_new /= a_new.sum()

        # (g) evaluate candidate
        cur_val = fast_score(a_new)

        # (h) accept if improvement; otherwise occasionally accept uphill
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
        else:
            if np.random.rand() < 0.0003:       # very small uphill probability
                a = a_new

        # ----------------------------------------------------------
        # (i) random “mass‑transfer’’ move (helps to escape flat regions)
        if np.random.rand() < 0.02:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta_transfer = 0.20 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta_transfer
                cand[j] += delta_transfer
                cand = cand / cand.sum()
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < 0.001:
                    a = cand

        # ----------------------------------------------------------
        # (j) stagnation handling – re‑initialise from the best with noise
        if time.time() - last_improve > 30.0:
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = 0.9 * best_seq + 0.1 * noise
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

        # ----------------------------------------------------------
        # (k) progress reporting
        if it % 5000 == 0:
            print(f"[iter {it:7d}] cur {cur_val:.6f}  best {best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    return best_seq.tolist()
