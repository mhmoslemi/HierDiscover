def gradient_of_softmax(a, conv, beta):
    n = a.shape[0]; m = conv.shape[0]; # m = 2n-1
    maxv = conv.max()
    w = np.exp(beta * (conv - maxv))
    w /= w.sum()
    # Use FFT
    # Determine length for convolution w * a_rev = len = m + n -1
    L = int(2**np.ceil(np.log2(m + n - 1)))
    # compute FFT of w padded L and a_rev padded L
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(W * A_rev, n=L)
    # take only first m+n-1 entries
    corr = corr[:m + n - 1]
    offset = m - 1
    grad = 2.0 * corr[offset - np.arange(n)]
    return grad

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i = (i+offset)^(-p)  (i=0…n‑1), normalised to unit L1‑norm."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full discrete self‑convolution x*x using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex { w ≥ 0, Σw = 1 }.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of  L(a)=log‑sum‑exp(β·conv(a,a))/β  w.r.t. a.
    The gradient is a weighted sum of the exact peak‑gradients.
    It is computed efficiently via a single FFT‑based correlation.
    """
    # soft‑max weights for each convolution entry
    maxv = conv.max()
    w = np.exp(beta * (conv - maxv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    n = a.shape[0]
    m = conv.shape[0]                     # = 2n‑1
    # correlation w ⋆ a  (cross‑correlation, not convolution)
    #   grad_i = 2 * Σ_k w_k * a_{k‑i}
    # This is exactly 2·corr[w, a] evaluated at lag = –i.
    L = int(2 ** np.ceil(np.log2(m + n - 1)))   # power‑of‑two length for FFT
    a_rev = a[::-1]                     # reversed a
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(W * A, n=L)[: m + n - 1]   # full correlation

    offset = m - 1                                    # zero‑lag index
    grad = 2.0 * corr[offset - np.arange(n)]
    return grad

            conv = conv_fft(a)
            grad = gradient_of_softmax(a, conv, beta)

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))
            step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
            step = np.minimum(step, a * 0.98)

            a_trial = a - step
            if rng.random() < 0.001:
                a_trial += (rng.random(n) - 0.5) * 1e-6
            a_trial = project_onto_simplex(a_trial)

            trial_score = 2.0 * n * conv_fft(a_trial).max()
            if trial_score < global_best_score:
                global_best_score = trial_score
                global_best_a = a_trial.copy()
                print(f"[restart {restart}] iter {it:07d} new global best = {trial_score:.6f}")

            cur_score = 2.0 * n * conv.max()
            if trial_score <= cur_score:
                a = a_trial
            else:
                temperature = 0.02 * np.exp(-it / 150_000.0)
                if rng.random() < np.exp(-(trial_score - cur_score) / max(temperature, 1e-12)):
                    a = a_trial

            if (it + 1) % 10_000 == 0:
                beta = min(beta + 0.5, 20.0)
                lr = max(lr * 0.9, 1e-4)

            it += 1

    # ------------------------------------------------------------------
    # final sanity check with the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[final] evaluated score = {final_score:.6f}")

    return global_best_a.tolist()
