def _refine_sequence_general(a, max_iters, deadline, mode='soft', step0=0.6, eps_start=0.05, eps_end=0.001, beta=0.9):
    n=...
    fft_len_conv=...
    fft_len_grad=...
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_of_seq(a)
    for it in range(1,max_iters+1):
        if time.time() >= deadline: break
        # conv
        Af = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(Af * Af, fft_len_conv)[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute w
        if mode == 'soft':
            eps_rel = eps_start + (eps_end - eps_start) * (it / max_iters)
            tau = max(eps_rel * max_val, 1e-14)
            w = np.exp((conv - max_val) / tau)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.full_like(w, 1.0 / w.shape[0])
        elif mode == 'hard':
            # find active indices
            # Use tolerance to include near maxima to avoid instability
            # e.g., rel tolerance 1e-12
            active = np.where(np.abs(conv - max_val) < 1e-14 * max_val + 1e-15)[0]
            # but just assign all maxima
            # compute w uniform
            w = np.zeros_like(conv)
            if active.size > 0:
                w[active] = 1.0 / active.size
            else:
                w = np.full_like(conv, 1.0 / conv.shape[0])
        # gradient
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[:len(w) + n - 1]
        g = 2.0 * conv2[n - 1 : n - 1 + n]
        # step size
        if mode == 'soft':
            step = step0 / math.sqrt(it)
        else:

"""
Multi‑restart combined soft‑max / hard‑max projected gradient descent with
progressive up‑sampling.
The algorithm keeps the coefficients on the simplex (∑a = 1, a_i ≥ 0) and
minimises 2·n·max(conv(a,a)), which is exactly the quantity returned by
``evaluate_sequence`` after the scale‑invariant clipping.  
For a fixed length we first perform a soft‑max smoothed gradient descent,
followed by a hard‑max sub‑gradient (the true sub‑gradient of the max‑norm
convolution).  Several diverse starts (uniform, cosine windows, Dirichlet,
sparse patterns and the previous best construction) are refined, the best
shape is up‑sampled (linear interpolation) and refined again.  The whole
search runs for at most ``budget_s`` seconds and returns the best sequence
found.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Score for a sequence that already sums to 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()

    a: np.ndarray,
    max_iters: int,
    deadline: float,
    mode: str = "soft",
    step0: float = 0.6,
    eps_start: float = 0.05,
    eps_end: float = 0.001,
    beta: float = 0.9,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Refine a probability vector ``a`` of length n.

    Parameters
    ----------
    a : np.ndarray
        Initial vector (already normalised to sum 1, non‑negative).
    max_iters : int
        Upper bound on the number of iterations.
    deadline : float
        Unix time after which the routine aborts.
    mode : {"soft", "hard"}
        ``soft`` – soft‑max smoothed gradient (temperature decays).
        ``hard`` – true sub‑gradient obtained from the active set of
        maximal convolution entries.
    step0 : float
        Initial step length (scaled as step0 / sqrt(iter)).
    eps_start, eps_end : float
        Only used in *soft* mode – temperature is eps_rel*max_val with
        eps_rel linearly decreasing from eps_start to eps_end.
    beta : float
        Momentum coefficient (Nesterov style).  Use beta≈0 for hard mode.

    Returns
    -------
    final_a, best_a, best_score
        ``final_a`` – the iterate after the last performed step,
        ``best_a`` – the best vector seen during the run,
        ``best_score`` – its associated score.
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    v = np.zeros_like(a)           # momentum term
    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution -----
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- weights -----
        if mode == "soft":
            eps_rel = eps_start + (eps_end - eps_start) * (it / max_iters)
            tau = max(eps_rel * max_val, 1e-14)
            w = np.exp((conv - max_val) / tau)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.full_like(w, 1.0 / w.shape[0])
        else:  # hard mode
            # pick all indices whose value is (numerically) maximal
            # a small absolute tolerance is used to be robust
            tol = max(1e-14 * max_val, 1e-15)
            active = np.where(np.abs(conv - max_val) <= tol)[0]
            if active.size == 0:
                # fallback – use the exact argmax
                active = np.array([np.argmax(conv)], dtype=int)
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size

        # ----- gradient of the smooth/max approximation -----
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        g = 2.0 * conv2[n - 1 : n - 1 + n]

        # ----- step size, momentum, projection -----
        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)

    return a, best_a, best_score

progressive up‑sampling.

    2·n·max(conv(a,a)) ,

internal clipping.  For a fixed length we first apply a soft‑max smoothed

starts from many diverse initial points (uniform, Hann window, Dirichlet
samples, sparse patterns and the previously found construction), keeps the
best shape, linearly upsamples it by roughly 1.5×, refines the larger shape,

The whole routine respects a time budget (default 1000 s) and returns the

            prev = _normalise(np.interp(x_new, x_old, prev))
        candidates.append(prev)
    except Exception:
        pass

    # b) deterministic shapes
    candidates.append(np.full(cur_n, 1.0 / cur_n))                    # uniform
    idx = np.arange(cur_n)
    tri = np.minimum(idx + 1, cur_n - idx)                          # triangular
    candidates.append(_normalise(tri.astype(float)))
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (cur_n - 1)))       # Hann window
    candidates.append(_normalise(hann))

    # c) random Dirichlet starts (smooth)
    for _ in range(5):
        candidates.append(_normalise(np.random.dirichlet(np.ones(cur_n))))

    # d) low‑concentration Dirichlet (spiky)
    for _ in range(5):
        alpha = np.full(cur_n, 0.2)
        candidates.append(_normalise(np.random.dirichlet(alpha)))

    # e) sparse starts – a few random locations carry the whole mass
    for _ in range(3):
        m = max(2, int(0.03 * cur_n))               # ~3 % non‑zero
        pos = np.random.choice(cur_n, m, replace=False)
        vals = np.random.rand(m)
        a = np.zeros(cur_n)
        a[pos] = vals
        candidates.append(_normalise(a))

    print(f"[{time.time() - start_time:6.2f}s] Starting refinement at n={cur_n}"
          f" with {len(candidates)} candidates")

    # --------------------------------------------------------------
    # 1. Refine all candidates at the current resolution
    # --------------------------------------------------------------
    for ci, cand in enumerate(candidates, 1):
        if time.time() >= deadline:
            break
        soft_iters = 15000
        hard_iters = 15000

        print(f"[{time.time() - start_time:6.2f}s] Candidate {ci}/{len(candidates)}"
              f" – soft phase ({soft_iters} iters)")
        _, best_soft, best_soft_score = _refine_sequence_general(
            cand,
            max_iters=soft_iters,
            deadline=deadline,
            mode="soft",
            step0=0.55,
            eps_start=0.06,
            eps_end=0.001,
            beta=0.9,
        )
        if time.time() >= deadline:
            break

        print(f"[{time.time() - start_time:6.2f}s] Candidate {ci}/{len(candidates)}"
              f" – hard phase ({hard_iters} iters)")
        _, best_hard, best_hard_score = _refine_sequence_general(
            best_soft,
            max_iters=hard_iters,
            deadline=deadline,
            mode="hard",
            step0=0.45,
            beta=0.0,
        )

        # keep the better of the two phases
        if best_hard_score < best_soft_score:
            local_best_seq = best_hard
            local_best_score = best_hard_score
        else:
            local_best_seq = best_soft
            local_best_score = best_soft_score

        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"   >>> New global best: score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Progressive up‑sampling (≈1.5× each step) up to a ceiling.
    # --------------------------------------------------------------
    MAX_LEN = 12000
    while time.time() < deadline and cur_n < MAX_LEN:
        next_n = int(cur_n * 1.5)
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n <= cur_n:
            break

        # linear interpolation of the current best shape
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        up = np.interp(x_new, x_old, best_seq)
        up = _normalise(up)

        cur_n = next_n
        print(f"[{time.time() - start_time:6.2f}s] Upsampled to n={cur_n}")

        # soft‑max refinement at the new size
        soft_iters = max(12000, int(20000 * cur_n / next_n))
        _, best_soft, best_soft_score = _refine_sequence_general(
            up,
            max_iters=soft_iters,
            deadline=deadline,
            mode="soft",
            step0=0.5,
            eps_start=0.05,
            eps_end=0.001,
            beta=0.9,
        )
        if time.time() >= deadline:
            break

        # hard‑max refinement at the new size
        hard_iters = max(12000, int(20000 * cur_n / next_n))
        _, best_hard, best_hard_score = _refine_sequence_general(
            best_soft,
            max_iters=hard_iters,
            deadline=deadline,
            mode="hard",
            step0=0.4,
            beta=0.0,
        )

        # pick the better version to carry forward
        if best_hard_score < best_soft_score:
            cand_seq = best_hard
            cand_score = best_hard_score
        else:
            cand_seq = best_soft
            cand_score = best_soft_score

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"   >>> Updated global best after up‑sampling: score={best_score:.6f}")

        # continue the chain with the refined version
        up = cand_seq

    # --------------------------------------------------------------
    # 3. Final long refinement on the best length discovered
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_soft_iters = 50000
        final_hard_iters = 50000

        print(f"[{time.time() - start_time:6.2f}s] Final soft refinement on n={best_seq.shape[0]},"
              f" {final_soft_iters} iters")
        _, best_soft, best_soft_score = _refine_sequence_general(
            best_seq,
            max_iters=final_soft_iters,
            deadline=deadline,
            mode="soft",
            step0=0.45,
            eps_start=0.03,
            eps_end=0.0005,
            beta=0.9,
        )
        if best_soft_score < best_score:
            best_score = best_soft_score
            best_seq = best_soft.copy()
            print(f"   >>> Improvement after final soft phase: score={best_score:.6f}")

        if time.time() < deadline:
            print(f"[{time.time() - start_time:6.2f}s] Final hard refinement on n={best_seq.shape[0]},"
                  f" {final_hard_iters} iters")
            _, best_hard, best_hard_score = _refine_sequence_general(
                best_seq,
                max_iters=final_hard_iters,
                deadline=deadline,
                mode="hard",
                step0=0.35,
                beta=0.0,
            )
            if best_hard_score < best_score:
                best_score = best_hard_score
                best_seq = best_hard.copy()
                print(f"   >>> Improvement after final hard phase: score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best sequence (plain Python floats)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200)

    best_seq = _normalise(best_seq)   # protect against numerical drift
    print(f"[{time.time() - start_time:6.2f}s] Finished. Best internal score = {best_score:.6f}")

    return [float(x) for x in best_seq]
