"""
Propose a step‑function by a long‐run exponentiated‑gradient (mirror descent)
optimizer on the probability simplex.

Key ideas
---------
* Work directly with a probability vector (sum = 1).  The evaluation
  `evaluate_sequence` returns `2·n·max(conv(a,a))` for such a vector,
  so minimising the max‑convolution is the sole goal.
* Use an *accurate* sub‑gradient of `max(conv)` – the previous code
  double‑counted off‑diagonal terms.
* Keep a relatively large active set (the top‑k convolution entries)
  to obtain a more informative descent direction.
* Apply an Adam‑like momentum to the gradient and a decreasing step size.
* Start from the best known construction (`height_sequence_1` if present),
  progressively up‑sample up to `MAX_LEN = 8000`, and then spend the
  remaining time performing only gradient updates (no extra up‑sampling).
* Finish with a cheap stochastic “mass‑transfer” local search.

The routine runs for at most ``budget_s`` seconds (default 1000) and
returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """Return the objective value 2·n·max(conv(a,a)) for a vector that sums to 1."""
    n = a.shape[0]
    # length for FFT (next power of two for speed)
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray):
    """Compute full auto‑convolution of ``a`` and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_set(conv: np.ndarray, max_val: float,
                top_k: int = 20, rel_eps: float = 0.001) -> np.ndarray:
    """
    Return a set of indices that are likely to be active at the optimum.
    We take all entries within ``rel_eps`` of the current maximum and,
    if they are fewer than ``top_k``, we supplement with the global top‑k.
    """
    if rel_eps > 0:
        eps = max_val * rel_eps
        idx = np.where(conv >= max_val - eps)[0]
    else:
        idx = np.array([], dtype=int)

    if idx.size < top_k:
        # add the globally largest entries
        extra = np.argpartition(conv, -top_k)[-top_k:]
        idx = np.unique(np.concatenate([idx, extra]))
    return np.sort(idx)


def _gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max(conv)`` restricted to the active set.
    For each active convolution index k we add a_{k‑i} to the gradient
    component i (and an extra a_i when i=j, i.e. k = 2·i).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_idx:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i = np.arange(low, high + 1, dtype=int)
        j = k - i
        # contribution a_{k-i}
        g[i] += a[j]
        # diagonal terms need a second a_i (since derivative of a_i^2 is 2·a_i)
        diag = i == j
        if np.any(diag):
            g[i[diag]] += a[i[diag]]
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` score.
    The algorithm runs for at most ``budget_s`` seconds and prints intermediate
    improvements.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise – use the best known construction if available.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        init = np.maximum(init, 0.0)
        if init.sum() == 0.0:
            raise ValueError
        init = init / init.sum()
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"Initialised from height_sequence_1: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        # fallback: random simplex point with moderate length
        n0 = 1800
        init = np.random.rand(n0)
        init = init / init.sum()
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"Random initialisation: n={n0}, score={best_score:.6f}")

    current = best_seq.copy()
    cur_n = current.shape[0]

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling (up to MAX_LEN) with a long exponentiated‑gradient phase.
    # ------------------------------------------------------------------
    MAX_LEN = 8000               # maximal length we will allow
    MOMENTUM = 0.9               # Adam‑style exponential moving average for the gradient
    ETA0 = 0.8                   # base step size for the mirror descent
    TOP_K = 30                  # active‑set size (larger than before)
    REL_EPS = 0.001

    # Store a momentum vector that will be re‑initialised whenever we change size.
    momentum = None

    while time.time() < deadline and cur_n < MAX_LEN:
        # decide next size (≈1.35× larger, capped by MAX_LEN)
        next_n = int(cur_n * 1.35) + 1
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n <= cur_n:
            break

        # ----- up‑sample current shape ---------------------------------
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a = np.interp(x_new, x_old, current)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        a = a / a_sum if a_sum > 0 else np.full(next_n, 1.0 / next_n)

        # reset momentum on size change
        momentum = np.zeros_like(a)

        # iteration budget for this size (enough to make progress)
        # we aim for roughly 40 000 iterations for the smallest size and scale down.
        target_iters = max(20000, int(40000 * 1800 / next_n))
        it = 0
        while time.time() < deadline and it < target_iters:
            it += 1

            conv, max_val = _conv_and_max(a)
            cur_score = 2.0 * next_n * max_val
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start:6.2f}s] Improve: n={next_n}, it={it}, score={cur_score:.6f}")

            # active‑set selection
            K = _active_set(conv, max_val, top_k=TOP_K, rel_eps=REL_EPS)

            # exact sub‑gradient on the active set
            g = _gradient(a, K)

            # Adam‑style momentum update (bias correction omitted – not needed for our schedule)
            momentum = MOMENTUM * momentum + (1.0 - MOMENTUM) * g

            # learning rate decays like 1/√t
            eta = ETA0 / math.sqrt(it)

            # exponentiated‑gradient (mirror descent on the simplex)
            a = np.maximum(a, 1e-15)                 # avoid log(0)
            log_a = np.log(a) - eta * momentum
            log_a -= np.max(log_a)                  # numerical stabilisation
            a = np.exp(log_a)
            a[a < 1e-15] = 1e-15
            a = a / a.sum()

        # prepare for next stage
        current = a.copy()
        cur_n = next_n

    # ------------------------------------------------------------------
    # 3. Long gradient phase at final size (use remaining time)
    # ------------------------------------------------------------------
    a = current.copy()
    n = a.shape[0]
    if momentum is None or momentum.shape[0] != n:
        momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv, max_val = _conv_and_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start:6.2f}s] Final‑phase improve: it={it}, score={cur_score:.6f}")

        K = _active_set(conv, max_val, top_k=TOP_K, rel_eps=REL_EPS)
        g = _gradient(a, K)
        momentum = MOMENTUM * momentum + (1.0 - MOMENTUM) * g
        eta = ETA0 / math.sqrt(it + 1000)   # a tiny offset prevents huge steps at the very start
        a = np.maximum(a, 1e-15)
        log_a = np.log(a) - eta * momentum
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a[a < 1e-15] = 1e-15
        a = a / a.sum()

    # ------------------------------------------------------------------
    # 4. Short stochastic local search (mass transfers) with remaining seconds.
    # ------------------------------------------------------------------
    # If there is still a couple of seconds left we do a cheap hill‑climb.
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        max_ls_iters = 15000
        for ls_it in range(max_ls_iters):
            if time.time() >= deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move a modest fraction of the source mass
            delta = a[i] * (np.random.rand() * 0.04)   # ≤ 4 % of a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            if a_new[i] < -1e-12:          # safeguard against numerical drift
                continue
            conv, max_val = _conv_and_max(a_new)
            new_score = 2.0 * n * max_val
            if new_score < best_score - 1e-10:
                best_score = new_score
                best_seq = a_new.copy()
                a = a_new
                print(f"[{time.time() - start:6.2f}s] LS improve: it={ls_it}, score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence seen.
    # ------------------------------------------------------------------
    print(f"Finished. Best score = {best_score:.6f}, length = {len(best_seq)}")
    # convert to plain Python floats (evaluate_sequence expects a list)
    return [float(x) for x in best_seq]
