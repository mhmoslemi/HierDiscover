def _run_gradient_descent(a0, n, deadline):

    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:

            best_a = a.copy()

"""
Hybrid optimizer for the step‑function objective.

* Works on the probability simplex (∑a_i = 1) – the objective
  `evaluate_sequence` is then `2·n·max(conv)`.
* Phase 1 – smoothed log‑sum‑exp gradient descent (as in the reference
  code) but with a more aggressive λ‑schedule and a larger step‑size.
* Phase 2 – guided simulated‑annealing on the final (largest) scale.
  The gradient of the active (maximal) convolution entries is used to
  bias the source of a mass‑transfer move, which dramatically speeds
  up convergence.
* A multi‑scale schedule up‑samples the current best distribution to
  larger lengths; early stages receive only a short gradient‑descent
  budget, the last (largest) length gets the remaining time for a long
  annealing sweep.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 2 * n - 1
    fft_len = 1 << ((size - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:size]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp gradient descent
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of the log‐sum‑exp surrogate."""
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)                # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """Smoothed gradient descent with momentum on the simplex."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 0.8
    lam_factor = 0.5          # initial λ = lam_factor * max(conv)
    lam_decay = 0.5
    lam_decay_interval = 500  # halve λ more often than the reference code
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] gradient improve: n={n} it={it} score={cur_score:.6f}")

        # update smoothing parameter
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Guided simulated‑annealing refinement (final stage)
# ----------------------------------------------------------------------


def _run_guided_annealing(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing where the source index is chosen
    proportionally to the sub‑gradient of the active (maximal) convolution.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    grad = None               # sub‑gradient (updated every 200 steps)
    it = 0
    len_conv = conv.size

    while time.time() < deadline:
        it += 1

        # temperature schedule (slow exponential decay)
        T = 0.05 * math.exp(-it / 200_000.0) + 1e-12

        # recompute sub‑gradient periodically
        if grad is None or it % 200 == 0:
            M = conv.max()
            mask = (conv == M).astype(float)          # shape (2n‑1)
            rev = a[::-1]
            full = _conv_fft(mask, rev)
            start = n - 1
            grad = 2.0 * full[start:start + n]       # all non‑negative
            grad_sum = grad.sum()
            if grad_sum > 0.0:
                prob_src = grad / grad_sum
            else:
                prob_src = np.full(n, 1.0 / n)

        # pick a source index (mass will be taken from here)
        i = np.random.choice(n, p=prob_src)
        if a[i] <= 1e-14:
            continue

        # pick a destination index uniformly (different from i)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a random fraction of the source mass
        eps = a[i] * np.random.rand() * 0.10   # ≤10 % of source
        if eps <= 0.0:
            continue

        # ----- incremental update of convolution -----
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        new_conv = conv.copy()
        # linear part: -2·eps·a shifted by i, +2·eps·a shifted by j
        new_conv[i:i + n] -= 2.0 * eps * a
        new_conv[j:j + n] += 2.0 * eps * a
        # quadratic delta = delta*delta
        idx2i = 2 * i
        if idx2i < len_conv:
            new_conv[idx2i] += eps * eps
        idx2j = 2 * j
        if idx2j < len_conv:
            new_conv[idx2j] += eps * eps
        idxij = i + j
        if idxij < len_conv:
            new_conv[idxij] -= 2.0 * eps * eps

        new_score = 2.0 * n * new_conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = new_a
            conv = new_conv
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n} it={it} score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Returns a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm runs for at most `budget_s` seconds
    and combines multi‑scale smoothed gradient descent with a final,
    guided simulated‑annealing polishing step.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Starter sequence (if the platform provides one)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    best_seq = None
    best_score = float("inf")
    global_a = starter

    # Length schedule – we focus most of the budget on the largest scale.
    lengths = [800, 1600, 3200, 6400, 9600]   # ascending
    # If we already have a starter of a different length, prepend it.
    if global_a is not None:
        init_len = global_a.shape[0]
        if init_len not in lengths:
            lengths = [init_len] + lengths
    lengths = sorted(set(lengths))

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise a0 at the current scale
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # Allocate a slice of the remaining budget for this scale
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # Early (smaller) scales get a modest gradient budget.
        # The last scale receives everything that is left.
        if idx < len(lengths) - 1:
            # give at most 30 s (or 5 % of the remaining time) to gradient
            grad_time = min(30.0, max(5.0, remaining * 0.05))
            grad_deadline = time.time() + grad_time
        else:
            # final scale – use almost the entire remaining time for gradient,
            # leaving a small window for the annealing sweep.
            grad_time = max(30.0, remaining * 0.6)
            grad_deadline = time.time() + grad_time

        # ------------------------------------------------------------------
        # Phase 1 – smoothed gradient descent
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_gradient_descent(a0, n, grad_deadline, start_time)

        # normalise (tiny numerical drift may appear)
        a_opt /= a_opt.sum()

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after gradient: n={n} score={score_opt:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – guided annealing (only on the final scale)
        # ------------------------------------------------------------------
        if idx == len(lengths) - 1 and time.time() < deadline:
            anneal_deadline = deadline
            a_ref, score_ref = _run_guided_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
            )
            a_ref /= a_ref.sum()
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n} score={score_ref:.6f}")

        # Remember the best distribution for the next up‑sampling step
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fall‑back (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
