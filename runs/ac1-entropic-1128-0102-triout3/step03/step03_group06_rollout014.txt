def project_simplex(v):
    if np.sum(v) == 0:
        # return uniform
        return np.full_like(v, 1.0/v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
        theta = 0
    else:
        rho = np.where(cond)[0][-1] + 1   # index
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def adam_optimize(a_start, time_budget):
    a = a_start.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    # hyperparameters
    lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)

    n = len(a)

    size2 = 1 << ((3*n-2).bit_length())
    deadline = time.time() + time_budget
    while time.time() < deadline:
        t += 1
        # compute conv a*a
        fft_a = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fft_a * fft_a, size1)[:2*n-1]
        # evaluate real objective
        max_conv = np.max(conv)
        # compute smoothed gradient
        # choose alpha maybe based on iteration or target: start small then increase
        alpha = 10.0 + (t/1e5)*40.0   # up to 50
        # compute weight using log-sum-exp for stability

        weight = exp_term / exp_term.sum()

def adam_optimize(a_init, time_limit):
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = len(a)
    size1 = 1 << ((2*n-1).bit_length())
    size2 = 1 << ((3*n-2).bit_length())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8; lr=0.1
    start = time.time()
    iteration = 0
    while time.time() - start < time_limit - 0.2:
        iteration += 1
        # compute convolution a*a
        fft_a = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fft_a * fft_a, size1)[:2*n-1]
        # compute smoothed gradient
        # schedule alpha
        elapsed = time.time() - start
        frac = elapsed / time_limit
        alpha = 5.0 + 40.0 * frac   # ranging 5 to 45
        maxc = conv.max()
        # compute weight
        exp_term = np.exp(alpha * (conv - maxc))
        w = exp_term / exp_term.sum()
        # compute gradient via FFT
        w_rev = w[::-1]
        fft_wr = np.fft.rfft(w_rev, size2)
        fft_a2 = np.fft.rfft(a, size2)
        conv_rev = np.fft.irfft(fft_wr * fft_a2, size2)[:3*n-2]
        grad = 2.0 * conv_rev[n-1:2*n-1][::-1]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = project_simplex(a)
        # check best
        if iteration % 200 == 0:
            val = fast_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[adam iter {iteration}] new best {best_val:.6f}")
    return best_a
```

def project_simplex(v):
    # Based on algorithm by Wang & Carreira-Perpinan (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)

    if not np.any(cond):
        # all zeros case
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

                break
        if time.time() >= greedy_deadline:

    # 2. Optimization stage
    time_left = deadline - time.time()
    if time_left > 5.0:
        if best_seq is None:
            # random Dirichlet init
            a0 = np.random.dirichlet(np.ones(n))

"""
Propose a low‑score step‑function.

The algorithm works in two stages:

1. **Greedy support‑set construction** – we try to find a sparse set
   S ⊂ {0,…,n‑1} of a given size k such that the ordered‑pair count
   for every possible sum does not exceed a bound that guarantees a
   score ≤ target (1.503).  The construction is fast and often yields a
   usable starting point.

2. **Smoothed‑max Adam optimisation** – starting from the best vector
   found in stage 1 (or a random Dirichlet draw) we minimise a smooth
   approximation of the maximal auto‑convolution value:
       f(a) = (1/α)·log Σₖ exp(α·(a∗a)ₖ) .
   The gradient of f is computed analytically using FFTs, then an
   Adam‑style update is performed and the iterate is projected onto the
   probability simplex (non‑negative, sum = 1).  The smoothing parameter
   α grows from 5 to ≈45 during the run, allowing the optimiser to first
   move in a well‑behaved landscape and later focus on the true maximum.

Both stages respect the 1000 s time limit and keep all numbers
non‑negative and ≤ 1000, so the final sequence can be passed directly
to `evaluate_sequence`.
"""

import numpy as np
import math
import time


# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence (no security checks) for a
    non‑negative vector a.  The vector is assumed to be already normalised
    (sum > 0).  Returns the evaluation score.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """
    For a support set S (array of indices) compute the ordered‑pair count
    for each possible convolution index 0 … 2·n‑2.
    """
    sums = np.add.outer(S, S)
    counts = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return counts


# ----------------------------------------------------------------------
def greedy_build_set(n: int, k_target: int, max_allowed: int) -> np.ndarray | None:
    """
    Build a support set of exactly `k_target` indices such that the ordered
    pair count for every sum never exceeds `max_allowed`.  The algorithm
    scans a random permutation of candidates and accepts an index only if
    the new counts stay within the bound.
    Returns the set as a 1‑D numpy array or None on failure.
    """
    order = np.random.permutation(n)
    S = []
    counts = np.zeros(2 * n - 1, dtype=np.int16)

    for idx in order:
        if len(S) >= k_target:
            break

        # self‑pair
        if counts[idx + idx] + 1 > max_allowed:
            continue

        # cross‑pair with existing members
        if S:
            existing = np.array(S, dtype=np.int32)
            sums = idx + existing
            if np.any(counts[sums] + 2 > max_allowed):
                continue

        # accept the index
        S.append(int(idx))
        counts[idx + idx] += 1
        if S[:-1]:
            counts[idx + existing] += 2

    return np.array(S, dtype=np.int32) if len(S) == k_target else None


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def adam_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Adam optimiser with a smoothly‑approximated max‑convolution loss.
    The optimization respects the simplex constraint via projection.
    """
    a = a_init.copy()
    n = a.size
    best_a = a.copy()
    best_val = fast_score(a)

    # FFT sizes (powers of two) for the two convolutions we need
    size1 = 1 << ((2 * n - 1).bit_length())          # for a*a
    size2 = 1 << ((3 * n - 2).bit_length())          # for w_rev * a

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.15

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    start = time.time()
    it = 0
    while time.time() - start < time_limit - 0.2:
        it += 1

        # ----- a * a (full convolution) -----
        fft_a = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fft_a * fft_a, size1)[: 2 * n - 1]

        # ----- smooth approximation of max(conv) -----
        # α grows linearly from 5 to 45 during the run
        elapsed = time.time() - start
        frac = elapsed / time_limit
        alpha = 5.0 + 40.0 * frac

        maxc = conv.max()
        # stable softmax
        exp_term = np.exp(alpha * (conv - maxc))
        w = exp_term / exp_term.sum()

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        fft_wr = np.fft.rfft(w_rev, size2)
        fft_a2 = np.fft.rfft(a, size2)
        conv_rev = np.fft.irfft(fft_wr * fft_a2, size2)[: 3 * n - 2]
        grad = 2.0 * conv_rev[n - 1: 2 * n - 1][::-1]   # shape (n,)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_simplex(a)

        # ----- bookkeeping -----
        if it % 250 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[adam iter {it:6d}] new best {best_val:.6f}")

    print(f"[adam] finished after {it} iterations, best {best_val:.6f}")
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with evaluation score ≤ 1.503.
    The routine first tries a greedy support‑set construction for several
    support sizes, then refines the best candidate with the Adam optimiser.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    n = 1500           # length of the step function
    target_score = 1.503

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1) Greedy support‑set search (quick, cheap)
    support_sizes = [500, 400, 300, 250, 200, 150, 120, 110, 100, 80]

    greedy_deadline = deadline - 200.0   # reserve at least 200 s for optimisation
    for k in support_sizes:
        if time.time() >= greedy_deadline:
            break
        allowed = max(1,
                      int(math.floor(target_score * k * k / (2.0 * n))))
        print(f"[greedy] try k={k}, allowed max count={allowed}")

        local_deadline = min(greedy_deadline, time.time() + 5.0)  # 5 s per size
        attempts = 0
        while time.time() < local_deadline:
            attempts += 1
            S = greedy_build_set(n, k, allowed)
            if S is None:
                continue
            cnt = compute_counts(S, n)
            maxc = int(cnt.max())
            if maxc > allowed:
                continue
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / k
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[found] k={k}, attempts={attempts}, maxc={maxc}, score={val:.6f}")
            # a modest early stop if we already beat the target comfortably
            if val <= target_score - 0.015:
                print("[greedy] strong candidate found, exiting greedy phase")
                break
        # stop early if a very good candidate already exists
        if best_seq is not None and best_val <= target_score - 0.015:
            break

    # ------------------------------------------------------------------
    # 2) Adam based optimisation (takes the remaining time)
    time_left = deadline - time.time()
    if time_left > 10.0:          # need a bit of room for the optimiser
        if best_seq is None:
            # random Dirichlet start if greedy failed
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = best_seq.copy()
        print(f"[optim] starting Adam optimiser with {time_left:.1f}s remaining")
        a_opt = adam_optimize(a0, time_left)
        val_opt = fast_score(a_opt)
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[optim] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)
        print("[fallback] using uniform distribution")

    print(f"[final] best value returned: {best_val:.6f}")
    return best_seq.tolist()
