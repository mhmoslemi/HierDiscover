"""
Propose a low‑value step‑function sequence.

Algorithm
---------
1.  Build a pool of analytically‑motivated probability vectors
    (power‑law, exponential, linear‑descending, uniform and the
    previously‑found best if available) on the simplex ∑a_i = 1,
    a_i ≥ 0.

2.  Pick the best pool member according to a fast version of
    ``evaluate_sequence`` (FFT based convolution).

3.  Optimise the chosen vector with a projected **Adam** optimiser.
    At each iteration:
        • compute the full self‑convolution via FFT,
        • select the K largest convolution entries (these are the
          active constraints of the max‑operator),
        • form the sub‑gradient of the max by aggregating the
          contributions of all active indices,
        • perform an Adam step and project back onto the probability
          simplex.
    A few stochastic “mass‑transfer’’ moves are interleaved to escape
    plateaus, and after a period without improvement the current iterate
    is re‑initialised from the best solution plus a small random perturbation.

4.  The loop runs for the allotted time budget (≈1000 s).  The best
    sequence encountered is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for ``evaluate_sequence`` – assumes a is a non‑negative
    1‑D array and uses the FFT for the convolution.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_exponential(n: int, base: float = 0.95, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ base^{i+offset}, normalised to sum 1."""
    idx = np.arange(n, dtype=np.float64) + offset
    a = np.power(base, idx)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n-i), normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def gradient_multi_active(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max_k (a*a)_k`` aggregated over the indices in
    ``top_idxs``.  For each k ∈ top_idxs the contribution is
        g_i ← g_i + 2·a_{k-i}   (when 0 ≤ k-i < n)
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in top_idxs:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i = np.arange(low, high + 1)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Search for a step‑function with a low evaluation value."""
    np.random.seed(seed)
    # small safety margin to guarantee graceful exit
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 3000 is a good trade‑off between quality and speed.
    n = 3000

    # ------------------------------------------------------------------
    # Build a cheap pool of analytically motivated initial vectors.
    candidates = []

    # power‑law family (various exponents & offsets)
    exponents = [0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60]
    offsets   = [0.0, 0.1, 0.5, 1.0, 2.0]
    for p in exponents:
        for off in offsets:
            candidates.append(build_power_law(n, p, off))

    # exponential family (different bases)
    bases = [0.98, 0.95, 0.92, 0.90]
    for b in bases:
        candidates.append(build_exponential(n, base=b))

    # linear descending and uniform baseline
    candidates.append(build_linear_descending(n))
    candidates.append(np.full(n, 1.0 / n))

    # previous best if it exists
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                seq = np.zeros(n, dtype=np.float64)
                seq[:prev.size] = prev
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0.0:
                seq /= seq.sum()
                candidates.append(seq)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Choose the best pool member according to the fast score.
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best pool value: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optimisation parameters
    a = best_seq.copy()                     # current iterate on the simplex
    best_seq = a.copy()
    best_val = fast_score(a)

    # Adam hyper‑parameters
    lr = 0.8                                 # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)                    # first moment
    v = np.zeros_like(a)                    # second moment
    t = 0                                    # iteration counter

    # Active‑set parameters
    K = 12                                   # number of top convolution entries to use
    random_move_prob = 0.025                # probability of a stochastic mass‑transfer
    random_move_frac = 0.20                 # fraction of source mass to move
    uphill_accept = 0.001                   # occasional uphill acceptance
    last_improve = time.time()

    # Pre‑compute FFT size (power of two) for the chosen n.
    conv_size = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # (1) self‑convolution via FFT
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]

        # (2) locate the K largest entries – these are the active constraints
        # (np.argpartition is O(n) and cheap)
        top_idxs = np.argpartition(-conv, K)[:K]

        # (3) sub‑gradient of the max‑operator
        grad = gradient_multi_active(a, top_idxs)

        # --------------------------------------------------------------
        # (4) Adam update (projected onto the simplex)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps) * m_hat
        a_new = proj_simplex(a - step)

        # (5) evaluate the candidate
        cur_val = fast_score(a_new)

        # --------------------------------------------------------------
        # (6) acceptance rule
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
        else:
            if np.random.rand() < uphill_accept:
                a = a_new

        # --------------------------------------------------------------
        # (7) occasional random “mass‑transfer’’ move
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = proj_simplex(cand)          # stays on the simplex
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < uphill_accept:
                    a = cand

        # --------------------------------------------------------------
        # (8) stagnation handling – re‑seed from the current best with a
        #     small random perturbation if no improvement for a while.
        if time.time() - last_improve > 30.0:
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(0.85 * best_seq + 0.15 * noise)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

        # --------------------------------------------------------------
        # (9) simple progress report
        if t % 4000 == 0:
            print(f"[iter {t:7d}] cur {cur_val:.6f}  best {best_val:.6f}")

    # ------------------------------------------------------------------
    print(f"[final] best value after {t} iterations: {best_val:.6f}")
    # External checker normalises automatically; we return the raw list.
    return best_seq.tolist()
