    """
    Return the first `n` coefficients of (1‑z)^‑½,
    i.e. a_i = C(2i,i) / 4**i, normalised so that they sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence a[i] = a[i-1] * (2i-1) / (2i)
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _score_from_a(a: np.ndarray, n: int, Nfft: int) -> float:
    """Compute the raw objective `2·n·max(conv(a,a)) / (sum(a)**2`."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    max_conv = conv.max()
    s = a.sum()
    # avoid division‑by‑zero – the optimizer never lets s become tiny,
    # but keep the guard for safety.
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Core optimisation routine

def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  A few seconds are kept as safety margin.
    **kwargs : dict
        Ignored – present only to match the required signature.

    Returns
    -------
    List[float]
        The best sequence found (plain Python list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimension – a larger `n` yields a lower theoretical bound.
    n = 60_000                      # feel free to adjust; 60 k fits easily in RAM
    L = 2 * n - 1                   # length of linear auto‑convolution
    m = n + L - 1                   # length needed for the correlation step
    Nfft = next_pow_two(m)          # power‑of‑two FFT size (fast)

    # ------------------------------------------------------------------
    # Initial point: half‑binomial series (already normalised to sum = 1)
    a = half_binomial_series(n)

    # Keep the best solution encountered
    best_a = a.copy()
    best_score = _score_from_a(a, n, Nfft)

    print(f"[init] n={n}, initial raw score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    beta = 0.5                      # start of soft‑max smoothing
    beta_target = 2500.0            # final β – the larger the tighter the max
    beta_mul = 1.003                # geometric increase per iteration

    eta = 0.18                      # step size for gradient descent
    eta_min = 4e-5
    eta_decay = 0.9996

    momentum = 0.85                 # exponential moving average of the update
    v = np.zeros_like(a)            # momentum buffer

    max_phase = 1                   # 1 → soft‑max, 2 → pure max‑subgradient
    last_improve = 0                # iteration index of the last improvement
    restart_gap = 300_000           # restart after this many iterations without progress
    noise_prob = 0.008
    noise_scale = 3e-4

    # timing
    deadline = time.time() + budget_s - 5.0   # leave a few seconds safety margin
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) current auto‑convolution and objective value
        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]                # linear part only
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() * a.sum())

        # --------------------------------------------------------------
        # 2) record best solution
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:09d}] new best raw score={best_score:.6f}")

        # --------------------------------------------------------------
        # 3) gradient of the (smoothed) max‑convolution
        if max_phase == 1:
            # soft‑max (log‑sum‑exp) smoothing of the max
            shift = max_conv                     # for numerical stability
            w_exp = np.exp(beta * (conv - shift))
            w = w_exp / w_exp.sum()              # weights sum to 1 (shape L)

            # correlation a ⋆ w = 2 · corr(a, w)
            Wr = np.fft.rfft(w[::-1], n=Nfft)    # FFT of reversed weights
            corr_full = np.fft.irfft(A * Wr, n=Nfft)   # length Nfft
            start = L - 1                        # index where linear correlation starts
            grad = 2.0 * corr_full[start:start + n]

            # increase temperature
            beta = min(beta_target, beta * beta_mul)
            if beta >= beta_target:
                max_phase = 2                    # switch to pure max‐subgradient
        else:
            # pure max‑subgradient (single active convolution index)
            k = int(conv.argmax())
            grad = np.zeros_like(a)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] = 2.0 * a[k - idx]

        # --------------------------------------------------------------
        # 4) occasional random perturbation (helps escaping plateaus)
        if rng.random() < noise_prob:
            a += (rng.random(n) - 0.5) * noise_scale
            a = project_onto_simplex(a)

        # --------------------------------------------------------------
        # 5) momentum + gradient step + projection onto simplex
        v = momentum * v - eta * grad
        a_trial = a + v                         # note: we move *against* grad
        # protect against stepping beyond the non‑negative orthant
        a_trial = np.maximum(a_trial, 0.0)
        a = project_onto_simplex(a_trial)

        # --------------------------------------------------------------
        # 6) step‑size schedule
        eta = max(eta * eta_decay, eta_min)

        # --------------------------------------------------------------
        # 7) restart iff stuck for a long time
        if it - last_improve > restart_gap:
            a = rng.dirichlet(np.full(n, 0.6))
            a = project_onto_simplex(a)
            v.fill(0.0)
            last_improve = it
            # compute quick estimate for logging
            est = _score_from_a(a, n, Nfft)
            print(f"[iter {it:09d}] restart triggered – new random start, est raw score≈{est:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Final evaluation with the official routine (adds the clamping etc.)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
