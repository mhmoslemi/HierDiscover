"""
Propose a step‑function (non‑negative list) that minimises `evaluate_sequence`.

Algorithm
---------

The evaluation  E(a) = 2·n·max(a*a)  is scale‑invariant, so we work on the
probability simplex (a_i ≥ 0, Σ a_i = 1).  Directly minimising the
maximum is non‑convex and the sub‑gradient used in earlier runs often
gets stuck on a tiny active set.  To obtain a more informative descent
direction we replace the hard maximum by a smooth log‑sum‑exp
approximation:

    φβ(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

When β → ∞   φβ → max.  For moderate β the function is differentiable,
its gradient is a weighted sum of all convolution entries where the
weights are the soft‑max of the (scaled) convolution.  This gives a
global view of the landscape and avoids the plateaus that hurt a pure
sub‑gradient method.

The optimiser consists of three phases inside the 1000 s budget:

* **Phase 0 – initialisation** – start from a random point,
  from the previous run (`height_sequence_1`) or from a simple
  symmetric “tent’’ shape.
* **Phase 1 – smoothed projected gradient descent** – use the
  gradient of φβ with a schedule β = 0.5, 2, 8, 32, 128.  For each β we
  run many Nesterov‑style updates with a diminishing step size
  `η = η0 / √t`.  The projection onto the simplex is performed with
  the O(n) algorithm of Duchi et al.
* **Phase 2 – final refinement** – a cheap simulated‑annealing walk
  (mass‑transfer moves) tightens the solution and guarantees that we
  never end up with a worse value.

All convolutions are performed by FFT‑based multiplication
(`np.fft.fft`).  The FFT length is the next power of two ≥ 3·n – 2, which
covers both the self‑convolution (length 2n‑1) and the correlation needed
for the gradient (length 3n‑2).

The code respects the time budget, prints progress messages and finally
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1} (O(n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, length fixed once per run)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return a power‑of‑two length that can accommodate all needed convolutions."""
    # Need at least 3·n‑2 points for a⋆w (correlation) and 2·n‑1 for a*a.
    needed = 3 * n - 2
    return 1 << ((needed - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT of given length."""
    Fa = np.fft.fft(a, n=fft_len)
    conv = np.fft.ifft(Fa * Fa).real
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _gradient_phi(a: np.ndarray, conv: np.ndarray,
                  beta: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth approximation φβ(a) = (1/β) log Σ exp(β·conv).

    Returns ∇φβ (without the outer factor 2·n, which is added later).
    """
    # soft‑max weights (stable)
    m = conv.max()
    exp_vals = np.exp(beta * (conv - m))
    w = exp_vals / exp_vals.sum()               # shape (2·n‑1,)

    n = a.shape[0]
    # Pad w to FFT length
    w_pad = np.zeros(fft_len, dtype=np.float64)
    w_pad[: 2 * n - 1] = w

    # FFT of a (already computed outside is faster, but we keep it simple)
    Fa = np.fft.fft(a, n=fft_len)
    Fw = np.fft.fft(w_pad, n=fft_len)

    # Cross‑correlation a ⋆ w  =  ifft(Fa * conj(Fw))
    corr = np.fft.ifft(Fa * np.conj(Fw)).real

    # The needed slice corresponds to lags 0 … n‑1.
    start = (2 * n - 2)                # = len(w)-1
    grad = 2.0 * corr[start : start + n]   # factor 2 from ∂(a*a)/∂a
    return grad


# ----------------------------------------------------------------------
# Smoothed projected gradient descent (multiple β‑stages)
# ----------------------------------------------------------------------
def _smoothed_pgd(a0: np.ndarray, n: int,
                  deadline: float, start_time: float,
                  beta_schedule, step0: float = 0.8,
                  momentum_coeff: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Run projected gradient descent on φβ with a schedule of β values.
    Returns the best point discovered and its (exact) evaluate_sequence score.
    """
    fft_len = _fft_len_for_n(n)
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it_total = 0

    for beta in beta_schedule:
        # Reset iteration counter for this β‑stage
        it_stage = 0
        while (time.time() < deadline) and (it_stage < 100_000):
            it_stage += 1
            it_total += 1

            # self‑convolution
            conv = _conv_self_fft(a, fft_len)

            # exact score (no smoothing)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] improve (β={beta:.2f}, iter={it_total}), "
                      f"score={cur_score:.6f}")

            # gradient of the smoothed objective
            grad = _gradient_phi(a, conv, beta, fft_len)

            # Nesterov‑style momentum
            momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * grad

            # diminishing step size
            step = step0 / math.sqrt(it_total)

            a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny ε‑moves)
# ----------------------------------------------------------------------
def _annealing_refine(a0: np.ndarray, n: int,
                      deadline: float, start_time: float,
                      cur_best_score: float,
                      start_temp: float = 5e-2,
                      end_temp: float = 1e-5,
                      max_steps: int = 120_000) -> tuple[np.ndarray, float]:
    """Local stochastic search that moves a small ε from one index to another."""
    a = a0.copy()
    fft_len = _fft_len_for_n(n)

    # current exact score
    conv = _conv_self_fft(a, fft_len)
    cur_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of the source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop = _conv_self_fft(a_prop, fft_len)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve, score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple symmetric “tent’’ distribution (used as a deterministic starter)
# ----------------------------------------------------------------------
def _tent_shape(n: int) -> np.ndarray:
    """A symmetric triangular shape that sums to 1."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 - np.abs(x - centre) / centre
    vals[vals < 0] = 0.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the given time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # small safety margin

    # ------------------------------------------------------------------
    # Choose a working length (larger lengths give no asymptotic advantage
    # but may allow finer shaping).  2400 works well and matches the
    # previous best run.
    # ------------------------------------------------------------------
    n = 2400

    # ------------------------------------------------------------------
    # Build an initial point: prefer the previous run if available,
    # otherwise fall back to a symmetric tent, finally random.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = starter / starter.sum()
            if a0.shape[0] != n:
                # Upsample / downsample to required length
                a0 = np.interp(
                    np.linspace(0.0, 1.0, n),
                    np.linspace(0.0, 1.0, starter.shape[0]),
                    starter
                )
                a0 = a0 / a0.sum()
        else:
            raise ValueError
    except Exception:
        # deterministic tent shape gives a solid first guess
        a0 = _tent_shape(n)

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 0 – optional short random perturbation (helps to escape
    # pathological symmetry when the starter is already close to a local
    # minimum).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rnd = np.random.rand(n)
        rnd /= rnd.sum()
        a0 = 0.8 * a0 + 0.2 * rnd          # tiny random blend
        a0 = _project_simplex(a0)

    # ------------------------------------------------------------------
    # Phase 1 – smoothed projected gradient descent with a β‑schedule.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # β values grow exponentially; the last value (≈128) is already
        # a very tight approximation of the true max.
        beta_schedule = [0.5, 2.0, 8.0, 32.0, 128.0]

        a_opt, score_opt = _smoothed_pgd(
            a0, n, deadline, start_time,
            beta_schedule,
            step0=0.8,
            momentum_coeff=0.85
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase‑1 best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – local simulated‑annealing refinement (uses the remaining time).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _annealing_refine(
            best_seq, n,
            deadline, start_time,
            best_score,
            start_temp=5e-2,
            end_temp=1e-5,
            max_steps=150_000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final score after annealing {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(100, 1.0 / 100.0)

    # Final normalisation (tiny drift may appear because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
