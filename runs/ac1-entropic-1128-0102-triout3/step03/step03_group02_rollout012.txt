def _lp_solve(f, M):
    n = f.shape[0]
    first_col = np.r_[f, np.zeros(n-1)]
    first_row = np.r_[f[0], np.zeros(n-1)]
    A = toeplitz(first_col, first_row)  # shape (2n-1, n)
    c = -np.ones(n)  # maximize sum(g)
    b = np.full(2 * n - 1, M)
    # SciPy's linprog expects A_ub and b_ub. Provide bounds.
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    return res
```

def _mix(f, g, target_sum, t):
    # scale g to target_sum
    sum_g = g.sum()
    if sum_g <= 0:
        return f
    g = g * (target_sum / sum_g)
    new_f = (1 - t) * f + t * g
    # maintain nonneg
    new_f = np.maximum(new_f, 0)
    # renormalise to be exactly target_sum
    sum_new = new_f.sum()
    if sum_new > 0:
        new_f = new_f * (target_sum / sum_new)
    return new_f

"""
Linear‑programming + simple mixing refinement.

The evaluation function is
    score = 2·n·max(conv(a,a)) / (sum(a)^2)

and is scale‑invariant.  We therefore fix the ℓ₁‑norm to
    S = √(2 n)
so that the score equals simply  max(conv(a,a)).
For a given sequence f with max‑convolution M we solve the LP

    maximise   Σ g_i
    subject to  (f * g)_k ≤ M   for all k = 0,…,2n‑2
                g_i ≥ 0

(the convolution is linear in g).  The optimal g has a larger sum
than f while respecting the same peak M.  After scaling g back to the
target norm S we mix it with f:
    f ← (1‑t) f + t g
with a decreasing step size t.  Repeating this step strictly reduces
M and quickly drives the sequence towards a very flat auto‑convolution.
The routine tries a few lengths (up to 1800) and several diverse
initial shapes (uniform, triangular, Hann, random Dirichlet, and the
previous best construction if available).  The best sequence found
within the time budget is returned as a plain list of floats.

If SciPy is not available the algorithm falls back to the
soft‑max projected gradient descent that was used previously.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (used only in the fallback)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fallback optimiser (the method shown in the statement)
def _score_of_seq(a: np.ndarray) -> float:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _refine_sequence(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.6,
    eps_start: float = 0.05,
    eps_end: float = 0.001,
    beta: float = 0.9,
) -> tuple[np.ndarray, np.ndarray, float]:
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        eps_rel = eps_start + (eps_end - eps_start) * (it / max_iters)
        tau = max(eps_rel * conv.max(), 1e-14)
        w = np.exp((conv - conv.max()) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.shape[0])

        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        g = 2.0 * conv2[n - 1 : n - 1 + n]

        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)

    return a, best_a, best_score

# Linear‑programming refinement (main method)
try:
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz
    _has_scipy = True
except Exception:               # pragma: no cover
    _has_scipy = False


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the full auto‑convolution of a."""
    return np.convolve(a, a).max()


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    One LP step: given a current shape f (sum = S) and its peak M,
    solve maximise Σ g_i  subject to (f * g)_k ≤ M, g ≥ 0.
    Returns the optimal g (still with sum = S after scaling inside the caller).
    """
    n = f.shape[0]

    # build the convolution matrix A (size (2n‑1) × n)
    first_col = np.r_[f, np.zeros(n - 1)]
    first_row = np.r_[f[0], np.zeros(n - 1)]
    A = toeplitz(first_col, first_row)

    c = -np.ones(n)                     # maximise Σg_i  → minimise -Σg_i
    b = np.full(2 * n - 1, M)

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        # In the very unlikely event of a failure fall back to the trivial solution.
        return np.copy(f)
    return res.x


def _lp_improve(init_a: np.ndarray,
                max_iters: int,
                deadline: float,
                t0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Repeated LP+mixing steps on a fixed length.
    Returns the best sequence seen and its maximal convolution value.
    """
    n = init_a.shape[0]
    S = math.sqrt(2 * n)                 # target ℓ₁‑norm
    a = init_a.copy()
    best_a = a.copy()
    best_M = _conv_max(a)

    t = t0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        M = _conv_max(a)
        if M < best_M:
            best_M = M
            best_a = a.copy()

        # LP to get a direction with larger sum but same bound M
        g = _lp_step(a, M)

        # scale g back to the target norm S
        sum_g = g.sum()
        if sum_g > 0:
            g = g * (S / sum_g)

        # convex combination
        a = (1.0 - t) * a + t * g

        # tiny renormalisation (numerical safety)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (S / s)

        # decay the mixing weight
        t = max(t * 0.94, 0.01)

    final_M = _conv_max(a)
    if final_M < best_M:
        best_M = final_M
        best_a = a.copy()
    return best_a, best_M

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function height vector.
    Tries a small set of lengths (up to 1800) and several diverse
    initial shapes.  For each shape a cheap LP‑based refinement is
    performed; the best sequence encountered is returned.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to safely fetch the historic construction if it exists
    def _fetch_previous():
        try:
            arr = np.array(height_sequence_1, dtype=float)
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 0:
                return arr / s   # put it on the probability simplex for convenience
        except Exception:
            pass
        return None

    prev_simplex = _fetch_previous()   # on the simplex (sum = 1)

    # ------------------------------------------------------------------
    # Lengths we will try (the previous best was at 1800)
    lengths = [600, 900, 1200, 1500, 1800]

    for n in lengths:
        if time.time() >= deadline:
            break

        S = math.sqrt(2 * n)                # target ℓ₁‑norm
        # ----- build a pool of initial candidates (all summing to S) -----
        candidates = []

        # uniform
        candidates.append(np.full(n, S / n))

        # triangular (tent) shape
        idx = np.arange(n)
        tri = np.minimum(idx + 1, n - idx)
        tri = tri / tri.sum() * S
        candidates.append(tri)

        # Hann window
        hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
        hann = hann / hann.sum() * S
        candidates.append(hann)

        # random Dirichlet
        candidates.append(np.random.dirichlet(np.ones(n)) * S)

        # previous construction, resampled if we have one
        if prev_simplex is not None:
            old_n = prev_simplex.shape[0]
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, prev_simplex)
            interp = np.maximum(interp, 0.0)
            s = interp.sum()
            if s > 0:
                interp = interp * (S / s)
            else:
                interp = np.full(n, S / n)
            candidates.append(interp)

        # ------------------------------------------------------------------
        # Run a cheap LP‑based refinement on each candidate
        for cand_id, cand in enumerate(candidates, 1):
            if time.time() >= deadline:
                break

            # number of LP iterations – a modest budget that still fits the time limit
            max_iters = 30 if n <= 900 else 20
            print(f"[{time.time() - start_time:6.2f}s] n={n}, candidate {cand_id}/{len(candidates)}, LP iters={max_iters}")

            refined, M = _lp_improve(cand,
                                     max_iters=max_iters,
                                     deadline=deadline,
                                     t0=0.6)

            # The actual evaluate_sequence score equals M (because sum = √(2n))
            if M < best_score:
                best_score = M
                best_seq = refined.copy()
                print(f"   >>> New global best: n={n}, score={M:.6f}")

    # ------------------------------------------------------------------
    # If we still have no sequence (unlikely) fall back to a simple uniform vector
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n)

    # Make sure we return plain Python floats
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total == 0:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    # do NOT renormalise to sum‑1 – keep the norm √(2 n) that was used during optimisation
    print(f"[{time.time() - start_time:6.2f}s] Finished. Best score (≈max‑conv) = {best_score:.6f}, length = {len(best_seq)}")
    return [float(x) for x in best_seq]
