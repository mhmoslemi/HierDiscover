def gradient_on_support(a, support):
    # a length n, but only support non-zero.
    # compute conv via FFT (need full a)
    conv = FFT convolution.
    k_max = np.argmax(conv)
    # compute gradient as earlier (full length) but set entries not in support to zero after projection.
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k_max - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    # gradient step
    a_new = a - eta * grad
    # projection onto simplex restricted to support
    # set a_new[outside] = 0
    a_new[~support] = 0
    # project onto simplex of length k: we can use same algorithm but on support subvector
    supp_idxs = np.where(support)[0]
    sub = a_new[supp_idxs]
    # project onto simplex:
    u = np.sort(sub)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(sub)+1)
    cond = u - cssv/ind > 0
    rho = ind[cond][-1] if np.any(cond) else 0
    theta = cssv[cond][-1] / rho if rho>0 else 0
    sub_proj = np.maximum(sub - theta, 0.0)
    a_proj = np.zeros_like(a)
    a_proj[supp_idxs] = sub_proj
    return a_proj
```

def subgradient_optimize_on_support(a_init, mask, time_limit):
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.5
    eta_decay = 1e-6
    random_move_prob = 0.02
    random_move_step = 0.3
    it = 0
    deadline = time.time() + time_limit - 0.1
    support_idx = np.where(mask)[0]
    k = support_idx.size
    while time.time() < deadline:
        # compute convolution
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))
        # gradient
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        grad[valid] = 2.0 * a[j[valid]]
        # enforce gradient zero outside support (optional)
        grad[~mask] = 0.0

        # step
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # projection onto simplex restricted to support
        # set elsewhere to zero
        v[~mask] = 0.0
        # project subvector
        sub = v[support_idx]
        # simplex projection
        u = np.sort(sub)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, k + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        sub_proj = np.maximum(sub - theta, 0.0)
        a[:] = 0.0
        a[support_idx] = sub_proj

        # random mass transfer move
        if np.random.rand() < random_move_prob:
            i_idx, j_idx = np.random.choice(support_idx, size=2, replace=False)
            if a[i_idx] > 0:
                delta = random_move_step * a[i_idx] * np.random.rand()
                cand = a.copy()
                cand[i_idx] -= delta
                cand[j_idx] += delta
                # projection again
                cand[~mask] = 0.0
                sub = cand[support_idx]
                u = np.sort(sub)[::-1]
                cssv = np.cumsum(u) - 1.0
                ind = np.arange(1, k + 1)
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand[support_idx] = np.maximum(sub - theta, 0.0)

                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_a = cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 5000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad_on_support iter {it}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad_on_support final] best value {best_val:.6f}")
    return best_a
```

"""
Propose a low‑score step‑function by a two‑phase strategy.

Phase 1 – Random sparse support search
    * Choose a support size k (starting from ~150 downwards).
    * Sample many random subsets S ⊂ {0,…,n‑1} of size k.
    * For each S compute the ordered‑pair count via a cheap
      O(k²) outer‐sum + bincount.  The evaluation score for the
      uniform weight 1/k on S is
          score = 2·n·max_count / k² .
    * Keep the best uniform‑weight vector found.

Phase 2 – Refine the weights on the best support
    * Starting from the uniform vector on the best support,
      run a projected sub‑gradient descent that only moves mass
      inside the support (the other entries stay zero).
    * The gradient of the max–convolution is obtained analytically,
      a step is taken, and the vector is projected onto the simplex
      restricted to the support.
    * Occasional random mass‑transfer moves are tried to escape
      plateaus.  The best vector seen during the run is returned.

If the two phases still do not reach the target bound
(1.503) a fallback full‑vector sub‑gradient optimisation is performed.
All steps respect the security checks of `evaluate_sequence`.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (assumes a ≥ 0)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def project_simplex_on_support(v: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """Project v onto the simplex while keeping entries outside mask at 0."""
    idx = np.where(mask)[0]
    sub = v[idx]
    proj = project_simplex(sub)
    out = np.zeros_like(v)
    out[idx] = proj
    return out

# ----------------------------------------------------------------------
def max_count_from_support(S: np.ndarray, n: int) -> int:
    """Return the maximal ordered‑pair count for support S (size k)."""
    sums = np.add.outer(S, S)            # shape (k, k)
    cnt = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return int(cnt.max())

# ----------------------------------------------------------------------
def random_support_search(n: int,
                          k: int,
                          time_limit: float,
                          cur_best_score: float):
    """
    Randomly sample subsets of size k for at most time_limit seconds.
    Return the best uniform‑weight vector (as ndarray), its score, and the support.
    """
    best_seq = None
    best_score = cur_best_score
    best_S = None
    start = time.time()
    attempts = 0
    while time.time() - start < time_limit:
        attempts += 1
        S = np.random.choice(n, size=k, replace=False)
        max_c = max_count_from_support(S, n)
        score = 2.0 * n * max_c / (k * k)          # uniform weight 1/k
        if score < best_score:
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / k
            best_seq = a
            best_score = score
            best_S = S
            print(f"[random_support] k={k}, attempts={attempts}, max_c={max_c}, score={score:.6f}")
        # early exit if we already beat the target
        if best_score <= 1.503:
            break
    return best_seq, best_score, best_S

# ----------------------------------------------------------------------
def subgradient_optimize_on_support(a_init: np.ndarray,
                                    mask: np.ndarray,
                                    time_limit: float) -> np.ndarray:
    """
    Projected sub‑gradient descent that never leaves the support specified by mask.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.5
    eta_decay = 1e-6
    random_move_prob = 0.025
    random_move_step = 0.30

    supp_idx = np.where(mask)[0]
    k = supp_idx.size
    it = 0
    deadline = time.time() + time_limit - 0.1

    while time.time() < deadline:
        # convolution & active index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active quadratic term
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask_ij = (j >= 0) & (j < n)
        grad[mask_ij] = 2.0 * a[j[mask_ij]]
        grad[~mask] = 0.0                     # keep us inside the support

        # step and projection onto the simplex restricted to the support
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        v[~mask] = 0.0
        a = project_simplex_on_support(v, mask)

        # occasional random mass‑transfer move (helps escaping plateaus)
        if np.random.rand() < random_move_prob:
            i_idx, j_idx = np.random.choice(supp_idx, size=2, replace=False)
            if a[i_idx] > 0:
                delta = random_move_step * a[i_idx] * np.random.rand()
                cand = a.copy()
                cand[i_idx] -= delta
                cand[j_idx] += delta
                cand[~mask] = 0.0
                cand = project_simplex_on_support(cand, mask)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 5000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad_support iter {it}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad_support final] best value {best_val:.6f}")
    return best_a

# ----------------------------------------------------------------------
def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on the full vector (fallback).
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    eta0 = 0.30
    eta_decay = 1e-5
    random_move_prob = 0.02
    random_move_step = 0.15
    it = 0
    deadline = time.time() + time_limit - 0.2

    while time.time() < deadline:
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        a = project_simplex(v)

        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_step * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = project_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand

                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a

    if best_seq is not None:
