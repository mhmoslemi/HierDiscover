def _solve_lp(a, M):
    n = a.shape[0]
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(a, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # bounds for variables
    bounds = [(0, None)] * n
    # objective to maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # linear inequality A @ g <= M
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a non‑negative step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1. Initialise a probability vector (sum = 1).  If the global variable
   `height_sequence_1` is present we up‑sample it, otherwise a random
   vector is used.

2. Short projected‑gradient descent (sub‑gradient of the max‑convolution)
   to obtain a decent starting shape.

3. Main optimisation loop (until the time budget is exhausted):
   * Compute the current maximal self‑convolution value `M`.
   * Solve a linear programme  
       maximise   1ᵀ g
       s.t.       a * g ≤ M      (linear convolution constraints)
                  g ≥ 0
     The LP finds a direction `g` that spreads the mass where `a`
     is small while respecting the current convolution bound.
   * Normalise `g` to a probability vector and move from `a` towards `g`
     by back‑tracking line search – the step size is reduced until the
     maximal convolution strictly decreases.
   * Every few iterations a short projected‑gradient descent refines the
     current point.

4. A final simulated‑annealing walk makes small mass‑transfer moves
   (ε from one index to another) to escape any remaining local minima.

The whole procedure respects the 1000 s budget, prints progress, and
returns the best sequence found (as a plain Python list).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient from all active convolution entries
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Average sub‑gradient of the max‑convolution."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if act.size > 0:
        g /= float(act.size)
    return g


# ----------------------------------------------------------------------
# Projected (Nesterov‑style) gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_gradient_descent_full(a0: np.ndarray, n: int,
                              deadline: float, start_time: float,
                              beta: float = 0.9,
                              step0: float = 0.6,
                              max_iter: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Projected gradient descent using the exact sub‑gradient."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: n={n}, iter={it}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing local refinement
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer walk with a geometric temperature schedule."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Cache for linear‑programming matrix pattern (depends only on n)
# ----------------------------------------------------------------------
_lp_pattern_cache = {}


def _prepare_lp_pattern(n: int):
    """Return (rows, cols) for the convolution matrix A (size (2n‑1)×n)."""
    if n not in _lp_pattern_cache:
        rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
        cols = np.repeat(np.arange(n), n)
        _lp_pattern_cache[n] = (rows, cols)
    return _lp_pattern_cache[n]

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The method combines a short gradient descent,
    an LP‑based redistribution step, and a final annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Reserve a second for the harness itself.
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------

    starter = None

        except Exception:
            starter = None

    n = 2400                     # length of the candidate
    if starter is not None:
        a = _upsample(starter, n)
    else:
        a = np.random.rand(n)
        a /= a.sum()

    best_seq = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    print(f"[{time.time() - start_time:6.2f}s] initial score: {best_score:.6f}")

    cur_score = 2.0 * n * _conv_self(a).max()
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()

    # --------------------------------------------------------------
    # Stage 2 – LP‑based redistribution loop
    # --------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 5.0:
        iteration += 1
        conv = _conv_self(a)
        M = conv.max()

        # Build the constraint matrix A (sparse) for the current a.
        rows, cols = _prepare_lp_pattern(n)
        data = np.repeat(a, n)                     # a_i appears n times
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # Solve: maximise sum(g)  subject to A @ g ≤ M , g ≥ 0.
        lp_res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                         bounds=[(0, None)] * n, method='highs')
        if not lp_res.success:
            print(f"[{time.time() - start_time:6.2f}s] LP failed at iter {iteration}")
            break
        g = lp_res.x
        if g.sum() == 0:
            print(f"[{time.time() - start_time:6.2f}s] LP returned zero vector, stop.")
            break
        g /= g.sum()                               # normalise to sum = 1

        # Back‑tracking line search from a towards g.
        t = 1.0
        improved = False
        while t > 1e-5:
            a_candidate = (1.0 - t) * a + t * g
            M_candidate = _conv_self(a_candidate).max()
            if M_candidate < M - 1e-12:            # strict improvement
                a = a_candidate
                M = M_candidate
                improved = True
                break
            t *= 0.5

        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve it {iteration}: score={best_score:.6f}")

        # Occasionally sprinkle a few gradient‑descent steps to smooth the shape.
        if iteration % 10 == 0:
            sub_deadline = min(deadline, time.time() + 5.0)
            a, _ = _run_gradient_descent_full(a, n, sub_deadline, start_time,
                                              beta=0.9, step0=0.4, max_iter=300_000)

    # Stage 3 – final simulated‑annealing refinement (uses remaining time)
    # --------------------------------------------------------------

                                          best_score, best_seq,

                                          steps=200_000)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety normalisation and conversion to plain Python list
    # ------------------------------------------------------------------

    s = best_seq.sum()
    if s > 0:

    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    return [float(x) for x in best_seq]
