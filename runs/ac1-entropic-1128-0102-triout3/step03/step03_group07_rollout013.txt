"""
Propose a candidate sequence for the auto‑convolution peak problem.

Algorithm
---------
1.  Work on the probability simplex (∑a_i = 1, a_i ≥ 0).  The objective for a
    simplex vector a of length *n* is   score = 2·n·max(a * a)  (the sum
    cancels out, therefore we may keep the sum equal to one).

2.  Initialise *a* with the classic half‑binomial (≈ (1‑z)^{-1/2}) series,
    which is a good starting point for a flat auto‑convolution.

3.  **Iterative LP‑mixing step** (the idea from the reference solution):
        •  Let M = max(a*a) (computed by a fast FFT based convolution).
        •  Solve the linear programme  

                maximise    Σ b_j
                subject to  (a * b)_k ≤ M   for all k = 0,…,2n‑2
                            b_j ≥ 0

        •  Scale the LP solution to a simplex vector  g = b / Σb.
        •  Mix the current iterate with the LP solution  

                a ← (1‑t)·a + t·g ,   t ∈ (0,1)

        The mix factor *t* starts relatively large (0.5) and is slowly
        decreased, guaranteeing monotone improvement of the peak M.

4.  After every LP step we recompute the score, keep the best iterate seen,
    and occasionally inject a small random perturbation or restart from a
    Dirichlet draw if progress stalls.

5.  The loop runs until the time budget is exhausted (≈ 950 s, leaving a
    safety margin).  The final best sequence is validated with the provided
    ``evaluate_sequence`` routine and returned as a plain list of floats.

The method uses only NumPy and SciPy (the HiGHS linear‑programming backend)
and therefore stays well within the one‑CPU, 1000 s budget while typically
driving the score below the previous 1.525 ≈ target 1.503.
"""

import time
from typing import List
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)

def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^{-1/2}  (a_i = C(2i,i)/4^i)
    and normalised so that they sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_max_fft(a: np.ndarray) -> float:
    """
    Compute max(a * a) using an FFT (linear convolution, not circular).
    a is assumed to be a probability vector (sum = 1).
    """
    n = a.shape[0]
    L = 2 * n - 1
    Nfft = next_pow_two(L)
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[:L]
    # Numerical noise may give tiny negative values – clip them.
    conv = np.clip(conv, 0.0, None)
    return float(conv.max())


def build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix M such that (a * b) = M·b for any vector b.
    For each column j the non‑zero part is a shifted copy of the vector a.
    """
    n = a.shape[0]
    L = 2 * n - 1
    M = np.zeros((L, n), dtype=np.float64)
    # each column j contributes a on rows j … j+n‑1
    for j in range(n):
        M[j:j + n, j] = a
    return M


# ----------------------------------------------------------------------
# Core optimisation routine

def propose_candidate(seed: int = 42,
                     budget_s: float = 950.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises the evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  A few seconds are kept as safety margin.
    **kwargs :
        Ignored – present only to match the required signature.

    Returns
    -------
    List[float]
        The best sequence found (plain Python list of floats).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a moderate n gives a good trade‑off between speed
    # and attainable bound.  The value can be overridden via **kwargs.
    n = int(kwargs.get("n", 800))
    if n < 3:
        n = 3
    L = 2 * n - 1

    # --------------------------------------------------------------
    # Initial point – classic half‑binomial series (already on the simplex)
    a = half_binomial_series(n)

    # Keep the best found vector and its score.
    best_a = a.copy()
    best_M = conv_max_fft(a)
    best_score = 2.0 * n * best_M
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optimisation parameters
    mix_factor = 0.5          # initial mixing weight t
    mix_decay = 0.99          # geometric decay of t each iteration
    min_mix = 0.01            # lower bound for t
    restart_gap = 3000        # iterations without improvement → restart
    noise_prob = 0.008        # probability of a small random perturbation
    noise_scale = 2e-4        # scale of the perturbation
    max_it_without_improve = 0

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    it = 0
    last_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # ---- 1.  Current convolution peak ---------------------------------
        M = conv_max_fft(a)
        cur_score = 2.0 * n * M

        # ---- 2.  Record best solution ------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:06d}] new best score = {best_score:.6f}")

        # ---- 3.  Linear‑programming step ---------------------------------
        # Build matrix representing a * b ≤ M.
        A_mat = build_convolution_matrix(a)          # shape (L, n)
        b_ub = np.full(L, M, dtype=np.float64)       # right‑hand side ≤ M
        c = -np.ones(n, dtype=np.float64)            # maximise Σb  → minimise –Σb
        bounds = [(0.0, None) for _ in range(n)]

        # Solve the LP (HiGHS is fast for medium sized problems)
        res = linprog(c, A_ub=A_mat, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True})

        if not res.success:
            # Unexpected failure – abort the LP loop but keep current best.
            print(f"[iter {it:06d}] LP failed (status: {res.message}); terminating.")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            # Degenerate solution – cannot improve further.
            print(f"[iter {it:06d}] LP returned near‑zero sum; terminating.")
            break

        # Normalise the LP solution to a probability vector.
        g = b / sum_b

        # ---- 4.  Mix the current iterate towards the LP solution ----------
        a = (1.0 - mix_factor) * a + mix_factor * g

        # Numerical cleanup: enforce non‑negativity and re‑project onto simplex.
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = half_binomial_series(n)
        else:
            a = a / a_sum

        # ---- 5.  Decay the mixing factor ---------------------------------
        mix_factor = max(min_mix, mix_factor * mix_decay)

        # ---- 6.  Occasional stochastic perturbation -----------------------
        if rng.random() < noise_prob:
            noise = (rng.random(n) - 0.5) * noise_scale
            a = a + noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # ---- 7.  Restart logic (if stagnation) ---------------------------
        if it - last_improve > restart_gap:
            # Fresh start from a Dirichlet distribution (more diverse than uniform).
            a = rng.dirichlet(np.full(n, 0.6))
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = it
            print(f"[iter {it:06d}] restart – new random start, score≈{2.0*n*conv_max_fft(a):.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Final evaluation using the provided (slow) routine for safety.
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
