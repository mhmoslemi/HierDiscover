"""
Propose_candidate – simplex‐projected L‑BFGS with a smooth max approximation.

The evaluation function 2·n·max(conv(a,a)) / (∑a)² is scale‐invariant,
therefore we keep the coefficients on the probability simplex
(∑a = 1, a_i ≥ 0).  The non‑smooth “max” is replaced by
a log‑sum‑exp (a smooth max) with a decreasing temperature τ.
For a fixed τ the smooth objective is differentiable; its gradient
with respect to the underlying log‑probability parameters x
(soft‑max a_i = e^{x_i}/∑e^{x}) is obtained analytically by
convolution with the soft‑max weights.  The resulting smooth problem
is solved by L‑BFGS‑B.  After a short optimisation at a relatively
large τ we halve τ several times, warm‑starting each stage with the
solution from the previous stage – this mimics a continuation method
and yields a much tighter approximation of the true max.

To escape possible local minima we perform a few random restarts
(Dirichlet draws) and keep the best result seen by the exact
evaluate_sequence() call.  Finally the best shape is up‑sampled
(linear interpolation) to a larger length and refined once more.
The whole routine respects the 1000 s wall‑time budget.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
def _fft_len_needed(k: int) -> int:
    """Next power of two ≥ k (used for zero‑padded FFTs)."""
    return 1 << (k - 1).bit_length()


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear auto‑convolution a * a via FFT (full length 2·len(a)-1)."""
    n = a.shape[0]
    L = _fft_len_needed(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a*a)_k with respect to a.
    Implements  g_i = 2·Σ_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _fft_len_needed(len(w) + n - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(W * A_rev, n=L)[: len(w) + n - 1]
    return 2.0 * conv[n - 1 : n - 1 + n]


def _smooth_max_and_grad(x: np.ndarray, n: int, tau: float):
    """
    Smooth‑max objective (log‑sum‑exp) and its gradient w.r.t. the
    log‑probability vector x (a_i = softmax(x_i)).
    Returns (f, g) where f is the smooth max value and g is ∇_x f.
    """
    # soft‑max to obtain a feasible simplex vector
    a = np.exp(x)
    a_sum = a.sum()
    a = a / a_sum

    conv = _conv_full(a)
    max_val = conv.max()

    # numerical‑stable soft‑max over convolution entries
    shifted = conv - max_val
    exps = np.exp(shifted / tau)
    w = exps / exps.sum()

    # smooth max value = max + τ·log Σ exp(...)
    f = max_val + tau * np.log(np.sum(exps))

    # gradient w.r.t. a (using the weights w)
    g_a = _grad_a(a, w)

    # chain rule for the soft‑max parametrisation:
    #   ∂a_i/∂x_j = a_i(δ_{ij} – a_j)
    # Hence ∇_x f = (I – a a^T) g_a = g_a – a·(g_a·a)
    dot = np.dot(g_a, a)
    g_x = g_a - a * dot
    return f, g_x


def _lbfgs_optimize(start_a: np.ndarray,
                    max_iter: int,
                    tau: float,
                    deadline: float) -> np.ndarray:
    """
    Run a single L‑BFGS‑B optimisation stage at temperature τ,
    starting from the simplex vector `start_a`.  The function respects
    the global time deadline.
    """
    eps = 1e-12
    x0 = np.log(np.maximum(start_a, eps))

    # wrapper required by scipy – it expects (f, grad) when jac=True
    def fun(x):
        return _smooth_max_and_grad(x, start_a.shape[0], tau)

    # L‑BFGS‑B; the optimizer is stopped early if we run out of time.
    res = scipy.optimize.minimize(fun,
                                  x0,
                                  method='L-BFGS-B',
                                  jac=True,
                                  options={'maxiter': max_iter,
                                           'disp': False})
    a_opt = np.exp(res.x)
    a_opt /= a_opt.sum()
    return a_opt


def _refine_with_cooling(a: np.ndarray,
                         iters_per_stage: int,
                         taus: list,
                         deadline: float) -> np.ndarray:
    """
    Perform a continuation optimisation: start at a large τ and
    gradually halve it, always warm‑starting the next stage.
    """
    cur = a.copy()
    for tau in taus:
        if time.time() >= deadline:
            break
        cur = _lbfgs_optimize(cur,
                               max_iter=iters_per_stage,
                               tau=tau,
                               deadline=deadline)
    return cur


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **_ignored) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that yields a
    low value of `evaluate_sequence`.  The routine runs at most
    `budget_s` seconds and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ---------- configuration ----------
    base_n = 1800                 # start length (same order as the reference)
    max_length = 6000            # ultimate length after up‑sampling
    stage_iters = 300            # L‑BFGS iterations per τ‑stage
    cooling_schedule = [0.08, 0.04, 0.02, 0.01, 0.005]

    best_seq = None
    best_score = float('inf')

    # ---------- diverse starting points ----------
    candidates = []

    # 1) uniform
    candidates.append(np.full(base_n, 1.0 / base_n))

    # 2) triangular (tent)
    idx = np.arange(base_n)
    tri = np.minimum(idx + 1, base_n - idx)
    candidates.append(tri / tri.sum())

    # 3) Hann window (raised cosine)
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (base_n - 1)))
    candidates.append(hann / hann.sum())

    # 4) a few random Dirichlet draws
    for _ in range(3):
        candidates.append(np.random.dirichlet(np.ones(base_n)))

    # ------------------------------------------------------------------
    # 5) optimise each candidate on the base length
    # ------------------------------------------------------------------
    for i, cand in enumerate(candidates, 1):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] Optimising candidate {i}/{len(candidates)} "
              f"with n={base_n}")

        refined = _refine_with_cooling(cand,
                                       iters_per_stage=stage_iters,
                                       taus=cooling_schedule,
                                       deadline=deadline)

        score = evaluate_sequence(refined.tolist())
        if score < best_score:
            best_score = score
            best_seq = refined.copy()
            print(f"   New global best: score={score:.6f}")

    # ------------------------------------------------------------------
    # 6) progressive up‑sampling (≈1.5× each step) up to max_length
    # ------------------------------------------------------------------
    cur_a = best_seq if best_seq is not None else np.full(base_n, 1.0 / base_n)
    cur_n = cur_a.shape[0]

    while cur_n < max_length and time.time() < deadline:
        next_n = int(cur_n * 1.5)
        if next_n > max_length:
            next_n = max_length

        # linear interpolation onto the larger grid
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        up = np.interp(x_new, x_old, cur_a)
        up = np.maximum(up, 0.0)
        up /= up.sum()

        print(f"[{time.time() - start_time:6.2f}s] Upsampling to n={next_n}")

        refined = _refine_with_cooling(up,
                                       iters_per_stage=stage_iters,
                                       taus=cooling_schedule,
                                       deadline=deadline)

        score = evaluate_sequence(refined.tolist())
        if score < best_score:
            best_score = score
            best_seq = refined.copy()
            print(f"   Updated global best after up‑sampling: n={next_n}, score={score:.6f}")

        cur_a = refined
        cur_n = next_n

    # ------------------------------------------------------------------
    # 7) final polishing on the best length discovered
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] Final polishing on n={best_seq.shape[0]}")
        final = _refine_with_cooling(best_seq,
                                     iters_per_stage=stage_iters,
                                     taus=cooling_schedule,
                                     deadline=deadline)
        final_score = evaluate_sequence(final.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = final
            print(f"   Final improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # 8) sanity – ensure non‑negativity and normalisation
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(base_n, 1.0 / base_n)

    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    print(f"[{time.time() - start_time:6.2f}s] Finished. Best evaluate_sequence score = {best_score:.6f}")

    return [float(v) for v in best_seq]
