"""
Propose a step‑function that minimizes
    2·n·max(conv(a,a)) / (sum(a)^2)

The objective is scale‑invariant, so we keep the coefficients on the
probability simplex (∑a_i = 1).  The algorithm works in several stages:

1.  A hierarchy of lengths (512 → … → 8192) is used.  After each stage the
    best distribution is linearly interpolated (upsampled) to the next
    length, preserving the total mass.

2.  For each length we run an Adam optimiser on the smooth surrogate
    L_λ(a) = λ·log(∑exp(conv(a)/λ)).  The gradient of the surrogate is
    computed via FFT (soft‑max weights × reversed a).  λ starts as a
    fraction of the current maximum convolution value and is halved every
    few thousand iterations; the learning‑rate decays as 1/√t.  After each
    step the iterates are projected onto the simplex.

3.  The best iterate of the Adam run is refined by a cheap simulated‑
    annealing sweep that transfers a small ε of mass between two random
    coordinates, accepting worsening moves with a geometric temperature
    schedule.

4.  The procedure is repeated for the next, larger length.  The final
    distribution (as a Python list of floats) is returned.

The whole run respects the supplied time budget (default 1000 s) and
prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, ∑x=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real‑to‑real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors x and y via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, renormalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of L_λ(a) = λ·log(∑exp(conv/λ)).
    Parameters
    ----------
    a   : np.ndarray, shape (n,)
        Current probability vector (∑a=1).
    conv: np.ndarray, shape (2n‑1,)
        Self‑convolution a*a.
    lam : float
        Smoothing parameter λ>0.
    Returns
    -------
    grad: np.ndarray, shape (n,)
        Gradient ∇L_λ(a).
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights, numerically stable
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # cross‑correlation of weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)                     # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------


def _run_adam_optimize(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       max_iter: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective on the probability simplex.
    Returns the best vector seen and its exact score (=2·n·max(conv)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.5

    # λ‑schedule
    lam_factor = 0.25          # λ = lam_factor * current max
    lam_decay = 0.5
    lam_interval = 3000

    beta1_pow = beta1
    beta2_pow = beta2

    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, it={it}, score={cur_score:.6f}")

        # update smoothing parameter
        if it % lam_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-12)
        lam = max(max_b * lam_factor, 1e-12)

        # gradient of the surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate schedule
        lr = lr0 / math.sqrt(it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

        # update the running powers of beta
        beta1_pow *= beta1
        beta2_pow *= beta2

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 40_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing: move a small ε from i to j.
    Returns possibly improved sequence and its score.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score
