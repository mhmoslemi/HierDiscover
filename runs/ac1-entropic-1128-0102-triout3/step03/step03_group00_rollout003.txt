"""
Propose a step‑function (list of non‑negative heights) that minimizes
`evaluate_sequence`.  The algorithm is a multi‑restart Adam optimiser
augmented with a short simulated‑annealing polish.

**Main ideas**

1. Work on the probability simplex (∑a = 1).  On this manifold the
   evaluation score is simply `2·n·max(conv(a,a))`, i.e. we only have to
   lower the peak of the self‑convolution.

2. Run several *independent* Adam optimisations on a modest size
   (`n≈1000`).  Random restarts explore different basins of attraction.
   The best run is kept.

3. Upsample the best vector to a larger size (`n≈4000`) using linear
   interpolation and continue optimisation with a longer Adam run.

4. A cheap simulated‑annealing walk (tiny mass‑transfer moves) finishes
   the search while the time budget permits.

All optimisation steps respect the time limit (`budget_s` seconds) and
periodically print progress for debugging.  The final sequence is
renormalised to sum = 1 (scaling does not affect the score) and returned
as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (averaged over the active set)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray,
                      conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Return an averaged sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] ≥ max(conv)*(1‑rel_tol) are treated as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    act = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # each pair (i,j) contributes a_j to ∂/∂a_i and a_i to ∂/∂a_j
        g[i] += a[j]
        g[j] += a[i]

    if act.size:
        g /= float(act.size)
    return g

# ----------------------------------------------------------------------
# Adam optimiser on the simplex (minimise 2·n·max(conv))
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex.
    Returns the best vector seen and its score (2·n·max(conv)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        maxb = conv.max()
        cur_score = 2.0 * n * maxb

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it}, score={cur_score:.6f}")

        # sub‑gradient of the objective (2·n·max(conv))
        grad = _subgradient_full(a, conv) * (2.0 * n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

        # occasional tiny random jitter to keep the dynamics alive
        if it % 20000 == 0:
            a += np.random.randn(n) * 1e-4
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_score

# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealing.
    Returns a possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.08   # up to 8 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis criterion
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best list found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin for the harness

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage 1 – several short Adam runs on a modest size (n≈1000)
    # ------------------------------------------------------------------
    n_small = 1000
    n_restarts = 4
    slice_time = max(15.0, 0.12 * (deadline - time.time()))   # total for stage 1
    slice_deadline = time.time() + slice_time
    time_per_restart = slice_time / n_restarts

    for r in range(n_restarts):
        if time.time() >= slice_deadline:
            break

        if r == 0 and starter is not None:
            a0 = _upsample(starter, n_small)
        else:
            a0 = np.random.rand(n_small)
            a0 /= a0.sum()

        run_deadline = time.time() + time_per_restart
        a_opt, score_opt = _adam_optimize(
            a0, n_small, run_deadline, start_time,
            lr=0.07, beta1=0.9, beta2=0.999
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 1 restart {r+1} improve: score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale and long Adam optimisation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution
        best_seq = np.full(n_small, 1.0 / n_small)

    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_large = 4000
    a0_large = _upsample(best_seq, n_large)

    # Allocate most of the remaining budget to Adam, keep ~10 % for annealing
    remaining = deadline - time.time()
    anneal_time = max(12.0, 0.07 * remaining)     # a few seconds for the final polish
    grad_deadline = time.time() + remaining - anneal_time

    a_large, score_large = _adam_optimize(
        a0_large, n_large, grad_deadline, start_time,
        lr=0.05, beta1=0.9, beta2=0.999
    )
    if score_large < best_score:
        best_score = score_large
        best_seq = a_large.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score_large:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – short simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq, n_large, anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=4e-2, end_temp=5e-6, steps=120000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
