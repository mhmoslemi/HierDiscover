"""
propose_candidate
-----------------
Search for a low‑value step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the linear convolution.  The routine uses three phases:

1.  **Parametric sweep** – generate many analytically motivated families
    (power‑law, exponential, polynomial‑decay, mixtures) and keep the best
    candidate according to a fast FFT‑based score.

2.  **Refinement** – starting from the best candidate run a projected Adam
    optimiser on the simplex (∑a = 1, a≥0).  The gradient is an aggregated
    sub‑gradient of the currently active convolution entries (those within a
    decreasing relative gap of the maximum).  Random “mass‑transfer” moves
    are interleaved to escape plateaus, and a simple restart strategy is used
    if no improvement occurs for a while.

3.  **Return** – the best sequence found (as a plain list of floats).  The
    external checker will apply its own normalisation, so the raw values are
    acceptable.

All helper functions are top‑level; no closures or file I/O are used.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` (no safety checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum‑1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i  (0<r<1), normalised to sum‑1."""
    if r <= 0.0 or r >= 1.0:
        return np.full(n, 1.0 / n)
    a = np.power(r, np.arange(n, dtype=np.float64))
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_polynomial_decay(n: int, q: float = 1.0) -> np.ndarray:
    """a_i ∝ (n-i)^{q}, normalised to sum‑1."""
    a = np.arange(n, 0, -1, dtype=np.float64) ** q
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def gradient_multi_active(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a∗a)_k  aggregated over the indices in `top_idxs`.
    For each active index k we add 2·a_{k-i} to the i‑th component whenever
    the pair (i, k‑i) is inside the vector.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in top_idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation value.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 1. problem size – 3000 works well and matches the previous runs.
    n = 3000
    conv_size = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # 2. generate an initial pool from several parametric families.
    candidates = []

    # power‑law family (wide grid)
    p_vals = np.linspace(0.20, 0.90, 15)
    offset_vals = np.array([0.0, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0])
    for p in p_vals:
        for off in offset_vals:
            candidates.append(build_power_law(n, p, off))

    # exponential family
    r_vals = np.concatenate([
        np.linspace(0.90, 0.99, 20, endpoint=False),
        np.linspace(0.990, 0.9999, 20)
    ])
    for r in r_vals:
        candidates.append(build_exponential(n, r))

    # polynomial‑decay (linear, quadratic, etc.)
    q_vals = np.linspace(0.2, 3.0, 12)
    for q in q_vals:
        candidates.append(build_polynomial_decay(n, q))

    # mixture: power‑law on a prefix + uniform low‑tail
    for p in [0.35, 0.45, 0.55]:
        for tail_frac in [0.05, 0.10, 0.20]:
            a_main = build_power_law(n, p, offset=0.0)
            tail_len = max(1, int(n * tail_frac))
            tail = np.full(tail_len, 1.0)
            a_mix = np.concatenate([a_main[: n - tail_len], tail])
            a_mix /= a_mix.sum()
            candidates.append(a_mix)

    # previous best if it exists
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                seq = np.zeros(n, dtype=np.float64)
                seq[: prev.size] = prev
            seq = np.maximum(seq, 0.0)
            total = seq.sum()
            if total > 0.0:
                seq /= total
                candidates.append(seq)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. pick the best among the coarse pool.
    best_seq = None
    best_val = np.inf
    for a in candidates:
        val = fast_score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    print(f"[init] best coarse value: {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. refined optimisation (projected Adam + occasional random moves).
    a = best_seq.copy()                 # current point (on simplex)
    best_seq = a.copy()
    best_val = fast_score(a)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule parameters
    eta0 = 0.6                # base learning rate
    max_active = 15          # maximum number of active indices
    delta_start = 0.15
    delta_end = 0.01
    rand_move_prob = 0.04
    rand_move_frac = 0.18
    uphill_prob = 0.001

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ---- convolution and active set ---------------------------------
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_conv = conv.max()

        # relative gap that shrinks linearly
        delta = max(delta_end,
                    delta_start * (1.0 - it / 250000.0))
        mask = conv >= max_conv * (1.0 - delta)
        active = np.flatnonzero(mask)
        if active.size > max_active:
            # keep the most violating ones
            top_sel = np.argpartition(-conv[active], max_active)[:max_active]
            active = active[top_sel]

        # ---- gradient (aggregated over the active set) -----------------
        grad = gradient_multi_active(a, active)

        # ---- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a - step)

        cur_val = fast_score(a_new)

        # ---- accept improvements ----------------------------------------
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
        else:
            # occasional uphill move (helps escaping plateaus)
            if np.random.rand() < uphill_prob:
                a = a_new

        # ---- random mass‑transfer move -----------------------------------
        if np.random.rand() < rand_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta_transfer = rand_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta_transfer
                cand[j] += delta_transfer
                cand = proj_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < uphill_prob:
                    a = cand

        # ---- logging ----------------------------------------------------
        if it % 5000 == 0:
            print(f"[iter {it:7d}] cur {cur_val:.6f}  best {best_val:.6f}")

        # ---- stagnation handling (restart) -------------------------------
        if time.time() - last_improve > 20.0:
            # perturb the best solution a little and re‑project
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(0.85 * best_seq + 0.15 * noise)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # Return plain Python list; the external checker will re‑normalise if needed.
    return best_seq.tolist()
