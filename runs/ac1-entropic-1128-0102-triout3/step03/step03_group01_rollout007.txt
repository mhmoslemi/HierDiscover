"""
Propose an improved step‑function using a two‑phase simplex Adam optimizer.

The algorithm works on the probability simplex (∑a_i = 1).  First we
smooth the ℓ∞‑objective (max self‑convolution) with a log‑sum‑exp
surrogate and run an Adam‑style optimiser (phase 1).  When the
smoothing becomes negligible we switch to the exact sub‑gradient of
the max (phase 2).  After each update the vector is projected onto the
simplex (Duchi et al. 2008).  The best solution found is refined by a
cheap simulated‑annealing sweep.  The procedure is applied repeatedly
to increasing problem sizes (up‑sampling the current best solution),
allowing the time budget (≈1000 s) to be spent where it is most
effective.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using real‑to‑real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv).
    """
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)               # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _subgrad_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution (average over all active
    indices).  This is used when the smoothing parameter becomes tiny.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat all entries within relative tolerance 1e-12 as active
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser with two phases (smooth → exact)
# ----------------------------------------------------------------------


def _run_combined_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          max_iter: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex that first uses the smooth surrogate,
    then switches to the exact max‑sub‑gradient.
    Returns best vector and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.5                      # base learning rate
    lam_factor = 0.3                 # initial smoothing factor
    lam_decay = 0.8
    lam_decay_interval = 2000       # halve smoothing every 2k iters

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] improve: n={n}, it={it}, score={cur_score:.6f}")

        # adapt smoothing factor
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)

        lam = max(max_b * lam_factor, 1e-12)

        # choose gradient: smoothed while lam is not tiny, otherwise exact
        if lam > 1e-6:
            grad = _gradient_from_conv(a, conv, lam)
        else:
            grad = _subgrad_max(a, conv)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)   # diminishing step size
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project onto the simplex
        a = _project_simplex(a)

        # occasional random jitter to escape shallow plateaus
        if it % 5000 == 0:
            jitter = np.random.rand(n) * 0.001
            a = a + jitter
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (same as before)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing: move a small ε from i to j.
    Returns possibly improved sequence and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Finds a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The search proceeds on a schedule of growing
    lengths, applying a combined Adam optimiser (smooth → exact) and a
    final simulated‑annealing polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None
    global_a = None

    # ------------------------------------------------------------------
    # Optional starter (previous best known construction)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # length schedule – start modest then grow; shuffled to avoid bias
    lengths = [2000, 4000, 6400, 9600, 12800]
    np.random.shuffle(lengths)

    # use the provided starter if any
    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------- initialise a0 ----------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ------------------- allocate time slice --------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # 70 % of the remaining budget for optimisation, the rest for annealing
        slice_deadline = time.time() + max(8.0, 0.7 * remaining)

        # ------------------- combined Adam descent ------------------
        a_opt, score_opt = _run_combined_descent(a0, n, slice_deadline,
                                                 start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after descent: n={n}, score={score_opt:.6f}")

        # ------------------- annealing refinement -------------------
        if time.time() < deadline:
            anneal_deadline = deadline
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=30_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal: n={n}, score={score_ref:.6f}")

        # remember the best distribution for the next upscale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # return a plain Python list (evaluate_sequence expects a list)
    return [float(x) for x in best_seq]
