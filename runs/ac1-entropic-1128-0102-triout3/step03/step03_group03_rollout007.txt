"""
Propose a candidate step‑function using a multi‑scale soft‑max gradient descent.
The algorithm works on the probability simplex (sum = 1) and repeatedly:

1.  Starts from a known construction (``height_sequence_1``) or a random point.
2.  Upsamples the shape to larger lengths (800 → 1500 → 3000 → 5000).
3.  At each length performs exponentiated‑gradient (mirror‑descent) steps.
    The update direction is the gradient of a *smoothed* maximum of the
    auto‑convolution, obtained with a soft‑max (log‑sum‑exp) weighting.
    The gradient is computed efficiently with two FFT‑based convolutions.
4.  A short stochastic mass‑transfer hill‑climbing phase refines the best
    solution found.
All operations respect the non‑negativity constraint and keep the vector
normalised to unit sum, so the score returned by ``evaluate_sequence`` is
simply ``2·n·max(conv(a,a))``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full auto‑convolution of a (length n) with itself using FFT.
    Returns an array of length 2*n‑1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_and_weights(a: np.ndarray, beta: float):
    """
    Compute the full auto‑convolution and a soft‑max weight vector.
    The weight w_i = exp(beta*(conv_i-max_conv))/sum_j exp(...)
    returns a vector of the same length (2*n-1) that sums to 1.
    """
    conv = _full_convolution(a)
    max_val = conv.max()
    # soft‑max (log‑sum‑exp) – shift by the maximum for stability
    exp_arg = beta * (conv - max_val)
    # under‑flow to 0 is fine; overflow impossible because values ≤ 0
    w = np.exp(exp_arg, dtype=np.float64)
    w_sum = w.sum()
    if w_sum == 0.0:
        # all entries far below max – fall back to uniform weight on the max location(s)
        w = np.zeros_like(conv)
        w[conv == max_val] = 1.0 / np.count_nonzero(conv == max_val)
    else:
        w /= w_sum
    return conv, max_val, w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given the weight vector w (length 2*n-1) compute the gradient of
    the weighted max (i.e. of the smoothed objective) w.r.t. a.
    The closed‑form is
        g[l] = 2 * Σ_i w_i * a_{i-l}   (with out‑of‑range a taken as 0)
    which can be evaluated by a single convolution of w with the reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed copy, length n
    m = w.shape[0]                      # = 2*n - 1
    conv_len = 1 << ((m + n - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=conv_len)
    W = np.fft.rfft(w, n=conv_len)
    full = np.fft.irfft(A_rev * W, n=conv_len)[: m + n - 1]   # length 3*n-2
    # slice n‑1 … n‑1+n‑1 (inclusive start, exclusive end) → length n
    g = 2.0 * full[n - 1: n - 1 + n]
    return g


def _max_convolution(a: np.ndarray) -> float:
    """Fast maximum of the auto‑convolution (no extra weights)."""
    conv = _full_convolution(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old to new length, re‑normalised to sum = 1."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        new_seq[:] = 1.0 / new_len
    else:
        new_seq /= s
    return new_seq

        best_seq = init.copy()
        best_score = 2.0 * n0 * _max_convolution(best_seq)
        print(f"[{time.time() - start_time:5.2f}s] Random init (n={n0}) "
              f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Multi‑scale optimisation (800 → 1500 → 3000 → 5000)
    # --------------------------------------------------------------
    scale_lengths = [800, 1500, 3000, 5000]   # final length respects the 5000‑item cap
    current = best_seq.copy()
    eps = 1e-15                               # numerical floor
    eta0 = 0.5                                 # initial step‑size for EG
    beta_max = 30.0                            # final smoothing parameter
    beta_start = 1.0

    for target_n in scale_lengths:
        if time.time() >= deadline:
            break

        # up‑sample if necessary
        if current.shape[0] != target_n:
            current = _upsample(current, target_n)

        n = target_n
        a = current.copy()                     # working copy
        # iteration budget – more for smaller n, fewer for larger n
        base_iters = 40000
        max_iter = int(base_iters * 2000 / n) + 4000
        max_iter = max(max_iter, 12000)

        # momenta for Adam‑like behaviour (optional – can be turned off)
        m = np.zeros_like(a)                   # first moment
        v = np.zeros_like(a)                   # second moment
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8

        it = 0
        while time.time() < deadline and it < max_iter:
            it += 1

            # ---- compute smoothed objective and its gradient -----------------
            # gradually increase beta (soft‑max temperature)
            beta = beta_start + (beta_max - beta_start) * min(1.0, it / max_iter)
            conv, max_val, w = _conv_and_weights(a, beta)

            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:5.2f}s] "
                      f"Improved (n={n}, iter={it}) score={cur_score:.6f}")

            # gradient of the smoothed max
            g = _gradient_from_weights(a, w)

            # ---- Adam‑style exponentiated‑gradient update on the simplex -----
            # (log‑domain mirror descent)
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # step‑size schedule – decreases like 1/sqrt(t)
            eta = eta0 / math.sqrt(it)

            # log‑update (ensure positivity)
            log_a = np.log(a + eps) - eta * (m_hat / (np.sqrt(v_hat) + eps_adam))
            a = np.exp(log_a)
            a = np.maximum(a, eps)               # avoid exact zeros
            a /= a.sum()

        # prepare for next scale
        current = a.copy()

    # --------------------------------------------------------------
    # 3. Final stochastic local search (mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        it_loc = 0
        max_local_iters = 20000
        while time.time() < deadline and it_loc < max_local_iters:
            it_loc += 1
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] < eps:
                continue
            # move a random fraction (up to 8 %) of the mass from i to j
            frac = np.random.rand() * 0.08
            delta = frac * a[i]
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # quick guard against negative due to rounding
            if a_new[i] < 0:
                continue
            new_score = 2.0 * n * _max_convolution(a_new)
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                a = a_new
                print(f"[{time.time() - start_time:5.2f}s] "
                      f"Random improve (n={n}, it={it_loc}) score={new_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best sequence discovered (as plain Python floats)
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (never happens in practice)
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
