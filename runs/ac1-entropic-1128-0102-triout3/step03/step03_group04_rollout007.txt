    max_c = conv.max()

    """Return the soft‑max of x with temperature beta (β·x)."""
    max_x = x.max()

    s = exp_shifted.sum()

        return np.full_like(x, 1.0 / x.shape[0])
    return exp_shifted / s

"""
Exponential‑gradient minimisation of the maximal self‑convolution.

The step‑function heights are treated as a probability vector (non‑negative,
sum = 1 – scaling does not change the score).  In every iteration we:

1. Compute the full self‑convolution a*a.
2. Form a soft‑max weighting over all convolution entries
   w_k ∝ exp(β·conv_k).  The temperature β is increased slowly so the
   optimiser first flattens the whole convolution and then focuses on the
   remaining peaks.
3. Compute the exact gradient of the smooth objective
   f(a)= (1/β)·log Σ_k exp(β·conv_k) .
   Using the identity ∂conv_k/∂a_i = 2·a_{k‑i} (when defined) the gradient
   can be written as a single convolution w * a_rev, which we evaluate with
   FFTs.
4. Perform an exponentiated‑gradient (mirror descent on the simplex):
   a_i ← a_i·exp(‑η·g_i)   followed by renormalisation.
5. Occasionally add a tiny random jitter to avoid stagnation.
6. Keep the best vector seen so far (according to the official evaluator).

Multiple independent restarts (one warm start from the provided global
candidate, the others random Dirichlet draws) are run as long as the
time budget allows.  The best sequence found before the budget expires is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)

def conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


def softmax_weights(x: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max of x with temperature β (i.e.  w_k ∝ exp(β·x_k))."""
    max_x = x.max()
    # shift for numerical stability
    shifted = beta * (x - max_x)
    exp_shifted = np.exp(shifted)
    s = exp_shifted.sum()
    if s == 0.0 or not np.isfinite(s):
        # fallback to uniform weights
        return np.full_like(x, 1.0 / x.shape[0])
    return exp_shifted / s


def gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max objective
        f(a) = (1/β)·log Σ_k exp(β·conv_k)
    using the exact sub‑gradient of each convolution entry.
    The result is
        g_i = 2· Σ_k w_k·a_{k‑i},
    where w_k are the soft‑max weights of conv.
    """
    w = softmax_weights(conv, beta)               # shape (2n‑1,)
    a_rev = a[::-1]                               # reversed a
    size = w.shape[0] + a_rev.shape[0] - 1        # = 3n‑2
    # convolution w * a_rev via FFT
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a_rev, n=size)
    full = np.fft.irfft(W * A, n=size)
    n = a.shape[0]
    grad = 2.0 * full[n - 1 : n - 1 + n]          # slice length n
    return grad

    """
