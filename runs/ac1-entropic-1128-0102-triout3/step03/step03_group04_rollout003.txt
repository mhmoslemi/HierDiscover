Propose a candidate step‑function sequence that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm follows three stages:

1. **Broad parametric sweep** –  generate monotone power‑law (and a
   simple exponential) candidates for several lengths.  A coarse grid
   over the exponent `p` and the offset is evaluated and the best
   candidate is retained.

2. **Refined local search** – around the best parameters a fine grid
   is explored and the best vector is taken as a seed for a projected
   gradient descent.  The descent uses the sub‑gradient of a weighted
   soft‑max of the top‑k convolution entries, Euclidean‑projects onto
   the probability simplex and employs momentum, step‑size decay and
   occasional jitter to escape plateaus.

3. **Multi‑restart polishing** – a handful of additional restarts are
   launched from random Dirichlet draws (and from the refined seed).  A
   simple simulated‑annealing acceptance rule allows occasional uphill

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Return a length‑n non‑negative vector a_i = (i+offset)^{-p},
    normalised to unit L1‑norm."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i = exp(-lam * i), normalised to unit sum."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w ≥ 0, Σw = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Exact sub‑gradient of the convolution entry at `peak_idx`
    w.r.t. the vector `a`.  Returns a vector of shape a.shape."""
    n = a.shape[0]
    grad = np.zeros_like(a)

    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    # each valid pair contributes 2·a_jdx
    grad[idx] = 2.0 * a[jdx]
    return grad


def gradient_of_top_k(a: np.ndarray, conv: np.ndarray, k: int,
                      beta: float = 5.0) -> np.ndarray:
    """Blend of gradients of the `k` largest convolution entries.
    The entries are soft‑maxed with temperature 1/β."""
    m = conv.shape[0]
    if k >= m:
        idxs = np.arange(m)
    else:
        idxs = np.argpartition(conv, -k)[-k:]
    vals = conv[idxs]
    maxv = conv.max()
    w = np.exp(beta * (vals - maxv))          # shift for numerical stability
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w_sum

    grad = np.zeros_like(a)
    for idx, weight in zip(idxs, w):
        grad += weight * gradient_of_peak(a, int(idx))
    return grad

                        break
                    a = power_law_sequence(n_best, p, offset=offset)
                    cur = fast_score(a)
                    if cur < best_global_score:
                        best_global_score = cur
                        best_global_seq = a.copy()
                        best_global_params = ("power", n_best, p, offset)
                        print(f"[fine] n={n_best:4d} p={p:.4f} off={offset:.3f} → {cur:.6f}")
        else:   # exponential
            _, _, lam0 = best_global_params
            lam_fine = np.arange(max(1e-5, lam0 - 0.0005),
                                 lam0 + 0.00051,
                                 0.00005)
            for lam in lam_fine:
                if time.time() > deadline:
                    break
                a = exponential_sequence(n_best, lam)
                cur = fast_score(a)
                if cur < best_global_score:
                    best_global_score = cur
                    best_global_seq = a.copy()
                    best_global_params = ("exp", n_best, lam)
                    print(f"[fine] n={n_best:4d} lam={lam:.5f} → {cur:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – multi‑restart projected gradient descent
    # ------------------------------------------------------------------
    print("[grad] phase 3 – multi‑restart projected gradient descent")
    max_restarts = 10
    top_k = 15                # more gradients give richer information
    beta_start = 2.0
    beta_inc = 0.004
    eta_start = 0.07
    decay_factor = 0.88
    decay_every = 8000

    # remember the globally best vector found so far
    global_best_seq = best_global_seq.copy() if best_global_seq is not None else None
    global_best_score = best_global_score

    for restart in range(max_restarts):
        if time.time() > deadline:
            break

        # initialise start vector for this restart
        if restart == 0 and best_global_seq is not None:
            a = best_global_seq.copy()
        else:
            # sample a random Dirichlet (concentration < 1 favours sparsity)
            conc = 0.6
            a = rng.gamma(conc, 1.0, size=n_best)
            a /= a.sum()

        cur_score = fast_score(a)
        eta = eta_start
        beta = beta_start
        momentum = np.zeros_like(a)

        it = 0
        # we allow a large number of iterations but cut off when time runs out
        while time.time() < deadline and it < 400_000:
            conv = conv_fft(a)
            # compute a soft‑max blend of top‑k gradients
            grad = gradient_of_top_k(a, conv, top_k, beta=beta)

            # momentum update
            momentum = 0.9 * momentum - eta * grad

            # propose new point and re‑project onto simplex
            a_trial = project_onto_simplex(a + momentum)

            # occasional jitter to shake off flat plateaus
            if rng.random() < 0.003:
                jitter = (rng.random_like(a) - 0.5) * 1e-6
                a_trial = project_onto_simplex(a_trial + jitter)

            trial_score = fast_score(a_trial)

            # record improvement globally
            if trial_score < global_best_score:
                global_best_score = trial_score
                global_best_seq = a_trial.copy()
                print(f"[restart {restart} iter {it:07d}] new global best = {global_best_score:.6f}")

            # accept if better, otherwise probabilistic Metropolis step
            if trial_score <= cur_score:
                a = a_trial
                cur_score = trial_score
            else:
                # simple temperature schedule
                temperature = 0.02 * np.exp(-it / 250_000.0)
                prob = exp(-(trial_score - cur_score) / max(temperature, 1e-12))
                if rng.random() < prob:
                    a = a_trial
                    cur_score = trial_score

            # step‑size decay and β increase
            if (it + 1) % decay_every == 0:
                eta = max(eta * decay_factor, 1e-6)
                beta = beta + beta_inc

            it += 1

        print(f"[restart {restart}] finished – local best score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(global_best_seq.tolist())
    print("[done] final reported score =", final_score)
    print("[info] best parameters :", best_global_params)

    return global_best_seq.tolist()
