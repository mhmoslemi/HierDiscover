def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    best_n = None

    # Starting point
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # schedule lengths
    lengths = [1024, 2048, 4096, 8192, 12288, 16384]  # to increase length gradually
    # randomize order maybe but we want increasing length to use best as warm start
    # We'll iterate in order but can skip if not enough time.

    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # init a0
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # allocate slice time proportionally: maybe 15% of remaining each length
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.15, 120.0)  # at most 2 minutes per length
        slice_deadline = time.time() + slice_time

        a_opt, score_opt = _run_optimized_descent(a0, n, slice_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            best_n = n

        # annealing

"""
Multi‑scale Adam + log‑sum‑exp smoothing with long simulated‑annealing polish.

* Works on the probability simplex (∑aᵢ = 1).  The objective
  2·n·max(conv) is scale‑invariant, so we keep the sum exactly 1.
* Uses an Adam‑style adaptive step on the smooth surrogate
  L_λ(a)=λ·log∑exp(conv/λ).  λ starts as a fraction of the current
  maximum convolution value and is multiplied by 0.9 every few thousand
  iterations (slow annealing → a close approximation of the true max).
* After each Adam step the vector is projected onto the simplex
  (Euclidean projection, Duchi et al. 2008).
* A hierarchy of lengths (doubling each stage) provides a warm‑start
  for larger problems.  The final length (≈16 k) receives the bulk of
  the time budget.
* When the time budget is almost exhausted a cheap mass‑transfer
  simulated‑annealing sweep (≈200 k proposals) is performed to polish
  the result.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Adam optimisation on the smoothed max (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      max_iter: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate of max(conv).
    Returns the best vector seen and its exact objective value
    (2·n·max(conv) because the sum of a is kept equal to 1).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    step_alpha = 0.7               # base learning‑rate factor
    best_a = a.copy()
    best_score = float('inf')
    it = 0

    # smoothing schedule
    lam_factor = 0.4               # initial λ = lam_factor * max(conv)
    lam_decay = 0.9
    lam_decay_interval = 5000

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ (slow annealing)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-12)
        lam = max(max_b * lam_factor, 1e-14)

        # --- gradient of the log‑sum‑exp surrogate -----------------------
        # soft‑max weights over convolution entries
        max_val = conv.max()
        w = np.exp((conv - max_val) / lam)
        w /= w.sum()
        # cross‑correlation w * a_rev gives dL/da
        a_rev = a[::-1]
        full = _conv_fft(w, a_rev)                # length 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]        # shape (n,)

        # --- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step_alpha / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        # project back to the simplex (guarantees non‑negativity & sum=1)
        a = _project_simplex(a)

    return best_a, best_score

def _run_annealing_refine(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated annealing.  Proposes moving a small ε from one
    coordinate to another and accepts according to a geometric temperature
    schedule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    # temperature schedule – start a little higher than in the previous code
    start_temp = 5e-2
    end_temp = 1e-6

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move: transfer a small fraction of mass from i to j
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Finds a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm runs for at most `budget_s`
    seconds and combines multi‑scale Adam optimisation with a final
    simulated‑annealing polishing step.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter sequence (provided by the harness)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – we go up to ~16 k (a power of two is FFT‑friendly)
    lengths = [800, 1600, 3200, 6400, 12800, 16384]
    # First few lengths get a modest slice of time, the last one receives the rest
    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # initialise a0 (warm‑start by up‑sampling the best so far)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # slice of the remaining budget for this stage
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        if idx < len(lengths) - 1:                # not the last (largest) stage
            # Give about 8 % of the remaining time, but at least a few seconds
            slice_time = max(5.0, 0.08 * remaining)
        else:
            # Last stage – consume everything that is left
            slice_time = remaining

        slice_deadline = time.time() + slice_time

        # --------------------------------------------------------------
        # Adam optimisation on the smoothed surrogate
        # --------------------------------------------------------------
        a_opt, score_opt = _run_adam_descent(a0, n, slice_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after Adam: n={n}, score={score_opt:.6f}")

        # --------------------------------------------------------------
        # Final annealing polish (if we still have time)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(3.0, 0.04 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing_refine(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                steps=200_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal: n={n}, score={score_ref:.6f}")

        # keep the best distribution for the next up‑sampling
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Convert to plain Python list – the harness will call evaluate_sequence()
    return [float(x) for x in best_seq]
