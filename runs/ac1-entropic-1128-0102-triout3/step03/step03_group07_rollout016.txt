```python
def half_binomial_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0*i - 1.0) / (2.0*i)
    a /= a.sum()
    return a
```

"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

using an alternating linear‑programming / mixing scheme.
The algorithm is inspired by the LP‑based iteration described in the
reference solution, but it differs from the original gradient descent
code.

Main ideas
----------
* Keep the sequence on the probability simplex (∑a_i = 1, a_i ≥ 0).
* For a current sequence f, compute its auto‑convolution peak M.
* Solve the LP

      maximise   Σ g_i
      subject to   (f * g)_k ≤ M   for k = 0,…,2n‑2
                   g_i ≥ 0

  (the convolution constraints are linear in g because f is fixed).
* Scale the LP solution to a probability vector  ĝ = g / Σg .
* Mix the current iterate with ĝ:
      f_new = (1‑t)·f + t·ĝ
  The mixing factor t is chosen from a small set of candidates by
  evaluating the exact convolution of the mixture (via FFT) and picking
  the t that yields the smallest new peak.
* Accept the mixture if it reduces the peak; otherwise keep f.
* Occasionally restart from a random Dirichlet draw when no improvement
  has been seen for many iterations.

The whole routine runs for at most ``budget_s`` seconds (default 950 s)
and returns the best sequence found.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def half_binomial_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑½, normalised to sum to 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_fft(x: np.ndarray, y: np.ndarray,
             nfft: int, L: int) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of two real vectors using FFT.
    Returns the first L entries (length of the linear convolution).
    """
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(X * Y, n=nfft)[:L]
    return conv


def build_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Return the (2n‑1)×n matrix representing the linear convolution
    (a * g) as a function of g.  Each row k contains the coefficients
    a_{k‑j} (or 0) for the variables g_j.
    """
    n = a.shape[0]
    L = 2 * n - 1
    # For offset d (≥0) the diagonal consists of the constant a[d].
    # Construct an (n, n) array where row d is filled with a[d].
    diag_data = np.broadcast_to(a[:, None], (n, n))
    offsets = np.arange(n)                     # 0,…,n‑1
    A = sp.diags(diag_data, offsets, shape=(L, n), format='csr')
    return A

    linear‑programming based alternating optimisation.
    """

    M = cf.max()

    # -------------------------- optimisation loop ------------------------
    deadline = time.time() + budget_s - 5.0   # safety margin
    it = 0
    it_since_best = 0
    restart_gap = 4000      # restart after this many iterations without improvement
    t_candidates = [0.0, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    while time.time() < deadline:

        # ----- 5. line‑search for the best mixing proportion -------------
        best_local_M = M
