"""
Boosted gradient‑descent search for a low‑max‑auto‑convolution step function.

The evaluation is scale‑invariant, so every candidate is kept on the
probability simplex (non‑negative, sums to one).  For a fixed length `n`
the objective is

    score = 2·n·max(conv(a,a))

with `conv` the linear auto‑convolution.  We minimise a smooth approximation
to the max via a soft‑max (temperature τ) and apply an Adam optimiser
with Euclidean projection onto the simplex (Duchi et al.).  The search
strategy is:

1.  Initialise a pool of diverse shapes (uniform, triangular, Hann,
    several Beta‑shaped densities, random Dirichlet, and the previous
    best if available) for a set of lengths.
2.  Refine each start with Adam for a length‑dependent budget of
    iterations, keeping the best sequence seen.
3.  Upsample the current best by linear interpolation (≈1.5× larger)
    repeatedly up to a configurable maximum length, refining at each
    resolution.
4.  Use the remaining time for many short random‑perturbation restarts
    on the final length to escape possible local minima.
5.  Return the best sequence (as plain Python floats).  All intermediate
    progress and scores are printed.

The code respects the 1000 s time budget by checking the deadline after
each iteration and each outer loop.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> float:
    """Return max(conv(a,a)) for a vector that already sums to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


def _score_of_seq(a: np.ndarray) -> float:
    """Score = 2·n·max(conv) for a normalized (sum=1) sequence."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)

    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the smoothed max‑conv objective.

    Returns the best sequence (still on the simplex) seen during the run
    and its score.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]

    # moments for Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score_of_seq(a)

    # pre‑allocate FFT lengths (they depend only on n)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    # the length of the soft‑max weight vector equals 2*n‑1
    len_w = 2 * n - 1
    fft_len_grad = 1 << ((len_w + n - 1).bit_length())

    a_rev = a[::-1]                     # will be recomputed each iter
    eps_range = eps_end - eps_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution of a with itself -----
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[:len_w]
        max_val = float(conv.max())
        cur_score = 2.0 * n * max_val

        # keep the best seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights (temperature τ) -----
        eps_rel = eps_start + eps_range * (it / max_iters)
        tau = max(eps_rel * max_val, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w.fill(1.0 / len_w)

        # ----- gradient of the smoothed maximum -----
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2_full = np.fft.irfft(W * A_rev, n=fft_len_grad)[:len_w + n - 1]
        # slice the part that corresponds to the gradient entries
        g = 2.0 * conv2_full[n - 1 : n - 1 + n]

        # ----- Adam moment updates -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- parameter update + projection onto simplex -----
        step = step0 / math.sqrt(it)
        a = a - step * (m_hat / (np.sqrt(v_hat) + 1e-8))
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to compute a score quickly (exact for normalized vectors)
    # ------------------------------------------------------------------
    def _record(seq: np.ndarray, score: float, msg: str):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] {msg} → new best score {score:.6f}")

    # ------------------------------------------------------------------
    # 1. Initialise a pool of lengths and diverse starting points
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1800, 2700, 4000, 6000, 9000]
    MAX_LEN = 20000

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] Exploring length n={n}")
        # ----- build a list of candidate initial vectors on the simplex -----
        candidates = []

        # a) uniform
        candidates.append(np.full(n, 1.0 / n))

        # b) triangular (tent)
        idx = np.arange(n)
        tri = np.minimum(idx + 1, n - idx)
        candidates.append(tri / tri.sum())

        # c) Hann window
        hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
        candidates.append(hann / hann.sum())

        # d) a few symmetric Beta shapes (p < 1 gives edge‑heavy mass)
        for p in (0.3, 0.5, 0.8, 1.5):
            x = (np.arange(n) + 0.5) / n
            beta = np.power(x, p - 1) * np.power(1 - x, p - 1)
            beta = np.maximum(beta, 0.0)
            candidates.append(beta / beta.sum())

        # e) a random Dirichlet point
        candidates.append(np.random.dirichlet(np.ones(n)))

        # f) reuse the previous best if it exists (interpolated to current n)
        if best_seq is not None:
            cur_n = best_seq.shape[0]
            a_interp = np.interp(
                np.linspace(0.0, 1.0, n),
                np.linspace(0.0, 1.0, cur_n),
                best_seq,
            )
            a_interp = np.maximum(a_interp, 0.0)
            s = a_interp.sum()
            if s > 0:
                candidates.append(a_interp / s)

        # ----- refinement budget – more iterations for smaller n ----------
        max_iters = int(30000 * 1200 / n) + 15000
        for i, a0 in enumerate(candidates, 1):
            if time.time() >= deadline:
                break
            print(
                f"   start {i}/{len(candidates)} – iters={max_iters}"
            )
            refined, score = _refine_adam(
                a0,
                max_iters=max_iters,
                deadline=deadline,
                step0=0.6,
                eps_start=0.05,
                eps_end=0.001,
                beta1=0.9,
                beta2=0.999,
            )
            _record(refined, score, f"len={n} start#{i}")

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling of the current best shape
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1000, 1.0 / 1000)

    cur_n = best_seq.shape[0]
    while time.time() < deadline:
        next_n = int(cur_n * 1.5)
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n <= cur_n:
            break

        # linear interpolation to the larger grid
        a_up = np.interp(
            np.linspace(0.0, 1.0, next_n),
            np.linspace(0.0, 1.0, cur_n),
            best_seq,
        )
        a_up = np.maximum(a_up, 0.0)
        a_up /= a_up.sum()

        # iteration budget – fewer iters for larger n
        max_iters = int(30000 * cur_n / next_n) + 15000
        print(
            f"[{time.time() - start_time:6.2f}s] Upsampling {cur_n}->{next_n}, iters={max_iters}"
        )
        refined, score = _refine_adam(
            a_up,
            max_iters=max_iters,
            deadline=deadline,
            step0=0.55,
            eps_start=0.04,
            eps_end=0.001,
            beta1=0.9,
            beta2=0.999,
        )
        _record(refined, score, f"upsampled to {next_n}")

        cur_n = next_n
        # keep the interpolated (but not yet refined) shape for the next round
        best_seq = refined

    # ------------------------------------------------------------------
    # 3. Final intensive random‑restart fine‑tuning on the best length
    # ------------------------------------------------------------------
    n = best_seq.shape[0]
    remaining = deadline - time.time()
    if remaining > 5.0:
        # estimate a cheap restart to take roughly 4–5 seconds
        # (empirically ~20000 iterations for n≈2000)
        iters_per_restart = max(15000, int(20000 * 1500 / n))
        # allow as many restarts as fit into the remaining budget
        restart_budget = 4.0  # seconds per restart (very rough)
        max_restarts = int(remaining // restart_budget)
        print(
            f"[{time.time() - start_time:6.2f}s] Final phase – {max_restarts} random restarts"
        )
        for r in range(max_restarts):
            if time.time() >= deadline:
                break
            # perturb the current best a little and renormalise
            perturb = 0.01 * np.random.randn(n)
            a0 = best_seq + perturb
            a0 = np.maximum(a0, 0.0)
            a0_sum = a0.sum()
            if a0_sum == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= a0_sum

            refined, score = _refine_adam(
                a0,
                max_iters=iters_per_restart,
                deadline=deadline,
                step0=0.45,
                eps_start=0.02,
                eps_end=0.0005,
                beta1=0.9,
                beta2=0.999,
            )
            _record(refined, score, f"restart {r+1}/{max_restarts}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total

    print(
        f"[{time.time() - start_time:6.2f}s] Finished – best score ≈ {best_score:.6f}"
    )
    return [float(x) for x in best_seq]
