def fft_convolve(x, y):
    n = x.size
    size = 1 << ((2*n-1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:2*n-1]
    return conv

"""
Efficient search for a low‑score step function.

The score is
    score = 2·n·max(conv(a, a)) / (sum(a))²,
with a ≥ 0.  The scale of a does not matter, therefore we keep
sum(a)=1 (simplex constraint).  The algorithm starts from the
uniform distribution and repeatedly smoothes the L∞‑objective
using a log‑sum‑exp surrogate:

    Φβ(a) = (1/β) log Σₖ exp(β·(a∗a)ₖ).

∇Φβ(a) = 2·(w ∗ a) where wₖ ∝ exp(β·(a∗a)ₖ) (the weights sum to 1).
We minimise Φβ with projected Adam on the simplex.  The smoothing
parameter β is increased during the run so that the optimisation
first behaves like a soft‑max and finally focuses on the true
maximum.  If the method does not reach the target, a fast greedy
search for a sparse support set (uniform weights on the support)
is tried as a fallback.

The whole procedure respects a global time‑budget (default 1000 s)
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective for a non‑negative vector a."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def self_convolution(a: np.ndarray) -> np.ndarray:
    """Return a∗a (ordered pair count) of length 2·n‑1."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (w ⋆ a) of lengths M and n.
    Returned length is M+n‑1, with lag 0 at index len(a)-1.
    """
    M = w.size
    n = a.size
    size = 1 << ((M + n - 1).bit_length())
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(fw * np.conj(fa), size)[: M + n - 1]
    return corr


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Simple O(n log n) implementation (the classic sorting method)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Compute the soft‑max weights wₖ ∝ exp(β·convₖ)."""
    # shift by the maximum to avoid overflow
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
def adam_step(a: np.ndarray,
              grad: np.ndarray,
              m: np.ndarray,
              v: np.ndarray,
              t: int,
              lr: float,
              b1: float,
              b2: float,
              eps: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Perform one Adam update (returns new a, new m, new v)."""
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * (grad * grad)
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    # projection onto the simplex (non‑negative, sum=1)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.size)
    else:
        a_new /= s
    return a_new, m, v


# ----------------------------------------------------------------------
def gradient_descent_smoothing(n: int,
                               time_limit: float,
                               target_score: float = 1.503) -> np.ndarray:
    """
    Minimise the smooth approximation of the L∞‑objective with Adam.
    Returns the best vector seen within the allotted time.
    """
    # start from the uniform distribution (sum = 1)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    lr_initial = 0.12

    best_a = a.copy()
    best_val = fast_score(a)

    start = time.time()
    deadline = start + time_limit - 0.2

    # schedule for the smoothing parameter β
    beta_min = 0.4
    beta_max = 12.0
    # reach beta_max roughly halfway through the allowed time
    beta_ramp_time = (deadline - start) * 0.5

    it = 1
    stagnation = 0  # counts iterations without improvement

    while time.time() < deadline:
        # compute self‑convolution and the soft‑max weights
        conv = self_convolution(a)

        # update β (linear ramp)
        now = time.time()
        beta = beta_min if now < start else min(
            beta_max,
            beta_min + (beta_max - beta_min) * (now - start) / beta_ramp_time
        )
        w = smooth_max_weights(conv, beta)

        # gradient = 2·(w ⋆ a) restricted to n entries
        corr = correlation(w, a)                     # length = (2n‑1)+n‑1 = 3n‑2
        grad = 2.0 * corr[n - 1 :: -1]               # take indices n‑1 … 0

        # Adam step
        a, m, v = adam_step(a, grad, m, v, it, lr_initial, b1, b2, eps)

        # occasional evaluation
        if it % 250 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                stagnation = 0
                print(f"[iter {it:6d}] new best {best_val:.6f} (β={beta:.2f})")
                if best_val <= target_score:
                    print("[success] target reached during smoothing.")
                    break
            else:
                stagnation += 1

            # if we see no improvement for a while, add a tiny random jitter
            if stagnation >= 30:
                jitter = np.random.randn(n) * 0.004
                a = a + jitter
                a = np.maximum(a, 0.0)
                s = a.sum()
                a = a / s if s > 0 else np.full_like(a, 1.0 / n)
                stagnation = 0

        it += 1

    print(f"[final smoothing] best score {best_val:.6f}")
    return best_a


# ----------------------------------------------------------------------
def greedy_support_set(n: int,
                       k_target: int,
                       max_allowed: int,
                       time_limit: float) -> np.ndarray | None:
    """
    Faster (but still greedy) construction of a support set S of size k_target
    such that the ordered‑pair count never exceeds max_allowed.
    Returns the binary indicator vector (length n) or None.
    """
    deadline = time.time() + time_limit
    while time.time() < deadline:
        # random permutation of indices, pick first k_target and test
        perm = np.random.permutation(n)
        S = perm[:k_target]
        # compute ordered counts via convolution of binary vector
        indicator = np.zeros(n, dtype=np.float64)
        indicator[S] = 1.0
        cnt = self_convolution(indicator)  # ordered pairs count
        if cnt.max() <= max_allowed:
            return indicator
    return None


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with evaluation score ≤ 1.503.
    Primary method: Adam optimisation on a smooth L∞ surrogate.
    If that does not meet the target, a quick greedy support‑set search is used
    as a fallback.  The returned sequence is a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ---- problem parameters ------------------------------------------------
    n = 1500                     # length of the step‑function
    target_score = 1.503

    # ----------------------------------------------------------------------
    # 1. Main optimisation – smoothed Adam
    remaining = max(0.0, deadline - time.time())
    opt_vec = gradient_descent_smoothing(n, remaining, target_score)
    best_val = fast_score(opt_vec)

    # early exit if we already beat the target
    if best_val <= target_score:
        print("[final] target achieved by gradient descent.")
        return opt_vec.tolist()

    # ----------------------------------------------------------------------
    # 2. Fallback: try to construct a uniform‑weight sparse support set
    #    (the greedy approach is very cheap compared to the Adam run).
    support_sizes = [120, 110, 100, 90, 80]      # try a few k values
    for k in support_sizes:
        allowed = max(1, int(math.floor(target_score * k * k / (2.0 * n))))
        print(f"[fallback] trying support size k={k}, allowed max count={allowed}")
        # give each size at most 30 s (or whatever time is left)
        time_left = max(0.0, deadline - time.time())
        if time_left < 5.0:
            break
        t_alloc = min(30.0, time_left)
        indicator = greedy_support_set(n, k, allowed, t_alloc)
        if indicator is None:
            continue
        # uniform weights on the support
        a = indicator / k
        val = fast_score(a)
        print(f"   -> found set, score={val:.6f}")
        if val < best_val:
            best_val, opt_vec = val, a.copy()
        if best_val <= target_score:
            print("[final] target reached by greedy support set.")
            return opt_vec.tolist()

    # ----------------------------------------------------------------------
    # 3. If everything failed, fall back to plain uniform distribution.
    if np.allclose(opt_vec, 0):
        print("[final] returning uniform distribution as last resort.")
        opt_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(opt_vec)

    print(f"[final] best value returned: {best_val:.6f}")
    return opt_vec.tolist()
