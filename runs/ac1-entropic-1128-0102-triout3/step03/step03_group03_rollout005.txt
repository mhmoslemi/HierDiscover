function project_simplex(v):
    # Returns w such that w_i >=0, sum w_i =1.
    # Use algorithm by Duchi et al.
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv/ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v):
    # sorts v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def softmax_weights(conv, beta):
    # conv may be large length; for numerical stability subtract max before exp.
    exps = np.exp(beta * (conv - conv.max()))
    w = exps / np.sum(exps)
    return w

"""
Propose a candidate step function using a smoothed‑max (soft‑max) gradient descent
on the probability simplex, combined with progressive up‑sampling and a final
random local refinement.  The method differs from the previous active‑set
exponentiated‑gradient approach: it uses the *full* convolution gradient (via
FFT) and a temperature schedule that gradually approximates the true maximum.
All updates keep the vector strictly positive and renormalise to sum = 1,
which respects the invariance of the evaluation function under scaling.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """
    Full auto‑convolution of a with itself (using FFT) and its maximum.
    Returns (conv, max_value) where conv has length 2*len(a)-1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())


def _gradient_from_weights(a: np.ndarray, w: np.ndarray):
    """
    Gradient of the soft‑max‑weighted max‑convolution.
    For each i, g_i = 2 * Σ_k w_k * a_{k-i} (out‑of‑range indices are ignored).
    Implemented as a convolution of w with the reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    total_len = w.shape[0] + a_rev.shape[0] - 1   # = 3n-2
    fft_len = 1 << (total_len.bit_length())
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(W * A_rev, n=fft_len)     # length = total_len
    start = n - 1                                 # index that aligns with i=0
    g = 2.0 * conv[start:start + n]               # length n
    return g


def _score_seq(a: np.ndarray) -> float:
    """Score = 2 * n * max(conv(a,a))  (a is assumed to sum to 1)."""
    n = a.shape[0]
    _, m = _conv_and_max(a)
    return 2.0 * n * m


def _optimize_seq(a: np.ndarray,
                  best_seq: np.ndarray,
                  best_score: float,
                  deadline: float,
                  start_time: float):
    """
    Smoothed‑max gradient descent on the simplex for a fixed length.
    Returns (a, best_seq, best_score) after the allotted iterations.
    """
    n = a.shape[0]

    # iteration budget – a few 10‑k steps are enough for the allowed time.
    if n <= 3000:
        max_iters = 20000
    elif n <= 5000:
        max_iters = 15000
    else:
        max_iters = 12000

    # schedule parameters
    beta_start = 5.0
    beta_end = 150.0
    lr0 = 0.8
    eps_floor = 1e-15

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # 1) compute convolution and current score
        # --------------------------------------------------------------
        conv_full, max_val = _conv_and_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"Improved (n={n}, it={it}): score={cur_score:.6f}")

        # --------------------------------------------------------------
        # 2) soft‑max weights (smooth approximation of the max)
        # --------------------------------------------------------------
        # temperature grows geometrically from beta_start to beta_end
        beta = beta_start * (beta_end / beta_start) ** ((it - 1) / max_iters)
        # soft‑max; shift by max_val for numerical stability
        w = np.exp(beta * (conv_full - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:                     # extremely unlikely
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of the smoothed objective
        # --------------------------------------------------------------
        g = _gradient_from_weights(a, w)

        # --------------------------------------------------------------
        # 4) exponentiated‑gradient update on the simplex
        # --------------------------------------------------------------
        lr = lr0 / math.sqrt(it)                     # decreasing step size
        exponent = -lr * g
        exponent = np.clip(exponent, -50.0, 50.0)     # avoid overflow
        a = a * np.exp(exponent)
        a = np.maximum(a, eps_floor)                 # keep strictly positive
        a /= a.sum()

    return a, best_seq, best_score

        if next_n <= cur_n:
            break

        # linear up‑sampling to the new grid
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a = np.interp(x_new, x_old, current)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        a = a / a_sum if a_sum > 0 else np.full(next_n, 1.0 / next_n)

        # optimise the up‑sampled shape
        a, best_seq, best_score = _optimize_seq(a, best_seq, best_score,
                                                deadline, start_time)

        # prepare for the next resolution level
        current = a.copy()
        cur_n = next_n
        print(f"[{time.time() - start_time:6.2f}s] Finished size {cur_n}, "
              f"best_score so far={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Final cheap random mass‑transfer local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        max_local_iters = 18000
        for it in range(max_local_iters):
            if time.time() >= deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move a modest random fraction from i to j
            eps = np.random.rand() * 0.04          # up to 4 % of source mass
            delta = eps * a[i]
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # safety: keep non‑negative (numerical)
            if a_new[i] < 0.0:
                continue
            _, max_val = _conv_and_max(a_new)
            new_score = 2.0 * n * max_val
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                a = a_new
                print(f"[{time.time() - start_time:6.2f}s] Local improve: "
                      f"n={n}, it={it}, score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(100, 1.0 / 100)

    return [float(v) for v in best_seq]
