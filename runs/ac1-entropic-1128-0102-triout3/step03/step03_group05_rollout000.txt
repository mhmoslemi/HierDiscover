"""
Propose a step‑function sequence with a low evaluation value.

Algorithm
---------
* Initialise a population of analytically‑motivated sequences (power‑law,
  linear, uniform, the previous best if available) and keep the best one.
* Run an Adam‑style stochastic optimisation on the *max* of the
  self‑convolution (the objective is scale‑invariant, we keep the sum
  equal to 1).  At each iteration we:
    – compute the convolution via FFT,
    – select the most critical indices (those within a shrinking relative
      gap of the current maximum),
    – build a sub‑gradient from those active indices,
    – apply an Adam update and project back onto the probability simplex.
* Occasionally perform random “mass‑transfer’’ moves to escape plateaus.
* If no improvement has been observed for a while, re‑seed from a noisy
  version of the current best sequence.
* After the time budget expires we return the best sequence found.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n-i) (i=0..n‑1) normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def gradient_multi_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)_k for a set of active indices.
    For each active k the contribution to the gradient is
        g_i ← g_i + 2·a_{k-i}   (when 0 ≤ k‑i < n)
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        grad[i] += 2.0 * a[k - i]
    return grad


# ----------------------------------------------------------------------
def fast_score_from_max(n: int, max_conv: float) -> float:
    """Given max of the convolution and length n, compute the evaluation."""
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation value.  The routine runs for at most `budget_s` seconds and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # 1. Problem size (adjustable – larger gives more freedom, slower per
    #    iteration).  5000 works comfortably within the time budget.
    n = 5000

    # --------------------------------------------------------------
    # 2. Build a small pool of analytically‑motivated initial sequences.
    candidates = []

    # Power‑law family
    exponents = [0.20, 0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60]
    offsets   = [0.0, 0.1, 0.5, 1.0, 2.0]
    for p in exponents:
        for off in offsets:
            candidates.append(build_power_law(n, p, off))

    # Linear descending shape
    candidates.append(build_linear_descending(n))

    # Uniform (baseline)
    candidates.append(np.full(n, 1.0 / n))

    # Include the previous best if it exists
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                seq = np.zeros(n, dtype=np.float64)
                seq[:prev.size] = prev
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                seq /= s
                candidates.append(seq)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (according to the exact objective).
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        conv = np.fft.irfft(np.fft.rfft(seq, 1 << ((2 * n - 1).bit_length())) *
                           np.fft.rfft(seq, 1 << ((2 * n - 1).bit_length())),
                           1 << ((2 * n - 1).bit_length()))[: 2 * n - 1]
        val = fast_score_from_max(n, conv.max())
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best initial value: {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation parameters
    a = best_seq.copy()                     # current iterate (on simplex)
    best_seq = a.copy()
    best_val = fast_score_from_max(n, np.fft.irfft(
        np.fft.rfft(a, 1 << ((2 * n - 1).bit_length())) *
        np.fft.rfft(a, 1 << ((2 * n - 1).bit_length())),
        1 << ((2 * n - 1).bit_length()))[:2 * n - 1].max())
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.8                # base learning rate (will be scaled per coordinate)

    # Active‑set annealing
    delta_start = 0.04
    delta_end   = 0.005
    K = 12                   # max number of active indices to keep
    it = 0
    last_improve = time.time()

    # FFT size (next power of two ≥ 2n‑1)
    conv_size = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # 5. Main optimisation loop
    while time.time() < deadline:
        # ------ Current convolution & objective ------
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_conv = conv.max()
        cur_val = fast_score_from_max(n, max_conv)

        # Update global best
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = time.time()

        # ------ Determine active indices (those close to the max) ------
        delta = max(delta_end,
                    delta_start * (1.0 - it / 250000.0))   # linear anneal
        mask = conv >= max_conv * (1.0 - delta)
        active = np.flatnonzero(mask)
        if active.size > K:
            # keep the K indices with largest convolution values
            topK = np.argpartition(-conv[active], K)[:K]
            active = active[topK]

        # ------ Gradient from active set ------
        grad = gradient_multi_active(a, active)

        # ------ Adam update ------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / (np.sqrt(v_hat) + eps) * m_hat

        a_candidate = proj_simplex(a - step)

        # ------ Evaluate candidate ------
        fft_c = np.fft.rfft(a_candidate, conv_size)
        conv_c = np.fft.irfft(fft_c * fft_c, conv_size)[: 2 * n - 1]
        cand_val = fast_score_from_max(n, conv_c.max())

        # ------ Acceptance rule (strict improvement, with occasional uphill) ------
        if cand_val < best_val:
            a = a_candidate
            best_val = cand_val
            best_seq = a.copy()
            last_improve = time.time()
        else:
            if np.random.rand() < 0.001:          # rare uphill moves
                a = a_candidate

        # ------ Random “mass‑transfer’’ hill‑climbing move ------
        if np.random.rand() < 0.03:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 1e-12:
                transfer = 0.25 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= transfer
                cand[j] += transfer
                cand = proj_simplex(cand)
                fft_c2 = np.fft.rfft(cand, conv_size)
                conv_c2 = np.fft.irfft(fft_c2 * fft_c2, conv_size)[:2 * n - 1]
                cand_val2 = fast_score_from_max(n, conv_c2.max())
                if cand_val2 < best_val:
                    a = cand
                    best_val = cand_val2
                    best_seq = cand.copy()
                    last_improve = time.time()
                elif np.random.rand() < 0.001:
                    a = cand

        # ------ Stagnation handling ------
        if time.time() - last_improve > 35.0:
            # Re‑seed from a noisy version of the current best
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(0.85 * best_seq + 0.15 * noise)
            # Reset Adam moments to avoid dragging old gradients
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

        # ------ Periodic reporting ------
        if it % 4000 == 0:
            print(f"[iter {it:7d}] cur {cur_val:.6f}  best {best_val:.6f}")

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # The external checker will renormalise if needed; we keep sum = 1.
    return best_seq.tolist()
