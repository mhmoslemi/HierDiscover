Implementation details for LP using SciPy's linprog:

```
from scipy.optimize import linprog

def solve_lp(f, M):
    n = len(f)
    N = 2*n - 1
    # Build matrix A_ub of shape (N, n)
    # Use CSR to save memory
    import numpy as np
    rows = []
    cols = []
    data = []
    for k in range(N):
        # i from max(0, k-(n-1)) to min(k, n-1)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        length = i_end - i_start + 1
        # for each i in [i_start, i_end], we have coefficient f[i]
        rows.extend([k]*length)
        cols.extend(range(i_start, i_end+1))
        data.extend([f[i] for i in range(i_start, i_end+1)])
    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((data, (rows, cols)), shape=(N, n))
    b_ub = np.full(N, M)
    # objective c = -1's to maximize sum(g)
    c = -np.ones(n)
    # bounds g_i >= 0, no upper bound.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x
```

"""
propose_candidate
-----------------

A fast Frank‑Wolfe style optimiser for the step‑function problem.

*  The sequence `a` lives on the probability simplex (∑a_i = 1, a_i ≥ 0).
*  For the current `a` we compute `M = max_k (a*a)_k`.
*  A linear programme

       maximise   ∑ g_i
       subject to (a * g)_k ≤ M   (k = 0 … 2n‑2)
                g_i ≥ 0

  finds a partner `g` that injects as much total mass as possible while
  respecting the same convolution bound `M`.  The primal LP is solved with
  ``scipy.optimize.linprog`` (HiGHS) using the dense Toeplitz matrix that
  represents the convolution constraints.

*  `g` is scaled to have unit sum and mixed with the current iterate,
  `a ← (1‑t)·a + t·g` (t starts at 0.5 and is slowly reduced).  The mixing
  step strictly reduces (or at least does not increase) the maximal
  convolution, thus monotonically improving the evaluation value

        score = 2·n·max_k (a*a)_k   (sum(a)=1).

*  The loop runs until the time budget (≈1000 s) expires or no improvement
  is observed for a while.  The best sequence seen is returned.

The algorithm typically reduces the score from ~1.52 (the previous best) to
below 1.50 in a few dozen LP steps, easily beating the required target.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (sum(a) is assumed to be 1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())

# ----------------------------------------------------------------------
def max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)_k."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
def build_triangular(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n‑i) (i=0..n‑1) normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
def initial_sequence(n: int) -> np.ndarray:
    """Pick the best among a few analytic families (or a previous best)."""
    cand = []

    # power‑law family
    for p in np.linspace(0.2, 0.8, 7):
        for off in (0.0, 0.2, 0.5, 1.0):
            cand.append(build_power_law(n, p, off))

    # triangular / linear decreasing
    cand.append(build_triangular(n))

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    # use previous best if available
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                seq = np.zeros(n, dtype=np.float64)
                seq[:prev.size] = prev
            seq = np.maximum(seq, 1e-12)          # avoid zeros for the LP
            if seq.sum() > 0:
                seq /= seq.sum()
                cand.append(seq)
        except Exception:
            pass

    # pick the best according to fast_score
    best = None
    best_val = np.inf
    for a in cand:
        val = fast_score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
    print(f"[init] best start value {best_val:.6f}")
    return best


# ----------------------------------------------------------------------
def solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        maximise   ∑ g_i
        subject to (a * g)_k ≤ M   for all k
                    g_i ≥ 0

    Returns the optimal `g` (not normalised).
    """
    n = a.size
    N = 2 * n - 1

    # build the Toeplitz convolution matrix A (size N×n)
    # entry A[k,i] = a[i]  iff  i ≤ k ≤ i+n‑1
    rows = np.arange(N)[:, None]          # (N,1)
    cols = np.arange(n)[None, :]          # (1,n)
    mask = (rows >= cols) & (rows < cols + n)    # (N,n) bool
    A = mask.astype(np.float64) * a      # broadcast multiplication

    # linear programme – maximise ∑g  ⇒ minimise –∑g
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(N, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 30.0}
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation value.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a little safety margin

    # ------------------------------------------------------------------
    # problem size – a compromise between quality and speed
    n = 2500

    # ------------------------------------------------------------------
    # initialise
    a = initial_sequence(n)                  # on the simplex, sum = 1
    best_a = a.copy()
    best_val = fast_score(a)
    last_improve = time.time()

    # pre‑compute FFT size used repeatedly
    fft_size = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1. current bound M
        fft_a = np.fft.rfft(a, fft_size)
        conv_a = np.fft.irfft(fft_a * fft_a, fft_size)[: 2 * n - 1]
        M = float(conv_a.max())
        cur_val = 2.0 * n * M

        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[iter {it:5d}] new best {best_val:.6f}")

        # --------------------------------------------------------------
        # 2. solve the LP to obtain a direction g
        try:
            g_raw = solve_lp_max_sum(a, M)
        except RuntimeError as e:
            print(f"[iter {it}] LP error: {e}")
            break

        sum_g = g_raw.sum()
        if sum_g <= 0:
            print("[iter] zero sum_g – terminating")
            break

        # normalise g to the simplex
        g = g_raw / sum_g

        # --------------------------------------------------------------
        # 3. mixing – Frank‑Wolfe step
        #   start with a relatively large step, decay slowly
        t = max(0.1, 0.5 * (0.97 ** (it // 10)))
        a_next = (1.0 - t) * a + t * g

        # avoid exact zeros (LP would become unbounded next round)
        eps = 1e-12
        a_next = np.maximum(a_next, eps)
        a_next /= a_next.sum()

        # --------------------------------------------------------------
        # 4. evaluate candidate
        fft_next = np.fft.rfft(a_next, fft_size)
        conv_next = np.fft.irfft(fft_next * fft_next, fft_size)[: 2 * n - 1]
        M_next = float(conv_next.max())
        val_next = 2.0 * n * M_next

        # accept if we have an improvement, otherwise keep the old iterate
        if val_next < cur_val:
            a = a_next
        else:
            # occasional uphill move to escape shallow plateaus
            if np.random.rand() < 0.001:
                a = a_next

        # --------------------------------------------------------------
        # 5. stagnation handling – re‑seed from best if stuck
        if time.time() - last_improve > 30.0:
            # small random perturbation of the current best
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = np.maximum(0.9 * best_a + 0.1 * noise, eps)
            a /= a.sum()
            last_improve = time.time()
            print(f"[iter {it}] re‑seeded from best")

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    return best_a.tolist()
