def exponentiated_gradient_step(a, grad, eta):
    a = a * np.exp(-eta * grad)
    sum_a = a.sum()
    if sum_a > 0:
        a /= sum_a
    else:
        a = np.full_like(a, 1.0 / len(a))

"""
Propose a step‑function (non‑negative list) that minimises `evaluate_sequence`.

Algorithm Overview
------------------
1. **Multiple restarts** – we start from the previous best construction (if
   available) and from a few random Dirichlet draws. Each start is processed
   independently; the best overall result is kept.

2. **Coarse optimisation (projected gradient)** – a short projected‑gradient
   descent on a modest size (n≈800) quickly discovers a good non‑uniform shape.

3. **Upsample & smooth optimisation** – the best coarse shape is linearly
   interpolated to a larger size (n≈2400).  A *smooth* Exponentiated‑Gradient
   descent is performed with a temperature‑controlled soft‑max of the
   convolution (λ increases from 0.1 to ~30).  This provides a dense,
   well‑behaved gradient that helps escape shallow local minima.

4. **Fine projected‑gradient refinement** – after the smooth phase we switch
   back to the exact sub‑gradient of the true max‑convolution and run a longer
   projected‑gradient descent with Nesterov momentum.

5. **Local stochastic refinement (annealing)** – a cheap simulated‑annealing
   walk (mass‑transfer moves) uses the remaining time budget to squeeze out
   any further improvement.

All operations respect the simplex constraint (non‑negative, unit sum), so
the objective `2·n·max(a*a)` is exactly the value returned by
`evaluate_sequence` (the clipping logic never activates).  The code is
self‑contained, uses only NumPy and the standard library, and fits comfortably
within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n) algorithm)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient from all active convolution entries (max‑set)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Average sub‑gradient of the max‑convolution (active set)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if act.size > 0:
        g /= float(act.size)
    return g


# ----------------------------------------------------------------------
# Projected (Nesterov‑style) gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_gradient_descent_full(a0: np.ndarray, n: int,
                              deadline: float, start_time: float,
                              beta: float = 0.9,
                              step0: float = 0.6,
                              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with Nesterov momentum using the exact
    max‑convolution sub‑gradient.
    Returns (best_point, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: n={n}, iter={it}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing local refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 150000) -> tuple[np.ndarray, float]:
    """Simple annealing walk that transfers a small ε from one index to another."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear interpolation (upsampling) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        out = seq.copy()
    else:
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, new_n)
        out = np.interp(new_x, old_x, seq)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.full_like(out, 1.0 / new_n)
    return out

# Smooth Exponentiated‑Gradient optimisation (soft‑max of the convolution)
# ----------------------------------------------------------------------
def _run_smooth_eg(a0: np.ndarray, n: int,

                   lam_start: float = 0.1,
                   lam_end: float = 30.0,
                   lam_steps: int = 18,
                   eta0: float = 0.6,
                   max_iter_per_lam: int = 4000) -> tuple[np.ndarray, float]:
    """
    Smooth optimisation using a soft‑max of the convolution.
    The temperature λ is increased geometrically; for each λ we perform a
    limited number of exponentiated‑gradient steps.
    Returns the best point encountered and its (true) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    lam_vals = np.geomspace(lam_start, lam_end, num=lam_steps)

    for lam in lam_vals:
        for it in range(1, max_iter_per_lam + 1):
            if time.time() >= deadline:
                return best_a, best_score

            conv = _conv_self(a)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:

            s = a.sum()

            else:

        remaining = deadline - time.time()
        smooth_deadline = time.time() + max(5.0, 0.55 * remaining)
        a_smooth, score_smooth = _run_smooth_eg(
            a0_large, n_large, smooth_deadline, start_time,
            lam_start=0.1, lam_end=30.0,
            lam_steps=20, eta0=0.5, max_iter_per_lam=3000
        )
        _update_best(a_smooth)

        # ---- Stage 3: fine projected gradient descent -----------------------
        if time.time() >= deadline:
            return
        remaining = deadline - time.time()
        grad_deadline = time.time() + max(5.0, 0.30 * remaining)
        a_fine, score_fine = _run_gradient_descent_full(
            a_smooth, n_large, grad_deadline, start_time,
            beta=0.9, step0=0.4, max_iter=10**7
        )
        _update_best(a_fine)

        # ---- Stage 4: simulated‑annealing refinement ------------------------
        if time.time() >= deadline:
            return
        a_anneal, score_anneal = _run_annealing(
            best_seq if best_seq is not None else a_fine,
            n_large,
            deadline,
            start_time,
            best_score,
            best_seq if best_seq is not None else a_fine,
            start_temp=5e-2,
            end_temp=1e-5,
            steps=150000
        )
        _update_best(a_anneal)

    # ------------------------------------------------------------------
    # Try to start from the previously discovered construction, if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = np.full_like(starter, 1.0 / starter.size)
            else:
                starter = starter / starter.sum()
            _pipeline(starter)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Additional random restarts while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # Dirichlet with concentration <1 encourages sparsity (often useful)
        size = 800
        rand_vec = rng.gamma(0.5, 1.0, size)
        if rand_vec.sum() == 0:
            rand_vec = np.full(size, 1.0 / size)
        else:
            rand_vec = rand_vec / rand_vec.sum()
        _pipeline(rand_vec)

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case we never found anything
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (protect against tiny drift)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
