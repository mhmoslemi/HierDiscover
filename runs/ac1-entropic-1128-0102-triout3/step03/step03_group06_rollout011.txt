"""
Propose a low‑score step‑function.

The search proceeds in three stages:

1. **Random support sampling** – for several support sizes `k` we draw many
   random subsets of `{0,…,n‑1}` and evaluate the uniform distribution on the
   set.  The evaluation uses a fast FFT‑based routine (identical to the
   provided `evaluate_sequence`).  This stage often finds a candidate already
   below the target 1.503.

2. **Hill‑climbing on the support** – if random sampling is not enough we
   start from the best support found and perform a stochastic local‑search.
   The move is a single swap: remove one index i∈S and insert a fresh index
   j∉S.  The ordered‑pair count array `counts` (the self‑convolution of the
   binary indicator of S) is updated incrementally with `np.add.at`, so each
   swap costs O(k).  Swaps that do not increase the maximal count are always
   accepted; occasional uphill moves help escape plateaus.

3. **Continuous refinement** – finally a short projected sub‑gradient descent
   (the same routine used previously) refines the real coefficients while
   keeping the simplex constraint ∑aᵢ=1.  The remaining time budget is used
   for this step.

The algorithm respects the 1000 s time limit, never writes to disk and
returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the scoring function (identical to `evaluate_sequence`)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """
    Ordered‑pair counts for a support set S (ordered sums, length 2*n-1).
    S must be a 1‑D array of integers.
    """
    # Outer sum yields all i+j (ordered pairs)
    sums = np.add.outer(S, S)                     # shape (k,k)
    # bincount works on flattened array; dtype int64 is safe.
    counts = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return counts

# ----------------------------------------------------------------------
def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent for the original objective."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    eta0 = 0.30
    eta_decay = 1e-5
    random_move_prob = 0.02
    random_move_step = 0.15
    it = 0
    deadline = time.time() + time_limit - 0.2

    while time.time() < deadline:
        # self‑convolution
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active index
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        rho = ind[cond][-1] if np.any(cond) else 0
        theta = cssv[cond][-1] / rho if rho > 0 else 0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass‑transfer move
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_step * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # re‑project
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                rho = ind[cond][-1] if np.any(cond) else 0
                theta = cssv[cond][-1] / rho if rho > 0 else 0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a

            np.add.at(new_counts, add_sums, 2)
            new_counts[new_idx + new_idx] += 1

            new_max = int(new_counts.max())

            # accept if we do not increase the maximal count
            if new_max <= max_c:
                # apply the swap permanently
                S[i_pos] = new_idx
                in_S[old_idx] = False
                in_S[new_idx] = True
                counts = new_counts
                max_c = new_max

                cur_score = 2.0 * n * max_c / (k * k)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = np.zeros(n, dtype=np.float64)
                    best_seq[S] = 1.0 / k
                    print(f"[hillclimb] iter {it:6d} improved score {cur_score:.6f}")

                if cur_score <= target_score:
                    print("[hillclimb] target reached!")
                    return best_seq.tolist()
            else:
                # very occasional uphill move to avoid being stuck
                if np.random.rand() < 0.001:
                    S[i_pos] = new_idx
                    in_S[old_idx] = False
                    in_S[new_idx] = True
                    counts = new_counts
                    max_c = new_max
            it += 1

    # ------------------------------------------------------------------
    # 3) Final continuous refinement with sub‑gradient descent (if time permits)
    remaining = deadline - time.time()
    if remaining > 5.0 and best_seq is not None:
        print(f"[refine] running sub‑gradient for ~{remaining:.1f}s")
        refined = subgradient_optimize(best_seq.copy(), remaining)
        refined_val = fast_score(refined)
        if refined_val < best_score:
            best_score = refined_val
            best_seq = refined
            print(f"[refine] improved to {refined_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – uniform distribution (score = 2) should never be reached,
    # but we keep it for completeness.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = fast_score(best_seq)

    print(f"[final] best score returned: {best_score:.6f}")
    return best_seq.tolist()
