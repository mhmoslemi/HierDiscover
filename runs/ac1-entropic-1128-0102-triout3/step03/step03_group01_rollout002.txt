def _run_adam_descent(a0, n, deadline, start_time):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_lr = 0.05 # learning rate
    lam_factor = 0.2  (some initial) maybe 0.2
    lam_decay = 0.7
    lam_decay_interval = 5000

    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # schedule lambda
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-10)
        lam = max(max_b * lam_factor, 1e-12)

        # compute gradients for log-sum-exp surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1*m + (1 - beta1)*grad
        v = beta2*v + (1 - beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # symmetrize
        a = (a + a[::-1]) / 2.0
        a = _project_simplex(a)

        # ensure not all zeros? but solver ensures sum 1.

    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes
`evaluate_sequence`.  

The algorithm runs a multi‑scale optimisation:

* start from a (possibly provided) distribution and up‑sample it
  through a schedule of lengths (800 → 1600 → … → 12800);
* on each scale run a fast Adam optimiser on a smooth
  log‑sum‑exp surrogate of the ℓ∞‑norm of the self‑convolution;
* then switch to a pure sub‑gradient descent that directly minimises
  the peak of the convolution;
* finally polish the best vector with a cheap simulated‑annealing
  mass‑transfer sweep.

All updates are projected onto the probability simplex
(∑aᵢ = 1, aᵢ ≥ 0) so the objective reduces to 2·n·max(a∗a).
The routine respects the 1000 s budget and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (real FFT)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate Lλ(a) of max(a∗a).
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate (projected onto simplex)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    step_lr = 0.12                 # learning‑rate scaling
    lam_factor = 0.25              # initial λ = lam_factor * max
    lam_decay = 0.5
    lam_decay_interval = 2_000
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        # λ schedule
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-10)
        lam = max(max_b * lam_factor, 1e-12)

        # gradient of the smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # update and projection
        step = step_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


def _run_subgrad_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent directly on max(a∗a).  At each iteration we
    compute the index of the maximal convolution entry and move a
    little mass away from the most responsible coordinates.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    step0 = 0.18

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        max_idx = conv.argmax()

        # build sub‑gradient of the max entry
        g = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        jdx = max_idx - idx
        diff = idx != jdx
        g[idx[diff]] = 2.0 * a[jdx[diff]]
        g[idx[~diff]] = a[idx[~diff]]

        # diminishing step size
        step = step0 / math.sqrt(it)
        a = a - step * g
        a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing: transfer a small ε from i to j.
    The objective is computed as 2·n·max(conv) (sum is 1).
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a mass move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # evaluate new candidate
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

            sub_deadline = time.time() + max(4.0, 0.10 * (deadline - time.time()))
            a_sub, score_sub = _run_subgrad_descent(a_opt, n, sub_deadline, start_time)
            if score_sub < best_score:
                best_score = score_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] Global improve (Subgrad) n={n} score={score_sub:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – cheap simulated‑annealing polishing
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(3.0, 0.07 * (deadline - time.time()))
            a_anneal, score_anneal = _run_annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-6,
                steps=60_000,
            )
            if score_anneal < best_score:
                best_score = score_anneal
                best_seq = a_anneal.copy()
                print(f"[{time.time() - start_time:6.2f}s] Global improve (Anneal) n={n} score={score_anneal:.6f}")

        # keep the current best as warm‑start for the next (larger) scale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no viable vector was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n)

    # Return a plain list of Python floats (required by the harness)
    return [float(x) for x in best_seq]
