def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    rng = np.random.default_rng(seed)
    # choose n
    n = 4200  # or 4000

    # initialize a
    init = None
    hs1 = globals().get('height_sequence_1')
    if hs1 is not None and isinstance(hs1, (list, np.ndarray)):
        init_arr = np.array(hs1, dtype=np.float64)
        if init_arr.size <= n:
            a = np.zeros(n)
            a[:init_arr.size] = init_arr
        else:
            a = init_arr[:n]  # truncated
        # ensure sum=1
        a = project_onto_simplex(a)
    else:
        a = half_binomial_series(n)  # already sum to 1

    # Precalc sizes
    L = 2 * n - 1
    Nfft = next_pow_two(3 * n - 2)

    # Adam params
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.4  # base learning rate

    # softmax param
    beta = 1.0
    beta_target = 4000.0
    beta_inc = 1.004

    best_a = a.copy()
    conv_full = np.fft.irfft(np.fft.rfft(a, n=Nfft) * np.fft.rfft(a, n=Nfft), n=Nfft)
    max_conv = conv_full[:L].max()
    best_score = 2.0 * n * max_conv
    last_improve = 0
    restart_gap = 200_000
    noise_prob = 0.004
    noise_scale = 5e-5

    deadline = time.time() + budget_s - 5.0
    it = 0

    while time.time() < deadline:
        # compute convolution
        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it}] best score {best_score:.6f}")

        # compute softmax weights
        shift = max_conv
        w_exp = np.exp(beta * (conv - shift))
        w_sum = w_exp.sum()
        if w_sum == 0:
            w = np.ones(L) / L
        else:
            w = w_exp / w_sum

        # gradient via fft
        Wr = np.fft.rfft(w[::-1], n=Nfft)
        corr_full = np.fft.irfft(A * Wr, n=Nfft)
        start = L - 1
        grad = 2.0 * corr_full[start:start + n]

        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_trial = a - step

        # optional noise
        if rng.random() < noise_prob:
            noise = (rng.random(n) - 0.5) * noise_scale
            a_trial = a_trial + noise

        # projection
        a = project_onto_simplex(a_trial)

        # schedule
        beta = min(beta_target, beta * beta_inc)
        lr *= 0.99995

        if it - last_improve > restart_gap:
            # random restart
            a = rng.dirichlet(np.full(n, 0.8))
            a = project_onto_simplex(a)
            m.fill(0)
            v.fill(0)
            last_improve = it
            print(f"[iter {it}] restart, current score {2.0*n*np.max(np.fft.irfft(np.fft.rfft(a,n=Nfft)**2,n=Nfft)[:L]):.6f}")

        it += 1

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] best score {final_score:.6f}")

    return best_a.tolist()

"""
Projected Adam optimiser for the auto‑convolution peak.

The algorithm works on the probability simplex (∑ a_i = 1, a_i ≥ 0).

* Initialise a (length *n*) from the best known construction if it is
  available, otherwise from the half‑binomial series (a good starting
  point).

* Repeatedly compute the auto‑convolution by FFT, build a soft‑max
  weighting of the convolution entries (β gradually grows, turning the
  smooth weighting into a true max), and obtain the gradient via a
  correlation (also FFT‑based).

* Use Adam (adaptive moment estimation) to take a step, then project
  the iterate back onto the simplex.

* Occasionally add a tiny random perturbation and restart from a fresh
  Dirichlet sample if no improvement has been seen for a long time.

* Keep the best sequence seen during the whole run and return it at the
  end.  The returned list can be fed directly to ``evaluate_sequence``.
"""

import time
from typing import List

import numpy as np


def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^‑½  (a_i = C(2i,i)/4^i) normalised
    so that they sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def propose_candidate(seed: int = 42,
                     budget_s: float = 950.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises the evaluation
    function (max auto‑convolution).  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size -------------------------------------------------------
    # A length of a few thousand gives a good trade‑off between flexibility
    # and speed.  Feel free to change it if you have a different budget.
    n = 4000
    L = 2 * n - 1                     # length of a*a (linear convolution)
    Nfft = next_pow_two(3 * n - 2)    # size for zero‑padded FFTs (≥ n+L‑1)

    # ------------------------------------------------------------------
    # Initialise a -------------------------------------------------------
    # Prefer the already‑found construction if it exists.
    hs1 = globals().get('height_sequence_1')
    if isinstance(hs1, (list, np.ndarray)):
        hs_arr = np.asarray(hs1, dtype=np.float64)
        if hs_arr.size >= n:
            a = hs_arr[:n].copy()
        else:
            a = np.zeros(n, dtype=np.float64)
            a[:hs_arr.size] = hs_arr
    else:
        a = half_binomial_series(n)

    # Ensure feasibility (sum‑to‑one simplex)
    a = project_onto_simplex(a)

    # ------------------------------------------------------------------
    # Adam hyper‑parameters ------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.07               # base learning rate
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    # Soft‑max (log‑sum‑exp) parameters for smoothing the max
    beta = 1.0
    beta_target = 4000.0
    beta_inc = 1.004

    # Restart & noise control
    last_improve = 0
    restart_gap = 250_000          # iterations without improvement → restart
    noise_prob = 0.005
    noise_scale = 6e-5

    # ------------------------------------------------------------------
    # Tracking best solution ---------------------------------------------
    conv_full = np.fft.irfft(
        np.fft.rfft(a, n=Nfft) * np.fft.rfft(a, n=Nfft), n=Nfft
    )
    max_conv = conv_full[:L].max()
    best_score = float(2.0 * n * max_conv)
    best_a = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop ---------------------------------------------
    deadline = time.time() + budget_s - 5.0   # safety margin
    it = 0
    while time.time() < deadline:
        # ----- 1) convolution ------------------------------------------------
        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]                     # linear part only
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ----- 2) keep the best ---------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:09d}] new best score = {best_score:.6f}")

        # ----- 3) soft‑max weights (smooth approximation of max) ------------
        shift = max_conv
        w_exp = np.exp(beta * (conv - shift))   # all ≤ 1, numerically safe
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full(L, 1.0 / L, dtype=np.float64)
        else:
            w = w_exp / w_sum                     # shape (L,)

        # ----- 4) gradient of the smoothed objective ------------------------
        # gradient = 2 * correlation(a, w)
        Wr = np.fft.rfft(w[::-1], n=Nfft)
        corr_full = np.fft.irfft(A * Wr, n=Nfft)
        start = L - 1
        grad = 2.0 * corr_full[start:start + n]

        # ----- 5) Adam update ------------------------------------------------
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- 6) stochastic perturbation (helps escaping flat regions) -----
        if rng.random() < noise_prob:
            noise = (rng.random(n) - 0.5) * noise_scale
            step = step - noise   # move a little in a random direction

        # ----- 7) projected gradient step -----------------------------------
        a_trial = a - step
        a = project_onto_simplex(a_trial)

        # ----- 8) schedules ---------------------------------------------------
        beta = min(beta_target, beta * beta_inc)        # sharpen soft‑max
        lr *= 0.99995                                   # gentle decay

        # ----- 9) restart logic -----------------------------------------------
        if it - last_improve > restart_gap:
            # fresh random start from a Dirichlet with moderate concentration
            a = rng.dirichlet(np.full(n, 0.7))
            a = project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            # quick diagnostic score of the restart point
            A_tmp = np.fft.rfft(a, n=Nfft)
            conv_tmp = np.fft.irfft(A_tmp * A_tmp, n=Nfft)[:L]
            print(f"[iter {it:09d}] restart – score≈{2.0 * n * conv_tmp.max():.6f}")

        it += 1

    # ----------------------------------------------------------------------
    # Final evaluation with the provided routine (clamps etc.) -------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    # Return as a plain Python list
    return best_a.tolist()
