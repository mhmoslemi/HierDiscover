import numpy as np
import time
import math

def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence without security checks."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """Return ordered pair counts for all sums using support set S."""
    # Use int64 to avoid overflow
    sums = np.add.outer(S, S)
    # Convert to 1D and bin count
    cnt = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return cnt

def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Fallback: simple projected sub‑gradient descent."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    random_move_prob = 0.02
    random_move_step = 0.15
    it = 0
    deadline = time.time() + time_limit - 0.2
    ind = np.arange(1, n + 1)  # reused in projection
    while time.time() < deadline:
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        # Euclidean projection onto simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)
        # occasional random mass‑transfer
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_step * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # projection again
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand
        if it % 2000 == 0:

            if cur_val < best_val:

                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")
        it += 1

    Primary approach: random construction of a sparse support set S.
    For a given support size k we compute the maximal ordered‑pair count.
    If max ≤ allowed = floor(target * k^2 / (2 n)) the uniform‑weight
    vector on S already satisfies the target bound.

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Fast version of the public evaluate_sequence (without the security checks)
def fast_score(a: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative vector a (L1‑norm = 1)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())          # power‑of‑two for FFT
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """
    For a support set S (array of indices) compute the ordered‑pair count
    for each possible convolution index (0 … 2·n‑2).
    Returns a length‑(2·n‑1) integer array.
    """
    # all pairwise sums (ordered, i.e. (i,j) and (j,i) are counted)
    sums = np.add.outer(S, S)                     # shape (|S|, |S|)
    cnt = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return cnt


# ----------------------------------------------------------------------
def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Simple projected sub‑gradient descent (fallback).
    Starts from a feasible a_init (already L1‑normalised) and runs until
    ``time_limit`` seconds have elapsed, returning the best vector seen.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    eta0 = 0.30
    eta_decay = 1e-5
    random_move_prob = 0.02
    random_move_step = 0.15

    it = 0
    deadline = time.time() + time_limit - 0.2
    ind = np.arange(1, n + 1)                    # used in the simplex projection

    while time.time() < deadline:
        # ---- self‑convolution and the index where it is maximal ----
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # ---- sub‑gradient of the active entry ----
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # ---- step size ----
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # ---- Euclidean projection onto the probability simplex ----
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # ---- occasional random mass‑transfer move (helps escaping plateaus) ----
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_step * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta

                # projection again
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)

                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # ---- bookkeeping ----
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with evaluation score ≤ 1.503.
    Primary strategy: random construction of a sparse support set S.
    For a chosen support size k we compute the maximal ordered‑pair count.
    If that maximum does not exceed

            allowed = ⌊target·k² / (2·n)⌋

    the uniform‑weight vector on S already satisfies the target bound.
    We try a descending list of k values (larger k → looser bound → easier
    to satisfy).  If no construction succeeds within the time budget we
    fall back to a simple projected sub‑gradient optimiser.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    n = 1500                                   # length of the step function
    target_score = 1.503

    # Larger k gives a larger allowed count and is therefore easier to meet.
    support_sizes = [800, 700, 600, 500, 400, 300, 250, 200,
                    150, 120, 100, 80]

    best_seq = None
    best_val = np.inf

    for k in support_sizes:
        allowed = int(math.floor(target_score * k * k / (2.0 * n)))
        allowed = max(1, allowed)               # avoid zero
        print(f"[search] Trying k={k}, allowed max count={allowed}")

        # Reserve at most 30 % of the remaining time (capped at 40 s)
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        time_for_k = min(40.0, 0.3 * remaining)

        attempts = 0
        local_deadline = time.time() + time_for_k
        while time.time() < local_deadline:
            attempts += 1
            # Randomly draw a support set of size k
            S = np.random.choice(n, k, replace=False)
            cnt = compute_counts(S, n)
            max_c = int(cnt.max())

            if max_c <= allowed:
                # Uniform weights on the support
                a = np.zeros(n, dtype=np.float64)
                a[S] = 1.0 / k
                val = fast_score(a)
                print(f"[found] k={k}, attempts={attempts}, max_c={max_c}, score={val:.6f}")
                if val <= target_score:
                    print("[success] target met, returning this sequence.")
                    return a.tolist()
                # keep it as a candidate even if a little above target
                if val < best_val:
                    best_val, best_seq = val, a.copy()

            if attempts % 1000 == 0:
                print(f"[search] k={k}, attempts={attempts}, best so far {best_val:.6f}")

        print(f"[search] Finished k={k} after {attempts} attempts, best so far {best_val:.6f}")

        if best_seq is not None and best_val <= target_score:
            break

    # ------------------------------------------------------------------
    # Fallback: if we never found a satisfactory support set, run the
    # sub‑gradient optimiser starting from a simple power‑law vector.
    if best_seq is None:
        print("[fallback] No random support succeeded, invoking sub‑gradient optimiser.")
        init = 1.0 / np.arange(1, n + 1, dtype=np.float64) ** 0.5
        init /= init.sum()
        best_seq = subgradient_optimize(init, deadline - time.time())
        best_val = fast_score(best_seq)

    # Ultimate safeguard – uniform distribution (score = 2) should never be needed.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    print(f"[final] best value returned: {best_val:.6f}")
    return best_seq.tolist()
