"""
Propose a low‑maximum self‑convolution step‑function.

The optimizer works on the probability simplex (∑aᵢ = 1) because the
evaluation function is scale‑invariant.  It combines three ideas:

1. **Smoothed log‑sum‑exp gradient descent** – the ℓ∞ norm of the
   convolution is approximated by a differentiable surrogate.
   A decreasing smoothing parameter λ and momentum give fast progress.
2. **Multi‑scale schedule** – the current best distribution is up‑sampled
   (linear interpolation) to a larger size and used as a warm‑start.
   This lets the algorithm exploit the extra degrees of freedom of longer
   sequences without restarting from scratch.
3. **Targeted simulated annealing** – after each gradient phase a cheap
   mass‑transfer annealing is run, moving a small ε from a source index to
   a destination index.  The temperature schedule is aggressive at the
   beginning and cools to 1e‑5, polishing the best vector found.

The code respects the 1000 s budget, allocates the remaining time
between the stages and prints occasional progress messages.
The returned list is a plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper utilities – all top‑level (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max over all convolution entries – numerically stable
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of the weight vector with the reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Smoothed gradient descent on the simplex
# ----------------------------------------------------------------------


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        slice_deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Smoothed log‑sum‑exp gradient descent with momentum.
    Returns the best vector encountered and its exact objective value
    (2·n·max(conv) because the vector is kept on the simplex).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 0.8

    # λ‑schedule: start relatively large, halve every few thousand steps
    lam_factor = 0.30
    lam_decay = 0.65
    lam_decay_interval = 2500

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b               # exact objective (sum(a)=1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve n={n} it={it} score={cur_score:.6f}")

        # update λ (never let it become exactly zero)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        # surrogate gradient
        grad = _gradient_from_conv(a, conv, lam)

        # add a tiny isotropic noise to help escape shallow plateaus
        if it % 500 == 0:
            noise_scale = 0.02 / math.sqrt(it)
            grad += noise_scale * np.random.randn(n)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 50000) -> tuple[np.ndarray, float]:
    """
    Very cheap annealing: move a small ε from i to j.
    Returns the (possibly) improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of the source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

    return [float(x) for x in best_seq]
