    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    # load init from global height_sequence_1 else random of modest length (2000)
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0)
        s = init.sum()
        if s == 0:
            init = np.random.rand(2000)
            init /= init.sum()
        else:
            init = init / s
        best_seq = init.copy()
    except NameError:
        n0 = 2000
        init = np.random.rand(n0)
        init = init / init.sum()
        best_seq = init.copy()

    best_score = _score_of_seq(best_seq)
    print(f"Initial score {best_score:.6f}, n={len(best_seq)}")

    # Optionally enlarge to longer length (e.g., 8000) using interpolation.
    # Use progressive upsampling gradually.

    # We'll use function to upsample to target length while keeping normalized.
    def upsample(seq, target_n):
        cur_n = seq.shape[0]
        if target_n == cur_n:
            return seq
        x_old = np.linspace(0, 1, cur_n)
        x_new = np.linspace(0, 1, target_n)
        up = np.interp(x_new, x_old, seq)
        up = np.maximum(up, 0)
        up /= up.sum()
        return up

    # First ensure length moderately large for better shape
    target_len = 8000
    if best_seq.shape[0] < target_len:
        best_seq = upsample(best_seq, target_len)
        best_score = _score_of_seq(best_seq)
        print(f"Upsampled to {target_len}, score {best_score:.6f}")

    a = best_seq.copy()
    n = a.shape[0]

    # Simulated annealing parameters
    T0 = 0.01 * best_score  # initial temperature fraction of score
    Tmin = 1e-7
    total_iters = 0
    max_iters = 2000000  # upper bound but will stop by deadline

    # schedule: T = T0 * (Tmin/T0)^(it/total_iters) approximated adaptively.
    # We'll compute exponent factor based on ratio to keep schedule.
    # We'll reduce temperature every 1000 iterations.

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # jitter temperature, schedule linear cooling:
        if it % 1000 == 0:
            # compute fraction of total allowed iterations based on remaining time
            elapsed = time.time() - start
            remaining = deadline - time.time()
            # linear schedule from T0 to Tmin across total budget:
            T = max(Tmin, T0 * (1 - elapsed / (budget_s - 2)))
        else:
            T = T

        # propose move
        # choose type: with probability 0.7 move piece of mass; 0.3 swap masses
        if np.random.rand() < 0.7:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move fraction up to 0.5 of source mass
            eps = np.random.rand() * 0.5
            delta = eps * a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # swap masses
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]

        # Ensure nonnegative due to rounding
        if np.any(a_new < -1e-12):
            continue

        # Compute score
        cur_score = _score_of_seq(a_new)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a_new.copy()
            a = a_new
            print(f"[{time.time()-start:.2f}s] New best {best_score:.6f} iter {it}")

            if T > 0 and np.random.rand() < math.exp(-(cur_score - best_score) / T):
                a = a_new

    print(f"Finished after {it} iterations, best_score={best_score:.6f}")

import numpy as np
import time, math

def _conv_max(a):
    n = a.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(Af * Af, n=N)[:2*n-1]
    return conv, conv.max()

def _score_of_seq(a):
    n = a.shape[0]
    _, max_val = _conv_max(a)
    return float(2.0 * n * max_val)

def _active_set(conv, max_val, rel_eps=0.001, max_add=32):
    eps = max_val * rel_eps
    idx = np.where(conv >= max_val - eps)[0]
    if idx.size < max_add:
        topk = min(max_add, conv.size)
        idx = np.unique(np.argpartition(conv, -topk)[-topk:])
    return idx

def _gradient_active(a, K):
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        # add 2*a[j_idx] to g[i_idx]
        g[i_idx] += 2.0 * a[j_idx]
    return g

def _upsample(seq, target_n):
    cur_n = seq.shape[0]
    if target_n == cur_n:
        return seq
    x_old = np.linspace(0.0, 1.0, cur_n)
    x_new = np.linspace(0.0, 1.0, target_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    tot = up.sum()
    if tot <= 0:
        up = np.full(target_n, 1.0/target_n)
    else:
        up /= tot
    return up

    # ------------------------------------------------------------------

        n = a.shape[0]
        # Pre‑compute initial score

        T = cur_score * 0.02   # start temperature at 2 % of current score

        decay = 0.99995  # multiplicative decay per iteration
        it_local = 0

"""
search for a non‑negative step function that minimizes

        2·n·max(conv(a,a))

where a is normalised (sum(a)=1).  The method consists of

1. initialise from the globally supplied ``height_sequence_1`` (or random);
2. progressively up‑sample to a length of up to 8000 while applying a
   mirror‑descent (exponentiated‑gradient) optimiser on the simplex.
   – an active set of the largest convolution entries supplies a sparse
     sub‑gradient;
   – a momentum term and a slowly decaying step‑size improve stability;
3. a long simulated‑annealing / hill‑climbing phase that moves small
   fractions of mass between positions (up‑hill moves are accepted
   according to a temperature schedule);
4. return the best sequence seen (as plain Python floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# convolution helpers (FFT – O(n log n))
def _conv_max(a: np.ndarray):
    """Return full auto‑convolution of a and its maximum value."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(Af * Af, n=N)[: 2 * n - 1]
    return conv, conv.max()


def _score_of_seq(a: np.ndarray) -> float:
    """Objective for a normalised vector a (sum(a)=1)."""
    n = a.shape[0]
    _, mx = _conv_max(a)
    return float(2.0 * n * mx)


# ----------------------------------------------------------------------
def _active_set(conv: np.ndarray, max_val: float,
                rel_eps: float = 0.001, max_add: int = 32):
    """
    Return a modest set of indices that are “near‑max”.
    All indices k with conv[k] ≥ max_val·(1‑rel_eps) are included.
    If fewer than ``max_add`` indices are collected, the remaining slots are
    filled with the globally largest entries.
    """
    eps = max_val * rel_eps
    idx = np.where(conv >= max_val - eps)[0]
    if idx.size < max_add:
        topk = min(max_add, conv.size)
        idx = np.unique(np.argpartition(conv, -topk)[-topk:])
    return idx


def _gradient_active(a: np.ndarray, K: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) using the active set ``K``.
    For each active convolution index k,
        ∂conv_k/∂a_i = 2·a_{k‑i}  (if the index is in range).
    The contributions are accumulated into a gradient vector.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]        # 2·a_{k‑i}
    return g


def _upsample(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation up‑sampling to ``target_n`` points (keeps sum=1)."""
    cur_n = seq.shape[0]
    if target_n == cur_n:
        return seq
    x_old = np.linspace(0.0, 1.0, cur_n)
    x_new = np.linspace(0.0, 1.0, target_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up = np.full(target_n, 1.0 / target_n)
    else:
        up /= s
    return up
