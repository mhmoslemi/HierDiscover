"""
Propose a low‑value step‑function sequence.

Algorithm
---------

* Initialise many candidates (power‑law, linear, uniform, random Dirichlet,
  plus the previous best if available) and keep the one with the smallest
  fast_score (i.e. smallest 2·n·max(a∗a) for sum(a)=1).

* Optimise the candidate on the probability simplex (a_i ≥ 0, Σa_i = 1)
  using a projected Adam update.  The loss is a smoothed version of the
  max‑convolution:
        L(a) = (1/λ)·log Σ_k exp(λ·(a∗a)_k) .
  For λ → ∞ this equals the true max.  λ is increased linearly during
  the run.

* The gradient   ∂L/∂a = 2·(a ⋆ w)   where w is the soft‑max weight
  vector on the convolution entries and “⋆” denotes cross‑correlation.
  Both the convolution a∗a and the cross‑correlation are obtained by
  FFTs, giving an O(n log n) per iteration cost.

* Random “mass‑transfer’’ moves (moving a fraction of mass from one index
  to another) are interleaved, and after 30 s without improvement the
  current iterate is re‑initialised as a slight perturbation of the best
  solution found so far.

The routine runs for at most the supplied time budget (≈1000 s) and
returns the best sequence (as a Python list of floats) discovered.

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` (no safety checks).
    Assumes a is a non‑negative 1‑D array on the simplex (sum(a)=1).
    Returns 2·n·max(a∗a), which equals the official score for sum=1.
    """
    n = a.size
    # size for a linear convolution a*a (length 2n‑1)
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n-i) for i=0..n‑1, normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation value.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size (the same size that gave the previous best)
    n = 3000

    # Pre‑computed FFT size for all convolutions (a*a and a⋆w)
    conv_size = 1 << ((3 * n - 2).bit_length())   # ≥ n + (2n‑1) – 1

    # --------------------------------------------------------------
    # 1. Build a diverse pool of analytically motivated initialisations
    candidates = []

    # power‑law family
    exponents = np.linspace(0.20, 0.60, 9)          # 0.20 … 0.60
    offsets   = [0.0, 0.1, 0.5, 1.0, 2.0]
    for p in exponents:
        for off in offsets:
            candidates.append(build_power_law(n, float(p), float(off)))

    # linear decreasing shape
    candidates.append(build_linear_descending(n))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # a few random Dirichlet samples (different concentration levels)
    for _ in range(30):
        alpha = np.random.uniform(0.05, 5.0)
        cand = np.random.dirichlet(np.full(n, alpha))
        candidates.append(cand)

    # previous best (if available)
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                seq = np.zeros(n, dtype=np.float64)
                seq[:prev.size] = prev
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0.0:
                seq /= seq.sum()
                candidates.append(seq)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2. Pick the best initial candidate (according to the fast score)
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best initial value: {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Optimisation loop (projected Adam + random moves)
    a = best_seq.copy()               # current iterate (on the simplex)
    best_seq = a.copy()
    best_val = fast_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05          # base learning rate

    beta1_pow = 1.0
    beta2_pow = 1.0

    # smoothing parameter schedule for log‑sum‑exp
    lam_start = 0.5
    lam_end = 80.0
    max_iters_est = 500000    # used only for λ scheduling

    # stochastic mass‑transfer controls
    random_move_prob = 0.015
    random_move_frac = 0.15

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ----------------------------------------------------------
        # (a) self‑convolution and its current max
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = time.time()

        # ----------------------------------------------------------
        # (b) smooth max (log‑sum‑exp) and its gradient
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iters_est)

        # soft‑max weights on the convolution entries
        shift = max_conv                         # for numerical stability
        exp_vals = np.exp(lam * (conv - shift))
        w = exp_vals / exp_vals.sum()           # Σ w = 1

        # cross‑correlation a ⋆ w = gradient (up to factor 2)
        w_rev = w[::-1]                         # reversed weight vector
        fft_w_rev = np.fft.rfft(w_rev, conv_size)
        grad_full = np.fft.irfft(fft_a * fft_w_rev, conv_size)

        L = 2 * n - 1
        idx = (L - 1) - np.arange(n)            # indices n‑1 … 2n‑2
        grad = 2.0 * grad_full[idx]

        # ----------------------------------------------------------
        # (c) Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a - step)

        a = a_new  # always accept the Adam step

        # ----------------------------------------------------------
        # (d) occasional random “mass‑transfer’’ move
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = proj_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < 0.001:   # tiny uphill chance
                    a = cand

        # ----------------------------------------------------------
        # (e) stagnation handling – re‑seed from the best solution
        if time.time() - last_improve > 30.0:
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(0.9 * best_seq + 0.1 * noise)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            last_improve = time.time()

        # ----------------------------------------------------------
        # (f) progress report
        if it % 5000 == 0:
            print(f"[iter {it:7d}] cur {cur_val:.6f}  best {best_val:.6f}  λ {lam:.2f}")

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # Return a plain Python list; the external checker will evaluate it.
    return best_seq.tolist()
