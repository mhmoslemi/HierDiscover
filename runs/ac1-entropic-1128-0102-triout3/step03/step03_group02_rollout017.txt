def conv_fft(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n -1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    return conv

Now gradient softmax method:

def softmax_gradient(a, tau):
    conv = conv_fft(a)
    max_val = conv.max()
    # w = exp((conv - max_val) / tau)
    w = np.exp((conv - max_val) / tau)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(w, 1.0 / len(w))
    # compute grad = 2 * (w * a_rev) conv
    a_rev = a[::-1]
    # length for conv_of_w_a_rev.
    n = len(a)
    fft_len = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len)[:len(w)+n-1]
    # grad indices from conv2[n-1: n-1+n] inclusive.
    grad = 2.0 * conv2[n-1: n-1+n]
    return grad, conv, max_val

def gradient_descent(a, max_iters, deadline, step0=0.5, tau_start=0.01, tau_end=0.0001, beta=0.9, warmup_frac=0.6):
    n = a.shape[0]
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_of_seq(a)
    warmup_iters = int(max_iters * warmup_frac)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute conv and max...
        if it <= warmup_iters:
            # compute conv and gradient via softmax
            conv = conv_fft(a)
            max_val = conv.max()
            # tau schedule:
            t_frac = it / warmup_iters if warmup_iters>0 else 1.0
            tau = max(tau_start + (tau_end - tau_start) * t_frac, 1e-12) * max_val
            # gradient
            grad, _, _ = softmax_gradient(a, tau)  # we need conv for best scoring.
        else:
            # compute conv and gradient via hard max
            conv = conv_fft(a)
            max_val = conv.max()
            idx = np.argmax(conv)
            # compute grad using slice:
            grad = np.zeros_like(a)
            start_i = max(0, idx - (n-1))
            end_i = min(n-1, idx)
            # i from start_i to end_i inclusive correspond to all a_i that have counterpart.
            # For each i in that range, counterpart is idx - i.
            i_range = np.arange(start_i, end_i+1)
            grad[i_range] = 2.0 * a[idx - i_range]
        # update best
        cur_score = 2.0 * n * max_val  # same as _score_of_seq
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # momentum update
        step = step0 / math.sqrt(it)
        v = beta * v - step * grad
        a = _project_simplex(a + v)
    return a, best_a, best_score

def optimize_via_l_bfgs(a0, max_iter, deadline, tau_rel=1e-4):
    n = a0.shape[0]
    # initialize z = log(a0) perhaps
    # Ensure a0 > 0; in case zeros present we add epsilon.
    epsilon = 1e-12
    a0_safe = np.maximum(a0, epsilon)
    z0 = np.log(a0_safe)

    def obj_and_grad(z):
        # check time
        if time.time() >= deadline:
            # raise exception to abort optimizer
            raise TimeoutError()
        a = np.exp(z)
        a_sum = a.sum()
        a = a / a_sum
        # compute conv via FFT
        conv = conv_fft(a)
        max_val = conv.max()
        # compute softmax weights with temperature = tau_rel * max_val
        tau = max(tau_rel * max_val, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0/len(w))
        # gradient w.r.t a:
        a_rev = a[::-1]
        fft_len = 1 << ((len(w) + n - 1).bit_length())
        W = np.fft.rfft(w, n=fft_len)
        A_rev = np.fft.rfft(a_rev, n=fft_len)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len)[:len(w) + n - 1]
        grad_a = 2.0 * conv2[n-1 : n-1 + n]  # shape n
        # objective
        obj = 2.0 * n * max_val

        # compute gradient w.r.t z
        grad_z = a * (grad_a - np.dot(grad_a, a))
        return obj, grad_z

    # Use minimize with L-BFGS-B, method='L-BFGS-B', jac=True, options {'maxiter': max_iter}
    res = minimize(fun=lambda z: obj_and_grad(z), x0=z0, method='L-BFGS-B', jac=True,
                   options={'maxiter': max_iter, 'disp': False, 'gtol': 1e-12})
    # after optimization, get a
    a_opt = np.exp(res.x)
    a_opt = a_opt / a_opt.sum()
    # compute final score
    final_score = _score_of_seq(a_opt)
    return a_opt, final_score

def refine_with_l_bfgs(a_init, max_iter, deadline, tau_rel=1e-4):
    n = a_init.shape[0]
    eps = 1e-12
    a0 = np.maximum(a_init, eps)  # ensure positive for log
    z0 = np.log(a0)
    def fun_and_grad(z):
        if time.time() >= deadline:
            raise TimeoutError()
        a = np.exp(z)
        total = a.sum()
        a = a / total
        # convolution
        conv = conv_fft(a)
        max_val = conv.max()
        tau = max(tau_rel * max_val, 1e-14)
        # compute softmax weights
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.shape[0])

        a_rev = a[::-1]
        # conv2 for gradient
        fft_len = 1 << ((len(w) + n - 1).bit_length())
        W = np.fft.rfft(w, n=fft_len)
        A_rev = np.fft.rfft(a_rev, n=fft_len)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len)[:len(w) + n - 1]
        grad_a = 2.0 * conv2[n-1 : n-1 + n]  # shape n

        obj = 2.0 * n * max_val

        grad_z = a * (grad_a - np.dot(grad_a, a))
        return obj, grad_z

    try:
        res = minimize(fun_and_grad, z0, method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter, 'disp': False, 'gtol': 1e-12})
    except TimeoutError:
        # reached deadline inside function; use current best
        # compute a from current parameters (z0 maybe unchanged)
        # Actually we don't have intermediate z. We could keep last feasible z as closure variable but not.
        # Simpler: bring the function to compute final a from current res.x if available else from a_init.
        # Since exception aborts, we may not have res. We'll fallback to a_init.
        return a_init, _score_of_seq(a_init)

    a_opt = np.exp(res.x)
    a_opt = a_opt / a_opt.sum()
    final_score = _score_of_seq(a_opt)

"""
A fast multi‑phase optimiser for the step‑function attack.

Strategy
--------
1. **Warm‑start** – use the previously discovered best sequence
   (global variable ``height_sequence_1``) if it exists.
2. **Smooth L‑BFGS** – optimise the normalised vector on the simplex
   via an unconstrained log‑softmax parametrisation.
   The objective is a smoothed version of the max‑convolution,
   gradients are obtained analytically (soft‑max weights convolved
   with the reversed signal).  A very small temperature (≈1e‑4·max)
   makes the smooth approximation almost exact while keeping the
   gradient well‑behaved.
3. **Progressive up‑sampling** – repeatedly interpolate the current
   best shape to a larger length (≈1.5×) up to 12000 entries and run
   another L‑BFGS phase at each scale.
4. **Local random perturbations** – perform many tiny mass‑shifts
   (Monte‑Carlo hill‑climbing) on the final shape; occasional
   short projected‑gradient refinements are applied to escape flat
   plateaux.
5. Return the best normalised non‑negative vector found within the
   allotted time (≈1000 s).

The implementation is fully self‑contained, uses only NumPy and
SciPy (if available) and respects the time budget by checking the
deadline at every iteration.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy optimiser
try:
    from scipy.optimize import minimize
    _SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """
    Score for a sequence that already sums to 1:
    2·n·max(conv(a,a))
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full discrete autocorrelation of a (length 2·len‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple projected‑gradient descent kept for fallback when SciPy is absent.
def _refine_sequence(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.6,
    eps_start: float = 0.05,
    eps_end: float = 0.001,
    beta: float = 0.9,
) -> tuple[np.ndarray, np.ndarray, float]:
    """Accelerated soft‑max projected gradient descent (same as in the reference)."""
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        eps_rel = eps_start + (eps_end - eps_start) * (it / max_iters)
        tau = max(eps_rel * max_val, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.shape[0])

        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        g = 2.0 * conv2[n - 1 : n - 1 + n]

        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)

    return a, best_a, best_score


# ----------------------------------------------------------------------
# L‑BFGS optimiser (smooth max with temperature τ = τ_rel·max)
def _refine_with_l_bfgs(
    a0: np.ndarray,
    max_iter: int,
    deadline: float,
    tau_rel: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """Optimise a normalised vector via L‑BFGS on a smooth max‑convolution."""
    if not _SCIPY_AVAILABLE:          # pragma: no cover
        # fall back to the slower projected gradient method
        return _refine_sequence(a0, max_iter, deadline)[0:2]

    n = a0.shape[0]
    eps = 1e-12
    a0 = np.maximum(a0, eps)
    z0 = np.log(a0)                     # log‑softmax parametrisation

    class _TimeoutError(RuntimeError):
        pass

    def _fun_and_grad(z):
        if time.time() >= deadline:
            raise _TimeoutError()
        a = np.exp(z)
        a = a / a.sum()

        conv = _conv_full(a)
        max_val = conv.max()
        tau = max(tau_rel * max_val, 1e-14)

        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.shape[0])

        a_rev = a[::-1]
        fft_len = 1 << ((len(w) + n - 1).bit_length())
        W = np.fft.rfft(w, n=fft_len)
        A_rev = np.fft.rfft(a_rev, n=fft_len)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len)[: len(w) + n - 1]
        grad_a = 2.0 * conv2[n - 1 : n - 1 + n]

        obj = 2.0 * n * max_val
        # chain rule for the softmax parametrisation
        grad_z = a * (grad_a - np.dot(grad_a, a))
        return obj, grad_z

    try:
        res = minimize(
            fun=_fun_and_grad,
            x0=z0,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "gtol": 1e-12, "disp": False},
        )
    except _TimeoutError:               # pragma: no cover
        # timeout happened inside the objective; return the current point
        a = np.exp(z0)
        a = a / a.sum()
        return a, _score_of_seq(a)

    a_opt = np.exp(res.x)
    a_opt = a_opt / a_opt.sum()
    return a_opt, _score_of_seq(a_opt)


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    n_iters: int,
    epsilon: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple Monte‑Carlo hill‑climbing: move a tiny amount of mass
    from a source index (chosen proportionally to its weight) to a
    random destination.  Accept only improving moves.
    """
    best_a = a.copy()
    best_score = _score_of_seq(best_a)
    n = best_a.shape[0]

    for _ in range(n_iters):
        if time.time() >= deadline:
            break
        # source weighted by current mass, destination uniform
        src = np.random.choice(n, p=best_a)
        dst = np.random.randint(0, n)
        if src == dst:
            continue
        delta = min(epsilon, best_a[src])
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # normalisation unchanged, only positivity matters
        score = _score_of_seq(cand)
        if score < best_score:
            best_score = score
            best_a = cand
            print(
                f"[{time.time():.2f}s] Random move improved score to {best_score:.6f}"
            )
    return best_a, best_score
