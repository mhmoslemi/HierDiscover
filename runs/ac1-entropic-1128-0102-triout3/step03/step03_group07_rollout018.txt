"""
Improved search for a step‑function minimizing the auto‑convolution peak.

The algorithm works in two phases:

* **Phase A – Random‑parameter shape search**  
  We generate many simple parametric families (power‑law and exponential
  decays), normalise them to unit ℓ¹‑norm and evaluate the objective
  ``2·n·max(conv)`` using an FFT‑based convolution.  The best sequence
  found is kept as the starting point for the second phase.

* **Phase B – Soft‑max gradient‑descent polishing**  
  Starting from the best sequence of phase A we run the smooth max‑
  optimisation introduced in the reference solution (soft‑max (log‑sum‑exp)
  smoothing of the peak, gradually increasing the smoothing parameter
  β).  After β reaches a large target we switch to a pure max‑subgradient
  step.  Each gradient step is projected onto the probability simplex to
  keep feasibility.  Random restarts and tiny Gaussian perturbations are
  used to escape plateaus.

The whole routine respects the 1000 s time budget (a small safety margin
is kept) and finally returns the best sequence found as a plain list of
floats.
"""
import time
import numpy as np
from typing import List
from scipy.optimize import linprog  # used only for the optional LP‑step (fast fallback)

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^‑½  (a_i = C(2i,i)/4^i) and normalised
    so that they sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_score(a: np.ndarray, n: int, Nfft: int, L: int) -> float:
    """
    Returns the value 2·n·max( a * a ) for a sequence a that already
    sums to 1.  Computed via FFT for speed.
    """
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[:L]
    return float(2.0 * n * conv.max())


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Builds the (2n‑1) × n linear convolution matrix C such that
    (C @ b) == a * b  (linear convolution, zero‑padded).
    """
    n = a.shape[0]
    L = 2 * n - 1
    C = np.zeros((L, n), dtype=np.float64)
    # each column j contributes a shifted copy of a
    for j in range(n):
        C[j:j + n, j] = a
    return C


# ----------------------------------------------------------------------
# Core optimisation routine

def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises the evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  A few seconds are kept as safety margin.
    **kwargs : dict
        Ignored – present only to match the required signature.

    Returns
    -------
    List[float]
        The best sequence found (plain Python list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 3500 is a good trade‑off between quality and speed.
    n = 3500
    L = 2 * n - 1
    Nfft = next_pow_two(L + n)          # a safe power‑of‑two size for all FFTs

    # ------------------------------------------------------------------
    # Phase A – random parametric shape search
    best_a = half_binomial_series(n)    # fallback if everything else fails
    best_score = conv_score(best_a, n, Nfft, L)
    print(f"[init] n={n}, initial score (half‑binomial) = {best_score:.6f}")

    phase_a_deadline = time.time() + 0.25 * budget_s   # ~25 % of total time
    it_a = 0
    while time.time() < phase_a_deadline:
        # randomly pick a family
        if rng.random() < 0.55:                     # power‑law family
            p = rng.uniform(0.35, 1.30)             # exponent
            shift = rng.uniform(0.001, 5.0)         # shift >0 to avoid singularity
            idx = np.arange(n, dtype=np.float64)
            raw = (idx + 1.0 + shift) ** (-p)       # (i+1+shift)^{-p}
        else:                                         # exponential family
            lam = rng.uniform(0.0005, 0.0100)        # decay rate
            raw = np.exp(-lam * np.arange(n, dtype=np.float64))

        # normalise to unit ℓ¹‑norm
        a = raw / raw.sum()

        # ignore pathological candidates with an almost‑zero a_0 (would make M tiny
        # but the rest of the convolution huge); keep a_0 reasonable.
        if a[0] < 1e-8:
            continue

        cur_score = conv_score(a, n, Nfft, L)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[phase A iter {it_a:07d}] new best score = {best_score:.6f}")

        it_a += 1

    # ------------------------------------------------------------------
    # Phase B – soft‑max gradient descent (the reference optimiser) with
    #             modified hyper‑parameters and occasional momentum.
    a = best_a.copy()                     # start from the best shape found
    best_score_B = best_score
    best_a_B = a.copy()

    # optimisation parameters
    beta = 1.0
    beta_target = 1800.0               # a larger final β gives a tighter max
    beta_mul = 1.003
    eta = 0.22                         # initial step size
    eta_min = 5e-5
    eta_decay = 0.9994
    max_phase = 1                      # 1 = soft‑max, 2 = pure max
    momentum = np.zeros_like(a)        # simple exponential momentum
    mu = 0.85                          # momentum coefficient

    # restart and perturbation control
    last_improve = 0
    restart_gap = 300_000
    noise_prob = 0.008
    noise_scale = 1e-4

    deadline = time.time() + budget_s - 3.0
    it = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) convolution and score
        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # 2) best‑so‑far tracking
        if cur_score < best_score_B:
            best_score_B = cur_score
            best_a_B = a.copy()
            last_improve = it
            print(f"[iter {it:09d}] new best (phase B) score = {best_score_B:.6f}")

        # --------------------------------------------------------------
        # 3) gradient computation
        if max_phase == 1:                     # soft‑max smoothing
            shift = max_conv
            w_exp = np.exp(beta * (conv - shift))
            w = w_exp / w_exp.sum()           # shape (L,)

            # correlation via FFT (gradient = 2·corr(a,w))
            Wr = np.fft.rfft(w[::-1], n=Nfft)
            corr_full = np.fft.irfft(A * Wr, n=Nfft)
            start = L - 1
            grad = 2.0 * corr_full[start:start + n]
            beta = min(beta_target, beta * beta_mul)
            if beta >= beta_target:
                max_phase = 2
        else:                                   # pure max‑subgradient
            k = int(conv.argmax())
            grad = np.zeros_like(a)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = k - idx
            grad[idx] = 2.0 * a[jdx]

        # --------------------------------------------------------------
        # 4) momentum & step
        momentum = mu * momentum + (1.0 - mu) * grad
        step = eta * momentum
        # avoid flipping any entry negative in a single step
        step = np.minimum(step, a * 0.95)

        a_trial = a - step
        a = project_onto_simplex(a_trial)

        # --------------------------------------------------------------
        # 5) occasional Gaussian perturbation
        if rng.random() < noise_prob:
            noise = (rng.random(n) - 0.5) * noise_scale
            a = project_onto_simplex(a + noise)

        # --------------------------------------------------------------
        # 6) step‑size schedule
        eta = max(eta * eta_decay, eta_min)

        # --------------------------------------------------------------
        # 7) restart if stuck for a long time
        if it - last_improve > restart_gap:
            # take a fresh random Dirichlet start
            a = rng.dirichlet(np.full(n, 0.6))
            a = project_onto_simplex(a)
            last_improve = it
            # recompute its score (for logging)
            A = np.fft.rfft(a, n=Nfft)
            conv = np.fft.irfft(A * A, n=Nfft)[:L]
            print(f"[iter {it:09d}] restart – new random start, score≈{2.0*n*conv.max():.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Choose the best sequence found in either phase
    final_a = best_a_B if best_score_B < best_score else best_a
    final_score = evaluate_sequence(final_a.tolist())
    print(f"[done] total iterations (phase B)={it}, best reported score={final_score:.6f}")

    return final_a.tolist()
