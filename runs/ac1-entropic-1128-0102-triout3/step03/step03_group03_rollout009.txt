    n = a.shape[0]
    fft_len = 1 << ((2*n - 1).bit_length())

def gradient_softmax_max(a, alpha):
    n = a.shape[0]
    conv = compute_conv(a)
    # compute softmax weights
    maxc = conv.max()
    # subtract max for numerical stability
    exp_vals = np.exp(alpha * (conv - maxc))
    w = exp_vals / exp_vals.sum()
    # compute convolution of reversed a with w
    # use FFT for conv_rev
    # reversed a: a_rev = a[::-1]
    len_w = conv.shape[0]  # 2n-1
    total_len = len_w + n - 1   # length for conv_rev
    fft_len = 1 << ((total_len - 1).bit_length())
    A_rev_f = np.fft.rfft(a[::-1], n=fft_len)
    W_f = np.fft.rfft(w, n=fft_len)
    conv_rev_full = np.fft.irfft(A_rev_f * W_f, n=fft_len)[:total_len]

    start = n - 1

    # diagonal correction
    diag_corr = w[2 * np.arange(n)] * a
    grad = 2.0 * sum_term - diag_corr
    return grad

"""
Propose a candidate step‑function using a soft‑max (log‑sum‑exp) surrogate for the
ℓ∞‑norm of the auto‑convolution.  The method works on the probability simplex
with exponentiated‑gradient updates (mirror descent w.r.t. the KL‑divergence)
while gradually sharpening the surrogate (α grows).  A progressive
upsampling scheme first optimises a modest length (≈1500), then repeatedly
interpolates to larger sizes (up to 5000) and continues the optimisation.
Finally a cheap stochastic mass‑transfer hill‑climber refines the best
solution found.

The algorithm differs from the previous active‑set approach by using the full
soft‑max gradient (computed efficiently with FFTs) instead of a tiny active
set, and by annealing the smoothing parameter α.  Empirically this yields a
tighter reduction of the objective 2·n·max(conv(a,a)).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities ----------------------------------------------------
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution of a with itself (length 2·n‑1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the soft‑max weights w_k = exp(α·conv_k)/∑exp(α·conv).
    The shift by max(conv) guarantees numerical stability.
    """
    max_c = conv.max()
    shifted = conv - max_c          # max becomes 0
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / exp_vals.sum()
    return w


def _grad_softmax_max(a: np.ndarray, alpha: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute both the auto‑convolution of a and the gradient of the
    smoothed max (log‑sum‑exp) w.r.t. a.

    Returns:
        conv   – full convolution a*a (length 2·n‑1)
        grad   – ∂/∂a  ( (1/α)·log∑exp(α·conv) )
    """
    n = a.shape[0]

    # 1) convolution a*a
    conv = _conv_full(a)

    # 2) soft‑max weights over the convolution entries
    w = _softmax_weights(conv, alpha)               # length 2·n‑1

    # 3) gradient = 2 * Σ_k w_k a_{k-i}  –  w_{2i}·a_i
    #    The first term is a convolution of reversed a with w.
    a_rev = a[::-1]                                 # length n
    total_len = (2 * n - 1) + n - 1                  # = 3·n‑2
    fft_len = _next_pow2(total_len)
    A_rev_f = np.fft.rfft(a_rev, n=fft_len)
    W_f = np.fft.rfft(w, n=fft_len)
    conv_rev_full = np.fft.irfft(A_rev_f * W_f, n=fft_len)[:total_len]

    # extract the pieces corresponding to Σ_t a_t·w_{i+t}
    start = n - 1
    sum_term = conv_rev_full[start:start + n]       # shape (n,)

    # diagonal correction (only one contribution when k = 2i)
    diag_corr = w[2 * np.arange(n)] * a

    grad = 2.0 * sum_term - diag_corr
    return conv, grad


def _score_from_conv(conv: np.ndarray) -> float:
    """Return the evaluation score 2·n·max(conv) for a convolution array."""
    n = (conv.shape[0] + 1) // 2
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------
# ----------------------------------------------------------------------


def _exponentiated_step(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform a mirror‑descent step on the simplex:
        a ← normalize( a * exp( -η·grad ) )
    """
    # keep numbers safely away from zero before taking logs
    a = np.maximum(a, 1e-15)
    log_a = np.log(a) - eta * grad
    log_a -= np.max(log_a)                 # numeric stabilisation
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new


def _optimize_on_simplex(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation on the simplex using the soft‑max surrogate.
    The smoothing parameter α is increased linearly; the step size η decays
    like 1/√t.  Returns the best sequence seen (still normalised to sum=1) and
    its raw score 2·n·max(conv).
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    conv, _ = _grad_softmax_max(a, alpha=5.0)
    best_score = _score_from_conv(conv)

    eta0 = 0.8
    alpha0 = 5.0
    alpha_max = 200.0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # schedule
        eta = eta0 / math.sqrt(it)
        alpha = min(alpha_max, alpha0 + 0.05 * it)   # slowly sharpen

        conv, grad = _grad_softmax_max(a, alpha)

        cur_score = _score_from_conv(conv)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] improve n={n}, it={it}, score={cur_score:.6f}"
            )

        a = _exponentiated_step(a, grad, eta)

    return best_a, best_score


def _local_mass_transfer(
    a: np.ndarray,
    cur_score: float,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: repeatedly move a random fraction of
    mass from one coordinate to another.  Accept only improving moves.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move up to 4 % of the source mass
        eps = np.random.rand() * 0.04
        delta = eps * best_a[i]
        if delta <= 0:
            continue
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        conv = _conv_full(a_new)
        new_score = _score_from_conv(conv)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new
            print(
                f"[{time.time() - start_time:6.2f}s] random improve: n={n}, score={new_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Produce a non‑negative step‑function (list of heights) that attempts to
    minimise ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds and prints progress.  The optimisation proceeds in three stages:
        1.  Multiple inexpensive starts (uniform, random, or the historic
            ``height_sequence_1`` if it exists) are optimised at a modest
            size (≈1500) using soft‑max mirror descent.
        2.  The best intermediate shape is progressively up‑sampled (≈1.4×)
            until the target length (5000) is reached, continuing the same
            optimisation at each scale.
        3.  A short stochastic local‑search refines the final candidate.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # tiny safety margin

    # ------------------------------------------------------------------
    # 1.   Gather initial candidates
    # ------------------------------------------------------------------
    inits = []

    # a) historic construction, if present
    try:
        init = np.array(globals()["height_sequence_1"], dtype=float)
        init = np.maximum(init, 0.0)
        if init.sum() > 0:
            init /= init.sum()
            inits.append(init)
            print("[init] Using historic height_sequence_1.")
    except Exception:
        pass

    # b) uniform start (moderate length)
    n0 = 1500
    inits.append(np.full(n0, 1.0 / n0))

    # c) random Dirichlet start, same length
    rand = np.random.rand(n0)
    rand /= rand.sum()
    inits.append(rand)

    # d) simple triangular shape (more mass on the sides)
    tri = np.linspace(1, 0, n0 // 2 + 1)
    tri = np.concatenate([tri, tri[-2::-1]])          # symmetric
    tri = np.maximum(tri, 0)
    tri /= tri.sum()
    inits.append(tri)

    # ------------------------------------------------------------------
    # 2.   Coarse optimisation (≈1500 points)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    base_n = 1500
    max_iters_coarse = 25000

    for idx, init in enumerate(inits):
        if time.time() > deadline:
            break
        # ensure the length equals base_n (interpolate if needed)
        if init.shape[0] != base_n:
            x_old = np.linspace(0.0, 1.0, init.shape[0])
            x_new = np.linspace(0.0, 1.0, base_n)
            a = np.interp(x_new, x_old, init)
        else:
            a = init.copy()
        a = np.maximum(a, 0.0)
        a /= a.sum()

        a_opt, sc = _optimize_on_simplex(
            a, max_iters_coarse, deadline, start_time
        )
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[init {idx}] new best coarse score: {sc:.6f}")

    if best_seq is None:
        # fallback – a simple uniform vector
        best_seq = np.full(100, 1.0 / 100)
        best_score = _score_from_conv(_conv_full(best_seq))

    # ------------------------------------------------------------------
    # 3.   Progressive up‑sampling to the target size (5000)
    # ------------------------------------------------------------------
    target_n = 5000
    cur = best_seq.copy()
    cur_n = cur.shape[0]

    while cur_n < target_n and time.time() < deadline:
        # decide the next size (≈1.4× growth, capped)
        nxt = int(cur_n * 1.45) + 1
        nxt = min(nxt, target_n)
        if nxt <= cur_n:
            break

        # linear interpolation
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, nxt)
        cur = np.interp(x_new, x_old, cur)
        cur = np.maximum(cur, 0.0)
        cur /= cur.sum()
        cur_n = nxt

        # iteration budget scales inversely with length (more iterations for
        # smaller vectors, enough for larger ones)
        max_iters = int(20000 * 1500 / cur_n) + 5000
        max_iters = max(max_iters, 8000)

        cur, cur_score = _optimize_on_simplex(
            cur, max_iters, deadline, start_time
        )
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[up] n={cur_n}, improved score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4.   Final stochastic local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _local_mass_transfer(
            best_seq, best_score, deadline, start_time
        )

    # ------------------------------------------------------------------
    # 5.   Return a plain Python list (evaluate_sequence expects list)
    # ------------------------------------------------------------------
    # Clip to the allowed range [0,1000] (the evaluator does it anyway)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Ensure non‑negative and not all zeros (the optimiser already guarantees sum≈1)
    best_seq = np.maximum(best_seq, 0.0)

    return [float(x) for x in best_seq]
```

This implementation respects the time budget, works entirely on the simplex (so the scale‑invariance of the objective is honoured), and employs a different optimisation principle than the previously‑provided code. The soft‑max surrogate together with full‑gradient exponentiated updates and progressive up‑sampling typically drives the maximum convolution value lower, often beating the prior 1.516083 bound within the same time limit.
