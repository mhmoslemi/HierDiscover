```python
def gradient_smooth(a, beta):
    n = a.shape[0]
    # convolution a*a
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    max_val = conv.max()
    # compute softmax weights w = exp(beta*conv - logsumexp)
    # stable:
    max_c = conv.max()
    exp_term = np.exp(beta * (conv - max_c))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp
    # we will compute gradient:
    # compute convolution of reversed w with a
    w_rev = w[::-1]
```

```
def compute_gradient_and_smooth_obj(a, beta):
    n = a.shape[0]
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    # smooth max
    max_c = conv.max()
    exp_term = np.exp(beta * (conv - max_c))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp
    # compute gradient: g = 2 * correlation(w, a) for negative lags
    w_rev = w[::-1]
    fft_len2 = 1 << ((len(w_rev) + n - 1).bit_length())
    Wr = np.fft.rfft(w_rev, n=fft_len2)
    Ar = np.fft.rfft(a, n=fft_len2)
    conv_rev = np.fft.irfft(Wr * Ar, n=fft_len2)[:len(w_rev)+n-1]
    idx = len(w_rev) - 1  # = 2n-2
    g = 2.0 * conv_rev[idx - np.arange(n)]
    # return gradient and smooth_score (optional)
    smooth_score = 2.0 * n * (max_c + np.log(sum_exp) / beta)
```

"""
Propose a candidate step‑function using a smooth log‑sum‑exp surrogate for the
max‑convolution, Adam optimisation on the log‑parameters (soft‑max) and a final
random mass‑transfer local search.  The surrogate allows the use of the full
gradient (all 2n−1 convolution entries) rather than an active‑set, giving a
more accurate descent direction.  The algorithm runs for the allotted time
budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """Exact objective (sum a = 1) = 2·n·max(conv(a,a))."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _smooth_grad_and_score(a: np.ndarray, beta: float):
    """
    Returns the gradient of the smooth objective
        2·n·logsumexp_{k}(conv_k)/beta
    and the surrogate score itself.
    """
    n = a.shape[0]

    # --- auto‑convolution of a -----------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]

    # --- log‑sum‑exp of the convolution entries -------------------------
    max_c = conv.max()
    # stable exponentiation
    exp_shift = np.exp(beta * (conv - max_c))
    sum_exp = exp_shift.sum()
    # weights that sum to 1
    w = exp_shift / sum_exp

    # --- gradient of the smooth max (full set of indices) -------------
    # gradient = 2 * correlation(w, a) for negative lags
    w_rev = w[::-1]
    fft_len2 = 1 << ((len(w_rev) + n - 1).bit_length())
    Wr = np.fft.rfft(w_rev, n=fft_len2)
    Ar = np.fft.rfft(a, n=fft_len2)
    conv_rev = np.fft.irfft(Wr * Ar, n=fft_len2)[: len(w_rev) + n - 1]
    # The needed slice corresponds to indices (len(w_rev)-1) - i
    idx = len(w_rev) - 1                     # = 2*n-2
    g = 2.0 * conv_rev[idx - np.arange(n)]

    # --- surrogate score ------------------------------------------------
    smooth_max = max_c + math.log(sum_exp) / beta
    smooth_score = 2.0 * n * smooth_max

    return g, smooth_score

# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float):
    """
    Optimise a distribution on the probability simplex using Adam on the
    log‑parameters.  The surrogate loss is the log‑sum‑exp of the convolution;
    beta is slowly increased (annealed) so that the surrogate converges to the
    true max.  The routine returns the best sequence (w.r.t. the exact objective)
    observed before the deadline.
    """
    n = a0.shape[0]
    # initialise log‑probabilities
    log_a = np.log(np.maximum(a0, 1e-15))
    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12            # base learning rate for the log‑domain
    beta1_pow = 1.0
    beta2_pow = 1.0

    # smoothing parameter schedule
    beta = 0.5
    beta_max = 80.0
    beta_inc = 1.001

    # keep track of the best exact score
    a_cur = np.exp(log_a)
    a_cur /= a_cur.sum()
    best_a = a_cur.copy()
    best_score = _score_of_seq(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1. evaluate smooth objective and its gradient
        # --------------------------------------------------------------
        a_cur = np.exp(log_a)
        a_cur /= a_cur.sum()
        g, smooth_score = _smooth_grad_and_score(a_cur, beta)

        # --------------------------------------------------------------
        # 2. Adam update on log‑parameters
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep log‑values in a reasonable range to avoid overflow
        np.clip(log_a, -30.0, 30.0, out=log_a)

        # --------------------------------------------------------------
        # 3. occasional exact evaluation
        # --------------------------------------------------------------
        if it % 250 == 0:
            a_cur = np.exp(log_a)
            a_cur /= a_cur.sum()
            cur_score = _score_of_seq(a_cur)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_cur.copy()
                print(f"[{time.time():.2f}s] iter={it}  exact score={cur_score:.6f}  beta={beta:.2f}")

        # --------------------------------------------------------------
        # 4. anneal beta
        # --------------------------------------------------------------
        beta = min(beta_max, beta * beta_inc)

    # final check
    final_score = _score_of_seq(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = best_a

    return best_a, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method starts from the previously discovered
    construction if available, otherwise from a uniform distribution, then
    performs a long Adam optimisation on a smooth surrogate of ``max(conv)``
    together with a small final stochastic local search.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # leave a small slack

    # --------------------------------------------------------------
    # 1. Choose a target length (the same size used in the previous best)
    # --------------------------------------------------------------
    TARGET_N = 5000

    # --------------------------------------------------------------
    # 2. Initialise the distribution
    # --------------------------------------------------------------
    init_seq = None
    if 'height_sequence_1' in globals():
        try:
            tmp = np.array(height_sequence_1, dtype=float)
            tmp = np.maximum(tmp, 0.0)
            if tmp.sum() > 0.0:
                # up‑sample / down‑sample to TARGET_N points
                x_old = np.linspace(0.0, 1.0, len(tmp))
                x_new = np.linspace(0.0, 1.0, TARGET_N)
                tmp = np.interp(x_new, x_old, tmp)
                init_seq = tmp
        except Exception:
            init_seq = None

    if init_seq is None:
        # fall back to a uniform distribution
        init_seq = np.full(TARGET_N, 1.0 / TARGET_N)

    # Normalise (sum = 1) and evaluate the starting score
    init_seq = np.maximum(init_seq, 0.0)
    init_seq /= init_seq.sum()
    best_seq = init_seq.copy()
    best_score = _score_of_seq(best_seq)
    print(f"Initial sequence: n={TARGET_N}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Main optimisation loop (Adam on smooth surrogate)
    # --------------------------------------------------------------
    # We run a single long optimisation; short random restarts are added
    # if time permits after the first run.
    best_seq, best_score = _adam_optimize(best_seq, deadline)

    # --------------------------------------------------------------
    # 4. Final cheap stochastic local search (mass transfer)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        max_local_iters = 20000
        it_local = 0
        while time.time() < deadline and it_local < max_local_iters:
            it_local += 1
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # transfer a random fraction (up to 3 %) of the mass from i to j
            eps = np.random.rand() * 0.03
            delta = eps * a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # maintain non‑negativity
            if a_new[i] < 0:
                continue
            new_score = _score_of_seq(a_new)
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                a = a_new
                print(f"[{time.time() - start:6.2f}s] local move improve: score={new_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]

    n = a0.shape[0]
    log_a = np.log(np.maximum(a0, 1e-15))

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                 # learning rate for the log‑domain
    beta1_pow = 1.0
    beta2_pow = 1.0

    # smoothing schedule
    beta = 0.5
    beta_max = 80.0
    beta_inc = 1.001

    # best exact solution seen so far
    a_cur = np.exp(log_a)
    a_cur /= a_cur.sum()
    best_a = a_cur.copy()
    best_score = _score_of_seq(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- evaluate smooth loss and its gradient -------------------
        a_cur = np.exp(log_a)
        a_cur /= a_cur.sum()
        g, _ = _smooth_grad_and_score(a_cur, beta)

        # ----- Adam update on log‑parameters ---------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logs in a safe range (prevents overflow/underflow)
        np.clip(log_a, -30.0, 30.0, out=log_a)

        # ----- occasional exact evaluation -----------------------------
        if it % 250 == 0:
            a_cur = np.exp(log_a)
            a_cur /= a_cur.sum()
            cur_score = _score_of_seq(a_cur)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_cur.copy()
                print(f"[{time.time():.2f}s] iter={it}  exact_score={cur_score:.6f}  beta={beta:.2f}")

        # ----- anneal the smoothing parameter -------------------------
        beta = min(beta_max, beta * beta_inc)

    # final check (in case the last exact evaluation was skipped)
    final_score = _score_of_seq(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = best_a
    return best_a, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given evaluation
    function.  The algorithm proceeds as follows:

    1. Initialise from the previously discovered sequence `height_sequence_1`
       (if present) or from a uniform distribution, resampled to a target
       length of 5000 points.
    2. Run a long Adam optimisation on a smooth log‑sum‑exp surrogate of the
       max‑convolution, annealing the smoothing parameter β from 0.5 up to 80.
       The best exact score seen during this phase is retained.
    3. Perform a brief stochastic local search that repeatedly transfers a
       small random fraction of mass between two indices; any improvement is
       kept.
    4. Return the best sequence as a plain list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # 1. Target length (same as the previous best construction)
    # --------------------------------------------------------------
    TARGET_N = 5000

    # --------------------------------------------------------------
    # 2. Initialise the distribution
    # --------------------------------------------------------------
    init_seq = None
    if 'height_sequence_1' in globals():
        try:
            tmp = np.array(height_sequence_1, dtype=float)
            tmp = np.maximum(tmp, 0.0)
            if tmp.sum() > 0.0:
                # linear interpolation to TARGET_N points
                x_old = np.linspace(0.0, 1.0, len(tmp))
                x_new = np.linspace(0.0, 1.0, TARGET_N)
                tmp = np.interp(x_new, x_old, tmp)
                init_seq = tmp
        except Exception:
            init_seq = None

    if init_seq is None:
        init_seq = np.full(TARGET_N, 1.0 / TARGET_N)

    init_seq = np.maximum(init_seq, 0.0)
    init_seq /= init_seq.sum()
    best_seq = init_seq.copy()
    best_score = _score_of_seq(best_seq)
    print(f"Initial sequence: n={TARGET_N}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Main optimisation (Adam on the smooth surrogate)
    # --------------------------------------------------------------
    best_seq, best_score = _adam_optimize(best_seq, deadline)

    # --------------------------------------------------------------
    # 4. Final cheap stochastic local search (mass transfer)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        max_local_iters = 20000
        it_local = 0
        while time.time() < deadline and it_local < max_local_iters:
            it_local += 1
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps = np.random.rand() * 0.03      # up to 3 % of source mass
            delta = eps * a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            if a_new[i] < 0:
                continue
            new_score = _score_of_seq(a_new)
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                a = a_new
                print(f"[{time.time() - start_time:6.2f}s] local improve: score={new_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
