def _adam_optimize(a, max_iters, deadline, step0=0.5, eps_start=0.05, eps_end=0.001, beta1=0.9, beta2=0.999):
    n=len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score(a)
    for it in range(1, max_iters+1):
        if time.time()>=deadline: break
        # compute eps schedule
        eps = eps_start + (eps_end-eps_start)*(it/max_iters)
        grad = _gradient_softmax(a, max(eps*max(_conv_linear(a)),1e-14))
        # updates (Adam)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        step = step0 / (np.sqrt(it)+1)
        a = _project_simplex(a - step * m_hat / (np.sqrt(v_hat)+1e-8))
        # evaluate
        s = _score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    return a, best_a, best_score

"""
Propose a low‑score step‑function using a three‑phase search:

1. **Diverse random starts** – uniform, triangular, Hann, a previous best,
   and a random Dirichlet.  Each start is refined with a fast Adam‑style
   optimiser that minimises the objective 2·n·max(conv(a,a)) while keeping
   a on the probability simplex (∑a=1, a_i≥0).

2. **Progressive up‑sampling** – the current best shape is linearly
   interpolated to a larger size (≈1.4× bigger, up to a configurable max).
   The larger problem is again refined with the optimiser; because the
   objective scales roughly like 1/n the longer sequences can achieve a
   smaller absolute value.

3. **Final polishing** – a short “hard‑max” stage that uses the exact
   sub‑gradient of the max‑convolution (the gradient of those entries that
   attain the current maximum) and a decreasing step‑size.  This finishes
   the search once the global time budget (default 1000 s) is exhausted.

All vector updates are projected onto the simplex by the Euclidean projection
of Duchi et al.  The code is fully self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` for the final check, and respects the 1 CPU
limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_linear(a: np.ndarray) -> np.ndarray:
    """Linear (full) auto‑convolution of a non‑negative vector."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())       # smallest power‑of‑2 ≥2n‑1
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv) (the same quantity the optimizer minimises)."""
    n = a.shape[0]
    conv = _conv_linear(a)
    return 2.0 * n * conv.max()


def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Soft‑max probabilities over the convolution entries."""
    max_val = conv.max()
    w = np.exp((conv - max_val) / eps)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(conv, 1.0 / conv.shape[0])
    return w


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_k w_k·conv_k w.r.t. a (using the convolution symmetry)."""
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((len(w) + n - 2).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    tmp = np.fft.irfft(W * A_rev, n=L)[: len(w) + n - 1]
    # extract the n entries that correspond to the original coordinates
    grad = 2.0 * tmp[n - 1 : n - 1 + n]
    return grad

) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam‑style optimisation of 2·n·max(conv) on the simplex.
    Returns (final_a, best_a_seen, best_score_seen).
    """
    n = a.shape[0]
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    best_a = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_linear(a)
        max_val = conv.max()

        # temperature schedule for the soft‑max
        eps = eps_start + (eps_end - eps_start) * (it / max_iters)
        tau = max(eps * max_val, 1e-14)

        w = _softmax_weights(conv, tau)
        g = _grad_from_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it + 1.0)
        a = _project_simplex(a - step * m_hat / (np.sqrt(v_hat) + 1e-8))

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return a, best_a, best_score


def _hardmax_optimize(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.4,
    beta: float = 0.9,
    tol: float = 1e-12,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimises using the exact sub‑gradient (indicator of the maximal
    convolution entries) for the final polishing stage.
    """
    n = a.shape[0]
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_linear(a)
        max_val = conv.max()
        # positions that are (almost) maximal
        mask = np.abs(conv - max_val) <= tol
        w = mask.astype(float)
        if w.sum() > 0:
            w /= w.sum()

        g = _grad_from_weights(a, w)

        step = step0 / math.sqrt(it + 1.0)
        v = beta * v - step * g
        a = _project_simplex(a + v)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return a, best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low ``evaluate_sequence`` score.
    The routine respects the time budget (default 1000 s) and returns the
    best sequence found as plain Python floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0         # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ---- 1. build a small pool of diverse starts (all on the simplex) ----
    n0 = 1200
    pool = []

    # a) previous construction, if it exists
    try:
        prev = np.array(height_sequence_1, dtype=float)
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            if prev.shape[0] >= n0:
                pool.append(prev[:n0])
            else:
                # pad with zeros to length n0 and renormalise
                pad = np.zeros(n0)
                pad[: prev.shape[0]] = prev
                pad = pad / pad.sum()
                pool.append(pad)
    except Exception:
        pass

    # b) uniform
    pool.append(np.full(n0, 1.0 / n0))

    # c) tent (triangular) shape
    idx = np.arange(n0)
    tri = np.minimum(idx + 1, n0 - idx)
    pool.append(tri / tri.sum())

    # d) Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n0 - 1)))
    pool.append(hann / hann.sum())

    # e) a random Dirichlet start
    pool.append(np.random.dirichlet(np.ones(n0)))

    # ---- 2. refine each start -------------------------------------------------
    for i, cand in enumerate(pool, 1):
        if time.time() >= deadline:
            break
        iters = 30000
        print(f"[{time.time() - start:6.2f}s] Refining start {i}/{len(pool)} "
              f"(n={n0}, iters={iters})")
        _, best_local, best_local_score = _adam_optimize(
            cand,
            max_iters=iters,
            deadline=deadline,
            step0=0.6,
            eps_start=0.05,
            eps_end=0.001,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"   New global best: score={best_local_score:.6f}")

    # ---- 3. progressive up‑sampling -----------------------------------------
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)

    cur = best_seq.copy()
    cur_n = cur.shape[0]
    MAX_LEN = 12000                       # a bit larger than the previous 8000

    while True:
        # decide next size (≈1.4× larger, never shrink)
        nxt = int(cur_n * 1.4)
        nxt = min(nxt, MAX_LEN)
        if nxt <= cur_n:
            break

        # linear interpolation to the new grid
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, nxt)
        a_up = np.interp(x_new, x_old, cur)
        a_up = np.maximum(a_up, 0.0)
        if a_up.sum() == 0:
            a_up = np.full(nxt, 1.0 / nxt)
        else:
            a_up = a_up / a_up.sum()

        # allocate a modest number of iterations (larger n ⇒ fewer iters)
        base = 45000
        iters = max(15000, int(base * n0 / nxt))
        print(f"[{time.time() - start:6.2f}s] Upsampling to n={nxt}, "
              f"iters={iters}")

        _, best_local, best_local_score = _adam_optimize(
            a_up,
            max_iters=iters,
            deadline=deadline,
            step0=0.55,
            eps_start=0.04,
            eps_end=0.001,
        )

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"   Updated global best after up‑sampling: n={nxt}, "
                  f"score={best_local_score:.6f}")

        # keep the final iterate for the next up‑sampling step
        cur = best_local
        cur_n = nxt

        if time.time() >= deadline:
            break

    # ---- 4. final polishing with the exact hard‑max sub‑gradient -------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        iters = 60000
        print(f"[{time.time() - start:6.2f}s] Final hard‑max polishing "
              f"on n={n}, iters={iters}")
        _, best_local, best_local_score = _hardmax_optimize(
            best_seq,
            max_iters=iters,
            deadline=deadline,
            step0=0.35,
            beta=0.9,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"   Final improvement: score={best_local_score:.6f}")

    # ---- 5. tidy up and return ------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    # Show the true evaluation (the function that will be used by the judge)
    try:
        from __main__ import evaluate_sequence  # the platform injects it
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start:6.2f}s] Finished. "
              f"True evaluate_sequence score = {true_score:.6f}")
    except Exception:
        print(f"[{time.time() - start:6.2f}s] Finished. "
              f"Internal best score (2·n·max(conv)) = {best_score:.6f}")

    return [float(x) for x in best_seq]
