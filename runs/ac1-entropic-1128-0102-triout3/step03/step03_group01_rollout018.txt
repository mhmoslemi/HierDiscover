def _run_fast_anneal(a0, deadline, start_time):
    a = a0.copy()
    n = len(a)
    conv = _conv_self(a)  # uses FFT.
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    cur_score = best_score
    cur_max = conv.max()
    anneal_start = time.time()
    # Set schedule parameters: start_temp = cur_max * maybe 0.5, end_temp = cur_max * 1e-5.
    start_temp = max(cur_max * 0.25, 1e-8)
    end_temp = max(cur_max * 1e-6, 1e-12)
    # Compute total time left
    total_time = deadline - anneal_start
    # We'll loop until deadline.
    it = 0
    while time.time() < deadline:
        it += 1
        # pick random i with a[i] > 1e-12
        i = np.random.randint(0, n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(0, n)
        if i == j:
            continue
        # choose epsilon
        eps = a[i] * np.random.rand() * 0.2
        if eps <= 0:
            continue
        # store this for revert
        a_i_old = a[i]
        a_j_old = a[j]
        # update conv
        # slices in view referencing a before modifications
        # Use a_save = a.copy() before modification? Actually we need old a for slice updates.
        # At this point a hasn't been changed yet.
        conv[i:i+n] -= 2.0 * eps * a
        conv[j:j+n] += 2.0 * eps * a
        # epsilon^2 adjustments
        idx = 2*i
        if idx < 2*n-1:
            conv[idx] += eps*eps
        idx = 2*j
        if idx < 2*n-1:
            conv[idx] += eps*eps
        idx = i + j
        if idx < 2*n-1:
            conv[idx] -= 2.0 * eps*eps

        # adjust a
        a[i] -= eps
        a[j] += eps

        # compute new max
        new_max = conv.max()
        new_score = 2.0 * n * new_max

        # temperature based acceptance
        if new_score < cur_score:
            accept = True
        else:
            # compute fraction of remaining time
            elapsed = time.time() - anneal_start
            proportion = min(elapsed / total_time, 1.0)
            T = start_temp * ((end_temp / start_temp) ** proportion)
            # if T may become zero; avoid division by zero
            if T <= 0:
                accept = False
            else:
                accept = np.exp(-(new_score - cur_score) / T) > np.random.rand()
        if accept:
            cur_score = new_score
            cur_max = new_max
            # maybe keep best if improved
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # optionally recompute conv sqrt to address drift? We'll keep conv.
                print(f"[{time.time() - start_time:6.2f}s] fast anneal improve: n={n}, it={it}, score={cur_score:.6f}")
        else:
            # revert
            a[i] = a_i_old
            a[j] = a_j_old
            # revert conv changes
            conv[i:i+n] += 2.0 * eps * a   # now a restored to old values
            conv[j:j+n] -= 2.0 * eps * a
            if 2*i < 2*n-1:
                conv[2*i] -= eps*eps
            if 2*j < 2*n-1:
                conv[2*j] -= eps*eps
            if i + j < 2*n-1:
                conv[i+j] += 2.0 * eps*eps

        # occasional full recompute to correct drift either every 50000/100000 steps or if time.
        if it % 50000 == 0:
            conv = _conv_self(a)   # recompute full
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
    return best_a, best_score

                best_a = a.copy()

"""
Hybrid optimisation for the step‑function problem.

1. Warm‑start from the previously found construction (`height_sequence_1`);
   normalise it to sum = 1 and, if necessary, up‑sample to a fixed
   length (`TARGET_N`).

2. A short run of the original smoothed‑gradient descent (`_run_smooth_descent`)
   offers a quick gain with virtually no extra cost.

3. The core of the new search is a high‑throughput simulated‑annealing loop
   (`_run_fast_anneal`).  The convolution is updated *incrementally*:
   moving a small amount `ε` from index `i` to `j` only changes the
   `n` entries `conv[i:i+n]` and `conv[j:j+n]`.  This avoids the O(n log n)
   FFT in every trial and allows millions of proposals within the 1000 s
   budget.

   * Temperature decays exponentially from a fraction of the current score
     to a tiny value.
   * Only moves that improve the objective are always taken; uphill moves are
     accepted according to the annealing temperature.
   * Every 100 k steps the convolution is recomputed from scratch to
     eliminate accumulated round‑off error.

The routine returns the best sequence found (as a plain Python list of
floats).  All numbers are non‑negative, sum to 1 (scale‑invariant for the
objective) and respect the limits imposed by `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real‑to‑real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = x.size + y.size - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length and renormalisation (∑=1)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed gradient descent (the method used previously)
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of the log‑sum‑exp surrogate."""
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)                # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _run_smooth_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Smoothed gradient descent with momentum on the simplex."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    beta = 0.9
    step0 = 0.8
    it = 0

    lam_factor = 0.25          # initial λ = factor * max(conv)
    lam_decay = 0.5
    lam_decay_interval = 2_000

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, "
                f"it={it}, score={cur_score:.6f}"
            )

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast incremental simulated annealing
# ----------------------------------------------------------------------


def _run_fast_anneal(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """High‑throughput annealing that updates the convolution incrementally."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)                     # exact convolution at start
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    best_a = a.copy()
    best_score = cur_score

    anneal_start = time.time()
    total_time = deadline - anneal_start
    # Temperature expressed on the *objective* scale
    start_temp = max(cur_score * 0.02, 1e-4)
    end_temp = max(cur_score * 1e-6, 1e-12)

    it = 0
    while time.time() < deadline:
        it += 1

        # ------- propose a move -------
        i = np.random.randint(0, n)
        ai = a[i]
        if ai <= 0.0:
            continue
        j = np.random.randint(0, n)
        if i == j:
            continue

        eps = ai * np.random.rand() * 0.2   # up to 20 % of the source mass
        if eps <= 0.0:
            continue

        # Store data required for a revert
        a_i_old = ai
        a_j_old = a[j]

        # ---- incremental convolution update (uses the *old* a) ----
        conv[i : i + n] -= 2.0 * eps * a
        conv[j : j + n] += 2.0 * eps * a

        # ε² terms (tiny, but keep for exactness)
        idx = 2 * i
        if idx < 2 * n - 1:
            conv[idx] += eps * eps
        idx = 2 * j
        if idx < 2 * n - 1:
            conv[idx] += eps * eps
        idx = i + j
        if idx < 2 * n - 1:
            conv[idx] -= 2.0 * eps * eps

        # ---- apply the mass transfer ----
        a[i] = a_i_old - eps
        a[j] = a_j_old + eps

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        # ----- acceptance test -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # exponential temperature schedule
            elapsed = time.time() - anneal_start
            frac = min(elapsed / total_time, 1.0) if total_time > 0 else 1.0
            T = start_temp * ((end_temp / start_temp) ** frac) if start_temp > 0 else 0.0
            if T > 0.0:
                accept = math.exp(-(new_score - cur_score) / T) > np.random.rand()

        if accept:
            cur_score = new_score
            cur_max = new_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] fast anneal improve: "
                    f"n={n}, it={it}, score={cur_score:.6f}"
                )
        else:
            # ---- revert everything ----
            a[i] = a_i_old
            a[j] = a_j_old

            conv[i : i + n] += 2.0 * eps * a
            conv[j : j + n] -= 2.0 * eps * a
            if 2 * i < 2 * n - 1:
                conv[2 * i] -= eps * eps
            if 2 * j < 2 * n - 1:
                conv[2 * j] -= eps * eps
            if i + j < 2 * n - 1:
                conv[i + j] += 2.0 * eps * eps

        # periodic full recomputation to keep numerical drift in check
        if it % 100_000 == 0:
            conv = _conv_self(a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm consists of:

    * a warm‑start from the previous best (`height_sequence_1`);
    * a quick smoothed‑gradient descent pass;
    * a long, fast simulated‑annealing phase that updates the convolution
      incrementally, allowing many millions of proposals within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # ------------------------------------------------------------------
    # Retrieve the previous construction (if available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = None

    # Target length – the previous best used 6400 items.
    TARGET_N = 6400

    if starter is not None:
        # Normalise and adjust length.
        total = starter.sum()
        if total > 0.0:
            starter = starter / total
        else:
            starter = np.random.rand(TARGET_N)
            starter /= starter.sum()
        if starter.shape[0] != TARGET_N:
            a = _upsample(starter, TARGET_N)
        else:
            a = starter.copy()
    else:
        a = np.random.rand(TARGET_N)
        a /= a.sum()

    # ------------------------------------------------------------------
    # Phase 1 – a short smoothed‑gradient descent warm‑up
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        # use at most 30 s or 20 % of the remaining budget, whichever is smaller
        slice_deadline = time.time() + min(30.0, 0.2 * remaining)
        a, _ = _run_smooth_descent(
            a, TARGET_N, slice_deadline, start_time, max_iter=30_000
        )

    # ------------------------------------------------------------------
    # Phase 2 – fast incremental simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _run_fast_anneal(a, deadline, start_time)
        print(
            f"[{time.time() - start_time:6.2f}s] final best score: {best_score:.6f}"
        )
    else:
        best_seq = a
        best_score = float("inf")

    # ------------------------------------------------------------------
    # Return a plain Python list (the harness expects this)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
