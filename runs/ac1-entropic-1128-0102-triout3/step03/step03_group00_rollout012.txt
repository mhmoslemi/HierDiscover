def _conv_full_fft(x, y):
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)[:n]
    return conv

"""
Improved step‑function search.

The algorithm:
1.  A fast “smooth‑Adam’’ optimizer works on the probability simplex.
    Instead of using the true sub‑gradient of the max‑convolution,
    it minimises a smooth approximation
        f_α(a) = (1/α)·log(∑_k exp(α·(a∗a)_k))
    whose gradient can be written as a convolution and is computed
    efficiently by FFT.  α is increased gradually (continuation
    method) so the optimiser first sees a gentle landscape and later
    focuses on the true peak.
    The Adam update is followed by Euclidean projection onto the
    simplex.

2.  After the main optimisation a cheap simulated‑annealing walk
    performs tiny mass‑transfer moves to fine‑tune the solution.

3.  The whole procedure respects the 1000 s budget, prints progress
    and finally returns the best sequence found.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# FFT based convolutions
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    fft_len = 1 << ((out_len - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:out_len]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    out_len = len(x) + len(y) - 1
    fft_len = 1 << ((out_len - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:out_len]
    return conv


# ------------------------------------------------------------
# Smooth (log‑sum‑exp) gradient of the max‑convolution
# ------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, alpha: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, obj) where:
    * obj = 2·n·max(a∗a) (the original objective)
    * grad = ∇_a ( (1/α)·log∑_k exp(α·(a∗a)_k) )
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    max_val = conv.max()
    # smooth weights (log‑sum‑exp trick)
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    p = w / w_sum                            # normalized weights, sum to 1
    # gradient = 2 * (p * a_rev)_{slice}
    a_rev = a[::-1]
    full = _conv_full_fft(p, a_rev)         # length 3n‑2
    grad = 2.0 * full[n - 1 : n - 1 + n]     # slice of length n
    # original (non‑smooth) objective for bookkeeping
    obj = 2.0 * n * max_val
    return grad, obj


# ------------------------------------------------------------
# Adam optimiser on the simplex (smooth continuation)
# ------------------------------------------------------------
def _run_smooth_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alpha_start: float = 0.1,
    alpha_factor: float = 1.001,
    alpha_max: float = 30.0,
    eta0: float = 0.8,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iter: int = 10**7,
) -> tuple[np.ndarray, float]:
    """Adam with smooth max‑convolution gradient + simplex projection."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    alpha = alpha_start
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        grad, cur_score = _smooth_gradient(a, alpha)

        # keep the best (according to the true objective)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={t}, score={cur_score:.6f}"
            )

        # Adam update (working on the raw vector)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta0 / math.sqrt(t)  # simple decay

        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # project back onto the simplex
        a = _project_simplex(a)

        # increase the smoothing parameter
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated annealing refinement (same as in the baseline)
# ------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 80000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ------------------------------------------------------------
# Linear interpolation to change length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
