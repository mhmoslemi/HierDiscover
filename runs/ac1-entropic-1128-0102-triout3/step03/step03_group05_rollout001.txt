"""
Propose a step‑function using a projected Adam optimizer on a *smooth*
approximation of the max‑convolution.  The max is replaced by a soft‑max
with an annealing temperature, allowing the gradient to take all (or many)
convolution entries into account.  At each iteration we:

  1. Compute the self‑convolution a*a via an FFT.
  2. Form soft‑max weights w_i ∝ exp(conv_i / T) (T slowly decreases).
  3. The gradient of the smooth objective is
         g = 2 * (w  *  a_rev)        (valid convolution)
     which we also obtain with an FFT.
  4. Perform an Adam update and project the result onto the probability
     simplex (∑a_i = 1, a_i ≥ 0).
  5. Keep the best sequence seen (according to the exact fast_score).

Occasional random “mass‑transfer” moves and a restart mechanism help to
escape plateaus.  The whole routine stops after the allotted time budget
(~1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (scale‑invariant)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n-i) (i=0..n-1), normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation value.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin for interpreter exit

    # --------------------------------------------------------------
    # 1. Problem size – can be tuned; 3000 works well within the time budget.
    n = 3000
    conv_size = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFT speed

    # --------------------------------------------------------------
    # 2. Build a small pool of analytically motivated initial points.
    candidates = [
        build_power_law(n, p=0.5, offset=0.0),
        build_power_law(n, p=0.4, offset=0.0),
        build_power_law(n, p=0.6, offset=0.0),
        build_linear_descending(n),
        np.full(n, 1.0 / n),
    ]

    # If a previous best sequence is present, reuse its first n entries.
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size < n:

            else:
                prev = prev[:n]
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0:
                prev /= prev.sum()
                candidates.append(prev)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Pick the best seed according to the fast score.
    best_seq = None
    best_val = np.inf
    for cand in candidates:

        if val < best_val:

            best_seq = cand.copy()
    print(f"[init] best seed value = {best_val:.6f}")

    # --------------------------------------------------------------

    a = best_seq.copy()                 # current iterate (on the simplex)
    best_seq = a.copy()
    best_val = fast_score(a)

    # Adam hyper‑parameters
    eta0 = 0.6          # base step‑size

    eps = 1e-8
    m = np.zeros_like(a)   # first moment
