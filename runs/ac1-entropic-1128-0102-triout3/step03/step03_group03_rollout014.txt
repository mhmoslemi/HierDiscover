def softmax_gradient(a, p, top_k=50):
    conv, max_val = _conv_and_max(a)  # conv array
    # compute weight = exp(p * (conv - max_val)) to avoid overflow.
    exp_vals = np.exp(p * (conv - max_val))
    # maybe zero-out small values.
    if top_k is not None and top_k < len(conv):
        # keep only top_k values of exp_vals (largest original conv)
        # get indices of top_k conv values
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        mask = np.zeros_like(exp_vals, dtype=bool)
        mask[top_idx] = True
        exp_vals[~mask] = 0.0
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        # fallback to active set gradient
        K = _active_indices(conv, max_val)
        g = _gradient_active(a, K)
        return g
    weights = exp_vals / sum_exp
    # compute gradient as weighted sum of active per index gradient.
    n = a.shape[0]
    g = np.zeros_like(a)
    # Loop over indices where weight > 0; we can get nonzero weights indices
    nz = np.nonzero(weights)[0]
    for k in nz:
        w = weights[k]
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * w * a[j_idx]
    return g

def weighted_gradient_fft(a, w):
    n = len(a)
    # compute reversed w
    w_rev = w[::-1]  # shape (2n-1,)
    # convolution length via FFT
    size = 1 << ((len(w_rev) + n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * W, n=size)
    # conv length size; we need indices from (2n-2 - (n-1)) to (2n-2) inclusive?
    # conv_rev = conv (since we convolved a and w_rev). The full conv_rev length = len(w_rev)+n-1 = (2n-1) + n - 1 = 3n-2.
    # Access slice:
    start = (len(w_rev) - 1)  # which is (2n-2)
    # Actually conv_rev result at index m (0-index) = sum_{i} a_i * w_rev[m - i].
    # We want conv_rev[2n-2 - i] for i=0..n-1. Since start = (len(w_rev)-1) = 2n-2.
    # So g_i = 2 * conv_rev[start - i]
    # So compute conv_rev = conv[:len(w_rev)+n-1]
    conv_full = conv[:len(w_rev) + n - 1]  # shape 3n-2
    g = 2.0 * conv_full[start - np.arange(n)]
    return g

def _weighted_gradient_fft(a, weights):
    n = a.shape[0]
    # weights length = 2n-1

"""
Propose a step‑function using a multi‑phase scheme:

1. **Initialisation** – start from the previously found sequence
   ``height_sequence_1`` if it exists, otherwise from a random simplex point.
2. **Progressive up‑sampling** – repeatedly enlarge the sequence (≈1.5×)
   up to a target length (default 6000).  On each size we run a
   mirror‑descent on the probability simplex with a *soft‑max* surrogate
   for the max‑convolution.  The surrogate makes the gradient a weighted
   average of many convolution entries instead of a tiny active set,
   giving a more informative direction.  The update is an exponentiated
   gradient (log‑space) with a diminishing step‑size.
3. **Guided stochastic hill‑climbing** – after the largest size is
   reached we perform many local mass‑transfer moves.  The source and
   target indices are chosen from the (approximate) sub‑gradient:
   move mass from indices with large gradient to indices with small
   gradient, accepting strictly improving moves and occasionally a
   worse move according to a simple annealing schedule.
4. **Return** – the best sequence seen (normalised to sum = 1).  The
   evaluator `evaluate_sequence` is scale‑invariant, so the normalisation
   does not affect the score.

The algorithm is deliberately lightweight (no external LP solver) yet
exploits a smoother objective and a longer search budget than the
reference implementation, aiming to push the bound below the target
1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full autocorrelation a*a using FFT.
    Returns a vector of length 2*len(a)-1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_and_max(a: np.ndarray):
    """
    Returns (conv, max(conv)) for the current sequence a.
    """
    conv = _full_convolution(a)
    return conv, conv.max()


def _score_of_seq(a: np.ndarray) -> float:
    """
    Objective used during the search.
    a must be normalised so that sum(a)=1.
    Returns 2 * n * max(conv(a,a)).
    """
    n = a.shape[0]
    conv, max_val = _conv_and_max(a)
    return float(2.0 * n * max_val)


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 8) -> np.ndarray:
    """
    Small set of indices that are either within `rel_eps * max_val` of the
    maximum or, if there are too few, the globally top‑`max_add` indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of `max(conv)` taken on the active set K.
    For each active convolution index k the contribution to component i is
    2 * a_{k-i} (if the index is inside the support).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _weighted_gradient_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} efficiently with FFT.
    `w` has length 2*n-1 (the same length as the full convolution).
    The result `g` has length n.
    """
    n = a.shape[0]
    # Length needed for the convolution of a (len n) with reversed w (len 2n-1)
    fft_len = 1 << ((len(w) + n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w[::-1], n=fft_len)
    prod = np.fft.irfft(Af * Wrev, n=fft_len)
    # Keep only the first len(w)+n-1 entries (full linear convolution)
    full = prod[: len(w) + n - 1]
    # The start index of the reversed slice (the element with index 2n-2)
    start = len(w) - 1
    g = 2.0 * full[start - np.arange(n)]
    return g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (a step function) with a low
    `evaluate_sequence` score.   The routine respects a time budget
    (default 1000 s) and prints occasional progress reports.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise – try to reuse a previously found construction.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)  # may be defined globally
        init = np.maximum(init, 0.0)
        if init.sum() == 0.0:
            raise ValueError
        init = init / init.sum()
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"[{0:6.2f}s] Init from height_sequence_1: n={len(best_seq)}, "
              f"score={best_score:.6f}")
    except Exception:
        # fall back to a random simplex point of moderate size
        n0 = 800
        init = np.random.rand(n0)
        init = init / init.sum()
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"[{0:6.2f}s] Random init: n={n0}, score={best_score:.6f}")

    cur_seq = best_seq.copy()
    cur_n = cur_seq.shape[0]

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling + mirror‑descent with a soft‑max surrogate
    # ------------------------------------------------------------------
    MAX_LEN = 6000                      # target maximal length
    UP_FACTOR = 1.45                    # geometric growth factor
    # parameters for the mirror‑descent phase
    ETA0 = 0.8                          # initial step size (log‑space)
    P_START = 4.0                       # initial soft‑max temperature
    P_MAX = 45.0                        # final temperature
    P_INC_FACTOR = 1.001                # multiplicative increase per iteration
    MIN_ITER = 8000                     # minimum inner iterations per size
    MAX_ITER = 30000                    # cap on inner iterations

    while time.time() < deadline and cur_n < MAX_LEN:
        # decide next length
        nxt_n = int(min(MAX_LEN, cur_n * UP_FACTOR))
        if nxt_n <= cur_n:
            break

        # ----- up‑sample current shape by linear interpolation ----------
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, nxt_n)
        cur_seq = np.interp(x_new, x_old, cur_seq)
        cur_seq = np.maximum(cur_seq, 0.0)
        cur_seq = cur_seq / cur_seq.sum()
        cur_n = nxt_n

        n = cur_n
        # iteration budget adapts loosely to size (more iterations for smaller n)
        iters = max(MIN_ITER, int(MAX_ITER * 8000 / n))
        iters = min(iters, MAX_ITER)

        # initialise mirror‑descent variables
        log_a = np.log(np.maximum(cur_seq, 1e-15))
        p = P_START

        for it in range(1, iters + 1):
            if time.time() >= deadline:
                break

            a = np.exp(log_a)               # back‑to‑vector (always positive)
            a = a / a.sum()                  # enforce simplex

            conv, max_val = _conv_and_max(a)
            cur_score = 2.0 * n * max_val

            # keep the best overall sequence seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"Improved (size={n}) iter={it} score={cur_score:.6f}")

            # ----- compute a smooth surrogate gradient --------------------
            # soft‑max weights on the convolution entries
            shift = p * (conv - max_val)          # max entry becomes 0
            # Clip to avoid underflow when p is large
            shift = np.clip(shift, -50, 0)
            exp_vals = np.exp(shift)
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or np.isnan(sum_exp):
                # fallback to the classic active‑set sub‑gradient
                K = _active_indices(conv, max_val)
                g = _gradient_active(a, K)
            else:
                weights = exp_vals / sum_exp
                g = _weighted_gradient_fft(a, weights)

            # ----- exponentiated‑gradient (mirror descent) update -------
            eta = ETA0 / math.sqrt(it)          # standard diminishing stepsize
            log_a = log_a - eta * g
            # keep numbers numerically stable
            log_a -= np.max(log_a)
            # optionally increase the soft‑max temperature
            if p < P_MAX:
                p = min(P_MAX, p * P_INC_FACTOR)

        # end of inner loop – the last `a` becomes the starting point for the next size
        cur_seq = np.exp(log_a)
        cur_seq = np.maximum(cur_seq, 1e-15)
        cur_seq = cur_seq / cur_seq.sum()

    # ------------------------------------------------------------------
    # 3. Guided stochastic local search (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        cur_score = _score_of_seq(a)

        # annealing schedule for occasional uphill moves
        T0 = 0.02
        LOCAL_ITERS = 60000

        for it_local in range(LOCAL_ITERS):
            if time.time() >= deadline:
                break

            # recompute the active set and its gradient
            conv, max_val = _conv_and_max(a)
            K = _active_indices(conv, max_val)
            g = _gradient_active(a, K)

            # choose source (large gradient) and target (small gradient)
            src = int(np.argmax(g))
            tgt = int(np.argmin(g))
            # with some probability pick completely random indices
            if np.random.rand() < 0.15:
                src, tgt = np.random.randint(0, n, size=2)
                while tgt == src:
                    tgt = np.random.randint(0, n)

            # propose moving a modest fraction of mass
            eps = np.random.rand() * 0.05           # up to 5 % of source
            delta = eps * a[src]
            if delta <= 0.0:
                continue

            a_new = a.copy()
            a_new[src] -= delta
            a_new[tgt] += delta
            if a_new[src] < 0.0:
                continue

            new_score = _score_of_seq(a_new)

            # greedy acceptance, plus occasional uphill moves (simulated annealing)
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # temperature slowly cools
                T = T0 * (1.0 - it_local / LOCAL_ITERS)
                if np.random.rand() < math.exp((cur_score - new_score) / max(T, 1e-6)):
                    accept = True

            if accept:
                a = a_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] "
                          f"Local improve iter={it_local} score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    # Ensure the returned list respects the non‑negative requirement.
    best_seq = np.maximum(best_seq, 0.0)

    # Normalise to sum = 1 (any scaling works for the external evaluator)
    best_seq = best_seq / best_seq.sum()

    return [float(x) for x in best_seq]
