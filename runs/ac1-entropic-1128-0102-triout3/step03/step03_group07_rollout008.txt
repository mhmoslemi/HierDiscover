"""
Propose a candidate step‑function sequence.

Algorithm
---------
* Initialise a on the probability simplex:
  – If the global variable ``height_sequence_1`` exists, use it and
    renormalise it to sum 1.
  – Otherwise start from the half‑binomial series (the classical
    “(1‑z)^‑½’’ coefficients) of length ``n`` (default 5000).

* Optimise on the simplex using an Adam optimiser.
  – The objective is 2·n·max(conv(a,a)), but we smooth the max with a
    soft‑max (log‑sum‑exp) while ``β`` is increased.
  – The gradient of the smoothed max is obtained by a single
    correlation via FFT (O(N log N)).
  – Once ``β`` reaches ``β_target`` we switch to the exact max‑sub‑gradient,
    i.e. an equal weight on all active indices where the convolution attains its
    maximum.

* After each Adam step the iterate is projected back onto the probability
  simplex (Euclidean projection, Duchi et al.).

* Random perturbations and occasional restarts prevent getting stuck in
  poor local minima.

* The best sequence seen during the run is stored and finally returned
  (as a plain Python list).  The true objective value is printed using the
  provided ``evaluate_sequence`` routine.

The routine respects the 1000 s time budget (default 950 s is used as a
safety margin) and works with a single CPU.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^‑½  (a_i = C(2i,i)/4^i) and normalised
    so that they sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

            s = a0.sum()
            if s <= 0:
                a = half_binomial_series(5000)
                n = a.size
            else:
                a = a0 / s
                n = a.size
    else:
        n = 5000                     # length of the sequence (feel free to adjust)
        a = half_binomial_series(n)  # already sums to 1

    # ------------------------------------------------------------------
    # static sizes for FFT based convolution / correlation
    L = 2 * n - 1                 # length of a*a (linear convolution)
    Nfft = next_pow_two(3 * n - 2)  # length needed for correlation a * rev(w)

    # keep the best solution seen so far
    best_a = a.copy()
    # surrogate objective (2·n·max(conv)) – fast to compute
    A = np.fft.rfft(a, n=Nfft)
    conv_full = np.fft.irfft(A * A, n=Nfft)
    conv = conv_full[:L]
    best_score = 2.0 * n * conv.max()
    print(f"[init] n={n}, initial surrogate score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) optimisation hyper‑parameters
    beta = 1.0                     # start of the soft‑max smoothing
    beta_target = 1200.0           # final β (large → true max)
    beta_mul = 1.003               # geometric increase per iteration

    # Adam parameters
    eta = 0.12                     # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0                          # iteration counter for bias correction

    # restart / perturbation control
    last_improve = 0
    restart_gap = 250_000          # iterations without improvement → restart
    noise_prob = 0.006
    noise_scale = 1.4e-4

    # timing
    deadline = time.time() + budget_s - 5.0   # leave a small safety margin
    it = 0

    # optional early exit target (the problem goal)
    target_score = 1.503

    # ------------------------------------------------------------------
    # main optimisation loop
    while time.time() < deadline:
        # --------------------------------------------------------------
        # compute convolution a*a (linear part) via FFT
        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv      # surrogate objective

        # --------------------------------------------------------------
        # store best solution
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:09d}] new best surrogate score = {best_score:.6f}")

        # early exit if we have already passed the target (with some tolerance)
        if cur_score < target_score:
            print(f"[iter {it:09d}] target reached (surrogate score {cur_score:.6f}) – stopping early")
            break

        # --------------------------------------------------------------
        # determine the weight vector w (soft‑max or pure max)
        if beta < beta_target:
            # soft‑max smoothing
            shift = max_conv                         # improve numerical stability
            w_exp = np.exp(beta * (conv - shift))
            w = w_exp / w_exp.sum()
            # increase β
            beta = min(beta_target, beta * beta_mul)
        else:
            # pure max – use all active indices (within a tiny tolerance)
            eps_active = 1e-12
            active = np.where(conv >= max_conv - eps_active)[0]
            w = np.zeros_like(conv)
            w[active] = 1.0 / len(active)

        # --------------------------------------------------------------
        # gradient = 2 * correlation(a, w_rev)
        # (the correlation is obtained via a single FFT with the reversed w)
        Wr = np.fft.rfft(w[::-1], n=Nfft)
        corr_full = np.fft.irfft(A * Wr, n=Nfft)
        start = L - 1               # index where shift 0 begins
        grad = 2.0 * corr_full[start:start + n]   # shape (n,)

        # --------------------------------------------------------------
        # Adam update
        it += 1
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # clamp the step so we never subtract more than 95 % of a component
        step = np.minimum(step, 0.95 * a)

        # take the step and re‑project onto the simplex
        a = a - step
        a = project_onto_simplex(a)

        # --------------------------------------------------------------
        # occasional random perturbation (helps escape flat regions)
        if rng.random() < noise_prob:
            noise = (rng.random(n) - 0.5) * noise_scale
            a = a + noise
            a = project_onto_simplex(a)

        # --------------------------------------------------------------
        # restart if stagnation is observed
        if it - last_improve > restart_gap:
            a = rng.dirichlet(np.full(n, 0.6))
            a = project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = it
            print(f"[iter {it:09d}] restart – new random start, surrogate score≈{2.0*n*np.max(np.fft.irfft(np.fft.rfft(a,n=Nfft)**2,n=Nfft)[:L]):.6f}")

    # ------------------------------------------------------------------
    # final evaluation with the official routine (adds the security checks)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
