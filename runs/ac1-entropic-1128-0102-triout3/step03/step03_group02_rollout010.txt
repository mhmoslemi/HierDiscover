def _refine_adam(a, max_iters, deadline, mode='softmax', lr=0.1, eps_start=0.05, eps_end=0.001):
    n = a.shape[0]
    a = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = _score_of_seq(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute conv via FFT
        fft_len_conv = 1 << ((2*n - 1).bit_length())
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        if mode == 'softmax':
            tau = max(eps_start + (eps_end - eps_start) * (it / max_iters), 1e-14) * max_val
            w = np.exp((conv - max_val) / tau)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.full_like(w, 1.0 / w.shape[0])
        else: # hardmax
            # indicator
            thresh = max_val - 1e-12
            mask = (conv >= thresh).astype(float)
            w_sum = mask.sum()
            if w_sum > 0:
                w = mask / w_sum
            else:
                w = np.full_like(conv, 1.0 / conv.shape[0])
        # compute gradient g = 2 * convolution of w with a_rev
        a_rev = a[::-1]
        fft_len_grad = 1 << ((len(w) + n - 1).bit_length())
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        g = 2.0 * conv2[n-1: n-1 + n]  # length n
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = _project_simplex(a)
        # Clip maybe to max 1000? Not needed
    return a, best_a, best_score

Propose a low‑score step‑function using a multi‑stage, Adam‑driven
sub‑gradient descent on the L∞‑norm of the auto‑convolution.
The algorithm works on the probability simplex (∑aᵢ = 1, aᵢ ≥ 0) so the
score returned by ``evaluate_sequence`` is simply

    score = 2·n·max(conv(a,a))

where ``conv`` is the full linear auto‑convolution.  The optimisation proceeds
as follows:

1.  Initialise a diverse pool of candidates (uniform, triangular, Hann,
    Blackman, a random Dirichlet and the previous best solution if available)
    and normalise each to the simplex.
2.  Refine every candidate with two Adam passes:
        – a soft‑max smoothing of the max‑operator (temperature decays)
        – a hard‑max sub‑gradient (weights are the active constraints)
3.  Iteratively up‑sample the current global best (≈1.5× larger each step)
    and repeat the two‑phase refinement, allocating fewer iterations to larger
    vectors so the total runtime stays within the 1000 s budget.
4.  Perform a final fine‑tuning on the best length found and a few
    random perturbations to escape possible local minima.
5.  Return the best sequence as plain Python floats.

All updates are projected onto the simplex (Euclidean projection) to keep
feasibility; Adam’s adaptive step‑sizes provide faster convergence than the
simple momentum used previously.  The implementation is fully self‑contained,
uses only NumPy and the supplied ``evaluate_sequence`` for the final check,
and respects the time limit.

import time
import math

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Scoring (internal, assumes sum(a)=1)
def _score_of_seq(a: np.ndarray) -> float:
    """Return the internal objective 2·n·max(conv(a,a)) for a simplex‑normalised a."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()

# Gradient from a weighted convolution (weights w sum to 1)
def _conv_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of  Σ_k w_k (a∗a)_k  w.r.t. a.
    The result is 2·(w ∗ a_rev) restricted to the original length.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # length of convolution w (len = 2n-1) with a_rev (len=n) -> 3n-2
    conv_len = len(w) + n - 1
    fft_len = 1 << (conv_len.bit_length())
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(W * A_rev, n=fft_len)[:conv_len]
    # extract the central n entries that correspond to gradient components
    g = 2.0 * conv[n - 1 : n - 1 + n]
    return g

# ----------------------------------------------------------------------
# Adam‑driven refinement (soft‑max or hard‑max mode)
def _refine_adam(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    mode: str = "softmax",
    lr: float = 0.1,
    eps_start: float = 0.05,
    eps_end: float = 0.001,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Refine a simplex‑normalised vector a using Adam updates.

    mode:
        "softmax" – w is a soft‑max over convolution entries (temperature decays)
        "hardmax" – w is uniform on the set of maximal entries (exact sub‑gradient)

    Returns (final_a, best_seen_a, best_score).
    """
    n = a.shape[0]
    a = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution and current score -----
        fft_len = 1 << ((2 * n - 1).bit_length())
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- weight vector w -----
        if mode == "softmax":
            # temperature proportional to current max (smooth annealing)
            tau = max(
                eps_start + (eps_end - eps_start) * (it / max_iters), 1e-14
            ) * max_val
            w = np.exp((conv - max_val) / tau)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.full_like(conv, 1.0 / conv.shape[0])
        else:  # hardmax
            # all entries within a tiny tolerance of the maximum share the weight
            thresh = max_val - 1e-12
            mask = (conv >= thresh).astype(float)
            w_sum = mask.sum()
            if w_sum > 0:
                w = mask / w_sum
            else:
                w = np.full_like(conv, 1.0 / conv.shape[0])

        # ----- gradient -----
        g = _conv_gradient(a, w)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # projection back onto simplex
        a = _project_simplex(a)

    return a, best_a, best_score

# ----------------------------------------------------------------------
# Small random perturbation (Gaussian) followed by a short Adam refinement
def _perturb_and_refine(
    a: np.ndarray,
    scale: float,
    iters: int,
    deadline: float,
    mode: str = "softmax",
) -> tuple[np.ndarray, float]:
    """Create a perturbed copy of a, run a short refinement, return best copy."""
    n = a.shape[0]
    noisy = a + scale * np.random.randn(n)
    noisy = np.maximum(noisy, 0.0)
    if noisy.sum() == 0:
        noisy = np.full(n, 1.0 / n)
    else:
        noisy = noisy / noisy.sum()
    _, best_local, best_sc = _refine_adam(
        noisy,
        max_iters=iters,
        deadline=deadline,
        mode=mode,
        lr=0.08,
        eps_start=0.03,
        eps_end=0.0005,
    )
    return best_local, best_sc

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Main entry‑point.  Runs a time‑bounded multi‑scale optimisation and returns
    a list of non‑negative heights with a low evaluate_sequence score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final checks

    # ------------------------------------------------------------------
    # 1. Assemble diverse initial candidates (all normalised to the simplex)
    # ------------------------------------------------------------------
    n0 = 2000                                   # initial resolution
    candidates = []

    # previous best (if present)
    try:
        prev = np.array(height_sequence_1, dtype=float)
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            # if it is shorter/longer than n0, resample it to n0
            if prev.shape[0] != n0:
                x_old = np.linspace(0.0, 1.0, prev.shape[0])
                x_new = np.linspace(0.0, 1.0, n0)
                prev = np.interp(x_new, x_old, prev)
                prev = np.maximum(prev, 0.0)
                if prev.sum() == 0:
                    prev = np.full(n0, 1.0 / n0)
                else:
                    prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    # uniform
    candidates.append(np.full(n0, 1.0 / n0))

    # triangular (tent)
    idx = np.arange(n0)
    tri = np.minimum(idx + 1, n0 - idx)
    candidates.append(tri / tri.sum())

    # Hann window
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n0 - 1)))
    candidates.append(hann / hann.sum())

    # Blackman window (more aggressive side‑lobe suppression)
    black = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * idx / (n0 - 1))
        + 0.08 * np.cos(4.0 * np.pi * idx / (n0 - 1))
    )
    candidates.append(black / black.sum())

    # random Dirichlet point
    candidates.append(np.random.dirichlet(np.ones(n0)))

    # ------------------------------------------------------------------
    # 2. Refine each candidate on the initial grid
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    for i, cand in enumerate(candidates, 1):
        if time.time() >= deadline:
            break

        print(
            f"[{time.time() - start_time:6.2f}s] Starting refinement of candidate {i}/{len(candidates)}"
        )

        # soft‑max phase
        _, best_soft, sc_soft = _refine_adam(
            cand,
            max_iters=18000,
            deadline=deadline,
            mode="softmax",
            lr=0.12,
            eps_start=0.07,
            eps_end=0.001,
        )
        # hard‑max phase (more aggressive near the true objective)
        _, best_hard, sc_hard = _refine_adam(
            best_soft,
            max_iters=18000,
            deadline=deadline,
            mode="hardmax",
            lr=0.07,
        )

        if sc_hard < best_score:
            best_score = sc_hard
            best_seq = best_hard.copy()
            print(f"   New global best from candidate {i}: score={sc_hard:.6f}")

    # ------------------------------------------------------------------
    # 3. Progressive up‑sampling and refinement
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)

    current = best_seq.copy()
    cur_n = current.shape[0]
    MAX_LEN = 12000                                 # upper bound for up‑sampling

    while True:
        next_n = int(cur_n * 1.5)
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n <= cur_n:
            break

        # Linear interpolation up‑sampling
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a = np.interp(x_new, x_old, current)
        a = np.maximum(a, 0.0)
        a = a / a.sum() if a.sum() > 0 else np.full(next_n, 1.0 / next_n)

        # Allocate a modest number of iterations (more for smaller sizes)
        base_iters = 25000
        iters = max(int(base_iters * n0 / next_n) + 10000, 12000)

        print(
            f"[{time.time() - start_time:6.2f}s] Upsampling to n={next_n}, iters≈{iters}"
        )

        # Soft‑max refinement
        _, best_soft, sc_soft = _refine_adam(
            a,
            max_iters=iters,
            deadline=deadline,
            mode="softmax",
            lr=0.11,
            eps_start=0.06,
            eps_end=0.001,
        )
        # Hard‑max refinement
        _, best_hard, sc_hard = _refine_adam(
            best_soft,
            max_iters=iters,
            deadline=deadline,
            mode="hardmax",
            lr=0.07,
        )

        if sc_hard < best_score:
            best_score = sc_hard
            best_seq = best_hard.copy()
            print(
                f"   Updated global best after up‑sampling: n={next_n}, score={sc_hard:.6f}"
            )

        # Small random shake‑ups to possibly escape shallow minima
        for _ in range(3):
            if time.time() >= deadline:
                break
            perturbed, sc_pert = _perturb_and_refine(
                best_seq,
                scale=0.03,
                iters=4000,
                deadline=deadline,
                mode="softmax",
            )
            if sc_pert < best_score:
                best_score = sc_pert
                best_seq = perturbed.copy()
                print(
                    f"   Improvement via perturbation at n={next_n}: score={sc_pert:.6f}"
                )

        current = best_hard.copy()
        cur_n = next_n

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final fine‑tuning on the best length found
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_final = best_seq.shape[0]
        final_iters = max(int(40000 * n0 / n_final) + 20000, 25000)

        print(
            f"[{time.time() - start_time:6.2f}s] Final refinement on n={n_final}, iters≈{final_iters}"
        )
        # A final soft‑max pass
        _, best_soft, sc_soft = _refine_adam(
            best_seq,
            max_iters=final_iters,
            deadline=deadline,
            mode="softmax",
            lr=0.09,
            eps_start=0.04,
            eps_end=0.0005,
        )
        # Followed by a hard‑max pass
        _, best_hard, sc_hard = _refine_adam(
            best_soft,
            max_iters=final_iters // 2,
            deadline=deadline,
            mode="hardmax",
            lr=0.06,
        )
        if sc_hard < best_score:
            best_score = sc_hard
            best_seq = best_hard.copy()
            print(f"   Final improvement: score={sc_hard:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity check using the provided evaluator (optional)
    # ------------------------------------------------------------------
    try:
        eval_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] evaluate_sequence reports {eval_score:.6f}")
    except Exception:
        eval_score = None

    # Ensure tidy output
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    total_time = time.time() - start_time
    print(
        f"[{total_time:6.2f}s] Finished. Best internal score = {best_score:.6f}"
    )
    if eval_score is not None:
        print(f"   (Score reported by evaluate_sequence = {eval_score:.6f})")

    return [float(x) for x in best_seq]
