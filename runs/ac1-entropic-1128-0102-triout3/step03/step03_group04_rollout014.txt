def mixed_power_uniform_sequence(n, p, offset, alpha):
    i = np.arange(n, dtype=np.float64)
    base = (i + offset) ** (-p)
    uniform = np.ones_like(i)
    a = alpha * base + (1 - alpha) * uniform

    return a

"""
propose_candidate – minimise 2·n·max(conv(a,a)) / (sum(a)^2)

The routine works in three stages:

1. **Coarse parametrised sweep** – candidates are generated from a
   power‑law a_i ∝ (i+offset)^(–p) and from a simple mixture
   (α·power‑law + (1‑α)·uniform).  All vectors are normalised to L1‑norm 1.
   The best vector from this cheap search starts the local optimiser.

2. **Projected Adam optimisation** – a full‑gradient of the smoothed
   maximum (soft‑max over the whole convolution) is computed efficiently
   via FFTs:  grad = 2·(rev(a) * w) where w are the soft‑max weights.
   Adam’s adaptive moments and a projection onto the probability simplex
   keep the iterates feasible.  The temperature β of the soft‑max is
   increased slowly so that the optimisation first smooths the landscape
   and then focuses on the true peak.

3. **Final check** – the best vector found is evaluated with the official
   ``evaluate_sequence`` routine and returned as a plain list.

The implementation respects a user supplied time budget (default 950 s,
leaving a safety margin) and prints progress messages.
"""

import time
import numpy as np

# Basic utilities
# ----------------------------------------------------------------------


def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+offset)^(-p), then normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:                # fallback to uniform
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def mixed_power_uniform_sequence(
    n: int, p: float, offset: float = 1.0, alpha: float = 0.5
) -> np.ndarray:
    """α·power‑law + (1‑α)·uniform, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    base = (i + offset) ** (-p)
    uniform = np.ones_like(i)
    a = alpha * base + (1.0 - alpha) * uniform
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full self‑convolution x*x using a real FFT (length 2·len(x)-1)."""
    n = x.shape[0]
    m = 2 * n - 1
    # next power of two for efficiency
    size = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(X * X, n=size)
    return conv[:m]


def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary 1‑D arrays via FFT."""
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {w ≥ 0, Σw = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def weighted_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max smoothed max‑convolution.
    w_i = exp(beta·(conv_i - max(conv))) / Σ_j exp(...).
    grad = 2·(rev(a) * w)   (linear convolution, slice of length n).
    """
    maxv = conv.max()
    # soft‑max weights (shifted by max to avoid overflow)
    w = np.exp(beta * (conv - maxv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    a_rev = a[::-1]
    grad_full = full_convolution_fft(a_rev, w)          # length 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * grad_full[start : start + a.shape[0]]
    return grad

    best_a = None
    best_score = np.inf
    best_params = None

    # --------------------------------------------------- phase 1 : parametric sweep
    print("[search] phase 1 – power‑law & mixed sweep")
    p_vals = np.arange(0.35, 0.75, 0.025)       # 0.35 … 0.73
    offsets = np.arange(0.1, 3.1, 0.3)          # 0.1 … 2.9
    alphas = [0.0, 0.25, 0.5, 0.75, 1.0]       # mixture weight

    # pure power‑law candidates
    for p in p_vals:
        for off in offsets:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=off)
            cur = 2.0 * n * conv_fft(a).max()
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                best_params = ("pure", p, off, None)
                print(f"[grid] new best (pure) p={p:.3f} off={off:.2f} score={cur:.6f}")

    # mixed power‑law / uniform candidates
    for p in p_vals:
        for off in offsets:
            for alpha in alphas:
                if time.time() > deadline:
                    break
                a = mixed_power_uniform_sequence(n, p, offset=off, alpha=alpha)
                cur = 2.0 * n * conv_fft(a).max()
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    best_params = ("mix", p, off, alpha)
                    print(
                        f"[grid] new best (mix) p={p:.3f} off={off:.2f} α={alpha:.2f} score={cur:.6f}"
                    )

    # --------------------------------------------------- fallback if sweep failed
    if best_a is None:
        print("[fallback] using half‑binomial seed")
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
        a /= a.sum()
        best_a = a
        best_score = 2.0 * n * conv_fft(a).max()
        best_params = ("half_binomial", None, None, None)

    # --------------------------------------------------- phase 2 : projected Adam optimisation
    print("[adam] phase 2 – projected Adam descent")
    a = best_a.copy()                     # current iterate
    cur_score = 2.0 * n * conv_fft(a).max()

    # Adam hyper‑parameters
    lr = 0.07                # base learning rate
    decay_factor = 0.90
    decay_every = 6_000
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta = 1.0               # start of soft‑max temperature
    beta_inc = 0.001         # increase per decay step
    max_iters = 500_000

    m = np.zeros_like(a)    # first moment
    v = np.zeros_like(a)    # second moment

    best_local_a = a.copy()
    best_local_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a)
        grad = weighted_gradient(a, conv, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_onto_simplex(a_new)

        # occasional tiny jitter to escape flat plateaux
        if rng.random() < 0.001:
            jitter = (rng.random(n) - 0.5) * 1e-6
            a_new = project_onto_simplex(a_new + jitter)

        new_score = 2.0 * n * conv_fft(a_new).max()

        # keep the best ever seen
        if new_score < best_local_score:
            best_local_score = new_score
            best_local_a = a_new.copy()
            print(
                f"[adam iter {it:07d}] new best local score = {best_local_score:.6f}"
            )

        a = a_new
        cur_score = new_score

        # scheduled decay of step size and increase of β
        if it % decay_every == 0:
            lr = max(lr * decay_factor, 1e-6)
            beta += beta_inc

    # --------------------------------------------------- final evaluation
    final_score = evaluate_sequence(best_local_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")
    print(f"[info] best parameters from sweep: {best_params}")

    return best_local_a.tolist()
