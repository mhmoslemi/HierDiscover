Pseudo:

def project_onto_simplex(v):
    if v.sum() == 1 and (v >= 0).all(): return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def compute_grad_softmax(a, alpha):
    n = a.shape[0]
    # compute convolution a*a
    conv_len = 2*n-1
    # compute FFT length as next power of 2 >= (n + conv_len - 1) = 3n-2 for convolution a*w
    fft_len = 1 << ((3*n - 2).bit_length())
    # compute autocorrelation
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:conv_len]
    # compute softmax weights
    maxc = conv.max()
    # compute exponentials scaling to avoid overflow
    exp_vals = np.exp(alpha * (conv - maxc))
    w = exp_vals / exp_vals.sum()
    # compute convolution of a with w (size conv_len + n -1 = 3n-2)
    Wf = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(Af * Wf, n=fft_len)[:(n + conv_len - 1)]
    grad = 2 * conv_aw[:n]  # first n entries correspond to i from 0..n-1
    return grad, conv.max()

def project_onto_simplex(v):
    # ensure v is numpy array
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all zero? then return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v):
    """Project a vector v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all entries are zero or negative -> return uniform
        w = np.full_like(v, 1.0 / n)
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # keeps best seq inside this length
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.2   # tune
    # iteration budget
    max_iters = 25000 if n <= 3000 else 18000 if n <= 4000 else 15000
    # Or compute based on time left: allocate maybe (deadline - time.time())/2 iterations left average each iteration ~ 0.001? We can approximate.
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute convolution and max
        conv, max_conv = compute_conv_and_max(a)  # shape (2n-1)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] length={n} it={it} new best {cur_score:.6f}")
        # temperature schedule
        # Use quadratic increase up to alpha_max after half the iterations
        alpha_min = 5.0
        alpha_max = 2000.0
        if it <= max_iters // 2:
            t = it / (max_iters // 2)
            alpha = alpha_min + (alpha_max - alpha_min) * (t**2)
        else:
            alpha = alpha_max
        # compute softmax weights
        # shift by max to avoid overflow
        exps = np.exp(alpha * (conv - max_conv))
        w = exps / exps.sum()
        # compute gradient via correlation
        grad = grad_corr(a, w)
        # Adam update
        lr = base_lr / math.sqrt(it)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # gradient descent step
        a = a - step
        # enforce positivity and simplex
        a = np.maximum(a, 1e-12)
        # project onto simplex
        a = project_onto_simplex(a)
        # keep minimal floor again
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
    return a, best_seq, best_score

"""
Propose a candidate step‑function by minimizing a smoothed version of the
max‑auto‑convolution objective.

The algorithm works in three stages:

1. **Initialisation** – use the globally supplied `height_sequence_1`
   (if present) or a random point on the probability simplex.
2. **Smoothed Adam optimisation** – minimise
   `F(a) = (1/α)·log Σ_k exp(α·(a*a)_k)` using Adam on the simplex.
   The temperature α is increased from a small value (≈5) to a large one
   (≈2000), so that the smooth objective turns into the true max.
   Gradients are obtained analytically via a correlation of `a` with the
   soft‑max weights, using FFTs for O(n log n) work.
3. **Local stochastic refinement** – cheap random mass‑transfer moves
   (moving a few percent of the mass from one index to another) are
   accepted only if they improve the exact score
   `2·n·max(conv(a,a))`.  The best sequence seen over all phases is
   returned as a plain Python list of non‑negative floats.

All updates keep the vector non‑negative and on the simplex (sum = 1).
The routine respects the 1000 s budget and prints progress
information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (components ≥ 0, sum = 1).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive ⇒ uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(a: np.ndarray) -> float:
    """
    Exact objective value `2·n·max(conv(a,a))` for a vector that already sums to 1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution of a (real FFT) and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _corr_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of `Σ_k w_k·(a*a)_k` with respect to a.
    This is 2·(a ⋆ w) where ⋆ denotes (linear) correlation.
    """
    n = a.shape[0]
    conv_len = w.shape[0]                     # = 2·n‑1
    fft_len = 1 << ((n + conv_len - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    Wf = np.fft.rfft(w, n=fft_len)
    # correlation = ifft(conj(A) * W)
    corr = np.fft.irfft(np.conj(Af) * Wf, n=fft_len)[: n + conv_len - 1]
    return 2.0 * corr[:n]                     # first n entries = desired gradient

def _optimise_length(a_init: np.ndarray,
                     best_seq: np.ndarray,
                     best_score: float,
                     deadline: float,
                     start_time: float):
    """
    Adam optimisation of the smoothed max‑convolution objective for a fixed
    length.  Returns the (possibly) improved vector together with the global
    best sequence and its score.
    """
    a = a_init.copy()
    n = a.shape[0]

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.25                # base learning rate before decay

    # iteration budget (scaled with dimension)
    if n <= 2500:
        max_iters = 26000
    elif n <= 3500:
        max_iters = 21000
    else:
        max_iters = 15000

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution, soft‑max weights, gradient -----------------
        # use a single large FFT length for both convolution and correlation
        fft_len = 1 << ((3 * n - 2).bit_length())
        Af = np.fft.rfft(a, n=fft_len)

        conv_full = np.fft.irfft(Af * Af, n=fft_len)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] length={n:<4d} "
                  f"it={it:<5d} new best={cur_score:.6f}")

        alpha_min, alpha_max = 5.0, 2000.0
        if it <= max_iters // 2:
            t = it / (max_iters // 2)
            alpha = alpha_min + (alpha_max - alpha_min) * (t ** 2)
        else:
            alpha = alpha_max

        exps = np.exp(alpha * (conv - max_conv))
        w = exps / exps.sum()

        # gradient via correlation
        grad = _corr_gradient(a, w)

        # ----- Adam update (gradient descent) -------------------------
        lr = base_lr / math.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step                     # descent direction

        # keep the vector on the simplex and strictly positive
        a = np.maximum(a, 1e-12)
        a = _project_onto_simplex(a)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()

        # occasional jitter to escape shallow plateaus
        if it % 2000 == 0:
            noise = np.random.randn(n) * 0.005
            a = np.maximum(a + noise, 1e-12)
            a = _project_onto_simplex(a)

    return a, best_seq, best_score


def _local_search(a: np.ndarray,
                  best_seq: np.ndarray,
                  best_score: float,
                  deadline: float,
                  start_time: float,
                  max_iters: int = 15000):
    """
    Simple stochastic hill‑climbing: move a few percent of the mass from one
    index to another and keep the move if the exact score improves.
    """
    n = a.shape[0]
    cur = a.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = np.random.rand() * 0.04          # up to 4 % of source mass
        delta = eps * cur[i]
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # compute exact score (no projection needed – sum unchanged)
        conv, max_conv = _conv_and_max(cand)
        new_score = 2.0 * n * max_conv
        if new_score < best_score:
            best_score = new_score
            best_seq = cand.copy()
            cur = cand
            print(f"[{time.time() - start_time:6.2f}s] local improve n={n:<4d} "
                  f"it={it:<5d} score={new_score:.6f}")
    return cur, best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low value of the provided
    `evaluate_sequence` objective.

    The search employs a smoothed max‑convolution objective optimised by Adam
    on the probability simplex, followed by a short stochastic local‑search
    phase.  The best sequence found within the time budget is returned as a
    plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0               # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise (use the known construction if available)
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # may be defined globally
        init = np.maximum(init, 0.0)
        if init.sum() == 0.0:
            raise ValueError
        init = init / init.sum()
        print(f"Initialised from height_sequence_1 (length={init.shape[0]})")
    except Exception:
        n0 = 4000
        init = np.random.rand(n0)
        init = init / init.sum()
        print(f"Random initialisation (length={n0})")

    MAX_LEN = 5000
    if init.shape[0] > MAX_LEN:
        # down‑sample to the allowed maximum length
        x_old = np.linspace(0.0, 1.0, init.shape[0])
        x_new = np.linspace(0.0, 1.0, MAX_LEN)
        init = np.interp(x_new, x_old, init)
        init = init / init.sum()
    best_seq = init.copy()
    best_score = _fast_score(best_seq)
    print(f"Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Main optimisation on the (current) length
    # ------------------------------------------------------------------
    a_opt, best_seq, best_score = _optimise_length(init, best_seq,
                                                   best_score, deadline,
                                                   start_time)

    # ------------------------------------------------------------------
    # 3. Short stochastic refinement (mass transfers)
    # ------------------------------------------------------------------
    a_opt, best_seq, best_score = _local_search(a_opt, best_seq,
                                                best_score, deadline,
                                                start_time, max_iters=12000)

    # ------------------------------------------------------------------
    # 4. Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        # fallback – uniform 100‑point vector
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
