def _gradient_logsumexp(a, conv, lam):
    n = a.shape[0]
    max_val = conv.max()
    exp_term = np.exp((conv - max_val) / lam)
    weights = exp_term / exp_term.sum()   # shape (2n-1)
    # compute correlation with reversed a
    a_rev = a[::-1]
    # conv of weights and a_rev yields length 3n-2 (full)
    # Use fft
    size = len(weights) + len(a_rev) - 1
    nfft = 1 << ((size - 1).bit_length())
    W = np.fft.rfft(weights, n=nfft)
    A = np.fft.rfft(a_rev, n=nfft)
    full = np.fft.irfft(W * A, n=nfft)[:size]
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad
```

def _run_annealing(a0, n, deadline, start_time, cur_best_score, cur_best_seq,
                   start_temp=5e-2, end_temp=1e-6, max_steps=200_000):
    a = a0.copy()

    max_conv = conv.max()
    score = 2.0 * n * max_conv
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    it = 0
    while time.time() < deadline and it < max_steps:
        it += 1
        # temperature schedule
        T = start_temp * ((end_temp / start_temp) ** (it / max_steps))
        # pick source i with a[i]>0
        i = np.random.randint(0, n)
        if a[i] <= 0:

        j = np.random.randint(0, n)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.2  # up to 20%

        # compute new score via conv (by recompute conv maybe heavy)

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().  The algorithm combines several ideas:

*  **Probability‑simplex parametrisation** – we optimise logits y and obtain
   the step heights a = softmax(y), guaranteeing a_i≥0 and Σa_i=1.
*  **Adam optimisation** on a smooth log‑sum‑exp surrogate of the
   ℓ∞‑norm  max(a*a).  The surrogate is  L_λ(a)=λ·log∑exp(conv/λ)  and
   λ is gradually reduced, so the optimisation converges to the true
   max‑norm problem.
*  **Multi‑scale warm‑starts** – we start with short vectors (≈1 k),
   improve them, then linearly up‑sample to longer lengths (≈16 k).
   Several deterministic initial shapes (Hann, triangular, Gaussian) and
   random restarts are tried.
*  **Final simulated‑annealing polish** – a cheap mass‑transfer annealer
   explores the neighbourhood of the best vector found by Adam.
*  **Time‑aware scheduling** – the 1000 s budget is split among the
   scales; everything stops automatically when the deadline is reached.

The whole routine runs on a single CPU and uses only NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (real‑to‑real) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = 2 * n - 1
    nfft = _next_pow2(size)
    Af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(Af * Af, n=nfft)[:size]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real‑to‑real)."""
    size = x.shape[0] + y.shape[0] - 1
    nfft = _next_pow2(size)
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


# ----------------------------------------------------------------------
# Initial shapes (deterministic)
# ----------------------------------------------------------------------


def _init_hann(n: int) -> np.ndarray:
    """Hann window (zero at the ends), normalised to sum = 1."""
    w = np.hanning(n)
    if w.sum() == 0:
        w = np.ones(n)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _init_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, normalised to sum = 1."""
    if n % 2 == 0:
        up = np.arange(1, n // 2 + 1)
        down = up[::-1]
    else:
        up = np.arange(1, (n + 1) // 2 + 1)
        down = up[:-1][::-1]
    w = np.concatenate([up, down]).astype(float)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _init_gaussian(n: int, sigma: float = None) -> np.ndarray:
    """Discrete Gaussian (unnormalised), normalised to sum = 1."""
    if sigma is None:
        sigma = n / 6.0
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0.0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _grad_logsumexp(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of   L_λ(a)=λ·log∑exp(conv/λ)   w.r.t. a.
    Parameters
    ----------
    a   : (n,)   probability vector (Σa=1)
    conv: (2n‑1,) self‑convolution a*a
    lam : smoothing parameter λ>0
    Returns
    -------
    g   : (n,)   gradient ∇_a L_λ
    """
    n = a.shape[0]

    # soft‑max weights over convolution entries (stable)
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()                     # shape (2n‑1,)

    # correlation of w with reversed a gives ∂conv/∂a (dense)
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)      # length = 3n‑2
    start = n - 1
    g = 2.0 * full[start:start + n]  # 2·∂conv/∂a weighted sum
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (via softmax logits)
# ----------------------------------------------------------------------


def _run_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate.
    Returns the best vector (sum = 1) and the corresponding exact score
    2·n·max(a*a) observed during the run.
    """
    # initialise logits (softmax parameters)
    eps = 1e-12
    y = np.log(np.maximum(a0, eps))

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    lr = 0.15      # base learning rate for logits
    it = 0

    best_a = None
    best_score = float("inf")

    # λ‑schedule parameters
    lam_factor = 0.5          # λ = lam_factor * max(conv)
    lam_decay = 0.6
    lam_decay_interval = 5000

    while time.time() < deadline and it < max_iter:
        it += 1

        # softmax → probability vector a (sum = 1)
        y_max = y.max()
        y_shift = y - y_max
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # convolution & exact objective
        conv = _conv_self(a)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c

        # keep the best seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, it={it}, score={cur_score:.6f}"
            )

        # update λ (smoothness)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)

        lam = max(max_c * lam_factor, 1e-12)

        # gradient of the surrogate w.r.t. a
        grad_a = _grad_logsumexp(a, conv, lam)

        # chain rule: ∂/∂y = a * (grad_a – <a,grad_a>)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    # safety – if no improvement was recorded, fall back to the last a
    if best_a is None:
        best_a = a
        best_score = 2.0 * n * _conv_self(best_a).max()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny mass‑transfer simulated annealing
# ----------------------------------------------------------------------


def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    max_steps: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing where a small ε is moved from one coordinate
    to another.  The total mass (hence Σa) stays constant and all entries
    remain non‑negative.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # propose a move
        i = np.random.randint(0, n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(0, n)
        if i == j:
            continue

        eps = a[i] * np.random.rand() * 0.2  # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # evaluate new score (full convolution – cheap enough for modest n)
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, it={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    evaluate_sequence().  The algorithm runs for at most ``budget_s`` seconds,
    using a multi‑scale Adam optimiser followed by a final annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # If the harness supplied a previous construction, use it as warm‑start
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Multiscale schedule (geometric growth, modest upper bound)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]
    # shuffle a little – if we run out of time we still have a variety
    np.random.shuffle(lengths)

    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Build a modest set of initial candidates for this scale
        # --------------------------------------------------------------
        candidates = []

        if global_a is not None:
            candidates.append(_upsample(global_a, n))
        else:
            # uniform random start
            a = np.random.rand(n)
            a /= a.sum()
            candidates.append(a)

        # deterministic windows
        candidates.append(_init_hann(n))
        candidates.append(_init_triangular(n))
        candidates.append(_init_gaussian(n))

        # a few extra random restarts
        for _ in range(3):
            a = np.random.rand(n)
            a /= a.sum()
            candidates.append(a)

        # --------------------------------------------------------------
        # Allocate a time slice for this resolution
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # give at least a few seconds, otherwise a proportion of the remaining
        slice_deadline = time.time() + max(6.0, 0.08 * remaining)

        # --------------------------------------------------------------
        # Adam optimisation for each candidate (stop early if time runs out)
        # --------------------------------------------------------------
        for a0 in candidates:
            if time.time() >= deadline:
                break
            a_opt, score_opt = _run_adam(a0, n, slice_deadline, start_time)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve (Adam) n={n}, score={score_opt:.6f}"
                )

        # --------------------------------------------------------------
        # Short annealing sweep to polish the best vector of this scale
        # --------------------------------------------------------------
        if best_seq is not None and time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, 0.04 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-6,
                max_steps=80_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve (anneal) n={n}, score={score_ref:.6f}"
                )

        # Remember the best distribution for the next upscale
        global_a = best_seq.copy() if best_seq is not None else None

    # ------------------------------------------------------------------
    # Final long‑run annealing on the best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            final_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-7,
            max_steps=200_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] final anneal best score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback in the unlikely event we never produced a vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return as a plain Python list of floats
    return [float(x) for x in best_seq]
