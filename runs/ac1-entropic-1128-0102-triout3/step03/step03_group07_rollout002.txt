# -*- coding: utf-8 -*-
"""
Propose a candidate step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

where `conv` is the linear autoconvolution.  The optimizer works on the
probability simplex (∑a_i = 1) and uses an Adam‑style exponentiated‑gradient
update on a smooth soft‑max approximation of the max.  The temperature β of
the soft‑max is increased slowly; when it reaches a large target the weight
vector becomes (practically) a delta at the current peak, i.e. we switch to
a pure sub‑gradient of the max.

Multiple random restarts and occasional log‑space noise are employed to
escape local minima.  The best sequence seen during the run is returned.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^(-1/2)  (a_i = C(2i,i)/4^i),
    normalised to sum to 1.  These coefficients decay like 1/√i and
    provide a good starting point for the problem.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Core optimisation routine

def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises the evaluation function.
    The routine runs for at most ``budget_s`` seconds (a small safety margin
    is kept) and returns the best sequence encountered as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine problem size and initialise a starting sequence.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        # Use the best known sequence from the previous run if it exists.
        base = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        # Clip to the domain allowed by evaluate_sequence.
        base = np.clip(base, 0.0, 1000.0)
        if base.sum() == 0.0:
            base = None
    else:
        base = None

    if base is None or base.ndim != 1 or base.size < 2:
        # fall back to the classic half‑binomial start
        n = 4000                     # length of the sequence (tunable)
        a = half_binomial_series(n)
    else:
        a = base.copy()
        n = a.shape[0]

    # Normalise to the simplex (sum = 1).  The objective is scale‑invariant.
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() == 0.0:
        a = rng.dirichlet(np.full(n, 0.5))
    a = a / a.sum()

    # Keep a copy of the best sequence seen so far.
    best_a = a.copy()
    best_score = np.inf
    best_iter = 0

    # ------------------------------------------------------------------
    # 2) FFT geometry.
    # ------------------------------------------------------------------
    L = 2 * n - 1                # length of linear self‑convolution
    m = n + L - 1                # length of a * rev(w) (3n‑2)
    Nfft = next_pow_two(m)       # power‑of‑two size for speed

    # ------------------------------------------------------------------
    # 3) Adam state (operating in log‑space = exponentiated gradient).
    # ------------------------------------------------------------------
    log_a = np.log(a + 1e-30)            # avoid log(0)
    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05                           # base learning rate for log‑space
    t_iter = 0

    # ------------------------------------------------------------------
    # 4) Soft‑max temperature schedule.
    # ------------------------------------------------------------------
    beta = 1.0
    beta_target = 4000.0
    beta_mul = 1.0012

    # ------------------------------------------------------------------
    # 5) Restart / perturbation control.
    # ------------------------------------------------------------------
    last_improve = 0
    restart_gap = 300_000               # iterations without improvement → restart
    noise_prob = 0.006
    noise_scale = 1e-3

    # ------------------------------------------------------------------
    # 6) Timing.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # safety margin
    it = 0

    # ------------------------------------------------------------------
    # 7) Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- a) current convolution (FFT based) -----
        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]                     # linear part only
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ----- b) best‑so‑far update -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            best_iter = it
            print(f"[iter {it:09d}] new best score = {best_score:.6f}")

        # ----- c) gradient of the smooth objective -----
        if beta < beta_target:
            # soft‑max weights (stable)
            shift = max_conv
            w_exp = np.exp(beta * (conv - shift))
            w = w_exp / w_exp.sum()

            # gradient = 2 * correlation(a, w_rev)
            Wr = np.fft.rfft(w[::-1], n=Nfft)
            corr_full = np.fft.irfft(A * Wr, n=Nfft)
            start = L - 1
            grad = 2.0 * corr_full[start:start + n]

            # increase temperature
            beta = min(beta_target, beta * beta_mul)
        else:
            # pure max sub‑gradient (single active index)
            k = int(conv.argmax())
            grad = np.zeros_like(a)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = k - idx
            grad[idx] = 2.0 * a[jdx]

        # ----- d) Adam update on log‑space -----
        # gradient w.r.t. log‑variables (chain rule)
        g_log = a * grad

        t_iter += 1
        m_adam = beta1 * m_adam + (1.0 - beta1) * g_log
        v_adam = beta2 * v_adam + (1.0 - beta2) * (g_log ** 2)

        m_hat = m_adam / (1.0 - beta1 ** t_iter)
        v_hat = v_adam / (1.0 - beta2 ** t_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        log_a = log_a - step

        # occasional stochastic perturbation in log‑space
        if rng.random() < noise_prob:
            log_a += (rng.random(n) - 0.5) * noise_scale

        # ----- e) map back to the simplex -----
        a = np.exp(log_a)
        # Prevent exact zeros (they would give -inf in log next round)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = rng.dirichlet(np.full(n, 0.5))
            a_sum = a.sum()
        a = a / a_sum
        log_a = np.log(a + 1e-30)   # re‑normalise log_a after projection

        # ----- f) learning‑rate schedule -----
        lr = max(lr * 0.9999, 1e-5)

        # ----- g) restart if stagnated -----
        if it - last_improve > restart_gap:
            a = rng.dirichlet(np.full(n, 0.6))
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
            log_a = np.log(a + 1e-30)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_iter = 0
            beta = 1.0
            last_improve = it
            # quick score report for the fresh start
            A_tmp = np.fft.rfft(a, n=Nfft)
            conv_tmp = np.fft.irfft(A_tmp * A_tmp, n=Nfft)[:L]
            print(f"[iter {it:09d}] restart triggered – score≈{2.0 * n * conv_tmp.max():.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 8) Final evaluation with the provided routine (safety check)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
