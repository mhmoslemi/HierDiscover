Propose a low‑value step‑function sequence.

The algorithm works in three stages:

1. **Diverse analytical seeds** – a large pool of power‑law,
   exponential and truncated‑linear shapes (plus the previous best
   sequence, if available) is sampled and the one with the lowest
   fast_score is kept.

2. **Adam‑based projected gradient descent** on a sub‑gradient of the
   max‑convolution.  The active set of convolution indices is obtained
   by a shrinking relative gap and only the largest few are kept.
   After each gradient step the iterate is projected onto the probability
   simplex (∑a_i=1, a_i≥0).  Adam’s per‑coordinate adaptive step sizes
   together with a lightweight learning‑rate decay give a more stable
   descent than plain SGD.

3. **Escaping local minima** – occasional random “mass‑transfer” moves,
   periodic injection of fresh random shapes and a restart after a
   stagnation interval keep the search from getting stuck.

The routine runs for at most ``budget_s`` seconds (≈1000 s) and returns
the best sequence discovered.  All numbers are non‑negative; the

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        # (a) self‑convolution of the current iterate
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_conv = conv.max()

        # (b) build active set (indices whose convolution value is close to max)
        delta = max(delta_end,
                    delta_start * (1.0 - it / 400000.0))   # linear annealing
        mask = conv >= max_conv * (1.0 - delta)
        active_idxs = np.flatnonzero(mask)
        if active_idxs.size > max_active:
            # keep the `max_active` largest entries
            top = np.argpartition(-conv[active_idxs], max_active)[:max_active]
            active_idxs = active_idxs[top]

        # (c) sub‑gradient for the current active set
        grad = gradient_multi_active(a, active_idxs)

        # (d) Adam update (with a mild decay on the global step size)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step_size = 0.7 / (np.sqrt(it) + 1.0)   # decaying base step
        update = - step_size * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a + update)

        # (e) evaluate the new candidate
        fft_new = np.fft.rfft(a_new, conv_size)
        conv_new = np.fft.irfft(fft_new * fft_new, conv_size)[: 2 * n - 1]
        cand_val = float(2.0 * n * conv_new.max())

        # (f) accept if improvement; otherwise occasional uphill acceptance
        if cand_val < best_val:
            best_val = cand_val
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
        else:
            if rng.random() < uphill_accept:
                a = a_new

        # ----------------------------------------------------------
        # (g) random “mass‑transfer” move to help escape plateaus
        if rng.random() < random_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta_transfer = random_move_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta_transfer
                cand[j] += delta_transfer
                cand = proj_simplex(cand)

                fft_c = np.fft.rfft(cand, conv_size)
                conv_c = np.fft.irfft(fft_c * fft_c, conv_size)[: 2 * n - 1]
                cand_val2 = float(2.0 * n * conv_c.max())

                if cand_val2 < best_val:
                    best_val = cand_val2
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif rng.random() < uphill_accept:
                    a = cand

        # ----------------------------------------------------------
        # (h) stagnation handling – re‑initialise from the best with a small noise
        if time.time() - last_improve > 30.0:
            noise = rng.random(n)
            noise /= noise.sum()
            a = proj_simplex(0.85 * best_seq + 0.15 * noise)
            last_improve = time.time()

        # ----------------------------------------------------------
        # (i) occasional injection of a fresh random shape
        if it % 20000 == 0:
            fresh = []
            for _ in range(5):
                fresh.append(build_power_law(n, rng.uniform(0.15, 0.85),
                                            rng.uniform(0.0, 2.0)))
                fresh.append(build_exponential(n, rng.uniform(0.80, 0.99)))
                fresh.append(build_truncated_linear(n,
                                                    rng.integers(int(0.25 * n), n + 1)))
            for seq in fresh:
                val = fast_score(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    a = seq.copy()
                    m.fill(0.0)
                    v.fill(0.0)
                    last_improve = time.time()
                    break

        # ----------------------------------------------------------
        # (j) reporting
        if it % 5000 == 0:
            print(f"[iter {it:7d}] current {cand_val:.6f}  best {best_val:.6f}")

    # --------------------------------------------------------------
    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # The external checker will re‑normalise, but we keep the sum = 1.
    return best_seq.tolist()
