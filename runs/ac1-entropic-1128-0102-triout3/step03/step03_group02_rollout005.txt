"""
Propose a low‑score step‑function using a fast Adam‑type projected gradient
descent on the exact sub‑gradient of  max(conv(a,a)).  The method works in three
phases:

1.  Initialise a pool of diverse shapes (uniform, triangular, Hann window,
    random Dirichlet and, if available, the previously found `height_sequence_1`).
    Each candidate is refined for a modest number of iterations with a
    projected Adam optimiser.

2.  Iteratively up‑sample the current best shape (≈1.5× larger each time) and
    repeat the refinement.  Larger lengths are allowed to run fewer iterations
    so that the overall wall‑clock budget (≈1000 s) is respected.

3.  A final long refinement on the best length found.

All updates stay on the probability simplex (∑a_i = 1, a_i ≥ 0) which makes the
objective  `evaluate_sequence` scale‑invariant.  The gradient of the
max‑convolution is obtained analytically via a second convolution with an
indicator vector of the active (max‑imposing) indices, which is cheap using FFT.
The optimiser records the best sequence seen and returns it as a list of Python
floats.
"""

import time
import math

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Score for a probability vector a (∑a=1).  Returns 2·n·max(conv(a,a))."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _refine_sequence_adam(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    base_lr: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam‑style projected descent for a fixed length.

    Returns (final_a, best_seen_a, best_seen_score) where the score is
    2·n·max(conv(a,a)) (the same value `evaluate_sequence` would return after
    normalisation).
    """
    n = a.shape[0]

    # FFT lengths (next power‑of‑two for speed)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- auto‑convolution ------------------------------------------------
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()

        # ---- sub‑gradient of max(conv) ---------------------------------------
        # active set (indices where the max is attained)
        # a tiny relative tolerance avoids missing the exact maximum due to fp noise
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros(2 * n - 1, dtype=float)
        w[active] = 1.0          # each active index contributes equally

        # gradient = 2 * (w * a_rev)_{slice}
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
        g = 2.0 * conv2[n - 1 : n - 1 + n]

        # ---- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr_t = base_lr / math.sqrt(it)   # modest decay
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # ---- keep feasible -----------------------------------------------------
        a = _project_simplex(a)

        # ---- record best --------------------------------------------------------
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return a, best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low `evaluate_sequence` score.

    The algorithm follows a three‑stage scheme (diverse starts → progressive
    up‑sampling → final long refinement) using the Adam optimiser defined above.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1.  Small‑scale search (n ≈ 1200) with several diverse seeds
    # ------------------------------------------------------------------
    n0 = 1200
    init_pool = []

    # uniform
    init_pool.append(np.full(n0, 1.0 / n0))

    # triangular (tent) shape
    idx = np.arange(n0)
    tri = np.minimum(idx + 1, n0 - idx)
    init_pool.append(tri / tri.sum())

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n0 - 1)))
    init_pool.append(hann / hann.sum())

    # one random Dirichlet draw
    init_pool.append(np.random.dirichlet(np.ones(n0)))

    # previous construction, if the global variable exists
    try:
        prev = np.array(height_sequence_1, dtype=float)
        if prev.shape[0] != n0:
            # linear interpolation to the required size
            x_old = np.linspace(0.0, 1.0, prev.shape[0])
            x_new = np.linspace(0.0, 1.0, n0)
            prev = np.interp(x_new, x_old, prev)
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0:
            init_pool.append(prev / s)
    except Exception:
        pass

    # refine each candidate
    for i, cand in enumerate(init_pool, 1):
        if time.time() >= deadline:
            break
        iters = 25000
        print(
            f"[{time.time() - start_time:6.2f}s] Refining n={n0}, candidate {i}/{len(init_pool)} ({iters} iters)"
        )
        _, best_local_a, best_local_score = _refine_sequence_adam(
            cand,
            max_iters=iters,
            deadline=deadline,
            base_lr=0.6,
            beta1=0.9,
            beta2=0.999,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"   New global best: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Progressive up‑sampling
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)

    cur_seq = best_seq.copy()
    cur_n = cur_seq.shape[0]
    MAX_LEN = 8000      # we keep within the budget used by the reference solution

    while True:
        # decide next size
        next_n = int(cur_n * 1.5)
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n <= cur_n:
            break

        # linear interpolation to increase resolution
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a_up = np.interp(x_new, x_old, cur_seq)
        a_up = np.maximum(a_up, 0.0)
        a_up /= a_up.sum()

        # iteration budget – larger n receives fewer iterations
        base_iters = 40000
        max_iter = int(base_iters * n0 / next_n) + 15000
        max_iter = max(max_iter, 15000)

        print(
            f"[{time.time() - start_time:6.2f}s] Upsampling to n={next_n}, {max_iter} iters"
        )
        _, best_local_a, best_local_score = _refine_sequence_adam(
            a_up,
            max_iters=max_iter,
            deadline=deadline,
            base_lr=0.55,
            beta1=0.9,
            beta2=0.999,
        )

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"   Updated global best after up‑sampling: n={next_n}, score={best_score:.6f}")

        cur_seq = a_up            # continue up‑sampling from the interpolated version
        cur_n = next_n

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3.  Final long refinement on the best length discovered
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        final_iters = 50000
        print(
            f"[{time.time() - start_time:6.2f}s] Final refinement on n={n}, {final_iters} iters"
        )
        _, best_local_a, best_local_score = _refine_sequence_adam(
            best_seq,
            max_iters=final_iters,
            deadline=deadline,
            base_lr=0.45,
            beta1=0.9,
            beta2=0.999,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"   Final improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Normalise and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total

    print(f"[{time.time() - start_time:6.2f}s] Finished. Best score = {best_score:.6f}")
    return [float(x) for x in best_seq]
