def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2
    target = 1.503
    # We'll try different n values:
    n_values = [200,250,300,350,400,500,600,800,1000] # but we might skip large ones if time low.
    best_seq = None
    best_val = float('inf')
    # iterate over n
    for n in n_values:
        # choose list of k fractions
        for frac in [0.5,0.6,0.7,0.8,0.9]:
            k = int(frac*n)
            if k < 2: continue
            allowed = int(math.floor(target * k*k / (2.0 * n)))
            if allowed < 1: allowed = 1
            # if allowed is >= k^2 maybe trivial - but not.
            # Greedy attempts
            attempts = 0
            max_attempts = 5000
            # early break if total time near deadline:
            while time.time() < deadline and attempts < max_attempts:
                attempts+=1
                # produce S via greedy_build_set, may fail -> None
                S = greedy_build_set(n,k,allowed)
                if S is None:
                    continue
                cnt = compute_counts(S,n)
                max_c = int(cnt.max())
                if max_c > allowed:
                    # not successful constraint (should not happen)
                    continue
                a = np.zeros(n, dtype=np.float64)
                a[S] = 1.0/k
                val = fast_score(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[found] n={n},k={k},allowed={allowed},attempts={attempts},score={val:.6f}")
                    # if below target, can return early (optional)
                    if val <= target:
                        print("[success] Target met")
                        return best_seq.tolist()
            # after attempts if best found, maybe continue using subgradient refinement
            if best_seq is not None:
                # attempt to refine with subgradient for a while
                # set time left for refinement
                time_left = deadline - time.time()
                if time_left > 10.0:   # ensure enough time
                    print(f"[refine] refining best solution, time left {time_left:.1f}s")
                    refined = subgradient_optimize(best_seq, time_left)
                    refined_val = fast_score(refined)
                    if refined_val < best_val:
                        best_val = refined_val
                        best_seq = refined.copy()
                        print(f"[refine] new best {best_val:.6f}")
                    # after refine just break maybe
                # continue searching maybe other n?
    # fallback: run subgradient on uniform distribution of size maybe 1500
    if best_seq is None:
        print("[fallback] no good support found, using subgradient on uniform init.")
        init = np.full(200, 1/200, dtype=np.float64)  # maybe n=200 uniform
        best_seq = subgradient_optimize(init, deadline - time.time())
        best_val = fast_score(best_seq)

    print(f"[final] best score {best_val:.6f}, n={len(best_seq)}")
    return best_seq.tolist()

"""
Propose a low‑score step function.

The search works in two phases:

1. **Sparse support construction (small n).**
   For several modest lengths `n` (200‑1000) we pick a support size
   `k = ⌈fraction·n⌉` and a maximum allowed ordered‑pair count
   derived from the target score (1.503).  Random subsets of the given
   size are generated until one satisfies the count bound.
   The corresponding uniform vector (mass 1/k on the support) gives a
   candidate sequence; its fast evaluation is kept if it improves the
   best value seen.

2. **Weight refinement.**
   Whenever a candidate is found we run a short projected‑sub‑gradient
   descent (the same routine used in the reference solution) to adjust
   the non‑zero weights and possibly lower the maximal self‑convolution.
   The refinement is limited by the remaining time budget.

If nothing is found we fall back to a generic sub‑gradient optimisation
starting from a uniform vector.  The whole routine respects the 1000 s
budget and returns a list of non‑negative heights suitable for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the score for a non‑negative vector `a`
    (the sum of `a` is assumed to be non‑zero).  This is equivalent to
    `evaluate_sequence` but skips the defensive checks.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # next power of two for FFT (zero‑padding)
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """
    Given a support set `S` (array of indices) and the length `n`,
    return the ordered‑pair count for each possible convolution index
    0 … 2·n‑2.
    """
    sums = np.add.outer(S, S)                # shape (|S|, |S|)
    return np.bincount(sums.ravel(), minlength=2 * n - 1)


# ----------------------------------------------------------------------
def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the original objective.
    The vector is kept on the probability simplex (sum = 1, a_i ≥ 0).
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    # step‑size parameters (tuned for small to medium n)
    eta0 = 0.40
    eta_decay = 5e-6
    random_move_prob = 0.03
    random_move_step = 0.12

    it = 0
    deadline = time.time() + time_limit - 0.1
    ind = np.arange(1, n + 1)  # used in simplex projection

    while time.time() < deadline:
        # current convolution and the most violating index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active term (ordered pairs)
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size decreasing with iteration count
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer move
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_step * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta

                # project back onto simplex
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)

                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # periodic bookkeeping
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        it += 1

    return best_a

                if time_left > 5.0:
                    refine_time = min(20.0, time_left / 2)
                    print(f"[refine] polishing best vector (n={len(best_seq)}) "
                          f"for {refine_time:.1f}s")
                    refined = subgradient_optimize(best_seq, refine_time)
                    refined_val = fast_score(refined)
                    if refined_val < best_val:
                        best_val, best_seq = refined_val, refined.copy()
                        print(f"[refine] improved score to {best_val:.6f}")

    # 2) fallback – generic sub‑gradient optimisation from a uniform start.
    if best_seq is None:
        print("[fallback] no feasible support found, running generic optimisation.")
        n0 = 500
        init = np.full(n0, 1.0 / n0, dtype=np.float64)
        remaining = deadline - time.time()
        if remaining > 0:
            best_seq = subgradient_optimize(init, remaining)
        else:
            best_seq = init
        best_val = fast_score(best_seq)

    print(f"[final] best score = {best_val:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
