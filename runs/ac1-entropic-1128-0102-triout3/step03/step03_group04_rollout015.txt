"""
propose_candidate
=================

A three‑phase optimizer that searches for a non‑negative step‑function sequence
with a low value of

    score = 2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm differs from the previous implementation in several ways:

1. **Exponentiated‑gradient descent on the probability simplex** – the
   parameters are updated by a multiplicative rule `a_i ← a_i·exp(‑η·g_i)`,
   which keeps all entries positive and automatically respects the unit‑sum
   constraint (no explicit projection is required).

2. **Soft‑max smoothing of the max‑convolution** – instead of a hard “max”
   sub‑gradient we minimise a smooth approximation
   `log‑sum‑exp(β·conv) / β`.  The gradient of this proxy is a weighted sum of
   the per‑index gradients, where the weights are a soft‑max over the
   convolution entries.  The temperature `β` is increased slowly, so the
   optimisation first reduces the overall shape of the convolution and later
   concentrates on the true peak.

3. **FFT‑based convolution & gradient** – both the convolution and the
   weighted gradient are obtained with a single pair of forward FFTs and two
   inverse FFTs per iteration, giving O(n log n) complexity.

4. **Multiple restarts + diverse seeds** – the best candidate from a fast
   power‑law/ exponential sweep is refined, and several random Dirichlet
   starts are explored.  An occasional jitter avoids stagnation.

The routine respects a user‑supplied time budget (default ≈ 950 s, leaving a
5 s safety margin) and returns the best sequence found as a plain list of
floats.  The final sequence is validated with the provided `evaluate_sequence`
function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities -----------------------------------------------------

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to unit L1‑norm."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def compute_score(a: np.ndarray, n: int, L: int) -> float:
    """
    Return the evaluation score 2·n·max(conv(a, a)).
    The vector `a` is assumed to sum to 1, so the denominator is 1.
    """
    A = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(A * A, n=L)
    m = 2 * n - 1
    max_conv = conv_full[:m].max()
    return 2.0 * n * max_conv


def weighted_gradient(a: np.ndarray, conv: np.ndarray, beta: float,
                      n: int, L: int) -> np.ndarray:
    """
    Compute ∇_a  (soft‑max_β(conv))  using FFTs.
    The weights are w_i ∝ exp(β·conv_i).
    The returned gradient is the exact gradient of Σ_i w_i·conv_i.
    """
    # soft‑max weights (shifted for numerical stability)
    shift = conv.max()
    w = np.exp(beta * (conv - shift))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # weighted gradient = 2 * (w_rev ⊛ a) evaluated at positions m‑1‑i
    w_rev = w[::-1]                     # reverse the weight vector
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(A * W, n=L)   # linear convolution of w_rev and a
    m = 2 * n - 1
    idx = m - 1 - np.arange(n)            # indices m‑1‑i
    grad = 2.0 * grad_full[idx]
    return grad


def exponentiated_update(a: np.ndarray, grad: np.ndarray,
                         eta: float) -> np.ndarray:
    """
    Perform a single multiplicative (exponentiated) gradient step and
    renormalise to the simplex.
    """
    # Clip the raw gradient to avoid overflow in the exponential
    max_clip = 20.0 / max(eta, 1e-12)   # heuristic bound
    grad_clipped = np.clip(grad, -max_clip, max_clip)

    a_new = a * np.exp(-eta * grad_clipped)
    # Guard against all zeroes after the multiplicative step
    if not np.isfinite(a_new).all() or a_new.sum() == 0.0:
        a_new = np.full_like(a, 1.0 / a.size)
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main search routine --------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative sequence that minimises the provided evaluation
    function.  The routine respects the time budget and returns the best list
    found (as plain Python floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------ configuration
    n = 2500                                          # length of the candidate
    m = 2 * n - 1
    L = next_pow2(3 * n - 2)                         # FFT length for all ops

    # --------------------------------------------------------------- phase 0
    # Use the global sequence if it exists (might already be good)
    best_a = None
    best_score = np.inf
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            cand = np.maximum(cand, 0.0)
            if cand.size > 0 and np.isfinite(cand).all():
                cand /= cand.sum()
                sc = compute_score(cand, n, L)
                best_a = cand
                best_score = sc
                print(f"[init] height_sequence_1 score = {sc:.6f}")
        except Exception as exc:                     # pragma: no cover
            print(f"[init] could not use height_sequence_1: {exc}")

    # ----------------------------------------------------------------- phase 1
    # Coarse sweep of power‑law and exponential families
    print("[search] phase 1 – coarse parametric sweep")
    p_grid = np.linspace(0.45, 0.65, 15)             # power‑law exponent
    offsets = np.linspace(0.3, 2.0, 9)               # offset for power‑law
    lambdas = np.linspace(0.0005, 0.0045, 18)       # pure exponential rates

    for offset in offsets:
        for p in p_grid:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset)
            sc = compute_score(a, n, L)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[grid] new best (p={p:.4f},off={offset:.2f}) → {sc:.6f}")

    for lam in lambdas:
        if time.time() > deadline:
            break
        a = exponential_sequence(n, lam)
        sc = compute_score(a, n, L)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[exp]  new best (λ={lam:.5f}) → {sc:.6f}")

    # If nothing was found (should never happen) fall back to uniform
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_score(best_a, n, L)
        print("[fallback] uniform start used")

    # -------------------------------------------------------------- phase 2
    # Exponentiated‑gradient descent with multiple restarts
    print("[search] phase 2 – exponentiated‑gradient descent")
    max_restarts = 5
    global_best_a = best_a.copy()
    global_best_score = best_score

    for restart in range(max_restarts):
        if time.time() > deadline:
            break

        # Initialise this restart
        if restart == 0:
            a = best_a.copy()
        else:
            # Sparse Dirichlet start (α<1 yields spikier vectors)
            a = rng.gamma(0.7, 1.0, size=n)
            a /= a.sum()

        cur_score = compute_score(a, n, L)
        if cur_score < global_best_score:
            global_best_score, global_best_a = cur_score, a.copy()
            print(f"[restart {restart}] initial improvement → {cur_score:.6f}")

        # Hyper‑parameters (will be annealed)
        eta = 0.08            # base step size for exponentiated update
        beta = 0.8            # start temperature for soft‑max
        eta_decay = 0.9997    # multiplicative decay per iteration
        beta_inc = 0.0004     # additive increase per iteration

        it = 0
        while time.time() < deadline:
            # ----- convolution & soft‑max weights
            A = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(A * A, n=L)
            conv = conv_full[:m]                     # linear self‑convolution

            # ----- gradient of the smooth approximation of the max
            grad = weighted_gradient(a, conv, beta, n, L)

            # ----- multiplicative (exponentiated) update
            a = exponentiated_update(a, grad, eta)

            # occasional jitter to escape plateaus
            if rng.random() < 0.004:
                jitter = (rng.random(n) - 0.5) * 1e-7
                a += jitter
                a = np.maximum(a, 0.0)
                a /= a.sum()

            # ----- evaluation of the true (non‑smooth) objective
            score = compute_score(a, n, L)
            if score < global_best_score:
                global_best_score, global_best_a = score, a.copy()
                print(f"[restart {restart} iter {it:07d}] new global best = {score:.6f}")

            # ----- schedule updates
            eta = max(eta * eta_decay, 1e-6)
            beta += beta_inc
            it += 1

            # safety break – stop if we have not improved for a while
            if it % 20000 == 0:
                pass  # could insert a stagnation check here

        print(f"[restart {restart}] finished after {it:,} iters, local best = {cur_score:.6f}")

    # --------------------------------------------------------------- phase 3
    # Final local jitter‑only refinement (tiny perturbations)
    print("[search] phase 3 – final jitter refinement")
    a = global_best_a.copy()
    for _ in range(3000):
        if time.time() > deadline:
            break
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        epsilon = 1e-7 * (rng.random() - 0.5) * a[i]
        a_candidate = a.copy()
        a_candidate[i] = max(a_candidate[i] - epsilon, 0.0)
        a_candidate[j] = a_candidate[j] + epsilon
        a_candidate /= a_candidate.sum()
        score = compute_score(a_candidate, n, L)
        if score < global_best_score:
            global_best_score, global_best_a = score, a_candidate.copy()
            a = a_candidate
            print(f"[jitter] improved to {score:.6f}")

    # ----------------------------------------------------------- final check
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[done] final evaluate_sequence score = {final_score:.6f}")
    print(f"[info] best raw score (2·n·max) = {global_best_score:.6f}")

    return global_best_a.tolist()
