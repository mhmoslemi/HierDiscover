"""
Propose a step‑function with a low evaluate_sequence score.

The core is an Adam‑style projected optimizer that minimises a smooth
approximation of  max(conv(a,a))  on the probability simplex.
A variety of seeds (uniform, triangular, Hann, arcsine‑U‑shaped and several
Dirichlet draws) are refined for several different lengths.  The best
candidate is up‑sampled progressively (linear interpolation) and refined
again, ending with a long final Adam run.

Key ideas beyond the previous baseline:
* Adam instead of simple momentum – adaptive learning rates help escape
  plateaus.
* A U‑shaped “arcsine” start that is known to flatten the sum‑distribution
  of two independent draws (its convolution is close to uniform).
* Several random Dirichlet starts at each length to increase diversity.
* Faster temperature schedule for the soft‑max that approximates the max;
  the same schedule is used for all runs.
* A final simulated‑annealing style perturb‑and‑refine phase for a last
  push.

The routine respects the 1000 s budget, prints intermediate improvements,
and finally returns a list of plain Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Compute a * a (linear convolution) using FFT of given length."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_of_seq(a: np.ndarray) -> float:
    """Score = 2·n·max(conv) for a vector already summing to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    conv = _conv_fft(a, fft_len)
    return 2.0 * n * conv.max()


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max over convolution entries with temperature tau."""
    max_val = conv.max()
    # avoid overflow / underflow
    shifted = (conv - max_val) / max(tau, 1e-14)
    w = np.exp(shifted)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w.fill(1.0 / w.size)
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max (soft‑max) w.r.t. a.
    g_i = 2 * Σ_k w_k * a_{k-i}.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len)[:conv_len]
    # slice the part that corresponds to the original length
    g = 2.0 * conv2[n - 1 : n - 1 + n]
    return g


def _refine_adam(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.12,
    tau_start: float = 0.06,
    tau_end: float = 0.001,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam‑style projected optimisation on the simplex.

    Returns (final_a, best_seen_a, best_seen_score).
    """
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- current score ----------
        fft_len = 1 << ((2 * n - 1).bit_length())
        conv = _conv_fft(a, fft_len)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---------- soft‑max ----------
        tau = tau_start + (tau_end - tau_start) * (it / max_iters)
        w = _softmax_weights(conv, tau)

        # ---------- gradient ----------
        g = _gradient_from_weights(a, w)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)   # modest decay
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- projection ----------
        a = _project_simplex(a_new)

    return a, best_a, best_score

    is performed at the very end.

    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # a small safety margin

    best_seq = None
    best_score = float("inf")

    # 1.  Work on a pool of lengths – start modest, finish with a larger one
    # ------------------------------------------------------------------

        a_up = np.maximum(a_up, 0.0)
        a_up /= a_up.sum() if a_up.sum() > 0 else np.full_like(a_up, 1.0 / next_n)

        # ----- allocate iteration budget (larger sizes get fewer iters) -----
        base_iters = 25000
        iters = max(12000, int(base_iters * 1800 / next_n))

        print(
            f"[{time.time() - start_time:6.2f}s] Upsampling {cur_n} → {next_n}, "
            f"{iters} iters"
        )

        _, best_local_a, best_local_score = _refine_adam(
            a_up,
            max_iters=iters,
            deadline=deadline,
            step0=0.1,
            tau_start=0.05,
            tau_end=0.001,
            beta1=0.9,
            beta2=0.999,
        )

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(
                f"   Updated global best after up‑sampling: n={next_n}, "
                f"score={best_local_score:.6f}"
            )

        current = best_local_a
        cur_n = next_n

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3.  Final long refinement on the best length found
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_final = best_seq.shape[0]
        final_iters = 60000
        print(
            f"[{time.time() - start_time:6.2f}s] Final refinement n={n_final}, "
            f"{final_iters} iters"
        )
        _, best_local_a, best_local_score = _refine_adam(
            best_seq,
            max_iters=final_iters,
            deadline=deadline,
            step0=0.08,
            tau_start=0.04,
            tau_end=0.0005,
            beta1=0.9,
            beta2=0.999,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"   Final improvement: score={best_local_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Light perturb‑and‑refine (simulated‑annealing flavour)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        perturb_iters = 12000
        print(
            f"[{time.time() - start_time:6.2f}s] Perturb‑refine phase, {perturb_iters} iters"
        )
        a = best_seq.copy()
        for it in range(perturb_iters):
            if time.time() >= deadline:
                break
            # small random transfer between two positions
            i, j = np.random.randint(0, a.size, size=2)
            delta = (np.random.rand() - 0.5) * 0.004   # ±0.2 % of total mass
            if a[i] + delta < 0.0:
                delta = -a[i]
            if a[j] - delta < 0.0:
                delta = a[j]
            a[i] += delta
            a[j] -= delta
            # keep on simplex (projection fixes tiny drift)
            a = _project_simplex(a)

            # occasional short Adam refinement (every 1500 steps)
            if (it + 1) % 1500 == 0:
                _, a, _ = _refine_adam(
                    a,
                    max_iters=3000,
                    deadline=deadline,
                    step0=0.06,
                    tau_start=0.03,
                    tau_end=0.0008,
                    beta1=0.9,
                    beta2=0.999,
                )
        # final evaluation
        final_score = _score_of_seq(a)
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(f"   Perturb phase improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(
        f"[{time.time() - start_time:6.2f}s] Finished. Best score "
        f"(2·n·max(conv)) = {best_score:.6f}"
    )
    return [float(x) for x in best_seq]
