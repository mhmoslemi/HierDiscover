def build_conv_matrix(f):
    n = len(f)
    row = np.empty(n*n, dtype=np.int32)
    col = np.empty(n*n, dtype=np.int32)
    data = np.empty(n*n, dtype=np.float64)
    idx = 0
    for i in range(n):
        base_row = i
        f_i = f[i]
        for j in range(n):
            row[idx] = base_row + j   # i+j
            col[idx] = j
            data[idx] = f_i
            idx += 1
    return csr_matrix((data, (row, col)), shape=(2*n-1, n))

def _run_adam_descent(a0, n, deadline, start_time, max_iter=200_000):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    lam_factor = 0.25
    lam_decay = 0.5
    lam_decay_interval = 2000

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n}, it={it}, score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = 0.5 / (np.sqrt(v_hat) + eps)   # step scaled by sqrt(variance)
        a = a - step * m_hat
        # ensure positivity and simplex
        a = _project_simplex(a)

    return best_a, best_score

def _run_lp_refine(a, n, deadline, start_time, max_iter=10):
    # iterative LP refinement: iterate up to max_iter, each time solve LP and combine.
    f = a.copy()
    best_f = f.copy()
    best_score = 2.0 * n * _conv_self(f).max()
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # Build constraints matrix for f
        M = _conv_self(f).max()
        A = _build_conv_matrix(f)   # shape (2n-1, n)
        b = np.full(2*n-1, M)
        # Solve LP: maximize sum(g)
        c = -np.ones(n)   # linprog minimizes, so minimize -sum(g)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            # no improvement possible
            break
        # scale to sum=1
        g_scaled = g / sum_g
        # mix
        t = 0.5  # mixing factor
        f = (1 - t) * f + t * g_scaled
        f = _project_simplex(f)  # maintain sum=1, nonnegative
        cur_score = 2.0 * n * _conv_self(f).max()
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve iter={it+1}, n={n}, score={cur_score:.6f}")
            # optionally decrease mixing factor as progress
        else:
            # if no improvement, break

"""
Propose a low‑value step‑function.

The algorithm mixes three ideas:

1. **Adam‑based smoothed‑gradient descent** on the log‑sum‑exp surrogate for
   the ℓ∞–norm of the self‑convolution.  The iterates are projected onto the
   probability simplex (∑aᵢ = 1, aᵢ ≥ 0).

2. **Linear‑programming polishing** (only on moderate sizes, n ≤ 2000):
   for the current vector f we solve
       maximise   ∑ gⱼ
       subject to f * g ≤ M   (pointwise) ,   g ≥ 0,
   where M = max(f*f).  The optimiser pushes mass into “free’’ slots while
   keeping the convolution bound; the result is mixed back into f, which
   provably reduces the maximal convolution value.

3. **Final cheap simulated‑annealing sweep** that transfers a small ε
   between two coordinates; this further reduces the peak.

The whole search is performed at several length scales
(· up‑sampled between stages) and stops after the time budget
(~1000 s).  The best sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a via real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed‑gradient descent (Adam)
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate.
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights (stable)
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)                # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam‑style smoothed‑gradient descent on the simplex.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.03          # base learning rate

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    # λ‑schedule
    lam_factor = 0.25
    lam_decay = 0.5
    lam_decay_interval = 2000

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, it={it}, score={cur_score:.6f}")

        # λ update (gradual annealing)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# LP polishing (only for modest sizes)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A of shape (2n‑1, n) such that (A @ g)_k = (f*g)_k.
    Each entry A[k, j] = f[k‑j] whenever 0 ≤ k‑j < n, else 0.
    """
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    idx = 0
    for i in range(n):          # i is the index from f
        f_i = f[i]
        base = i                # k = i + j
        for j in range(n):
            rows[idx] = base + j
            cols[idx] = j
            data[idx] = f_i
            idx += 1
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _run_lp_refine(a: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iter: int = 6) -> tuple[np.ndarray, float]:
    """
    Iteratively solve the LP
        maximise   1ᵀ g
        subject to f*g ≤ M , g ≥ 0
    and mix the result back into the current vector.
    The loop stops when the bound cannot be improved or the time budget expires.
    """
    f = a.copy()
    best_f = f.copy()
    best_score = 2.0 * n * _conv_self(f).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        M = _conv_self(f).max()
        A = _build_conv_matrix(f)                     # (2n‑1) × n
        b = np.full(2 * n - 1, M)

        # linprog minimizes; maximise Σg → minimise −Σg
        c = -np.ones(n, dtype=np.float64)

        # Use the high‑performance HiGHS interior‑point solver
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # fallback – stop polishing
            break

        g = np.maximum(res.x, 0.0)          # numerical safety
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-9:
            # no extra mass can be added without violating the bound
            break

        # Scale to the simplex (the bound then becomes stricter)
        g = g / sum_g

        # Convex combination (t = 0.5 works well in practice)
        t = 0.5
        f = (1.0 - t) * f + t * g
        f = _project_simplex(f)

        cur_score = 2.0 * n * _conv_self(f).max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve: n={n}, it={it}, score={cur_score:.6f}")
        else:
            # no improvement – stop early
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (same as in the reference code)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """Simple mass‑transfer annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Finds a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm combines Adam‑based smoothed gradient
    descent, an LP polishing phase (for n ≤ 2000) and a final simulated‑annealing
    sweep.  It works on several length scales, up‑sampling the best distribution
    found so far.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    best_score = float("inf")
    best_seq = None
    global_a = None

    # Length schedule – grow geometrically, shuffled to avoid bias if we stop early
    lengths = [512, 1024, 2048, 4096, 8192]
    np.random.shuffle(lengths)

    # optional starter from a previous run
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- initialise a0 -------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ----- allocate a time slice for this length ------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # devote roughly 60‑% to gradient descent, 25‑% to LP (if allowed), rest to annealing
        grad_deadline = time.time() + max(4.0, 0.55 * remaining)

        # ----- Adam‑based smoothed gradient descent -------------------------
        a_opt, score_opt = _run_adam_descent(a0, n, grad_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after Adam: n={n}, score={score_opt:.6f}")

        # ----- LP polishing (only for modest n) -----------------------------
        if n <= 2000 and time.time() < deadline:
            lp_deadline = time.time() + max(2.0, 0.2 * (deadline - time.time()))
            a_lp, score_lp = _run_lp_refine(a_opt, n, lp_deadline, start_time)
            if score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after LP: n={n}, score={score_lp:.6f}")

        # ----- final annealing refinement ------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(2.0, 0.15 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=50_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal: n={n}, score={score_ref:.6f}")

        # keep the best for the next upscale
        global_a = best_seq.copy()

    # safety fallback
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # return plain Python list (evaluate_sequence will clamp/normalise anyway)
    return [float(x) for x in best_seq]
