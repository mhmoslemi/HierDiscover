"""
Optimiser for the “max‑convolution” score.

The function ``propose_candidate`` builds a non‑negative step‑function (a list of
floats) of length *n*≈2500 that attempts to minimise

    score = 2·n·max(conv(a,a)) / (sum(a)²)

(the denominator is scale‑invariant, so we keep the sequence normalised to
unit L¹‑norm).  The optimiser works in three stages:

1.  A quick grid‑search over power‑law families a_i ∝ (i+offset)^(-p) to obtain
    a decent seed (also falls back to the supplied ``height_sequence_1`` if
    available).

2.  Several multi‑restart Adam‑style projected gradient descents.  The gradient
    is a weighted sum of the gradients of the *k* largest convolution entries;
    a soft‑max weighting (temperature β) focuses the descent on the dominant
    peaks while still providing a smooth direction.  After each step the vector
    is projected onto the probability simplex (non‑negative, sum = 1).

3.  A handful of linear‑programming refinements.  For a fixed vector **f** we
    solve

        maximise   Σ b_i
        subject to  (f * b) ≤ M   (pointwise) ,   b ≥ 0

    where M = max(f*f).  The solution **b** places additional mass where the
    convolution bound is not tight.  A convex combination ``f←(1‑t)f + t·b̂``
    (with b̂ normalised) is performed and projected back onto the simplex.
    The LP is solved with ``scipy.optimize.linprog`` using the HIGH‑S solver.

The whole routine respects a user‑supplied time budget (default 950 s, leaving a
few seconds safety margin) and prints progress.  The returned list is the best
candidate found; the official ``evaluate_sequence`` is called once at the end
to obtain the final reported score.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# Helper utilities (no closures, no lambdas)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Return a non‑negative vector a_i = (i+offset)^(-p) (i=0…n‑1) normalised
    to unit L¹‑norm."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:                     # defensive fallback
        a.fill(1.0 / n)
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full discrete self‑convolution x*x using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w ≥ 0, Σw = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Exact sub‑gradient of the convolution entry conv[peak_idx] w.r.t. a.
    The gradient is g_i = 2·a_{peak_idx-i} for all valid i, else 0."""
    n = a.shape[0]
    grad = np.zeros_like(a)

    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


def gradient_of_top_k(a: np.ndarray, conv: np.ndarray, k: int,
                      beta: float = 5.0) -> np.ndarray:
    """Weighted blend of gradients of the top‑k convolution entries.
    Weights are given by a soft‑max with temperature 1/β."""
    m = conv.shape[0]
    if k >= m:
        idxs = np.arange(m)
    else:
        # argpartition is O(m) – fast enough for the typical sizes
        idxs = np.argpartition(conv, -k)[-k:]
    vals = conv[idxs]
    maxv = conv.max()
    # soft‑max (shifted by max to avoid overflow)
    w = np.exp(beta * (vals - maxv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w_sum

    grad = np.zeros_like(a)
    for idx, weight in zip(idxs, w):
        grad += weight * gradient_of_peak(a, int(idx))
    return grad

        # Adam moments
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-12

        cur_score = 2.0 * n * conv_fft(a).max()
        beta = beta_start
        eta = eta_start

        it = 0
        while time.time() < deadline and it < max_iter_per_restart:
            conv = conv_fft(a)

            # weighted gradient of the top‑k peaks
            grad = gradient_of_top_k(a, conv, top_k, beta=beta)

            # Adam update (move opposite to gradient)
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = eta * m_hat / (np.sqrt(v_hat) + eps)
            a_trial = a - step
            a_trial = np.maximum(a_trial, 1e-12)             # avoid exact zeros
            a_trial = project_onto_simplex(a_trial)

            # occasional tiny jitter to shake out plateaus
            if rng.random() < 0.003:
                jitter = (rng.random(n) - 0.5) * 1e-6
                a_trial += jitter
                a_trial = project_onto_simplex(a_trial)

            trial_score = 2.0 * n * conv_fft(a_trial).max()

            # accept if better, otherwise Metropolis‑type acceptance
            if trial_score <= cur_score:
                a = a_trial
                cur_score = trial_score
            else:
                temperature = 0.015 * np.exp(-it / 120_000.0)
                if rng.random() < np.exp(-(trial_score - cur_score) / max(temperature, 1e-12)):
                    a = a_trial
                    cur_score = trial_score

            # keep global best
            if cur_score < global_best_score:
                global_best_score = cur_score
                global_best_a = a.copy()
                print(f"[restart {restart} it {it:07d}] global best = {global_best_score:.6f}")

            # periodic decay of step‑size and increase of β
            if (it % decay_every) == 0:
                eta = max(eta * decay_factor, 5e-7)
                beta += beta_inc

        print(f"[restart {restart}] finished – local best score = {cur_score:.6f}")

    print("[grad] finished – best score after gradient phase =", global_best_score)

    # ------- 4. linear‑programming refinements ---------------------------
    print("[lp] phase 4 – LP refinements")
    a = global_best_a.copy()
    cur_score = global_best_score
    lp_iter = 0
    max_lp_iters = 8

    while time.time() < deadline and lp_iter < max_lp_iters:
        conv = conv_fft(a)
        M = conv.max()

        # Build the full convolution matrix A (shape (2n‑1, n))
        # A @ b = conv(f, b)
        c = np.concatenate([a, np.zeros(n - 1)])
        r = np.concatenate([a[:1], np.zeros(n - 1)])
        A = toeplitz(c, r)                     # dense matrix, size ≈ 5k × 2.5k

        # Solve: maximise Σ b_i  ⟺ minimise -Σ b_i
        lin_obj = -np.ones(n)
        ub = np.full(2 * n - 1, M)

        # Bound each variable to be non‑negative
        bounds = [(0.0, None)] * n

        # Use the HiGHS interior‑point/simplex solver (fast for moderate size)
        res = linprog(lin_obj, A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print("[lp] linprog failed (status:", res.message, "); stopping refinements")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # Normalise b to unit L¹‑norm
        g = b / sum_b

        # Convex combination with a modest step size
        t = 0.35
        a = (1.0 - t) * a + t * g
        a = project_onto_simplex(a)

        new_score = 2.0 * n * conv_fft(a).max()
        lp_iter += 1
        print(f"[lp] iter {lp_iter:02d} – score {new_score:.6f}")

        if new_score < cur_score:
            cur_score = new_score
            global_best_a = a.copy()
            global_best_score = new_score

        # early exit if we are already under the target threshold
        if cur_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")
    print(f"[info] best internal score = {global_best_score:.6f}")

    return global_best_a.tolist()
