"""
Propose a low‑value step‑function using a smoothed‑max stochastic
gradient descent on the simplex.  The objective  f(a)=max_k (a*a)_k
is approximated by a soft‑max with a temperature that is annealed.
The gradient of the smooth objective can be written as a
convolution of the soft‑max weights with the reversed current
distribution, which we evaluate by two FFTs per iteration.
Updates are performed with Euclidean projection onto the probability
simplex, combined with occasional random mass‑transfer moves and
restart‑on‑stagnation.  This differs from the previous active‑set
SGD and typically reaches a lower bound within the same time budget.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence for a normalized (sum=1) array."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)   # sum(a)=1 → denominator =1


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Generate a power‑law shape a_i ∝ (i+offset)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n-i) (i=0..n‑1), normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Search for a non‑negative step‑function with a low evaluation value."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 3000

    # --------------------------------------------------------------

    for p in exponents:
        for off in offsets:
            candidates.append(build_power_law(n, p, off))

    candidates.append(build_linear_descending(n))

    # use previous best if it exists

        try:

                prev = tmp
            else:

            prev = np.maximum(prev, 0.0)
            s = prev.sum()

        except Exception:

    # --------------------------------------------------------------
