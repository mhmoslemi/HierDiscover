"""
Propose a step‑function (non‑negative list) that minimizes `evaluate_sequence`.

Algorithm
---------
1. **Parametric warm‑start** –  several simple families (geometric,
   power‑law) are generated on a modest length (n≈800).  Each candidate is
   normalised to sum‑one and evaluated; the best parameters are kept.
2. **Upscaling** – the chosen shape is created directly at a larger length
   (n≈2400) and becomes the start point for a long projected‑gradient
   descent on the probability simplex.  The sub‑gradient is the exact
   average over all active convolution entries (those attaining the
   current maximum).  Nesterov‑style momentum and a step size η₀/√t give a
   fast, stable descent.  The loop runs until the time budget is nearly
   exhausted, always remembering the best vector seen.
3. **Local stochastic refinement** – a cheap simulated‑annealing walk that
   transfers a tiny amount of mass between random indices accepts
   downhill moves and occasional uphill moves according to a geometric
   temperature schedule.  This squeezes a little more out of the solution.

All helpers are top‑level functions; the code respects the 1000 s budget
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simple helpers
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are used.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thr = max_val * (1.0 - rel_tol)
    act = np.where(conv >= thr)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if act.size > 0:
        g /= float(act.size)
    return g


# ------------------------------------------------------------
# Shape generators for the warm‑start
# ------------------------------------------------------------
def _geom_shape(n: int, eps: float) -> np.ndarray:
    """Geometric (exponential) shape a_i = eps * (1‑eps)^i, normalised."""
    if eps <= 0.0 or eps >= 1.0:
        return np.full(n, 1.0 / n)
    r = 1.0 - eps
    a = eps * (r ** np.arange(n, dtype=float))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^{-p}, normalised."""
    if p <= 0.0:
        return np.full(n, 1.0 / n)
    a = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ------------------------------------------------------------
# Gradient descent on the simplex
# ------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         beta: float = 0.95,
                         step0: float = 1.0,
                         max_iter: int = 10**8) -> tuple[np.ndarray, float]:
    """
    Projected (Nesterov‑style) gradient descent.
    Returns best sequence and its score (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing refinement
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 80000) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer annealing."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal iter={it:,} score={score:.6f}")

    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list) that minimises
    `evaluate_sequence`.  The routine:
      * searches a few parametric families for a good warm‑start,
      * up‑scales the best shape to n≈2400,
      * runs a long projected‑gradient descent,
      * finishes with a brief annealing phase.
    The best sequence found before the budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a little safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Stage 0 – parametric warm‑start on a modest size
    # ------------------------------------------------------------------
    n_small = 800
    best_small = None
    best_small_score = float("inf")
    best_kind = None
    best_param = None

    # geometric (exponential) family – eps = (1‑r)
    eps_grid = np.logspace(-5, -1, num=30)   # from 1e‑5 to 0.1
    for eps in eps_grid:
        a = _geom_shape(n_small, eps)
        score = float(2 * n_small * _conv_self(a).max())
        if score < best_small_score:
            best_small_score = score
            best_small = a
            best_kind = "geom"
            best_param = eps

    # power‑law family
    p_grid = np.linspace(0.2, 4.0, num=30)
    for p in p_grid:
        a = _powerlaw_shape(n_small, p)
        score = float(2 * n_small * _conv_self(a).max())
        if score < best_small_score:
            best_small_score = score
            best_small = a
            best_kind = "pow"
            best_param = p

    # fallback to a random start if nothing succeeded (should not happen)
    if best_small is None:
        best_small = np.random.rand(n_small)
        best_small /= best_small.sum()
        best_kind = "rand"
        best_param = None

    print(f"[{time.time() - start_time:6.2f}s] warm‑start ({best_kind}) score={best_small_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – up‑scale to a larger length and run prolonged GD
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_small]

    n_large = 2400
    if best_kind == "geom":
        a0_large = _geom_shape(n_large, best_param)
    elif best_kind == "pow":
        a0_large = _powerlaw_shape(n_large, best_param)
    else:  # random
        a0_large = np.random.rand(n_large)
        a0_large /= a0_large.sum()

    # Reserve a small slice for final annealing (≈5 % of remaining time)
    remaining = deadline - time.time()
    anneal_time = max(5.0, 0.05 * remaining)
    grad_deadline = deadline - anneal_time

    a_best, best_score = _run_gradient_descent(
        a0_large,
        grad_deadline,
        start_time,
        beta=0.95,
        step0=1.0,
        max_iter=10**7,
    )
    print(f"[{time.time() - start_time:6.2f}s] GD finished best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – brief simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            a_best,
            deadline,
            start_time,
            start_temp=5e-2,
            end_temp=1e-5,
            steps=80000,
        )
        if score_ref < best_score:
            best_score = score_ref
            a_best = a_ref
            print(f"[{time.time() - start_time:6.2f}s] annealing improved to {score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    a_best = np.asarray(a_best, dtype=float)
    s = a_best.sum()
    if s > 0:
        a_best /= s
    else:
        a_best = np.full_like(a_best, 1.0 / a_best.size)

    return [float(x) for x in a_best]
