"""
Propose a candidate step sequence by stochastic Adam optimisation on the simplex,
with a soft‑max smoothing of the auto‑convolution peak that is annealed to a
pure sub‑gradient phase.  Random restarts, occasional Gaussian perturbations,
and a simple replay of the best solution are used.  The routine respects the
time‑budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def half_binomial_series(n: int) -> np.ndarray:
    """Return the first n half‑binomial coefficients normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.int64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def next_pow_two(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
def compute_full_convolution(a: np.ndarray, Nfft: int) -> np.ndarray:
    """Linear convolution a*a, returned length 2*len(a)-1."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)
    L = 2 * a.shape[0] - 1
    return conv[:L]

def softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float, Nfft: int) -> np.ndarray:
    """Gradient of the smooth max (log‑sum‑exp) of the auto‑convolution."""
    maxc = conv.max()
    # stabilised exponentials
    w = np.exp(beta * (conv - maxc))
    w /= w.sum()
    # correlation between a and reversed soft‑max weights
    Wr = np.fft.rfft(w[::-1], n=Nfft)
    A = np.fft.rfft(a, n=Nfft)
    corr = np.fft.irfft(A * Wr, n=Nfft)
    start = conv.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad

def maxconv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the true max of the auto‑convolution."""
    k = int(conv.argmax())
    n = a.shape[0]
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs) -> list:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The algorithm uses an Adam optimiser on the probability simplex,
    a temperature‑controlled soft‑max approximation of the convolution peak,
    periodic random restarts and tiny Gaussian perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 3500 works well, increase a little if time permits.
    n = 3800
    L = 2 * n - 1
    Nfft = next_pow_two(3 * n)   # safe size for all FFTs

    # --------------------------------------------------------------
    # Initial point – half‑binomial series, already normalised to sum 1.
    a = half_binomial_series(n)
    a = project_onto_simplex(a)   # safety

    best_a = a.copy()
    conv = compute_full_convolution(a, Nfft)
    best_score = 2.0 * n * conv.max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Soft‑max annealing
    beta = 0.5
    beta_target = 2500.0
    beta_mul = 1.002

    # Learning‑rate schedule
    eta = 0.25
    eta_min = 5e-6
    eta_decay = 0.9994

    # Restart / noise control
    last_improve = 0
    restart_gap = 300_000           # iterations without progress → restart
    noise_prob = 0.004
    noise_scale = 4e-4

    # Main loop
    deadline = time.time() + budget_s - 5.0   # safety margin
    it = 0
    while time.time() < deadline:
        # ---------- convolution and score ----------
        conv = compute_full_convolution(a, Nfft)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ---------- bookkeeping ----------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:09d}] new best {best_score:.6f}")

        # ---------- gradient ----------
        if beta < beta_target:
            grad = softmax_gradient(a, conv, beta, Nfft)
            beta = min(beta_target, beta * beta_mul)
        else:
            grad = maxconv_gradient(a, conv)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        step = eta * m / (np.sqrt(v) + eps)

        # prevent any coordinate from becoming negative after the step
        step = np.minimum(step, a * 0.995)

        a_candidate = a - step
        a = project_onto_simplex(a_candidate)

        # ---------- learning‑rate decay ----------
        eta = max(eta * eta_decay, eta_min)

        # ---------- occasional Gaussian jitter ----------
        if rng.random() < noise_prob:
            noise = (rng.random(n) - 0.5) * noise_scale
            a = a + noise
            a = project_onto_simplex(a)

        # ---------- restart logic ----------
        if it - last_improve > restart_gap:
            # try a sparse Dirichlet start (alpha<1 encourages zeros)
            alpha = 0.5
            a = rng.dirichlet(np.full(n, alpha))
            a = project_onto_simplex(a)
            last_improve = it
            # a cheap score estimate for logging
            tmp_conv = compute_full_convolution(a, Nfft)
            tmp_score = 2.0 * n * tmp_conv.max()
            print(f"[iter {it:09d}] restart – score≈{tmp_score:.6f}")

        # ------------------------------------------------------
        it += 1

    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")
    return best_a.tolist()

    restart_gap = 300_000          # iterations without improvement → restart
    noise_prob = 0.004
    noise_scale = 4e-4

    # ------------------- main optimisation loop --------------------
    deadline = time.time() + budget_s - 5.0   # safety margin
    it = 0
    while time.time() < deadline:
        # ---- convolution and current score -----------------------
        conv = compute_full_convolution(a, Nfft)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ---- keep the best solution seen -------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:09d}] new best {best_score:.6f}")

        # ---- gradient (soft‑max → pure sub‑gradient) ------------
        if beta < beta_target:
            grad = softmax_gradient(a, conv, beta, Nfft)
            beta = min(beta_target, beta * beta_mul)
        else:
            grad = maxconv_gradient(a, conv)

        # ---- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        step = eta * m / (np.sqrt(v) + eps)

        # prevent a coordinate from becoming negative after the step
        step = np.minimum(step, a * 0.995)

        a_candidate = a - step
        a = project_onto_simplex(a_candidate)

        # ---- learning‑rate decay ---------------------------------
        eta = max(eta * eta_decay, eta_min)

        # ---- occasional Gaussian jitter ---------------------------
        if rng.random() < noise_prob:
            noise = (rng.random(n) - 0.5) * noise_scale
            a = a + noise
            a = project_onto_simplex(a)

        # ---- restart if stuck ------------------------------------
        if it - last_improve > restart_gap:
            # 20 % of restarts start from a uniform distribution
            if rng.random() < 0.2:
                a = np.full(n, 1.0 / n)
            else:
                # Dirichlet with α < 1 encourages sparsity
                alpha = 0.5
                a = rng.dirichlet(np.full(n, alpha))
            a = project_onto_simplex(a)

            # reset optimiser state
            m.fill(0.0)
            v.fill(0.0)
            beta = 0.5
            eta = 0.25
            last_improve = it

            tmp_conv = compute_full_convolution(a, Nfft)
            tmp_score = 2.0 * n * tmp_conv.max()
            print(f"[iter {it:09d}] restart – score≈{tmp_score:.6f}")

        it += 1

    # -----------------------------------------------------------------
    # final evaluation via the provided routine (clipping etc. is handled there)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
