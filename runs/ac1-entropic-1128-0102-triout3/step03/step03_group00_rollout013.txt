"""
Propose a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`.  The algorithm follows three phases:

1. **Fast simplex gradient descent** on a modest size (n≈800) to obtain a
   good non‑uniform shape.
2. **LP‑guided mixing** – repeatedly solve a linear program that finds a
   direction `b` with maximal L1‑mass while keeping the convolution
   `f*b` below the current worst convolution value `M`.  Mix `b` into the
   current sequence and run a short projected‑gradient refinement.
   This tends to flatten the auto‑convolution beyond what pure gradient
   descent can achieve.
3. **Upsample** the best small‑size solution to a larger size (n≈2400) and
   perform a long gradient‑descent run followed by a brief simulated‑annealing
   refinement.

All steps respect the 1000 s time budget and keep the vector on the
probability simplex (sum = 1), which makes the objective scale‑invariant.
"""

import time
import math
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) convolution a * a using FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of max‑convolution (average over active set)
# ------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)
    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]
    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    if act.size > 0:
        g /= float(act.size)
    return g


# ------------------------------------------------------------
# Projected gradient descent on the simplex (Nesterov momentum)
# ------------------------------------------------------------
def _run_gradient_descent_full(a0: np.ndarray, n: int,
                              deadline: float, start_time: float,
                              beta: float = 0.9,
                              step0: float = 0.6,
                              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Projected gradient descent; returns best point found and its score."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, iter={it}, score={cur_score:.6f}")
        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)
    return best_a, best_score


# ------------------------------------------------------------
# Linear programme: maximise sum(b) subject to f*b ≤ M  (entry‑wise)
# ------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """Solve  max Σ b_j  s.t.  b ≥ 0  and  (f * b) ≤ M (linear convolution)."""
    n = f.shape[0]
    # Build sparse matrix A where (A @ b)[k] = (f * b)[k]
    i = np.repeat(np.arange(n, dtype=np.int32), n)
    j = np.tile(np.arange(n, dtype=np.int32), n)
    rows = i + j                     # k = i + j
    cols = j                         # variable index
    data = f[i]                      # coefficient f_i for each pair
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    # maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    # fallback – uniform distribution
    return np.full(n, 1.0 / n, dtype=np.float64)


# ------------------------------------------------------------
# Linear interpolation (upsampling) with renormalisation
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ------------------------------------------------------------
# Simulated annealing refinement (cheap mass‑transfer moves)
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 80000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, iter={it}, score={score:.6f}")
    return best_seq, best_score
