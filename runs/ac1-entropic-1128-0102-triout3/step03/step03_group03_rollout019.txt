Propose a candidate step‑function using a fast Adam‑based mirror‑descent
optimizer.  The algorithm

* starts from the best known construction (global ``height_sequence_1``) or
  a random Dirichlet point,
* progressively upsamples to a target length (≤8000),
* repeatedly evaluates the full auto‑convolution via FFT,
* computes a sub‑gradient from the (up to 64) indices whose convolution is
  within a tiny relative tolerance of the current maximum,
* performs a multiplicative Adam update on the logarithms of the
  coefficients (mirror‑descent on the probability simplex) and
* finishes with a cheap stochastic mass‑transfer hill‑climber.

All updates keep the vector strictly positive; after each update the
vector is renormalised to sum 1, which leaves the original objective
invariant.  The routine runs for at most ``budget_s`` seconds and returns
the best sequence seen.
"""

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full auto‑convolution a*a using FFT."""
    n = a.shape[0]
    length = 2 * n - 1
    fft_len = 1 << ((length).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:length]
    return conv

def _score_of_seq(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a sequence that already sums to 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())

def _active_set(conv: np.ndarray, max_val: float,
                rel_eps: float = 0.001, max_add: int = 64) -> np.ndarray:
    """Return indices whose values are within ``rel_eps`` of the maximum,
    padded up to ``max_add`` indices if needed."""
    eps = max_val * rel_eps
    idx = np.where(conv >= max_val - eps)[0]
    if idx.size < max_add:
        topk = min(max_add, conv.shape[0])
        idx = np.unique(np.argpartition(conv, -topk)[-topk:])
    return idx

def _gradient_active(a: np.ndarray, K: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) obtained from the active set K.
    For each active convolution index k the contribution to component i is
    2·a_{k‑i} whenever the index is in range.

    n = a.shape[0]

        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g

        idx = np.unique(np.argpartition(conv, -topk)[-topk:])
    return idx


def _gradient_active(a: np.ndarray, K: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) obtained from the active set K.
    For each active convolution index k the contribution to component i is
    2·a_{k‑i} whenever the index is in range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) with a low
    `evaluate_sequence` score.  The routine runs for at most ``budget_s``
    seconds and prints intermediate improvements.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise – try to reuse the best known construction.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # global variable, may exist
        init = np.maximum(init, 0.0)
        if init.sum() == 0.0:
            raise ValueError
        init = init / init.sum()
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"Initialised from height_sequence_1 (n={len(best_seq)}), "
              f"score={best_score:.6f}")
    except Exception:
        # fallback: random Dirichlet point
        n0 = 2000
        init = np.random.rand(n0)
        init = init / init.sum()
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"Random initialisation (n={n0}), score={best_score:.6f}")

    current = best_seq.copy()
    cur_n = current.shape[0]

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling towards a target size.
    # ------------------------------------------------------------------
    TARGET_LEN = 5000          # size where we will spend most iterations
    MAX_LEN = 8000             # absolute ceiling (still cheap enough)

    while cur_n < TARGET_LEN and time.time() < deadline:
        next_n = min(int(cur_n * 1.45) + 1, TARGET_LEN)
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a = np.interp(x_new, x_old, current)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        current = a
        cur_n = next_n
        print(f"Upsampled to n={cur_n}")

    # ------------------------------------------------------------------
    # 3. Main optimisation on the final size (mirror‑descent with Adam).
    # ------------------------------------------------------------------
    if cur_n > MAX_LEN:
        cur_n = MAX_LEN
        current = current[:MAX_LEN]
        current = current / current.sum()

    # Adam state (operating on log‑probabilities)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(current)
    v = np.zeros_like(current)
    t = 0
    log_a = np.log(np.maximum(current, 1e-15))

    # How many gradient steps shall we attempt?  We cap at a large number
    # but also stop once the time budget is exhausted.
    MAX_ITERS = 600_000
    it = 0

    while time.time() < deadline and it < MAX_ITERS:
        it += 1

        # ----- evaluate current sequence --------------------------------
        a = np.exp(log_a)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        conv = _full_convolution(a)
        max_val = conv.max()
        cur_score = 2.0 * cur_n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start:6.2f}s] iter={it}, n={cur_n}, "
                  f"improved score={cur_score:.6f}")

        # ----- sub‑gradient from the active set ------------------------
        K = _active_set(conv, max_val, rel_eps=0.001, max_add=64)
        g = _gradient_active(a, K)

        # ----- Adam on log‑variables (mirror descent) -----------------
        t += 1
        lr = 0.3 / math.sqrt(t)                 # decreasing step size
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep numbers in a stable range
        log_a -= np.max(log_a)                  # prevents overflow in exp()
        a = np.exp(log_a)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        log_a = np.log(a)

        # occasional tiny random perturbation to break stagnation
        if it % 5000 == 0 and (deadline - time.time()) > 2.0:
            i, j = np.random.randint(0, cur_n, size=2)
            eps_move = np.random.rand() * 0.02
            delta = eps_move * a[i]
            if delta > 0:
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                log_a = np.log(a)

    # ------------------------------------------------------------------
    # 4. Final cheap stochastic hill‑climbing (mass transfers).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    local_iters = 30_000
    for it_local in range(local_iters):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps_move = np.random.rand() * 0.08          # up to 8 % of source mass
        delta = eps_move * a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        conv = _full_convolution(a_new)
        new_score = 2.0 * n * conv.max()
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            print(f"[{time.time() - start:6.2f}s] local improve it={it_local}, "
                  f"score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # Convert to plain Python floats for the final output
    return [float(x) for x in best_seq]
