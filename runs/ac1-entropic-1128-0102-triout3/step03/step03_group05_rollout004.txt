"""
Propose a step‑function sequence with a low evaluation value.

Algorithm overview
------------------
* Choose a sequence length *n* (larger than the previous 3000 when time allows).
* Build a small pool of analytically motivated initial shapes
  (power‑law, linear, uniform, and an up‑sampled previous best).
* Pick the best of those initial candidates according to a fast FFT based score.
* Run a projected gradient descent on the simplex:
    – Each iteration computes the self‑convolution of the current vector.
    – The *K* largest convolution entries are taken and a soft‑max weighting
      (controlled by a temperature parameter that rises during the run) builds
      a smooth approximation of the max‑operator.
    – A weighted sub‑gradient of the smoothed max is obtained by summing the
      individual gradients of the selected convolution entries.
    – The gradient is normalised and a diminishing step is taken;
      the new point is projected back onto the probability simplex.
* Occasionally perform random “mass‑transfer’’ moves to escape plateaus.
* If no improvement has been observed for a while, re‑seed the iterate from a
  noisy perturbation of the best solution seen so far.
* The loop runs until the allotted time (≈1000 s) expires and the best sequence
  found is returned.

The implementation is fully self‑contained, uses only NumPy, and respects all
constraints of the evaluation function (non‑negative entries, no I/O, etc.).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Efficient version of `evaluate_sequence` (no safety checks).
    The score is invariant under scaling, so the vector is assumed to be
    normalised (sum == 1).  Returns the same value as `evaluate_sequence`.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, ∑ x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n-i) (i=0..n‑1) normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def weighted_gradient_multi_active(a: np.ndarray,
                                   idxs: np.ndarray,
                                   weights: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of the (soft‑max of) convolution max.
    For each selected convolution index k we add 2·a_{k‑i} weighted by
    the corresponding weight.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k, w in zip(idxs, weights):
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i = np.arange(low, high + 1)
        j = k - i
        grad[i] += 2.0 * w * a[j]
    return grad

            best_val = cur_val
            best_seq = a_new.copy()
            last_improve = time.time()

        # (g) occasional uphill acceptance (helps escape shallow plateaus)
        elif np.random.rand() < uphill_prob:
            a = a_new

        # --------------------------------------------------------------
        # (h) random “mass‑transfer’’ moves (exploration)
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 1e-12:
                delta_transfer = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta_transfer
                cand[j] += delta_transfer
                cand = proj_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < uphill_prob:
                    a = cand

        # --------------------------------------------------------------
        # (i) stagnation handling – re‑seed from a noisy perturbation of the best
        if time.time() - last_improve > 30.0:
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(0.85 * best_seq + 0.15 * noise)
            last_improve = time.time()

        # --------------------------------------------------------------
        # (j) periodic reporting
        if it % 2000 == 0:
            print(f"[iter {it:7d}] cur {cur_val:.6f}  best {best_val:.6f}")

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # Return as plain Python list; the external checker will renormalise if needed.
    return best_seq.tolist()
