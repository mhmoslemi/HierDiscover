    """Project vector v onto the probability simplex (sum=1, all >=0)."""
    # Implementation based on algorithm by Wang & Carreira-Perpinan.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _conv_and_max(a):
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()

"""
Propose a low‑value step function for `evaluate_sequence`.

The search proceeds in three stages:

1. **Parametric power‑law initialization** – for a given length `n`
   we generate vectors `a_i ∝ (i+1)^{-p}` for a grid of exponents `p`
   (including the previous best construction if it exists) and keep the
   best‐scoring one.

2. **Gradient‑directed mass‑transfer refinement** – the objective
   `2·n·max(conv(a,a))` (with `sum(a)=1`) is minimized by repeatedly
   moving a small amount of mass from the index with the largest
   sub‑gradient to the index with the smallest sub‑gradient.  The
   sub‑gradient is obtained from the active set of convolution entries
   that are (near) maximal.  If a move does not improve the score a
   small exponentiated‑gradient (mirror‑descent) step on the simplex is
   taken as a fallback.

3. **Random hill‑climbing** – a final phase of cheap random
   mass‑transfer moves (up to 4 % of the source mass) is performed
   until the time budget expires.

All updates keep the vector non‑negative and normalised, so the final
sequence can be passed directly to `evaluate_sequence`.  Intermediate
scores and elapsed time are printed for progress monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """
    Return the objective value 2·n·max(conv(a,a)) for a vector that already
    sums to 1 (the denominator in `evaluate_sequence` is then 1).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """
    Return a modest set of indices that are (almost) maximal.
    All indices with value >= max_val - rel_eps*max_val are kept;
    if fewer than `max_add` are found, the globally largest entries are
    added to reach `max_add` indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of `max(conv)` for the active set `K`.
    For each active convolution index k the contribution to component i is
    2·a_{k-i} whenever the index lies in the valid range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    (non‑negative entries summing to 1).
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are negative – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  The routine respects a ~1000 s wall‑clock budget and prints
    intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    # --------------------------------------------------------------
    # 1. Choose problem size and a first feasible vector.
    # --------------------------------------------------------------
    try:
        # use the previously found construction if it exists

        init = init / s
        n = init.shape[0]
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] Init from height_sequence_1 "
              f"(n={n}) → score={best_score:.6f}")
    except Exception:
        # fall back to a power‑law shape (length≈5000)
        n = 5000
        best_seq = None
        best_score = float("inf")
        print(f"[{time.time() - start_time:6.2f}s] No previous construction – "

    # --------------------------------------------------------------
    # 2. Power‑law grid search (executed only if we did not have init).
    # --------------------------------------------------------------
    if best_seq is None:

        for p in p_vals:
            if time.time() > deadline:

            i = np.arange(n, dtype=float) + 1.0            # 1‑based index

            best_seq = current.copy()
            print(f"[{time.time() - start_time:6.2f}s] Iter {it:5d} "
                  f"→ new best score={cur_score:.6f}")

        # (a) active‑set sub‑gradient
        # ----------------------------------------------------------

        g = _gradient_active(current, K)

        # ----------------------------------------------------------
        # (b) greedy mass transfer from high‑gradient to low‑gradient
        # ----------------------------------------------------------
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        if i_max != i_min and current[i_max] > 0:
            # move at most 5 % of the source mass
            delta = min(0.05 * current[i_max], current[i_max])
            cand = current.copy()
            cand[i_max] -= delta
            cand[i_min] += delta
            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < cur_score - 1e-12:
                current = cand
                continue          # successful greedy step, restart loop

        # ----------------------------------------------------------
        # (c) fallback: mirror‑descent (exponentiated‑gradient) step
        # ----------------------------------------------------------
        eta = 0.5 / math.sqrt(it)          # decaying step size
        log_a = np.log(np.maximum(current, 1e-15)) - eta * g
        log_a -= np.max(log_a)             # stabilise
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        current = a_new
        # loop continues

    # --------------------------------------------------------------
    # 5. Final cheap random hill‑climbing (small mass swaps).
    # --------------------------------------------------------------
    if time.time() < deadline:
        local_iters = 15000
        for _ in range(local_iters):
            if time.time() > deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j or current[i] == 0:
                continue
            # propose moving up to 4 % of the source mass
            eps = np.random.rand() * 0.04
            delta = eps * current[i]
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            # safety (should not happen)
            if cand[i] < 0:
                continue
            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = cand.copy()
                current = cand
                print(f"[{time.time() - start_time:6.2f}s] Random hill‑climb "
                      f"→ score={new_score:.6f}")

    # --------------------------------------------------------------
    # 6. Return the best vector found (as plain Python floats).
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
