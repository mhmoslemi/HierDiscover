import numpy as np
import time

def fast_score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence: computes 2 * n * max(conv) / sum(a)**2.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Return a power‑law shape a_i ∝ (i+offset)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

def build_linear_descending(n: int) -> np.ndarray:
    """Return a linearly decreasing shape a_i ∝ (n‑i), normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a

    for p in exponents:
        for off in offsets:

    stagnation_secs = 30.0   # time after which we re‑seed from best
    last_improve = time.time()

    # pre‑compute FFT length (need at least 3n‑2 points for gradient)
    fft_len = 1 << ((3 * n - 2).bit_length())
    grad_start = n - 1
    grad_end   = 2 * n - 1   # exclusive

    it = 0
    while time.time() < deadline:
        # ---- a) self‑convolution (used for the smooth max) ------------
        fft_a = np.fft.rfft(a, fft_len)
        conv_self = np.fft.irfft(fft_a * fft_a, fft_len)[: 2 * n - 1]
        max_conv = conv_self.max()

        # ---- b) soft‑max weights ---------------------------------------
        beta = min(beta_max, beta_start + beta_inc * it)
        # shift for numerical stability
        exp_arg = beta * (conv_self - max_conv)
        # clamp to avoid overflow on very large negative numbers
        exp_arg = np.clip(exp_arg, -700, 0)   # np.exp(-700) ~ 5e-305
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to uniform weighting (should not happen)
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- c) gradient of the smoothed max -------------------------
        # gradient = 2 * (a * w_rev)_{n-1 : 2n-2}
        w_rev = w[::-1]
        fft_w_rev = np.fft.rfft(w_rev, fft_len)
        grad_full = np.fft.irfft(fft_a * fft_w_rev, fft_len)
        grad = 2.0 * grad_full[grad_start:grad_end]   # length n

        # ---- d) exponentiated gradient update -------------------------
        # step size decays like 1/sqrt(iter)
        step = eta0 / np.sqrt(it + 1.0)
        # multiplicative update; keep a strictly positive to avoid log(0)
        a_mul = a * np.exp(-step * grad)
        a_new = a_mul / a_mul.sum()   # automatically on simplex

        # ---- e) evaluate -----------------------------------------------
        cur_val = fast_score(a_new)

        # ---- f) accept -------------------------------------------------
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
        else:
            if rng.random() < uphill_prob:
                a = a_new

        # ---- g) occasional random mass‑transfer ------------------------
        if rng.random() < random_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-12:
                transfer = random_move_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= transfer
                cand[j] += transfer
                cand = proj_simplex(cand)   # keep on simplex
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif rng.random() < uphill_prob:
                    a = cand

        if time.time() - last_improve > stagnation_secs:
            # inject a small random component into the current best
            noise = rng.random(n)
            noise /= noise.sum()
            a = proj_simplex(0.9 * best_seq + 0.1 * noise)
            last_improve = time.time()

            print(f"[iter {it:7d}] cur {cur_val:.6f}  best {best_val:.6f}  beta {beta:.1f}")

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # The external checker normalises; returning the raw vector is fine.
    return best_seq.tolist()

import numpy as np
import time


def fast_score(a: np.ndarray) -> float:
    """
    Fast version of the official ``evaluate_sequence``.
    Returns 2·n·max(a∗a) / (sum(a)²).  The input is assumed to be non‑negative.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
        {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n‑i), normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a
