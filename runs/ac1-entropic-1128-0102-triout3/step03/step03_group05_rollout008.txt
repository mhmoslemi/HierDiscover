"""
propose_candidate
-----------------

Search for a low‑value step‑function (non‑negative vector on the simplex) that
minimises

    f(a) = 2·n·max_k (a * a)_k       (the sum of a is fixed to 1)

The code uses a *smooth* approximation of the max – a log‑sum‑exp over the
convolution entries – and performs an exponentiated‑gradient (mirror‑descent)
update with Adam‑style moment adaptation.  The smoothing parameter α is
annealed from a small value to a large one, which gradually turns the smooth
objective into the true max.  Occasionally random “mass‑transfer’’ moves are
tried to escape plateaus and a short stagnation‑restart is performed.

All heavy lifting (convolution, correlation) is carried out with FFTs, so a
single iteration costs only a few milliseconds for n≈3000.  The routine runs
for at most the supplied time budget (≈1000 s) and returns the best sequence
found during the run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (assumes a is non‑negative, sum≈1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n-i) for i=0..n-1, normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def convolve_self(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution a * a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def weighted_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the cross‑correlation (a ⋆ w) needed for the gradient.
    Returns an array of length n containing
        2 * Σ_j a_j w_{j+i}  for i=0..n‑1
    """
    n = a.shape[0]
    L = 1 << ((3 * n - 2 - 1).bit_length())          # length ≥ n+(2n‑1)-1
    fft_a = np.fft.rfft(a, L)
    fft_w = np.fft.rfft(w, L)
    # cross‑correlation = irfft(conj(fft_a) * fft_w)
    corr_full = np.fft.irfft(np.conj(fft_a) * fft_w, L)
    offset = w.shape[0] - 1                           # =2n‑2
    grad = 2.0 * corr_full[offset: offset + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, seed: int = 42) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluate_sequence() value.  The routine respects the given time budget
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    n = 3000

    # --------------------------------------------------------------
    # Build a small pool of analytically motivated starting points.
    candidates = []

    # power‑law family (various exponents & offsets)
    for p in [0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60]:
        for off in [0.0, 0.1, 0.5, 1.0, 2.0]:
            candidates.append(build_power_law(n, p, off))

    # linear descending shape
    candidates.append(build_linear_descending(n))

    # uniform – baseline
    candidates.append(np.full(n, 1.0 / n))

    # if a previous best is available, reuse its first n entries
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                seq = np.zeros(n, dtype=np.float64)
                seq[:prev.size] = prev
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0.0:
                seq /= seq.sum()
                candidates.append(seq)
        except Exception:
            pass

    # --------------------------------------------------------------
    # Pick the best initial candidate (according to fast_score).
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best initial value: {best_val:.6f}")

    # --------------------------------------------------------------
    # Optimisation parameters (Adam‑style exponentiated gradient).
    a = best_seq.copy()               # always stays on the simplex
    a /= a.sum()                      # safeguard, should already be 1
    best_seq = a.copy()
    best_val = 2.0 * n * convolve_self(a).max()

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters
    eta0 = 0.30                     # base step size for EG update
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smoothing schedule (log‑sum‑exp ↦ max)
    alpha_start = 5.0
    alpha_end = 400.0

    # exploration
    random_move_prob = 0.02
    random_move_frac = 0.25

    it = 0
    last_improve = start_time

    # --------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) self‑convolution and its maximum (needed for the true objective)
        conv = convolve_self(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv           # because Σa = 1

        # ------------------------------------------------------------------
        # 2) smooth‑max gradient
        #    anneal α linearly in time
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (budget_s * 0.9))
        alpha = alpha_start + frac * (alpha_end - alpha_start)

        # stable exponentiation
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()           # normalized weights

        grad = weighted_correlation(a, w)       # length n, non‑negative

        # ------------------------------------------------------------------
        # 3) Adam moments (first & second) for the exponentiated‑gradient
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step size (slow decay)
        eta = eta0 / np.sqrt(it + 1)

        # 4) EG update on the simplex
        step = eta * (m_hat / (np.sqrt(v_hat) + eps))
        # protect against extreme values that could cause underflow/overflow
        step = np.clip(step, -20.0, 20.0)
        a = a * np.exp(-step)
        a_sum = a.sum()
        if a_sum == 0.0:                     # extremely unlikely, but safe‑guard
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ------------------------------------------------------------------
        # 5) keep the best solution seen so far
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = time.time()

        # ------------------------------------------------------------------
        # 6) occasional random mass‑transfer move (exploration)
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand /= cand.sum()
                cand_val = 2.0 * n * convolve_self(cand).max()
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < 0.001:   # tiny uphill probability
                    a = cand

        # ------------------------------------------------------------------
        # 7) stagnation handling – if no improvement for ~30 s, reseed
        if time.time() - last_improve > 30.0:
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = 0.7 * best_seq + 0.3 * noise
            a /= a.sum()
            last_improve = time.time()

        # ------------------------------------------------------------------
        # 8) progress report (lightweight)
        if it % 2000 == 0:
            print(f"[it {it:6d}] cur {cur_val:.6f}  best {best_val:.6f}")

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # The external checker will normalise if necessary – we already have Σ=1.
    return best_seq.tolist()
