"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max_k (a*a)[k] / (∑ a_i)^2
the same objective that ``evaluate_sequence`` uses.

The search consists of two phases:

1. **Parametric sweep** – the sequence is taken from a small family of
   analytically‑motivated shapes:
        • pure power‑law  a_i ∝ (i+off)^{-p}
        • symmetric power‑law a_i ∝ (i+off)^{-p} + (n‑i+off)^{-p}
   The exponent ``p`` is searched on a fine grid (0.4…0.7) and several
   offsets (0…5).  Every candidate is normalised to unit L1‑norm and
   scored with a fast FFT based evaluator.  The best candidate found is
   kept as a starting point for the next phase.

2. **Refinement by projected stochastic gradient descent** – starting
   from the best candidate we perform a few hundred (or up to the time
   budget) projected gradient steps on the simplex (∑a_i = 1, a_i≥0).
   The gradient is a sub‑gradient of the max‑convolution term, built by
   aggregating the contributions of the currently active convolution
   indices (those whose value is within a small relative gap of the
   maximum).  Occasionally small random “mass‑transfer’’ moves are tried
   to escape plateaus.

The whole routine respects the 1000 s time limit; it returns the best
sequence (as a plain Python list) that it discovered before the deadline.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` (no safety checks).
    Assumes a is a non‑negative 1‑D array and that sum(a) > 0.
    The returned value equals evaluate_sequence(a.tolist()).
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # next power‑of‑two for a clean FFT (speed‑up for large n)
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, ∑ x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def gradient_multi_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)[k]   w.r.t. a,
    aggregating the contributions of the indices in ``active_idxs``.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i = np.arange(low, high + 1)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float, off: float) -> np.ndarray:
    """
    a_i ∝ (i+1+off)^{-p},   i = 0 … n‑1,  normalised to sum 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + off
    a = np.power(idx, -p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑value step‑function within the supplied time budget.
    Returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # 1. Problem size – 3000 works well for the current budget.
    n = 3000

    # --------------------------------------------------------------
    # 2. Helper to keep track of the best sequence.
    best_seq = None
    best_val = np.inf

    def consider(seq: np.ndarray):
        """evaluate ``seq`` and update the global best if improved."""
        nonlocal best_seq, best_val
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[new best] value={val:.6f}")

    # --------------------------------------------------------------
    # 3. Add a few deterministic starters (uniform, linear, previous run).
    consider(np.full(n, 1.0 / n))                     # uniform baseline
    consider(np.arange(n, 0, -1, dtype=np.float64) / np.arange(1, n + 1).sum())  # descending linear

    # previous global solution, if present
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                seq = np.zeros(n, dtype=np.float64)
                seq[:prev.size] = prev
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0:
                seq /= seq.sum()
                consider(seq)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 4. Exhaustive sweep over the parametric families.
    #    (a) pure power‑law
    #    (b) symmetric sum of a power‑law and its reverse
    p_vals = np.linspace(0.40, 0.70, 61)   # step ≈0.005
    offsets = np.linspace(0.0, 5.0, 21)    # step 0.25

    for p in p_vals:
        for off in offsets:
            if time.time() > deadline:
                break
            base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64) + off, p)

            # (a) plain power‑law
            seq_pow = base / base.sum()
            consider(seq_pow)

            # (b) symmetric version – higher mass at both ends
            seq_sym = base + base[::-1]
            seq_sym = seq_sym / seq_sym.sum()
            consider(seq_sym)

        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 5. If the sweep already beat the target, we can stop early.
    target = 1.503
    if best_val <= target:
        print(f"[early stop] target reached with value {best_val:.6f}")
        return best_seq.tolist()

    # --------------------------------------------------------------
    # 6. Fine‑grained refinement using projected stochastic gradient descent.
    a = best_seq.copy()
    it = 0
    eta0 = 0.5                     # initial step‑size
    max_active = 8                # how many indices to aggregate
    delta_start = 0.04            # initial relative gap for the active set
    delta_end = 0.005             # final relative gap
    random_move_prob = 0.02
    random_move_frac = 0.15
    uphill_accept = 0.001
    conv_size = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # (a) convolution and its maximum
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_conv = conv.max()

        # (b) active set – indices whose value is within a shrinking gap of the max
        delta = max(delta_end, delta_start * (1.0 - it / 200000.0))
        mask = conv >= max_conv * (1.0 - delta)
        active = np.flatnonzero(mask)
        if active.size > max_active:
            # keep the truly largest ones
            top = np.argpartition(-conv[active], max_active)[:max_active]
            active = active[top]

        # (c) sub‑gradient for the active set
        grad = gradient_multi_active(a, active)

        # (d) projected gradient step (square‑root decay)
        eta = eta0 / np.sqrt(it + 1)
        a_new = proj_simplex(a - eta * grad)

        # (e) evaluate the candidate
        cand_val = fast_score(a_new)
        if cand_val < best_val:
            best_val = cand_val
            best_seq = a_new.copy()
            a = a_new
        else:
            # occasional uphill acceptance (helps escaping flat plateaus)
            if np.random.rand() < uphill_accept:
                a = a_new

        # (f) occasional random mass‑transfer move
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta_transfer = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()

                cand[j] += delta_transfer
                cand = proj_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                elif np.random.rand() < uphill_accept:
                    a = cand

        # (g) occasional progress report
        if it % 2000 == 0:
            print(f"[iter {it:6d}] cur={cand_val:.6f}  best={best_val:.6f}")

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # The external checker normalises the sequence, so we can return it as‑is.
    return best_seq.tolist()
