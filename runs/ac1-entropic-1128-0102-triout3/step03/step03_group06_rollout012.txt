def adam_optimize(a_init, time_limit, max_iter_per_sec=, alpha=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    start = time.time()
    deadline = start + time_limit
    best_a = a.copy()
    best_val = fast_score(a)
    while time.time() < deadline:
        # compute convolution to find max
        conv = np.fft.irfft(np.fft.rfft(a, conv_size) * np.fft.rfft(a, conv_size), conv_size)[:2*n-1]
        k_max = int(np.argmax(conv))
        # grad
        grad = np.zeros_like(a)
        i = np.arange(a.size)
        j = k_max - i
        mask = (j >= 0) & (j < a.size)
        grad[mask] = 2.0 * a[j[mask]]
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # project onto simplex
        # using existing projection code
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, a_new.size + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(a_new - theta, 0.0)
        # evaluate occasionally
        if t % 5000 == 0:
            val = fast_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[adam iter {t}] new best {best_val:.6f}")
    return best_a

"""
Optimise a non‑negative step‑function a (length n) to minimise

        score = 2·n·max_k (a*a)[k] / (sum_i a_i)^2 .

The optimisation is performed with a smooth approximation of the
maximum (soft‑max) and an Adam‑style projected optimiser that works
directly on the simplex.  The algorithm proceeds as follows

1.  Initialise a on the simplex (a decreasing power‑law works well).
2.  In each iteration
    •  compute the self‑convolution a*a via FFT,
    •  form a soft‑max weight vector w over the convolution entries
       (temperature β is increased slowly during the run),
    •  obtain the gradient of the smooth objective:
          ∇ = 2·corr(a, w)   (linear correlation, computed by FFT),
    •  apply an Adam update and project back onto the probability simplex.
3.  Keep the best (lowest score) vector seen and stop as soon as the
    target score (≤ 1.503) is reached or the time budget expires.

If the smooth optimiser fails to meet the target, a simple projected
sub‑gradient descent (the previous baseline) is used as a fallback.
The returned list is guaranteed to contain non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the score for a non‑negative vector a.
    The vector is assumed to be already normalised (sum>0).
    """
    n = a.size
    total = a.sum()
    if total < 1e-12:
        return float('inf')
    # convolution size – next power of two for efficiency
    size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(a, size)
    conv = np.fft.irfft(ft * ft, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (total * total))


# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray,
                  time_limit: float,
                  target_score: float = 1.503) -> np.ndarray:
    """
    Adam optimiser with a smooth (soft‑max) surrogate for the max.
    Returns the best vector encountered within the allotted time.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    # FFT sizes
    conv_size = 1 << ((2 * n - 1).bit_length())          # for a*a
    grad_size = 1 << ((3 * n - 2).bit_length())          # for correlation a ⋆ w

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.10                     # initial learning rate
    it = 0

    # schedule for the soft‑max temperature β
    beta_start = 0.5
    beta_end = 20.0
    beta_ramp_iters = 400_000      # after this many iters β reaches beta_end

    deadline = time.time() + time_limit - 0.1   # safety margin

    while time.time() < deadline:
        it += 1

        # ----- self‑convolution (used for the surrogate) -----
        ft = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(ft * ft, conv_size)[: 2 * n - 1]

        # ----- soft‑max weights over convolution entries -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp_iters)
        max_conv = conv.max()
        # stabilise the exponentials
        exp_arg = beta * conv - beta * max_conv
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth objective (correlation a ⋆ w) -----
        ft_a = np.fft.rfft(a, grad_size)
        ft_w = np.fft.rfft(w, grad_size)
        corr = np.fft.irfft(np.conj(ft_a) * ft_w, grad_size)[:n]
        grad = 2.0 * corr

        # ----- Adam update -----
        lr = lr0 / (1.0 + it * 1e-6)          # slowly decaying step size
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = project_onto_simplex(a_candidate)

        # ----- occasional random mass‑transfer move (helps escape plateaus) -----
        if np.random.rand() < 0.001:
            i, j = np.random.randint(0, n, size=2)
            if a_candidate[i] > 0.0:
                delta = 0.02 * a_candidate[i] * np.random.rand()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = project_onto_simplex(a_candidate)

        a = a_candidate

        # ----- bookkeeping -----
        if it % 20000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adam iter {it:7d}] new best {best_val:.6f}")
            if best_val <= target_score:
                print("[adam] target reached, exiting early")
                break

    # final check
    final_val = fast_score(best_a)
    print(f"[adam final] best value {final_val:.6f}")
    return best_a

        # sub‑gradient for the active index
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        a = project_onto_simplex(v)

        # occasional random move
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = random_move_step * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = project_onto_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with evaluation score ≤ 1.503.
    Primary strategy – Adam optimiser with a soft‑max surrogate for the
    max‑convolution; fallback – the classic projected sub‑gradient method.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # problem size – the same n as in the previous best run
    n = 1500
    target = 1.503

    # --- 1. initialise a simple decreasing distribution (power‑law) ---
    a0 = 1.0 / np.arange(1, n + 1, dtype=np.float64) ** 0.5
    a0 /= a0.sum()

    # --- 2. try the Adam optimiser (most of the time budget) ---
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("[stage 1] running Adam optimiser")
        a_best = adam_optimize(a0, remaining, target_score=target)
        val_best = fast_score(a_best)
        print(f"[stage 1] best score after Adam: {val_best:.6f}")
    else:
        a_best = a0.copy()
        val_best = fast_score(a_best)

    # --- 3. if still too large, fall back to sub‑gradient descent ---
    if val_best > target:
        remaining = deadline - time.time()
        if remaining > 5.0:
            print("[stage 2] falling back to sub‑gradient optimiser")
            a_refined = subgradient_optimize(a_best, remaining)
            val_refined = fast_score(a_refined)
            print(f"[stage 2] score after sub‑gradient: {val_refined:.6f}")
            if val_refined < val_best:
                a_best, val_best = a_refined, val_refined

    # --- 4. final safety net (uniform distribution) ------------------
    if val_best == float('inf') or np.isnan(val_best):
        print("[stage 3] using uniform fallback")
        a_best = np.full(n, 1.0 / n, dtype=np.float64)
        val_best = fast_score(a_best)

    print(f"[final] returning sequence with score {val_best:.6f}")
    # ensure plain Python list of floats as required
    return a_best.tolist()
