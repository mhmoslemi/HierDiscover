"""
Projected gradient descent on a smooth approximation of the
max‑convolution objective.

The original evaluation is  f(a)=2·n·max_k (a∗a)_k / (∑a_i)^2.
Because the objective is scale‑invariant we work on the simplex
{a_i≥0, ∑a_i=1}.  Instead of the hard max we minimise the
log‑sum‑exp of the self‑convolution, i.e.

    F_β(a)= (1/β)·log Σ_k exp(β·(a∗a)_k) ,

which converges to the max as β→∞.  The gradient of F_β can be
written in closed form using the soft‑max weights w_k∝exp(β·c_k)
where c_k=(a∗a)_k.  Using the FFT we evaluate the convolution,
the weights and the gradient in O(N log N) time.

A small amount of momentum, occasional stochastic ‘mass‑transfer’
moves and a simple stagnation‑re‑seed strategy are added to escape
plateaus.  The routine runs for at most the supplied time budget and
returns the best sequence found (as a plain list of floats).  The
sequence is already normalised to sum = 1, the external checker will
renormalise if necessary.

"""

import numpy as np
import time

# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of `evaluate_sequence` (assumes a is non‑negative,
    already on the simplex)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    a /= a.sum()
    return a


def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = n-i (i=0..n‑1), normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


def build_exponential(n: int, lam: float = 0.001) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def gradient_softmax(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    a : (n,) probability vector
    w : (2n‑1,) soft‑max weights (non‑negative, sum may be 1)
    size_fft : power‑of‑two ≥ 3n‑2 (length for the FFTs)
    Returns ∇_a ( Σ_k w_k·(a∗a)_k ).
    """
    n = a.shape[0]
    # FFT of a (padded to size_fft)
    A_fft = np.fft.rfft(a, size_fft)

    # FFT of w (zero‑padded)
    W_fft = np.fft.rfft(w, size_fft)
    conv_aw = np.fft.irfft(A_fft * W_fft, size_fft)[: 3 * n - 2]

    # FFT of reversed w
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, size_fft)
    conv_arev = np.fft.irfft(A_fft * W_rev_fft, size_fft)[: 3 * n - 2]

    # indices for the reversed part
    idx = (2 * n - 2) - np.arange(n)        # length n
    grad = 2.0 * (conv_aw[:n] + conv_arev[idx])
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation value.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safe margin

    # --------------------------------------------------------------
    # 1. Problem size
    n = 3000

    # --------------------------------------------------------------
    # 2. Build a modest pool of analytically‑motivated starts
    candidates = []

    # power‑law family
    for p in [0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60]:
        for off in [0.0, 0.1, 0.5, 1.0, 2.0]:
            candidates.append(build_power_law(n, p, off))

    # exponential families
    for lam in [0.0005, 0.001, 0.002, 0.005, 0.01]:
        candidates.append(build_exponential(n, lam))

    # linear descending, uniform
    candidates.append(build_linear_descending(n))
    candidates.append(np.full(n, 1.0 / n))

    # previous best, if available
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            seq = np.zeros(n, dtype=np.float64)
            seq[: min(prev.size, n)] = prev[:n]
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0:
                seq /= seq.sum()
                candidates.append(seq)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Choose the best initial candidate (fast_score)
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best initial value: {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop (smooth‑max PGD + random moves)
    a = best_seq.copy()                # current iterate (always on simplex)
    best_seq = a.copy()
    best_val = fast_score(a)

    # hyper‑parameters
    beta = 5.0                 # start of soft‑max temperature (low = smooth)
    beta_max = 300.0           # final beta (more focused on the true max)
    beta_inc = (beta_max - beta) / 300000.0   # linear increase
    eta0 = 0.8                 # initial step size
    mu = 0.9                   # momentum coefficient
    random_move_prob = 0.02    # prob. of a stochastic mass‑transfer move
    random_move_frac = 0.15    # max fraction of source mass to move
    uphill_prob = 0.001        # occasional uphill acceptance
    it = 0
    last_improve = time.time()

    # pre‑compute FFT size (must accommodate convolution of length 3n‑2)
    size_fft = 1 << ((3 * n - 2).bit_length())
    idx_rev = (2 * n - 2) - np.arange(n)   # used in gradient computation

    while time.time() < deadline:
        # ------------------------------------------------------
        # (a) self‑convolution and its maximum (via a single FFT)
        A_fft = np.fft.rfft(a, size_fft)
        conv_full = np.fft.irfft(A_fft * A_fft, size_fft)[: 2 * n - 1]
        max_conv = conv_full.max()
        cur_val = 2.0 * n * max_conv

        # (b) soft‑max weights
        w = np.exp(beta * (conv_full - max_conv))      # w_i ≤ 1, w_max = 1

        # (c) gradient of the smooth max
        # FFT of w and w_rev (real arrays)
        W_fft = np.fft.rfft(w, size_fft)
        conv_aw = np.fft.irfft(A_fft * W_fft, size_fft)[: 3 * n - 2]

        w_rev = w[::-1]
        W_rev_fft = np.fft.rfft(w_rev, size_fft)
        conv_arev = np.fft.irfft(A_fft * W_rev_fft, size_fft)[: 3 * n - 2]

        grad = 2.0 * (conv_aw[:n] + conv_arev[idx_rev])

        # (d) step size (sqrt decay) and momentum update
        eta = eta0 / np.sqrt(it + 1)
        # momentum (Nesterov‑like)
        if it == 0:
            mom = np.zeros_like(a)
        mom = mu * mom - eta * grad
        a_new = proj_simplex(a + mom)

        # (e) evaluate candidate (fast, exact for our normalisation)
        new_val = fast_score(a_new)

        # (f) accept improvement, otherwise occasional uphill
        if new_val < best_val:
            best_val = new_val
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
        else:
            if np.random.rand() < uphill_prob:
                a = a_new

        # ------------------------------------------------------
        # (g) random mass‑transfer move (helps escape plateaus)
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = proj_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < uphill_prob:
                    a = cand

        # ------------------------------------------------------
        # (h) anneal beta (make the smooth max sharper)
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # ------------------------------------------------------
        # (i) stagnation handling – re‑initialise from a noisy perturbation
        if time.time() - last_improve > 30.0:
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(0.9 * best_seq + 0.1 * noise)
            mom = np.zeros_like(a)
            last_improve = time.time()
            print(f"[re‑seed] iteration {it}, best {best_val:.6f}")

        # ------------------------------------------------------
        # (j) logging
        if it % 5000 == 0:
            print(f"[iter {it:7d}] cur {cur_val:.6f}  best {best_val:.6f}  beta {beta:.1f}")

        # ------------------------------------------------------
        # (k) early stop if target reached
        if best_val <= 1.503:
            print(f"[target] reached {best_val:.6f} at iteration {it}")
            break

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # Return plain Python list (checker will re‑normalise if needed)
    return best_seq.tolist()
