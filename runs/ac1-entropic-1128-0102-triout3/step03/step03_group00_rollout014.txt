def _run_mirror_descent_full(a0, n, deadline, start_time, step0=0.5, max_iter=10**7):

    a = a0.copy()

    best_a = a.copy()

    best_score = float('inf')

    for it in range(1, max_iter+1):

        if time.time() >= deadline:

            break

        conv = _conv_self(a)

        max_val = conv.max()

        # compute objective: 2 * n * max_val / sum_a**2 (sum_a = 1 as we normalise)

        # Since we keep sum=1 after each step, sum_a=1.

        cur_score = 2.0 * n * max_val

        if cur_score < best_score:

            best_score = cur_score

            best_a = a.copy()

            print(f"[{time.time() - start_time:6.2f}s] mirror improve: n={n}, iter={it}, score={cur_score:.6f}")

        # gradient:

        grad = _subgradient_full(a, conv, rel_tol=1e-12)

        # shift to avoid overflow

        min_grad = grad.min()

        # ensure non negative shift

        grad_shift = grad - min_grad

        eta = step0 / math.sqrt(it)

        # exponentiate

        # Use np.exp(-eta * grad_shift) possibly large exponent negative, safe.

        mult = np.exp(-eta * grad_shift)

        a = a * mult

        # renorm to sum 1

        s = a.sum()

        if s > 0:

            a = a / s

        else:

            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

def _run_adam_descent(a0, n, deadline, start_time, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=10**7):

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0

    while time<deadline and t<max_iter:
        t+=1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score=2*n*max_val # since sum=1

        if cur_score < best_score: update.

        grad = _subgradient_full(a, conv)
        # note grad is nonnegative.

        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)  # ensures nonnegative sum=1

"""
Propose a non‑negative step function (list of heights) that minimizes
`evaluate_sequence`.  The method consists of three phases:

1. **Exploration with many random restarts** – a small problem (n≈800)
   is solved repeatedly using a mirror‑descent (exponentiated‑gradient)
   update on the probability simplex.  The update uses a sub‑gradient
   of the max‑convolution that is averaged over an active set whose
   tolerance is gradually tightened.  The best candidate from this stage
   becomes the seed for the next phase.

2. **Fine‑tuning on a larger scale** – the seed is linearly up‑sampled
   to a longer length (n≈2400) and refined for a long time with the
   same mirror‑descent scheme (now with a tighter tolerance).  The
   objective `2·n·max(conv)` is monitored and the best point is kept.

3. **Local stochastic refinement** – a cheap simulated‑annealing walk
   (mass transfers between two positions) is performed on the remaining
   time budget to escape possible local minima.

All operations stay on the simplex (non‑negative entries summing to 1)
so the scale‑invariance of `evaluate_sequence` is respected.
The algorithm respects the 1000 s budget; if time runs out the best
sequence found so far is returned.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution (averaged over an active set)
# ------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Compute the (average) sub‑gradient of max_k conv_k.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if act.size > 0:
        g /= float(act.size)
    return g


# ------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the simplex
# ------------------------------------------------------------
def _run_mirror_descent(a0: np.ndarray, n: int,
                        deadline: float, start_time: float,
                        step0: float = 0.5,
                        rel_tol: float = 1e-6,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of exponentiated‑gradient descent.
    Returns the best point discovered and its (approximate) objective:
        score = 2·n·max(conv)   (sum(a) is kept equal to 1)
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max  # sum(a) == 1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] mirror improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv, rel_tol=rel_tol)

        # shift to avoid overflow/underflow in the exponent
        grad_shift = grad - grad.min()

        eta = step0 / math.sqrt(it)

        # exponentiated update (mirrors steps of natural gradient)
        a = a * np.exp(-eta * grad_shift)

        # renormalise to the simplex
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ------------------------------------------------------------
# Adam‑style projected gradient descent on the simplex
# ------------------------------------------------------------
def _run_adam_descent(a0: np.ndarray, n: int,
                      deadline: float, start_time: float,
                      lr: float = 0.05,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation with projection onto the simplex after each step.
    Returns best point and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv, rel_tol=1e-6)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated annealing local refinement
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Tiny “ε‑transfer” annealing (move a fraction of mass from i to j).
    Returns possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # transfer up to 10 % of the mass at i
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling (linear interpolation) to a larger length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny numerical drift may appear)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # convert to plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
