def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs) -> List[float]:
    """
    Adam optimisation of a smoothed max of the auto‑convolution.
    Starts from a half‑binomial distribution, runs a soft‑max
    smoothing parameter β that grows to a large value, and keeps
    the best sequence seen (evaluated with the true objective
    2·n·max(f*f)).  Returns the sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    # problem size
    n = 8000
    L = 2 * n - 1
    m = n + L - 1   # 3n-2
    Nfft = next_pow_two(m)
    # initial a
    a = half_binomial_series(n)  # sum 1
    # initial logits: log of a (or uniform if zero)
    eps = 1e-12
    z = np.log(a + eps)  # shape (n,)
    # Adam parameters
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m_t = np.zeros_like(z)
    v_t = np.zeros_like(z)
    t_iter = 0
    # smoothing parameter beta
    beta = 1.0
    beta_target = 5000.0
    beta_mul = 1.0015
    # best tracking
    best_score = np.inf
    best_a = a.copy()
    # timing
    deadline = time.time() + budget_s - 5.0
    # restart controls
    last_improve = 0
    restart_gap = 500_000  # iterations without improvement trigger
    # noise schedule
    noise_prob = 0.005
    noise_scale = 1e-2

    while time.time() < deadline:
        # compute FFT of a
        A = np.fft.rfft(a, n=Nfft)
        # compute linear convolution a*a
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = t_iter
            print(f"[iter {t_iter}] new best: {best_score:.6f}")

        # compute soft‑max weights
        # shift for stability
        w_exp = np.exp(beta * (conv - max_conv))
        w = w_exp / w_exp.sum()
        # gradient w.r.t a
        Wr = np.fft.rfft(w[::-1], n=Nfft)
        grad_a_full = np.fft.irfft(A * Wr, n=Nfft)
        start = L - 1
        grad_a = 2.0 * grad_a_full[start:start + n]

        # gradient w.r.t logits (softmax)
        # grad_z = a * (grad_a - np.dot(a, grad_a))
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        t_iter += 1
        m_t = beta1 * m_t + (1 - beta1) * grad_z
        v_t = beta2 * v_t + (1 - beta2) * (grad_z ** 2)
        m_hat = m_t / (1 - beta1 ** t_iter)
        v_hat = v_t / (1 - beta2 ** t_iter)
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # apply step to logits
        z += step

        # optional random perturbation
        if rng.random() < noise_prob:
            noise = rng.normal(scale=noise_scale, size=n)
            z += noise

        # recompute a from logits (softmax)
        # subtract max for numeric stability
        z_shift = z - np.max(z)
        a = np.exp(z_shift)
        a_sum = a.sum()
        a = a / a_sum

        # increase beta
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # restart if stuck
        if t_iter - last_improve > restart_gap:
            # re‑initialize logits from a Dirichlet
            a = rng.dirichlet(np.full(n, 0.7))
            a = a / a.sum()
            z = np.log(a + eps)
            m_t.fill(0.0)
            v_t.fill(0.0)
            last_improve = t_iter
            print(f"[iter {t_iter}] restart – score≈{2.0*n*np.max(np.fft.irfft(np.fft.rfft(a, n=Nfft)**2, n=Nfft)[:L]):.6f}")

    # final check using evaluate_sequence (clamping already neutral)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={t_iter}, best reported score={final_score:.6f}")
    return best_a.tolist()

import time
import numpy as np
from typing import List

def next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def half_binomial_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑½ (a_i = C(2i,i)/4^i) normalised to sum to 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs) -> List[float]:
    """
    Adam optimisation of a smoothed maximum of the auto‑convolution.
    The algorithm works on the probability simplex using a soft‑max
    smoothing parameter β that is increased gradually.  The best sequence
    (according to the true objective 2·n·max(f*f)) seen during the run is
    returned.
    """
    rng = np.random.default_rng(seed)

    # problem dimension – larger n gives more freedom at modest cost
    n = 8000
    L = 2 * n - 1
    m = n + L - 1            # = 3n‑2
    Nfft = next_pow_two(m)

    # start from a smooth half‑binomial distribution (sum = 1)
    a = half_binomial_series(n)

    # logits for a softmax – guarantees positivity and sum 1
    eps = 1e-12
    z = np.log(a + eps)

    # Adam hyper‑parameters
    lr = 0.12
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m_t = np.zeros_like(z)
    v_t = np.zeros_like(z)

    # smoothing parameter β
    beta = 1.0
    beta_target = 6000.0
    beta_mul = 1.0025

    # book‑keeping
    best_score = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    restart_gap = 300_000   # restart if no improvement for many iterations
    noise_prob = 0.004
    noise_scale = 1e-2

    deadline = time.time() + budget_s - 5.0

    while time.time() < deadline:
        # ----- convolution on the simplex -----
        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]                     # linear part
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ----- keep the best seen -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it}] new best score = {best_score:.6f}")

        # ----- soft‑max weights (smooth approximation of the max) -----
        w_exp = np.exp(beta * (conv - max_conv))
        w = w_exp / w_exp.sum()

        # ----- gradient w.r.t. a (2 * correlation(a, w)) -----
        Wr = np.fft.rfft(w[::-1], n=Nfft)
        corr_full = np.fft.irfft(A * Wr, n=Nfft)
        start = L - 1
        grad_a = 2.0 * corr_full[start:start + n]

        # ----- gradient w.r.t. logits (softmax Jacobian) -----
        dot = np.dot(a, grad_a)                 # scalar a·grad_a
        grad_z = a * (grad_a - dot)

        # ----- Adam update on logits -----
        it += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad_z
        v_t = beta2 * v_t + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m_t / (1.0 - beta1 ** it)
        v_hat = v_t / (1.0 - beta2 ** it)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z += step

        # ----- occasional random perturbation to escape flat regions -----
        if rng.random() < noise_prob:
            z += rng.normal(scale=noise_scale, size=n)

        # ----- map logits back to the simplex -----
        z_shift = z - np.max(z)                 # for numerical stability
        a = np.exp(z_shift)
        a_sum = a.sum()
        a = a / a_sum

        # ----- increase smoothing parameter -----
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # ----- restart if stuck for a long while -----
        if it - last_improve > restart_gap:
            a = rng.dirichlet(np.full(n, 0.7))
            a = a / a.sum()
            z = np.log(a + eps)
            m_t.fill(0.0)
            v_t.fill(0.0)
            last_improve = it

import time
from typing import List

import numpy as np


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^‑½  (a_i = C(2i,i)/4^i)
    normalised so that they sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs) -> List[float]:
    """
    Adam optimiser for a smoothed version of the max of the auto‑convolution.
    The search works on the probability simplex (∑a_i = 1, a_i ≥ 0) using a
    soft‑max smoothing parameter β that is increased gradually.  The best
    sequence (according to the true objective 2·n·max(f*f)) encountered during
    the run is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimension – a larger n gives more freedom at modest cost.
    n = 8000                     # length of the step‑function
    L = 2 * n - 1                # length of the linear auto‑convolution
    m = n + L - 1                # length needed for correlations (3n‑2)
    Nfft = next_pow_two(m)       # FFT size (power of two for speed)

    # ------------------------------------------------------------------
    # Initialise from a smooth half‑binomial distribution (already sums to 1)
    a = half_binomial_series(n)

    # Logits for a softmax – guarantees positivity and simplex constraint
    eps = 1e-12
    z = np.log(a + eps)

    # Adam hyper‑parameters
    lr = 0.12                    # base learning rate for logits
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m_t = np.zeros_like(z)
    v_t = np.zeros_like(z)

    # Smoothing (soft‑max) parameter β
    beta = 1.0
    beta_target = 6000.0
    beta_mul = 1.0025

    # Book‑keeping
    best_score = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    restart_gap = 300_000       # restart if no improvement for many iterations
    noise_prob = 0.004
    noise_scale = 1e-2

    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # ----- 1) linear auto‑convolution of the current a -----
        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)          # circular convolution
        conv = conv_full[:L]                             # keep linear part only
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ----- 2) keep the best solution seen so far -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it}] new best score = {best_score:.6f}")

        # ----- 3) soft‑max weights (smooth approximation of the max) -----
        #   w_k = exp(β (conv_k − max_conv)) / Σ exp(·)
        w_exp = np.exp(beta * (conv - max_conv))
        w = w_exp / w_exp.sum()

        # ----- 4) gradient w.r.t. a: 2 * correlation(a, w) -----
        Wr = np.fft.rfft(w[::-1], n=Nfft)                # FFT of reversed weights
        corr_full = np.fft.irfft(A * Wr, n=Nfft)        # circular correlation
        start = L - 1
        grad_a = 2.0 * corr_full[start:start + n]

        # ----- 5) gradient w.r.t. logits (soft‑max Jacobian) -----
        dot = np.dot(a, grad_a)                          # a·grad_a
        grad_z = a * (grad_a - dot)

        # ----- 6) Adam update on the logits -----
        it += 1
        m_t = beta1 * m_t + (1.0 - beta1) * grad_z
        v_t = beta2 * v_t + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m_t / (1.0 - beta1 ** it)
        v_hat = v_t / (1.0 - beta2 ** it)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z += step

        # ----- 7) occasional random perturbation (helps escape plateaus) -----
        if rng.random() < noise_prob:
            z += rng.normal(scale=noise_scale, size=n)

        # ----- 8) map logits back to the simplex (soft‑max) -----
        z_shift = z - np.max(z)          # numerical‑stability shift
        a = np.exp(z_shift)
        a = a / a.sum()

        # ----- 9) increase the smoothing parameter -----
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # ----- 10) restart if we have been stuck for a long time -----
        if it - last_improve > restart_gap:
            a = rng.dirichlet(np.full(n, 0.7))
            a = a / a.sum()
            z = np.log(a + eps)
            m_t.fill(0.0)
            v_t.fill(0.0)
            last_improve = it
            # quick diagnostic for the fresh start
            A_tmp = np.fft.rfft(a, n=Nfft)
            conv_tmp = np.fft.irfft(A_tmp * A_tmp, n=Nfft)[:L]
            print(f"[iter {it}] restart – initial score≈{2.0 * n * conv_tmp.max():.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided routine (clamps, safety checks, etc.)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
