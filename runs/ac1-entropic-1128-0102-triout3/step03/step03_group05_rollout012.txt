"""
Propose a low‑value step‑function sequence.

Algorithm
---------
1. Generate a diverse pool of analytical candidates (power‑law, exponential,
   symmetric exponential, Gaussian, cosine, linear, uniform and the previous
   best sequence if available).  All candidates are normalised to sum = 1.
2. Pick the best candidate according to a fast implementation of
   `evaluate_sequence` (FFT based self‑convolution).
3. Starting from this candidate run a long projected sub‑gradient descent
   on the simplex (∑a_i = 1, a_i≥0) using the following tricks:
   * The sub‑gradient is the sum of contributions from the indices of the
     convolution that are within a shrinking relative gap of the current
     maximum.
   * Adam‑style moment estimates are used to stabilise the noisy sub‑gradient.
   * A simple back‑tracking line‑search (at most two FFTs per iteration)
     accepts only moves that improve the objective; occasional uphill
     moves (probability ≈ 0.001) help to escape plateaus.
   * Random “mass‑transfer’’ moves are interleaved (≈1 % of iterations) with
     Metropolis acceptance.
   * If no improvement is seen for 30 s the iterate is re‑initialised by a
     small random perturbation of the current best solution.
4. The loop runs until the allocated time budget (≈1000 s) is exhausted.
5. The best sequence found is returned as a plain Python list of floats.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (no closures, top‑level as required)

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` (no safety checks).
    Assumes a is a non‑negative 1‑D array; sum(a) should be > 0.
    Returns the same value as `evaluate_sequence`.
    """
    n = a.size
    total = a.sum()
    if total < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(a, size)
    conv = np.fft.irfft(ft * ft, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (total * total))


def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def build_exponential(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def build_exponential_sym(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential (both ends heavy), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a /= a.sum()
    return a


def build_gaussian(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian, normalised."""
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def build_cosine(n: int) -> np.ndarray:
    """Cosine‑shaped bump shifted to be non‑negative, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2.0) / n) + 1.0
    a[a < 0] = 0.0
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n-i), normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


def gradient_multi_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution w.r.t. a.
    For each active index k we add 2·a_{k-i} to grad_i whenever the pair
    (i, k‑i) is inside the array bounds.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i = np.arange(low, high + 1)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


def get_active_set(conv: np.ndarray, max_conv: float,
                   delta: float, K: int) -> np.ndarray:
    """Return up to K indices whose convolution value is within (1‑δ)·max."""
    mask = conv >= max_conv * (1.0 - delta)
    idxs = np.flatnonzero(mask)
    if idxs.size > K:
        # keep the K largest values among the candidates
        topk = idxs[np.argpartition(-conv[idxs], K - 1)[:K]]
        return topk
    return idxs


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation value.  Runs for at most `budget_s` seconds and returns the
    best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    n = 3000

    # ------------------------------------------------------------------
    # 2. Build a pool of analytically motivated initial sequences
    candidates = []

    # power‑law family
    for p in np.linspace(0.05, 0.90, 18):
        for off in [0.0, 0.2, 0.5, 1.0]:
            candidates.append(build_power_law(n, p, off))

    # pure exponential (decreasing)
    for lam in np.logspace(-4, -1.5, 12):
        candidates.append(build_exponential(n, lam))

    # symmetric exponential (both ends)
    for lam in np.logspace(-4, -1.5, 10):
        candidates.append(build_exponential_sym(n, lam))

    # centred Gaussian
    for sigma in np.linspace(0.5 * n / 20, 0.5 * n, 8):
        candidates.append(build_gaussian(n, sigma))

    # cosine bump
    candidates.append(build_cosine(n))

    # linear descending
    candidates.append(build_linear_descending(n))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # previous best (if available)
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size < n:
                tmp = np.zeros(n, dtype=np.float64)
                tmp[:prev.size] = prev
                prev = tmp
            else:
                prev = prev[:n].copy()
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s > 0:
                prev /= s
                candidates.append(prev)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate (according to fast_score)
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best initial value: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    a = best_seq.copy()                  # current iterate (on simplex)
    best_seq = a.copy()
    best_val = fast_score(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    eta0 = 1.0          # base learning rate
    K_active = 30       # maximal number of active indices
    delta_start = 0.06
    delta_end = 0.001
    random_move_prob = 0.015      # frequency of random mass‑transfer moves
    random_move_frac = 0.20       # max fraction of source mass to move
    uphill_prob = 0.001           # occasional uphill acceptance
    last_improve = time.time()
    it = 0

    # pre‑compute FFT size for this n
    conv_size = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # --------------------------------------------------------------
        # (a) self‑convolution and its maximum (current state)
        ft = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(ft * ft, conv_size)[: 2 * n - 1]
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv / (a.sum() ** 2)

        # (b) active set for the sub‑gradient
        delta = max(delta_end,
                    delta_start * (1.0 - it / 5_000_000.0))   # linear anneal
        active_idxs = get_active_set(conv, max_conv, delta, K_active)

        # (c) sub‑gradient of the max‑convolution
        grad = gradient_multi_active(a, active_idxs)

        # (d) Adam moment updates
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # (e) step size (sqrt decay) and actual step
        eta = eta0 / np.sqrt(it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # (f) projected gradient move (descent direction)
        a_candidate = proj_simplex(a - step)

        # evaluate candidate
        ft_c = np.fft.rfft(a_candidate, conv_size)
        conv_c = np.fft.irfft(ft_c * ft_c, conv_size)[: 2 * n - 1]
        max_c = conv_c.max()
        cand_val = 2.0 * n * max_c / (a_candidate.sum() ** 2)

        # (g) accept if we improve, else maybe accept uphill
        if cand_val < cur_val:
            a = a_candidate
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_candidate.copy()
                last_improve = time.time()
        else:
            if np.random.rand() < uphill_prob:
                a = a_candidate
                cur_val = cand_val

        # --------------------------------------------------------------
        # (h) random “mass‑transfer’’ hill‑climbing move
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta_transfer = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta_transfer
                cand[j] += delta_transfer
                # stay on simplex (sum unchanged) – just project to restore exactness
                cand = proj_simplex(cand)
                ft_r = np.fft.rfft(cand, conv_size)
                conv_r = np.fft.irfft(ft_r * ft_r, conv_size)[: 2 * n - 1]
                val_r = 2.0 * n * conv_r.max() / (cand.sum() ** 2)
                if val_r < best_val:
                    best_val = val_r
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < uphill_prob:
                    a = cand

        # --------------------------------------------------------------
        # (i) stagnation handling – re‑seed from a slight perturbation of best
        if time.time() - last_improve > 30.0:
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(0.94 * best_seq + 0.06 * noise)
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            last_improve = time.time()
            print("[restart] re‑initialised from best after stagnation")

        # --------------------------------------------------------------
        # (j) periodic logging
        if it % 8000 == 0:
            now = time.time()
            elapsed = now - start_time
            print(f"[iter {it:8d}] cur {cur_val:.6f}  best {best_val:.6f}  "
                  f"time {elapsed:.1f}s")

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # The external checker will normalise; we simply return the raw sequence.
    return best_seq.tolist()
