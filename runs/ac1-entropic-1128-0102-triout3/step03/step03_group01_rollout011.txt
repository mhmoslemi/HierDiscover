    a = a0.copy()
    best_a = a0.copy()
    best_score = float('inf')
    it = 0

        it += 1
        conv = _conv_self(a)
        max_b = conv.max()

"""
propose_candidate
-----------------
A multi‑stage optimizer for the “max‑self‑convolution” problem.

Strategy
~~~~~~~~
1. **Warm start** – if a previous solution (``height_sequence_1``) is
   available it is normalised to a probability vector and used as the
   initial distribution.
2. **Multi‑scale schedule** – we optimise on a sequence of lengths
   (400 … 9600).  For each length the current best distribution is
   up‑sampled (linear interpolation) to obtain a warm start.
3. **Phase A: smoothed ℓ∞ surrogate descent** – the same log‑sum‑exp
   gradient descent that was used before, but with a slightly more
   aggressive step‑size schedule.
4. **Phase B: exact sub‑gradient / exponentiated‑gradient descent** –
   after the smooth phase we focus on the true max‑convolution entries
   (λ→0).  The update is multiplicative (mirror descent on the simplex)
   which often converges faster to a flatter autocorrelation.
5. **Phase C: LP‑based “mass‑filling’’ step (small sizes only)** –
   for n ≤ 2000 we solve a linear program that maximises the total
   mass of a new distribution **g** under the constraint
   ``f * g ≤ M`` (where ``M`` is the current maximum convolution of the
   current distribution ``f``).  Normalising ``g`` and mixing it with the
   current solution yields a provable non‑increase of the objective.
6. **Phase D: cheap simulated‑annealing polishing** – stochastic mass
   transfers between two coordinates, accepting worsening moves with a
   slowly decreasing temperature.
7. **Best‑of‑all** – the best sequence seen throughout the run is
   returned.

All operations stay on the probability simplex (∑aᵢ = 1) so the
objective ``2·n·max(conv)`` is invariant to scaling; the final list is
converted back to plain Python floats.

The implementation respects the 1000 s budget, checks the deadline
frequently and prints progress messages.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors x and y via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n; re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate.
    """
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)                # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]              # 2·∂conv/∂a weighted sum
    return grad


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent with momentum on the simplex.
    Returns the best vector and its exact (2·n·max(conv)) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.95                     # a tad more momentum than before
    step0 = 0.9                     # slightly larger initial step
    it = 0

    # λ‑schedule: start relatively large and halve regularly
    lam_factor = 0.45
    lam_decay = 0.5
    lam_decay_interval = 2_000

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, it={it}, score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-10)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (exact ℓ∞ sub‑gradient)
# ----------------------------------------------------------------------


def _run_exp_descent(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     max_iter: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a sub‑gradient that only sees the
    entries attaining the current maximum.  The update is multiplicative:
        a_i ← a_i·exp(−η·g_i)
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    eta0 = 0.5                     # initial learning rate

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp improve: n={n}, it={it}, score={cur_score:.6f}")

        # mask of entries that are (almost) maximal
        eps = 1e-12
        mask = (conv >= max_b - eps).astype(float)

        # sub‑gradient = 2·(mask * a_rev) (slice of length n)
        grad = 2.0 * _conv_fft(mask, a[::-1])[n - 1:n - 1 + n]

        # clip to avoid overflow in the exponential
        grad = np.clip(grad, -50.0, 50.0)

        eta = eta0 / math.sqrt(it)   # diminishing step
        a = a * np.exp(-eta * grad)

        # project back onto the simplex by simple renormalisation
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming based mass‑filling step (used only for modest n)
# ----------------------------------------------------------------------


def _run_lp_improve(f: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    max_lp_iters: int = 6) -> tuple[np.ndarray, float]:
    """
    Solve the LP:
        maximise   sum g_j
        subject to f * g ≤ M   (elementwise)
                    g ≥ 0
    and return the best normalised g found (or the original f if no
    improvement).  The LP is solved with scipy.optimize.linprog.
    """
    best_f = f.copy()
    best_score = 2.0 * n * _conv_self(best_f).max()
    cur_f = f.copy()

    for lp_it in range(max_lp_iters):
        if time.time() >= deadline:
            break

        conv_f = _conv_self(cur_f)
        M = conv_f.max()

        # Build the (2n‑1)×n constraint matrix A where
        #   (A g)_k = sum_i cur_f[i]·g_{k‑i}
        rows = 2 * n - 1
        cols = n
        A = np.zeros((rows, cols), dtype=np.float64)
        # each column j is cur_f shifted by j positions
        for j in range(cols):
            A[j:j + n, j] = cur_f

        # Linear programme: maximise sum(g)  → minimise -sum(g)
        c = -np.ones(cols, dtype=np.float64)
        bounds = [(0, None)] * cols
        res = linprog(c, A_ub=A, b_ub=np.full(rows, M),
                      bounds=bounds, method='highs', options={'presolve': True})

        if not res.success:
            # LP failure – abort further LP attempts
            print(f"[{time.time() - start_time:6.2f}s] LP failed (iter {lp_it}): {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        g_norm = g / sum_g

        new_score = 2.0 * n * _conv_self(g_norm).max()
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_f = g_norm.copy()
            cur_f = g_norm.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve: n={n}, iter={lp_it}, score={new_score:.6f}")
        else:
            # No improvement – stop early
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing: move a small ε from i to j.
    Returns possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function that minimises `evaluate_sequence`.
    The algorithm runs for at most ``budget_s`` seconds, combining:
      * multi‑scale smoothed gradient descent,
      * exact sub‑gradient exponentiated‑gradient descent,
      * (for modest sizes) a linear‑programming mass‑filling step,
      * a final simulated‑annealing polishing.
    The best distribution found is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Use a previous solution if the runner supplied one
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – start small, finish with the largest size we can fit.
    lengths = [400, 800, 1600, 3200, 4800, 6400, 9600]
    # Randomise order only to avoid pathological early termination.
    np.random.shuffle(lengths)

    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise a0 for this length
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # Allocate a time slice for this length (roughly a fraction of the
        # remaining budget, but never less than a few seconds)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_deadline = time.time() + max(8.0, 0.18 * remaining)

        # ------------------------------------------------------------------
        # Phase A – smoothed gradient descent
        # ------------------------------------------------------------------
        a_smooth, score_smooth = _run_smooth_descent(
            a0, n, slice_deadline, start_time)

        if score_smooth < best_score:
            best_score = score_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after smooth: n={n}, score={score_smooth:.6f}")

        # ------------------------------------------------------------------
        # Phase B – exact sub‑gradient / exponentiated‑gradient descent
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 6.0:
            exp_deadline = time.time() + max(5.0, 0.12 * remaining)
            a_exp, score_exp = _run_exp_descent(
                a_smooth, n, exp_deadline, start_time)
            if score_exp < best_score:
                best_score = score_exp
                best_seq = a_exp.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after exp: n={n}, score={score_exp:.6f}")
        else:
            a_exp = a_smooth

        # ------------------------------------------------------------------
        # Phase C – LP‑based improvement (only for modest sizes)
        # ------------------------------------------------------------------
        if n <= 2000:
            remaining = deadline - time.time()
            if remaining > 4.0:
                lp_deadline = time.time() + max(3.0, 0.07 * remaining)
                a_lp, score_lp = _run_lp_improve(
                    a_exp, n, lp_deadline, start_time)
                if score_lp < best_score:
                    best_score = score_lp
                    best_seq = a_lp.copy()
                    print(f"[{time.time() - start_time:6.2f}s] global improve after LP: n={n}, score={score_lp:.6f}")
                a_exp = a_lp   # continue from the LP‑improved point

        # ------------------------------------------------------------------
        # Phase D – cheap simulated‑annealing polishing
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 3.0:
            anneal_deadline = time.time() + max(2.5, 0.04 * remaining)
            a_ref, score_ref = _run_annealing(
                a_exp,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=30_000,
            )
            if score_ref < best_score:
                best_score = score_ref

                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n}, score={score_ref:.6f}")

        # ------------------------------------------------------------------
        # Remember the best distribution for the next upscale
        # ------------------------------------------------------------------
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Final long‑run refinement on the best size we have so far
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0 and best_seq is not None:

            best_seq, best_seq.size, final_deadline, start_time)
        if score_final < best_score:

    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list of floats
    return [float(x) for x in best_seq]
