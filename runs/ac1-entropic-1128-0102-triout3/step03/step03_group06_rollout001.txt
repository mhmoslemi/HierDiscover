"""
Annealing + subgradient hybrid optimizer for the step‑function evaluation.
- Starts from a smooth decreasing distribution (1/(i+1)) that already
  yields a modest score.
- Performs a temperature‑controlled simulated‑annealing on the simplex:
  repeatedly transfers a random amount of mass between two positions,
  accepting worsening moves with probability exp(-Δ/T).  The temperature
  decays linearly with elapsed time.
- The best vector found is handed to a short projected sub‑gradient
  descent (the same routine used previously) to polish the solution.
- The whole search runs within the given time budget (≈1000 s) and
  returns the best sequence discovered.
"""

import numpy as np
import time
import math
import random

def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the score (identical to evaluate_sequence for a
    normalized non‑negative vector)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent on the simplex (same as the earlier
    fallback implementation but with slightly tweaked hyper‑parameters)."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    eta0 = 0.30          # initial step size
    eta_decay = 2e-5     # slower decay gives more steps
    random_move_prob = 0.03
    random_move_step = 0.2
    it = 0
    deadline = time.time() + time_limit - 0.1

    while time.time() < deadline:
        # self‑convolution, locate active index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient for the active index (ordered pairs)
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass‑transfer move (helps escape plateaus)
        if np.random.rand() < random_move_prob:
            i_rand, j_rand = np.random.randint(0, n, size=2)
            if i_rand != j_rand and a[i_rand] > 0.0:
                delta = random_move_step * a[i_rand] * np.random.rand()
                cand = a.copy()
                cand[i_rand] -= delta
                cand[j_rand] += delta
                # re‑project
                u2 = np.sort(cand)[::-1]
                cssv2 = np.cumsum(u2) - 1.0
                cond2 = u2 - cssv2 / ind > 0
                if np.any(cond2):
                    rho2 = ind[cond2][-1]
                    theta2 = cssv2[cond2][-1] / rho2
                else:
                    theta2 = 0.0
                cand = np.maximum(cand - theta2, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand                     # keep the improvement
                elif np.random.rand() < 0.001:
                    a = cand                     # occasional uphill move

        # bookkeeping
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Hybrid search:
    1) Simulated annealing on the simplex (mass transfers) for the bulk of time.
    2) A short projected sub‑gradient polish on the best vector found.
    Returns a list of floats (the step‑function heights).
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    n = 1500                     # length of the step function
    target_score = 1.503         # early‑stop target (optional)

    # ------------------------------------------------------------------
    # 0. Simple, smooth initial guess – a decreasing harmonic shape.
    a = 1.0 / np.arange(1, n + 1, dtype=np.float64)
    a /= a.sum()                 # normalize to sum = 1

    best_a = a.copy()
    best_val = fast_score(best_a)
    cur_a = a.copy()
    cur_val = best_val

    print(f"[init] initial score {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Simulated annealing – mass transfers between two positions.
    #    Temperature decays linearly with elapsed time.
    it = 0
    while time.time() < deadline:
        # pick a source with non‑zero mass
        i = np.random.randint(0, n)
        if cur_a[i] == 0.0:
            continue
        # pick a distinct destination
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a random fraction of the source mass
        max_delta = cur_a[i]
        # the fraction shrinks as the temperature drops – makes moves finer later
        elapsed = time.time() - start
        frac = 0.6 * (1.0 - elapsed / (deadline - start)) + 0.02   # between ~0.02 and 0.62
        delta = max_delta * np.random.rand() * frac

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = fast_score(cand)

        # acceptance test
        if cand_val < cur_val:
            cur_a, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_a = cand_val, cand.copy()
                print(f"[anneal iter {it:6d}] new best {best_val:.6f}")
        else:
            # linear temperature schedule
            t_frac = elapsed / (deadline - start)
            T = max(0.01 * (1.0 - t_frac), 1e-6)   # small but positive
            # Metropolis acceptance
            if np.random.rand() < math.exp(-(cand_val - cur_val) / T):
                cur_a, cur_val = cand, cand_val

        # occasional restart from the current best with a random perturbation
        if it % 50000 == 0 and it > 0:
            # blend the best vector with a small random Dirichlet noise
            noise = np.random.dirichlet(np.ones(n) * 0.1)
            cur_a = 0.9 * best_a + 0.1 * noise
            cur_a /= cur_a.sum()
            cur_val = fast_score(cur_a)

        # early exit if we have already beaten the target comfortably
        if best_val <= target_score:
            print(f"[early stop] target score achieved at iter {it}")
            break

        it += 1

    print(f"[anneal] finished after {it} iterations, best score {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Polish the obtained vector with a short sub‑gradient descent.
    time_left = deadline - time.time()
    if time_left > 5.0:        # leave some slack for final conversion
        print(f"[polish] running sub‑gradient for {time_left:.1f}s")
        polished = subgradient_optimize(best_a, time_left)
        polished_val = fast_score(polished)
        print(f"[polish] polished score {polished_val:.6f}")
        if polished_val < best_val:
            best_val, best_a = polished_val, polished.copy()

    # ------------------------------------------------------------------
    # 3. Final sanity – ensure we return a plain Python list of floats.
    #    The evaluation routine will clamp values to [0,1000] and enforce
    #    a positive total, both of which are satisfied by construction.
    print(f"[final] returning best score {best_val:.6f}")
    return best_a.tolist()

"""
Hybrid optimizer for the step‑function score.

1. **Initialisation** – a smooth decreasing harmonic distribution that already
   yields a modest score (≈2).

2. **Simulated annealing** – repeatedly transfer a random amount of mass between
   two positions while staying on the probability simplex.  Moves that increase
   the score are accepted with Metropolis probability `exp(-Δ/T)` where the
   temperature `T` decays linearly with elapsed time.  The best vector seen
   throughout the run is recorded.  Periodically we re‑inject a small random
   perturbation of the current best to avoid stagnation.

3. **Polishing** – the best vector is handed to a short projected sub‑gradient
   descent (the same routine used as a fallback previously) to squeeze a bit
   more improvement.

The whole search respects the 1000 s budget and returns the best sequence
found as a plain list of floats.
"""

import numpy as np
import time
import math
import random


def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (identical to the provided
    `evaluate_sequence` for a non‑negative, normalised vector)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent on the simplex.

    The routine follows the classic approach used previously but with a few
    tuned hyper‑parameters (slower learning‑rate decay and a slightly higher
    random‑move probability) to act as a fine‑grained polish after the
    annealing phase.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    eta0 = 0.30          # initial step size
    eta_decay = 2e-5     # slower decay → more iterations
    random_move_prob = 0.03
    random_move_step = 0.2
    it = 0
    deadline = time.time() + time_limit - 0.1

    while time.time() < deadline:
        # compute self‑convolution and the index attaining the maximum
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active term
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass‑transfer move (helps escape plateaus)
        if np.random.rand() < random_move_prob:
            i_rand, j_rand = np.random.randint(0, n, size=2)
            if i_rand != j_rand and a[i_rand] > 0.0:
                delta = random_move_step * a[i_rand] * np.random.rand()
                cand = a.copy()
                cand[i_rand] -= delta
                cand[j_rand] += delta
                # re‑project
                u2 = np.sort(cand)[::-1]
                cssv2 = np.cumsum(u2) - 1.0
                cond2 = u2 - cssv2 / ind > 0
                if np.any(cond2):
                    rho2 = ind[cond2][-1]
                    theta2 = cssv2[cond2][-1] / rho2
                else:
                    theta2 = 0.0
                cand = np.maximum(cand - theta2, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Hybrid search for a low‑score step function.

    Returns a list of non‑negative floats (the heights) whose score is
    as low as possible within the allotted time.  The algorithm performs
    a long simulated‑annealing phase followed by a brief sub‑gradient
    polishing step.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for the caller

    n = 1500
    target_score = 1.503   # optional early‑stop threshold

    # ------------------------------------------------------------------
    # 0. Initial guess – a decreasing harmonic distribution.
    a = 1.0 / np.arange(1, n + 1, dtype=np.float64)
    a /= a.sum()                           # normalise to sum = 1

    best_a = a.copy()
    best_val = fast_score(best_a)
    cur_a = a.copy()
    cur_val = best_val

    print(f"[init] initial score {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Simulated annealing on the simplex.
    it = 0
    while time.time() < deadline:
        # pick a source index with positive mass
        i = np.random.randint(0, n)
        if cur_a[i] == 0.0:
            continue

        # pick a distinct destination
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # amount to transfer – fraction shrinks as temperature falls
        max_delta = cur_a[i]
        elapsed = time.time() - start
        total = deadline - start
        frac = 0.6 * (1.0 - elapsed / total) + 0.02   # ≈0.62 → 0.02
        delta = max_delta * np.random.rand() * frac

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_val = fast_score(cand)

        # acceptance test (Metropolis)
        if cand_val < cur_val:
            cur_a, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_a = cand_val, cand.copy()
                print(f"[anneal iter {it:6d}] new best {best_val:.6f}")
        else:
            # linear temperature schedule (max 0.01, linearly to almost 0)
            T = max(0.01 * (1.0 - elapsed / total), 1e-6)
            if np.random.rand() < math.exp(-(cand_val - cur_val) / T):
                cur_a, cur_val = cand, cand_val

        # every ~50 k iterations inject a small random perturbation of the best seen
        if it % 50000 == 0 and it > 0:
            noise = np.random.dirichlet(np.ones(n) * 0.1)
            cur_a = 0.9 * best_a + 0.1 * noise
            cur_a /= cur_a.sum()
            cur_val = fast_score(cur_a)

        # early stop if we have already beaten the target
        if best_val <= target_score:
            print(f"[early stop] target achieved at iteration {it}")
            break

        it += 1

    print(f"[anneal] finished after {it} iterations, best score {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Polishing with a short sub‑gradient descent (uses remaining time).
    time_left = deadline - time.time()
    if time_left > 5.0:
        print(f"[polish] running sub‑gradient for {time_left:.1f}s")
        polished = subgradient_optimize(best_a, time_left)
        polished_val = fast_score(polished)
        print(f"[polish] polished score {polished_val:.6f}")
        if polished_val < best_val:
            best_val, best_a = polished_val, polished.copy()

    # ------------------------------------------------------------------
    # 3. Return the best sequence as a plain Python list.
    print(f"[final] returning best score {best_val:.6f}")
    return best_a.tolist()
