def gradient_softmax(a, conv, beta):
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # compute conv of a and reversed w
    # with FFT
    n = a.shape[0]
    m = conv.shape[0]
    w_rev = w[::-1]
    L = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)
    conv2 = np.fft.irfft(A * W, n=L)
    # grad slice
    grad = 2.0 * conv2[(m - 1) - np.arange(n)]
    return grad
```

def adam_update(a, grad, m, v, t, alpha, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = alpha * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    return a_new, m, v
```

    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()

        a[:] = 1.0 / n
    else:
        a /= total
    return a

    i = np.arange(n, dtype=np.float64)

    else:

    conv = np.fft.irfft(X * X, n=m)
    return conv

    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    ind = np.arange(1, n + 1)

        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=off)
            conv = conv_fft(a)
            cur = 2.0 * n * conv.max()          # sum(a)=1 ⇒ denominator = 1
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                print(f"[grid] new best p={p:.3f} off={off:.2f} score={cur:.6f}")

    # exponential grid
    lambdas = np.arange(0.0005, 0.0105, 0.0010)   # ~10 values
    for lam in lambdas:
        if time.time() > deadline:
            break
        a = exponential_sequence(n, lam)
        conv = conv_fft(a)
        cur = 2.0 * n * conv.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[exp] new best λ={lam:.4f} score={cur:.6f}")

    # fallback (should never be needed)
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * conv_fft(best_a).max()
        print("[fallback] using uniform distribution")

    global_best_a = best_a.copy()
    global_best_score = best_score
    print(f"[phase 1] best initial score = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # phase 2 – Adam‑style projected gradient descent on a soft‑max proxy
    max_restarts = 4
    beta_start = 1.0
    beta_max = 30.0
    beta_inc_per_iter = (beta_max - beta_start) / 200_000.0   # ≈1e‑4
    alpha = 0.15                # base learning rate for Adam
    eps = 1e-12
    eval_every = 5000           # how often to call evaluate_sequence

    print("[phase 2] Adam‑style projected gradient descent (soft‑max)")

    for restart in range(max_restarts):
        if time.time() > deadline:
            break

        # initialise
        if restart == 0:
            a = global_best_a.copy()
        else:
            a = rng.gamma(0.7, 1.0, size=n)
            a /= a.sum()

        m = np.zeros_like(a)       # first moment
        v = np.zeros_like(a)       # second moment
        beta = beta_start
        it = 0

        while time.time() < deadline:
            conv = conv_fft(a)

            # periodic true evaluation
            if it % eval_every == 0:
                cur_score = evaluate_sequence(a.tolist())
                if cur_score < global_best_score:
                    global_best_score = cur_score
                    global_best_a = a.copy()
                    print(f"[restart {restart} iter {it}] new global best = {global_best_score:.6f}")

            # compute smooth gradient
            grad = gradient_softmax(a, conv, beta)

            # Adam update (bias correction omitted for speed)
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            step = alpha * m / (np.sqrt(v) + eps)

            # take step and project back onto the simplex
            a = a - step
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a)

            # tiny random jitter to avoid stagnation
            if rng.random() < 0.001:
                a += (rng.random(n) - 0.5) * 1e-5
                a = project_onto_simplex(a)

            # increase soft‑max temperature gradually
            if beta < beta_max:
                beta = min(beta_max, beta + beta_inc_per_iter)

            it += 1

        approx_score = 2.0 * n * conv_fft(a).max()
        print(f"[restart {restart}] finished – approx score = {approx_score:.6f}")

    # ------------------------------------------------------------------
    # phase 3 – top‑k refinement with exact sub‑gradients
    top_k = 5
    beta_top = 8.0               # moderate sharpening for the top‑k blend
    print("[phase 3] top‑k refinement (exact sub‑gradients)")

    a = global_best_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline and it < 200_000:
        conv = conv_fft(a)

        if it % eval_every == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < global_best_score:
                global_best_score = cur_score
                global_best_a = a.copy()
                print(f"[top‑k] iter {it} new global best = {global_best_score:.6f}")

        # top‑k indices and soft‑max weights
        idxs = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idxs]
        maxv = conv.max()
        w = np.exp(beta_top * (vals - maxv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # combine exact gradients of the selected entries
        grad = np.zeros_like(a)
        for idx, weight in zip(idxs, w):
            grad += weight * gradient_of_peak(a, int(idx))

        # Adam‑style update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        step = alpha * m / (np.sqrt(v) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a)

        it += 1

    # ------------------------------------------------------------------
    # final sanity check & return
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")
    return global_best_a.tolist()
