def lp_optimize(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # Build convolution matrix
    # Use SciPy's toeplitz or manual loops
    # conv constraint length = 2n-1
    constraints = [C_f @ g <= M * np.ones(2*n-1)]
    # No sum constraint; we want maximize sum(g)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)

def refine_adam(a, max_iters, deadline, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_of_seq(a)
    n = len(a)
    fft_len_conv = 1 << ((2*n-1).bit_length())
    fft_len_grad = 1 << ((3*n-2).bit_length())
    a_rev = a[::-1]
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # conv and max
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute softmax weights
        # cut off top values: use tau = max(1e-14, t_factor * max_val)
        tau = max(1e-14, 0.05*(1.0/(it**0.5))*max_val)   # adapt gradually; maybe decays
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        # gradient via FFT
        # conv of w with reversed a
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[:len(w)+n-1]
        g = 2.0 * conv2[n-1: n-1 + n]   # g vector length n
        # Adam updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        # step size maybe lr / sqrt(it)
        step = lr / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score
```

def _adam_refine(a, max_iters, deadline, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8, temp_start=0.1, temp_end=0.001, use_softmax=True):
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_of_seq(a)
    # precompute length for FFT for conv and grad. For gradient using softmax we need conv2 length (3n-2). If use argmax gradient we can compute directly.
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    if use_softmax:
        fft_len_grad = 1 << ((3 * n - 2).bit_length())
        a_rev_full = a[::-1]  # will update each iteration.
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # conv
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        # best update
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute temperature: linear schedule from start to end
        tau = max(1e-14, (temp_start + (temp_end - temp_start) * it / max_iters) * max_val)
        if use_softmax:
            # compute w
            w = np.exp((conv - max_val) / tau)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.full_like(w, 1.0 / w.shape[0])
            # gradient conv via FFT
            W = np.fft.rfft(w, n=fft_len_grad)
            # update a_rev each iter
            a_rev = a[::-1]  # compute each iteration
            A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
            conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[:len(w)+n-1]
            g = 2.0 * conv2[n-1:n-1+n]
        else:
            # argmax gradient
            arg = np.argmax(conv)
            i_start = max(0, arg - n + 1)
            i_end = min(n, arg + 1)
            g = np.zeros_like(a)
            # indices
            i = np.arange(i_start, i_end)
            g[i] = 2.0 * a[arg - i]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        # ensure nonneg + simplex
        a = _project_simplex(a)
    return best_a, best_score

* Adam‑style adaptive moments replace the simple Nesterov momentum.
* The temperature for the soft‑max approximation of the “max” is
  decreased linearly from a relatively large value to a tiny one, and
  for the final fine‑tune the pure sub‑gradient (arg‑max) is used.
* Multiple diverse seeds (uniform, triangular, Hann, previous solution,
  Dirichlet) are refined on a modest size, then the current best shape
  is progressively up‑sampled (≈1.3×) up to a maximum length.
* The iteration budget for each resolution is scaled with the length so
  that the overall time stays well under the 1000 s budget.
* After every refinement the best‑seen sequence and its score are kept;
  the final result is the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Score for a sequence that already sums to 1: 2·n·max(conv)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _adam_refine(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    lr: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp_start: float = 0.2,
    temp_end: float = 0.001,
    use_softmax: bool = True,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam‑projected (onto the simplex) descent.

    Returns (final_a, best_a, best_score).  `a0` is assumed to sum to 1.
    """
    a = a0.copy()
    n = a.shape[0]

    # pre‑compute FFT lengths
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    if use_softmax:
        fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)  # 1st moment
    v = np.zeros_like(a)  # 2nd moment

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution and current score -----
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- gradient of the smooth (or exact) max -----
        if use_softmax:
            # temperature schedule (linear in iteration count)
            temp_factor = temp_start + (temp_end - temp_start) * (it / max_iters)
            tau = max(temp_factor * max_val, 1e-14)

            w = np.exp((conv - max_val) / tau)      # w_i ≤ 1, w_max = 1
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w.fill(1.0 / w.shape[0])

            # gradient = 2 * (w ⊛ a_rev)   (⊛ = linear convolution)
            W = np.fft.rfft(w, n=fft_len_grad)
            a_rev = a[::-1]
            A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
            conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
            g = 2.0 * conv2[n - 1 : n - 1 + n]
        else:
            # exact sub‑gradient w.r.t. the active (max) index
            arg = int(np.argmax(conv))
            i_start = max(0, arg - n + 1)
            i_end = min(n, arg + 1)
            g = np.zeros_like(a)
            idx = np.arange(i_start, i_end, dtype=int)
            g[idx] = 2.0 * a[arg - idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex -----
        a = _project_simplex(a)

    return a, best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimizes `evaluate_sequence`.
    The routine respects the global time budget (≈1000 s) and prints
    intermediate progress.  It returns the best sequence found as plain
    Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # a small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Build a small pool of diverse length‑n0 candidates (sum = 1)
    # ------------------------------------------------------------------
    n0 = 1800
    candidates = []

    # a) previous construction if available
    try:
        prev = np.array(height_sequence_1, dtype=float)
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0:
            candidates.append(prev / s)
    except Exception:
        pass

    # b) uniform
    candidates.append(np.full(n0, 1.0 / n0))

    # c) triangular (tent)
    idx = np.arange(n0)
    tri = np.minimum(idx + 1, n0 - idx)
    candidates.append(tri / tri.sum())

    # d) Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n0 - 1)))
    candidates.append(hann / hann.sum())

    # e) random Dirichlet point
    candidates.append(np.random.dirichlet(np.ones(n0)))

    # ------------------------------------------------------------------
    # 2. Short refinement of every start point
    # ------------------------------------------------------------------
    for ci, cand in enumerate(candidates, 1):
        if time.time() >= deadline:
            break
        iters = 18000
        print(f"[{time.time() - start_time:6.2f}s] Initial refinement {ci}/{len(candidates)} "
              f"n={n0} iters={iters}")
        final_a, best_a, best_local_score = _adam_refine(
            cand,
            max_iters=iters,
            deadline=deadline,
            lr=0.6,
            temp_start=0.25,
            temp_end=0.003,
            use_softmax=True,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_a.copy()
            print(f"   ** New global best (init {ci}): score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Progressive up‑sampling
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)

    current = best_seq.copy()
    cur_n = current.shape[0]
    MAX_LEN = 8000

    while cur_n < MAX_LEN and time.time() < deadline:
        next_n = int(cur_n * 1.3)
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n == cur_n:
            break

        # linear interpolation to the new grid
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a_up = np.interp(x_new, x_old, current)
        a_up = np.maximum(a_up, 0.0)
        s = a_up.sum()
        if s == 0:
            a_up = np.full(next_n, 1.0 / next_n)
        else:
            a_up /= s

        # iteration budget – fewer iterations for larger n
        base_iters = 25000
        max_iters = int(base_iters * n0 / next_n) + 12000
        max_iters = max(max_iters, 12000)

        print(f"[{time.time() - start_time:6.2f}s] Upsampling to n={next_n} "
              f"iters={max_iters}")

        final_a, best_a, best_local_score = _adam_refine(
            a_up,
            max_iters=max_iters,
            deadline=deadline,
            lr=0.55,
            temp_start=0.20,
            temp_end=0.002,
            use_softmax=True,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_a.copy()
            print(f"   ++ Updated global best after up‑sampling: n={next_n} "
                  f"score={best_score:.6f}")

        # prepare for next up‑sample
        current = final_a
        cur_n = next_n

    # ------------------------------------------------------------------
    # 4. Final long refinement (exact sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        remaining_time = deadline - time.time()
        # Roughly allocate up to 60 % of the remaining time to this stage
        est_iters = 60000
        print(f"[{time.time() - start_time:6.2f}s] Final refinement n={n} "
              f"estimated iters≈{est_iters}")
        final_a, best_a, best_local_score = _adam_refine(
            best_seq,
            max_iters=est_iters,
            deadline=deadline,
            lr=0.45,
            temp_start=0.00,   # not used when use_softmax=False
            temp_end=0.00,
            use_softmax=False,  # pure arg‑max sub‑gradient
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_a.copy()
            print(f"   !! Final improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Normalise and return the best sequence found
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    print(f"[{time.time() - start_time:6.2f}s] Finished. Best score (2·n·max(conv)) = {best_score:.6f}")
    return [float(x) for x in best_seq]
