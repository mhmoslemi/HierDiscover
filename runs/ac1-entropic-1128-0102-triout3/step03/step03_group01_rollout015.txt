def annealing_fast(a, conv, n, deadline, start_time, best_seq, best_score):
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_score = 2*n*cur_conv.max()
    best_seq_local = best_seq.copy()
    best_score_local = best_score
    step = 0
    max_steps = int(5e6)
    start_temp = 0.05
    end_temp = 1e-6
    while time.time() < deadline and step < max_steps:
        step += 1
        # temperature schedule based on step fraction
        frac = step / max_steps
        T = start_temp * (end_temp/start_temp)**frac
        # sample i based on probabilities
        i = np.random.choice(n, p=cur_a)
        # ensure a_i >0
        if cur_a[i] <= 0:
            continue
        # sample j not equal i
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1
        eps = cur_a[i] * np.random.rand() * 0.2
        if eps == 0:
            continue
        # save slices for revert
        slice_i_start = i
        slice_i_end = i + n
        slice_j_start = j
        slice_j_end = j + n

        # backup slices
        old_slice_i = cur_conv[slice_i_start:slice_i_end].copy()
        old_slice_j = cur_conv[slice_j_start:slice_j_end].copy()
        old_2i = cur_conv[2*i]
        old_2j = cur_conv[2*j]
        old_ij = cur_conv[i + j]

        # update conv
        cur_conv[slice_i_start:slice_i_end] -= 2.0 * eps * cur_a
        cur_conv[slice_j_start:slice_j_end] += 2.0 * eps * cur_a

        cur_conv[2*i] += eps * eps
        cur_conv[2*j] += eps * eps
        cur_conv[i + j] -= 2.0 * eps * eps

        # update a
        cur_a[i] -= eps
        cur_a[j] += eps

        # compute new score
        new_max = cur_conv.max()
        new_score = 2.0 * n * new_max

        # acceptance test
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-20)):
            cur_score = new_score
            # update best if improved
            if new_score < best_score_local:
                best_score_local = new_score
                best_seq_local = cur_a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: step {step}, score {new_score:.6f}")
        else:
            # revert
            cur_conv[slice_i_start:slice_i_end] = old_slice_i
            cur_conv[slice_j_start:slice_j_end] = old_slice_j
            cur_conv[2*i] = old_2i
            cur_conv[2*j] = old_2j
            cur_conv[i + j] = old_ij
            cur_a[i] += eps
            cur_a[j] -= eps

def _run_fast_annealing(a0, n, deadline, start_time, cur_best_seq, cur_best_score,
                        start_temp=5e-2, end_temp=1e-6, total_steps=None):
    a = a0.copy()
    conv = _conv_self(a)  # length 2n-1
    cur_score = 2 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    step = 0
    if total_steps is None:
        # estimate steps from remaining time and expected per step
        # We can guess 10k steps per second? Actually we measured maybe 10k per second
        # Let's set total_steps = int((deadline - time.time()) * 8000) to be safe
        total_steps = int((deadline - time.time()) * 8000)
        if total_steps < 1000:
            total_steps = 1000
    while time.time() < deadline and step < total_steps:
        step += 1
        # temperature schedule
        frac = step / total_steps
        T = start_temp * (end_temp / start_temp) ** frac

        # sample source i according to probability proportional to a
        if a.sum() <= 0:
            break
        i = np.random.choice(n, p=a)    # vectorized choice uses probability distribution
        if a[i] <= 0:
            continue
        # choose destination
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1
        # epsilon up to 20% of source mass
        eps = a[i] * np.random.rand() * 0.2
        if eps <= 0:
            continue

        # backup slices
        s_i_start = i
        s_i_end = i + n
        s_j_start = j
        s_j_end = j + n

        old_slice_i = conv[s_i_start:s_i_end].copy()
        old_slice_j = conv[s_j_start:s_j_end].copy()
        old_2i = conv[2*i]
        old_2j = conv[2*j]
        old_ij = conv[i + j]

        # update conv
        conv[s_i_start:s_i_end] -= 2.0 * eps * a
        conv[s_j_start:s_j_end] += 2.0 * eps * a
        conv[2*i] += eps * eps
        conv[2*j] += eps * eps
        conv[i + j] -= 2.0 * eps * eps

        # update a
        a[i] -= eps
        a[j] += eps

        # compute new score
        new_max = conv.max()
        new_score = 2.0 * n * new_max

        # acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-20)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] fast annealing improve: step={step}, score={new_score:.6f}")
        else:
            # revert
            conv[s_i_start:s_i_end] = old_slice_i
            conv[s_j_start:s_j_end] = old_slice_j
            conv[2*i] = old_2i
            conv[2*j] = old_2j
            conv[i + j] = old_ij
            a[i] += eps
            a[j] -= eps

    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) whose self‑convolution
has a small ℓ∞‑norm.  The routine runs for at most ``budget_s`` seconds.
It combines three ideas:

1.  A multi‑scale smoothed gradient descent on the probability simplex
    (the same surrogate that the reference solution uses) to obtain a
    good starting point.
2.  A fast, incremental simulated‑annealing / hill‑climbing phase that
    updates the convolution exactly in O(n) time per move, allowing
    many millions of moves in the remaining budget.
3.  Simple up‑sampling between scales so that a promising distribution
    at a smaller size seeds the optimisation at a larger size.

The algorithm is deterministic apart from the random seed supplied to
``np.random``.  All helper functions are top‑level (no closures) and
the final returned object is a plain Python ``list`` of floats, ready
for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (real FFT, returns length 2·n‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors x and y via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, keeping sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate for ``max(conv)``.
    When ``lam`` → 0 it becomes a sub‑gradient of ``max(conv)``.
    """
    n = a.shape[0]
    # soft‑max weights
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)                # length 3·n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Smoothed gradient descent (same idea as the reference code)
# ----------------------------------------------------------------------


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent with momentum on the simplex.
    Returns the best vector seen and its exact objective
    (2·n·max(conv) because the vector is kept normalised).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 0.8
    it = 0

    # λ‑schedule
    lam_factor = 0.25
    lam_decay = 0.5
    lam_decay_interval = 2_000

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, it={it}, score={cur_score:.6f}")

        # adapt λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast incremental simulated annealing / hill‑climbing
# ----------------------------------------------------------------------


def _run_fast_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_seq: np.ndarray,
                        cur_best_score: float,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Incremental SA that updates the self‑convolution exactly in O(n)
    per move.  The move transfers a small ε from a source index i to a
    destination j.  The temperature is exponentially annealed.
    Returns the best sequence discovered and its exact objective.
    """
    a = a0.copy()
    conv = _conv_self(a)                     # length 2·n‑1
    cur_score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    # estimate a feasible number of steps from remaining time
    remaining = max(0.0, deadline - time.time())
    # roughly 8 000 updates per second on a modern CPU for n≈6400
    max_steps = int(remaining * 8_000) + 1
    if max_steps < 5_000:
        max_steps = 5_000

    step = 0
    while time.time() < deadline and step < max_steps:
        step += 1
        # temperature schedule (exponential)
        frac = step / max_steps
        T = start_temp * (end_temp / start_temp) ** frac

        # pick source proportional to its mass (more likely to move mass from large piles)
        if a.sum() <= 0:
            break
        i = np.random.choice(n, p=a)
        if a[i] <= 0.0:
            continue

        # destination – uniform among the other indices
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # ε up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        if eps <= 0.0:
            continue

        # backup the parts of conv that will be overwritten
        s_i_start, s_i_end = i, i + n
        s_j_start, s_j_end = j, j + n

        old_slice_i = conv[s_i_start:s_i_end].copy()
        old_slice_j = conv[s_j_start:s_j_end].copy()
        old_2i = conv[2 * i]
        old_2j = conv[2 * j]
        old_ij = conv[i + j]

        # --- apply the move (exact update) ---------------------------------
        # linear terms
        conv[s_i_start:s_i_end] -= 2.0 * eps * a
        conv[s_j_start:s_j_end] += 2.0 * eps * a
        # quadratic (δ*δ) terms
        conv[2 * i] += eps * eps
        conv[2 * j] += eps * eps
        conv[i + j] -= 2.0 * eps * eps

        # update the probability vector
        a[i] -= eps
        a[j] += eps

        # compute new objective
        new_max = conv.max()
        new_score = 2.0 * n * new_max

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-20)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] fast anneal improve: n={n}, step={step}, score={new_score:.6f}")
        else:
            # revert
            conv[s_i_start:s_i_end] = old_slice_i
            conv[s_j_start:s_j_end] = old_slice_j
            conv[2 * i] = old_2i
            conv[2 * j] = old_2j
            conv[i + j] = old_ij
            a[i] += eps
            a[j] -= eps

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm works in several size stages,
    runs a smoothed gradient descent at each stage and finishes with a
    fast incremental simulated‑annealing sweep.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # optional starter from the notebook environment
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # length schedule – grow geometrically, finish with the largest size
    lengths = [2000, 4000, 6400, 9600, 12800]
    # we keep the order increasing so that the final largish vector gets most time
    best_seq = None
    best_score = float('inf')

    for n in lengths:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # initialise a0
        # --------------------------------------------------------------
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # allocate a slice of the remaining budget for this scale
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # Roughly 30 % of the remaining budget for gradient descent,
        # the rest for the final annealing.
        gd_budget = min(remaining * 0.3, 40.0)          # never exceed ~40 s for GD
        gd_deadline = time.time() + gd_budget

        # --------------------------------------------------------------
        # smoothed gradient descent (same method as the reference code)
        # --------------------------------------------------------------
        a_opt, score_opt = _run_smooth_descent(a0, n, gd_deadline, start_time)

        # exact objective for the best vector found by GD
        cur_score = 2.0 * n * _conv_self(a_opt).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after GD: n={n}, score={cur_score:.6f}")

        # --------------------------------------------------------------
        # fast incremental annealing on the remaining time for this scale
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = deadline  # use all remaining time on the final size
            a_ref, score_ref = _run_fast_annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_seq,
                best_score,
                start_temp=5e-2,
                end_temp=1e-6,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n}, score={score_ref:.6f}")

        # keep the best vector for the next up‑sampling step
        best_seq = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fall‑back (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(200, 1.0 / 200.0)

    # return a plain Python list
    return [float(x) for x in best_seq]
