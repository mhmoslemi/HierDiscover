"""
Propose a step‑function sequence with a low evaluation_score.

The method combines several ideas that improve on the previous
soft‑max projected gradient descent:

*   Adam‑style adaptive steps instead of simple Nesterov momentum.
*   A temperature schedule that starts warm (≈0.5·max) and cools to
    ≈0.001·max, focusing the gradient on the large convolution entries.
*   Enforced symmetry a_i = a_{n-1-i} after each update – the optimal
    solution is believed to be (approximately) symmetric, and this halves
    the effective search space.
*   Multi‑start random Dirichlet initialisations together with a few
    deterministic shapes (uniform, triangular, Hann, previous best).
*   Progressive up‑sampling (≈1.5×) with a larger number of Adam steps at
    each resolution.
*   A final stochastic “annealing” phase that adds small Gaussian
    perturbations and accepts strictly improving moves (simple hill‑climb).

All iterates are kept on the probability simplex (sum = 1, a_i ≥ 0) by a
Euclidean projection.  The score that is optimised is
`2·n·max(conv(a,a))`, which equals the value returned by
`evaluate_sequence` when the sequence is normalised to sum = 1.

The routine respects the 1000 s time limit, prints progress, and returns
the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """
    Score for a sequence that already sums to 1:
    score = 2·n·max(convolution(a, a)).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _optimize_length(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    eps_start: float = 0.5,
    eps_end: float = 0.001,
    lr: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam‑based soft‑max projected gradient descent for a fixed length.

    Returns (final_a, best_seen_a, best_seen_score).
    The input and all intermediate vectors are kept on the simplex (sum = 1).
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT lengths (next powers of two for speed)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())  # length of w ⊛ a_rev

    # Adam accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- convolution and current score ----
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---- soft‑max weights (temperature schedule) ----
        # exponential decay of epsilon
        eps = eps_start * (eps_end / eps_start) ** (t / max_iters)
        tau = max(eps * max_val, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # ---- gradient of the smooth approximation ----
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        g = 2.0 * conv2[n - 1 : n - 1 + n]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + 1e-8)

        a = a - step * m_hat

        # ---- enforce symmetry and project onto simplex ----
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a)

    return a, best_a, best_score


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation up‑sampling preserving end‑points."""
    old_len = seq.shape[0]
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    up = up / up.sum()
    # enforce symmetry after interpolation
    up = (up + up[::-1]) * 0.5
    return up
