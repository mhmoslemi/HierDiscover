"""
propose_candidate – a fast multi‑phase optimizer for the step‑function bound.

The search consists of three stages:

1. **Initialisation** – start from the provided `height_sequence_1` (if any) or a
   random simplex point.

2. **Progressive up‑sampling + smooth mirror descent** – the current shape is
   repeatedly interpolated to a larger support (≈1.4× each stage, up to a
   configurable maximum size).  For each size we run a mirror‑descent loop on
   the probability simplex:

   * The objective  `2·n·max(conv(a,a))` (with `∑a = 1`) is smoothed by a
     soft‑max with a temperature that rises from `α_min` to `α_max`.  The
     gradient of the smoothed objective is
       `∇ = 2 * (p  *  a_rev)`   (convolution of the soft‑max weights `p`
       with the reversed vector `a_rev`).
   * An exponential update `log a ← log a – η·∇` is performed, where the step
     size η decays as `η₀/√t`.  After each update the vector is renormalised
     to stay on the simplex.
   * Every few thousand iterations a tiny uniform jitter is blended in to
     avoid stagnation.

   The best sequence seen so far (across all sizes) is kept.

3. **Final stochastic mass‑transfer hill‑climbing** – starting from the best
   sequence a short random‑move local search is performed until the time
   budget expires.

All operations are implemented with real FFT‑based convolutions ( O(N log N) ),
so the method easily handles sequences of several‑tens of thousands of
entries within the 1000 s limit.

The returned list consists of non‑negative floats; `evaluate_sequence`
will therefore accept it directly.

def real_fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full real convolution of x and y using FFT."""
    n = len(x) + len(y) - 1
    L = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Objective for a vector that already satisfies sum(a)=1."""
    n = a.shape[0]
    conv = real_fft_convolve(a, a)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


def compute_gradient(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max smoothed objective.
    p – soft‑max probabilities (length 2·n‑1),
    a – current probability vector (length n).
    The formula is  ∇_i = 2·∑_k p_k·a_{k-i}  (indices out of range are ignored).
    This is obtained by convolving p with the reversed a.
    """
    a_rev = a[::-1]
    conv_all = real_fft_convolve(p, a_rev)                # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * conv_all[n - 1 : n - 1 + n]               # slice of length n
    return grad


def run_opt_on_size(a: np.ndarray,
                    best_seq: np.ndarray,
                    best_score: float,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 20000,
                    eta0: float = 1.0,
                    alpha_min: float = 2.0,
                    alpha_max: float = 30.0) -> tuple[np.ndarray, np.ndarray, float]:
    """Mirror‑descent optimisation for a fixed support size."""
    n = a.shape[0]
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- evaluate current vector -----
        conv = real_fft_convolve(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
