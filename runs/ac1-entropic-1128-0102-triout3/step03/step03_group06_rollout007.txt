        # compute new counts efficiently via delta update:
        # Remove i_remove contributions
        # However for simplicity in prototyping compute counts new each time (O(k^2))

def fast_score(a):
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s*s))
```

"""
Propose a low‑score step‑function by searching for a sparse support set
with a small maximal ordered‑pair count.  The algorithm works in three
phases:

1.  Randomly initialise a support set S of size k (k≈100) inside a
    window of length n=1500.  The convolution of the indicator of S
    with itself yields the ordered‑pair count for each possible sum.
2.  Perform a hill‑climbing / simulated‑annealing style local search:
    repeatedly swap one element of S with a new index and update the
    count vector incrementally (O(k) work).  Swaps that do not increase
    the maximal count are always accepted; occasional worsening moves
    are allowed with a temperature schedule.  The search is repeatedly
    restarted until a set with max‑count ≤ allowed (derived from the
    target score) is found or time runs out.
3.  If a suitable set is found, the final sequence is uniform on the
    support (mass 1/k).  Otherwise a fallback projected sub‑gradient
    descent (the previous approach) is applied starting from the best
    vector seen.

The whole routine respects the 1000 s budget, prints progress and
returns a list of non‑negative floats suitable for ``evaluate_sequence``.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (identical to evaluate_sequence)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """
    Ordered‑pair count of a support set S ⊂ {0,…,n‑1}.
    Returns a length‑(2·n‑1) integer array where entry s equals
    |{(i,j)∈S×S : i+j=s}|.
    """
    sums = np.add.outer(S, S)                # includes both (i,j) and (j,i)
    return np.bincount(sums.ravel(), minlength=2 * n - 1)


# ----------------------------------------------------------------------
def attempt_find_set(n: int,
                    k: int,
                    allowed_max: int,
                    time_limit: float) -> np.ndarray | None:
    """
    Try to construct a support set of size ``k`` with ordered‑pair count
    never exceeding ``allowed_max``.  A simple stochastic hill climbing
    with occasional random restarts is used.  Returns the set as a 1‑D
    numpy array of indices, or ``None`` if the time budget expires.
    """
    deadline = time.time() + time_limit
    best_S = None
    best_max = math.inf
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ---- initialise a random set ---------------------------------
        S = rng.choice(n, size=k, replace=False).astype(np.int32)
        S_set = set(int(x) for x in S)
        counts = compute_counts(S, n)
        cur_max = int(counts.max())

        if cur_max < best_max:
            best_max, best_S = cur_max, S.copy()

        if cur_max <= allowed_max:
            return S

        # ---- local search -------------------------------------------
        # limit the number of swaps for this start to avoid endless loops
        max_swaps = 20000
        no_improve = 0
        for _ in range(max_swaps):
            if time.time() >= deadline:
                break

            # pick a random element to remove
            idx_rm = rng.integers(k)
            i_rm = int(S[idx_rm])

            # pick a new index not currently used
            i_add = int(rng.integers(n))
            while i_add in S_set:
                i_add = int(rng.integers(n))

            # ----- compute the new counts in a copy (vectorised) -----
            new_counts = counts.copy()

            # remove contributions of i_rm (both ordered directions)
            new_counts[i_rm + S] -= 2
            new_counts[2 * i_rm] += 1               # correct self‑pair

            # build the set without the removed element
            mask = np.arange(k) != idx_rm
            S_others = S[mask]

            # add contributions of i_add
            new_counts[i_add + S_others] += 2
            new_counts[2 * i_add] += 1

            new_max = int(new_counts.max())

            # acceptance rule (strict improvement or occasional uphill)
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                # a simple temperature that slowly cools
                elapsed = time.time() - (deadline - time_limit)
                frac = elapsed / time_limit
                temperature = max(0.5, (1.0 - frac) * 5.0)
                if math.exp(-(new_max - cur_max) / temperature) > rng.random():
                    accept = True

            if accept:
                # commit the swap
                S_set.remove(i_rm)
                S_set.add(i_add)
                S[idx_rm] = i_add
                counts = new_counts
                cur_max = new_max
                no_improve = 0

                if cur_max < best_max:
                    best_max, best_S = cur_max, S.copy()

                if cur_max <= allowed_max:
                    return S
            else:
                no_improve += 1

            # give up on this start if we see no improvement for a while
            if no_improve >= 3000:
                break

    # after time out, return the best set found (if it already satisfies)
    if best_max <= allowed_max:
        return best_S
    return None


# ----------------------------------------------------------------------
def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Simple projected sub‑gradient descent (fallback when the combinatorial
    search fails).  The method works on the original (non‑uniform) vector.
    """
    a = a_init.copy()
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    best_a = a.copy()
    best_val = fast_score(a)

    eta0 = 0.30
    eta_decay = 1e-5
    random_move_prob = 0.02
    random_move_step = 0.15

    it = 0
    deadline = time.time() + time_limit - 0.2
    ind = np.arange(1, n + 1)

    while time.time() < deadline:
        # convolution and active index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient for the active index
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the probability simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass‑transfer move
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_step * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # projection again
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        it += 1

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function whose evaluation score is ≤ 1.503.
    The primary strategy builds a sparse support set with a bounded
    ordered‑pair count; if that fails a fallback sub‑gradient optimiser
    is invoked.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # reserve a little safety margin

    n = 1500                             # length of the step function
    target_score = 1.503

    # Try several support sizes, roughly decreasing difficulty.
    support_sizes = [110, 100, 120, 90, 80]

    best_seq = None
    best_val = np.inf

    for k in support_sizes:
        allowed = max(1,
                      int(math.floor(target_score * k * k / (2.0 * n))))
        print(f"[search] trying k={k}, allowed max count={allowed}")

        # allocate at most 200 s for a single size (or whatever is left)
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        local_budget = min(200.0, remaining)

        S = attempt_find_set(n, k, allowed, local_budget)
        if S is not None:
            # construct the uniform‑weight vector on the support
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / k
            val = fast_score(a)
            print(f"[found] k={k} -> score {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a.copy()
            if val <= target_score:
                print("[success] target met, returning sequence.")

        else:

    # ------------------------------------------------------------------

    if best_seq is None:

    # Final safety: ensure we return a list of plain floats.

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    print(f"[final] best value returned: {best_val:.6f}")
    return best_seq.tolist()
