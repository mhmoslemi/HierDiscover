def _adam_optimize(a, max_iters, deadline, alpha, beta1, beta2, eps, temp_start, temp_end):
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    # compute initial score
    conv = np.fft.irfft(np.fft.rfft(a, n=fft_len_conv) * np.fft.rfft(a, n=fft_len_conv), n=fft_len_conv)[:2*n-1]
    max_val = conv.max()
    best_score = 2.0 * n * max_val

    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        # conv and max
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule
        tau = temp_start + (temp_end - temp_start) * (t / max_iters)
        # compute w
        w = np.exp((conv - max_val) / max(tau, 1e-14))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient via convolution w * a_rev
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        g = 2.0 * conv2[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # project onto simplex
        a = _project_simplex(a)

    return a, best_a, best_score

"""
Multi‑scale projected Adam optimisation for the auto‑convolution max‑norm.

The evaluation
    score = 2·n·max(conv(a,a)) / (sum(a)^2)
is invariant to scaling, therefore we keep the vector on the probability
simplex (sum = 1, a_i ≥ 0).  For a fixed length `n` we minimise
`2·n·max(conv)` using a smooth approximation of the max (soft‑max) and an
Adam‑style stochastic‑gradient step followed by Euclidean projection onto the
simplex (Duchi et al.).  The optimiser is run at several resolutions:
starting from moderate lengths (≈1200) we up‑sample the current best shape,
run several random restarts and finally devote the remaining time budget to
a long refinement of the best sequence found.  The algorithm exploits FFTs for
fast convolution and adapts the soft‑max temperature from 0.05 down to 1e‑4.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full convolution (real).  `a` is assumed to sum to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective w.r.t. `a`.
    Implements g_i = 2 * Σ_k w_k * a_{k-i}
    using FFT for speed.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    # slice the centre n entries
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.05,
                   temp_end: float = 0.0005) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimisation of 2·n·max(conv) on the simplex.

    Returns (final_a, best_seen_a, best_score).
    """
    n = a.shape[0]
    # FFT lengths – cached for the whole run
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)        # first moment
    v = np.zeros_like(a)        # second moment

    # initialise best
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- current convolution and score ----
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---- soft‑max weights over convolution entries ----
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        # avoid division by zero
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # ---- gradient of the smooth max ----
        g = _gradient_from_weights(a, w, fft_len_grad)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ---- projection onto the simplex ----
        a = _project_simplex(a)

    return a, best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step function within the given time budget.
    The routine uses several scales, random restarts and a long Adam refinement.
    Returns a plain list of floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin to avoid overrunning the allotted time
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. size schedule (≈1.5× growth, capped at 8000)
    # ------------------------------------------------------------------
    sizes = []
    n = 1200
    MAX_N = 8000
    while n <= MAX_N:
        sizes.append(n)
        n = int(n * 1.5)

    # try to load the previously discovered construction, if any
    prev_seq = None
    try:
        prev_seq = np.array(height_sequence_1, dtype=float)
        prev_seq = np.maximum(prev_seq, 0.0)
        if prev_seq.sum() > 0:
            prev_seq = prev_seq / prev_seq.sum()
    except Exception:
        prev_seq = None

    # ------------------------------------------------------------------
    # 2. iterate over sizes
    # ------------------------------------------------------------------
    for stage, n in enumerate(sizes, 1):
        if time.time() >= deadline:
            break

        print(f"[{time.time() - start_time:6.2f}s] Stage {stage}/{len(sizes)} – n = {n}")

        # ---- collect a small pool of initial candidates for this size ----
        candidates = []

        # up‑sample current global best (if it exists and has a different length)
        if best_seq is not None and best_seq.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, best_seq.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, best_seq)
            up = np.maximum(up, 0.0)
            s = up.sum()
            if s == 0:
                up = np.full(n, 1.0 / n)
            else:
                up = up / s
            candidates.append(up)

        # previous construction (rescaled if necessary)
        if prev_seq is not None:
            if prev_seq.shape[0] != n:
                x_old = np.linspace(0.0, 1.0, prev_seq.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                up = np.interp(x_new, x_old, prev_seq)
                up = np.maximum(up, 0.0)
                s = up.sum()
                if s == 0:
                    up = np.full(n, 1.0 / n)
                else:
                    up = up / s
                candidates.append(up)
            else:
                candidates.append(prev_seq.copy())

        # deterministic shapes
        candidates.append(np.full(n, 1.0 / n))                     # uniform
        idx = np.arange(n)
        tri = np.minimum(idx + 1, n - idx)                      # triangular
        candidates.append(tri / tri.sum())
        if n > 1:
            hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))  # raised‑cosine
            candidates.append(hann / hann.sum())
        else:
            candidates.append(np.full(1, 1.0))

        # a few random Dirichlet starts for diversity
        for _ in range(2):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # ------------------------------------------------------------------
        # 3. optimise each candidate at the current resolution
        # ------------------------------------------------------------------
        for cid, cand in enumerate(candidates, 1):
            if time.time() >= deadline:
                break

            # allocate a generous iteration budget, but respect the global deadline
            # rough estimate of iteration time grows roughly linearly with n
            est_iter_time = 6e-4 * (n / 1200.0) + 1e-4
            remaining = deadline - time.time()
            max_iters = int(min(200000, max(15000, remaining / est_iter_time)))
            if max_iters <= 0:
                break

            print(
                f"   [{time.time() - start_time:5.2f}s]  cand {cid}/{len(candidates)} "
                f"– max_iters={max_iters}"
            )

            _, local_best_a, local_best_score = _adam_optimize(
                cand.copy(),
                max_iters=max_iters,
                deadline=deadline,
                alpha=0.2,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                temp_start=0.05,
                temp_end=0.0005,
            )

            if local_best_score < best_score:
                best_score = local_best_score
                best_seq = local_best_a.copy()
                print(
                    f"       NEW GLOBAL BEST (n={n}) – score = {best_score:.6f}"
                )

        # optional quick refinement of the current best at this size
        if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
            remaining = deadline - time.time()
            est_iter_time = 6e-4 * (n / 1200.0) + 1e-4
            extra_iters = int(min(50000, remaining / est_iter_time))
            if extra_iters > 0:
                print(
                    f"   [{time.time() - start_time:5.2f}s]  extra refinement (n={n}) – iters={extra_iters}"
                )
                _, local_best_a, local_best_score = _adam_optimize(
                    best_seq.copy(),
                    max_iters=extra_iters,
                    deadline=deadline,
                    alpha=0.15,
                    beta1=0.9,
                    beta2=0.999,
                    eps=1e-8,
                    temp_start=0.02,
                    temp_end=0.0002,
                )
                if local_best_score < best_score:
                    best_score = local_best_score
                    best_seq = local_best_a.copy()
                    print(
                        f"       IMPROVEMENT AFTER EXTRA REFINEMENT – score = {best_score:.6f}"
                    )

    # ------------------------------------------------------------------
    # 4. final long refinement on the overall best sequence
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        remaining = deadline - time.time()
        est_iter_time = 6e-4 * (n / 1200.0) + 1e-4
        final_iters = int(min(300000, max(50000, remaining / est_iter_time)))
        if final_iters > 0:
            print(
                f"[{time.time() - start_time:6.2f}s] Final refinement – n={n}, iters={final_iters}"
            )
            _, local_best_a, local_best_score = _adam_optimize(
                best_seq.copy(),
                max_iters=final_iters,
                deadline=deadline,
                alpha=0.1,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                temp_start=0.01,
                temp_end=0.0001,
            )
            if local_best_score < best_score:
                best_score = local_best_score
                best_seq = local_best_a.copy()
                print(f"   FINAL IMPROVEMENT – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. post‑process and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # extreme fallback – small uniform vector
        best_seq = np.full(100, 1.0 / 100)

    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    print(
        f"[{time.time() - start_time:6.2f}s] Finished – best 2·n·max(conv) = {best_score:.6f}"
    )
    return [float(x) for x in best_seq]
