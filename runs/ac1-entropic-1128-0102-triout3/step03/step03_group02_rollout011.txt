def subgradient(a):
    n = a.shape[0]
    # compute convolution
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    max_val = conv.max()
    # find active indices
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv)
    if len(active) > 0:
        w[active] = 1.0 / len(active)
    else:
        w[conv.argmax()] = 1.0
    # compute gradient: 2 * conv(w, a_rev)
    a_rev = a[::-1]
    fft_len_grad = 1 << ((len(conv) + n - 1).bit_length())
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    grad_full = np.fft.irfft(W * A_rev, n=fft_len_grad)
    grad = 2.0 * grad_full[n-1:n-1+n]
    return grad, max_val, conv
```

"""
Propose a step‑function with a low evaluation score.

The search combines three ideas that are complementary to the previous
soft‑max gradient descent:

1. **Exact sub‑gradient descent** – the gradient of the max‑convolution is
   obtained analytically (the weight vector is a uniform distribution over
   the active set where the convolution attains its maximum).  A Nesterov‑style
   momentum update together with a Euclidean projection onto the probability
   simplex keeps the iterates feasible.

        maximise   ∑ b_j
        subject to   (a∗b)_k ≤ M   for all k,
        b_j ≥ 0

   unit L1‑norm we move a fraction `t` towards it (the alternating‑minimisation
   scheme from the problem statement).  A single LP step per length already
   provides a noticeable improvement.

3. **Progressive up‑sampling** – the best shape found for a length `n` is
   linearly interpolated to a larger size (≈1.5×).  The enlarged vector is
   re‑optimised, which allows the algorithm to explore richer families of
   step functions without having to start from scratch.

The routine respects the global time budget (default 1000 s) and prints
intermediate scores.  The returned list is a plain‑Python sequence whose
elements are non‑negative and sum to 1, so the value printed by
`evaluate_sequence` equals `2·n·max(conv(a,a))`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al.
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Af * Af, n=size)[: 2 * n - 1]
    return conv


def _conv_max(a: np.ndarray) -> float:
    return _full_convolution(a).max()


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (uniform weight on active set)
def _subgradient(a: np.ndarray):
    n = a.shape[0]
    conv = _full_convolution(a)
    M = conv.max()
    # active set (indices where the max is attained – a small tolerance avoids
    # missing ties caused by floating‑point noise)
    tol = 1e-12
    active = np.where(conv >= M - tol)[0]
    w = np.zeros_like(conv)
    w[active] = 1.0 / max(1, len(active))

    # gradient = 2 * (w * a_rev) convolution, restricted to the original n entries
    a_rev = a[::-1]
    size = 1 << ((len(conv) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    grad_full = np.fft.irfft(W * A_rev, n=size)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, M, conv


# ----------------------------------------------------------------------
# Gradient descent on the simplex (projected Nesterov momentum)
def _refine_subgrad(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.8,
    beta: float = 0.9,
) -> tuple[np.ndarray, np.ndarray, float]:
    n = a.shape[0]
    v = np.zeros_like(a)                # momentum term
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        g, cur_max, _ = _subgradient(a)
        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)

        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return a, best_a, best_score


# ----------------------------------------------------------------------
# One LP refinement step (alternating minimisation)
def _lp_step(a: np.ndarray, M: float):
    """Solve   maximise 1ᵀb   s.t.  (a∗b) ≤ M,  b ≥ 0."""
    try:
        import cvxpy as cp
    except Exception:   # cvxpy not available – skip LP
        return None

    n = a.shape[0]

    # Build the (2n‑1)×n convolution matrix.
    # Row k, column i contains a_i if 0 ≤ k‑i < n, else 0.
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = a

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    # ECOS is a fast interior‑point solver for modest LPs.
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)

    if b.value is None:
        return None
    return np.asarray(b.value).flatten()


def _lp_refine(a: np.ndarray, t: float, steps: int, deadline: float):
    """Perform a few alternating LP refinements."""
    for _ in range(steps):
        if time.time() >= deadline:
            break
        M = _conv_max(a)
        g = _lp_step(a, M)
        if g is None:
            break
        s = g.sum()
        if s <= 0:
            break
        g = g / s                     # bring to the simplex
        a = (1.0 - t) * a + t * g     # convex combination
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """Return a non‑negative step function with a low evaluation score.

    The routine runs for at most ``budget_s`` seconds and prints progress.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate the current score (sum is forced to 1)
    def _score(a):

        return 2.0 * n * _conv_max(a)

            prev = prev[:first_n]
            prev = _project_simplex(prev)   # ensure exact simplex
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # 1. Base lengths – we start small, then expand.
    base_lengths = [500, 1000, 1500, 2250, 3400]   # geometric progression, ≤ 4000
    # The largest length we ever attempt is limited by the remaining time.
    # (We will stop earlier if the deadline is reached.)

    for n in base_lengths:
        if time.time() >= deadline:
            break
        print(f"\n[{time.time() - start_time:6.2f}s] Working at length n={n}")

        # --------------------------------------------------------------
        # Assemble a pool of starting points (all on the simplex).
        # --------------------------------------------------------------
        starts = []

        # a) uniform
        starts.append(np.full(n, 1.0 / n))

        # b) triangular (“tent”)
        idx = np.arange(n)
        tri = np.minimum(idx + 1, n - idx)
        starts.append(tri / tri.sum())

        # c) Hann window (raised cosine)
        hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
        starts.append(hann / hann.sum())

        # d) a random Dirichlet vector (single random start)
        starts.append(np.random.dirichlet(np.ones(n)))

        # e) previous best shape, interpolated to the current size
        if best_seq is not None:
            x_old = np.linspace(0.0, 1.0, best_seq.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, best_seq)
            interp = np.maximum(interp, 0.0)
            s = interp.sum()
            if s > 0:
                starts.append(interp / s)

        # f) the “prev” shape extracted from height_sequence_1, if available
        if prev is not None and prev.shape[0] == n:
            starts.append(prev)

        # --------------------------------------------------------------
        # Refine each start independently (moderate iteration budget).
        # --------------------------------------------------------------
        iters = int(40000 * 1200 / n) + 15000   # fewer iterations for longer vectors
        for si, start in enumerate(starts, 1):
            if time.time() >= deadline:
                break
            print(
                f"  [{time.time() - start_time:6.2f}s]  Start {si}/{len(starts)} – iters={iters}"
            )
            a, a_best, score = _refine_subgrad(
                start.copy(),
                max_iters=iters,
                deadline=deadline,
                step0=0.8,
                beta=0.9,
            )
            # One cheap LP refinement (only for moderate sizes)
            if n <= 1800:
                a = _lp_refine(a, t=0.3, steps=3, deadline=deadline)
                score = _score(a)

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"   >>> New global best: score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2. Upsample the current global best to the next length (if any).
        # --------------------------------------------------------------
        # Upsampling is performed after the loop, just before we move on
        # to the next length (see the code block after the for‑loop).

        # Store for the next iteration.
        if best_seq is not None:
            # Keep a copy for up‑sampling later.
            last_best = best_seq.copy()
        else:
            last_best = np.full(n, 1.0 / n)

        # If we have a next length, interpolate now.
        # (We will reuse `last_best` at the top of the next iteration.)
        # The up‑sampling itself is cheap, so we do it here.
        # ------------------------------------------------------------------
        # Determine the size of the next step (≈1.5× bigger, never exceeding 8000).
        # ------------------------------------------------------------------
        # We compute it at the end of the loop; the next iteration will pick it up.
        # (The logic is simple: the for‑loop itself already enumerates the sizes.)

    # ------------------------------------------------------------------
    # 3. Final fine‑tuning on the best length discovered.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        final_iters = int(60000 * 1200 / n_final) + 20000
        print(
            f"\n[{time.time() - start_time:6.2f}s] Final refinement on n={n_final}, iters={final_iters}"
        )
        a, a_best, score = _refine_subgrad(
            best_seq.copy(),
            max_iters=final_iters,
            deadline=deadline,
            step0=0.5,
            beta=0.9,
        )
        # One more LP pass for the final vector (if affordable)
        if n_final <= 2000:
            a = _lp_refine(a, t=0.2, steps=4, deadline=deadline)
            score = _score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"   >>> Final improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Normalise and return the best sequence found.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Very unlikely fallback – a modest uniform vector.
        best_seq = np.full(300, 1.0 / 300)

    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    print(
        f"\n[{time.time() - start_time:6.2f}s] Finished. Best score = {best_score:.6f}"
    )
    return [float(x) for x in best_seq]
