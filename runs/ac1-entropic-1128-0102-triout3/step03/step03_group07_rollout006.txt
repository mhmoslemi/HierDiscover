"""
Propose a low‑score coefficient sequence for the auto‑convolution max problem.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) and
combines three ideas:

1.  A smooth “soft‑max” surrogate for the max‑operator (log‑sum‑exp) with a
    gradually increasing temperature β.  The gradient of the surrogate is
    computed by a single FFT‑based correlation.

2.  Adaptive Adam‑style updates with Euclidean projection back onto the
    simplex, which give a fast, momentum‑driven descent in the original
    non‑smooth objective.

3.  Simple stochastic perturbations and occasional restarts (Dirichlet
    draws) to escape flat regions.

The routine adapts the learning‑rate, β‑schedule and restart
conditions on the fly and returns the best sequence seen before the
time budget expires.
"""

import time
import numpy as np
from typing import List


def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^‑½  (a_i = C(2i,i)/4^i) and normalised
    so that they sum to 1.  This provides a strong deterministic start.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

    a = project_onto_simplex(init_seq)
