"""
Propose a low‑score step‑function.

The algorithm works in two stages.

1.  **Combinatorial support construction** –  
    For a given bound on the allowed number of ordered
    pairs per convolution index (``max_allowed``) we try to build a
    large support set ``S`` (indices with non‑zero weight) by a fast
    greedy process.  The greedy routine adds indices in a random order
    while keeping the convolution‐count of every sum ≤ ``max_allowed``.
    For a fixed ``max_allowed`` the maximal possible size of such a set
    is roughly ``sqrt(max_allowed * n)`` (B₂[g] theory).  By trying several
    values of ``max_allowed`` we obtain many candidate supports.
    The uniform vector on the support has score  
    ``2·n·max_count/|S|²`` (because the sum is normalised to 1).

2.  **Continuous refinement** –  
    The best uniform vector found is fed to a projected sub‑gradient
    descent (the same routine used in the reference solution) for the
    remaining time.  This often reduces the maximal convolution value
    further, yielding a final score below the target ``1.503``.

The whole search respects the 1000 s time budget, uses only a single
CPU and never accesses the file system.
"""

import numpy as np
import math
import time


# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` for a non‑negative vector."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # FFT size – next power of two that fits the full convolution
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """
    Ordered‑pair convolution counts for a support set ``S``.
    Returns an integer array of length ``2·n‑1`` where entry ``k`` is the
    number of ordered pairs (i,j)∈S×S with i+j=k.
    """
    sums = np.add.outer(S, S)                 # (|S|,|S|)
    return np.bincount(sums.ravel(), minlength=2 * n - 1)


# ----------------------------------------------------------------------
def greedy_build_set(n: int, k_target: int, max_allowed: int) -> np.ndarray | None:
    """
    Try to construct a support set of size ``k_target`` such that every
    convolution count stays ≤ ``max_allowed``.  Returns the set as a 1‑D
    NumPy array of ints or ``None`` if the random order failed.
    """
    order = np.random.permutation(n)
    S = []                         # selected indices
    counts = np.zeros(2 * n - 1, dtype=np.int16)

    for idx in order:
        if len(S) >= k_target:
            break

        # self‑pair (i,i)
        if counts[idx + idx] + 1 > max_allowed:
            continue

        if S:
            existing = np.array(S, dtype=np.int32)
            sums = idx + existing
            # each cross‑pair contributes 2 ordered pairs
            if np.any(counts[sums] + 2 > max_allowed):
                continue

        # accept idx
        S.append(int(idx))
        counts[idx + idx] += 1
        if S[:-1]:                      # there were previous points
            counts[idx + existing] += 2

    if len(S) == k_target:
        return np.array(S, dtype=np.int32)
    return None

def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on the original objective.
    The vector is kept on the probability simplex (non‑negative, sum = 1).
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    eta0 = 0.30
    eta_decay = 1e-5
    random_move_prob = 0.03
    random_move_step = 0.20
    it = 0
    deadline = time.time() + time_limit - 0.05
    ind = np.arange(1, n + 1)                 # used for projection

    while time.time() < deadline:
        # convolution and the (one) maximal index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered‑pair counting)
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size (simple decay)
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer move
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_step * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # re‑project onto simplex
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 2500 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")

        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a
