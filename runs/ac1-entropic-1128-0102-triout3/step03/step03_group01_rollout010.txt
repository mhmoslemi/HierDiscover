"""
Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm works in three stages:

1. **Warm‑start** – if a previous construction is available
   (``height_sequence_1``) it is normalised to the probability simplex and
   up‑sampled to a target length.

2. **Smoothed gradient descent** – a few seconds of the log‑sum‑exp
   surrogate optimisation (the same routine that was used previously) are
   performed on the target length to obtain a good coarse shape.

3. **Fast simulated‑annealing refinement** – the self‑convolution is updated
   incrementally (O(n) per move) so that millions of cheap mass‑transfer
   moves can be tried within the remaining time.  The temperature is
   exponentially cooled and the best vector seen is kept.

All operations stay on the probability simplex (∑aᵢ=1), which makes the
objective exactly ``2·n·max(conv)`` and eliminates any scale‑dependence.
The final list of heights is returned as plain Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n (renormalised to sum = 1)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of the log‑sum‑exp surrogate."""
    n = a.shape[0]
    # stable soft‑max over conv entries
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)                # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iter: int = 150_000) -> tuple[np.ndarray, float]:
    """Smoothed gradient descent with momentum on the simplex."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 0.8
    it = 0

    lam_factor = 0.25       # initial λ = lam_factor * current max
    lam_decay = 0.5
    lam_decay_interval = 2_000

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, "
                  f"it={it}, score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast simulated‑annealing (incremental convolution update)
# ----------------------------------------------------------------------


def _run_fast_annealing(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform many cheap mass‑transfer moves using an O(n) incremental update
    of the convolution.  The temperature is exponentially cooled.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    max_b = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * max_b

    init_T = max_b * 0.5          # start temperature as a fraction of current max
    final_T = 1e-8                # final temperature

    while time.time() < deadline:
        # fraction of the annealing schedule that has elapsed
        frac = (time.time() - start_time) / (deadline - start_time)
        T = init_T * (final_T / init_T) ** frac

        # pick source index i (must have positive mass)
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue

        # pick a distinct target index j
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose moving a small epsilon from i to j (up to 10 % of source mass)
        eps = a[i] * np.random.rand() * 0.1
        if eps <= 0.0:
            continue

        # ---- apply the update to the convolution (in‑place) ----
        conv[j:j + n] += 2.0 * eps * a
        conv[i:i + n] -= 2.0 * eps * a
        conv[2 * i] += eps * eps
        conv[2 * j] += eps * eps
        conv[i + j] -= 2.0 * eps * eps

        new_max_b = conv.max()
        delta = new_max_b - max_b

        # ---- Metropolis acceptance test ----
        accept = (delta < 0.0) or (np.random.rand() < math.exp(-delta / (T + 1e-12)))

        if accept:
            # keep the move
            a[i] -= eps
            a[j] += eps
            max_b = new_max_b
            cur_score = 2.0 * n * max_b
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] fast anneal improve: "
                      f"n={n}, score={best_score:.6f}")
        else:
            # revert the convolution update (a unchanged)
            conv[j:j + n] -= 2.0 * eps * a
            conv[i:i + n] += 2.0 * eps * a
            conv[2 * i] -= eps * eps
            conv[2 * j] -= eps * eps
            conv[i + j] += 2.0 * eps * eps
            # a stays as before

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm runs for at most ``budget_s`` seconds
    and combines a short smoothed‑gradient descent with a long fast
    simulated‑annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 1 s safety margin
    deadline = start_time + budget_s - 1.0

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start from a previously discovered solution (if any)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # Choose a target length.  Larger n gives a lower theoretical bound,
    # but also costs more per iteration.  We aim for ≈12 k when time allows.
    remaining_time = deadline - time.time()
    if remaining_time > 200.0:
        target_n = 12000
    elif remaining_time > 100.0:
        target_n = 8000
    else:
        target_n = 5000

    # initialise a0
    if starter is not None:
        a0 = _upsample(starter, target_n)
    else:
        a0 = np.random.rand(target_n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent (quick coarse optimisation)
    # ------------------------------------------------------------------
    # Allocate roughly 10 % of the remaining budget to this phase
    now = time.time()
    gd_deadline = now + min(10.0, 0.07 * (deadline - now))
    a_opt, score_opt = _run_smooth_descent(a0, target_n, gd_deadline, start_time)

    if score_opt < best_score:
        best_score = score_opt
        best_seq = a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] after gradient: n={target_n}, "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Fast simulated‑annealing refinement (the heavy work)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_ref, score_ref = _run_fast_annealing(best_seq, anneal_deadline, start_time)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal: n={target_n}, "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list (the evaluator will clamp any out‑of‑range
    # values, but our construction stays well inside the allowed interval).
    return [float(x) for x in best_seq]
