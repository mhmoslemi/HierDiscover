def _lp_boost(f, M=None):
    n = len(f)
    if M is None:
        M = _conv_max(f)
    # build A matrix
    A = np.zeros((2*n - 1, n), dtype=np.float64)
    # for each column j, fill rows j ... j+n-1 with f[0:len]
    for j in range(n):
        rows = j + np.arange(n)
        mask = rows < (2*n - 1)
        rows_valid = rows[mask]
        A[rows_valid, j] = f[:len(rows_valid)]
    # Set up LP
    c = -np.ones(n)  # maximize sum => minimize negative sum
    bounds = [(0, None)] * n
    b_ub = np.full(2*n - 1, M, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        # fallback: skip
        return None
    if not res.success:
        return None
    g = res.x
    return g

"""
search for a low‑score step‑function (non‑negative heights).

The routine mixes three ideas:

1. **Projected soft‑max gradient descent** on the probability simplex.
   The objective 2·n·max(conv(a,a)) is approximated by a smooth
   log‑sum‑exp (soft‑max) and we use Nesterov momentum with a decreasing
   step size.  The projection onto the simplex guarantees feasibility.

2. **LP “boost” steps**.
   For the current shape f we solve a small linear program
        maximise   Σ g_i
        subject to f∗g ≤ M   (M = max(f∗f) )
                  g_i ≥ 0 .
   The LP finds a feasible direction that carries as much mass as possible
   without violating the current autocorrelation bound.  A short line‑search
   mixes f and the normalised LP solution, often lowering the max‑convolution.

3. **Progressive up‑sampling**.
   Starting from a modest length (≈1200) we repeatedly refine, up‑sample
   (linear interpolation) and again refine.  After the final length is
   decided we run a long high‑precision gradient descent.

The whole process respects the supplied time budget (default 1000 s) and
returns the best sequence (as plain Python floats) found before the deadline.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of a (assumes sum(a)=1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a))  (a is already normalised)."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)


# ----------------------------------------------------------------------
def _refine_gradient(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.6,
    eps_start: float = 0.05,
    eps_end: float = 0.001,
    beta: float = 0.9,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Accelerated soft‑max projected gradient descent on a fixed length.
    Returns (final_a, best_seen_a, best_seen_score).
    """
    n = a.shape[0]
    # FFT lengths for convolution and for the gradient w⊛a_rev
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())
    v = np.zeros_like(a)                     # momentum term
    best_a = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --------- current convolution & score ----------
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]
        max_val = float(conv.max())
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # --------- soft‑max weights ----------
        eps_rel = eps_start + (eps_end - eps_start) * (it / max_iters)
        tau = max(eps_rel * max_val, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.shape[0])

        # --------- gradient of the smooth approximation ----------
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        g = 2.0 * conv2[n - 1 : n - 1 + n]          # length‑n gradient

        # --------- Nesterov momentum + projection ----------
        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)

    return a, best_a, best_score


# ----------------------------------------------------------------------
def _lp_boost(f: np.ndarray, M: float | None = None) -> np.ndarray | None:
    """
    Solve the linear program
        maximise   Σ g_i
        subject to  f∗g ≤ M ,  g_i ≥ 0 .
    Returns the feasible vector g (not normalised) or None if the LP fails.
    """
    n = f.shape[0]
    if M is None:
        M = _conv_max(f)

    # Build the (2n‑1)×n constraint matrix A such that (A g)_k = (f∗g)_k .
    # For each column j we copy f into rows j … j+n‑1.
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        r = j + np.arange(n)
        mask = r < rows
        A[r[mask], j] = f[: np.sum(mask)]

    c = -np.ones(n)                     # minimise –Σ g_i  → maximise Σ g_i
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M, dtype=np.float64)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception:
        return None

    if not res.success:
        return None
    return res.x


def _mix_improve(f: np.ndarray, g: np.ndarray, alphas: list[float]) -> tuple[np.ndarray, float]:
    """
    Try a few convex combinations (1‑α)·f + α·g (both already sum to 1)
    and keep the best.
    """
    best_seq = f
    best_sc = _score(f)
    for a in alphas:
        if a <= 0.0:
            continue
        h = (1.0 - a) * f + a * g
        h = _project_simplex(h)          # tiny numeric drift correction
        sc = _score(h)
        if sc < best_sc:
            best_sc = sc
            best_seq = h
    return best_seq, best_sc

        _, local_best, local_score = _refine_gradient(
            cand,
            max_iters=iters,
            deadline=deadline,
            step0=0.6,
            eps_start=0.05,
            eps_end=0.001,
            beta=0.9,
        )
        if local_score < best_score:
            best_score = local_score
            best_seq = local_best.copy()
            print(f"   New global best: score={local_score:.6f}")

    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)

    # ---- 3.  progressive up‑sampling with LP boosts ----
    cur_seq = best_seq.copy()
    cur_n = cur_seq.shape[0]
    MAX_LEN = 5000           # a safe upper bound for the time budget

    while True:
        # decide next size (≈1.4× larger, capped)
        next_n = int(cur_n * 1.4)
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n <= cur_n:
            break

        # linear interpolation to the new grid
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a_up = np.interp(x_new, x_old, cur_seq)
        a_up = np.maximum(a_up, 0.0)
        s = a_up.sum()
        if s == 0.0:
            a_up = np.full(next_n, 1.0 / next_n)
        else:
            a_up = a_up / s

        # iteration budget for the new size (fewer iters for larger n)
        base_iters = 50000
        max_iters = int(base_iters * n0 / next_n) + 20000
        max_iters = max(max_iters, 20000)

        print(
            f"[{time.time() - start_time:6.2f}s] Upsample to n={next_n}, iters={max_iters}"
        )
        _, local_best, local_score = _refine_gradient(
            a_up,
            max_iters=max_iters,
            deadline=deadline,
            step0=0.55,
            eps_start=0.04,
            eps_end=0.001,
            beta=0.9,
        )
        if local_score < best_score:
            best_score = local_score
            best_seq = local_best.copy()
            print(f"   Updated global best after up‑sampling: score={local_score:.6f}")

        # ----- occasional LP boost (only for moderate sizes) -----
        if cur_n <= 2000 and time.time() < deadline - 1.0:
            M = _conv_max(local_best)
            g_raw = _lp_boost(local_best, M=M)
            if g_raw is not None:
                sum_g = g_raw.sum()
                if sum_g > 1e-12:
                    g_norm = g_raw / sum_g               # now also sums to 1
                    alphas = [0.05, 0.10, 0.20, 0.30, 0.40]
                    mixed, mixed_sc = _mix_improve(local_best, g_norm, alphas)
                    if mixed_sc < best_score:
                        best_score = mixed_sc
                        best_seq = mixed.copy()
                        print(f"   LP‑boost improvement: score={mixed_sc:.6f}")

        cur_seq = local_best
        cur_n = next_n
        if time.time() >= deadline:
            break

    # ---- 4.  final long refinement on the best length found ----
    if time.time() < deadline:
        n_best = best_seq.shape[0]
        final_iters = 300000
        print(
            f"[{time.time() - start_time:6.2f}s] Final refinement on n={n_best}, iters={final_iters}"
        )
        _, final_best, final_score = _refine_gradient(
            best_seq,
            max_iters=final_iters,
            deadline=deadline,
            step0=0.45,
            eps_start=0.02,
            eps_end=0.0003,
            beta=0.9,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_best.copy()
            print(f"   Final refinement improvement: score={final_score:.6f}")

    # ---- 5.  tiny random hill‑climbing moves if time still remains ----
    while time.time() < deadline - 0.5:
        a = best_seq.copy()
        n = a.shape[0]
        i, j = np.random.choice(n, 2, replace=False)
        # transfer a small amount from i to j
        delta = min(a[i], 0.01) * np.random.rand()
        a[i] -= delta
        a[j] += delta
        a = _project_simplex(a)
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a
            print(f"[{time.time() - start_time:6.2f}s] Random perturb improvement: score={sc:.6f}")
        # sanity break to avoid endless loop when the budget is almost exhausted
        if time.time() - start_time > budget_s - 5:
            break

    # ---- 6.  tidy up and return ----
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total

    print(
        f"[{time.time() - start_time:6.2f}s] Finished. Best score (2·n·max(conv)) = {best_score:.6f}"
    )
    return [float(x) for x in best_seq]
