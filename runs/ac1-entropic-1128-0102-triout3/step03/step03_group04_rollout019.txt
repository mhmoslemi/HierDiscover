def adam_optimize(a0, n, deadline, max_iters=200000, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        conv = conv_fft(a)
        max_idx = conv.argmax()
        grad = gradient_of_peak(a, max_idx)
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # clip step to avoid large jumps
        step = np.clip(step, -a*0.5, a*0.5)
        a_new = a - step
        a_new = project_onto_simplex(a_new)
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            print(f"[adam] iter {t:07d} new best {best_score:.6f}")
        # Accept if improvement else keep old a (projected onto simplex)
        if score_new <= compute_score(a):
            a = a_new
        # else possibly accept with small probability? we can skip for simplicity.
    return best_a, best_score
```

def simulated_annealing(a0, n, deadline, max_iters=200000, init_temp=1e-2, max_move_ratio=0.02):
    a = a0.copy()
    score = compute_score(a)
    best_a = a.copy()
    best_score = score
    t = 0
    while time.time() < deadline and t < max_iters:
        temp = init_temp * (0.9995 ** t)
        # select two indices
        i = rng.integers(n)
        # ensure we can move from i
        if a[i] <= 1e-12:
            t += 1
            continue
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # random transfer amount
        delta = a[i] * rng.random() * max_move_ratio
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # no need project as sum same and no negative
        # compute new score
        new_score = compute_score(a_new)
        diff = new_score - score
        if diff < 0 or rng.random() < np.exp(-diff / max(temp, 1e-12)):
            a = a_new
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[anneal] iter {t:07d} new best {best_score:.6f}")
        t += 1
    return best_a, best_score
```

def adam_optimize(a0, n, deadline, rng, max_iters=100000, lr=0.05):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    t = 0
    while time.time() < deadline and t < max_iters:
        conv = conv_fft(a)
        max_idx = conv.argmax()
        grad = gradient_of_peak(a, int(max_idx))
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        # Clip step to avoid too large moves that would cause artifacts
        step = np.clip(step, -a * 0.5, a * 0.5)
        a_new = a - step
        a_new = project_onto_simplex(a_new)
        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[adam] iter {t:07d} new best {best_score:.6f}")
        # Accept if better, else keep old a (no metropolis)
        if new_score <= cur_score:
            a = a_new
            cur_score = new_score
        else:
            # small chance to accept worse if temperature positive (optional)
            if rng.random() < 0.001:
                a = a_new
                cur_score = new_score
        # optional jitter
        if rng.random() < 0.005:
            jitter = (rng.random(n) - 0.5) * 1e-6
            a = a + jitter
            a = project_onto_simplex(a)
    return best_a, best_score

def simulated_annealing(a0, n, deadline, rng, max_iters=200000, init_temp=0.01, max_move_ratio=0.02):
    a = a0.copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        temp = init_temp * (0.9995 ** t)
        i = rng.integers(n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * max_move_ratio
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # a_new sum unchanged, nonnegatives guaranteed
        new_score = compute_score(a_new)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / max(temp, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[anneal] iter {t:07d} new best {best_score:.6f}")
    return best_a, best_score

"""
Propose a step‑function that minimises

        score = 2·n·max(conv(a,a)) / (sum(a)²)

using three stages:

1.  **Broad grid sweep** – generate monotone and symmetric power‑law
    families a_i ∝ (i+offset)^(‑p) (or mirrored) for many (p,offset) pairs,
    keep the best.

2.  **Multi‑restart Adam optimisation** – starting from the best grid point
    and from random Dirichlet seeds, run projected Adam on the sub‑gradient
    of the maximal convolution entry (exact gradient of the active peak).
    The optimizer respects the probability simplex at every step.

3.  **Fine‑grained simulated annealing** – starting from the best Adam
    result, perform a simple Metropolis walk that moves a tiny amount of mass
    between two coordinates; accepts worsening moves with a decaying temperature.
    This helps to escape shallow basins.

All heavy work is done with FFT‑based convolutions, the routine respects a
~950 s internal time budget and prints progress.  The final candidate is
checked with the provided ``evaluate_sequence`` before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def powerlaw_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Monotone decreasing a_i = (i+offset)^(‑p), normalised to unit L1‑norm."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def symmetric_powerlaw_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Symmetric version: a_i = (min(i,n‑1‑i)+offset)^(‑p), L1‑normalised."""
    i = np.arange(n, dtype=np.float64)
    j = np.minimum(i, n - 1 - i)
    a = (j + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full self‑convolution x*x using a real FFT (length 2·len(x)-1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (Σw=1, w≥0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Exact sub‑gradient of the convolution entry at peak_idx w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)

    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for evaluate_sequence when sum(a)=1:
    score = 2·n·max(conv(a,a)) .
    """
    n = a.shape[0]
    conv = conv_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray, n: int, deadline: float,
                  rng: np.random.Generator,
                  max_iters: int = 200_000,
                  lr: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the sub‑gradient of the peak convolution entry.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    cur_score = compute_score(a)
    best_a, best_score = a.copy(), cur_score

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = conv_fft(a)
        peak = int(conv.argmax())
        grad = gradient_of_peak(a, peak)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Prevent absurd overshoots
        step = np.clip(step, -a * 0.5, a * 0.5)

        a_new = a - step
        a_new = project_onto_simplex(a_new)

        new_score = compute_score(a_new)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[adam] iter {t:07d} new best {best_score:.6f}")

        # Accept only improvements (optional tiny Metropolis hook)
        if new_score <= cur_score or rng.random() < 0.001:
            a = a_new
            cur_score = new_score

        # Occasional jitter to shake out of plateaus
        if rng.random() < 0.004:
            jitter = (rng.random(n) - 0.5) * 1e-6
            a = project_onto_simplex(a + jitter)

    return best_a, best_score


def simulated_annealing(a0: np.ndarray, n: int, deadline: float,
                        rng: np.random.Generator,
                        max_iters: int = 300_000,
                        init_temp: float = 0.02,
                        max_move_ratio: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis walk that shifts a tiny amount of mass between two
    coordinates while respecting the simplex.
    """
    a = a0.copy()
    cur_score = compute_score(a)
    best_a, best_score = a.copy(), cur_score

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        temp = init_temp * (0.9995 ** t)

        i = rng.integers(n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * max_move_ratio
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta   # sum is preserved, non‑negativity guaranteed

        new_score = compute_score(a_new)
        diff = new_score - cur_score

        if diff < 0 or rng.random() < np.exp(-diff / max(temp, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[anneal] iter {t:07d} new best {best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score.  The routine respects the supplied time budget (≈ 950 s) and
    returns the best list it discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size.  If a previous construction is present, reuse
    # its length; otherwise fall back to the classic n = 2500.
    # ------------------------------------------------------------------
    global_init = globals().get('height_sequence_1')
    if isinstance(global_init, list) and len(global_init) > 0:
        n = len(global_init)
    else:
        n = 2500

    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – coarse grid over monotone and symmetric power‑law families
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    p_grid = np.arange(0.30, 0.81, 0.025)
    offsets = [0.3, 0.5, 1.0, 2.0]

    print("[search] phase 0 – grid sweep (monotone & symmetric)")

    for symmetric in (False, True):
        for offset in offsets:
            for p in p_grid:
                if time.time() > deadline:
                    break
                if symmetric:
                    a = symmetric_powerlaw_sequence(n, p, offset=offset)
                else:
                    a = powerlaw_sequence(n, p, offset=offset)
                cur = compute_score(a)
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    print(f"[grid] new best sym={symmetric} p={p:.3f} off={offset} "
                          f"score={cur:.6f}")

    # ------------------------------------------------------------------
    # If a previously found construction is available, compare it now.
    # ------------------------------------------------------------------
    if isinstance(global_init, list) and len(global_init) == n:
        a = np.asarray(global_init, dtype=np.float64)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total < 1e-12:
            a[:] = 1.0 / n
        else:
            a /= total
        cur = compute_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[init] using supplied height_sequence_1 score={cur:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation with a few restarts
    # ------------------------------------------------------------------
    print("[search] phase 1 – Adam optimisation")
    global_best_a = best_a.copy()
    global_best_score = best_score

    max_restarts = 6
    for restart in range(max_restarts):
        if time.time() > deadline:
            break

        if restart == 0:
            a_start = best_a.copy()
        else:
            # Random Dirichlet-ish start (Gamma(0.5) → a bit sparse)
            a_start = rng.gamma(0.5, 1.0, size=n)
            a_start /= a_start.sum()

        # give each restart roughly a fixed slice of the remaining budget
        remaining = deadline - time.time()
        phase_deadline = time.time() + min(120.0, remaining * 0.5)

        a_opt, score_opt = adam_optimize(a_start, n, phase_deadline, rng,
                                         max_iters=200_000, lr=0.07)

        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_a = a_opt.copy()
            print(f"[restart {restart}] improved global best to {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – fine‑grained simulated annealing starting from the best Adam result
    # ------------------------------------------------------------------
    print("[search] phase 2 – simulated annealing fine‑tuning")
    if time.time() < deadline:
        a_opt, score_opt = simulated_annealing(global_best_a, n, deadline, rng,
                                               max_iters=300_000,
                                               init_temp=0.02,
                                               max_move_ratio=0.02)
        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_a = a_opt.copy()
            print(f"[anneal] final global best {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator and return the list
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")

    return global_best_a.tolist()
