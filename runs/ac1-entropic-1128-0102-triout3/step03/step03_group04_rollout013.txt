"""
candidate_proposer.py

A fast, deterministic search for a non‑negative step‑function sequence that
minimises the evaluation score

    2·n·max(conv(a,a)) / (sum(a)**2)

(the score is scale‑invariant, so we keep sum(a)=1).

The algorithm consists of four stages:

1.  **Initialisation** – if a global variable ``height_sequence_1`` is
    present, it is used as a warm start.  The best score seen so far is
    stored.

2.  **Parametric sweeps** – a coarse grid of power‑law sequences
    ``a_i ∝ (i+offset)^(-p)`` (n=2500) and a family of exponential decays
    ``a_i∝exp(-γ i)`` are evaluated.  The best parameters are remembered.
    A fine sweep around the best ``p`` and ``offset`` follows.

3.  **Mirror‑descent optimisation** – a soft‑max approximation of the
    maximum convolution entry is constructed:

        w_k ∝ exp(β·(conv_k – max(conv)))

    The gradient of the smoothed objective is the cross‑correlation
    ``grad = 2·(a ⋆ w)`` and the update is performed by exponentiated
    gradient (mirror descent on the probability simplex):

        log a ← log a – η·grad
        a ← softmax(log a)

    The step size ``η`` decays, the temperature ``β`` grows, and occasional
    jitter helps escaping plateaus.

4.  **Result** – the best vector found during the whole run is returned
    (as a plain ``list[float]``) after a final sanity check with the
    official ``evaluate_sequence`` routine.

The whole search respects a user‑supplied time budget (default 950 s,
leaving a safety margin) and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Return a length‑n non‑negative vector a_i = (i+offset)^(-p),
    normalised to L1‑norm 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a real vector via FFT."""
    n = x.shape[0]
    m = 2 * n - 1                       # length of the linear convolution
    N = 1 << ((m - 1).bit_length())     # next power of two
    X = np.fft.rfft(x, n=N)
    conv = np.fft.irfft(X * X, n=N)[:m]
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights for a convolution vector (numerically stable)."""
    maxv = conv.max()
    w = np.exp(beta * (conv - maxv))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    w /= s
    return w


def gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed objective  f(a)= (1/β)·log∑exp(β·conv_k)
    with respect to a (under the constraint sum(a)=1).
    The gradient equals 2·(a ⋆ w) where w are the soft‑max weights.
    """
    n = a.shape[0]
    w = softmax_weights(conv, beta)          # length 2n‑1
    # cross‑correlation a ⋆ w = a convolved with reversed w
    w_rev = w[::-1]
    m = w.shape[0]                           # =2n‑1
    N = 1 << ((n + m - 1).bit_length())      # power‑of‑two for linear convolution
    A = np.fft.rfft(a, n=N)
    W = np.fft.rfft(w_rev, n=N)
    corr = np.fft.irfft(A * W, n=N)           # linear convolution a * w_rev
    base = m - 1                              # index that aligns with a[0]
    grad = 2.0 * corr[base - np.arange(n)]
    return grad


def internal_score(a: np.ndarray) -> float:
    """Fast proxy for the official score when sum(a)==1."""
    conv = conv_fft(a)
    return 2.0 * a.shape[0] * conv.max()

        sc = internal_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_p, best_offset = None, None
            print(f"[exp] new best γ={gamma:.4f}, score={sc:.6f}")

    # -------------------------------------------------- phase 3 : fine power‑law refinement
    if best_p is not None and best_offset is not None:
        p_fine = np.arange(max(0.01, best_p - 0.02),
                           min(0.99, best_p + 0.02) + 1e-6, 0.001)
        off_fine = np.arange(max(0.0, best_offset - 0.2),
                             best_offset + 0.2 + 1e-6, 0.01)
        print("[search] phase 3 – fine power‑law refinement")
        for off in off_fine:
            for p in p_fine:
                if time.time() > deadline:
                    break
                a = power_law_sequence(n, p, offset=off)
                sc = internal_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    best_p, best_offset = p, off
                    print(f"[fine] new best p={p:.4f}, off={off:.2f}, score={sc:.6f}")

    # -------------------------------------------------- phase 4 : mirror‑descent optimisation
    if best_a is None:
        # fall back to uniform distribution
        best_a = np.full(n, 1.0 / n)
        best_score = internal_score(best_a)

    global_best_a = best_a.copy()
    global_best_score = best_score

    # hyper‑parameters for the mirror‑descent loop
    beta = 0.5                # start temperature for soft‑max
    beta_inc = 0.0005         # increase per iteration
    eta = 0.12                # step size (log‑space)
    decay_factor = 0.99
    decay_every = 4000        # decay interval
    max_iters = 500_000       # generous upper bound, truncated by time

    a = best_a.copy()
    it = 0
    print("[optim] phase 4 – mirror descent (soft‑max gradient)")
    while time.time() < deadline and it < max_iters:
        conv = conv_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < global_best_score:
            global_best_score = cur_score
            global_best_a = a.copy()
            print(f"[optim] iter {it:07d}  new global best = {cur_score:.6f}")

        #--- gradient (soft‑max of all convolution entries) -----------------
        grad = gradient_softmax(a, conv, beta)

        #--- exponentiated / mirror update ----------------------------------
        eps = 1e-30
        log_a = np.log(a + eps)
        # clip gradient to avoid overflow in the exponent
        grad_clip = np.clip(grad, -20.0, 20.0)
        log_a -= eta * grad_clip
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        #--- schedule --------------------------------------------------------
        if (it + 1) % decay_every == 0:
            eta = max(eta * decay_factor, 1e-6)
            beta += beta_inc

        # occasional tiny jitter to break possible cycles
        if rng.random() < 0.001:
            jitter = (rng.random(n) - 0.5) * 1e-6
            a = np.clip(a + jitter, 0.0, None)
            a /= a.sum()

        it += 1

    # -------------------------------------------------- final sanity check
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")
    return global_best_a.tolist()
