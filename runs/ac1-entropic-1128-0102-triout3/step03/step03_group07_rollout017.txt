"""
Propose a non‑negative step‑function (probability distribution) that
minimises the evaluation function

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear auto‑convolution.  The algorithm

* works directly on the probability simplex (∑a_i = 1, a_i ≥ 0);
* uses a smooth‐max (log‑sum‑exp) surrogate of the peak together with
  an Adam‑style adaptive learning rate;
* keeps the iterate on the simplex by Euclidean projection;
* starts from several diverse initialisations (half‑binomial,
  exponential, double‑half‑binomial, Dirichlet, uniform);
* periodically injects small random perturbations and performs full
  restarts when no improvement has been seen for long;
* after the main optimisation phase scans a few one‑parameter
  families (power‑law, exponential) to try to improve the best
  solution found.

The routine respects the 1000 s budget, prints progress, and returns
the best sequence (as a plain Python list of floats) discovered.
"""

import time
import numpy as np
from typing import List

def half_binomial_series(n: int) -> np.ndarray:
    """First `n` coefficients of (1‑z)^‑½  (a_i = C(2i,i)/4^i) normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def exponential_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Truncated exponential distribution a_i ∝ r**i with r close to 1."""
    # draw r in (0.99, 0.9999) – the closer to 1 the flatter the tail
    r = 0.99 + 0.0099 * rng.random()
    a = r ** np.arange(n, dtype=np.float64)
    a = a / a.sum()
    return a


def double_half_binomial(n: int) -> np.ndarray:
    """Symmetric “U‑shaped’’ distribution: two half‑binomials glued together."""
    half = half_binomial_series((n + 1) // 2)
    # mirror and truncate to exact length n
    if n % 2 == 0:
        a = np.concatenate([half, half[::-1]])
    else:
        a = np.concatenate([half, half[-2::-1]])
    a = a[:n]
    a = a / a.sum()
    return a


def powerlaw_initial(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+½)^{-p} (p≈0.5 for the half‑binomial shape)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = idx ** (-p)
    a = a / a.sum()
    return a

    # main optimisation loop
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) current convolution and objective (soft‑max works on the same conv)
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[:L]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # 2) record best solution
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:09d}] new best {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) soft‑max (log‑sum‑exp) weights – numerically stable
        shift = max_conv
        w_exp = np.exp(beta * (conv - shift))
        w = w_exp / w_exp.sum()                     # shape (L,)

        # gradient of the smoothed objective w.r.t. a
        #   grad = 2 * correlation(a, w)
        Wr = np.fft.rfft(w[::-1], n=Nfft)           # FFT of reversed weights
        corr_full = np.fft.irfft(A * Wr, n=Nfft)
        start = L - 1                               # index where shift == 0 starts
        grad = 2.0 * corr_full[start:start + n]

        # --------------------------------------------------------------
        # 4) Adam update on the *additive* simplex representation
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # limit the step size so we never push an entry below zero in one move
        max_step = a * 0.9
        step = np.clip(step, -max_step, max_step)

        # simple projected‑gradient step
        a_tmp = a - step
        a = project_onto_simplex(a_tmp)

        # --------------------------------------------------------------
        # 5) step‑size & smoothing schedule
        eta = max(eta * eta_decay, 5e-5)
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # --------------------------------------------------------------
        # 6) occasional random perturbation (helps escaping flat regions)
        if rng.random() < perturb_prob:
            delta = (rng.random(n) - 0.5) * perturb_scale
            a = a + delta
            a = project_onto_simplex(a)

        # --------------------------------------------------------------
        # 7) restart if stagnated for a long time
        if it - last_improve > restart_gap:
            a = random_start()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"[iter {it:09d}] restart – fresh start, score≈{scaled_objective(a):.6f}")

        it += 1

    # ------------------------------------------------------------------
    # post‑optimisation scan of a few simple parametric families
    # (cheap – a few hundred extra evaluations)

    # power‑law sweep around the half‑binomial exponent p≈0.5
    for p in np.linspace(0.35, 0.65, 21):
        cand = powerlaw_initial(n, p)
        sc = scaled_objective(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[grid] power‑law p={p:.3f} improved to {best_score:.6f}")

    # exponential sweep (different decay rates)
    for r in np.linspace(0.985, 0.9995, 20):
        a_tmp = r ** np.arange(n, dtype=np.float64)
        a_tmp = a_tmp / a_tmp.sum()
        sc = scaled_objective(a_tmp)
        if sc < best_score:
            best_score = sc
            best_a = a_tmp.copy()
            print(f"[grid] exponential r={r:.4f} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # final safety evaluation using the provided routine
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
