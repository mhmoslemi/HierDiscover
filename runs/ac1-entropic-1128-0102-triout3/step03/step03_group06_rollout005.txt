def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1
    best_score = float('inf')
    best_seq = None
    best_n = None

    # Parameter grid
    ns = [400, 500, 600, 800, 1000, 1200]   # candidate length
    ratios = [0.35, 0.4, 0.45, 0.5] # k/n ratio
    attempts = 0
    while time.time() < deadline:
        # randomly select n and ratio
        n = np.random.choice(ns)
        ratio = np.random.choice(ratios)
        k = int(ratio * n)
        if k < 2:
            continue
        # sample support
        S = np.random.choice(n, size=k, replace=False)
        counts = compute_counts(S, n)   # int counts
        max_c = int(counts.max())
        score = 2.0 * n * max_c / (k * k)
        if score < best_score:
            best_score = score
            # build full sequence
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / k
            best_seq = a.tolist()
            best_n = n
            print(f"New best: score={best_score:.6f}, n={n}, k={k}, max_c={max_c} at attempt {attempts}")
            # early exit if under target
            if best_score <= 1.503:
                print("Target achieved.")
                break
        attempts += 1

    # If after loop best_seq is None, fallback to uniform distribution maybe
    if best_seq is None:
        n = 500
        best_seq = np.full(n, 1.0/n).tolist()
        best_score = fast_score(np.array(best_seq))
        print(f"Fallback uniform, score {best_score}")

    # Return best sequence
    return best_seq

"""
Random-sparse subset search for a step function.
The algorithm repeatedly samples a random support set S of size k in a

"""
Random‑sparse set search.

The evaluation function depends only on the largest ordered‑pair count
max_c of the indices in the support set S (the self‑convolution of a
uniform distribution on S).  For a uniform weight 1/|S| we have

    score = 2·n·max_c / |S|² .

For a random set S the counts are roughly uniformly spread, so the
expected score is close to 1 – well below the target 1.503.
The algorithm repeatedly draws a random support S (size k) inside a
vector of length n, computes max_c via an O(k²) outer‑sum, evaluates the
score and keeps the best sequence found.  The search runs until the
time budget (≈1000 s) expires or a sequence with score ≤ 1.503 is
discovered.  If nothing is found, a simple uniform distribution is
returned as a safe fallback.
"""

import time
import math
import numpy as np


def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """
    Ordered‑pair count for each possible convolution index.

    Parameters
    ----------
    S : np.ndarray
        1‑D array of distinct indices (the support set).
    n : int
        Length of the step function.

    Returns
    -------
    np.ndarray
        Integer array of length 2·n‑1 where entry k holds the number of
        ordered pairs (i, j)∈S×S with i + j = k.
    """
    # outer sum creates every i+j (ordered, because both (i,j) and (j,i)
    # are present).  np.bincount tallies the frequencies.
    sums = np.add.outer(S, S)
    counts = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return counts


def fast_score(a: np.ndarray) -> float:
    """
    Quick evaluation of the provided sequence using FFT.
    This is only used as a fallback; the main search evaluates the score
    analytically from max_c.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with evaluation score ≤ 1.503.

    The procedure randomly samples support sets of various sizes and
    lengths, evaluates the score analytically, and keeps the best
    candidate seen before the time limit expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1        # safety margin

    # Parameters to explore – lengths and density ratios.
    n_candidates = [400, 500, 600, 800, 1000, 1200]
    ratio_candidates = [0.25, 0.30, 0.35, 0.40, 0.45]

    best_val = float('inf')
    best_seq = None
    attempts = 0

    # Main random‑search loop
    while time.time() < deadline:
        # pick a random problem size
        n = int(np.random.choice(n_candidates))
        ratio = float(np.random.choice(ratio_candidates))
        k = max(2, int(ratio * n))          # support size (must be ≥2)

        # draw a random support without replacement
        S = np.random.choice(n, size=k, replace=False).astype(np.int32)

        # compute ordered‑pair counts and the maximum
        cnts = compute_counts(S, n)
        max_c = int(cnts.max())

        # analytic score for the uniform‑weight vector on S
        score = 2.0 * n * max_c / (k * k)

        # keep the best one encountered
        if score < best_val:
            best_val = score
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / k
            best_seq = a.tolist()
            attempts += 1
            print(f"[{time.time() - start:6.2f}s] new best: "
                  f"score={best_val:.6f}, n={n}, k={k}, max_c={max_c}, attempts={attempts}")

            # stop early if we already satisfy the target bound
            if best_val <= 1.503:
                print("[info] target achieved – exiting search.")
                break
        else:
            attempts += 1

    # ------------------------------------------------------------------
    # 2. Fallback – if nothing succeeded (extremely unlikely)
    if best_seq is None:
        print("[fallback] no candidate found, using uniform distribution.")
        n = 500
        best_seq = (np.full(n, 1.0 / n, dtype=np.float64)).tolist()
        best_val = fast_score(np.array(best_seq))
        print(f"[fallback] uniform score = {best_val:.6f}")

    print(f"[final] best score returned: {best_val:.6f}")
    return best_seq
