def max_subgradient(a, conv_max_idx):
    n = len(a)
    k = conv_max_idx
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi+1, dtype=int)
    jdx = k - idx
    grad[idx] = 2.0 * a[jdx]
    return grad

"""
Propose a candidate sequence that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

where `conv` is the linear auto‑convolution.  The objective is invariant
under scaling, so we restrict the search to the probability simplex
(∑a_i = 1, a_i ≥ 0).  The algorithm is a hybrid of:

* a smoothed ‑max optimisation (log‑sum‑exp) with a gradually
  increasing inverse temperature β,
* an Adam‑style momentum update on the Euclidean simplex,
* projection onto the simplex after each update,
* occasional Gaussian perturbations and long‑run restarts.

The code is deliberately self‑contained, uses only NumPy, and prints
progress information.  It runs for at most ``budget_s`` seconds and
returns the best sequence (as a plain Python list) found within the
budget.
"""

import time
import numpy as np
from typing import List

# Helper utilities – all top‑level, no closures

def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of the series (1‑z)^‑½,
    i.e. a_i = C(2i,i) / 4^i, normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def linear_convolution_fft(a: np.ndarray, Nfft: int, L: int):
    """
    Compute the linear auto‑convolution a*a using FFT.
    Returns the full linear convolution (length L = 2·n‑1) and the max value.
    """
    A = np.fft.rfft(a, n=Nfft)
    conv_full = np.fft.irfft(A * A, n=Nfft)
    conv = conv_full[:L]
    max_conv = conv.max()
    return conv, max_conv, A


def softmax_gradient(A: np.ndarray, conv: np.ndarray, max_conv: float,
                     beta: float, Nfft: int, L: int, n: int) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) objective.
    """
    # stable soft‑max on the convolution values
    shift = max_conv
    w_exp = np.exp(beta * (conv - shift))
    w = w_exp / w_exp.sum()                      # shape (L,)

    # correlation of a with reversed weights
    Wr = np.fft.rfft(w[::-1], n=Nfft)
    corr_full = np.fft.irfft(A * Wr, n=Nfft)
    start = L - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient when the exact max is used.
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = k - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[jdx]
    return grad
