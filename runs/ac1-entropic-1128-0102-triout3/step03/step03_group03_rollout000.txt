def compute_score(seq):
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    # clip negative to 0 not needed; assume positive
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    # compute convolution via FFT
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af*Af, n=fft_len)[:2*n-1]
    max_val = conv.max()
    return float(2.0 * n * max_val / (sum_a*sum_a))

    start=time.time()

    best_seq=None; best_score=np.inf

    # Stage 1: Parameter shape search for a set of lengths
    for n in [800, 1500, 3000, 5000]:
        if time.time()>=deadline: break
        for alpha in np.linspace(0.3, 4.0, 20):
            seq = shape_sine(n, alpha)
            score = compute_score(seq)
            if score<best_score:

    # Stage 2: anisotropic gradient descent (mirror descent) with upsampling &

"""
Hybrid search for a low‑peak non‑negative step function.

The algorithm consists of three phases:

1. **Parametric sweep** – try simple analytic families (half‑sine raised to a
   power) for several lengths and exponents.  These give good starting points.

2. **Mirror‑descent (exponentiated gradient)** – on each length we run many
   EG steps using an *active‑set* sub‑gradient of the max‑convolution.  The
   active set is larger than in the previous baseline (up to 20 indices) and
   the step‑size decays as 1/√t.  After a block we up‑sample the current best
   shape (linear interpolation) and continue on a larger grid, up to 5000
   points.

3. **Simulated‑annealing local search** – starting from the best sequence
   found, repeatedly move a random fraction of mass from one index to another,
   accepting worsening moves with a temperature that decays exponentially.
   This helps escape shallow local minima.

All intermediate improvements are printed.  The returned sequence is a
plain list of Python floats; it is non‑negative, and the evaluation objective
is 2·n·max(conv(a,a)) / (sum(a)²) (the same objective used by
`evaluate_sequence`).  The search respects the 1000‑second time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_score(seq: np.ndarray) -> float:
    """
    Return the objective value 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The sequence may contain any non‑negative numbers; the function does
    not apply the clipping performed by ``evaluate_sequence``.
    """
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    if n == 0:
        return np.inf
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    # FFT length as next power of two >= 2*n-1
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()
    return float(2.0 * n * max_val / (sum_a * sum_a))


def conv_and_max(a: np.ndarray):
    """Return the full autocorrelation (via FFT) and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def active_indices(conv: np.ndarray, max_val: float,
                   rel_eps: float = 0.01, max_add: int = 20) -> np.ndarray:
    """
    Return a modest set of convolution indices that are near the maximum.
    All indices with value >= (1‑rel_eps)·max_val are selected; if this set
    is smaller than ``max_add`` we also add the globally largest entries.
    """
    eps = max_val * rel_eps
    idx_near = np.where(conv >= max_val - eps)[0]
    if idx_near.size < max_add:
        # add the top‑max_add entries (unique)
        topk = min(max_add, conv.size)
        idx_near = np.unique(np.argpartition(conv, -topk)[-topk:])
    elif idx_near.size > max_add:
        # truncate to max_add by keeping the largest among them
        vals = conv[idx_near]
        cut = np.argpartition(vals, -max_add)[-max_add:]
        idx_near = idx_near[cut]
    return idx_near


def gradient_active(a: np.ndarray, K: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv(a,a))`` w.r.t. the active indices K.
    For each active convolution index k the contribution to component i is
    2·a_{k-i} (if k‑i is a valid index).  The returned vector has the same
    length as ``a``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        # accumulate 2·a[j_idx] into g[i_idx]
        g[i_idx] += 2.0 * a[j_idx]
    return g


def shape_sine(n: int, alpha: float) -> np.ndarray:
    """
    Generate a half‑sine shape raised to the power ``alpha`` and normalised
    to unit sum: a_i ∝ sin(π·(i+0.5)/n)^α.
    """
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** alpha
    # Numerical safety – values are non‑negative by construction
    total = vals.sum()
    if total <= 0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) with a low
    ``evaluate_sequence`` score.  The routine runs for at most ``budget_s``
    seconds and prints intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # safety margin

    # ------------------------------------------------------------------
    # 0. Utility to keep the global best
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    def maybe_update(candidate: np.ndarray, label: str):
        """Update global best if candidate improves."""
        nonlocal best_seq, best_score
        score = compute_score(candidate)
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"[{time.time() - start_time:6.2f}s] {label} -> n={len(candidate)} "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # 1. Try a parametric family (half‑sine) for several lengths & exponents
    # ------------------------------------------------------------------
    lengths = [800, 1500, 3000, 5000]
    alphas = np.linspace(0.3, 4.5, 22)   # 22 values gives decent granularity
    for n in lengths:
        if time.time() >= deadline:
            break
        for alpha in alphas:
            seq = shape_sine(n, alpha)
            maybe_update(seq, f"Sine(alpha={alpha:.2f})")

    # ------------------------------------------------------------------
    # 2. If a historic sequence is available, compare it as well
    # ------------------------------------------------------------------
    try:
        hist = np.asarray(height_sequence_1, dtype=np.float64)
        hist = np.maximum(hist, 0.0)
        if hist.sum() > 0:
            hist = hist / hist.sum()
            maybe_update(hist, "Historical")
    except Exception:
        pass  # ignore if not present

    # ------------------------------------------------------------------
    # 3. Mirror‑descent (exponentiated gradient) with progressive up‑sampling
    # ------------------------------------------------------------------
    # start from the current best, otherwise a random start of moderate size
    if best_seq is not None:
        cur = best_seq.copy()
    else:
        cur = np.random.rand(2000)
    cur = np.maximum(cur, 1e-15)
    cur = cur / cur.sum()
    cur_n = cur.shape[0]

    MAX_LEN = 5000
    # We limit the number of size upgrades to keep within the time budget
    while time.time() < deadline and cur_n < MAX_LEN:
        # Determine next length (≈1.5× larger)
        next_n = int(cur_n * 1.5)
        next_n = min(next_n, MAX_LEN)
        if next_n <= cur_n:
            break

        # ----- up‑sample current shape ---------------------------------
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a = np.interp(x_new, x_old, cur)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        # ----- exponentiated‑gradient iterations -----------------------
        # iteration budget scaled with size (more iterations for smaller n)
        base_iters = 25000
        max_iter = int(base_iters * 2000 / next_n) + 4000
        max_iter = max(max_iter, 12000)  # enforce a sane minimum
        it = 0
        while time.time() < deadline and it < max_iter:
            it += 1

            # evaluate current sequence (fast FFT based)
            conv, max_val = conv_and_max(a)
            cur_score = 2.0 * next_n * max_val   # sum(a)=1 by construction
            if cur_score < best_score:
                maybe_update(a, f"EG (n={next_n})")
                # keep a copy for the up‑sampling step later
                best_local = a.copy()

            # active‑set sub‑gradient
            K = active_indices(conv, max_val, rel_eps=0.01, max_add=20)
            g = gradient_active(a, K)

            # mirror‑descent step (entropy mirror map → exponentiated update)
            eta = 0.6 / math.sqrt(it)   # diminishing step size
            a = np.maximum(a, 1e-15)
            log_a = np.log(a) - eta * g
            # numerical stabilisation
            log_a -= np.max(log_a)
            a = np.exp(log_a)
            a = np.clip(a, 1e-15, None)
            a /= a.sum()
        # advance to next size
        cur = a.copy()
        cur_n = next_n

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing / stochastic mass‑transfer local search
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        n = a.shape[0]
        cur_score = compute_score(a)

        # temperature schedule – start low enough to accept only near‑improvements
        temp = max(best_score * 0.015, 1e-6)
        decay = 0.99996
        max_local_iters = 25000
        it_local = 0
        while time.time() < deadline and it_local < max_local_iters:
            it_local += 1
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move a random fraction (up to 5 %) of mass from i to j
            eps = np.random.rand() * 0.05
            delta = eps * a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            if a_new[i] < 0:
                continue
            new_score = compute_score(a_new)
            # Metropolis acceptance
            if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / temp):
                a = a_new
                cur_score = new_score
                if new_score < best_score:
                    maybe_update(a, f"Anneal")
            temp *= decay

    # ------------------------------------------------------------------
    # 5. Fallback: uniform distribution if nothing was found (should not happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # convert to plain Python floats for the external caller
    return [float(x) for x in best_seq]
