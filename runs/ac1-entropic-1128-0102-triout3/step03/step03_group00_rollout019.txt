"""
propose_candidate
-----------------
Searches for a non‑negative step‑function (list of heights) that minimizes
the provided ``evaluate_sequence`` objective.

The method consists of three phases:

1. **Coarse optimisation** – a short projected‑Adam descent on a modest
   length (≈ 1200).  The objective 2·n·max(conv) is used (the sum of the
   sequence is kept equal to 1, so the division by ``sum_a**2`` in the
   real objective is irrelevant).

2. **Upsampling & long optimisation** – the best coarse solution is linearly
   interpolated to a larger length (≈ 5000).  A much longer projected‑Adam
   descent runs for the bulk of the remaining time budget.  The Adam step
   is combined with a simple Euclidean projection onto the probability
   simplex after each update.

3. **Local stochastic refinement** – a cheap simulated‑annealing walk that
   repeatedly transfers a small amount of mass between two random indices.
   It keeps any improvement and occasionally accepts uphill moves.

The routine respects the 1000 s time limit, prints progress messages and
finally returns the best sequence discovered (as a plain ``list`` of
floats).  The returned sequence always satisfies the non‑negativity and
sum‑≥ 0.01 requirements of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – O(n) algorithm
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x | x_i ≥ 0, ∑x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution via FFT (exact, zero‑padded)
# ----------------------------------------------------------------------
def conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (average over all active indices)
# ----------------------------------------------------------------------
def subgradient_maxconv(a: np.ndarray, conv: np.ndarray,
                        rel_tol: float = 1e-12) -> np.ndarray:
    """
    Compute the (average) sub‑gradient of max(conv) w.r.t. a.
    All indices k where conv[k] ≥ max(conv)·(1‑rel_tol) are treated as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val <= 0.0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    # For each active convolution index k we add the contributions of the
    # corresponding pairs (i, j) with i + j = k.
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i = np.arange(i_min, i_max + 1, dtype=int)
        j = k - i
        # derivative from a_i * a_j
        g[i] += a[j]
        g[j] += a[i]

    if active.size > 0:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Projected Adam optimiser on the simplex
# ----------------------------------------------------------------------
def adam_descent(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 lr: float = 0.3,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8,
                 max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected Adam descent on the probability simplex.
    Returns the best vector found (as a NumPy array) and the corresponding
    objective value 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={t}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        g = subgradient_maxconv(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # gradient descent step + projection onto simplex
        a = project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def anneal_refine(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  cur_best_score: float,
                  cur_best_seq: np.ndarray,
                  start_temp: float = 5e-2,
                  end_temp: float = 1e-5,
                  steps: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Tiny “transfer ε from i to j’’ simulated‑annealing walk.
    Returns an improved sequence (if found) and its score.
    """
    a = a0.copy()
    score = 2.0 * n * conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 10 % of the mass at i
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (simple interpolation) preserving the total mass
# ----------------------------------------------------------------------
def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a 1‑D array to a new length, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    start_time = time.time()

    best_seq = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
