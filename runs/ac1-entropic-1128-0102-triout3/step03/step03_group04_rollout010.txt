"""
Propose a step‑function that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1. **Coarse power‑law sweep** – generate candidates a_i ∝ (i+off)^{‑p}
   for a moderate length (default n=2500) and keep the best grid point.

2. **Full‑soft‑max projected gradient descent** – from the best grid
   point (and a few random Dirichlet restarts) we minimise a smooth
   approximation of the max‑convolution:
        F(a) = log‑sum‑exp(β·conv(a,a))
   The gradient of F is obtained efficiently by a single FFT‑based
   cross‑correlation.  After each step the vector is projected onto the
   probability simplex (non‑negative, unit sum).  β is increased
   gradually so that the smooth objective converges to the true
   max‑convolution.  The optimiser uses an adaptive step size and a
   tiny Metropolis‑type acceptance to escape plateaus.

3. **Optional LP refinement** – after gradient descent we solve a small
   linear programme (maximise Σg_i subject to g≥0 and (a*g)[k] ≤ M)
   to obtain a “direction” that strictly reduces the current peak M.
   The current solution is mixed with this direction and re‑projected.
   This step is inexpensive (one LP) and often yields a final drop.

The routine respects a ~950 s internal budget (leaving a safety margin)
and prints progress information.  The returned list is checked with
``evaluate_sequence`` before being output.
"""

import time
import numpy as np
from scipy.signal import correlate
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Return a non‑negative vector a_i = (i+offset)^{‑p} (i=0…n‑1)
    normalised to L1‑norm = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:          # defensive fallback
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length = 2·len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_weights(x: np.ndarray, beta: float) -> np.ndarray:
    """Return a soft‑max distribution over x with temperature 1/β."""
    max_x = x.max()
    w = np.exp(beta * (x - max_x))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.shape[0])
    return w / s


def gradient_full_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max objective  log‑sum‑exp(β·conv)
    with respect to a.
    Uses the identity   ∂conv_k/∂a = 2·a_{k‑i}
    and computes the weighted sum via a single FFT‑based correlation.
    """
    w = softmax_weights(conv, beta)                     # shape (2n‑1,)
    # weighted gradient = 2 * correlation(w, a)  (valid mode = n entries)
    grad = 2.0 * correlate(w, a, mode="valid", method="fft")
    return grad


def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A where A[k,i] = f_{k‑i} if 0≤k‑i<n else 0.
    Used for the LP refinement step.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # for each shift k we copy the overlapping part of f
    for k in range(m):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices i in [lo, hi] correspond to f_{k‑i}
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        A[k, idx] = f[k - idx]
    return A
