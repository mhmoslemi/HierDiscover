"""
Low‑score step function generator.

The algorithm tries to construct a sparse support set S of size k
and assign a uniform weight 1/k on this support.
For a random set of moderate size (k≈300‑600) the ordered
pair counts are close to their average k²/(2n), so the
evaluation score 2·n·max_count/k² is typically near 1.
We therefore attempt many random supports for a few different k
values until a sequence with a score ≤ TARGET (1.503) is found.
If the random search fails we fall back to a simple projected
sub‑gradient descent starting from the best sequence seen.
All computation respects the 1000 s time budget.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on the max‑convolution objective.
    The implementation follows the one used previously, but a few
    parameters are slightly tuned.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_sz = 1 << ((2 * n - 1).bit_length())

    eta0 = 0.35          # initial step size (a bit larger)
    eta_decay = 1e-5
    random_move_prob = 0.03
    random_move_step = 0.20
    it = 0
    deadline = time.time() + time_limit - 0.2

    while time.time() < deadline:
        # convolution and index of the active (maximum) entry
        fft_a = np.fft.rfft(a, conv_sz)
        conv = np.fft.irfft(fft_a * fft_a, conv_sz)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size and Euclidean projection onto the simplex
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # projection onto the probability simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        rho = ind[cond][-1] if np.any(cond) else 0
        theta = cssv[cond][-1] / rho if rho > 0 else 0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer (helps escaping plateaus)
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_step * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # re‑project
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                rho = ind[cond][-1] if np.any(cond) else 0
                theta = cssv[cond][-1] / rho if rho > 0 else 0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # keep track of the best seen so far
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it:6d}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with evaluation score ≤ 1.503.
    Primary method: random uniform support sets of moderate size.
    If the random search does not meet the target, a fallback
    projected sub‑gradient optimiser is invoked.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    target_score = 1.503
    n = 1500                               # length of the step function

    # ------------------------------------------------------------------
    # 1) Random sparse‑support search
    support_sizes = [600, 550, 500, 450, 400, 350,
                    300, 250, 200, 150, 120, 100]

    best_seq = None
    best_val = np.inf

    for k in support_sizes:
        # quick estimate of the maximal count that would still satisfy the target
        allowed = int(math.floor(target_score * k * k / (2.0 * n)))
        print(f"[search] trying support size k={k} (allowed max count ≈ {allowed})")
        attempts = 0
        # give each size a modest time slice (≈20 s) but respect the global deadline
        local_deadline = min(deadline, time.time() + 20.0)

        while time.time() < local_deadline and attempts < 5000:
            attempts += 1
            # random subset without replacement
            S = np.random.choice(n, size=k, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / k

            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[improve] k={k} attempt={attempts} score={val:.6f}")

            if val <= target_score:
                print("[success] target reached, returning sequence.")
                return best_seq.tolist()

        print(f"[search] finished k={k} after {attempts} attempts, "
              f"best so far {best_val:.6f}")

        if best_seq is not None and best_val <= target_score:
            break   # no need to continue with smaller supports

    # ------------------------------------------------------------------
    # 2) If the random search failed, fall back to sub‑gradient descent.
    if best_seq is None:
        print("[fallback] no sequence found, initializing from uniform distribution.")
        init = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        init = best_seq.copy()

    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[fallback] running sub‑gradient optimizer for {remaining:.1f}s")
        best_seq = subgradient_optimize(init, remaining)
        best_val = fast_score(best_seq)

    # ------------------------------------------------------------------
    # 3) Ensure we have a valid sequence to return.
    if best_seq is None:
        print("[ultimate fallback] using uniform distribution.")
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    print(f"[final] best score returned: {best_val:.6f}")
    return best_seq.tolist()
