"""
Optimise the evaluation score of a non‑negative step function.

The algorithm proceeds in two stages:

1. **Combinatorial set search** –  For several support sizes `k`
   we look for a set `S ⊂ {0,…,n‑1}` with a small maximal
   ordered‑pair count `max_k count(k)`.  The score for a uniform
   distribution on `S` is

        score = 2·n·max_count / k² .

   A random initialise is refined by a simple stochastic hill‑climbing
   that repeatedly swaps one element of the current support with a
   non‑support element.  Swaps are accepted only if they do not increase
   the current maximum count (occasional neutral moves are allowed to
   escape plateaus).  The counts are updated incrementally, i.e. in
   O(k) time per swap, using the fact that each ordered pair contributes
   2 to the convolution count (1 for a self‑pair).

   If a configuration meets the target bound `score ≤ 1.503` the
   corresponding sequence (uniform mass `1/k` on the support) is returned
   immediately.

2. **Fallback / polishing** –  If the combinatorial phase fails to reach
   the target within the time budget, we fall back to a short projected
   sub‑gradient descent (the same routine used in the previous
   implementation) starting from the best uniform sequence found so
   far.  The optimiser works on the continuous weights while keeping the
   simplex constraint `∑ a_i = 1, a_i ≥ 0`.

The whole search is bounded by the provided `budget_s` (default 1000 s)
and always returns a valid list of floats.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Fast score (identical to evaluate_sequence for a normalised vector)
def fast_score(a: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative vector a (sum‑1)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def compute_counts_np(support: list[int], n: int) -> np.ndarray:
    """
    Ordered‑pair count for a binary support set.
    Returns an integer array of length 2·n‑1 containing the counts.
    """
    if not support:
        return np.zeros(2 * n - 1, dtype=np.int32)
    arr = np.array(support, dtype=np.int32)
    sums = np.add.outer(arr, arr).ravel()
    cnt = np.bincount(sums, minlength=2 * n - 1)
    return cnt.astype(np.int32)


# ----------------------------------------------------------------------
def uniform_sequence(support: set[int], n: int) -> np.ndarray:
    """Build a vector of length n with uniform mass 1/|support| on the support."""
    k = len(support)
    a = np.zeros(n, dtype=np.float64)
    if k:
        a[list(support)] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Incremental updates used by the hill‑climber
def _apply_removal(counts: np.ndarray, r: int, support_arr: np.ndarray) -> None:
    """Subtract the contribution of element r (still present in support_arr)."""
    sums = r + support_arr
    counts[sums] -= 2          # (r,i) and (i,r)
    counts[2 * r] += 1         # self‑pair counted only once (net -1)


def _revert_removal(counts: np.ndarray, r: int, support_arr: np.ndarray) -> None:
    """Undo the removal of element r."""
    sums = r + support_arr
    counts[sums] += 2
    counts[2 * r] -= 1


def _apply_addition(counts: np.ndarray, p: int, support_arr: np.ndarray) -> None:
    """Add the contribution of element p (p not yet in support_arr)."""
    sums = p + support_arr
    counts[sums] += 2          # (p,i) and (i,p)
    counts[2 * p] -= 1         # adjust self‑pair (net +1)


def _revert_addition(counts: np.ndarray, p: int, support_arr: np.ndarray) -> None:
    """Undo the addition of element p."""
    sums = p + support_arr
    counts[sums] -= 2
    counts[2 * p] += 1


# ----------------------------------------------------------------------
def hill_climb(
    support_set: set[int],
    counts: np.ndarray,
    n: int,
    target_max: int,
    max_steps: int,
    time_deadline: float,
) -> int:
    """
    Stochastic hill climbing that tries to lower the maximal count.
    Returns the final maximal count (may be > target_max if the deadline
    is reached).  The routine works *in‑place* on `support_set` and
    `counts`.
    """
    # internal mutable list for fast conversion to np.array
    support_list = list(support_set)
    cur_max = int(counts.max())
    steps = 0

    while steps < max_steps and cur_max > target_max:
        # time guard
        if time.time() >= time_deadline:
            break

        # choose a sum index that currently attains the maximum
        max_indices = np.flatnonzero(counts == cur_max)
        sum_idx = int(np.random.choice(max_indices))

        # all elements of the support that take part in this sum
        critical = [
            i for i in support_set if 0 <= sum_idx - i < n and (sum_idx - i) in support_set
        ]

        # pick the element to remove (biased towards critical ones)
        if critical:
            r = int(np.random.choice(critical))
        else:
            r = int(np.random.choice(list(support_set)))

        # --------------------------------------------------------------
        # 1) remove r (tentatively)
        supp_arr_before = np.array(support_list, dtype=np.int32)  # contains r
        _apply_removal(counts, r, supp_arr_before)
        support_set.remove(r)
        support_list.remove(r)

        # --------------------------------------------------------------
        # 2) try a handful of random insertions; keep the first that does
        #    not increase the maximal count.
        accepted = False
        for _ in range(7):                     # a few trials per removal
            p = int(np.random.randint(0, n))
            if p in support_set:
                continue
            supp_arr_after = np.array(support_list, dtype=np.int32)  # after removal, no r
            _apply_addition(counts, p, supp_arr_after)
            new_max = int(counts.max())
            if new_max <= cur_max:                     # good move
                support_set.add(p)
                support_list.append(p)
                cur_max = new_max
                accepted = True
                break
            # reject this addition
            _revert_addition(counts, p, supp_arr_after)

        # --------------------------------------------------------------
        if not accepted:
            # revert the removal (no addition succeeded)
            _revert_removal(counts, r, supp_arr_before)
            support_set.add(r)
            support_list.append(r)

        steps += 1

        # occasional neutral random move to help escape plateaus
        if not accepted and np.random.rand() < 0.001:
            # perform a random swap without checking the max count
            # (this is a cheap way to introduce diversity)
            r2 = int(np.random.choice(list(support_set)))
            p2 = int(np.random.randint(0, n))
            while p2 in support_set:
                p2 = int(np.random.randint(0, n))
            arr_before = np.array(support_list, dtype=np.int32)
            _apply_removal(counts, r2, arr_before)
            support_set.remove(r2)
            support_list.remove(r2)

            arr_after = np.array(support_list, dtype=np.int32)
            _apply_addition(counts, p2, arr_after)
            support_set.add(p2)
            support_list.append(p2)

            cur_max = int(counts.max())

    return cur_max

def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Simple projected sub‑gradient descent (fallback) – identical to the
    version used in the reference implementation.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())

    eta0 = 0.30
    eta_decay = 1e-5
    random_move_prob = 0.02
    random_move_step = 0.15
    it = 0
    deadline = time.time() + time_limit - 0.2

    while time.time() < deadline:
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        grad = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        rho = ind[cond][-1] if np.any(cond) else 0
        theta = cssv[cond][-1] / rho if rho > 0 else 0
        a = np.maximum(v - theta, 0.0)

        # random small mass transfer
        if np.random.rand() < random_move_prob:
            i_rand, j_rand = np.random.randint(0, n, size=2)
            if i_rand != j_rand and a[i_rand] > 0:
                delta = random_move_step * a[i_rand] * np.random.rand()
                cand = a.copy()
                cand[i_rand] -= delta
                cand[j_rand] += delta
                # project again
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                rho = ind[cond][-1] if np.any(cond) else 0
                theta = cssv[cond][-1] / rho if rho > 0 else 0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 2500 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")

        it += 1

    print(f"[subgrad final] best value found: {best_val:.6f}")
    return best_a

            # random initial support
            S_arr = np.random.choice(n, k, replace=False).astype(np.int32)
            S_set = set(int(x) for x in S_arr)
            counts = compute_counts_np(list(S_set), n)
            cur_max = int(counts.max())
            cur_score = 2.0 * n * cur_max / (k * k)

            # keep best overall
            if cur_score < best_val:
                best_val = cur_score
                best_seq = uniform_sequence(S_set, n)
                print(f"[found] k={k} attempts={attempts} score={cur_score:.6f}")

            # if already good enough – return
            if cur_score <= target_score:
                print("[success] target met on initial random set")
                return best_seq.tolist()

            # ----------------------------------------------------------
            # hill‑climbing phase
            # limit the number of internal steps by the remaining time
            steps_budget = int(1e5)                     # generous upper bound
            steps_budget = min(steps_budget,
                               int((deadline - time.time()) * 1200))  # ~1200 steps per second
            if steps_budget <= 0:
                break

            final_max = hill_climb(
                support_set=S_set,
                counts=counts,
                n=n,
                target_max=allowed_max,
                max_steps=steps_budget,
                time_deadline=deadline,
            )
            final_score = 2.0 * n * final_max / (k * k)

            if final_score < best_val:
                best_val = final_score
                best_seq = uniform_sequence(S_set, n)
                print(f"[improved] k={k} attempts={attempts} score={final_score:.6f}")

            if final_score <= target_score:
                print("[success] target met after local search")
                return best_seq.tolist()

            # early termination if we have exhausted the global budget
            if time.time() >= deadline:
                break

        print(f"[search] finished k={k} after {attempts} attempts, best so far {best_val:.6f}")

        # stop early if we already reached the target globally
        if best_val <= target_score:
            break

    # ------------------------------------------------------------------
    # 2. fallback: sub‑gradient refinement from the best uniform sequence
    if best_seq is None:
        # very first fallback – start from a simple power‑law
        init = 1.0 / np.arange(1, n + 1, dtype=np.float64) ** 0.5
        init /= init.sum()
    else:
        init = best_seq.copy()

    remaining = deadline - time.time()
    if remaining > 5.0:                     # ensure we have some time left
        print(f"[fallback] running sub‑gradient optimisation for {remaining:.1f}s")
        refined = subgradient_optimize(init, remaining)
        refined_val = fast_score(refined)
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined
            print(f"[fallback] improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. final safety net
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    print(f"[final] best value returned: {best_val:.6f}")
    return best_seq.tolist()
