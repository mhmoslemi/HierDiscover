Propose a step‑function sequence that (hopefully) beats the previous
best bound 1.525761.

Algorithm
---------
1. **Parametric warm‑start** – generate a few hundred candidate
   distributions of the form (i+δ)^‑p (power‑law) and exponential
   decay, together with the classic half‑binomial series.  The best
   candidate (measured by the exact objective
   ``2·n·max(conv)/sum²``) is kept.

2. **Simplex Adam optimisation** – starting from that candidate we run
   a projected Adam optimiser on the probability simplex.
   * In the early stage we use a smooth approximation of the max
     (soft‑max with a temperature β) to obtain a dense gradient.
   * Once β reaches a large target we switch to the true max‑sub‑gradient.
   * After each Adam step the iterate is projected onto the simplex
     (Euclidean projection, Duchi et al.).
   * A tiny amount of Gaussian noise is injected occasionally to escape
     flat regions.
   * If no improvement is seen for many iterations we restart from the
     current best solution.

3. The best sequence found within the time budget is returned as a plain
   list of floats.  The final score is printed using the provided
   ``evaluate_sequence`` routine.

The code respects the 1000 s time budget, uses only NumPy (no external
solvers) and stays within the memory limits of a typical single‑CPU
environment.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions (no closures)

def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^‑½  (a_i = C(2i,i)/4^i) normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def power_law_series(n: int, p: float, shift: float) -> np.ndarray:
    """
    a_i ∝ (i + shift)^‑p  (i = 0,…,n‑1), normalised to sum = 1.
    shift must be ≥ 0; p > 0.
    """
    idx = np.arange(n, dtype=np.float64) + shift + 1e-12   # avoid zero
    a = idx ** (-p)
    a /= a.sum()
    return a


def exponential_series(n: int, lam: float) -> np.ndarray:
    """
    a_i ∝ exp(‑lam·i) , normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}
    (algorithm of Duchi et al.).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def objective_value(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq))/sum(seq)².
    Assumes seq is a 1‑D numpy array of non‑negative numbers.
    """
    n = seq.shape[0]
    N = next_pow_two(3 * n - 2)
    A = np.fft.rfft(seq, n=N)
    conv_full = np.fft.irfft(A * A, n=N)
    L = 2 * n - 1
    conv = conv_full[:L]
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (s * s)

                     budget_s: float = 950.0,
                     **kwargs) -> List[float]:

    Search for a non‑negative step‑function sequence that minimises the
    evaluation function (the lower the better).  The routine combines a
    cheap parametric sweep with a projected Adam optimiser on the
    probability simplex.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  A few seconds are kept as safety margin.

    best_seq = None
    best_score = np.inf
    candidates_tried = 0

    n_list = [2000, 3500, 5000]

        candidates_tried += 1

                sc = objective_value(a)

                candidates_tried += 1

            if sc < best_score:

            candidates_tried += 1

    print(f"[init] examined {candidates_tried} candidates, best score = {best_score:.6f}")

    # --------------------------------------------------------------

    a = best_seq.copy()
    n = a.shape[0]
    L = 2 * n - 1

    last_improve = 0

    it = 0

        t += 1

        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]                     # linear part only
        max_conv = conv.max()

        if cur_score < best_score:
