"""
Propose a low‑value step‑function sequence.

Algorithm
---------
* Initialise a pool of analytically‑motivated candidates:
  - decreasing power‑law,
  - linear decreasing,
  - symmetric “U‑shaped’’ power‑law (large values near the ends,
    small values near the centre),
  - a triangular bump (optional).
  Choose the best‑scoring one as the starting point.

* Optimise the simplex‑constrained vector `a` with a **mirror‑descent**
  (exponentiated‑gradient) scheme:
    - At iteration `t` compute the self‑convolution `c = a * a`
      by FFT.
    - Determine an *active set* of indices whose convolution value
      is within a relative gap `δ(t)` from the current maximum.
      The gap is annealed from ~0.07 down to ~0.005.
    - The sub‑gradient of the max‑convolution over the active set is
      `g_i = 2 Σ_{k∈active} a_{k‑i}`.
      It is obtained efficiently by a second FFT:
          g = 2·corr(active_mask, a_rev)  (restricted to the first n entries).
    - Apply exponential update on the simplex:
          a ← a ∘ exp(‑η·g)   (∘ is element‑wise product)
      followed by normalisation to sum = 1.
      `η` decays as `η₀ / sqrt(t+1)`.
    - Keep the best sequence seen so far (`best_seq`, `best_val`).

* To escape plateaus:
      – occasional random “mass‑transfer’’ moves between two coordinates,
      – periodic restart from a noisy perturbation of the current best,
      – a tiny uphill‑acceptance probability.

The routine runs for at most `budget_s` seconds (default 1000 s) and
returns the best list of non‑negative heights found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of `evaluate_sequence` (scale‑invariant)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv_size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, conv_size)
    conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Decreasing power‑law a_i ∝ (i+offset)^{‑p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n‑i) (i=0..n‑1) normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


def build_u_shaped_powerlaw(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """
    Symmetric “U‑shaped’’ power‑law:
      a_i ∝ 1 / (dist_to_edge + offset)^{p},
    where `dist_to_edge = min(i+1, n-i)` (1‑based distance to the nearest end).
    The vector is normalised to sum = 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    dist = np.minimum(idx, n - idx + 1)           # distance to the nearest edge
    a = 1.0 / np.power(dist + offset, p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def build_triangular(n: int) -> np.ndarray:
    """Symmetric triangular bump (rise then fall), normalised."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Search for a non‑negative step‑function with a low evaluation value."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # 1. Problem size – keep the size used in previous runs for fair comparison.
    n = 3000

    # --------------------------------------------------------------
    # 2. Assemble a diverse pool of analytically motivated starts.
    candidates = []

    # decreasing power‑law family
    for p in [0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60]:
        for off in [0.0, 0.1, 0.5, 1.0, 2.0]:
            candidates.append(build_power_law(n, p, off))

    candidates.append(build_linear_descending(n))

    # symmetric U‑shaped power‑law (often yields lower peaks)
    for p in [0.30, 0.40, 0.50, 0.60, 0.80]:

            candidates.append(build_u_shaped_powerlaw(n, p, off))

    # uniform baseline

    # incorporate the previous best sequence if it exists
    if 'height_sequence_1' in globals():

            seq = np.zeros(n, dtype=np.float64)

        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate according to the fast score.
    best_seq = None
    best_val = np.inf
    for seq in candidates:

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best initial value: {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Mirror‑descent optimisation (exponential updates on the simplex).
    a = best_seq.copy()                # current iterate (always on simplex)
    best_seq = a.copy()
    best_val = fast_score(a)

    # hyper‑parameters
    eta0 = 0.8                         # initial step size for the exponential update
    beta = 0.90                        # momentum coefficient
    delta_start = 0.07                 # initial relative gap for the active set
    delta_end   = 0.005                # final relative gap

    random_move_frac = 0.25            # max fraction of source mass to move

    it = 0

    v = np.zeros_like(a)               # momentum buffer

    # pre‑compute FFT sizes

    grad_size = 1 << ((3 * n - 2).bit_length())          # size for gradient (active * a_rev)

    while time.time() < deadline:
        # ----------------------------------------------------------
        # (a) self‑convolution and its maximum
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_conv = conv.max()

            i, j = np.random.randint(0, n, size=2)

                cand[j] += transfer

                if cand_sum > 0:

                cand_val = fast_score(cand)
                if cand_val < best_val:

                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < uphill_accept:

            noise /= noise.sum()
            a = proj_simplex(0.90 * best_seq + 0.10 * noise)
            v = np.zeros_like(a)
            last_improve = time.time()

        # ----------------------------------------------------------
        # (j) logging
        if it % 5000 == 0:
            print(f"[iter {it:7d}] cur {cur_val:.6f}  best {best_val:.6f}")

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # The external checker normalises itself, so we can return the raw vector.
    return best_seq.tolist()
