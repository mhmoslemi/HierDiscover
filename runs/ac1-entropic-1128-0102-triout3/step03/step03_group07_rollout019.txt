def propose_candidate(seed=0, budget_s=950, **kwargs):
    rng = np.random.default_rng(seed)
    n = 8000
    # initial from half_binomial_series
    a = half_binomial_series(n)
    # project onto simplex (already)
    best_a = a.copy()
    best_score = obj(a)
    # compute FFT size
    L = 2*n-1
    m = n+L-1   # 3n-2
    Nfft = next_pow_two(m)

    beta = 1.0
    beta_target = 5000.0
    beta_mul = 1.002
    eta = 0.5
    eta_min = 1e-5
    eta_decay = 0.9997

    momentum = np.zeros_like(a)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    it = 0
    last_improve = 0
    restart_gap = 200_000

    # helper to evaluate objective fast
    def evaluate_obj(v):
        A = np.fft.rfft(v, n=Nfft)
        conv = np.fft.irfft(A*A, n=Nfft)[:L]
        maxc = conv.max()
        return float(2.0 * n * maxc)

    # main loop
    while time.time() < deadline:
        # compute conv and gradient based on phase
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A*A, n=Nfft)[:L]
        maxc = conv.max()
        cur_score = 2.0 * n * maxc

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it}] new best {best_score:.6f}")

        # decide gradient
        if beta < beta_target:
            # softmax weighting
            shift = maxc
            w_exp = np.exp(beta * (conv - shift))
            w = w_exp / w_exp.sum()
            # gradient = 2 * correlation(a, w)
            Wr = np.fft.rfft(w[::-1], n=Nfft)
            corr_full = np.fft.irfft(A * Wr, n=Nfft)

    print(f"[done] best_score={final_score:.6f}, iterations={it}")

"""
Propose a step‑function that minimises the evaluation function.

The method is an enhanced version of the gradient‑descent / annealing
optimiser that was used previously.  It works on the probability simplex
(∑a_i = 1, a_i ≥ 0) and consists of three phases:

1.  Soft‑max (log‑sum‑exp) smoothing of the max‑convolution with a
    slowly increasing β.  The gradient is the correlation of a with the
    soft‑max weights and the iterate is projected back onto the simplex.

2.  Pure max‑sub‑gradient descent (the “half‑binomial” optimiser) once
    β has become large, i.e. when the smoothing is essentially the true
    max.

3.  Simulated‑annealing style random perturbations and occasional restarts
    from a Dirichlet distribution to escape local minima.

A larger problem size (n = 10 000) is used – the lower bound of the
objective tends to 1 as n grows, so a bigger n gives more room for
improvement.  The routine runs for at most ``budget_s`` seconds and
returns the best sequence found.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required)

def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^‑½   (a_i = C(2i,i)/4^i)
    and normalised so that they sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries will be equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def max_convolution(a: np.ndarray, Nfft: int, L: int) -> float:
    """
    Return the maximum value of the linear auto‑convolution of a.
    FFT size (Nfft) and output length (L) are supplied to avoid recomputing them.
    """
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[:L]
    return float(conv.max())


# ----------------------------------------------------------------------
# Core optimisation routine

def propose_candidate(seed: int = 0,
                     budget_s: float = 950.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence that minimises the evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  A few seconds are kept as safety margin.
    **kwargs : dict
        Ignored – present only to match the required signature.

    Returns
    -------
    List[float]
        The best sequence found (plain Python list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – a larger n gives a tighter theoretical lower bound.
    n = 10_000
    L = 2 * n - 1                      # length of a*a (linear part)
    m = n + L - 1                      # length needed for a * rev(w) = 3n‑2
    Nfft = next_pow_two(m)             # power‑of‑two FFT size (speed)

    # ------------------------------------------------------------------
    # initialise – classic half‑binomial series (good start)
    a = half_binomial_series(n)        # already lies on the simplex
    best_a = a.copy()
    best_score = 2.0 * n * max_convolution(a, Nfft, L)

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    beta = 1.0                         # start of soft‑max smoothing
    beta_target = 5_000.0              # final β (large → true max)
    beta_mul = 1.002                   # multiplicative increase per iteration

    eta = 0.5                          # step size for soft‑max phase
    eta_min = 5e-5
    eta_decay = 0.9997

    momentum = np.zeros_like(a)        # Nesterov‑like momentum

    # restart / perturbation control
    last_improve = 0
    restart_gap = 300_000              # iterations without improvement → restart
    noise_prob = 0.004
    noise_scale = 1e-4

    # timing
    deadline = time.time() + budget_s - 5.0   # leave a safety margin
    it = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) current convolution (FFT‑based, linear part only)
        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # 2) update best solution seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:09d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) choose gradient according to the current phase
        if beta < beta_target:
            # ----- soft‑max (log‑sum‑exp) gradient ------------------
            shift = max_conv
            # avoid overflow – clip exponent to a reasonable range
            exp_arg = beta * (conv - shift)
            np.clip(exp_arg, -700, 700, out=exp_arg)
            w_exp = np.exp(exp_arg)
            w = w_exp / w_exp.sum()               # shape (L,)

            # gradient = 2 * correlation(a, w)
            Wr = np.fft.rfft(w[::-1], n=Nfft)      # FFT of reversed weights
            corr_full = np.fft.irfft(A * Wr, n=Nfft)
            start = L - 1                         # index where shift = 0 starts
            grad = 2.0 * corr_full[start:start + n]

            # increase β gradually
            beta = min(beta_target, beta * beta_mul)
        else:
            # ----- pure max‑sub‑gradient phase -----------------------
            k = int(conv.argmax())
            grad = np.zeros_like(a)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = k - idx
            grad[idx] = 2.0 * a[jdx]

        # --------------------------------------------------------------
        # 4) momentum update (helps traversing shallow valleys)
        momentum = 0.9 * momentum + grad

        # 5) gradient step and projection back onto the simplex
        step = eta * momentum
        # do not erase an entry completely – keep a small safety factor
        step = np.minimum(step, a * 0.95)
        a_trial = a - step
        a = project_onto_simplex(a_trial)

        # --------------------------------------------------------------
        # 6) stochastic perturbation (helps escaping flat regions)
        if rng.random() < noise_prob:
            noise = (rng.random(n) - 0.5) * noise_scale
            a = a + noise
            a = project_onto_simplex(a)

        # --------------------------------------------------------------
        # 7) step‑size schedule
        eta = max(eta * eta_decay, eta_min)

        # --------------------------------------------------------------
        # 8) restart logic
        if it - last_improve > restart_gap:
            # fresh random start from a Dirichlet distribution
            a = rng.dirichlet(np.full(n, 0.6))
            a = project_onto_simplex(a)
            last_improve = it
            tmp_score = 2.0 * n * max_convolution(a, Nfft, L)
            print(f"[iter {it:09d}] restart triggered – score≈{tmp_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # final evaluation using the provided routine (adds the extra security
    # checks and the exact denominator)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
