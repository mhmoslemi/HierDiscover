def subgradient_advanced(a_init, time_limit):
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = len(a)
    conv_size = 1 << ((2*n - 1).bit_length())
    eta0 = 0.2
    eta_decay = 1e-6
    momentum = np.zeros_like(a)
    beta = 0.9

    # active set parameter
    active_width = 5   # consider indices whose conv value within certain threshold from max.

    deadline = time.time() + time_limit - 0.1
    it = 0

    while time.time() < deadline:
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[:2*n-1]
        max_val = conv.max()
        # get indices within (max_val - eps)
        eps = max_val * 0.01   # 1% tolerance maybe
        active_idxs = np.where(conv >= max_val - eps)[0]
        if len(active_idxs) > 1:
            # compute subgradient as average of gradients from these indices
            grad = np.zeros_like(a)
            i = np.arange(n)
            for k in active_idxs:
                j = k - i
                mask = (j >= 0) & (j < n)
                grad[mask] += a[j[mask]]
            grad = 2 * grad / len(active_idxs)
        else:
            k_max = int(np.argmax(conv))
            i = np.arange(n)
            j = k_max - i
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(a)
            grad[mask] = 2 * a[j[mask]]

        # momentum
        momentum = beta * momentum + (1 - beta) * grad

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        a = a - eta * momentum

        # projection onto simplex
        a = simplex_projection(a)

        # occasional random mass transfer moves
        if np.random.rand() < 0.02:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.2 * a[i] * np.random.rand()
                a[i] -= delta
                a[j] += delta
                a = simplex_projection(a)

        # evaluate
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adv iter {it}] new best {best_val:.6f}")

        it += 1

    print(f"[adv final] best {best_val:.6f}")
    return best_a

def simplex_projection(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def adam_optimize(a_init, time_limit):
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    p = 5.0
    p_max = 40.0
    start = time.time()
    deadline = start + time_limit - 0.1

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # size for FFT: we need for conv and for gradient
    conv_size = 1 << ((2*n - 1).bit_length())
    grad_size = 1 << ((3*n - 2).bit_length())  # for cross correlation length

    while time.time() < deadline:
        # compute conv
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2*n - 1]

        # stability for softmax
        scaled = p * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)  # avoid overflow
        w = exp_scaled / np.sum(exp_scaled)

        # compute gradient via cross-correlation using FFT

        fft_w_rev = np.fft.rfft(w_rev, grad_size)
        fft_a2 = np.fft.rfft(a, grad_size)

        grad = 2.0 * corr[2*n - 2 : 2*n - 2 + n]  # slice n elements

        # Adam update
        it += 1
        lr = 0.5 * np.sqrt(1 - beta2**it) / (1 - beta1**it)  # base lr 0.5
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        a = a - lr * m / (np.sqrt(v) + eps)

        # projection onto simplex
        a = simplex_projection(a)

        # optional momentum update for p

        p = min(p_max, 5.0 + (p_max - 5.0) * (elapsed / time_limit))

        # evaluate occasionally
        if it % 500 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

    return best_a

"""
Propose a low‑score step function.

The algorithm consists of three phases:

1.  **Greedy support attempt** – try to build a small support set with
    a uniformly distributed weight (the same approach as the reference
    code).  This is very fast and occasionally finds a good candidate.

2.  **Fast projected sub‑gradient with momentum** – optimises the
    true objective  max(a*a) (i.e. the max of the self‑convolution) on
    the probability simplex.  A momentum term and occasional random
    mass transfers improve escape from local plateaus.  This phase
    consumes the bulk of the time budget and typically brings the score
    below 1.55.

3.  **Smoothed Adam optimisation** – the non‑smooth max is approximated
    by a soft‑max with an increasing temperature parameter p.  Adam
    updates are performed on the surrogate loss, and after each step the
    vector is projected back onto the simplex.  This refinement often
    pushes the score below the target 1.503.

All steps respect the non‑negativity and normalisation constraints.
The best sequence found is returned as a plain Python list.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (non‑negative, sum=1)
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# Fast evaluation used internally (identical to the provided evaluate_sequence
# but without the security checks).
def fast_score(a: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative vector a (sum‑1)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

# Phase‑1: greedy support set construction (identical to the reference one)
def compute_counts(S: np.ndarray, n: int) -> np.ndarray:
    """Ordered‑pair counts for a support set S."""
    sums = np.add.outer(S, S)
    counts = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return counts


def greedy_build_set(n: int, k_target: int, max_allowed: int) -> np.ndarray | None:
    """Try to build a support of size k_target with count ≤ max_allowed."""
    order = np.random.permutation(n)
    S = []
    counts = np.zeros(2 * n - 1, dtype=np.int16)

    for idx in order:
        if len(S) >= k_target:
            break
        # self‑pair
        if counts[idx + idx] + 1 > max_allowed:
            continue
        # cross‑pairs
        if S:
            existing = np.array(S, dtype=np.int32)
            sums = idx + existing
            if np.any(counts[sums] + 2 > max_allowed):
                continue
        # accept
        S.append(int(idx))
        counts[idx + idx] += 1
        if S[:-1]:
            counts[idx + existing] += 2

    if len(S) == k_target:
        return np.array(S, dtype=np.int32)
    return None

# Phase‑2: fast projected sub‑gradient descent with momentum
def subgradient_momentum(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent on max(a*a) with momentum."""
    a = a_init.copy()
    n = a.size
    conv_sz = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_val = fast_score(a)

    # hyper‑parameters
    eta0 = 0.25          # initial step size
    eta_decay = 5e-6     # slower decay than the reference
    beta = 0.9           # momentum factor
    momentum = np.zeros_like(a)

    rand_move_prob = 0.02
    rand_move_step = 0.2

    it = 0
    deadline = time.time() + time_limit - 0.15

    while time.time() < deadline:
        # convolution
        fft_a = np.fft.rfft(a, conv_sz)
        conv = np.fft.irfft(fft_a * fft_a, conv_sz)[: 2 * n - 1]

        # active set of near‑max indices (within 1 % of the maximum)
        max_val = conv.max()
        eps = max_val * 0.01
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient (average over the active set)
        grad = np.zeros_like(a)
        i = np.arange(n)
        for k in active:
            j = k - i
            mask = (j >= 0) & (j < n)
            grad[mask] += a[j[mask]]
        grad = 2.0 * grad / max(1, active.size)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        a = a - eta * momentum

        # projection onto the simplex
        a = simplex_projection(a)

        # occasional random mass transfer move
        if np.random.rand() < rand_move_prob:
            i1, i2 = np.random.randint(0, n, size=2)
            if i1 != i2 and a[i1] > 0:
                delta = rand_move_step * a[i1] * np.random.rand()
                cand = a.copy()
                cand[i1] -= delta
                cand[i2] += delta
                cand = simplex_projection(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best = cand_val, cand.copy()
                # accept with a tiny probability to increase exploration
                elif np.random.rand() < 0.001:
                    a = cand

        # bookkeeping
        if it % 2500 == 0:
            cur = fast_score(a)
            if cur < best_val:
                best_val, best = cur, a.copy()
                print(f"[subgrad it {it:6d}] new best {best_val:.6f}")

        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best

# Phase‑3: smoothed Adam optimisation (soft‑max surrogate)
def adam_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Adam optimiser on a soft‑max surrogate of max(a*a)."""
    a = a_init.copy()
    n = a.size

    # FFT sizes
    conv_sz = 1 << ((2 * n - 1).bit_length())
    grad_sz = 1 << ((3 * n - 2).bit_length())

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    p = 5.0                 # start temperature
    p_target = 40.0
    start = time.time()
    deadline = start + time_limit - 0.15

    best = a.copy()
    best_val = fast_score(a)

    it = 0
    while time.time() < deadline:
        # ----- convolution -------------------------------------------------
        fft_a = np.fft.rfft(a, conv_sz)
        conv = np.fft.irfft(fft_a * fft_a, conv_sz)[: 2 * n - 1]

        # ----- soft‑max (stable) -------------------------------------------
        scaled = p * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / np.sum(exp_scaled)        # soft‑max weights

        # ----- gradient of the surrogate ------------------------------------
        # cross‑correlation a ⋆ w (equivalent to convolution with reversed w)
        w_rev = w[::-1]
        fft_w_rev = np.fft.rfft(w_rev, grad_sz)
        fft_a_big = np.fft.rfft(a, grad_sz)
        corr = np.fft.irfft(fft_a_big * fft_w_rev, grad_sz)[: 3 * n - 2]

        # grad_i = 2 * sum_j a_j w_{i+j}  == 2 * corr[2*n-2 - i]
        idx = 2 * n - 2 - np.arange(n)
        grad = 2.0 * corr[idx]

        # ----- Adam update --------------------------------------------------
        it += 1
        lr = base_lr * np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        a = a - lr * m / (np.sqrt(v) + eps)

        # projection onto simplex
        a = simplex_projection(a)

        # ----- temperature annealing ---------------------------------------
        elapsed = time.time() - start
        if p < p_target:
            p = 5.0 + (p_target - 5.0) * (elapsed / time_limit)
            p = min(p, p_target)

        # ----- occasional random kick (helps escaping) --------------------
        if np.random.rand() < 0.015:
            i1, i2 = np.random.randint(0, n, size=2)
            if i1 != i2 and a[i1] > 0:
                delta = 0.12 * a[i1] * np.random.rand()
                a[i1] -= delta
                a[i2] += delta
                a = simplex_projection(a)

        # ----- bookkeeping --------------------------------------------------
        if it % 500 == 0:
            cur = fast_score(a)
            if cur < best_val:
                best_val, best = cur, a.copy()
                print(f"[adam it {it:6d}] new best {best_val:.6f}")

    print(f"[adam final] best value {best_val:.6f}")
    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with evaluation score ≤ 1.503.
    The routine runs three stages (greedy, sub‑gradient, Adam) inside the
    allotted time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – a value around 1500 works well in practice
    n = 1500
    target_score = 1.503

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1) Greedy support attempt (very cheap, may succeed quickly)
    support_sizes = [120, 110, 100, 90, 80]   # try a few values
    for k in support_sizes:
        allowed = int(math.floor(target_score * k * k / (2.0 * n)))
        allowed = max(1, allowed)
        attempts = 0
        max_attempts = 8000
        while attempts < max_attempts and time.time() < deadline:
            attempts += 1
            S = greedy_build_set(n, k, allowed)
            if S is None:
                continue
            cnt = compute_counts(S, n)
            if cnt.max() > allowed:
                continue
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / k
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[greedy] k={k} attempts={attempts} score={val:.6f}")
            if val <= target_score:
                print("[greedy] target achieved")
                return best_seq.tolist()
        print(f"[greedy] k={k} finished, best so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Fast sub‑gradient with momentum (main workhorse)
    remaining = deadline - time.time()
    subgrad_time = min(300.0, remaining * 0.5)   # allocate up to half of remaining time
    if subgrad_time > 30.0:                     # ensure we have enough time
        # initialise: try to use the global sequence if present, otherwise uniform
        if 'height_sequence_1' in globals():
            a0 = np.array(height_sequence_1, dtype=np.float64)
            if a0.size != n:
                if a0.size < n:
                    a0 = np.pad(a0, (0, n - a0.size), mode='constant')
                else:
                    a0 = a0[:n]
        else:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        cand = subgradient_momentum(a0, subgrad_time)
        val = fast_score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[phase 2] sub‑gradient best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Multiple Adam restarts for final polishing
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 30.0:
            break
        # allocate a small slice (≈ 120 s) for each restart
        run_time = min(120.0, remaining)

        # random Dirichlet initialisation
        alpha = np.random.uniform(0.2, 4.0)
        a0 = np.random.gamma(alpha, 1.0, size=n)
        a0 = a0 / a0.sum()

        cand = adam_optimize(a0, run_time)
        val = fast_score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[re‑start] new best {best_val:.6f}")
        if best_val <= target_score:
            print("[final] target reached, returning sequence")
            break

    # ------------------------------------------------------------------
    # fallback – uniform distribution (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    print(f"[final] best score returned {best_val:.6f}")
    return best_seq.tolist()
