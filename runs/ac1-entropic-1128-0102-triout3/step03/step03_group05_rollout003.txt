"""
Propose a low‑value step‑function using a smooth‑max Adam optimizer.

The algorithm:
1. Build a pool of analytically‑motivated initial vectors (power‑law,
   exponential, Gaussian, linear, a symmetric two‑peak shape and the
   previous best if available).  All candidates are normalised to sum = 1
   and the best one (according to a fast exact objective) is selected.
2. Optimise the vector on the probability simplex with Adam.
   The true objective is  max_k (a*a)_k, which is non‑smooth.
   We approximate the max by a soft‑max:
        φ_β(a) = max_k (a*a)_k + (1/β)·log Σ_k exp(β·((a*a)_k‑max))
   whose gradient can be written in closed form using a convolution.
   β is annealed from a small value (smooth) to a large value (sharp)
   so that the optimisation first follows a smooth landscape and then
   focuses on the true max.
3. After each Adam step the iterate is projected onto the simplex.
   Every 200 iterations the exact objective (via fast_score) is
   evaluated and the best sequence so far is stored.
4. To escape plateaus a cheap “mass‑transfer’’ move is performed with a
   small probability; the move is accepted if it improves the exact
   objective (or occasionally uphill with a tiny probability).
5. If no improvement has been observed for a while the current iterate
   is re‑initialised by mixing the best sequence with a small random
   perturbation.

The whole routine respects the 1000 s budget and returns the best
sequence found (as a plain Python list).  The sequence is already
normalised to sum = 1, non‑negative and thus valid for ``evaluate_sequence``.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no safety checks).  a must be non‑negative."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    a /= a.sum()
    return a


def build_exponential(n: int, lam: float = 0.001) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    a /= a.sum()
    return a


def build_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a


def build_linear_descending(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


def build_two_peaks(n: int, p: float = 0.5) -> np.ndarray:
    """Symmetric two‑tail power‑law (large mass at both ends)."""
    left = build_power_law(n, p)
    right = left[::-1]
    a = (left + right) / 2.0
    a /= a.sum()
    return a

    Search for a non‑negative step‑function of length ~3000 that minimises
    the evaluation function.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence found (as a list of floats).
    """

    # safety margin so that we return before the timeout
    deadline = time.time() + budget_s - 2.0

    # --------------------------------------------------------------

    n = 3000
    conv_len = 2 * n - 1
    # Size needed for convolutions up to length 3n‑2 (gradient step)
    fft_size = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # 2. Build a pool of initial candidates
    candidates = []

    # power‑law family
    for p in [0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60]:
        for off in [0.0, 0.1, 0.5, 1.0, 2.0]:
            candidates.append(build_power_law(n, p, off))

    # exponential decay
    for lam in [0.0005, 0.001, 0.002, 0.004]:
        candidates.append(build_exponential(n, lam))

    # Gaussian bump
    for sf in [0.15, 0.20, 0.25]:
        candidates.append(build_gaussian(n, sigma_factor=sf))

    v = np.zeros_like(a)

    beta_start = 5.0

    move_frac = 0.25

        else:

        # (d) gradient of the smooth max
        # gradient = 2 * correlation(a, w)  (see analysis)
        w_rev = w[::-1]
        fft_w_rev = np.fft.rfft(w_rev, fft_size)
        corr_full = np.fft.irfft(fft_a * fft_w_rev, fft_size)
        # slice indices n‑1 … 2n‑2 (inclusive) and reverse
        grad = 2.0 * corr_full[n - 1 : 2 * n - 1][::-1]

        # (e) Adam update

                best_seq = a_new.copy()

            else:
                # uphill acceptance (rare)
                if np.random.rand() < uphill_prob:
                    a = a_new
        else:
            # always keep the iterate (helps momentum)
            a = a_new

        # (g) random mass‑transfer move (hill‑climbing)

            a = proj_simplex(0.9 * best_seq + 0.1 * noise)

            v.fill(0.0)
            last_improve = time.time()
            print(f"[iter {it:7d}] re‑seeded after stagnation")

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # Return as plain Python list; the external checker will renormalise if needed.
    return best_seq.tolist()
