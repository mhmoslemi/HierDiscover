"""
Propose a step‑function (non‑negative list) that minimises
`evaluate_sequence`.  

The algorithm consists of three phases

1. **Warm‑start** – use the previously found construction (global
   `height_sequence_1` if present) and up‑sample it to a larger length.
2. **Smoothed projected gradient descent** – minimise a smooth
   approximation of the `max` in the objective:
   ```
        S_τ(a) = τ·log Σ_k exp( (a*a)_k / τ )
   ```
   The gradient of `S_τ` is a soft‑weighted sum of the individual
   convolution‑gradients and can be computed with a single FFT‑based
   cross‑correlation.  The iterate stays on the probability simplex
   (non‑negative, sum = 1) by Euclidean projection after each step.
   The temperature τ is exponentially cooled; early iterations see a
   very smooth landscape, later ones approach the true max.
3. **Local stochastic refinement** – a cheap simulated‑annealing walk
   (small mass transfers) finishes the search.

All heavy lifting (convolution, cross‑correlation) uses FFTs, so even
with a few hundred thousand iterations the 1000 s budget is respected.
The code prints progress messages (time, iteration, current score) so
that the user can see how the bound improves.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    # length needed for exact linear convolution
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    prod = Af * Af
    conv_full = np.fft.irfft(prod, n=fft_len)[:conv_len]
    return conv_full


# ----------------------------------------------------------------------
# 3.  Cross‑correlation a ⋆ w (used for the gradient of the smoothed max)
# ----------------------------------------------------------------------
def _cross_corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_j a_j·w_{i+j}  (i = 0 … n‑1) using FFT.
    Here w has length 2·n‑1 (the same length as a*a).
    """
    n = a.shape[0]
    full_len = n + w.shape[0] - 1            # = 3·n‑2
    fft_len = 1 << ((full_len).bit_length())
    a_rev = a[::-1]                          # reversed a
    A = np.fft.rfft(a_rev, n=fft_len)
    W = np.fft.rfft(w, n=fft_len)
    conv_full = np.fft.irfft(A * W, n=fft_len)[:full_len]
    start = n - 1
    return conv_full[start:start + n]        # length n


# ----------------------------------------------------------------------
# 4.  Smoothed projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_soft_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    *,
    step0: float = 0.6,
    beta: float = 0.9,
    tau0: float = 0.02,
    tau_min: float = 1e-5,
    tau_decay: float = 0.9995,
    max_iter: int = 10 ** 7,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(a*a) via a smooth surrogate.
    Returns the best point encountered and its (unsmoothed) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # 1) convolution and current (unsmoothed) objective
        conv = _conv_self_fft(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"soft‑descent improve: n={n}, iter={it}, score={cur_score:.6f}"
            )

        # 2) temperature schedule (exponential cooling)
        tau = max(tau_min, tau0 * (tau_decay ** it))

        # 3) soft‑max weights over the convolution entries
        #    shift by max for numerical stability
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # 4) gradient of the smooth surrogate (cross‑correlation)
        grad = _cross_corr_fft(a, w)            # shape (n,)

        # 5) Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # 6) diminishing step size
        step = step0 / math.sqrt(it)

        # 7) gradient step + projection onto simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# 5.  Local stochastic refinement (tiny simulated annealing)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    *,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 80000,
) -> tuple[np.ndarray, float]:
    """
    Perform a cheap simulated‑annealing walk that only moves a small
    amount of mass ε from one index to another.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self_fft(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self_fft(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"annealing improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# 6.  Upsampling (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
