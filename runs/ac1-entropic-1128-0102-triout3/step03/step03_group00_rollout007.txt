"""
Lightweight “soft‑max” projected gradient descent for the L∞ self‑convolution
problem.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) which makes
the objective scale‑invariant.  A smooth approximation of the max‑convolution
is used:
    f_ε(a) = ε·log(∑_k exp( (a*a)_k / ε ))
Its gradient is a weighted sum of the convolution derivatives with the
weights given by a soft‑max over the convolution entries.  As ε → 0 the
gradient focuses on the active (max) entries, while for larger ε it provides
a smoother signal that helps escape shallow local minima.

The optimisation proceeds in three stages (all respecting the supplied time
budget):

1. **Soft‑max GD** – a long projected gradient descent with an exponentially
   decreasing ε (from 0.1 down to 5·10⁻⁴).  Momentum (β=0.9) and a step size
   η₀/√t are used.  The best iterate (according to the true objective
   `2·n·max(a*a)`) is stored.

2. **Exact sub‑gradient refinement** – a short projected sub‑gradient descent
   that only looks at the (near) maximal convolution entries.  This sharpens
   the solution found by the smooth phase.

3. **Local annealing** – a cheap simulated‑annealing walk that transfers a
   small amount of mass between two random coordinates.  It runs for the
   remaining time budget and may discover a final improvement.

If a global variable `height_sequence_1` is present it is used as a warm‑start,
otherwise a random simplex point is generated.  The final sequence is
normalised to sum to 1 and returned as a plain Python list.

The whole routine fits into a 1000 s wall‑clock budget and prints progress
messages useful for debugging.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1} (O(n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear convolution a * a via FFT (full length 2·n‑1)
# ------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear self‑convolution of a (first 2·n‑1 entries)."""
    Af = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(Af * Af, n=fft_len)
    n = a.shape[0]
    return conv_full[: 2 * n - 1], Af  # also return the FFT of a for reuse


# ------------------------------------------------------------
# Gradient of the smooth (soft‑max) objective
# ------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      Af: np.ndarray,
                      eps: float,
                      fft_len: int) -> np.ndarray:
    """
    Compute ∇f_ε(a) where f_ε(a)=ε·log∑exp(conv/ε).
    The gradient is 2·(a * w_rev) where w_i = softmax(conv_i/ε).
    """
    n = a.shape[0]

    # soft‑max weights (numerically stable)
    max_c = conv.max()
    w = np.exp((conv - max_c) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:               # extreme underflow – fallback to uniform
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # reverse the weight vector for the (cross‑correlation) gradient
    w_rev = w[::-1]

    # FFT of the reversed weight vector (same length as Af)
    Bf = np.fft.rfft(w_rev, n=fft_len)

    # convolution a * w_rev (full length = 3·n‑2)
    grad_full = np.fft.irfft(Af * Bf, n=fft_len)
    # slice the relevant part: indices n‑1 … 2·n‑2 (inclusive)
    grad = 2.0 * grad_full[n - 1 : 2 * n - 1][::-1]
    return grad


# ------------------------------------------------------------
# Stage 1 – soft‑max projected gradient descent
# ------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    eps_start: float = 0.10,
                    eps_end: float = 5e-4,
                    beta: float = 0.90,
                    step0: float = 0.55,
                    max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth soft‑max surrogate.
    Returns the best point seen and its true objective value
    (2·n·max(a*a)).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # FFT length must accommodate the longest convolution (a * w_rev) -> 3n‑2
    fft_len = 1 << ((3 * n - 2).bit_length())

    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- evaluate current objective -----
        conv, Af = _conv_self(a, fft_len)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] soft‑max GD improve: iter={it}, score={cur_score:.6f}")

        # ----- schedule ε (exponential decay) -----
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        eps = eps_start * (eps_end / eps_start) ** frac

        # ----- compute gradient -----
        grad = _softmax_gradient(a, conv, Af, eps, fft_len)

        # ----- momentum & projection -----
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ------------------------------------------------------------
# Sub‑gradient of max‑convolution (active‑set version)
# ------------------------------------------------------------
def _subgradient_full(a: np.ndarray,
                      conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices within (1‑rel_tol) of the maximum contribute equally.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    if act.size:
        g /= float(act.size)
    return g


# ------------------------------------------------------------
# Stage 2 – projected sub‑gradient descent (exact max)
# ------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        beta: float = 0.90,
                        step0: float = 0.60,
                        max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent focusing on the true max‑conv.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    fft_len = 1 << ((2 * n - 1).bit_length())   # enough for a*a

    while time.time() < deadline and it < max_iter:
        it += 1

        conv, _ = _conv_self(a, fft_len)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ------------------------------------------------------------
# Stage 3 – cheap simulated annealing refinement
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 80000) -> tuple[np.ndarray, float]:
    """
    Small “move ε from i to j’’ annealing.  Returns the best sequence seen.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a, 1 << ((2 * n - 1).bit_length()))[0].max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.10      # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop, 1 << ((2 * n - 1).bit_length()))[0].max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: iter={it}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a simplex vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ------------------------------------------------------------
# Main entry point required by the evaluation harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget (≈1000 s) and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a tiny safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Warm‑start: use the globally supplied sequence if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    n_large = 5000                         # feel free to tweak
    if starter is not None:
        a0 = _upsample(starter, n_large)
    else:
        a0 = np.random.rand(n_large)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    # 60 % for the smooth soft‑max GD, 25 % for exact sub‑gradient, rest for annealing
    soft_deadline = start_time + 0.60 * total_time
    subgrad_deadline = soft_deadline + 0.25 * total_time
    anneal_deadline = deadline

    # ------------------------------------------------------------------
    # Stage 1 – smooth soft‑max gradient descent
    # ------------------------------------------------------------------
    a_soft, score_soft = _run_softmax_gd(
        a0, n_large, soft_deadline, start_time,
        eps_start=0.10, eps_end=5e-4,
        beta=0.90, step0=0.55
    )

    # ------------------------------------------------------------------
    # Stage 2 – exact sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < subgrad_deadline:
        a_sub, score_sub = _run_subgradient_gd(
            a_soft, n_large, subgrad_deadline, start_time,
            beta=0.90, step0=0.60
        )
    else:
        a_sub, score_sub = a_soft, score_soft

    best_seq = a_sub
    best_score = min(score_soft, score_sub)

    # ------------------------------------------------------------------
    # Stage 3 – local simulated annealing
    # ------------------------------------------------------------------
    if time.time() < anneal_deadline:
        a_final, score_final = _run_annealing(
            best_seq, n_large, anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, steps=80000
        )
        if score_final < best_score:
            best_seq = a_final
            best_score = score_final
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # Final normalisation (tiny drift may appear due to floating‑point)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
