def convex_refine(a_init, n, time_deadline):
    a_cp = cp.Variable(n, nonneg=True)
    t_cp = cp.Variable()
    constraints = [cp.sum(a_cp) == 1]
    # We'll start with constraints for top k indices from conv of a_init.
    conv = conv_fft(a_init)
    sorted_idx = np.argsort(-conv)  # descending
    selected = set(sorted_idx[:5])  # start with 5 top

    # Add constraints for each i in selected:
    for i in selected:
        Q = Q_matrix_for_index(i, n)
        constraints.append(cp.quad_form(a_cp, Q) <= t_cp)

    problem = cp.Problem(cp.Minimize(t_cp), constraints)
    problem.solve(solver=cp.SCS, verbose=False, max_iters=2000, eps=1e-4, warm_start=True)

    # Extract a_opt
    a_opt = np.array(a_cp.value)
    if a_opt is None or np.isnan(a_opt).any():
        # fallback: return initial

def build_Q(i,n):
    rows = []
    cols = []
    data = []
    lo = max(0, i-(n-1))
    hi = min(i, n-1)
    for j in range(lo, hi+1):
        k = i - j
        if j < k:
            rows.append(j)
            cols.append(k)
            data.append(0.5)
            rows.append(k)
            cols.append(j)
            data.append(0.5)
        elif j == k:
            rows.append(j)
            cols.append(j)
            data.append(1.0)
    return sp.csr_matrix((data, (rows, cols)), shape=(n, n))

"""
Propose candidate step‑function coefficients that minimise

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1. **Power‑law sweep** – a fast grid search over the family
   a_i ∝ (i+offset)^(-p) (normalised to L1‑norm 1).  The best
   point supplies a good initialisation.

2. **Cheap coordinate descent** – many cheap moves that keep the sum
   of the coefficients constant.  A move transfers a small amount of
   mass from a source index to a target index and updates the
   convolution incrementally (O(n) per move).  The current best
   sequence (lowest max‑convolution) is kept.

3. **Optional convex refinement** – if CVXPY is available, a small
   cutting‑plane SDP is solved: minimise t subject to
   a ≥ 0, Σa = 1 and aᵀQ_i a ≤ t for a selection of the most
   violated convolution entries.  New constraints are added
   iteratively until time runs out.

The whole routine respects the supplied time budget (≈ 950 s,
leaving a safety margin) and finally returns the best list of
coefficients found.  All numbers are non‑negative; the routine
prints occasional progress information.

# Basic linear‑algebra helpers
def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i = (i+offset)^(-p)  (i = 0 … n‑1)  normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Gradient utilities (kept from the previous approach – useful for a short
# refinement phase before the cheap coordinate descent)
def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Exact sub‑gradient of conv[peak_idx] w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)

    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


def gradient_of_top_k(a: np.ndarray, conv: np.ndarray, k: int,
                      beta: float = 5.0) -> np.ndarray:
    """Weighted mixture of the gradients of the k largest convolution entries."""
    m = conv.shape[0]
    if k >= m:
        idxs = np.arange(m)
    else:
        idxs = np.argpartition(conv, -k)[-k:]
    vals = conv[idxs]
    maxv = conv.max()
    w = np.exp(beta * (vals - maxv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w_sum
    grad = np.zeros_like(a)
    for idx, weight in zip(idxs, w):
        grad += weight * gradient_of_peak(a, int(idx))
    return grad

# Quadratic‑constraint helpers for the optional convex refinement
def build_Q_sparse(idx: int, n: int) -> sp.csr_matrix:
    """
    Build the symmetric PSD matrix Q such that
    aᵀ Q a = (a * a)[idx] .
    Entries:
        Q[j,k] = 0.5 for j+k = idx, j≠k
        Q[j,j] = 1   for 2j = idx
    """
    rows = []
    cols = []
    data = []
    lo = max(0, idx - (n - 1))
    hi = min(idx, n - 1)
    for j in range(lo, hi + 1):
        k = idx - j
        if j < k:
            rows.extend([j, k])
            cols.extend([k, j])
            data.extend([0.5, 0.5])
        elif j == k:
            rows.append(j)
            cols.append(j)
            data.append(1.0)
    return sp.csr_matrix((data, (rows, cols)), shape=(n, n))


def convex_refine(a_start: np.ndarray, n: int,
                  deadline: float, rng: np.random.Generator) -> np.ndarray:
    """Small cutting‑plane SDP that directly reduces the maximal convolution.
    Returns a (possibly) improved sequence; if cvxpy is unavailable, returns a_start."""
    try:
        import cvxpy as cp
    except Exception:                     # pragma: no cover
        return a_start

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    # initialise with a modest active set (the 12 largest peaks)
    conv = conv_fft(a_start)
    active = set()
    for idx in np.argsort(-conv)[:12]:
        constraints.append(cp.quad_form(a_var, build_Q_sparse(int(idx), n)) <= t_var)
        active.add(idx)

    best_a = a_start.copy()
    best_score = 2.0 * n * conv.max()

    while time.time() < deadline:
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, warm_start=True,
                       max_iters=5000, eps=1e-5, verbose=False)
        except Exception as e:              # pragma: no cover
            print("[convex] solver error:", e)
            break
        if a_var.value is None:
            break
        a_opt = np.asarray(a_var.value, dtype=np.float64).flatten()
        a_opt = np.clip(a_opt, 0.0, None)
        a_opt /= a_opt.sum()
        conv = conv_fft(a_opt)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a_opt.copy()
            print(f"[convex] new best score = {best_score:.6f}")

        # add constraints for the currently largest peak(s) that are not yet active
        new_added = False
        for idx in np.argsort(-conv)[:8]:
            if idx not in active:
                constraints.append(cp.quad_form(a_var,
                                                build_Q_sparse(int(idx), n)) <= t_var)
                active.add(idx)
                new_added = True
        if not new_added:
            # no new active constraints → convergence (or time limit)
            break
    return best_a

# Cheap coordinate descent (mass‑transfer moves with incremental convolution updates)
def coordinate_descent(a0: np.ndarray, n: int,
                       deadline: float, rng: np.random.Generator) -> np.ndarray:
    """Perform many cheap mass‑transfer moves, updating the convolution
    incrementally (O(n) per move).  Returns the best vector seen."""
    a = a0.copy()
    conv = conv_fft(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    start = time.time()
    it = 0
    recompute_interval = 4000   # recompute the full convolution every few thousand steps

    while time.time() < deadline:
        # source chosen with probability proportional to current mass
        src = rng.choice(n, p=a)
        if a[src] < 1e-12:
            continue
        # target uniformly random (must differ from src)
        tgt = rng.integers(n - 1)
        if tgt >= src:
            tgt += 1

        # step size slowly shrinks as we approach the deadline
        remaining = max(0.0, deadline - time.time())
        total = max(1e-12, deadline - start)
        frac = remaining / total
        step_ratio = 0.02 + 0.18 * frac          # from 0.20 → 0.02
        max_delta = a[src] * step_ratio
        if max_delta <= 1e-12:
            continue
        delta = max_delta * rng.random()

        a_old = a.copy()                         # values *before* the move
        a[src] -= delta
        a[tgt] += delta

        # incremental convolution update (exact up to O(delta²), ignored)
        conv[src:src + n] -= 2.0 * delta * a_old
        conv[tgt:tgt + n] += 2.0 * delta * a_old

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[coord iter {it:07d}] new best {best_score:.6f}")

        it += 1
        # occasional full recomputation to avoid drift from the ignored delta² terms
        if it % recompute_interval == 0:
            a = best_a.copy()                     # restart from the current best
            conv = conv_fft(a)
            a /= a.sum()                          # renormalise (tiny numerical drift)

        if best_score <= 1.503:                    # early stop if target already reached
            break

    return best_a

def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative step‑function vector that minimises the
    evaluation score.  Returns the best list found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------ configuration
    n = 2500                      # length of the sequence (adjustable)
    print("[search] length =", n)

    # --------------------------------------------------- phase 1 : power‑law sweep
    best_a = None
    best_score = np.inf
    p_grid = np.arange(0.40, 0.71, 0.015)
    offsets = [0.3, 0.5, 1.0, 2.0]

    print("[phase 1] coarse power‑law sweep")
    for off in offsets:
        for p in p_grid:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=off)
            conv = conv_fft(a)
            cur = 2.0 * n * conv.max()          # sum(a)=1 → denominator = 1
            if cur < best_score:
                best_score, best_a = cur, a.copy()
                print(f"[grid] new best: p={p:.4f}, off={off}, score={cur:.6f}")

    # fine sweep around the best (if we have a candidate)
    if best_a is not None:
        fine_range = 0.02
        p_fine = np.arange(max(0.01, best_a.shape[0] * 0.0 - fine_range),   # dummy to keep shape
                           max(0.99, best_a.shape[0] * 0.0 + fine_range) + 0.001,
                           0.001)                                          # overridden below
        # the above line is only to keep the variable defined – we recompute below
        p_fine = np.arange(max(0.01, best_a.shape[0] * 0.0 - fine_range),   # placeholder
                           max(0.01, best_a.shape[0] * 0.0 + fine_range) + 0.001,
                           0.001)                                          # placeholder
        # actual fine sweep using the previously stored best_p and best_offset
        # (they are stored implicitly in best_a, we just reuse the same grid around the best)
        # For simplicity we reuse the original coarse grid but with finer steps.
        p_center = None
        # recover the centre parameters from the best vector – we approximate by fitting a power law
        # (this is just a quick heuristic, not mathematically exact)
        idx = np.arange(n)
        # avoid log(0)
        with np.errstate(divide='ignore'):
            p_guess = -np.polyfit(np.log(idx + 1), np.log(best_a + 1e-12), 1)[0]
        p_center = np.clip(p_guess, 0.40, 0.70)
        off_center = 0.5  # keep the offset that gave the best coarse result (often 0.5)

        p_fine = np.arange(max(0.01, p_center - fine_range),
                           min(0.99, p_center + fine_range) + 0.001,
                           0.001)
        print("[phase 2] fine sweep")
        for off in [off_center]:
            for p in p_fine:
                if time.time() > deadline:
                    break
                a = power_law_sequence(n, p, offset=off)
                conv = conv_fft(a)
                cur = 2.0 * n * conv.max()
                if cur < best_score:
                    best_score, best_a = cur, a.copy()
                    print(f"[fine] new best: p={p:.4f}, off={off}, score={cur:.6f}")

    # --------------------------------------------------- optional gradient refinement (short)
    # A few fast projected‑gradient steps can improve the coarse candidate.
    # We keep this short because the subsequent coordinate descent is far cheaper.
    max_grad_iters = 30_000
    eta = 0.07
    beta = 3.0
    decay_every = 4000
    decay_factor = 0.85
    top_k = 6

    if best_a is not None and time.time() < deadline:
        print("[phase 3] brief projected gradient descent")
        a = best_a.copy()
        cur_score = 2.0 * n * conv_fft(a).max()
        for it in range(max_grad_iters):
            if time.time() > deadline:
                break
            conv = conv_fft(a)
            grad = gradient_of_top_k(a, conv, top_k, beta=beta)
            step = eta * grad
            step = np.minimum(step, a * 0.99)          # avoid zeroing any entry
            a_new = project_onto_simplex(a - step)
            new_score = 2.0 * n * conv_fft(a_new).max()
            if new_score < cur_score:
                a, cur_score = a_new, new_score
                if new_score < best_score:
                    best_score, best_a = new_score, a_new.copy()
                    print(f"[grad] improved best to {best_score:.6f}")
            # simple decay schedule
            if (it + 1) % decay_every == 0:
                eta = max(eta * decay_factor, 1e-6)
                beta += 0.005

    # --------------------------------------------------- phase 4 : cheap coordinate descent
    if best_a is None:                         # safety fallback
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * conv_fft(best_a).max()

    print("[phase 4] massive coordinate descent")
    coord_deadline = deadline - 10.0   # keep a few seconds for the final polishing step
    best_a = coordinate_descent(best_a, n, coord_deadline, rng)
    best_score = 2.0 * n * conv_fft(best_a).max()
    print(f"[coord] after descent best score = {best_score:.6f}")

    # --------------------------------------------------- optional convex refinement (if time permits)
    if time.time() < deadline - 5.0:
        print("[phase 5] convex refinement (if cvxpy present)")
        final_deadline = deadline
        best_a = convex_refine(best_a, n, final_deadline, rng)
        best_score = 2.0 * n * conv_fft(best_a).max()
        print(f"[convex] final refined score = {best_score:.6f}")

    # --------------------------------------------------- final sanity check & return
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")
    return best_a.tolist()
