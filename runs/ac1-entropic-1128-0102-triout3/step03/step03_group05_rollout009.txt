---------

       guarantees that the new point does not increase the objective.
       A momentum term (β≈0.9) smooths the updates.
    4. With a small probability a random “mass‑transfer’’ move is performed
       (move a fraction of the mass from one index to another).  The move is
       accepted if it improves the best value, otherwise it may be accepted
       with a tiny uphill probability that also anneals.
    5. Occasionally mix the current iterate with a fixed power‑law base
       (helps to escape plateaus) and, if no improvement has been observed
       for a while, re‑seed the iterate as a small perturbation of the best
       solution found so far.

*  The best sequence seen during the run is returned as a plain Python list.
   The external checker will normalise it, clip negatives and cap values
   automatically.

The whole search fits comfortably into the 1000 s budget on a single CPU.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of `evaluate_sequence` (no safety checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # size = next power of two ≥ 2n‑1
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def gradient_multi_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution (active‑set version).

    For each active index k the contribution is
        g_i ← g_i + 2·a_{k‑i}    (when 0 ≤ k‑i < n)
    The implementation follows the reference code.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i = np.arange(low, high + 1)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_exponential(n: int, lam: float = 0.001) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function (list of floats) with a low
    evaluation value.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0                 # safety margin

    # --------------------------------------------------------------
    # 1. Problem size (the benchmark used 3000)
    n = 3000

    # --------------------------------------------------------------
    # 2. Build a modest pool of analytically motivated starts.
    candidates = []

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Linear decreasing shape
    candidates.append(np.arange(n, 0, -1, dtype=np.float64) / np.arange(1, n + 1).sum())

    # Power‑law family (wide range of exponents & offsets)
    for p in np.linspace(0.15, 1.0, 18):
        for off in [0.0, 0.1, 0.5, 1.0, 2.0]:
            candidates.append(build_power_law(n, p, off))

    # Exponential family
    for lam in [0.0005, 0.001, 0.002, 0.004, 0.008]:
        candidates.append(build_exponential(n, lam))

    # Use the previous best if it exists
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)
        if prev.size > 0:
            seq = np.zeros(n, dtype=np.float64)
            seq[: min(n, prev.size)] = prev[: min(n, prev.size)]
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0:
                seq /= seq.sum()
                candidates.append(seq)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate according to the fast score.
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best start value = {best_val:.6f}")

    a = best_seq.copy()                 # current iterate (always on the simplex)
    best_seq = a.copy()
    best_val = fast_score(a)

    # --------------------------------------------------------------
    # 4. Hyper‑parameters
    max_K = 30                          # maximal active indices used for the gradient
    delta_start = 0.07                  # initial relative gap for the active set
    delta_end = 0.005                   # final relative gap
    eta0 = 0.6                          # initial step size (will be annealed)
    momentum_beta = 0.9                 # momentum coefficient
    random_move_prob = 0.02             # probability of a stochastic mass‑transfer move
    random_move_frac = 0.25             # largest fraction of source mass that may be moved
    uphill_initial = 0.02               # initial uphill‑accept probability
    uphill_final = 0.0005               # final uphill‑accept probability
    stagnation_secs = 15.0              # re‑seed if no improvement for this long

    # momentum buffer
    avg_grad = np.zeros_like(a)

    # Pre‑compute FFT size (nearest power of two ≥ 2n‑1)
    conv_size = 1 << ((2 * n - 1).bit_length())

    # A fixed power‑law vector for occasional mixing moves
    base_power = build_power_law(n, p=0.5, offset=0.0)

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ----------------------------------------------------------
        # (a) self‑convolution and current maximum
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_conv = conv.max()
        cur_val = fast_score(a)

        # (b) annealed parameters (based on elapsed fraction)
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)           # 0 … 1
        delta = max(delta_end, delta_start * (1.0 - frac))
        eta = eta0 * (1.0 - frac)                          # step size shrinks

        # (c) active set (indices whose convolution is close to the maximum)
        mask = conv >= max_conv * (1.0 - delta)
        active = np.flatnonzero(mask)
        if active.size > max_K:
            # keep the max_K largest ones
            idx = np.argpartition(-conv[active], max_K)[:max_K]
            active = active[idx]

        # (d) sub‑gradient for the current active set
        grad = gradient_multi_active(a, active)

        # (e) momentum update
        avg_grad = momentum_beta * avg_grad + (1.0 - momentum_beta) * grad

        # (f) back‑tracking line search (ensure non‑increase)
        cand = proj_simplex(a - eta * avg_grad)
        cand_val = fast_score(cand)
        # simple back‑track: halve eta until we do not worsen the objective
        bt = 0
        while cand_val > cur_val and eta > 1e-8 and bt < 10:
            eta *= 0.5
            cand = proj_simplex(a - eta * avg_grad)
            cand_val = fast_score(cand)
            bt += 1

        # (g) accept / uphill logic
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand.copy()
            a = cand
            last_improve = time.time()
        else:
            uphill_prob = uphill_initial * (1.0 - frac) + uphill_final * frac
            if np.random.rand() < uphill_prob:
                a = cand

        # ----------------------------------------------------------
        # (h) random “mass‑transfer’’ move (helps escaping plateaus)
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 1e-12:
                delta_transfer = random_move_frac * a[i] * np.random.rand()
                cand2 = a.copy()
                cand2[i] -= delta_transfer
                cand2[j] += delta_transfer
                cand2 = proj_simplex(cand2)
                cand2_val = fast_score(cand2)
                if cand2_val < best_val:
                    best_val = cand2_val
                    best_seq = cand2.copy()
                    a = cand2
                    last_improve = time.time()
                else:
                    uphill_prob = uphill_initial * (1.0 - frac) + uphill_final * frac
                    if np.random.rand() < uphill_prob:
                        a = cand2

        # ----------------------------------------------------------
        # (i) occasional mixing with a fixed power‑law shape
        if np.random.rand() < 0.01:
            alpha = np.random.rand() * 0.1          # at most 10 % mixing
            cand3 = proj_simplex((1.0 - alpha) * a + alpha * base_power)
            cand3_val = fast_score(cand3)
            if cand3_val < best_val:
                best_val = cand3_val
                best_seq = cand3.copy()
                a = cand3
                last_improve = time.time()

        # ----------------------------------------------------------
        # (j) stagnation handling – re‑seed from a small perturbation of the best
        if time.time() - last_improve > stagnation_secs:
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(0.95 * best_seq + 0.05 * noise)
            last_improve = time.time()

        # ----------------------------------------------------------
        # (k) progress reporting
        if it % 2000 == 0:
            print(f"[iter {it:6d}] cur {cur_val:.6f}  best {best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # The external checker normalises the sequence, so we may return it as‑is.
    return best_seq.tolist()
```

This implementation starts from a diverse set of analytically motivated sequences, performs an annealed projected gradient descent with momentum, augments the search with stochastic mass‑transfer moves, occasional mixing with a base power‑law distribution, and a simple stagnation‑re‑seeding strategy.  
All operations respect the non‑negativity and simplex constraints, and the fast FFT‑based score is used throughout to stay well within the 1000 s budget.
