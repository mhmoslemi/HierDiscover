"""
Propose a non‑negative step function that minimizes `evaluate_sequence`.

Algorithm
---------
1. **Coarse optimisation (small size)** – Projected gradient descent on the
   probability simplex for a short time (n≈800).  This quickly discovers a
   useful non‑uniform shape.

2. **Upsample + long optimisation (large size)** – The best small vector is
   linearly interpolated to a larger dimension (n≈2400) and refined by a long
   projected gradient descent (momentum + diminishing step‑size).

3. **Linear‑programming refinement** – For the current vector `f` we compute
   `M = max(conv(f,f))`.  The linear programme

        maximise    1ᵀ g
        subject to   f * g ≤ M          (convolution is linear in g)
                     g ≥ 0

   finds a direction `g` that carries more L¹‑mass while keeping the
   convolution bound.  After normalising `g` to unit sum we mix it with the
   current vector (`f ← (1‑t)·f + t·g_norm`).  Repeating this step reduces the
   maximum convolution and typically improves the objective by a few per‑cent.

4. **Local stochastic refinement** – A cheap simulated‑annealing walk
   (`ε`‑transfer moves) finishes the search within the remaining time budget.

All stages respect a total wall‑clock limit of 1000 s and keep the vector
normalised (sum = 1), which makes the evaluation function equal to `2·n·max(conv)`.
"""

import time
import math
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) convolution via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution
# ------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Average sub‑gradient from all indices attaining (or near) the maximum."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    act = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    if act.size:
        g /= float(act.size)
    return g


# ------------------------------------------------------------
# Projected (Nesterov‑style) gradient descent on the simplex
# ------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9, step0: float = 0.6,
                         max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Many iterations of projected gradient descent; returns best point."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: n={n}, iter={it}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ------------------------------------------------------------
# Linear‑programming refinement step
# ------------------------------------------------------------
def _lp_refinement(a0: np.ndarray, deadline: float, start_time: float,
                   max_steps: int = 12) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve   maximise 1ᵀg   subject to  f*g ≤ M, g≥0,
    then mix the solution back into f.  Returns the best vector found.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_M = _conv_self(a).max()
    best_score = 2.0 * n * best_M
    t = 0.5                      # mixing coefficient

    for step in range(max_steps):
        if time.time() >= deadline:
            break
        # current bound
        conv = _conv_self(a)
        M = conv.max()

        # Build convolution matrix A such that A @ g = f * g
        first_col = np.concatenate([a, np.zeros(n - 1)])
        first_row = np.concatenate([a[:1], np.zeros(n - 1)])
        A = toeplitz(first_col, first_row)

        # Linear programme: maximise sum(g)  ↔ minimise -sum(g)
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=M * np.ones(2 * n - 1),
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'presolve': True}
        )
        if not res.success:
            print(f"[{time.time() - start_time:6.2f}s] LP failed at step {step}")
            break

        g = res.x
        S = g.sum()
        if S <= 1.0 + 1e-12:
            # No increase in L1‑mass → stop
            break

        # Normalise g to unit L1‑mass and mix
        g_norm = g / S
        a = (1.0 - t) * a + t * g_norm
        a = a / a.sum()           # protect against rounding drift

        # New score
        M_new = _conv_self(a).max()
        score_new = 2.0 * n * M_new
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
            best_M = M_new
            print(f"[{time.time() - start_time:6.2f}s] LP improve step {step}: score={score_new:.6f}")

        # If we did not improve, slowly reduce the mixing factor
        if score_new >= best_score:
            t = max(0.1, t * 0.7)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing local refinement
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 80000) -> tuple[np.ndarray, float]:
    """Tiny “move ε from i to j’’ annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling (linear interpolation)
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ------------------------------------------------------------
# Main entry point required by the evaluation harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm obeys the given time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0       # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage 1 – short optimisation at modest size (n≈800)
    # ------------------------------------------------------------------
    n_small = 800
    slice_time = max(30.0, 0.15 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    if starter is not None:
        a0_small = _upsample(starter, n_small)
    else:
        a0_small = np.random.rand(n_small)
        a0_small /= a0_small.sum()

    a_small, score_small = _run_gradient_descent(
        a0_small, n_small, slice_deadline, start_time,
        beta=0.9, step0=0.6, max_iter=10**7
    )
    if score_small < best_score:
        best_score = score_small
        best_seq = a_small.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score_small:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale & long gradient descent (n≈2400)
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_large = 2400
    a0_large = _upsample(best_seq if best_seq is not None else a_small, n_large)

    remaining = deadline - time.time()
    grad_time = max(10.0, 0.55 * remaining)       # reserve ~45 % for later phases
    grad_deadline = time.time() + grad_time

    a_large, score_large = _run_gradient_descent(
        a0_large, n_large, grad_deadline, start_time,
        beta=0.9, step0=0.6, max_iter=10**7
    )
    if score_large < best_score:
        best_score = score_large
        best_seq = a_large.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score_large:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – linear‑programming refinement (still at n≈2400)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + max(20.0, 0.30 * (deadline - time.time()))
        a_lp, score_lp = _lp_refinement(best_seq.copy(), lp_deadline, start_time,
                                        max_steps=12)
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP stage improve: score={score_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – local stochastic refinement (annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq.copy(), n_large, deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, steps=120000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback and final normalisation
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return plain Python list
    return [float(x) for x in best_seq]
