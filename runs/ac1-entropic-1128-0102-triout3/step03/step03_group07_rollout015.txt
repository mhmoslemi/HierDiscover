"""
Propose a high‑quality step‑function (non‑negative sequence) that minimizes

    2·n·max(a*a) / (sum(a)^2)

(the same quantity returned by `evaluate_sequence`).  

The algorithm consists of three stages:

1. **Parameter sweep** – generate a few hundred analytically‑motivated
   candidates (half‑binomial, power‑law, symmetric Beta‑shaped, etc.),
   evaluate them quickly with an FFT‑based convolution and keep the best.

2. **Smooth Adam optimisation on the probability simplex** – the current
   best candidate is re‑parameterised as a soft‑max of an unconstrained
   vector `u` (so that `a = softmax(u)` automatically satisfies
   `a_i ≥ 0` and `∑a_i = 1`).  A smooth approximation of the
   max‑operator (soft‑max with a growing temperature `β`) provides a
   differentiable objective.  Adam’s moment estimates are used together
   with a Euclidean projection (via the simplex re‑parameterisation) to
   descend for the remaining time budget.

3. **Final random perturbations** – a short hill‑climbing phase adds a
   tiny Gaussian noise to the best vector and accepts the move if it
   improves the objective.  This helps to escape any remaining flat
   regions.

All steps respect the 1000 s time limit; the best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def half_binomial_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑½, normalized to sum to 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def power_law_sequence(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+shift+1)^{-p}, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def beta_shape_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped density on {0,…,n‑1}: a_i ∝ i^{α‑1}(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    # protect against NaNs from negative exponents on zeros
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def symmetric_power_law_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^{-p}, symmetric w.r.t. centre."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) * (n - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def compute_score(a: np.ndarray, Nfft: int, L: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) for a vector a that already
    sums to 1.  Uses FFT based convolution.
    """
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[:L]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation routine

def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    n = 6000                       # length of the sequence
    L = 2 * n - 1                  # length of linear auto‑convolution
    Nfft = next_pow_two(3 * n - 2)  # size sufficient for correlation steps

    # ------------------------------------------------------------------
    # 1️⃣  Parameter sweep – generate a diverse set of analytically motivated candidates.
    best_a = None
    best_score = np.inf

    def try_candidate(a_vec: np.ndarray, label: str) -> None:
        nonlocal best_a, best_score
        score = compute_score(a_vec, Nfft, L, n)
        if score < best_score:
            best_score = score
            best_a = a_vec.copy()
            print(f"[init] {label:25s} → score {best_score:.6f}")

    # half‑binomial (classic start)
    try_candidate(half_binomial_series(n), "half‑binomial")

    # power‑law families
    for p in np.linspace(0.30, 1.10, 17):
        for shift in (0.0, 0.5, 1.0, 2.0):
            cand = power_law_sequence(n, p, shift)
            try_candidate(cand, f"pow p={p:.2f} sh={shift}")

    # symmetric power‑law (both ends heavy)
    for p in np.linspace(0.30, 0.80, 11):
        cand = symmetric_power_law_sequence(n, p)
        try_candidate(cand, f"sym pow p={p:.2f}")

    # Beta‑shaped densities
    al_vals = np.linspace(0.30, 2.00, 9)
    be_vals = np.linspace(0.30, 2.00, 9)
    for alpha in al_vals:
        for beta in be_vals:
            cand = beta_shape_sequence(n, alpha, beta)
            try_candidate(cand, f"beta a={alpha:.2f} b={beta:.2f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimisation on the simplex (soft‑max + pure max phases)
    # Re‑parameterise a = softmax(u) to enforce the constraints automatically.
    u = np.log(best_a + 1e-16)               # initialise u
    m = np.zeros_like(u)                    # first moment
    v = np.zeros_like(u)                    # second moment

    # Adam hyper‑parameters
    step_size = 0.35
    step_decay = 0.9997
    step_min = 5e-6
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Soft‑max temperature schedule
    beta_soft = 1.0
    beta_target = 2500.0
    beta_mul = 1.004

    phase = 1            # 1 → soft‑max, 2 → max sub‑gradient
    it = 0
    last_improve = 0
    restart_gap = 150_000   # if stuck for a long time, perform a tiny random jitter

    deadline = time.time() + budget_s - 8.0   # safety margin for final post‑processing

    while time.time() < deadline:
        # --------------------------------------------------------------
        # a from current u
        a = np.exp(u)
        a_sum = a.sum()
        a = a / a_sum                        # now a lies on the simplex

        # --------------------------------------------------------------
        # convolution and current objective
        A = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A * A, n=Nfft)
        conv = conv_full[:L]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:07d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # compute gradient w.r.t. a
        if phase == 1:                               # soft‑max (smooth) gradient
            shift = max_conv
            w_exp = np.exp(beta_soft * (conv - shift))
            w_sum = w_exp.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w_exp / w_sum

            # correlation a * w_rev   (gradient = 2 * correlation)
            Wr = np.fft.rfft(w[::-1], n=Nfft)
            corr_full = np.fft.irfft(A * Wr, n=Nfft)
            start = L - 1
            grad_a = 2.0 * corr_full[start:start + n]

            # update temperature
            beta_soft = min(beta_target, beta_soft * beta_mul)
            if beta_soft >= beta_target:
                phase = 2                               # switch to pure max phase
        else:                                          # pure max sub‑gradient
            k = int(conv.argmax())
            grad_a = np.zeros_like(a)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = k - idx
            grad_a[idx] = 2.0 * a[jdx]

        # --------------------------------------------------------------
        # transform gradient to the unconstrained u‑space:
        #  ∂f/∂u = a * (∂f/∂a – (a·∂f/∂a))
        scalar = np.dot(a, grad_a)
        grad_u = a * (grad_a - scalar)

        # --------------------------------------------------------------
        # Adam update on u
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step_size * m_hat / (np.sqrt(v_hat) + eps)
        u -= step

        # decay learning‑rate
        step_size = max(step_min, step_size * step_decay)

        # --------------------------------------------------------------
        # occasional tiny random jitter to avoid stagnation
        if rng.random() < 0.004:
            u += rng.normal(scale=0.02, size=n)

        # --------------------------------------------------------------
        # simple restart‑like perturbation if no improvement for a long while
        if it - last_improve > restart_gap:
            # add a small random direction while staying on the simplex
            delta = rng.normal(loc=0.0, scale=0.005, size=n)
            a_pert = a + delta
            a_pert = np.clip(a_pert, 0, None)
            a_pert /= a_pert.sum()
            u = np.log(a_pert + 1e-16)
            m.fill(0)
            v.fill(0)
            last_improve = it
            print(f"[iter {it:07d}] jitter reset – score≈{compute_score(a_pert, Nfft, L, n):.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final short hill‑climbing phase (pure random perturbations)
    remaining = deadline - time.time()
    if remaining > 2.0:               # allocate at most a couple of seconds
        a = best_a.copy()
        move_scale = 0.002 * a.mean()   # typical step size
        while time.time() < deadline:
            noise = (rng.random(n) - 0.5) * move_scale
            cand = a + noise
            cand = np.clip(cand, 0, None)
            if cand.sum() == 0.0:
                continue
            cand /= cand.sum()
            sc = compute_score(cand, Nfft, L, n)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                a = cand
                print(f"[hill] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety evaluation (using the provided routine)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] best reported score = {final_score:.6f}")

    return best_a.tolist()
