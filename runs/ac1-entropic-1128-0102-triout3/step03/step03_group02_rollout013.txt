def _subgradient_refine(a, max_iters, deadline, step0=0.1, beta=0.9):
    v= np.zeros_like(a) # momentum
    best_a=a.copy()
    best_score=_score_of_seq(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        #compute conv and max
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af*Af, n=fft_len)[:2*n-1]
        cur_max = conv.max()
        cur_score = 2*n*cur_max
        if cur_score < best_score:
            best_score=cur_score
            best_a=a.copy()
        # find max indices (within tolerance)
        tol = 1e-12
        max_idxs = np.where(conv >= cur_max - tol)[0]
        grad = compute_gradient_max(a, max_idxs)  # an array shape n
        # step schedule
        step = step0 / math.sqrt(it)
        v = beta*v - step * grad
        a = _project_simplex(a + v)
    return a, best_a, best_score

"""
Propose a low‑score step‑function (non‑negative sequence).

Algorithm
---------
1. Build a small pool of diverse initial shapes (uniform, triangular,
   Hann, a random Dirichlet and, if available, the previous best
   `height_sequence_1`).  All are projected onto the probability simplex
   (sum = 1) so the evaluation becomes scale invariant.

2. For each candidate run a short *soft‑max projected gradient descent*
   (the same routine that gave the previous state‑of‑the‑art results) to
   obtain a reasonable starting shape.

3. Refine the best shape with a *sub‑gradient method* that uses the exact
   gradient of the maximal auto‑convolution entry (the index where
   `conv(a,a)` attains its maximum).  The update is a Nesterov‑style
   momentum step followed by Euclidean projection onto the simplex.
   This phase runs for as many iterations as the remaining time budget
   allows (hundreds of thousands of cheap FFT‑based steps).

4. Occasionally improve the sequence by solving a tiny linear programme:
   maximise the total mass `∑ g_i` subject to `a * g ≤ M` (where `M`
   is the current maximal convolution value) and `g ≥ 0`.  The solution
   is normalised back to the simplex and blended with the current sequence.
   A few LP steps rapidly push mass into positions where `a` is small,
   lowering the maximal convolution.

5. A final sub‑gradient sweep finishes the optimisation.

The whole routine respects the 1000 s budget by checking the deadline
after each iteration and aborting early if needed.  The returned list
contains plain Python floats and satisfies the non‑negativity requirement.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Score 2·n·max(conv(a,a)) for a vector that already sums to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _softmax_refine(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.6,
    eps_start: float = 0.05,
    eps_end: float = 0.001,
    beta: float = 0.9,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Accelerated soft‑max projected gradient descent on a fixed length.
    Returns (final_a, best_seen_a, best_score).
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution & current score
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights over convolution entries
        eps_rel = eps_start + (eps_end - eps_start) * (it / max_iters)
        tau = max(eps_rel * cur_max, 1e-14)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.shape[0])

        # gradient of smooth approximation
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        g = 2.0 * conv2[n - 1 : n - 1 + n]

        # Nesterov‑style momentum & projection
        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)

    return a, best_a, best_score


def _max_indices(conv: np.ndarray, atol: float = 1e-12) -> np.ndarray:
    """Return indices where conv attains its maximum (within atol)."""
    mx = conv.max()
    return np.where(conv >= mx - atol)[0]


def _gradient_max_convolution(a: np.ndarray, max_inds: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a * a)[k] w.r.t. a.
    For each maximal index k we add 2 * a shifted by k.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_inds:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        g[idx_i] += 2.0 * a[idx_j]
    return g


def _subgradient_refine(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.8,
    beta: float = 0.9,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Sub‑gradient descent using the exact gradient at the maximal convolution
    entry, with momentum and simplex projection.
    Returns (final_a, best_seen_a, best_score).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        max_inds = _max_indices(conv)
        grad = _gradient_max_convolution(a, max_inds)

        step = step0 / math.sqrt(it)
        v = beta * v - step * grad
        a = _project_simplex(a + v)

    return a, best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low `evaluate_sequence` score.
    The routine respects the supplied time budget (default 1000 s) and
    prints progress messages.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a working length (the current best was found for n≈1800)
    # ------------------------------------------------------------------
    n = 1800

    # ------------------------------------------------------------------
    # 2. Assemble a pool of diverse starting points (all on the simplex)
    # ------------------------------------------------------------------
    candidates: list[np.ndarray] = []

    # a) previous best if available
    try:
        prev = np.array(height_sequence_1, dtype=float)
        prev = np.maximum(prev, 0.0)
        if prev.size != n:
            # linear interpolation to the target length
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, prev)
        s = prev.sum()
        if s > 0:
            prev = prev / s
            candidates.append(prev)
    except Exception:
        pass

    # b) uniform
    candidates.append(np.full(n, 1.0 / n))

    # c) triangular (tent)
    idx = np.arange(n)
    tri = np.minimum(idx + 1, n - idx)
    candidates.append(tri / tri.sum())

    # d) Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
    candidates.append(hann / hann.sum())

    # e) a random Dirichlet start
    candidates.append(np.random.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 3. Light soft‑max refinement of every candidate
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for i, cand in enumerate(candidates, 1):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.2f}s] Soft‑max start {i}/{len(candidates)}")
        _, local_best, local_score = _softmax_refine(
            cand,
            max_iters=20000,
            deadline=deadline,
            step0=0.6,
            eps_start=0.05,
            eps_end=0.001,
            beta=0.9,
        )
        if local_score < best_score:
            best_score = local_score
            best_seq = local_best.copy()
            print(f"    New global best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Large sub‑gradient sweep (as many iterations as time permits)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    remaining = deadline - time.time()
    if remaining > 10.0:
        # allocate roughly half of the remaining budget to this phase
        sub_deadline = time.time() + remaining * 0.5
        max_iters = 200000                       # upper bound, will stop earlier if out of time
        print(f"[{time.time() - start_time:.2f}s] Sub‑gradient refinement (≤{max_iters} iters)")
        _, sub_best, sub_score = _subgradient_refine(
            best_seq,
            max_iters=max_iters,
            deadline=sub_deadline,
            step0=0.8,
            beta=0.9,
        )
        if sub_score < best_score:
            best_score = sub_score
            best_seq = sub_best.copy()
            print(f"    Sub‑gradient improved best score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional linear‑programming boost (if SciPy is available)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
        have_lp = True
    except Exception:
        have_lp = False

    if have_lp:
        # run a few LP‑based updates while time permits
        while time.time() < deadline - 5.0:
            # current maximal convolution value M
            fft_len = 1 << ((2 * n - 1).bit_length())
            Af = np.fft.rfft(best_seq, n=fft_len)
            conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
            M = conv.max()

            # Build matrix A (2n‑1) × n such that A @ g = conv(best_seq, g)
            rows = 2 * n - 1
            A = np.zeros((rows, n), dtype=float)
            # column j influences rows j … j+n‑1 with coefficients best_seq[0 …]
            for j in range(n):
                start = j
                length = min(n, rows - start)
                A[start:start + length, j] = best_seq[:length]

            # Linear programme: maximise ∑ g_i  ⇔ minimise −∑ g_i
            c = -np.ones(n)
            bounds = [(0, None)] * n
            res = linprog(
                c,
                A_ub=A,
                b_ub=np.full(rows, M),
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": max(0.5, (deadline - time.time()) / 2)},
            )
            if not res.success:
                print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
                break

            g0 = res.x
            g_sum = g0.sum()
            if g_sum == 0:
                break
            g = g0 / g_sum                     # bring back to the simplex

            # blend a bit towards g
            t = 0.2
            new_seq = (1.0 - t) * best_seq + t * g
            new_seq = np.maximum(new_seq, 0.0)
            new_seq /= new_seq.sum()

            # evaluate the new sequence
            Af = np.fft.rfft(new_seq, n=fft_len)
            conv_new = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
            new_score = 2.0 * n * conv_new.max()
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq
                print(f"[{time.time() - start_time:.2f}s] LP step improved score to {best_score:.6f}")
            else:
                # if no improvement, shrink the blending factor and retry
                t *= 0.5
                if t < 1e-4:
                    break

    # ------------------------------------------------------------------
    # 6. Final fine‑tuning with sub‑gradient (using whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:

        _, final_best, final_score = _subgradient_refine(
            best_seq,
            max_iters=final_iters,
            deadline=deadline,
            step0=0.6,
            beta=0.9,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_best.copy()
            print(f"    Final sweep improved best score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Normalise one last time and return plain floats
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    print(f"[{time.time() - start_time:.2f}s] Finished. Best score (2·n·max(conv)) = {best_score:.6f}")
    return [float(x) for x in best_seq]
