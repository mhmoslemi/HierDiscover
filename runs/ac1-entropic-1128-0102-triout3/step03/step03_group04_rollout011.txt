def compute_conv(a):
    n = len(a)
    m = 2*n-1
    X = np.fft.rfft(a, n=m)

   # compute softmax weights
   maxc = conv.max()
   w = np.exp(beta * (conv - maxc))

def weighted_gradient(a, conv, beta):
    # conv shape m = 2*n-1.
    maxc = conv.max()
    # compute weights as softmax of (conv - maxc) * beta.
    # Use stable exponent.
    exp_vals = np.exp(beta * (conv - maxc))
    w = exp_vals / exp_vals.sum()
    # compute convolution of a and w_rev
    w_rev = w[::-1]
    n = a.shape[0]
    # convolution length = n + (2n-1) - 1 = 3n-2.
    # use rfft for speed.
    L = 1
    while L < n + len(w_rev) - 1:
        L <<= 1
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)
    conv_full = np.fft.irfft(A * W, n=L)
    # slice gradient:
    grad = 2.0 * conv_full[n-1:2*n-1][::-1]
    return grad

Potential numerical issues: w values may be near zero; but we compute softmax over all. Beta controls temperature: small beta = soft smoothing emphasizing all entries; large beta focuses on peak.

"""
Propose a step‑function (non‑negative vector) with a low
``evaluate_sequence`` score.

The search consists of three stages:

1.  **Parameterized sweep** – a few simple families (power‑law,
    exponential, logistic) are scanned on a coarse grid.  The best
    vector found (normalised to L1‑norm 1) seeds the local search.

2.  **Smooth‐max Adam / exponentiated‑gradient descent** – the objective
    ``max(conv(a,a))`` is approximated by a softmax (log‑sum‑exp) with a
    temperature that is slowly increased.  The gradient of the smooth
    surrogate is obtained analytically via a second FFT, giving an
    O(n log n) update.  Adam‑style moments (or a multiplicative‑weights
    update) handle the simplex constraint without an explicit projection.

3.  **Multiple restarts** – the descent is started from the best sweep
    point and several random Dirichlet seeds.  The best vector over all
    runs is finally evaluated with ``evaluate_sequence`` and returned.

The whole routine respects a user‑supplied time budget (default ≈950 s)
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+offset)^(–p) , normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(‑lam·i) , normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def logistic_sequence(n: int, alpha: float, shift: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp(alpha·(i‑shift))) , normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(alpha * (i - shift)))
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def compute_conv_and_weighted_grad(a: np.ndarray,
                                   beta: float,
                                   L: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Return (conv, grad) where
        conv = a * a  (full linear convolution, length 2·n‑1)
        grad = ∇   ( (1/β)·log∑_k exp(β·conv_k) )
    using FFTs of length L (L ≥ 3·n‑2). ``beta`` is the soft‑max temperature.
    """
    n = a.shape[0]
    # FFT of a (zero‑padded to L)
    A = np.fft.rfft(a, n=L)

    # ---- convolution a*a (full length L) ----
    conv_full = np.fft.irfft(A * A, n=L)
    m = 2 * n - 1                     # length of the true convolution
    conv = conv_full[:m]

    # ---- soft‑max weights (stable) ----
    maxc = conv.max()
    # shift by maxc to avoid overflow
    exp_vals = np.exp(beta * (conv - maxc))
    w = exp_vals / exp_vals.sum()    # shape (m,)

    # ---- gradient of the smooth max ----
    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, n=L)      # FFT of the reversed weight vector
    grad_full = np.fft.irfft(A * W, n=L)   # length L
    # slice indices n‑1 … 2·n‑2 (inclusive), then reverse
    grad = 2.0 * grad_full[n - 1:2 * n - 1][::-1]
    return conv, grad

            m_hat = m / (1.0 - adam_beta1 ** it)
            v_hat = v / (1.0 - adam_beta2 ** it)

            # learning‑rate decay (inverse‑sqrt schedule)
            lr = max(lr_initial / np.sqrt(it), lr_min)

            # multiplicative (exponentiated) mirror step
            step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
            # protect against huge steps – cap at 90 % of current value
            step = np.minimum(step, a * 0.90)

            # exponentiated‑gradient style update
            exp_arg = -step
            exp_arg = np.clip(exp_arg, -30.0, 30.0)   # avoid overflow/underflow
            a_new = a * np.exp(exp_arg)
            a_new = np.clip(a_new, 1e-30, None)      # keep strictly positive
            a_new /= a_new.sum()                     # simplex projection

            # occasional tiny jitter to escape flat plateaux
            if rng.random() < 0.001:
                jitter = (rng.random(n) - 0.5) * 1e-7
                a_new = project_onto_simplex(a_new + jitter)

            # ----- evaluate new candidate -----
            conv_new, _ = compute_conv_and_weighted_grad(a_new, beta, L)
            new_score = 2.0 * n * conv_new.max()

            # accept if improved, otherwise keep old point
            if new_score <= cur_score:
                a = a_new
                cur_score = new_score
            else:
                # small probability to accept worse moves (simulated‑annealing flavour)
                temperature = 0.02 * np.exp(-it / 180_000.0)
                if rng.random() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
                    a = a_new
                    cur_score = new_score

            # record the best over all restarts
            if cur_score < global_best_score:
                global_best_score = cur_score
                global_best_a = a.copy()
                print(f"[restart {restart:02d} iter {it:07d}] global best = {global_best_score:.6f}")

            # update temperature schedule
            beta = min(beta + beta_inc, beta_target)

        print(f"[restart {restart}] finished – local best score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final check with the official evaluator
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")
    print(f"[info] best coarse‑grid parameters: score={best_score:.6f}")

    return global_best_a.tolist()
