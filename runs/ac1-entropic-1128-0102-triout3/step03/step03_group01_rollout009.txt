    step0 = 0.5

        it += 1
        conv = _conv_self(a)

            best_a = a.copy()

"""
Multi‑stage optimisation for the step‑function bound.

1. Initialise from a supplied solution (``height_sequence_1``) or from a random
   simplex vector.
2. Work on a schedule of increasing lengths (up‑sampling by linear interpolation).
3. For each length:
   * Smoothed log‑sum‑exp gradient descent on the simplex (as in the reference
     implementation) – fast convergence to a good basin.
   * A short sub‑gradient phase that uses the *exact* max‑convolution gradient
     (average of the active constraints) – refines the solution beyond the
     smoothing bias.
   * A cheap simulated‑annealing sweep that moves small amounts of mass between
     two coordinates – polishes the final result.
4. The best vector seen over all lengths is returned.

All operations respect the simplex (non‑negative, sum = 1) and the routine
stops automatically when the time budget (``budget_s``) is exhausted.
"""

import time
import math
import random
from typing import List

import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real‑to‑real)."""
    size = x.size + y.size - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length ``seq`` to ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate.
    Parameters
    ----------
    a   : np.ndarray (n,)
    conv: np.ndarray (2n‑1,)
    lam : smoothing parameter λ>0
    Returns
    -------
    grad: np.ndarray (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)           # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """Smoothed (log‑sum‑exp) gradient descent on the simplex."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    beta = 0.9
    step0 = 0.8
    it = 0

    lam_factor = 0.25          # initial λ = lam_factor * current max
    lam_decay = 0.5
    lam_decay_interval = 2_000

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, "
                  f"it={it}, score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _run_subgrad_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(a*a) with projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    step0 = 0.5
    it = 0
    # small tolerance for picking the active set
    rel_eps = 1e-9

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"it={it}, score={cur_score:.6f}")

        # identify active indices (within relative tolerance)
        eps = max_b * rel_eps
        active = np.where(conv >= max_b - eps)[0]
        # cap the active set size to keep work modest
        if active.size > 10:
            active = np.argsort(-conv)[:10]

        # gradient = average of (a + shift_rev) over active set
        grad = np.zeros_like(a)
        active_cnt = float(active.size)
        grad += a * active_cnt                     # sum of the “a” part

        idx_range = np.arange(n)
        for k in active:
            idx = k - idx_range
            mask = (idx >= 0) & (idx < n)
            grad[mask] += a[idx[mask]]

        grad /= active_cnt

        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 50_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing (move ε from i to j)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        # keep simplex property – projection is unnecessary because we preserve sum

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, "
                  f"it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm combines multi‑scale
    smoothed‑gradient descent, exact sub‑gradient refinement and a final
    annealing sweep, staying within the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter sequence (if provided by the harness)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – geometric growth, modest to keep within time budget.
    lengths = [800, 1600, 3200, 4800, 6400, 9600, 12800]
    random.shuffle(lengths)          # avoid systematic bias if we stop early

    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---------------------------------------------
        # initialise a0 for this scale
        # ---------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # give a minimum of a few seconds to every scale, otherwise split proportionally
        slice_budget = max(8.0, 0.12 * remaining)
        slice_deadline = time.time() + slice_budget

        # -------------------------------------------------
        # 1) Smoothed gradient descent (fast basin search)
        # -------------------------------------------------
        a_smooth, score_smooth = _run_smooth_descent(a0, n, slice_deadline, start_time)

        if score_smooth < best_score:
            best_score = score_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after gradient: "
                  f"n={n}, score={score_smooth:.6f}")

        # -------------------------------------------------
        # 2) Exact sub‑gradient refinement (tighten bound)
        # -------------------------------------------------
        if time.time() < deadline:
            # give a smaller slice for sub‑gradient stage
            sub_deadline = time.time() + max(6.0, 0.08 * (deadline - time.time()))
            a_sub, score_sub = _run_subgrad_descent(a_smooth, n, sub_deadline, start_time)
            if score_sub < best_score:
                best_score = score_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad: "
                      f"n={n}, score={score_sub:.6f}")

        # -------------------------------------------------
        # 3) Final annealing polish (optional)
        # -------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(5.0, 0.04 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=50_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: "
                      f"n={n}, score={score_ref:.6f}")

        # remember the best distribution for the next upscale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return as plain Python list
    return [float(x) for x in best_seq]
