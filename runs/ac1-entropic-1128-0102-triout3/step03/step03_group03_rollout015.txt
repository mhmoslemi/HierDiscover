"""
Propose a step‑function via a three‑phase optimisation:

1. **Progressive up‑sampling** – start from a known construction (or a random
   simplex point) and repeatedly double the resolution while performing a fast
   exponentiated‑gradient descent that only looks at a handful of “active”
   convolution entries (those close to the current maximum).  This quickly
   produces a decent coarse shape.

2. **Refined smooth‑max descent** – on the final length we minimise a smooth
   approximation of the L∞‑norm of the convolution:
        Sβ(a) = (1/β)·log∑ₖ exp(β·(a∗a)ₖ) .
   For a fixed β the gradient of Sβ can be written in closed form and is
   evaluated by an FFT‑based cross‑correlation (O(n log n)).  The temperature β
   is increased slowly so that the optimisation eventually targets the true
   max‑convolution.  The update is multiplicative (mirrored‑gradient) which
   automatically enforces non‑negativity; after each step we renormalise the
   vector to sum to one.

3. **Cheap stochastic local search** – finally we perform many random
   mass‑transfer moves (move a small fraction of the mass from one index to
   another) and keep any improvement.  This can squeeze a little extra
   performance out of the already‑good candidate.

The routine respects a time budget (default 1000 s) and always returns a
non‑negative list whose sum is 1, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (all top‑level, no closures)
# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """
    Full auto‑convolution of a with itself and the maximum entry.
    Returned ``conv`` has length 2·n‑1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 12):
    """Select a small active set of convolution positions."""
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of max(conv) using only the active indices K.
    For each active index k we add 2·a_{k‑i} to the i‑th component whenever
    the pair (i,k‑i) is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _gradient_smooth(a: np.ndarray, w: np.ndarray):
    """
    Gradient of the smooth‑max (soft‑max) of the convolution.
    It is 2·(a ⋆ w) (cross‑correlation).  Computed via FFT in O(n log n).
    """
    n = a.shape[0]            # length of a
    m = w.shape[0]            # length of w (=2·n‑1)
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w, n=fft_len)
    # linear cross‑correlation a ⋆ w  (non‑circular because of zero padding)
    corr = np.fft.irfft(A * np.conj(W), n=fft_len)
    start = m - 1            # index that corresponds to lag 0
    g = 2.0 * corr[start:start + n]
    return g

    ``evaluate_sequence`` score.  The routine respects the given time budget.
    """

        n0 = 1200
        init = np.random.rand(n0)
        init = init / init.sum()
        best_seq = init.copy()

    # --------------------------------------------------------------
    # 2. Progressive up‑sampling with a cheap exponentiated‑gradient.
    # --------------------------------------------------------------

    cur_n = cur_seq.shape[0]
    MAX_LEN = 8000                     # final length we aim for
    eta0 = 0.7
    rel_eps = 0.001
    max_add = 12                       # keep active set modest

        x_new = np.linspace(0.0, 1.0, next_n)

        for it in range(1, max_iter + 1):

            cur_score = 2.0 * next_n * max_val

                best_seq = a.copy()

            K = _active_indices(conv, max_val, rel_eps=rel_eps,
                                max_add=max_add)
            g = _gradient_active(a, K)

            # multiplicative (mirror) update
            eta = eta0 / math.sqrt(it)
            a = np.maximum(a, 1e-15)
            log_a = np.log(a) - eta * g
            log_a -= np.max(log_a)            # stabilise

        cur_seq = a.copy()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    # Ensure we are exactly at the desired final length.
    if cur_n < MAX_LEN:

        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, MAX_LEN)
        a = np.interp(x_new, x_old, cur_seq)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        cur_n = MAX_LEN
    else:
        a = cur_seq.copy()

    n = cur_n
    beta = 5.0                     # start temperature for soft‑max
    alpha0 = 0.6                   # base step size for exponentiated update
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1

        # convolution and smooth‑max weights
        conv, max_val = _conv_and_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] SmoothIter {iter_cnt:,} "
                  f"score={cur_score:.6f}")

        # soft‑max (smooth‑max) weighting of convolution entries
        # shifting by max_val keeps the exponent well‑behaved
        exp_weights = np.exp(beta * (conv - max_val))
        weight_sum = exp_weights.sum()
        if weight_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_weights / weight_sum

        # gradient of the smooth‑max (via cross‑correlation)
        g = _gradient_smooth(a, w)

        # multiplicative update (mirrored gradient descent)
        alpha = alpha0 / math.sqrt(iter_cnt)
        a = np.maximum(a, 1e-15)
        log_a = np.log(a) - alpha * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a[a < 1e-15] = 1e-15
        a /= a.sum()

        # slowly tighten the temperature – after every few thousand steps
        if iter_cnt % 2000 == 0 and beta < 200.0:
            beta *= 1.25

    # --------------------------------------------------------------
    # 4. Final cheap stochastic local search (mass transfers).
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    max_local_iters = 60000
    local_it = 0
    while time.time() < deadline and local_it < max_local_iters:
        local_it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a small random fraction from i to j
        eps = np.random.rand() * 0.04          # up to 4 % of source mass
        delta = eps * a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        conv, max_val = _conv_and_max(a_new)
        new_score = 2.0 * n * max_val
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            print(f"[{time.time() - start_time:6.2f}s] RandomMove it={local_it:,} "
                  f"score={new_score:.6f}")

    # 5. Return the best sequence found (already normalised to sum 1).
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s == 0.0:
        best_seq = np.full(100, 1.0 / 100)
    else:
        best_seq = best_seq / s

    return [float(x) for x in best_seq]
