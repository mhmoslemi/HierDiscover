            else:

                x_old = np.linspace(0,1,arr.shape[0])

        else:
            a = np.random.rand(n)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        
        # hyperparameters
        max_iters = 50000 # for each length
        eta0 = 0.5

                break
            # compute conv and objective
            conv = fft_convolve(a, a)
            max_conv = conv.max()
            cur_score = float(2 * n * max_conv)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time()-start:.2f}s] New best n={n} it={it} score={cur_score:.6f}")
            
            # annealing beta
            beta = min(beta_max, beta0 * (beta_factor ** it))
            # compute softmax weights
            # subtract max for stability
            shift = max_conv

def fft_convolve(x, y):
    n = len(x) + len(y) - 1
    fft_len = 1 << (n-1).bit_length()
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X*Y, n=fft_len)[:n]
    return conv
```

Propose a candidate step‑function using a soft‑max smoothed max‑convolution
objective and exponentiated‑gradient (mirror‑descent) on the probability
simplex.  The algorithm proceeds by progressive up‑sampling of a base shape,
a temperature (β) schedule that slowly turns the smooth surrogate into the
true L∞‑norm, and a final short stochastic local‑search phase.  The
implementation is fully vectorised and relies on FFT‑based convolutions,
so many thousands of iterations fit easily into the 1000 s budget.
"""
import time
import math
import numpy as np

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution of two 1‑D real arrays using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    L = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv

def _auto_conv(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution a*a (length 2·len(a)-1)."""
    return _fft_convolve(a, a)

def _grad_softmax(a: np.ndarray, beta: float, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective (log‑sum‑exp of the convolution)
    with respect to the probability vector a.
    """
    n = a.shape[0]
    # soft‑max weights, stabilised by the maximum convolution entry
    max_val = conv.max()
    exp_vals = np.exp(beta * (conv - max_val))
    p = exp_vals / exp_vals.sum()          # shape (2n-1,)

    # convolution p * a_rev, then extract the middle n entries
    a_rev = a[::-1]
    full = _fft_convolve(p, a_rev)        # length 3n-2
    grad = 2.0 * full[n - 1 : 2 * n - 1]   # length n
    return grad

  replace the hard max by a log‑sum‑exp (soft‑max) with a temperature β.
* β starts small (the surrogate is close to an average) and is linearly
  annealed to a large value (≈ 180) so that the surrogate converges to the
  true L∞‑norm.
* Updates are performed with exponentiated‑gradient (mirror descent using
  KL‑divergence), i.e.   a_i ← a_i·exp(−η·∂f/∂a_i)  followed by a
  normalisation – this automatically keeps the vector non‑negative and on
  the simplex (∑a_i = 1).  The step size η decays as 1/√t.
* The optimisation proceeds progressively: start from a modest size,
  repeatedly up‑sample the current shape (linear interpolation) and run the
  mirror‑descent for a number of iterations that scales roughly with the
  inverse length.  This mirrors the successful “progressive up‑sampling”
  strategy used previously but uses a full‑gradient (soft‑max) instead of a
  tiny active set.
* After the gradient phase a cheap random mass‑transfer hill‑climbing
  (stochastic local search) is executed to polish the solution.
* All convolutions are performed with FFTs, so each iteration costs
  O(n log n) and thousands of iterations comfortably fit into the 1000 s
  budget.

The function respects the required signature and prints intermediate
improvements.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines – FFT based convolutions
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real 1‑D arrays using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = x.shape[0] + y.shape[0] - 1
    L = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv


def _auto_conv(a: np.ndarray) -> np.ndarray:
    """Auto‑convolution a*a (length 2·len(a)-1)."""
    return _fft_convolve(a, a)


def _grad_softmax(a: np.ndarray, beta: float, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective
        f(a) = (1/β)·log Σ_i exp(β·conv_i(a))
    with respect to the probability vector a (∑a_i = 1, a_i ≥ 0).
    Returns a vector of length len(a).
    """
    n = a.shape[0]

    # soft‑max weights, stabilised by the maximum entry of conv
    max_val = conv.max()
    exp_vals = np.exp(beta * (conv - max_val))
    p = exp_vals / exp_vals.sum()                 # shape (2n‑1,)

    # gradient = 2·(p * a_rev) evaluated on the central n entries
    a_rev = a[::-1]
    full = _fft_convolve(p, a_rev)                # length 3n‑2
    grad = 2.0 * full[n - 1 : 2 * n - 1]           # slice of length n
    return grad

def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights (a step function) with a low
    `evaluate_sequence` score.  The routine runs for at most ``budget_s``
    seconds and prints intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise – try to reuse a previously discovered construction.
    # ------------------------------------------------------------------

        init = np.maximum(init, 0.0)
        s = init.sum()
        if s == 0.0:

        init = init / s
        cur_seq = init.copy()
        cur_n = cur_seq.shape[0]
        print(f"Initialised from height_sequence_1, n={cur_n}")
    except Exception:
        cur_n = 512
        cur_seq = np.random.rand(cur_n)
        cur_seq = cur_seq / cur_seq.sum()
        print(f"Random initialisation, n={cur_n}")

    # store the best shape seen so far (always normalised to sum‑1)
    best_seq = cur_seq.copy()
    best_score = 2.0 * cur_n * _auto_conv(best_seq).max()
    print(f"Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling + mirror‑descent on a smoothed objective.
    # ------------------------------------------------------------------
    MAX_LEN = 5000
    # Build an increasing schedule of lengths (≈1.45× factor, capped at MAX_LEN)
    lengths = [cur_n]
    while lengths[-1] < MAX_LEN:
        nxt = int(lengths[-1] * 1.45)
        if nxt > MAX_LEN:
            nxt = MAX_LEN
        if nxt <= lengths[-1]:
            break
        lengths.append(nxt)

    for target_n in lengths[1:]:
        if time.time() > deadline:
            break

        # ----- up‑sample current shape to the new length -----
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, target_n)
        a = np.interp(x_new, x_old, cur_seq)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        cur_n = target_n

        # ----- hyper‑parameters for this size -----
        # allocate more iterations for smaller problems, but keep a lower bound
        max_iters = int(25000 * 5000 / cur_n) + 8000
        if max_iters < 15000:
            max_iters = 15000
        eta0 = 0.5                # base step for exponentiated‑gradient
        beta0 = 0.4               # temperature at iteration 0
        beta_max = 180.0          # final temperature (close to hard max)

        print(f"--- Optimising length {cur_n}, max_iters={max_iters} ---")

        # ----- mirror‑descent loop -----
        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break

            conv = _auto_conv(a)                # auto‑convolution a*a
            max_conv = conv.max()
            cur_score = 2.0 * cur_n * max_conv

            # keep the overall best solution seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] New best – n={cur_n} it={it} score={cur_score:.6f}")

            # linear temperature schedule (soft‑max → hard max)
            beta = beta0 + (beta_max - beta0) * it / max_iters

            # gradient of the smoothed objective
            grad = _grad_softmax(a, beta, conv)

            # exponentiated‑gradient (KL‑mirror descent) update
            eta = eta0 / math.sqrt(it)
            a = a * np.exp(-eta * grad)

            # enforce strict positivity and renormalise onto the simplex
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

        # keep the shape for the next up‑sampling stage
        cur_seq = a.copy()

        # ----- 3. Short stochastic local search on this size -----
        local_iters = 4000
        for _ in range(local_iters):
            if time.time() > deadline:
                break
            i, j = np.random.randint(0, cur_n, size=2)
            if i == j:
                continue
            eps = np.random.rand() * 0.04            # move up to 4 % of source mass
            delta = eps * a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            if a_new[i] < 0:
                continue
            conv_new = _auto_conv(a_new)
            new_score = 2.0 * cur_n * conv_new.max()
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                a = a_new
                print(f"[{time.time() - start_time:6.2f}s] Local improve – n={cur_n} it={_} score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence (plain Python floats).
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(100, 1.0 / 100)

    best_seq = np.maximum(best_seq, 0.0)    # safety
    return [float(v) for v in best_seq]
