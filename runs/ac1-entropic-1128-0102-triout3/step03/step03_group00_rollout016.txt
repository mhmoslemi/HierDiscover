"""
Propose a step‑function (non‑negative list) that minimizes `evaluate_sequence`.

Main ideas
---------
* Work on the probability simplex (∑a_i = 1) – the objective is homogeneous,
  so any scaling of a does not change the score.
* Use a fast FFT‑based linear convolution.
* Optimise the (smooth) max‑convolution with Adam.  The gradients are the
  exact sub‑gradients of the active convolution entries, optionally
  weighted by a softmax (temperature τ) so that nearby peaks also guide the
  descent.
* Run the optimiser on a small size, up‑sample the best candidate to a
  larger size and continue optimisation.  When time permits a still larger
  size is tried.
* Finish with a cheap simulated‑annealing walk that moves a tiny amount of
  mass between two coordinates.
* The whole pipeline respects the 1000 s time budget and keeps the best
  sequence ever seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex using the O(n) algorithm of
    Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active set version)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Gradient of the max‑convolution (averaged over the active set)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    act = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    if act.size:
        g /= float(act.size)
    return g


# ----------------------------------------------------------------------
# Smoothed sub‑gradient (softmax weighting of nearby peaks)
# ----------------------------------------------------------------------
def _smoothed_subgradient(a: np.ndarray, conv: np.ndarray,
                          tau_rel: float = 0.02) -> np.ndarray:
    """Weighted sub‑gradient where each convolution entry gets weight
    w_k ∝ exp((c_k – max)/τ).  τ = tau_rel * max(c)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    tau = max_val * tau_rel
    # avoid division by zero if max is tiny
    if tau <= 0.0:
        tau = 1e-12

    diff = conv - max_val
    # ignore entries far below the maximum (exponential under‑flow)
    cap = -10.0 * tau
    mask = diff >= cap
    if not np.any(mask):
        # fall back to the pure active set
        return _subgradient_full(a, conv)

    w = np.exp(diff[mask] / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _subgradient_full(a, conv)
    w /= w_sum

    grad = np.zeros_like(a)
    active = np.where(mask)[0]
    for wk, k in zip(w, active):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += wk * a[j]
        grad[j] += wk * a[i]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (projected Adam)
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int,
              deadline: float, start_time: float,
              step_size: float = 0.1,
              tau_rel: float = 0.02,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation with projection onto the simplex.
    Returns the best vector encountered and its score (2·n·max(conv)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it}, score={cur_score:.6f}")

        # smoothed gradient of the max‑convolution
        grad = _smoothed_subgradient(a, conv, tau_rel=tau_rel)

        # Adam update (without bias‑correction – sufficient for our purpose)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        step = step_size / (np.sqrt(v) + eps)
        a = a - step * m

        # keep the iterate on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny ε‑mass moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 80000) -> tuple[np.ndarray, float]:
    """A cheap Monte‑Carlo walk that moves a small amount of mass between
    two positions.  Accepts downhill moves and occasional uphill moves."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # move up to 10 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

        best_score = score_mid
        best_seq = a_mid.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score_mid:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 (optional): a still larger problem if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:   # keep at least ~30 s for the final step
        n_large = 4800
        a0_large = _upsample(best_seq, n_large)

        large_time = deadline - time.time() - 20.0   # reserve ~20 s for annealing
        large_deadline = time.time() + large_time

        a_large, score_large = _run_adam(
            a0_large, n_large, large_deadline, start_time,
            step_size=0.08, tau_rel=0.012, max_iter=10 ** 7
        )
        if score_large < best_score:
            best_score = score_large
            best_seq = a_large.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 3 improve: score={score_large:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – short simulated‑annealing refinement (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        cur_n = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq, cur_n, anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, steps=80000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny drift may occur because of floating‑point noise)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
