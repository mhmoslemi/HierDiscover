def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0
    best_seq = None
    best_score = float('inf')
    # lengths list
    lengths = [512, 1024, 2048, 4096, 8000]
    cur_seq = None
    for n in lengths:
        # upsample previous best to length n, or start uniform if none
        if cur_seq is not None:
            # upsample
            x_old = np.linspace(0, 1, len(cur_seq))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, cur_seq)
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            a = np.full(n, 1.0 / n)

        # generate multiple restarts
        for restart in range(3):
            if time.time() >= deadline:
                break
            # for first restart use deterministic a, else random perturb
            if restart == 0:
                cand = a.copy()
            else:
                # random Dirichlet then blend with a
                rand = np.random.dirichlet(np.ones(n))
                # weighted blend
                cand = 0.7 * a + 0.3 * rand
                cand = cand / cand.sum()
            # Softmax refinement
            cand, best_soft_a, best_soft_score = _refine_softmax(cand, max_iters=20000, deadline=deadline,
                                                                step0=0.6, eps_start=0.05, eps_end=0.001, beta=0.9)
            # Subgradient refinement (more iterations)
            cand, best_sub_a, best_sub_score = _refine_subgrad(cand, max_iters=40000, deadline=deadline,
                                                               step0=0.5, beta=0.9)
            # pick better of soft and sub
            local_best_a = best_sub_a if best_sub_score < best_soft_score else best_soft_a
            local_best_score = min(best_sub_score, best_soft_score)
            if local_best_score < best_score:
                best_score = local_best_score
                best_seq = local_best_a.copy()
                print(f"[{time.time() - start:6.2f}s] New global best at n={n}, score={best_score:.6f}")

        # after restarts, set cur_seq to upsample of best_seq at current n
        cur_seq = best_seq.copy()
        # grow to next length will be upsampled later

    # final fine-tune on best length (if not already at max)
    if cur_seq is not None and time.time() < deadline:
        n = cur_seq.shape[0]
        cur_seq, best_final_a, best_final_score = _refine_subgrad(cur_seq, max_iters=80000, deadline=deadline,

def _refine_softmax(a, max_iters, deadline, step0, eps_start, eps_end, beta):
    n = a.shape[0]
    fft_len_conv = 1 << ((2*n - 1).bit_length())
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_of_seq(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # temperature schedule
        tau = eps_start + (eps_end - eps_start) * (it / max_iters)
        # use log-sum-exp: weights w_i = exp((conv_i - max_val)/tau)
        w = np.exp((conv - max_val) / max(tau, 1e-14))
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.full_like(w, 1.0 / w.shape[0])
        # gradient
        a_rev = a[::-1]
        fft_len_grad = 1 << ((len(w) + n - 1).bit_length())
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[:len(w) + n - 1]
        # extract gradient for components 0..n-1
        g = 2.0 * conv2[n-1:n-1+n]
        # momentum update
        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)
    return a, best_a, best_score

def _refine_subgrad(a, max_iters, deadline, step0=0.5, beta=0.9):
    n = a.shape[0]
    fft_len_conv = 1 << ((2*n - 1).bit_length())
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_of_seq(a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[:2*n-1]
        max_val = conv.max()
        i_max = conv.argmax()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # exact subgradient for max
        g = np.zeros_like(a)
        k_start = max(0, i_max - (n - 1))
        k_end = min(i_max, n - 1)
        idx = np.arange(k_start, k_end + 1)
        g[idx] = 2.0 * a[i_max - idx]
        # momentum
        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)
        # occasional perturbation to avoid stagnation
        if it % 10000 == 0:
            # add tiny noise
            noise = np.random.normal(scale=1e-4, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
    return a, best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimises
    score = 2·n·max(conv(a,a)) / (sum(a)²)

The evaluation is scale‑invariant, so we keep the coefficients on the
probability simplex (sum = 1, a_i ≥ 0).  The optimiser works in three
phases:

1.  **Coarse optimisation** – a soft‑max approximation of the
    max‑function gives a smooth gradient.  Several diverse starts
    (uniform, random Dirichlet and the previous best construction) are
    refined on a moderate size (≈1 k‑2 k entries).

2.  **Fine optimisation** – once a promising shape is found we switch
    to the exact sub‑gradient of the true max (i.e. pick the index of
    the maximal convolution entry and move opposite to its gradient).
    A Nesterov‑style momentum term and a decreasing step size are used,
    together with occasional tiny random perturbations to escape
    plateaus.

3.  **Progressive up‑sampling** – the best shape is linearly
    interpolated to larger lengths (≈1.5× each time, up to 8000
    coefficients).  Every new resolution receives a budget of
    iterations proportional to its cost and is refined with the same
    two‑stage scheme.  The final length gets a long dedicated run.

All updates are projected onto the simplex with the Euclidean
projection of Duchi et al.; this guarantees feasibility at every step.
The routine prints intermediate improvements and finally returns the
best sequence found as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """
    Score for a sequence already normalised to sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _refine_softmax(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.6,
    eps_start: float = 0.05,
    eps_end: float = 0.001,
    beta: float = 0.9,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Accelerated projected gradient descent using a soft‑max
    approximation of the max‑operator.
    Returns (final_a, best_seen_a, best_seen_score).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    v = np.zeros_like(a)                     # momentum term

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution and current score -----
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights -----
        tau = eps_start + (eps_end - eps_start) * (it / max_iters)
        # safe temperature
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:                                   # fallback: uniform weights
            w = np.full_like(w, 1.0 / w.shape[0])

        # ----- gradient of the smooth objective -----
        a_rev = a[::-1]
        grad_len = len(w) + n - 1
        fft_len_grad = 1 << (grad_len.bit_length())
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[:grad_len]
        g = 2.0 * conv2[n - 1 : n - 1 + n]       # slice the part that matches a

        # ----- Nesterov momentum and projection -----
        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)

    return a, best_a, best_score


# ----------------------------------------------------------------------
def _refine_subgradient(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.5,
    beta: float = 0.9,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Projected sub‑gradient descent using the *exact* gradient of the
    max‑convolution entry (i.e. only the worst index contributes).
    Periodically adds a tiny random jitter to avoid stagnation.
    Returns (final_a, best_seen_a, best_seen_score).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    v = np.zeros_like(a)                     # momentum term

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution, max position and score -----
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        i_max = conv.argmax()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- exact sub‑gradient for the maximal entry -----
        g = np.zeros_like(a)
        k_start = max(0, i_max - (n - 1))
        k_end = min(i_max, n - 1)
        idx = np.arange(k_start, k_end + 1, dtype=int)
        g[idx] = 2.0 * a[i_max - idx]

        # ----- momentum, step‑size and projection -----
        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)

        # ----- occasional jitter to leave flat plateaus -----
        if it % 10000 == 0:
            jitter = np.random.normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

    return a, best_a, best_score
