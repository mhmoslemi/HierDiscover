"""
Proposed optimiser for the auto‑convolution peak.

The routine works on the probability simplex (∑ a_i = 1, a_i ≥ 0).

* Phase 1 – a smoothed objective (log‑sum‑exp) is minimised with a
  projected Adam optimiser.  The smoothing factor β is increased
  slowly so that the optimiser first sees a “soft” version of the max
  and then a true max once β is large.

* Phase 2 – after β reaches a large target we switch to a pure
  max‑sub‑gradient (the sub‑gradient of the max of the convolution).
  The same projected Adam scheme is used, only the gradient changes.

* Restarts – if the algorithm makes no improvement for a long stretch
  we restart from a fresh Dirichlet sample; this helps escaping bad
  basins.

* Throughout we keep the best sequence ever seen and return it at the
  end.  The code respects the 1000 s time budget (a safety margin of a
  few seconds is kept).
"""
import time
from typing import List

import numpy as np


def half_binomial_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^‑½   (a_i = C(2i,i)/4^i)
    and normalised so that they sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

    max‑convolution objective, followed by a pure max‑sub‑gradient phase.

    # Initial point – classic half‑binomial series (already on the simplex).
    a = half_binomial_series(n)

    # Adam state
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.08               # base learning rate
    lr_min = 0.001

    # Soft‑max (log‑sum‑exp) parameters
    beta = 1.0
    beta_target = 1500.0
    beta_mul = 1.003

    # Phase control
    phase = 1               # 1 = smoothed; 2 = pure max

    # Restart logic
    last_improve = 0
    restart_gap = 300_000   # iterations without improvement → restart

    # Timing
    deadline = time.time() + budget_s - 5.0   # keep a safety margin

    # Keep the best ever found
    # (objective = 2·n·max(conv) because a is already on the simplex)
    A_fft = np.fft.rfft(a, n=Nfft)
    conv_full = np.fft.irfft(A_fft * A_fft, n=Nfft)
    conv = conv_full[:L]
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    print(f"[init] n={n}, initial score={best_score:.6f}")

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) compute convolution (FFT based, linear part only)
        A_fft = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(A_fft * A_fft, n=Nfft)
        conv = conv_full[:L]                     # linear convolution
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # 2) record best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:09d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) gradient (different in the two phases)
        if phase == 1:          # soft‑max smoothing
            # stable soft‑max weights
            shift = max_conv
            w_exp = np.exp(beta * (conv - shift))
            w = w_exp / w_exp.sum()

            # correlation a ⋆ w  (gradient = 2·corr)
            Wr_fft = np.fft.rfft(w[::-1], n=Nfft)
            corr_full = np.fft.irfft(A_fft * Wr_fft, n=Nfft)
            start = L - 1
            grad = 2.0 * corr_full[start:start + n]

            # increase smoothing factor
            beta = min(beta_target, beta * beta_mul)
            if beta >= beta_target:
                phase = 2
        else:                  # pure max sub‑gradient
            k = int(conv.argmax())
            grad = np.zeros_like(a)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = k - idx
            grad[idx] = 2.0 * a[jdx]

        # --------------------------------------------------------------
        # 4) Adam update (projected)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_trial = a - step
        a = project_onto_simplex(a_trial)

        # --------------------------------------------------------------
        # 5) learning‑rate schedule (slow decay)
        lr = max(lr_min, lr * 0.9999)

        # --------------------------------------------------------------
        # 6) restart if we are stuck
        if it - last_improve > restart_gap:
            a = rng.dirichlet(np.full(n, 0.6))
            a = project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            # recompute current score for the log output
            A_fft = np.fft.rfft(a, n=Nfft)
            conv_full = np.fft.irfft(A_fft * A_fft, n=Nfft)
            cur_score = 2.0 * n * conv_full[:L].max()
            print(f"[iter {it:09d}] restart – new random start, score≈{cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final check with the provided evaluation routine
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
