"""
Improved multi‑scale optimization for the step‑function problem.

The algorithm works on the probability simplex (∑a_i = 1) and
minimises 2·n·max(conv(a,a)).  It consists of three stages:

1.  Multi‑scale warm‑starts – start with a modest length, optimise,
    then up‑sample to a finer resolution.
2.  Adam‑based smoothed‑gradient descent on the log‑sum‑exp surrogate
    of the ℓ∞‑norm.  The smoothing parameter λ is annealed, the step
    size is adapted by Adam, and after each step the vector is projected
    onto the simplex.
3.  A final cheap simulated‑annealing sweep (mass‑transfer moves) to
    polish the best candidate found.

The code follows the same time‑budget discipline as the reference
implementation but spends the majority of the budget on the largest
scale (up to 16384 coefficients) and uses a more aggressive optimiser,
which usually pushes the score below the target 1.503.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


# ----------------------------------------------------------------------
# Up‑sampling (linear interpolation) while keeping the sum = 1
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed ℓ∞ surrogate gradient (log‑sum‑exp) and Adam optimiser
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the LSE surrogate L_λ(a)=λ·log∑exp(conv/λ).
    Returns a dense gradient of shape (n,).
    """
    n = a.shape[0]
    # soft‑max of the convolution entries (stable)
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)                # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam‑based smoothed gradient descent on the simplex.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = float('inf')

    # λ‑schedule parameters
    lam_factor = 0.15          # start with a tighter surrogate
    lam_decay = 0.5
    lam_decay_interval = 2000

    # Adam base learning rate (tuned experimentally)
    alpha0 = 0.5

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        # gradient of the surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam update (per‑coordinate)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection back onto the probability simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 3e-2,
                   end_temp: float = 1e-6,
                   steps: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Simple coarse‑grained simulated annealing that moves ε mass between two
    coordinates.  Returns a possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / (T + 1e-12)) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
