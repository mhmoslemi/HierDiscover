def _focused_hill_climb(a, n, L_fft, deadline):
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())
        # compute contributions vector for each i
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k_max - idx]
        # find i of max contribution
        maxpos = np.argmax(contrib)
        i = int(idx[maxpos])
        j = int(k_max - i)
        src = i if best_a[i] >= best_a[j] else j
        # compute partner values for potential destination indices
        # partner weight w_d = best_a[k_max - d] for d such that 0 <= k_max - d < n
        # For indices where partner out of range, weight = 0
        partner = np.zeros(n)
        d_valid_start = max(0, k_max - (n - 1))
        d_valid_end = min(k_max, n - 1)
        d_idxs = np.arange(d_valid_start, d_valid_end + 1)
        partner[d_idxs] = best_a[k_max - d_idxs]
        # exclude src
        partner[src] = np.inf
        # choose dest as with smallest partner
        dst = int(np.argmin(partner))
        if best_a[src] < 1e-12:
            # can't move
            continue
        # amount to move: fraction of src
        delta = min(best_a[src] * 0.5, 0.005)   # 50% of src or 0.005
        # ensure positive
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # renorm due to any floating errors
        cand = np.maximum(cand, 0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand = cand / cand_sum
        new_score = _objective_score(cand, n, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] Hill src={src} dst={dst} delta={delta:.6f} score={new_score:.6f}")
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative vector) that minimizes
    score = 2·n·max(conv(a,a)) / (sum(a)²) .
The search consists of:

1️⃣ Warm‑start with many Dirichlet draws (and an optional global starter).
2️⃣ “p‑norm annealing” gradient descent – minimise the Lp‑norm of the
   self‑convolution for a sequence of growing p (2 → 4 → 8 → 16 → 32).
   The gradient of the p‑norm is obtained analytically via FFT.
3️⃣ Direct sub‑gradient descent on the maximum convolution entry.
4️⃣ Focused hill‑climb: repeatedly move a little mass from the larger
   member of the dominating pair (i,j) for the current peak to an index
   whose partner at that peak is smallest, accepting only improving moves.
5️⃣ Final refinement – convex mixing with the uniform distribution and a
   cheap random‑pair hill‑climb.

All iterates are projected onto the probability simplex (non‑negative,
sum = 1) using the Euclidean Duchi projection.  The routine respects a
hard 1000 s budget and prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used by helper functions for logging)
# ----------------------------------------------------------------------
start_time = None  # will be set in propose_candidate()


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (assumes sum(a)=1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 0️⃣ Warm‑start (Dirichlet draws, optional global starter)
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                L_fft: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Random Dirichlet draws with varying concentration."""
    while time.time() < deadline:
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        if sample.sum() == 0:
            continue
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 1️⃣ p‑norm annealing gradient descent
# ----------------------------------------------------------------------
def _pnorm_descent(a0: np.ndarray,
                   n: int,
                   L_fft: int,
                   L_corr: int,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the Lp‑norm of the self‑convolution for an increasing sequence
    of p (2 → 4 → 8 → 16 → 32).  Gradient is obtained analytically via FFT.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    # sequence of p’s (larger p approximates the true max)
    p_list = [2, 4, 8, 16, 32]

    # small epsilon to keep logarithms safe
    eps = 1e-15

    for p in p_list:
        if time.time() >= deadline:
            break
        it = 0
        eta0 = 0.7 / p                      # step‑size scales with p
        while time.time() < deadline:
            it += 1
            conv = _conv_self_fft(a, L_fft)  # length 2n‑1
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] p={p} iter={it}  score={cur_score:.6f}")

            # ---------------------------------------------------------
            # gradient of (∑ conv^p)^{1/p}  (the Lp‑norm)
            # ---------------------------------------------------------
            log_conv = np.log(conv + eps)               # safe log
            w = np.exp((p - 1) * log_conv)              # w_k = conv_k^{p‑1}
            # denominator = (∑ conv^p)^{(p‑1)/p}
            sum_p = np.sum(conv * w)                    # = ∑ conv_k^{p}
            denom = sum_p ** ((p - 1) / p) if sum_p > 0 else 1.0

            # convolution of reversed w with a
            w_rev = w[::-1]
            Dr = np.fft.rfft(w_rev, n=L_corr)
            Ar = np.fft.rfft(a, n=L_corr)
            full = np.fft.irfft(Dr * Ar, n=L_corr)
            start = n - 1
            grad_raw = 2.0 * full[start:start + n]      # shape (n,)
            grad = grad_raw / denom

            # classic 1/√t step‑size schedule
            eta = eta0 / np.sqrt(it + 1)
            a_new = a - eta * grad
            a = _project_simplex(a_new)

            # occasional jitter to escape plateaus
            if rng.random() < 0.003:
                a += rng.normal(scale=0.001, size=n)
                a = _project_simplex(np.maximum(a, 0.0))

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Direct sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------
def _max_subgradient_descent(a0: np.ndarray,
                             n: int,
                             L_fft: int,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true objective
    f(a) = max_k (a*a)_k .
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    it = 0
    eta0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # gradient = 2 * a_{k_max‑i} on the feasible indices
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it + 1)
        a_new = a - eta * grad
        a = _project_simplex(a_new)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Max‑subgrad iter={it} score={cur_score:.6f}")

        # tiny random jitter
        if rng.random() < 0.002:
            a += rng.normal(scale=0.001, size=n)
            a = _project_simplex(np.maximum(a, 0.0))

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣ Focused hill‑climb on the maximal convolution entry
# ----------------------------------------------------------------------
def _focused_hill_climb(a: np.ndarray,
                       n: int,
                       L_fft: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly reduces the current peak by moving mass from the larger
    member of the most‑contributing pair to an index whose partner at the
    peak is smallest.
    """
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # contributions a_i * a_{k_max-i}
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        contrib = best_a[idx] * best_a[k_max - idx]
        max_pos = int(np.argmax(contrib))
        i = int(idx[max_pos])
        j = int(k_max - i)

        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-12:
            continue

        # partner weight for each possible destination d:
        #   w_d = a_{k_max‑d}   (zero when out‑of‑range)
        partner = np.full(n, np.inf)
        d_start = max(0, k_max - (n - 1))
        d_end = min(k_max, n - 1)
        d_idx = np.arange(d_start, d_end + 1, dtype=int)
        partner[d_idx] = best_a[k_max - d_idx]
        # exclude the source
        partner[src] = np.inf

        dst = int(np.argmin(partner))
        if dst == src:
            continue

        # amount to transfer – modest fraction of src weight
        delta = min(best_a[src] * 0.5, 0.005)
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        new_score = _objective_score(cand, n, L_fft)
        if new_score < best_score:
            best_score, best_a = new_score, cand
            print(f"[{time.time() - start_time:.2f}s] Hill src={src}→{dst} δ={delta:.6f} → score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣ Final mixing + cheap random‑pair hill‑climb
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Convex mix with uniform + tiny random‑pair adjustments."""
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing (λ from 0 to 1)
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        max_move = min(best_a[i], 0.004)
        if max_move <= 0:
            continue
        delta = rng.uniform(0.0, max_move)
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill i={i} j={j} δ={delta:.6f} → score={sc:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of non‑negative floats (approximately unit sum).
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # small safety margin

    rng = np.random.default_rng(seed)

    # problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    L_fft = _next_pow2(2 * n - 1)       # for a*a
    L_corr = _next_pow2(3 * n - 2)      # for the p‑norm gradient

    # ------------------------------------------------------------------
    # 0️⃣ Initialiser (uniform) and optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # optional global starter (if the variable exists in the environment)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start (Dirichlet draws)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.04 * (deadline - start_time))
    if time.time() < warm_deadline:
        best_a, best_score = _warm_start(n, best_a, best_score,
                                         L_fft, warm_deadline, rng)

    # ------------------------------------------------------------------
    # 2️⃣ p‑norm annealing gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        grad_deadline = time.time() + 0.60 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Starting p‑norm descent")
        best_a, best_score = _pnorm_descent(best_a, n, L_fft,
                                            L_corr, grad_deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] After p‑norm best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Direct sub‑gradient on the maximum entry
    # ------------------------------------------------------------------
    if time.time() < deadline:
        maxgrad_deadline = time.time() + 0.22 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Max‑subgradient descent")
        best_a, best_score = _max_subgradient_descent(best_a, n,
                                                      L_fft, maxgrad_deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] After max‑subgradient best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Focused hill‑climb on the maximal convolution entry
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + 0.10 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Focused hill‑climb")
        best_a, best_score = _focused_hill_climb(best_a, n,
                                                 L_fft, hill_deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final mixing + light random‑pair hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement")
        best_a, best_score = _final_refinement(best_a, n,
                                               L_fft, deadline, rng)

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    # log the true score according to evaluate_sequence (if available)
    try:
        real_score = evaluate_sequence(best_a.tolist())
        print(f"[{time.time() - start_time:.2f}s] Real evaluate_sequence score = {real_score:.6f}")
    except Exception:
        pass

    return [float(v) for v in best_a]
