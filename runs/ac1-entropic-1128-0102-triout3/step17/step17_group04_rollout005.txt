    a = a0.copy()
    n = a.shape[0]
    iteration = 0
    # initial step size derived from maybe 0.1
    eta0 = 0.5
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        score = float(2.0 * n * max_val)    # because sum=1
        # ensure scoreboard consistent
        if score < best_score - 1e-9:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad iter={iteration} score={score:.6f}")
        # compute gradient
        ks = np.flatnonzero(conv == max_val)
        grad = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)

def _subgradient_descent_one_start(a0, max_iters, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        max_val = conv.max()
        score = 2.0 * n * max_val   # sum=1
        if score < best_score - 1e-12:
            best_seq, best_score = a.copy(), score
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter {it} new best {score:.6f}")
        # gradient
        ks = np.flatnonzero(conv == max_val)
        grad = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            grad[i_start:i_end+1] += 2.0 * a[k - idx]
        grad /= len(ks)
        # step size schedule
        eta = 0.5 / np.sqrt(it + 1)
        a = a - eta * grad
        # ensure non-negativity and project onto simplex
        a = _project_simplex(a)
        # occasional random jitter
        if (it % 200) == 0 and it > 0:
            rng = np.random.default_rng()
            eps = rng.normal(scale=1e-4, size=n)
            a = np.clip(a + eps, 0, None)
            a = a / a.sum()
    return best_seq, best_score

"""
A fast, multi‑phase optimiser for the step‑function objective

    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1️⃣  **Rich parametric sweep** – evaluates many simple families (geometric,
    power‑law, linear, sinusoidal, …) covering a broad part of the search
    space and keeps the best candidate.

2️⃣  **Random Dirichlet starts + short sub‑gradient runs** – draws a handful of
    random probability vectors, runs a few iterations of a sub‑gradient
    descent that directly minimises the (non‑smooth) max‑convolution, and
    remembers any improvement.

3️⃣  **Long sub‑gradient descent from the current best** – a classic
    projected (mirror) descent with a decreasing step size and occasional
    random jitter to escape plateaus.  The gradient of the max‑convolution
    at a maximizing shift *k* is simply 2·a_{k‑i} (zero outside the support),
    which can be assembled in O(n).  After each step the iterate is projected
    onto the probability simplex (∑a_i = 1, a_i ≥ 0).

All heavy work (self‑convolution) uses a real FFT, so each iteration is cheap
and we can afford many thousands of steps well within the 1000 s budget.

The final sequence is returned as a plain Python list (the evaluator will
clip to [0,1000] automatically, but all our values are ≪ 1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Score for a vector that already sums to 1:
       score = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣  Rich parametric sweep
# ----------------------------------------------------------------------
def _extensive_sweeps(n, best_seq, best_score, deadline):
    """Explore a variety of simple families; keep the best vector seen."""
    idx = np.arange(n, dtype=np.float64)

    # ---- Geometric (decreasing) ---------------------------------------
    for r in np.linspace(0.20, 0.97, 40):
        if time.time() >= deadline:
            break
        geo = r ** idx
        geo /= geo.sum()
        sc = _objective_score(geo)
        if sc < best_score:
            best_seq, best_score = geo.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] geo r={r:.3f} score={sc:.6f}")

    # ---- Reverse geometric (increasing) --------------------------------
    for r in np.linspace(0.20, 0.97, 40):
        if time.time() >= deadline:
            break
        rev = r ** (n - 1 - idx)
        rev /= rev.sum()
        sc = _objective_score(rev)
        if sc < best_score:
            best_seq, best_score = rev.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] rev‑geo r={r:.3f} score={sc:.6f}")

    # ---- Linear decreasing ---------------------------------------------
    lin = (n - idx).astype(np.float64)
    lin /= lin.sum()
    sc = _objective_score(lin)
    if sc < best_score:

        print(f"[{time.time() - start_time:.2f}s] linear decreasing score={sc:.6f}")

            break
        pw = (idx + 1) ** (-p)
        pw /= pw.sum()
        sc = _objective_score(pw)
        if sc < best_score:
            best_seq, best_score = pw.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] power p={p:.2f} score={sc:.6f}")

    # ---- Sinusoidal half‑wave (raised to a power) ----------------------
    half = np.sin((idx + 1) * np.pi / (n + 1))
    for p in np.linspace(0.5, 5.0, 20):
        if time.time() >= deadline:
            break
        s = half ** p
        s /= s.sum()
        sc = _objective_score(s)
        if sc < best_score:
            best_seq, best_score = s.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] sine^p={p:.2f} score={sc:.6f}")

    # ---- Symmetric exponential (already used in the reference, but with a wider λ range)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-5, -0.2, 60):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] sym‑exp λ={lam:.1e} score={sc:.6f}")

    return best_seq, best_score

# 2️⃣  Sub‑gradient descent (single start)
# ----------------------------------------------------------------------
def _subgradient_descent_one_start(a0, max_iters, deadline, best_seq, best_score):
    """
    Perform a projected sub‑gradient descent directly on   max(conv(a,a))
    while keeping the sum of a equal to 1.  The step size is η_t = 0.5/√(t+1)
    and a tiny random jitter is injected every few hundred steps.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_val = conv.max()
        # current score (sum(a)=1 → denominator = 1)
        cur_score = 2.0 * n * max_val
        if cur_score + 1e-12 < best_score:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it} new best {cur_score:.6f}")

        # --- sub‑gradient of the max‑convolution -----------------------
        ks = np.flatnonzero(conv == max_val)           # possibly several maximisers
        grad = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += 2.0 * a[k - idx]
        grad /= len(ks)                                 # average if multiple ks

        # --- step size (decreasing) -----------------------------------
        eta = 0.5 / np.sqrt(it + 1.0)
        a = a - eta * grad

        # --- projection back onto the simplex ---------------------------
        a = _project_simplex(a)

        # --- occasional random jitter to hop out of plateaux -----------
        if (it % 250) == 0 and it > 0:
            eps = rng.normal(scale=1e-4, size=n)
            a = np.clip(a + eps, 0.0, None)
            a /= a.sum()

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine performs a fast parametric sweep, many Dirichlet starts with
    short sub‑gradient runs and a final long optimisation phase.  The best
    sequence (normalised to unit sum) is returned as a plain list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – extensive parametric sweep
    # --------------------------------------------------------------
    phase1_deadline = deadline - 600.0   # keep ~600 s for later phases
    if time.time() < phase1_deadline:
        best_seq, best_score = _extensive_sweeps(
            n, best_seq, best_score, phase1_deadline
        )

    # --------------------------------------------------------------
    # Phase 2 – many random Dirichlet starts with short descent runs
    # --------------------------------------------------------------
    phase2_deadline = deadline - 200.0
    rng = np.random.default_rng()
    while time.time() < phase2_deadline:
        # concentration – log‑uniform between 10⁻² and 10⁰·⁵
        logc = rng.uniform(-2.0, 0.5)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        # a few descent steps (fast)
        best_seq, best_score = _subgradient_descent_one_start(
            a, max_iters=400, deadline=phase2_deadline,
            best_seq=best_seq, best_score=best_score
        )
        # break if time is almost up
        if time.time() + 1.0 >= phase2_deadline:
            break

    # --------------------------------------------------------------
    # Phase 3 – long sub‑gradient descent from the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Long sub‑gradient descent from best")
        best_seq, best_score = _subgradient_descent_one_start(
            best_seq, max_iters=20000, deadline=deadline,
            best_seq=best_seq, best_score=best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After long descent best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a clean Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – final best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
