"""
Heuristic optimiser for the step‑function evaluation problem.

The algorithm consists of four stages:

0️⃣ **Initialisation** – start from the uniform distribution,
   optionally replace it by the provided global starter
   (``height_sequence_1``), and keep the best of a few Dirichlet draws.

1️⃣ **Smooth‑max projected gradient descent** – minimise a
   differentiable approximation of  max(conv(a,a))  by
   using the soft‑maximum (log‑sum‑exp) with an increasing temperature.
   The gradient is obtained by a convolution with the soft‑max‐weights.
   After each step the vector is projected onto the probability simplex.

2️⃣ **Targeted hill‑climb** – repeatedly locate the current maximal
   convolution entry, identify the pair of coordinates that contributes
   most to it and move a modest amount of mass from the larger one to a
   random destination, accepting only improving moves.

3️⃣ **Final refinement** – convex mixing with the uniform vector and a
   light random‑pair hill‑climb to polish the solution.

The whole routine respects a 1000 s budget (default) and prints
intermediate scores.  The returned list sums to one and contains only
non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution x * y via FFT (full length)."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: x.shape[0] + y.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score used by `evaluate_sequence` when sum(a)=1."""
    return float(2.0 * n * conv.max())


def _smooth_max_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Soft‑max weights for a given temperature `tau`.
    The vector is scaled so that the maximum entry has weight 1,
    then the soft‑max is applied.
    """
    max_c = conv.max()
    if max_c == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    # scale to [0,1]
    scaled = conv / max_c
    # exponent = tau * (scaled - 1)  ≤ 0
    exp_vals = np.exp(tau * (scaled - 1.0))
    w = exp_vals / exp_vals.sum()
    return w


def _gradient_smooth_max(a: np.ndarray, weights: np.ndarray,
                         L_corr: int) -> np.ndarray:
    """
    Gradient of the smooth‑max (soft‑max) of the self‑convolution.
    Formula:  ∇ = 2·(a * w_rev)_{valid} reversed,
    where w_rev is the reversed weight vector.
    """
    w_rev = weights[::-1]
    # full convolution a * w_rev  (length 3·n‑2)
    conv_full = _conv_fft(a, w_rev, L_corr)
    n = a.shape[0]
    # middle part gives the needed values; reverse to align with indices
    grad = 2.0 * conv_full[n - 1 : 2 * n - 1][::-1]
    return grad


def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    Return the pair (i, j) with i+j=k that maximises a_i·a_j,
    together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    prod = a[idx] * a[k - idx]
    max_pos = int(np.argmax(prod))
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(prod[max_pos])


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of non‑negative floats (approximately unit sum).
    """
    # -------------------- timing and RNG ------------------------------
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin
    rng = np.random.default_rng(seed)

    # -------------------- problem size -------------------------------
    n = 4096                     # trade‑off between quality and speed
    L_fft = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n - 2)         # for gradient convolution

    # -------------------- initial candidate ----------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _score_from_conv(_conv_self_fft(best_a, L_fft), n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # optional global starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter /= s
                sc = _score_from_conv(_conv_self_fft(starter, L_fft), n)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # -------------------- warm‑start (Dirichlet draws) ---------------
    warm_deadline = min(deadline, start_time + 0.04 * budget_s)
    while time.time() < warm_deadline:
        # concentration drawn log‑uniformly
        log_c = rng.uniform(np.log(0.1), np.log(15.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        conv = _conv_self_fft(a, L_fft)
        sc = _score_from_conv(conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Warm‑start c={c:.2f} → score={sc:.6f}")

    # -------------------- smooth‑max projected gradient descent ------
    grad_deadline = min(deadline, time.time() + 0.70 * (deadline - time.time()))
    print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max gradient descent")
    a = best_a.copy()
    iteration = 0
    eta0 = 0.6            # initial step size
    tau = 0.5             # initial temperature for soft‑max
    tau_max = 30.0
    tau_step_iter = 300   # increase τ every this many iterations

    while time.time() < grad_deadline:
        iteration += 1

        # self‑convolution of the current candidate
        conv = _conv_self_fft(a, L_fft)
        cur_score = _score_from_conv(conv, n)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD iter {iteration:,} score={cur_score:.6f}")

        # soft‑max weights centred at the current maximum
        weights = _smooth_max_weights(conv, tau)

        # gradient of the smooth maximum
        grad = _gradient_smooth_max(a, weights, L_corr)

        # diminishing step size (1/√t schedule)
        eta = eta0 / np.sqrt(iteration)

        # gradient step followed by projection onto the simplex
        a_new = a - eta * grad
        a = _project_simplex(a_new)

        # slowly raise the temperature to focus more on the true maximum
        if iteration % tau_step_iter == 0 and tau < tau_max:
            tau = min(tau_max, tau * 1.5)

    # -------------------- targeted hill‑climb -----------------------
    hill_deadline = min(deadline, time.time() + 0.20 * (deadline - time.time()))
    print(f"[{time.time() - start_time:.2f}s] Targeted hill‑climb")
    a = best_a.copy()

    while time.time() < hill_deadline:
        conv = _conv_self_fft(a, L_fft)
        cur_score = _score_from_conv(conv, n)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={cur_score:.6f}")

        k_max = int(conv.argmax())

        # largest contributing pair for the peak position
        i, j, _ = _max_contributions(a, k_max)
        if i == -1:
            continue

        src = i if a[i] >= a[j] else j
        if a[src] < 1e-12:
            continue

        dst = rng.integers(0, n)
        if dst == src:
            continue

        # move a modest fraction of the source mass (up to 30 % or 0.02)
        max_delta = min(a[src] * 0.30, 0.02)
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # re‑project onto the simplex (keeps sum = 1)
        cand = _project_simplex(cand)

        new_score = _score_from_conv(_conv_self_fft(cand, L_fft), n)
        if new_score < best_score:
            a = cand
            best_score = new_score
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s]  move {src}->{dst} δ={delta:.6f} → score={new_score:.6f}")

    # -------------------- final mixing & light hill‑climb ----------
    final_deadline = deadline
    print(f"[{time.time() - start_time:.2f}s] Final refinement")
    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= final_deadline:
            break
        cand = lam * best_a + (1.0 - lam) * uniform
        cand = _project_simplex(cand)  # ensures exact normalisation
        sc = _score_from_conv(_conv_self_fft(cand, L_fft), n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    while time.time() < final_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_a[i] * 0.30, 0.02)
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)

        sc = _score_from_conv(_conv_self_fft(cand, L_fft), n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Random pair i={i} j={j} δ={delta:.6f} → score={sc:.6f}")

    # -------------------- final normalisation & return -------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")

    # The evaluation function will also clamp values to [0,1000] and
    # protect against tiny sums, so the returned vector is safe.
    return [float(v) for v in best_a]
