Algorithm:

- Start with f = uniform or seed.

- Normalize f to sum=1.

- Compute M = max(conv(f,f))

- Define LP variables g_j >= 0.

- Constraints: For each k=0..2n-2: sum_{j=0}^{n-1} f_j * g_{k-j} <= M. Implementation: For each k, we build dot product f shifted.

- Objective: maximize sum(g_j).

- Solve LP using scipy.optimize.linprog with method='highs'.

- Get g (vector). Compute sum_g. If sum_g nearly 0 due to constraints, abort.

- Rescale g to have sum=1 (multiply by 1/sum_g).

- Update f ← (1-t) f + t g, with small t (like 0.2). (Or we can set t=0.5). Perhaps adapt.

- Repeat until M does not improve significantly or time running out.

def smooth_max_grad_desc(a, deadline, best_seq, best_score):
    a = a.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 1.0
    tau = max(_conv_self(a).max() * 0.5, 1e-6)
    while time.time() < deadline and tau > 1e-12:
        conv = _conv_self(a)
        max_conv = conv.max()
        # compute softmax probabilities
        log_w = (conv - max_conv) / tau
        w = np.exp(log_w)
        w = w / w.sum()
        # gradient of smooth max: grad = 2 * convolution of w with reversed a
        grad = 2.0 * np.convolve(w, a[::-1], mode='full')[n-1:2*n-1]
        # ADAM update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a_new = _project_simplex(a_new)
        # evaluate true objective
        new_obj = _objective_score(a_new)
        cur_obj = _objective_score(a)
        if new_obj < cur_obj:
            a = a_new
            cur_score = new_obj
            if new_obj < best_score:
                best_score, best_seq = new_obj, a.copy()
                print(f"[time] improved to {new_obj}")
        else:
            # possibly keep? we could accept if random
            pass
        tau *= 0.99  # cooling
        t += 1

Propose a non‑negative step function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search proceeds in three stages (under a 1000 s budget)

1️⃣  Large‑scale parametric sweeps (geometric, power, exponential,
    inverse‑power, cosine valley).  A few hundred candidates are tried
    quickly; the best seed is kept.

2️⃣  Massive Dirichlet sampling – up to several ten‑thousands of random
    points on the simplex.  Concentrations range from very spiky
    (c≈0.01) to near‑uniform (c≈5).

3️⃣  Smooth‑max gradient descent with Adam‑style adaptive steps.
    The non‑smooth max is replaced by τ·log‑sum‑exp, gradients are taken
    on the simplex (Euclidean projection) and the temperature τ is
    slowly cooled.  After a run finishes we jitter the current best
    point and restart; this multi‑restart loop continues until the
    time budget is exhausted.

All utilities are top‑level, there are no closures, and the code
relies only on numpy and scipy (high‑speed FFT based convolution).
The final sequence is normalised to sum = 1 and returned as plain
Python floats.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, not used in the final version

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Assumes sum(seq)=1 and returns 2·n·max(conv(seq,seq))."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # From: https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_inverse_power(n, best_seq, best_score, deadline):
    """a_i = (i+ε)^p + (n-1-i+ε)^p, ε avoids division by zero."""
    idx = np.arange(n, dtype=np.float64)
    eps = 1e-8
    for p in np.linspace(-3.0, 3.0, 61):
        if time.time() >= deadline:
            break
        seq = np.power(idx + eps, p) + np.power(n - 1 - idx + eps, p)
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] InvPower p={p:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_cosine_valley(n, best_seq, best_score, deadline):
    """High at the ends, low in the centre: 1 + a·cos(2π·i/(n‑1)), a∈[0,0.9]."""
    idx = np.arange(n, dtype=np.float64)
    for amp in np.linspace(0.1, 0.9, 17):
        if time.time() >= deadline:
            break
        seq = 1.0 + amp * np.cos(2.0 * np.pi * idx / (n - 1))
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Cosine amp={amp:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the ends with tunable split."""
    for L_frac in (0.02, 0.05, 0.1, 0.2, 0.3, 0.4):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        # concentration between 0.01 and 10 (log‑uniform)
        logc = rng.uniform(-2.0, 1.0)      # 10^-2 … 10^1
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        trials += 1
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet #{trials:,} c={c:.3f} score={sc:.6f}")
        # occasional larger jumps to keep diversity
        if trials % 5000 == 0:
            # tiny jitter to avoid stagnation
            eps = 1e-7
            best_seq = np.clip(best_seq + rng.uniform(-eps, eps, size=n), 0, None)
            best_seq = _project_simplex(best_seq)
    return best_seq, best_score

# 3️⃣ Smooth‑max gradient descent (Adam‑style)
# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(a0, deadline, best_seq, best_score, max_iters=20000):
    """
    Gradient descent on a τ·log‑sum‑exp surrogate of max(conv).
    Uses Adam‑style step size adaptation and back‑tracking line search
    on the true (non‑smoothed) objective.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    t_iter = 0

    # start temperature – a fraction of the current max‑conv
    cur_conv = _conv_self(a)
    tau = max(cur_conv.max() * 0.4, 1e-6)
    min_tau = 1e-12
    decay = 0.998

    # true objective of the current point
    cur_obj = _objective_score(a)

    while time.time() < deadline and tau > min_tau and t_iter < max_iters:
        t_iter += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # ---- soft‑max over the convolutions (smooth max) ----
        shift = max_conv                     # for numerical stability
        log_w = (conv - shift) / tau
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0:
            tau *= decay
            continue
        p = w / w_sum                       # soft‑max probabilities

        # ---- gradient of the smooth max (2 * p * a_rev) ----
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]
        grad = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # ---- Adam update (pre‑projection) ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = 0.5 / (np.sqrt(v_hat) + eps_adam)    # 0.5 is a base learning‑rate

        # ---- line search on the true objective ----
        a_candidate = a - step * m_hat
        a_candidate = _project_simplex(a_candidate)

        obj_candidate = _objective_score(a_candidate)

        # accept if we obtain a strict improvement
        if obj_candidate < cur_obj - 1e-12:
            a = a_candidate
            cur_obj = obj_candidate
            if obj_candidate < best_score:
                best_score, best_seq = obj_candidate, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SmoothGD it={t_iter:,} obj={obj_candidate:.6f} τ={tau:.2e}")
        else:
            # no improvement – take a tiny random jitter to escape flat plateaus
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
            cur_obj = _objective_score(a)       # recompute (cheap)

        # cool the temperature slowly
        tau *= decay
    return best_seq, best_score


def _smooth_max_multi_restart(a_start, deadline, best_seq, best_score):
    """
    Run several Adam‑style smooth‑max descents, resetting the
    temperature after each run and jittering the start point.
    """
    remaining = deadline
    a_cur = a_start.copy()
    while time.time() < remaining:
        # give each run at most 200 s (or whatever is left)
        run_deadline = min(time.time() + 200.0, remaining)
        best_seq, best_score = _smooth_max_gradient_descent(
            a_cur, run_deadline, best_seq, best_score
        )
        # If we have already reached the target we can stop early
        if best_score <= 1.503:
            break
        # jitter the current best to start a fresh descent
        rng = np.random.default_rng()
        eps = 1e-6
        a_cur = best_seq + rng.uniform(-eps, eps, size=a_cur.shape)
        a_cur = np.clip(a_cur, 0.0, None)
        a_cur = _project_simplex(a_cur)
    return best_seq, best_score
