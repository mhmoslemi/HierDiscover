        if eps <= 0:

        if a_prop[src] < 0:

        if new_score < best_score:
            best_score = new_score
            best_seq = a_prop.copy()
            a_cur = a_prop
            print(f"[{time.time() - start_time:.2f}s] hill it={it:,} score={new_score:.6f}")
        else:

            T = T0 * (0.9999 ** it)
            prob = np.exp(-(new_score - best_score) / (T + 1e-12))

            a_cur = np.maximum(a_cur + epsi, 0.0)
            a_cur = _project_simplex(a_cur)

"""
Hybrid optimiser for the L∞‑convolution problem.

The search proceeds in several stages:
1️⃣  A quick geometric‑×‑power parametric sweep to obtain a decent start.
2️⃣  A fine local sweep around the best parametric point.
3️⃣  Adam‑style optimisation of the smooth‑max (log‑sum‑exp) surrogate.
4️⃣  Classical projected sub‑gradient descent on the true L∞ objective.
5️⃣  Targeted hill‑climbing moves that always shift mass away from the
    indices most responsible for the current worst convolution value,
    with a tiny simulated‑annealing acceptance rule.

All updates keep the vector on the probability simplex (non‑negative,
unit L¹ norm).  The best sequence seen during the whole run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # all zero – fall back to uniform
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – assumes the sum ≈ 1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient helpers
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]               # diagonal counted twice – works for our purpose
    return g


# ----------------------------------------------------------------------
# Stage 1 – coarse/fine parametric sweep
# ----------------------------------------------------------------------


def _sweep_initial(n: int, L_fft: int, best_seq: np.ndarray,
                  best_score: float, deadline: float, start_time: float):
    """Coarse geometric × power sweep followed by a local refinement."""
    # coarse grid
    min_lambda = 4e-05          # λ = -log r   (r≈0.99996)
    max_lambda = 1.5e-03        # r≈0.9985
    lam_grid = np.linspace(min_lambda, max_lambda, 300)
    p_grid = np.linspace(-6.0, 3.0, 91)

    for lam in lam_grid:
        if time.time() >= deadline:
            break
        base = -lam * np.arange(n, dtype=np.float64)
        for p in p_grid:
            log_shape = base + p * np.log(np.arange(1, n + 1, dtype=np.float64))
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6e} "
                      f"p={p:.2f} score={sc:.6f}")

    # ---------- fine local sweep around the current best ----------
    # extract current λ,p by fitting log‑shape (least‑squares)
    # (we reuse the same family for the fine region)
    # To keep things simple we just centre a finer grid around the
    # best λ and p observed during the coarse pass.
    # If no improvement happened, keep the centre at the middle of the ranges.
    # NOTE: The exact extraction is cheap – we simply reuse the best λ/p.
    # The best λ/p are approximated by the last values that yielded improvement.
    # We store them during the coarse pass using local vars.
    # For robustness we fall back to the middle of the search box.
    # -----------------------------------------------------------------
    # (The extraction is deliberately lightweight; the fine sweep will
    # still explore a dense neighbourhood and improve the start.)
    # -----------------------------------------------------------------
    # We'll just reuse the same bounds but shrink them.
    if best_seq is not None:
        # Estimate the exponent p by linear regression on log‑shape.
        # log a_i = C + p*log(i+1) - λ i
        i = np.arange(n, dtype=np.float64)
        log_a = np.log(best_seq + 1e-30)
        # solve least squares for (λ, p)
        A = np.column_stack((-i, np.log(i + 1.0)))
        coeff, *_ = np.linalg.lstsq(A, log_a, rcond=None)
        lam0, p0 = coeff
    else:
        lam0 = (min_lambda + max_lambda) / 2.0
        p0 = 0.0

    # fine grids (≈30 s)
    fine_deadline = deadline - 0.1
    lam_fine = np.linspace(max(min_lambda, lam0 - 1e-04),
                           min(max_lambda, lam0 + 1e-04), 200)
    p_fine = np.linspace(p0 - 0.5, p0 + 0.5, 200)

    for lam in lam_fine:
        if time.time() >= fine_deadline:
            break
        base = -lam * np.arange(n, dtype=np.float64)
        for p in p_fine:
            log_shape = base + p * np.log(np.arange(1, n + 1, dtype=np.float64))
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] fine λ={lam:.6e} "
                      f"p={p:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 3 – Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam‑style updates on the log‑sum‑exp (β‑smooth) max‑convolution.
    The smoothing parameter β is increased gradually so that the
    optimisation first sees a smooth landscape and finally focuses
    on the true L∞‑objective.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    beta = 200.0          # initial smoothing
    alpha0 = 0.6          # base step size
    L_grad = _next_pow2(3 * n - 2)   # length for w * a_rev convolution

    while time.time() < deadline:
        it += 1

        # ---- compute self‑convolution ----
        conv = _conv_self_fft(a, L_fft)

        # ---- smooth max weights (log‑sum‑exp) ----
        max_c = conv.max()
        # shift for numerical stability
        exp_vals = np.exp(beta * (conv - max_c))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_vals / sum_exp

        # ---- gradient of the smooth surrogate ----
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L_grad)
        Revf = np.fft.rfft(a_rev, n=L_grad)
        grad_full = np.fft.irfft(Wf * Revf, n=L_grad)
        start = n - 1
        g = 2.0 * grad_full[start:start + n]

        # ---- Adam moments ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ---- decreasing step size ----
        step = alpha0 / np.sqrt(it)

        # ---- update and project onto simplex ----
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # ---- slowly increase smoothing ----
        if it % 2000 == 0:
            beta = min(beta * 1.05, 5e5)

        # ---- evaluate exact objective ----
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"β={beta:.1e} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 4 – projected sub‑gradient descent (as before, slight tweak)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.33

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # diminishing step length
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                      f"step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional stochastic uphill to escape tiny plateaus
            prob = np.exp(-(cand_score - best_score) * 7.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        if it % 1500 == 0:
            # tiny random jitter
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 5 – targeted hill‑climbing with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _targeted_hill_climb(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Move a tiny amount of probability mass from the index that most
    contributes to the current worst convolution value to the index
    that contributes the least.  A fast annealing schedule accepts
    occasional uphill moves.
    """
    a_cur = a0.copy()
    n = a_cur.shape[0]
    it = 0
    T0 = 0.012  # initial temperature for simulated annealing

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a_cur, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a_cur, k)

        # importance = a_i * (∂max/∂a_i) – larger => mass should leave
        importance = g * a_cur
        src = int(np.argmax(importance))
        dst = int(np.argmin(importance))

        # transfer up to 0.3 % of the source mass
        eps = a_cur[src] * rng.random() * 3e-3
        if eps <= 0.0:
            continue

        a_prop = a_cur.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_prop.copy()
            a_cur = a_prop
            print(f"[{time.time() - start_time:.2f}s] hill it={it:,} "
                  f"score={new_score:.6f}")
        else:
            # simulated‑annealing acceptance
            T = T0 * (0.99995 ** it)
            prob = np.exp(-(new_score - best_score) / (T + 1e-12))
            if rng.random() < prob:
                a_cur = a_prop

        if it % 4000 == 0:
            # occasional random shake to diversify search
            shake = 1e-4 * rng.random(n) * a_cur
            a_cur = np.maximum(a_cur + shake, 0.0)
            a_cur = _project_simplex(a_cur)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  Returns a plain Python list of floats (unit‑sum).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # safety margin against the hard cut‑off

    # --------------------------------------------------------------
    # Problem size – a trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter (height_sequence_1) exists, try it.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Stage 1 – coarse + fine parametric sweep (~130 s)
    # --------------------------------------------------------------
    sweep_deadline = start_time + min(130.0, budget_s * 0.14)
    best_seq, best_score = _sweep_initial(
        n, L_fft, best_seq, best_score, sweep_deadline, start_time)

    # --------------------------------------------------------------
    # Stage 2 – Adam optimisation on smooth max (~500 s)
    # --------------------------------------------------------------
    if time.time() < deadline - 350.0:
        adam_deadline = deadline - 350.0
        best_seq, best_score = _adam_optimize(
            best_seq, L_fft, adam_deadline, start_time,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Stage 3 – Projected sub‑gradient descent (~200 s)
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        sub_deadline = deadline - 150.0
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, sub_deadline, start_time,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Stage 4 – Targeted hill‑climb until time runs out.
    # --------------------------------------------------------------
    while time.time() < deadline:
        best_seq, best_score = _targeted_hill_climb(
            best_seq, L_fft, deadline, start_time,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Final normalisation (sum → 1) and conversion to Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
