def _adam_optimize(a0, cur_score, deadline, rng):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.1
    best_a = a.copy()
    best_score = cur_score
    iter = 0
    last_improve_iter = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        # compute grad sum
        grad = _subgradient_maxconv(a, conv)  # sum version (no avg)
        # objective gradient scaling
        grad = 2.0 * n * grad
        iter += 1
        # update moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iter)
        v_hat = v / (1 - beta2 ** iter)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        # projection onto simplex
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            last_improve_iter = iter
            print(f"[{time.time() - start_time:.2f}s] Adam iter {iter} new best {best_score:.6f}")
        a = cand
        # occasional random perturb
        if iter % 5000 == 0:
            # if no improvement in last 5000 iters, add noise
            if iter - last_improve_iter > 5000:
                noise = rng.normal(scale=1e-4, size=a.shape)
                a = a + noise
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                print(f"[{time.time() - start_time:.2f}s] Adam iter {iter} perturb added")
    return best_a, best_score

"""
Search for a non‑negative step‑function minimising
    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm:
- diverse deterministic random initialisations (uniform, Dirichlet draws,
  sparse, exponential and power‑law shapes);
- a fast Adam‑style projected‑gradient optimiser on the simplex using the
  sub‑gradient of the maximal self‑convolution entry;
- a cheap random local refinement at the end.

The whole routine respects the 1000 s wall‑time budget and returns the best
vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing
# ----------------------------------------------------------------------
start_time = None  # will be set in propose_candidate


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_sumconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) (not averaged) – sum over all active
    convolution indices.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    return grad


def _initial_candidates(n: int, rng: np.random.Generator, deadline: float):
    """Yield a sequence of promising initial vectors."""
    # uniform
    a = np.full(n, 1.0 / n, dtype=float)
    yield a

    # Dirichlet draws (log‑uniform concentration)
    for _ in range(200):

        conc_log = rng.uniform(-3.0, 0.7)

        a /= a.sum()
        yield a

        idx = rng.choice(n, size=k, replace=False)

        a[idx] = vals
        a /= a.sum()

        yield a

    except Exception:

def _adam_optimize(a0: np.ndarray, start_score: float, deadline: float,
                   rng: np.random.Generator):
    """
    Adam‑style projected gradient descent on the simplex.
    Uses the sub‑gradient of the maximal self‑convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.2          # initial step size (will decay)
    iteration = 0
    best_a = a.copy()
    best_score = start_score
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1

        conv = _conv_self(a)
        grad = _subgradient_sumconv(a, conv)          # sum‑type sub‑gradient
        grad = 2.0 * n * grad                         # gradient of the objective

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # linear decay of the base learning rate to zero
        t_rem = max(deadline - time.time(), 0.0)
        total_rem = max(deadline - start_time, 1e-12)
        alpha = alpha0 * (t_rem / total_rem)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _objective_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            last_improve_iter = iteration
            print(f"[{time.time() - start_time:.2f}s] Adam iter {iteration} new best {best_score:.6f}")

        a = cand

        # occasional random perturbation if stuck for a while
        if iteration % 5000 == 0:
            if iteration - last_improve_iter > 5000:
                noise = rng.normal(scale=1e-4, size=n)
                a = a + noise
                a = np.clip(a, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = best_a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter {iteration} perturbation added")

    return best_a, best_score


def _random_local_refinement(a0: np.ndarray, start_score: float,
                             deadline: float, rng: np.random.Generator):
    """
    Very cheap pairwise mass transfers that only accept strict improvements.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = start_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] <= 1e-15:
            continue
        dst = rng.integers(n)
        if dst == src:
            continue
        delta = a[src] * rng.random() * 0.01   # at most 1 % of source mass
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Entry point used by the judge.
    Searches for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best vector found as a plain Python list of floats.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – keep the same size as in the reference solution.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 1️⃣  Diversified initialisation
    # --------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=float)   # uniform start
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    for cand in _initial_candidates(n, rng, init_deadline):
        sc = _objective_score(cand)
        if sc < best_score:
            best_seq, best_score = cand.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] New start candidate score = {best_score:.6f}")

    # --------------------------------------------------------------

        # allocate roughly 60 % of the remaining budget to Adam
        remaining = deadline - time.time()

        best_seq, best_score = _adam_optimize(best_seq, best_score, adam_deadline, rng)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:

    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]

    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.2          # base learning rate (decays linearly)
    iteration = 0
    best_a = a.copy()
    best_score = start_score
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1

        conv = _conv_self(a)
        grad = _subgradient_sumconv(a, conv)          # sum‑type sub‑gradient
        grad = 2.0 * n * grad                         # gradient of the objective

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # linear decay of the learning rate to zero
        t_rem = max(deadline - time.time(), 0.0)
        total_rem = max(deadline - start_time, 1e-12)
        alpha = alpha0 * (t_rem / total_rem)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _objective_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            last_improve_iter = iteration
            print(f"[{time.time() - start_time:.2f}s] Adam iter {iteration} new best {best_score:.6f}")

        a = cand

        # occasional random perturbation if we have been stuck
        if iteration % 5000 == 0:
            if iteration - last_improve_iter > 5000:
                noise = rng.normal(scale=1e-4, size=n)
                a = a + noise
                a = np.clip(a, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = best_a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter {iteration} perturbation added")

    return best_a, best_score


def _random_local_refinement(a0: np.ndarray, start_score: float,
                             deadline: float, rng: np.random.Generator):
    """
    Very cheap pairwise mass transfers that only accept strict improvements.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = start_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] <= 1e-15:
            continue
        dst = rng.integers(n)
        if dst == src:
            continue
        delta = a[src] * rng.random() * 0.01   # move at most 1 % of source mass
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Main entry point.  Returns a list of non‑negative heights (probability
    vector) that minimises ``evaluate_sequence`` within the time budget.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # a tiny safety margin

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – keep the same size as the reference implementation.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 1️⃣  Diversified initialisation
    # --------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=float)   # uniform start
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    init_deadline = min(deadline, start_time + 120.0)  # limit init to ~2 min
    for cand in _initial_candidates(n, rng, init_deadline):
        sc = _objective_score(cand)
        if sc < best_score:
            best_seq, best_score = cand.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] New start candidate score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam optimisation (major search phase)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.60 * remaining   # about 60 % of what is left
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation (budget {adam_deadline - time.time():.1f}s)")
        best_seq, best_score = _adam_optimize(best_seq, best_score, adam_deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] Adam phase finished – best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final cheap random local refinement (polishing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = min(deadline, time.time() + 30.0)   # at most 30 s
        print(f"[{time.time() - start_time:.2f}s] Starting random local refinement")
        best_seq, best_score = _random_local_refinement(best_seq, best_score, local_deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] Local refinement finished – best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise once more and return as plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
