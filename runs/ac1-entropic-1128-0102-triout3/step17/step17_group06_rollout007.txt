def project_onto_simplex(v):
    # v is array
    if v.sum() == 1 and np.all(v>=0): return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent on the soft‑max of the self‑convolution.

The algorithm works on a probability vector `a` (∑a = 1, a_i ≥ 0).  
For a given `a` we compute the full linear convolution `c = a⋆a`.  
The evaluation metric is  2·n·max(c)  (because the sum is normalised).

We turn the non‑smooth “max” into a smooth soft‑max:
    w_k ∝ exp( -α·(max(c) – c_k) )   ,   ∑ w_k = 1 .
For this weight vector the gradient of the weighted sum
    S = Σ_k w_k·c_k
with respect to `a` is  g = 2·( w ⋆ a ), where `⋆` denotes the
cross‑correlation (g_i = Σ_j w_{i+j} a_j).  Using a mirror‑descent
update  a_i ← a_i·exp( –η·g_i )  and projecting back onto the simplex
keeps all entries non‑negative and the sum equal to one.

The routine:
  * starts from a (slightly mixed) Beta(½,½) “arcsine’’ distribution,
  * performs many cheap FFT‑based iterations while respecting the time budget,
  * adapts the learning rate when progress stalls,
  * finishes with a tiny random‑pair hill‑climb (as in the reference code).

The returned list is a valid step‑function; its score is printed and
available to the outer evaluator via `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _beta_shape(n: int, p: float) -> np.ndarray:
    """Discretised Beta(p,p) density on the grid i+½ / n, normalised to sum‑1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum = 1, v≥0). """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‐function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget, prints
    progress, and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # small safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives a finer discretisation.
    # --------------------------------------------------------------
    n = 8192                                       # feel free to change (power of two is convenient)
    L = _next_pow2(3 * n - 2)                      # length for FFTs (≥ 2n‑1 + n‑1)

    # --------------------------------------------------------------
    # 0️⃣  Initial distribution
    # --------------------------------------------------------------
    # start from a mixed arcsine (Beta(½,½)) distribution
    arcsine = _beta_shape(n, 0.5)
    uniform = np.full(n, 1.0 / n)
    a = 0.90 * arcsine + 0.10 * uniform
    a = a / a.sum()

    # If a global starter is present, adopt it (padded / trimmed to length n)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)

            if starter.size != n:

                else:

            starter = np.maximum(starter, 0.0)

    except Exception:

    # --------------------------------------------------------------
    # Helper to compute the evaluation score (2·n·max(conv)) for a vector with sum≈1
    # --------------------------------------------------------------
    def _current_score(vec: np.ndarray) -> float:

    best_a = a.copy()

    # --------------------------------------------------------------
    # Normalise & return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:

        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
