def _smoothmax_grad(a, beta, L_corr):
    conv = _conv_self_fft(a, L_fft)
    maxc = conv.max()
    # compute weights
    w = np.exp(beta * (conv - maxc))

    grad = 2.0 * conv_tmp[:len(a)][::-1]  # as earlier

Propose a step‑function (non‑negative vector) that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The implementation follows four stages:

0️⃣  Build a cheap pool of candidates:
    - uniform and several handcrafted “U‑shaped”, “tent”, “parabolic”,
      inverse‑√ and arcsine‑like profiles.
    - many Dirichlet draws with random concentration.

    For a fixed vector f compute M = max(f∗f).  Solve a LP that maximises

    Repeating this rapidly lowers the peak of the self‑convolution.

    β is increased gradually so the optimisation focuses on the true maximum.

    member of the most‑contributing pair (i,j) of the current maximal
    convolution entry to a random other coordinate, accepting only improving

# Low‑level utilities (FFT based, simplex projection, scoring)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (no normalisation assumed)."""
    conv = _conv_self_fft(a, L_fft)
    maxb = conv.max()
    s = a.sum()
    if s < 0.01:          # avoid division by zero – will be ∞ in the real evaluator
        return np.inf
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Hand‑crafted deterministic seeds
# ----------------------------------------------------------------------
def _deterministic_seeds(n: int) -> list[np.ndarray]:
    """Return a list of hand‑crafted candidate vectors (all normalised to sum=1)."""
    i = np.arange(n, dtype=float)

    # uniform
    candidates = [np.full(n, 1.0 / n)]

    # symmetric tent (mountain)
    tent = np.minimum(i + 1, n - i)
    candidates.append(tent / tent.sum())

    # symmetric parabola (peak in middle, zero at ends)
    par = 1.0 - ((i - (n - 1) / 2) / ((n - 1) / 2)) ** 2
    par[par < 0] = 0.0
    candidates.append(par / par.sum())

    # inverse‑sqrt (higher near left edge)
    invsqrt = 1.0 / np.sqrt(i + 1)
    candidates.append(invsqrt / invsqrt.sum())

    # symmetric inverse‑sqrt (U‑shaped)
    inv_sqrt_u = 1.0 / np.sqrt(i + 1) + 1.0 / np.sqrt(n - i)
    candidates.append(inv_sqrt_u / inv_sqrt_u.sum())

    # arcsine‑like (strong U‑shape, singular at the ends, regularised)
    eps = 1e-4
    arcs = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    candidates.append(arcs / arcs.sum())

    return candidates


# ----------------------------------------------------------------------
# Warm‑start with Dirichlet draws
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_sc: float,
                L_fft: int,
                deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # log‑uniform concentration in a wide interval
        log_c = rng.uniform(np.log(0.02), np.log(30.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _score(a, n, L_fft)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming “mass pump’’ step
# ----------------------------------------------------------------------
def _lp_improvement(a: np.ndarray,
                    n: int,
                    L_fft: int,
                    L_corr: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a small LP:
        maximise Σ b_j
        s.t.   (a * b)_k ≤ M   for k where a*a is near its peak,
               b_j ≥ 0.
    The solution b is rescaled to the same ℓ¹‑norm as a and mixed.
    """
    best_a = a.copy()
    best_sc = _score(best_a, n, L_fft)
    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        M = conv.max()
        # rows where the current peak is almost attained (within 0.5 %)
        active_idx = np.where(conv >= M * 0.995)[0]
        if len(active_idx) == 0:
            break

        # build sparse A such that (a*b)_k = Σ_i a_i b_{k-i}
        rows, cols, data = [], [], []
        for r, k in enumerate(active_idx):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            # i = k - j
            for j in range(j_start, j_end + 1):
                i = k - j
                rows.append(r)
                cols.append(j)
                data.append(best_a[i])
        A = csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))

        # linprog: maximise Σ b  ⇔ minimise –Σ b
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        ub = np.full(len(active_idx), M)

        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'disp': False})
        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        sum_a = best_a.sum()
        # if the LP cannot increase the total mass we stop
        if sum_b <= sum_a + 1e-12:
            break

        # rescale to keep ℓ¹‑norm constant, then convexly mix
        b_scaled = b * (sum_a / sum_b)
        t = 0.5                     # mixing strength
        cand = (1.0 - t) * best_a + t * b_scaled
        cand = _project_simplex(cand)

        sc = _score(cand, n, L_fft)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] LP step → score={sc:.6f}")
        else:
            # no improvement – stop the LP phase early
            break
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (approximates L∞)
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray,
                         beta: float,
                         L_corr: int,
                         L_fft: int) -> np.ndarray:
    """
    Gradient of the smooth‑max functional
        Φβ(a) = Σ_k w_k (a∗a)_k ,   w_k ∝ exp(β·((a∗a)_k – max)).
    Returns ∇Φβ(a).
    """
    n = a.shape[0]
    # convolution a*a (full)
    conv = _conv_self_fft(a, L_fft)
    mx = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(beta * (conv - mx))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * correlation(w, a)
    a_rev = a[::-1]                     # reversed a
    Wa = np.fft.rfft(w, n=L_corr)
    Aa = np.fft.rfft(a_rev, n=L_corr)
    tmp = np.fft.irfft(Wa * Aa, n=L_corr)   # length L_corr
    grad = 2.0 * tmp[:n][::-1]              # first n entries, reversed back
    return grad


def _smoothmax_descent(a0: np.ndarray,
                       n: int,
                       L_fft: int,
                       L_corr: int,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max approximation.
    β is increased slowly so the optimisation ultimately focuses on the
    true maximum of the self‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = _score(best_a, n, L_fft)

    it = 0
    eta0 = 0.6          # initial step size
    beta = 5.0          # start with a gentle smoothing
    while time.time() < deadline:
        it += 1
        # current score (exact, not the surrogate)
        sc = _score(a, n, L_fft)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] G‑descent it={it:,} β={beta:.1f} → score={sc:.6f}")

        # compute surrogate gradient
        grad = _smooth_max_gradient(a, beta, L_corr, L_fft)

        # decreasing stepsize (1/√t) and a tiny random jitter
        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * grad
        a = _project_simplex(a_new)

        # slowly raise β (focus more on the true max)
        beta = min(beta * 1.02, 200.0)

        # occasional random shake to escape tiny plateaus
        if np.random.rand() < 0.006:
            eps = np.random.normal(scale=0.001, size=n)
            a = _project_simplex(a + eps)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Targeted hill‑climb on the maximal convolution entry
# ----------------------------------------------------------------------
def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """Return the pair (i,j) with maximal product a[i]*a[j] for i+j = k."""
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    prod = a[idx] * a[k - idx]
    best = prod.argmax()
    i = int(idx[best])
    j = int(k - i)
    return i, j, float(prod[best])


def _targeted_hill_climb(a: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Aggressive local search that attacks the current maximal convolution entry."""
    best_a = a.copy()
    best_sc = _score(best_a, n, L_fft)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        k_max = int(conv.argmax())
        i, j, _ = _max_contributions(best_a, k_max)
        if i == -1:
            continue
        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-12:
            continue
        dst = rng.integers(0, n)
        if dst == src:
            continue
        delta = rng.uniform(0.0, min(best_a[src], 0.004))
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        sc = _score(cand, n, L_fft)
        if sc < best_sc:
            best_sc, best_a = sc, cand
            print(f"[{time.time() - start_time:.2f}s] Hill i={src}→{dst} δ={delta:.5f} → score={sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # tiny safety margin

    # problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    L_fft = _next_pow2(2 * n - 1)       # for a*a
    L_corr = _next_pow2(3 * n - 2)      # for correlations in gradient steps

    # ------------------------------------------------------------------
    # 0️⃣  Initialise (uniform + deterministic seeds + optional global starter)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_sc = _score(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_sc:.6f}")

    # optional global starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _score(starter, n, L_fft)
                if sc < best_sc:
                    best_sc, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # add deterministic handcrafted seeds
    for cand in _deterministic_seeds(n):
        sc = _score(cand, n, L_fft)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            print(f"[0.00s] Deterministic seed score={sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start with Dirichlet draws (≈5 % of total budget)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * (deadline - start_time))
    if time.time() < warm_deadline:
        best_a, best_sc = _warm_start(n, best_a, best_sc, L_fft, warm_deadline)

    # ------------------------------------------------------------------
    # 2️⃣  LP‑based mass‑pump improvement (~30 % of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + 0.30 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Starting LP‑mass‑pump phase")
        best_a, best_sc = _lp_improvement(best_a, n, L_fft, L_corr, lp_deadline)
        print(f"[{time.time() - start_time:.2f}s] LP phase best ≈{best_sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max projected gradient descent (~45 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        grad_deadline = time.time() + 0.45 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_a, best_sc = _smoothmax_descent(best_a, n, L_fft, L_corr, grad_deadline)
        print(f"[{time.time() - start_time:.2f}s] After descent best ≈{best_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Targeted hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb")
        best_a, best_sc = _targeted_hill_climb(best_a, n, L_fft, hill_deadline)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_sc:.6f}")
    return [float(v) for v in best_a.tolist()]
