def _simulated_annealing(a0, L_fft, deadline, start_time, best_seq, best_score, rng):
    a = a0.copy()
    n = a.shape[0]
    cur_score = best_score
    # Temperature schedule: start maybe 1e-3 *cur_score? Or based on difference typical.
    T = 0.001
    # internal parameters: steps per iteration maybe 1 move.
    while time.time() < deadline:
        # propose move
        src = rng.integers(n)
        if a[src] <= 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 5e-3  # up to 0.5% moved
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # ensure non-negative due to numerical error
        a_prop = np.maximum(a_prop, 0.0)
        # renorm ensure sum=1
        a_prop /= a_prop.sum()
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improve score={new_score:.6f}")
        # schedule temperature
        T = max(T * 0.99995, 1e-10)
    return best_seq, best_score

def _grad_softmax_max(a, conv, tau, L_fft_grad):
    # conv is full convolution (size 2n-1)
    max_c = conv.max()
    # stable exponent
    w = np.exp((conv - max_c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)  # uniform
    else:
        w /= w_sum
    w_rev = w[::-1]
    # convolution a (len n) with w_rev (len 2n-1)
    full = np.fft.irfft(np.fft.rfft(a, L_fft_grad) * np.fft.rfft(w_rev, L_fft_grad), n=L_fft_grad)
    # slice indices n-1 to 2*n-1 exclusive
    grad = 2.0 * full[n-1:2*n-1][::-1]
    return grad

"""Propose a step‑function that minimises the L∞ self‑convolution bound.

The algorithm proceeds in three stages:

1️⃣ **Initialisation** – use the supplied global sequence ``height_sequence_1``
   (the best result from the previous run) or fall back to a uniform vector.

2️⃣ **Adam on a smoothed max‑convolution** – we replace the nonsmooth
   objective  `max(conv(a,a))` by a temperature‑controlled log‑sum‑exp.
   Its gradient can be obtained analytically via a single FFT‑based
   cross‑correlation.  Adam (with simplex projection) iteratively reduces
   the true objective while the temperature is slowly annealed.

3️⃣ **Final simulated‑annealing hill‑climb** – tiny random mass transfers
   are accepted with a Metropolis rule; this captures any remaining
   improvements on the exact objective.

All heavy work uses FFTs (`O(n log n)`) and the whole search respects the
1000 s time budget.  The returned list is normalised to sum = 1, all entries
being non‑negative. """

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – assumes the sum is ≈ 1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    # sum is (almost) 1 – keep the expression simple and fast
    n = a.shape[0]
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Gradient of a smoothed max‑convolution (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _grad_softmax_max(
    a: np.ndarray, conv: np.ndarray, tau: float, L_fft_grad: int
) -> np.ndarray:
    """
    Gradient of  τ·log∑exp(conv_i/τ)  w.r.t. a.
    Parameters
    ----------
    a          : current sequence (len n)
    conv       : full self‑convolution of a (len 2·n‑1)
    tau        : smoothing temperature (τ>0)
    L_fft_grad : FFT length for the cross‑correlation (≥3·n‑2)
    Returns
    -------
    grad : np.ndarray, shape (n,)
    """
    # soft‑max weights (numerically stable)
    max_c = conv.max()
    w = np.exp((conv - max_c) / tau)
    w_sum = w.sum()
    w = w / w_sum

    # cross‑correlation a ⋆ w  (length n)
    # compute via convolution with reversed w
    w_rev = w[::-1]                       # length 2·n‑1
    full = np.fft.irfft(
        np.fft.rfft(a, L_fft_grad) * np.fft.rfft(w_rev, L_fft_grad),
        n=L_fft_grad,
    )
    n = a.shape[0]
    # slice indices n‑1 … 2·n‑2  (inclusive) and reverse to obtain increasing k
    grad = 2.0 * full[n - 1 : 2 * n - 1][::-1]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------


def _adam_smooth_opt(
    a0: np.ndarray,
    L_self: int,
    L_grad: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Adam with simplex projection, temperature‑annealed log‑sum‑exp."""
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.25

    # temperature schedule
    tau = 2e-3            # start fairly smooth
    tau_min = 5e-7
    tau_decay = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1

        # exact convolution (needed for both score and gradient)
        conv = _conv_self_fft(a, L_self)
        cur_score = 2.0 * n * conv.max()          # exact objective (sum≈1)

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                f"τ={tau:.2e} score={cur_score:.6f}"
            )

        # gradient of the smoothed max‑convolution
        grad = _grad_softmax_max(a, conv, tau, L_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)          # stay on the simplex (sum=1, ≥0)

        # anneal temperature
        tau = max(tau * tau_decay, tau_min)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb on the exact objective
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    L_self: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer moves with Metropolis acceptance."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = best_score

    # start temperature – a small fraction of the current score
    T = max(1e-4, cur_score * 5e-4)

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a modest fraction of the source mass
        eps = a[src] * rng.random() * 4e-3   # ≤0.4 % of source
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # re‑normalise to stay on the simplex (protect against tiny rounding loss)
        a_prop = _project_simplex(a_prop)

        # exact score
        conv = _conv_self_fft(a_prop, L_self)
        new_score = 2.0 * n * conv.max()

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] SA improve score={new_score:.6f}"
                )

        # geometric cooling
        T = max(T * 0.9999, 1e-12)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a tiny safety margin to avoid overrunning the global timeout
    deadline = start_time + budget_s - 0.15

    # --------------------------------------------------------------
    # Problem size – 4096 provides a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_self = _next_pow2(2 * n - 1)          # for a*a
    L_grad = _next_pow2(3 * n - 2)          # for a ⋆ w (cross‑correlation)

    # --------------------------------------------------------------
    # Starting point – use the globally stored sequence if it exists.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                raise ValueError
            starter = starter / starter.sum()
            a = starter
            best_seq = starter.copy()
            best_score = _eval_fast(a, L_self)
            print(
                f"[{time.time() - start_time:.2f}s] global start score={best_score:.6f}"
            )
        else:
            raise ValueError
    except Exception:
        # fallback: uniform distribution
        a = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = a.copy()
        best_score = _eval_fast(a, L_self)
        print(
            f"[{time.time() - start_time:.2f}s] uniform start score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # 1️⃣  Adam optimisation on a smoothed max‑convolution.
    # --------------------------------------------------------------
    if time.time() < deadline - 250.0:      # reserve time for the final SA phase
        adam_deadline = deadline - 250.0
        best_seq, best_score = _adam_smooth_opt(
            best_seq,
            L_self,
            L_grad,
            adam_deadline,
            start_time,
            best_seq,
            best_score,
            rng,
        )

    # --------------------------------------------------------------
    # 2️⃣  Simulated‑annealing hill‑climb – captures any remaining gains.
    # --------------------------------------------------------------
    while time.time() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq,
            L_self,
            deadline,
            start_time,
            best_seq,
            best_score,
            rng,
        )
        # a few very tiny jitter moves to keep the chain moving
        eps = 1e-5 * rng.random(n) * best_seq
        best_seq = np.maximum(best_seq + eps, 0.0)
        best_seq = _project_simplex(best_seq)

    # --------------------------------------------------------------
    # Final normalisation and return.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:

    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    final_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
