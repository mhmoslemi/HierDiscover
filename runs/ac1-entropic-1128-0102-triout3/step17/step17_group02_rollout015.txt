"""
Propose a step‐function (non‑negative vector) with a low
`evaluate_sequence` score.

Algorithm Sketch
----------------
• Choose a moderate problem size (n = 4096).  The bound 2·n·max(conv) /
  sum² is invariant to scaling, therefore we keep the vector on the
  probability simplex (sum = 1) and minimise 2·n·max(conv).

• Phase‑0: several deterministic “window” shapes (sine, Hann, triangular,
  Gaussian, Kaiser, Tukey).  All are projected onto the simplex and the
  best one is kept.

• Phase‑1: a large batch of random Dirichlet draws (different
  concentration parameters) to obtain a good starting point.

• Phase‑2: smooth projected‑gradient descent.  The nonsmooth max is
  replaced by a log‑sum‑exp (temperature β).  We use an Adam‑style
  optimiser with a schedule β = [0.5,1,2,5,10,20,50,100,200,500,1000,2000].
  After each β‑stage we re‑project onto the simplex.

• Phase‑3: a long random‑mass‑transfer hill‑climb (swap a small fraction
  of the mass between two coordinates).  This fine‑tunes the solution.

All heavy convolutions are performed with an FFT of length a power of two.
The whole routine respects a 1000 s budget; the deadline is split so that
the gradient stage consumes ≈ 90 % of the remaining time, leaving a short
final hill‑climb.

The function returns a plain Python list of floats.
"""

import time
import numpy as np
from math import sqrt, ceil, log

# ----------------------------------------------------------------------
# Global timing handle (used only for printing)
# ----------------------------------------------------------------------
_start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    (v_i ≥ 0, sum v_i = 1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (the internal version – sum = 1, denominator fixed)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Implements  grad = 2 * (a_rev * w)[offset],
    where w are the soft‑max weights over the convolution entries.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()                     # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)     # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Deterministic shapes (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_shape_sine(n: int) -> np.ndarray:
    x = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    return _project_simplex(x)


def _make_shape_hann(n: int) -> np.ndarray:
    x = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    return _project_simplex(x)


def _make_shape_triangular(n: int) -> np.ndarray:
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    return _project_simplex(tri)


def _make_shape_gaussian(n: int, sigma: float = 0.2) -> np.ndarray:
    i = np.arange(n)
    center = (n - 1) / 2.0
    x = np.exp(-0.5 * ((i - center) / (sigma * n)) ** 2)
    return _project_simplex(x)


def _make_shape_kaiser(n: int, beta: float = 8.0) -> np.ndarray:
    # Kaiser window using the modified Bessel function of the first kind
    i = np.arange(n)
    t = (2.0 * i) / (n - 1) - 1.0
    # Use scipy's i0 if available; otherwise fall back to a simple approximation.
    try:
        from scipy.special import i0
        w = i0(beta * np.sqrt(1 - t ** 2)) / i0(beta)
    except Exception:
        # Rough approximation when scipy is not present.
        w = np.exp(beta * np.sqrt(1 - t ** 2))
        w /= w.max()
    return _project_simplex(w)


def _make_shape_tukey(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey window – cosine tapered at the edges."""
    x = np.empty(n)
    for i in range(n):
        r = i / (n - 1)
        if r < alpha / 2:
            x[i] = 0.5 * (1 + np.cos(np.pi * (2 * r / alpha - 1)))
        elif r <= 1 - alpha / 2:
            x[i] = 1.0
        else:
            x[i] = 0.5 * (1 + np.cos(np.pi * (2 * r / alpha - 2 / alpha + 1)))
    return _project_simplex(x)


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return best vector seen among many Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_sc = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in [1e‑3, 5]
        log_c = rng.uniform(-3.0, 0.7)
        c = 10.0 ** log_c
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_sc:
            best_sc = sc
            best_seq = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_sc


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth max)
# ----------------------------------------------------------------------
def _adam_smooth_descent(a0: np.ndarray,
                         beta_vals,
                         deadline: float):
    """
    Adam‑style descent on the smooth max.  After each β‑stage we
    re‑project onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = _objective_score(a)

    # Adam hyper‑parameters
    lr0 = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for beta in beta_vals:
        # reset Adam moments for each temperature
        m.fill(0.0)
        v.fill(0.0)
        step = 0
        while time.time() < deadline:
            step += 1
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            # Adam update (on the raw vector before projection)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            update = lr0 * m_hat / (np.sqrt(v_hat) + eps)

            a_tmp = a - update
            a_tmp = np.maximum(a_tmp, 0.0)
            a_tmp = _project_simplex(a_tmp)

            sc = _objective_score(a_tmp)
            if sc < best_sc:
                best_sc = sc
                best_a = a_tmp.copy()
                print(f"[{time.time() - _start_time:.2f}s] β={beta:.1f} new best {best_sc:.6f}")
                a = a_tmp
                # keep Adam state (m,v) aligned with the new point
                # (simple re‑initialisation keeps things stable)
                m.fill(0.0)
                v.fill(0.0)
            else:
                # accept the move only if it improves the true objective;
                # otherwise stay where we are (basic greedy policy)
                a = a_tmp

            # safety abort if we are out of time
            if time.time() >= deadline:
                break
    return best_a, best_sc


# ----------------------------------------------------------------------
# Random hill‑climbing (fine‑tuning)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float) -> np.ndarray:
    """Simple random mass‑transfer hill‑climb."""
    a = a0.copy()
    cur_sc = _objective_score(a)
    best_sc = cur_sc
    best_a = a.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        delta = a[src] * rng.random() * 0.05  # move ≤5 % of source mass
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(np.maximum(cand, 0.0))
        sc = _objective_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            a = cand
            cur_sc = sc
            print(f"[{time.time() - _start_time:.2f}s] hill‑climb new best {best_sc:.6f}")
        elif sc < cur_sc:
            a = cand
            cur_sc = sc
    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine follows the phases described above
    and returns the best vector (as a plain Python list) found before the
    time deadline.
    """
    # reproducibility
    np.random.seed(seed)
    global _start_time
    _start_time = time.time()
    deadline = _start_time + budget_s - 0.1  # safety margin

    # ------------------------------------------------------------------
    # Problem dimension (4096 is the size used in the reference run)
    # ------------------------------------------------------------------
    n = 4096

    # start from uniform – this gives score = 2 (baseline)
    best_seq = np.full(n, 1.0 / n)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - _start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – deterministic window shapes (several parameter sweeps)
    # ------------------------------------------------------------------
    shape_fns = [
        _make_shape_sine,
        _make_shape_hann,
        _make_shape_triangular,
        lambda n: _make_shape_gaussian(n, sigma=0.15),
        lambda n: _make_shape_gaussian(n, sigma=0.30),
        lambda n: _make_shape_kaiser(n, beta=4.0),
        lambda n: _make_shape_kaiser(n, beta=8.0),
        lambda n: _make_shape_tukey(n, alpha=0.3),
        lambda n: _make_shape_tukey(n, alpha=0.7),
    ]

    for fn in shape_fns:
        try:
            shp = fn(n)
            sc = _objective_score(shp)
            if sc < best_score:
                best_seq, best_score = shp.copy(), sc
                print(f"[{time.time() - _start_time:.2f}s] deterministic shape score {sc:.6f}")
        except Exception:
            continue

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet draws (≈30 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, _start_time + 30.0)
    if time.time() < dirichlet_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=1200, deadline=dirichlet_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - _start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth Adam descent on the simplex
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.90
        beta_schedule = [0.5, 1, 2, 5, 10, 20, 50, 100, 200, 500, 1000, 2000]

        restarts = 12
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.01, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

            sub_deadline = min(descent_deadline,
                               time.time() + (descent_deadline - time.time()) / (restarts - r))
            a_opt, sc_opt = _adam_smooth_descent(a0, beta_schedule, sub_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - _start_time:.2f}s] restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement (last ~10 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - _start_time:.2f}s] starting final hill‑climb")
        best_seq = _hill_climb(best_seq, deadline)

    # ------------------------------------------------------------------
    # Normalise once more for safety and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = _objective_score(best_seq)
    print(f"[{time.time() - _start_time:.2f}s] finished – best internal score ≈ {final_score:.6f}")

    # The external evaluator works on the raw list (it will clamp and
    # re‑normalise internally), so we just return the vector.
    return [float(x) for x in best_seq]
