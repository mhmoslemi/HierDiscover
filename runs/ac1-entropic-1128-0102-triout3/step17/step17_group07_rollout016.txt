def compute_subgradient_max(a, argmax):
    n = a.shape[0]
    M = 2*n - 1
    I = np.zeros(M, dtype=float)
    # average weight among argmax
    weight = 1.0 / len(argmax) if len(argmax)>0 else 0.0
    I[argmax] = weight
    rev_a = a[::-1]
    L = next_pow2(M + n - 1)
    # compute convolution I * rev_a
    conv_full = convolve_fft(I, rev_a) # length M + n - 1
    # central slice
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad

            a = project_simplex(a)

    Af = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _max_contributions(a: np.ndarray, k: int):
    """Return the pair (i,j) with the largest product a_i·a_j for a fixed sum k."""
    n = a.shape[0]

        cur_score = _score_from_conv(conv, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubIter={it:,}  score={cur_score:.6f}")
        eta = eta0 / np.sqrt(it + 1)
        a = _subgradient_max_step(a, conv, n, eta)
        # occasional jitter to escape flat plateaus
        if rng.random() < 0.004:
            eps = rng.normal(scale=0.0005, size=n)
            a = a + eps
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
    return best_a, best_score


def _max_contributions(a: np.ndarray, k: int):
    """For a fixed sum k return the pair (i,j) with maximal a_i·a_j."""
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    max_pos = np.argmax(contrib)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(contrib[max_pos])


def _targeted_hill_climb(a: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float,
                         start_time: float):
    """Aggressive hill‑climb that attacks the current maximal convolution entry."""
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self_fft(best_a, L_fft), n)
    rng = np.random.default_rng()
    n_trials = 12        # how many destination candidates are examined per iteration

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        k_max = int(conv.argmax())
        i, j, _ = _max_contributions(best_a, k_max)
        if i == -1:
            continue
        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-12:
            continue

        best_candidate = None
        best_candidate_score = best_score

        for _ in range(n_trials):
            dst = rng.integers(0, n)
            if dst == src:
                continue
            delta = rng.uniform(0.0, min(best_a[src], 0.004))
            if delta <= 0.0:
                continue
            cand = best_a.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _score_from_conv(_conv_self_fft(cand, L_fft), n)
            if sc < best_candidate_score:
                best_candidate_score = sc
                best_candidate = cand

        if best_candidate is not None:
            best_a = best_candidate
            best_score = best_candidate_score
            print(f"[{time.time() - start_time:.2f}s] Hill src={src}→dst δ={delta:.5f} → score={best_score:.6f}")

    return best_a, best_score


def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float,
                      start_time: float):
    """Mix with the uniform distribution and perform a cheap random‑pair hill‑climb."""
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self_fft(best_a, L_fft), n)

    uniform = np.full(n, 1.0 / n)

    # convex mixing with uniform
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _score_from_conv(_conv_self_fft(cand, L_fft), n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.003))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _score_from_conv(_conv_self_fft(cand, L_fft), n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The search consists of:

    1️⃣ Warm‑start with Dirichlet draws (≈5 % of the budget).
    2️⃣ Projected sub‑gradient descent on the L∞‑objective (≈65 %).
    3️⃣ Targeted hill‑climb on the peak of the convolution (≈20 %).
    4️⃣ Final mixing with the uniform vector and a light random‑pair hill‑climb
       (the remaining time).

    The routine respects the 1000 s wall‑clock limit and returns a plain
    Python list of floats (approximately unit‑sum).
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Initialise (uniform or global starter if available)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _score_from_conv(_conv_self_fft(best_a, L_fft), n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # Try to use a supplied global starter ``height_sequence_1`` if it exists.
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0.0:
                starter = starter / total
                sc = _score_from_conv(_conv_self_fft(starter, L_fft), n)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start (Dirichlet draws)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * (deadline - start_time))
    if time.time() < warm_deadline:
        best_a, best_score = _initial_candidate(
            n, rng, best_a, best_score, L_fft, warm_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent on max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        grad_deadline = time.time() + 0.65 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_a, best_score = _subgradient_descent(
            best_a, n, L_fft, grad_deadline, start_time
        )
        print(f"[{time.time() - start_time:.2f}s] After descent best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Targeted hill‑climb on the peak of the convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + 0.20 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Targeted hill‑climb")
        best_a, best_score = _targeted_hill_climb(
            best_a, n, L_fft, hill_deadline, start_time
        )

    # ------------------------------------------------------------------
    # 4️⃣  Final mixing + light random‑pair hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement")
        best_a, best_score = _final_refinement(
            best_a, n, L_fft, deadline, start_time
        )

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    tot = best_a.sum()
    if tot > 0.0:
        best_a = best_a / tot
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
