    best_score = cur_score

        a_new[i] -= delta
        a_new[j] += delta
        # compute new score via convolution
        conv = _conv_self(a_new)
        new_score = 2.0 * n * conv.max()
        if new_score < best_score - 1e-12:
            # accept
            a = a_new
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
            print(...)
        else:
            no_improve += 1
            # optional occasional acceptance
            if no_improve % 10000 == 0:
                # restart from best random candidate
                a = best_a.copy()
        # maybe after many iterations without improvement, shuffle

import time, numpy as np
from scipy.optimize import linprog

def _conv_self(a):
    n = len(a)
    L = 1 << ((2*n-2).bit_length())
    # using real FFT
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    return conv

def _objective(seq):
    n = len(seq)
    conv = _conv_self(seq)
    maxb = conv.max()
    return float(2.0 * n * maxb / (seq.sum()**2))

def _lp_refine(f, max_iter=3, time_deadline=None):
    # f is numpy array sum=1
    n = f.shape[0]
    # precompute coefficient matrix for conv constraints
    rows = 2*n-1
    # Build A_ub as dense matrix for simplicity
    A_ub = np.zeros((rows, n+1), dtype=np.float64)
    # fill coefficients for g_j where j from 0..n-1
    for k in range(rows):
        # compute for each j: f_{k-j} if 0 <= k-j < n else 0
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # For j = lo..hi, we have f_{k-j}
        # So assign A_ub[k, j] = f_{k-j}
        # vectorized:
        j_vals = np.arange(lo, hi+1)
        i_vals = k - j_vals
        A_ub[k, j_vals] = f[i_vals]
    # set coefficient for M variable (last column)
    A_ub[:, -1] = -1.0
    # RHS zeros
    b_ub = np.zeros(rows, dtype=np.float64)
    # Equality sum(g)=1
    A_eq = np.zeros((1, n+1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    # bounds
    bounds = [(0.0, None)] * n + [(0.0, None)]  # g_j >=0, M>=0
    best_seq = f.copy()
    best_score = _objective(f)
    # iterative refinement
    for it in range(max_iter):
        # Solve LP
        res = linprog(c=np.concatenate([np.zeros(n), np.array([1.0])]), A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x[:n]
        M = res.x[-1]
        # Evaluate self-convolution objective for g
        score_g = _objective(g)
        if score_g < best_score - 1e-12:
            best_score, best_seq = score_g, g.copy()
        # Prepare for next iteration
        f = g
        # recompute A_ub for new f (since constraints depend on f)
        # optional early break if time
        if time_deadline is not None and time.time() >= time_deadline:
            break
        # recompute rows
        for k in range(rows):

def _hill_climb(a, deadline, best_overall_score):
    n = a.shape[0]
    a = a.copy()
    best_seq = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        # randomly select source i with enough mass
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        # fraction
        frac = rng.uniform(0.0, 0.2)
        delta = frac * a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # maintain numerical positivity
        if a_new[i] < 0:
            a_new[i] = 0.0
        # compute score
        sc = _objective_score(a_new)
        if sc < best_score - 1e-12:
            a = a_new
            best_seq = a_new.copy()
            best_score = sc
            # optional log

"""
Propose a non‑negative step‑function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines three ideas that were not used in the previous
submission:

1️⃣  A cheap random‑Dirichlet search on several modest problem sizes
    (n = 256, 512, 1024).  For each size we keep the K best candidates.

2️⃣  For each of those candidates we solve a **linear program**

        minimise   M
        subject to  (f * g)[k] ≤ M   for every k
                    sum(g) = 1
                    g ≥ 0

    where *f* is the candidate we start from.  The constraints are linear
    in the unknown vector g (the convolution with a fixed f is linear),
    therefore a true LP can be used.  The LP is solved with SciPy’s
    HiGHS backend; a few refinement iterations (re‑solve after updating f
    ← g) usually push the maximal convolution noticeably below the
    previous best value.

3️⃣  Finally a lightweight greedy “hill‑climbing” phase moves a tiny
    amount of mass from one position to another and accepts the move if
    the objective improves.  Because the score is cheap to evaluate with
    an FFT‑based convolution, millions of such moves can be tried within
    the 1000 s budget.

All steps respect the security checks of ``evaluate_sequence``:
the returned list is non‑negative, finite and the sum is forced to 1,
so the objective is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT convolution, objective, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a fast FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple hill‑climbing on the simplex
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, best_overall: float):
    """Greedy moves of a tiny amount of mass; accept only improvements."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_score = _objective_score(a)

    # parameters of the random move
    max_frac = 0.25          # at most 25 % of the source weight is moved
    min_move = 1e-12

    while time.time() < deadline:
        i = rng.integers(n)
        if a[i] < min_move:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        frac = rng.uniform(0.0, max_frac)
        delta = frac * a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # numerical safety
        a_new[i] = max(a_new[i], 0.0)
        sc = _objective_score(a_new)
        if sc < best_score - 1e-12:
            a = a_new
            best_seq = a_new.copy()
            best_score = sc
            if best_score < best_overall:
                # a tiny progress report
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (conv(f,g) ≤ M)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, max_iter: int = 4, deadline: float = None):
    """
    Starting from a distribution f (sum = 1),
    repeatedly solve the LP
        minimise   M
        subject to (f * g)[k] ≤ M   ∀k,
                    sum(g) = 1,
                    g ≥ 0 .
    After each solve we set f ← g and repeat.
    Returns the best g encountered and its score.
    """
    try:
        from scipy.optimize import linprog
    except Exception:                # scipy unavailable → skip refinement
        return f, _objective_score(f)

    n = f.shape[0]
    rows = 2 * n - 1                     # one inequality per convolution entry
    # allocate the full (dense) matrix once; it will be overwritten each iteration
    A_ub = np.zeros((rows, n + 1), dtype=np.float64)
    b_ub = np.zeros(rows, dtype=np.float64)

    # equality (sum(g)=1)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    # variable bounds: g_i ≥ 0, M ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    best_seq = f.copy()
    best_score = _objective_score(f)

    for it in range(max_iter):
        # ----- build the convolution constraints for the current f -----
        # each row k contains the coefficients f_{k-j} for those j that keep
        # the index k‑j inside [0,n‑1].
        for k in range(rows):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            j = np.arange(lo, hi + 1, dtype=np.intp)
            i = k - j
            A_ub[k, j] = f[i]            # coefficient for g_j
        A_ub[:, -1] = -1.0                # coefficient for –M ( ≤ 0 )
        # ----- solve the LP -----
        c = np.concatenate([np.zeros(n), np.array([1.0])])   # minimise M
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            # something went wrong – abort further refinement
            break
        g = res.x[:n]
        # occasionally numerical noise may break the sum‑to‑1 constraint,
        # so we renormalise explicitly.
        g = np.maximum(g, 0.0)
        s = g.sum()
        if s == 0:
            break
        g /= s
        # ----- evaluate the new candidate -----
        sc = _objective_score(g)
        if sc < best_score - 1e-12:
            best_seq, best_score = g.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] LP iter {it+1} improved to {sc:.6f}")
        # prepare for the next iteration
        f = g
        # stop early if the global deadline is near
        if deadline is not None and time.time() >= deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    evaluation function.  The routine mixes random Dirichlet sampling,
    a small linear‑programming refinement and a final greedy hill‑climb.
    The best sequence (renormalised to sum = 1) is returned as a plain
    Python list.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # optional global starter (height_sequence_1) if the host supplies it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                sc = _objective_score(starter)
                best_seq = starter.copy()
                best_score = sc
                print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
        else:
            raise ValueError
    except Exception:
        # fallback – start from a uniform distribution of length 256
        n0 = 256
        best_seq = np.full(n0, 1.0 / n0)
        best_score = _objective_score(best_seq)
        print(f"[{time.time() - start_time:.2f}s] No starter, uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameter sets (different lengths allow a better trade‑off between
    #   runtime and achievable bound)
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024]   # increasing size – early passes are cheap
    top_per_len = 15            # keep this many Dirichlet draws per length
    dirichlet_draws = 2000      # how many random draws per length

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.2f}s] ==== Exploring length {n} ====")

        # ---- uniform baseline for this size --------------------------------
        uniform = np.full(n, 1.0 / n)
        sc = _objective_score(uniform)
        if sc < best_score:
            best_score, best_seq = sc, uniform.copy()
            print(f"[{time.time() - start_time:.2f}s] Uniform (n={n}) score={sc:.6f}")

        # ---- a quick geometric sweep (few points, cheap) -------------------
        idx = np.arange(n, dtype=np.float64)
        for eps in np.logspace(-5, -1, 40):
            if time.time() >= deadline:
                break
            r = 1.0 - eps
            geo = np.power(r, idx)
            geo /= geo.sum()
            sc = _objective_score(geo)
            if sc < best_score:
                best_score, best_seq = sc, geo.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} score={sc:.6f}")

        # ---- random Dirichlet draws -----------------------------------------
        rng = np.random.default_rng(seed + n)
        candidates = []   # will hold tuples (score, sequence)
        for _ in range(dirichlet_draws):
            if time.time() >= deadline:
                break
            # concentration between 10⁻² and 5 (log‑uniform)
            logc = rng.uniform(-2.0, 0.7)
            c = 10.0 ** logc
            a = rng.gamma(shape=c, scale=1.0, size=n)
            if a.sum() == 0:
                continue
            a /= a.sum()
            sc = _objective_score(a)

            # keep the K best Dirichlet candidates
            if len(candidates) < top_per_len:
                candidates.append((sc, a))
                candidates.sort(key=lambda x: x[0])
            else:
                if sc < candidates[-1][0]:
                    candidates[-1] = (sc, a)
                    candidates.sort(key=lambda x: x[0])

            # update global best if we happen to beat it
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Dirichlet improved global to {sc:.6f}")

        # ---- linear‑programming refinement of the top candidates ------------
        for idx_c, (sc, cand) in enumerate(candidates):
            if time.time() >= deadline:
                break
            print(f"[{time.time() - start_time:.2f}s] LP‑refine candidate {idx_c+1}/{len(candidates)} "
                  f"(pre‑score={sc:.6f})")
            refined_seq, refined_sc = _lp_refine(cand, max_iter=4, deadline=deadline)
            if refined_sc < best_score:
                best_score, best_seq = refined_sc, refined_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] ** New global best after LP {refined_sc:.6f}")

        # ---- final greedy hill‑climb on the current global best (of this n) --
        # make sure the sequence length matches n
        cur_seq = best_seq
        if cur_seq.shape[0] != n:
            if cur_seq.shape[0] > n:
                cur_seq = cur_seq[:n]
            else:
                # pad with tiny values to keep the sum = 1
                pad = np.full(n - cur_seq.shape[0], 1e-12)
                cur_seq = np.concatenate([cur_seq, pad])
            cur_seq = np.maximum(cur_seq, 0.0)
            cur_seq /= cur_seq.sum()
        hill_seq, hill_sc = _hill_climb(cur_seq, deadline=deadline, best_overall=best_score)
        if hill_sc < best_score:
            best_score, best_seq = hill_sc, hill_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] ** New global best after hill‑climb {hill_sc:.6f}")

    # ----------------------------------------------------------------------
    # final clean‑up and return
    # ----------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        # revert to a safe uniform vector (should never happen)
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
