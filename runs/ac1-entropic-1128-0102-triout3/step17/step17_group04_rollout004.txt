"""
Propose a step‑function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of several cheap but diverse search phases:

1️⃣  Global starter – use the previously found sequence if available.
2️⃣  Random parametric families (two‑ended geometric blocks) – many random
    draws of block‑length, weight split and decay factor.
3️⃣  Random sparse Dirichlet draws – concentrate mass near the ends of the
    interval.
4️⃣  Projected sub‑gradient descent on the true max‑convolution objective.
5️⃣  A linear‑programming “LP‑improve’’ step (the classical alternating
    linear‑program method).  Given the current f (scaled to sum √(2n)),
    we maximise Σb subject to f∗b ≤ max(f∗f) and b≥0, then mix the new g
    back with f.  Repeating this non‑increasingly reduces the bound.

All phases respect the time budget (default 1000 s) and keep the best
sequence found.  The final list is returned in plain Python format.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Objective 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣  Two‑ended geometric block families (randomised)
# ----------------------------------------------------------------------
def _geometric_two_blocks(n: int, L: int, w: float, r: float) -> np.ndarray:
    """Return a symmetric two‑ended geometric distribution.

    - Left block of length L decays with ratio r (a_0 largest).
    - Right block mirrors the left block.
    - Total left weight = w, right weight = 1‑w.
    """
    seq = np.zeros(n, dtype=np.float64)
    if L == 0:
        return seq

    # left block
    left_raw = r ** np.arange(L, dtype=np.float64)
    left_raw /= left_raw.sum()
    seq[:L] = w * left_raw

    # right block (mirrored)
    right_raw = left_raw[::-1]   # same shape, highest at the outermost index
    seq[-L:] = (1.0 - w) * right_raw
    return seq


# ----------------------------------------------------------------------
# 2️⃣  Random sparse Dirichlet draws (mass near the ends)
# ----------------------------------------------------------------------
def _random_sparse_dirichlet(n: int, rng: np.random.Generator,
                             trials: int, deadline: float,
                             best_seq: np.ndarray, best_score: float):
    """Draw Dirichlet vectors on a small random support (biased to the ends)."""
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # support size – between 2 % and 25 % of n
        K = rng.integers(int(0.02 * n), max(int(0.25 * n), 1) + 1)
        # split the support between left and right halves
        k_left = rng.integers(0, K + 1)
        k_right = K - k_left
        left_idxs = rng.choice(np.arange(0, n // 2), size=k_left, replace=False) \
            if k_left > 0 else np.array([], dtype=int)
        right_idxs = rng.choice(np.arange(n // 2, n), size=k_right,
                                replace=False) if k_right > 0 else np.array([], dtype=int)
        idx = np.concatenate([left_idxs, right_idxs])
        if idx.size == 0:
            continue
        vals = rng.gamma(shape=1.0, scale=1.0, size=idx.size)
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = vals
        seq_sum = seq.sum()
        if seq_sum <= 0:
            continue
        seq /= seq_sum
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sparse Dirichlet "
                  f"K={K} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Projected sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float):
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    eta = 0.02          # initial step size
    min_eta = 1e-6
    stagnation = 0
    max_stagnation = 2000

    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t a:
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]               # derivative 2*a_{k-i}

        # try a simple back‑tracking line search
        improved = False
        try_eta = eta
        while try_eta >= min_eta:
            a_new = a - try_eta * grad
            a_new = _project_simplex(a_new)        # stay on the simplex
            new_score = _objective_score(a_new)
            if new_score < cur_score - 1e-12:
                improved = True
                break
            try_eta *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            stagnation = 0
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Sub‑grad improved "
                      f"score={new_score:.6f}")
            eta = min(1.0, eta * 1.05)      # slowly grow step size when succeeding
        else:
            # no progress – stochastic jitter + shrink step size
            rng = np.random.default_rng()
            eps = 1e-4 * rng.random() * a
            a = a + eps
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.5)
            stagnation += 1

        if stagnation >= max_stagnation:
            break
        if best_score <= 1.503:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  LP‑based improvement (classical alternating LP scheme)
# ----------------------------------------------------------------------
def _lp_improve(a0: np.ndarray, deadline: float,
                best_seq: np.ndarray, best_score: float):
    """Repeated linear‑program step described in the statement."""
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except Exception as e:
        print("SciPy not available – skipping LP phase.")
        return best_seq, best_score

    a = a0.copy()
    n = a.shape[0]
    s = math.sqrt(2.0 * n)          # scaling factor so that sum(f)=s
    t = 0.2                         # mixing coefficient (can be tuned)

    while time.time() < deadline:
        # build scaled vector f
        f = a * s
        conv_ff = _conv_self(f)
        M = float(conv_ff.max())    # this is exactly the current score

        # construct convolution matrix A (f * b) ≤ M
        # A has shape (2n‑1, n) with constant diagonals equal to f[d]
        diags_data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
        offsets = np.arange(n)      # 0,…,n‑1
        A = sparse.diags(diags_data, offsets,
                         shape=(2 * n - 1, n), format='csr')

        # linear program: maximise sum(b)  ⇔ minimise -1ᵀb
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})

        if res.status != 0 or res.x is None:
            print("LP failed or unbounded – terminating LP phase.")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            print("LP returned zero sum – terminating LP phase.")
            break

        # scale to meet the normalisation Σg = s
        scale = s / sum_b
        g = b_opt * scale

        # convex combination f_new = (1‑t)f + t g
        f_new = (1.0 - t) * f + t * g
        a_new = f_new / s
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        new_score = _objective_score(a_new)
        if new_score < best_score - 1e-12:
            a = a_new
            best_seq = a_new.copy()
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] LP improve new score={new_score:.6f}")
            # optional: increase mixing when successful
            t = min(0.5, t * 1.1)
        else:
            # no progress – shrink mixing
            t *= 0.7
            if t < 1e-3:
                break

        if best_score <= 1.503:
            break

    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence found (list of floats).
    """
    import numpy as np
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Determine problem size n
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        n = starter.size
        if n == 0:
            raise ValueError
    except Exception:
        n = 4096
        starter = None

    # ------------------------------------------------------------------
    # Baseline – uniform distribution
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Use global starter if it improves the baseline
    # ------------------------------------------------------------------
    if starter is not None:
        seq = np.maximum(starter, 0.0)
        total = seq.sum()
        if total > 0:
            seq = seq / total
            sc = _objective_score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] Global starter improved score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random two‑ended geometric blocks
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    max_phase = deadline - 200.0                 # reserve time for later phases
    while time.time() < max_phase:
        # random parameters
        L = rng.integers(int(0.02 * n), int(0.35 * n) + 1)
        w = rng.random()                     # left weight ∈ (0,1)
        r = rng.uniform(0.4, 0.99)            # decay factor
        seq = _geometric_two_blocks(n, L, w, r)
        if seq.sum() == 0:
            continue
        sc = _objective_score(seq)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] GeoBlock L={L} w={w:.3f} r={r:.3f} score={sc:.6f}")
        # occasional break to stay within time budget
        if time.time() >= max_phase:
            break

    # ------------------------------------------------------------------
    # Phase 2 – random sparse Dirichlet draws (favor ends)
    # ------------------------------------------------------------------
    trials = 3000
    best_seq, best_score = _random_sparse_dirichlet(
        n, rng, trials, deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_seq, best_score = _subgradient_descent(best_seq, deadline,
                                                    best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – LP improvement (classical alternating LP scheme)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting LP improvement")
        best_seq, best_score = _lp_improve(best_seq, deadline,
                                           best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After LP best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    final_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score≈{final_score:.6f}")

    return [float(x) for x in best_seq]
