def _smooth_descent_phase(a0, L_fft, deadline, start_time, best_seq, best_score, rng):
    a = a0.copy()
    n = a.shape[0]
    it = 0
    # schedule for alpha
    alpha_vals = np.logspace(np.log10(5), np.log10(2000), num=15) # from 5 to 2000
    alpha = alpha_vals[0]
    alpha_idx = 0
    step0 = 1.0 # initial step multiplier
    while time.time() < deadline:
        it += 1
        if it % 2000 == 0 and alpha_idx < len(alpha_vals)-1:
            alpha_idx += 1
            alpha = alpha_vals[alpha_idx]
            print(f"[{time.time() - start_time:.2f}s] increasing alpha to {alpha:.1f}")

        # compute conv
        conv = _conv_self_fft(a, L_fft)
        # softmax
        z = alpha * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)

        # gradient via convolution with reversed a
        a_rev = a[::-1]

"""
A fast 1000 s search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  **Coarse parametric sweep** – same family as before (geometric × power) to
    obtain a good initial point.

2️⃣  **Continuation of a smooth max‑approximation** – we replace the
    non‑smooth “max” by a soft‑max (log‑sum‑exp) with a temperature parameter
    α.  Starting with a low α (very smooth) we perform projected gradient descent,
    gradually raising α (continuation) so that the iterates converge to a true
    minimiser of the original L∞‑objective.  The gradient of the smooth proxy
    is obtained analytically via two FFT‑based convolutions, which makes each
    iteration O(n log n).  The step size is scaled by the gradient norm and
    decays like 1/√t.  Small random jitter is added periodically to escape flat
    plateaus.

3️⃣  **Final random hill‑climb** – tiny mass‑transfer moves with a simple
    simulated‑annealing acceptance rule capture any last improvement.

All phases respect the 1000 s budget, keep the vector on the probability simplex
and finally return the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Full linear self‑convolution a * a via FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the exact objective (assumes ∑a ≈ 1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Compute 2·n·max(conv(a,a)) for a vector with sum≈1.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric family a_i ∝ (i+1)^p · r^i
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """
    Build a vector a_i ∝ (i+1)^p * exp(-lam*i) (lam = -log r)
    and normalise to unit L1 norm.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = -lam * i + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)          # numerical stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (continuation on α)
# ----------------------------------------------------------------------


def _smooth_descent_phase(a0: np.ndarray,
                          L_fft: int,
                          deadline: float,
                          start_time: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the L∞‑objective via a smooth log‑sum‑exp surrogate.
    α (the smoothness temperature) is increased gradually (continuation).
    """
    a = a0.copy()
    n = a.shape[0]

    # α schedule – start very smooth, finish very sharp
    alphas = np.logspace(np.log10(5), np.log10(2000), num=15)   # 5 → 2000
    alpha_idx = 0
    alpha = alphas[alpha_idx]

    # FFT length for the second convolution (w * a_rev)
    len_w = 2 * n - 1
    len_a_rev = n
    L_rev = _next_pow2(len_w + len_a_rev - 1)

    it = 0
    step0 = 0.8                     # base step size multiplier

    while time.time() < deadline:
        it += 1

        # increase α occasionally
        if it % 2000 == 0 and alpha_idx < len(alphas) - 1:
            alpha_idx += 1
            alpha = alphas[alpha_idx]
            print(f"[{time.time() - start_time:.2f}s] α increased to {alpha:.1f}")

        # -------- 1) compute self‑convolution ----------
        conv = _conv_self_fft(a, L_fft)                 # length 2n‑1

        # -------- 2) soft‑max weights ----------
        z = alpha * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)                       # stabilise
        w = exp_z / exp_z.sum()                         # sum_i w_i = 1

        # -------- 3) gradient of the smooth proxy ----------
        a_rev = a[::-1]                                 # reversed a
        Wf = np.fft.rfft(w, n=L_rev)
        Ar = np.fft.rfft(a_rev, n=L_rev)
        conv_wr = np.fft.irfft(Wf * Ar, n=L_rev)[: len_w + len_a_rev - 1]
        # extract the slice that corresponds to the gradient
        g = 2.0 * conv_wr[n - 1: n - 1 + n]             # length n

        # -------- 4) step size ----------
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            continue
        step = step0 / (np.sqrt(it) * norm_g)

        # -------- 5) candidate update ----------
        cand = a - step * g
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] smooth it={it:,} α={alpha:.1f} "
                  f"step={step:.3e} score={cand_score:.6f}")
            continue

        # back‑track a few times if no improvement
        improved = False
        step_bt = step
        for _ in range(5):
            step_bt *= 0.5
            cand_bt = a - step_bt * g
            cand_bt = np.maximum(cand_bt, 0.0)
            cand_bt = _project_simplex(cand_bt)
            cand_bt_score = _eval_fast(cand_bt, L_fft)
            if cand_bt_score < best_score:
                best_score = cand_bt_score
                best_seq = cand_bt.copy()
                a = cand_bt
                improved = True
                print(f"[{time.time() - start_time:.2f}s] backtrack it={it:,} α={alpha:.1f} "
                      f"step={step_bt:.3e} score={cand_bt_score:.6f}")
                break
        if improved:
            continue

        # occasional jitter to escape flat plateaus
        if it % 3000 == 0:
            eps = 1e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score
