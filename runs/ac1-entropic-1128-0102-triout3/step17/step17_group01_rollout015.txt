"""
Propose a non‑negative step function that (approximately) minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three layers

1️⃣  A rich “dictionary’’ of simple parametric shapes (geometric‑power,
    power‑law, Dirichlet draws, random permutations).  Every shape is
    normalised to sum = 1.

2️⃣  Linear‑programming mixing: the self‑convolution of a convex combination
    a = Σ w_i·shape_i is a linear combination of the individual convolutions.
    We solve a small LP
        minimise M  s.t.   Σ w_i·conv_i[k] ≤ M  ∀k,
                         Σ w_i = 1, w_i ≥ 0,
    to obtain a mixture whose maximal convolution is (usually) smaller than for
    any single dictionary element.

3️⃣  Local polishing – exponentiated‑gradient, projected sub‑gradient descent
    and a brief simulated‑annealing hill‑climb – are run on the mixed vector
    while time permits.

All steps respect the simplex constraint; the final vector is normalised
to sum = 1 and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Numerical noise may produce tiny negative imag parts – keep real part.
    return conv.real


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2·Σ_j a_j·p_{i+j}.
    Implemented as a convolution of a_rev with p.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast objective (exact)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    # log‑space to avoid overflow/underflow
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)          # stabilise exponentiation
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _power_law_shape(n: int, alpha: float) -> np.ndarray:
    """Pure power‑law shape a_i ∝ (i+1)^{-alpha}, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = i ** (-alpha)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient for the true L∞‑objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g

def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated‑gradient) updates on the exact L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score

        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move (simulated annealing flavour)
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – we keep 4096 which is a good trade‑off.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline – uniform (just to have a reference).
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Incorporate the global starter if present.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Build a dictionary of candidate shapes.
    # --------------------------------------------------------------
    dict_vectors = [uniform]   # start with uniform

    # --- geometric × power shapes -------------------------------------------------
    r_vals = np.linspace(0.985, 0.9999,  thirty := 30)  # more points near 1
    p_vals = np.linspace(-6.0, 0.0,  thirteen := 13)   # mostly front‑loaded
    lam_vals = [0.0, 0.2, 0.5, 0.8]

    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                dict_vectors.append(_mix_with_uniform(shape, lam))

    # --- pure power‑law shapes ----------------------------------------------------
    alphas = np.linspace(0.5, 4.0,  fifteen := 15)
    for alpha in alphas:
        shape = _power_law_shape(n, alpha)
        dict_vectors.append(shape)
        for lam in lam_vals:
            dict_vectors.append(_mix_with_uniform(shape, lam))

    # --- random Dirichlet draws ---------------------------------------------------
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(4):
            dict_vectors.append(rng.dirichlet(np.full(n, alpha)))

    # --- random permutations of a few base shapes (break symmetry) ---------------
    for _ in range(6):
        vec = dict_vectors[rng.integers(len(dict_vectors))].copy()
        rng.shuffle(vec)
        dict_vectors.append(vec)

    # --------------------------------------------------------------
    # Remove obvious duplicates (tiny L2 distance) – cheap way.
    # --------------------------------------------------------------
    uniq = []
    uniq_norm = []
    for v in dict_vectors:
        if len(uniq) == 0:
            uniq.append(v)
            uniq_norm.append(np.linalg.norm(v))
            continue
        # compare with first few stored vectors only (heuristic)
        keep = True
        for u, un in zip(uniq[:10], uniq_norm[:10]):
            if np.linalg.norm(v - u) < 1e-12:
                keep = False
                break
        if keep:
            uniq.append(v)
            uniq_norm.append(np.linalg.norm(v))
    dict_vectors = uniq
    M = len(dict_vectors)
    print(f"[{time.time() - start:.2f}s] dictionary size = {M}")

    # --------------------------------------------------------------
    # 2️⃣  Pre‑compute all self‑convolutions (full length 2n‑1).
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    conv_matrix = np.empty((conv_len, M), dtype=np.float64)   # columns = vectors
    for idx, vec in enumerate(dict_vectors):
        conv_matrix[:, idx] = _conv_self_fft(vec, L_fft)

    # --------------------------------------------------------------
    # 3️⃣  Linear‑programming mixing of dictionary elements.
    # --------------------------------------------------------------
    try:
        import cvxpy as cp
        w = cp.Variable(M)
        M_var = cp.Variable()
        constraints = [w >= 0, cp.sum(w) == 1,
                       conv_matrix @ w <= M_var]   # element‑wise ≤
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        # ECOS works well for moderate sizes; cap iterations to stay within budget.
        solve_deadline = start + 150.0   # leave ~850 s for polishing
        prob.solve(solver=cp.ECOS, max_iters=4000, warm_start=True,
                   verbose=False)
        if w.value is not None:
            mix_weights = np.maximum(w.value, 0.0)
            mix_weights /= mix_weights.sum()
            mixed = np.zeros(n, dtype=np.float64)
            for coeff, vec in zip(mix_weights, dict_vectors):
                mixed += coeff * vec
            mixed = np.maximum(mixed, 0.0)
            mixed /= mixed.sum()
            mixed_score = _eval_fast(mixed, L_fft)
            if mixed_score < best_score:
                best_score = mixed_score
                best_seq = mixed.copy()
                print(f"[{time.time() - start:.2f}s] LP‑mix score={mixed_score:.6f}")
        else:
            print("LP solve failed – using best seed only.")
    except Exception as e:
        print(f"LP step skipped (exception: {e})")

    # --------------------------------------------------------------
    # 4️⃣  Polishing – exponentiated‑gradient & projected sub‑gradient.
    # --------------------------------------------------------------
    # Give the polishing phases roughly the remaining half of the budget.
    remaining = deadline - time.time()
    if remaining > 300.0:
        # exp‑grad (up to deadline‑150 s)
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 150, start,
            best_seq, best_score, rng)

    if time.time() < deadline - 70:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 70, start,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 5️⃣  Final stochastic hill‑climb (tiny mass transfers + occasional swaps).
    # --------------------------------------------------------------
    while time.time() < deadline:
        # move a tiny fraction of mass from src to dst
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 4e-3   # ≤0.4 % of source mass
        cand = best_seq.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            print(f"[{time.time() - start:.2f}s] hill‑climb move score={cand_score:.6f}")
            continue

        # occasionally try a full swap of two entries
        if rng.random() < 0.001:
            i, j = rng.integers(0, n, size=2)
            cand2 = best_seq.copy()
            cand2[i], cand2[j] = cand2[j], cand2[i]
            cand2_score = _eval_fast(cand2, L_fft)
            if cand2_score < best_score:
                best_seq = cand2
                best_score = cand2_score
                print(f"[{time.time() - start:.2f}s] hill‑climb swap score={cand2_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
