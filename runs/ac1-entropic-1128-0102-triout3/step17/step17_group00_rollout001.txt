"""
Propose a non‑negative step‑function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines three ideas:

1️⃣  A fast parametric sweep over the 2‑parameter family
    a_i ∝ (i+1)^p · r^i (geometric × power law) to obtain a good starter.

2️⃣  Smoothed gradient descent on a *soft‑max* approximation of the
    max‑convolution.  For a temperature β the objective is
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k) .
    Its gradient can be written as a convolution and is evaluated by FFT.
    β is increased in stages (10 → 30 → 80 → 200 → 500) so the optimisation
    first sees a smooth landscape and finally focuses on the true max.

3️⃣  A final random hill‑climb (tiny mass transfers) with a tiny
    simulated‑annealing acceptance rule to polish the result.

All updates keep the vector on the probability simplex (non‑negative,
unit L¹‑norm) via Euclidean projection.  The total run time is limited by
the `budget_s` argument (default 1000 s).  The best sequence found is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ v_i = 1, v_i≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – assumes the sum is ≈ 1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with unit L1 norm."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Family a_i ∝ (i+1)^p * r^i  (geometric × power)
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """
    Build a vector a_i ∝ (i+1)^p * exp(-lam*i) (lam = -log r) and L1‑normalise.
    """
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)                # stabilise exp
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smoothed gradient of the soft‑max of the convolution
# ----------------------------------------------------------------------


def _softmax_grad(a: np.ndarray,
                  beta: float,
                  L_fft_self: int,
                  L_fft_cross: int) -> np.ndarray:
    """
    Gradient of  (1/β)·log Σ_k exp(β·conv_k)  w.r.t. a,
    where conv = a * a (full linear convolution).
    Returns a vector of length n.
    """
    n = a.shape[0]

    # 1) self‑convolution (needed for the soft‑max weights)
    conv = _conv_self_fft(a, L_fft_self)               # length 2n‑1
    max_conv = conv.max()
    # soft‑max weights (stable)
    shifted = conv - max_conv
    exp_beta = np.exp(beta * shifted)                 # non‑negative, max = 1
    w = exp_beta / exp_beta.sum()                     # length 2n‑1, sums to 1

    # 2) gradient = 2 * (a_rev * w)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    L = L_fft_cross
    Af = np.fft.rfft(a_rev, n=L)
    Bf = np.fft.rfft(w, n=L)
    conv_rev = np.fft.irfft(Af * Bf, n=L)[: len(a_rev) + len(w) - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (unit L¹‑norm).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_fft_self = _next_pow2(2 * n - 1)
    L_fft_cross = _next_pow2(3 * n - 2)              # for a_rev * w

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft_self)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global starter if it exists (previous best from the system).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / s
            cur = _eval_fast(starter, L_fft_self)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Fine parametric sweep over (λ, p) – ≈70 s.
    # --------------------------------------------------------------
    sweep_deadline = start_time + 70.0
    min_lambda = 4e-05       # very close to 0 (r≈0.99996)
    max_lambda = 1.5e-03     # r≈0.9985
    lam_grid = np.linspace(min_lambda, max_lambda, 300)   # 300 points
    p_grid = np.linspace(-6.0, 3.0, 91)                    # step ≈0.1

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * np.arange(n, dtype=np.float64)
        for p in p_grid:
            log_shape = base + p * np.log(np.arange(1, n + 1, dtype=np.float64))
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft_self)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6e} p={p:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent (soft‑max of the convolution).
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        # β schedule – start smooth, end very sharp.
        beta_schedule = [10.0, 30.0, 80.0, 200.0, 500.0]

        # start from the best candidate found so far
        a = best_seq.copy()
        curr_score = best_score
        for beta in beta_schedule:
            if time.time() >= deadline - 150.0:
                break
            # initial step size for this β
            step = 0.5
            it = 0
            while time.time() < deadline - 150.0:
                it += 1
                grad = _softmax_grad(a, beta, L_fft_self, L_fft_cross)

                # line‑search style acceptance
                cand = a - step * grad
                cand = _project_simplex(cand)
                cand_score = _eval_fast(cand, L_fft_self)

                if cand_score < curr_score:
                    a = cand
                    curr_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] β={beta:.0f} gd it={it:,} "
                              f"step={step:.2e} score={cand_score:.6f}")
                    # be a bit bolder after a success
                    step *= 1.2
                else:
                    # step too large – shrink
                    step *= 0.5
                    if step < 1e-12:
                        break

    # --------------------------------------------------------------
    # 3️⃣  Final random hill‑climb (tiny mass transfers + simulated annealing).
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to 0.5 % of the source mass
        eps = best_seq[src] * rng.random() * 5e-3
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft_self)

        # accept improvement or occasionally accept a slight worsening
        accept = False
        if new_sc < best_score:
            accept = True
        else:
            # simulated‑annealing style acceptance; temperature slowly drops
            elapsed = time.time() - start_time
            temperature = max(1e-4, 0.02 * (1.0 - elapsed / budget_s))
            prob = np.exp(-(new_sc - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
