"""
Smoothed‑max (soft‑max) gradient descent on the step‑function.

The score (with a normalised sum) is   S(a)=2·n·max(conv(a,a)).
Rather than minimising the variance of the convolution (the previous
baseline) we directly smooth the max with a log‑sum‑exp and follow its
gradient on the probability simplex using exponentiated updates
(mirror‑descent with KL‑divergence).  The algorithm proceeds in four
stages:

0️⃣  initialise – uniform + optional global starter and a quick Dirichlet
   warm‑start (few seconds).

1️⃣  soft‑max gradient descent – at each iteration:
       • compute the full self‑convolution,
       • form the soft‑max weights w_k ∝ exp(α·conv_k),
       • obtain the gradient g = 2·corr(w,a) via FFT,
       • update a_i ← a_i·exp(−η·g_i) and renormalise.
   The smoothing parameter α is slowly increased (α≈10 000 at the end)
   and the stepsize η decays like η₀/√t.  This directly attacks the
   current maximal entry of the convolution.

2️⃣  targeted hill‑climb – the same aggressive “move mass from the larger
   partner of the dominant (i,j) pair at the current peak” move as in the
   reference code, but using the best solution from the previous stage.

3️⃣  final polishing – convex mixtures with the uniform vector and cheap
   random‑pair mirror‑descent steps.

The whole routine respects the 1000 s budget and prints intermediate
scores.  The returned list is a non‑negative step‑function with sum≈1,
ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (sum(a)=1 is assumed)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


def _softmax_max_gradient(a: np.ndarray,
                          conv: np.ndarray,
                          alpha: float,
                          L_corr: int) -> np.ndarray:
    """
    Gradient of   L(a)= (1/α)·log Σ_k exp(α·conv_k)   w.r.t. a.
    Using that   ∂conv_k/∂a_i = 2·a_{k-i} (valid indices only),
    we obtain   g = 2·corr(w,a)   where w are the soft‑max weights.
    """
    # soft‑max weights (stable)
    m = conv.max()
    exp_term = np.exp(alpha * (conv - m))
    w = exp_term / exp_term.sum()               # length 2·n‑1

    # correlation w ⋆ a  (valid part)
    a_rev = a[::-1]
    W_fft = np.fft.rfft(w, n=L_corr)
    A_rev_fft = np.fft.rfft(a_rev, n=L_corr)
    full = np.fft.irfft(W_fft * A_rev_fft, n=L_corr)
    n = a.shape[0]
    # extract the “valid” centre part of length n
    grad = 2.0 * full[n - 1: n - 1 + n]
    return grad


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated (mirror‑descent) update on the simplex."""
    a_new = a * np.exp(-eta * grad)
    # numerical safety – avoid all zeros
    if a_new.sum() == 0.0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / a_new.sum()
    return a_new


def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    For a given convolution index k, return the pair (i, j) with the
    largest product a_i·a_j (where i+j=k) together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    max_pos = np.argmax(contrib)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(contrib[max_pos])


# ----------------------------------------------------------------------
# 0️⃣  Warm‑start (uniform, global starter, Dirichlet draws)
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                L_fft: int,
                deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration drawn log‑uniformly in a sensible range
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 1️⃣  Soft‑max gradient descent (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _softmax_gradient_descent(a0: np.ndarray,
                              n: int,
                              L_fft: int,
                              L_corr: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected mirror‑descent minimising a smooth approximation of
    max(conv).  The smoothing parameter α is increased gradually.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    # schedule parameters
    eta0 = 800.0          # initial stepsize for the exponentiated update
    alpha = 2000.0        # start smoothing
    alpha_max = 15000.0
    alpha_inc = 1.001    # multiplicative increase per iteration

    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] GDIter={it:,}  score={cur_score:.6f}")

        # gradient of the smoothed max (log‑sum‑exp)
        grad = _softmax_max_gradient(a, conv, alpha, L_corr)

        # classic 1/√t stepsize
        eta = eta0 / np.sqrt(it + 1.0)

        # mirror‑descent update on the simplex
        a = _exp_update(a, grad, eta)

        # occasional tiny random jitter to escape flat plateaus
        if rng.random() < 0.002:
            eps = rng.normal(scale=5e-4, size=n)
            a = a + eps
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        # increase smoothing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_inc)

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Targeted hill‑climb on the maximal convolution entry
# ----------------------------------------------------------------------
def _targeted_hill_climb(a: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Aggressive peak‑reduction: repeatedly identify the largest contributing
    pair at the current maximal convolution entry and move a tiny amount of
    mass from the larger partner to a random other coordinate.
    """
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        i, j, contrib = _max_contributions(best_a, k_max)
        if i == -1:
            continue

        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-14:
            continue
        dst = rng.integers(0, n)
        if dst == src:
            continue

        # move a modest fraction of the source weight
        delta = rng.uniform(0.0, min(best_a[src], 0.008))
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()

        new_score = _objective_score(cand, n, L_fft)
        if new_score < best_score:
            best_score, best_a = new_score, cand
            print(f"[{time.time() - start_time:.2f}s] Hill i={src}→{dst} δ={delta:.6f} → score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final polishing (mix with uniform + light random‑pair steps)
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair mirror‑descent steps
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.006))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill i={i} j={j} δ={delta:.6f} → score={sc:.6f}")

    return best_a, best_score

            if s > 0.0:

                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
