"""
Search for a non‑negative step function that minimises the objective

    2·n·max(conv(a,a)) / (sum(a)²)

All vectors are kept on the probability simplex (sum = 1) so the denominator is
fixed and the objective reduces to 2·n·max(conv(a,a)).  The optimizer consists
of three phases:

1️⃣  *Broad random sampling*: many Dirichlet draws together with random‑phase
    constructions (the latter approximate a flat autocorrelation).

2️⃣  *Exact sub‑gradient descent* on the simplex.  The gradient of the
    non‑smooth “max‑convolution’’ is obtained analytically (it is the sum of a
    shifted copy of the current vector).  An exponentiated‑gradient (mirror‑
    descent) update keeps iterates inside the simplex without a separate
    projection step.

3️⃣  *Fine stochastic hill‑climbing*: tiny mass‑transfer moves with a
    Metropolis‑type acceptance allow the final few seconds to polish the best
    candidate.

The whole routine respects a 1000 s budget and returns a plain ``list`` of
floats ready for ``evaluate_sequence``.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using an FFT of length a power‑of‑2."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a ∗ a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

def _random_dirichlet(n: int, max_trials: int, deadline: float, rng: np.random.Generator):

# ----------------------------------------------------------------------
# Exact sub‑gradient descent (mirror‑descent on the simplex)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, step0: float, time_limit: float):
    """
    Mirror‑descent on the simplex using the exact sub‑gradient of
    max(conv(a,a)).  The update is   a ← a·exp(-η·grad)  (component‑wise)  and
    finally re‑normalised.
    """
    a = a0.copy()
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score
    step = step0
    n = a.shape[0]

    while time.time() < time_limit:
        conv = _conv_self(a)
        max_val = conv.max()
        # random tie‑breaking among all maximal indices
        max_idxs = np.flatnonzero(conv == max_val)
        k = max_idxs[np.random.randint(len(max_idxs))]
        offset = abs(k - (n - 1))

        # g = a shifted left + a shifted right (the sub‑gradient of the max entry)
        if offset < n:
            g = np.zeros_like(a)
            g[: n - offset] += a[offset:]
            g[offset:] += a[: n - offset]
        else:                     # offset == n‑1 is the extreme case (no overlap)
            g = np.zeros_like(a)

        # sub‑gradient of f(a) = 2·n·max(conv)  (sum(a)=1)
        grad = g - 2.0 * max_val * np.ones_like(a)

        # exponentiated‑gradient update (mirror‑descent with KL geometry)
        clip_val = 50.0                # avoid overflow in exp()
        exp_factor = np.exp(-step * np.clip(grad, -clip_val, clip_val))
        a = a * exp_factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        new_score = _objective_score(a)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] sub‑grad new best {best_score:.6f}")
            # slowly increase step when we make progress
            step = min(step * 1.05, 2.0)
        else:
            # otherwise shrink the step
            step = max(step * 0.5, 1e-6)

    return best_a, best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        delta = a[src] * rng.random() * 0.05          # move up to 5 % of src mass
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(cand)                  # stay on the simplex
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            # tiny Metropolis acceptance of uphill moves
            a = cand
            cur_score = cand_score
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Locate a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain list) found within the time budget.
    """
    global start_time
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between expressive power and speed
    # ------------------------------------------------------------------
    n = 4096
    best_a = np.full(n, 1.0 / n)                    # uniform start
    best_score = _objective_score(best_a)
    print(f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_a = starter.copy()
                    best_score = sc
                    print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Phase 1 – massive random sampling (Dirichlet + random‑phase)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 120.0)   # first two minutes
    if time.time() < phase1_deadline:
        seq, sc = _random_dirichlet(n, max_trials=3000, deadline=phase1_deadline, rng=rng)
        if seq is not None and sc < best_score:
            best_a, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

        seq2, sc2 = _random_phase_init(n, max_trials=800, deadline=phase1_deadline, rng=rng)
        if seq2 is not None and sc2 < best_score:
            best_a, best_score = seq2, sc2
            print(f"[{time.time() - start_time:.2f}s] after phase‑1 phase‑2 best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient (mirror) descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.80   # most of the remaining budget
        restarts = 8
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break
            # small random jitter around the current best
            jitter = rng.normal(scale=0.02, size=n)
            a0 = best_a + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

            sub_deadline = min(descent_deadline,
                               time.time() + (descent_deadline - time.time()) / (restarts - r))
            a_opt, sc_opt = _subgradient_descent(a0, step0=0.5, time_limit=sub_deadline)
            if sc_opt < best_score:
                best_a, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting hill‑climbing")
        best_a, best_score = _hill_climb(best_a, best_score, deadline)

    # final sanity normalisation
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
