ℓ∞‑norm of the autocorrelation (self‑convolution).

The algorithm starts from the best known vector (the global variable

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2
    length.  ``a`` is assumed to be a 1‑D real array.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # extreme case – numerical
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector (∑a=1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_gradient(a: np.ndarray, beta: float, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max,
        fβ(a) = (1/β)·log Σ_k exp(β·conv_k(a))

    ∇fβ = 2· Σ_k w_k·shift_k(a)   with   w_k = exp(β·conv_k)/Σ exp(β·conv)
    The computation is performed via a single FFT.
    """
    n = a.shape[0]
    # soft‑max weights
    maxc = conv.max()
    exp_beta = np.exp(beta * (conv - maxc))          # subtract max for stability
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp                          # shape (2·n‑1,)

    # convolution w * a_rev  → extracts Σ_k w_k·a_{k-j}
    a_rev = a[::-1]
    L = _next_pow2((2 * n - 1) + n - 1)              # length needed for linear conv
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A, n=L)[: (2 * n - 1) + n - 1]
    # the needed slice starts at index (n‑1) and has length n
    grad = 2.0 * conv_full[n - 1: n - 1 + n]
    return grad

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_seq = a.copy()
            best_score = cur_score
            print(f"[{time.time() - _start_time:.2f}s] New best (exact) = {best_score:.6f}")

        # --- gradient of smoothed max ---------------------------------
        grad = _smooth_gradient(a, beta, conv)

        # momentum update (Nesterov‑style)
        momentum = mom_beta * momentum + (1.0 - mom_beta) * grad
        direction = momentum

        # --- simple back‑tracking line‑search on the exact objective ----
        improved = False
        for factor in (1.0, 0.5, 0.25):
            step = lr * factor
            cand = a - step * direction
            cand = _project_simplex(cand)      # keep on simplex
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                lr = step * 1.05                # slightly increase for next step
                improved = True
                break
        if not improved:
            lr = max(lr * 0.5, 1e-12)          # shrink learning rate

        # --- anneal β -------------------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # --- occasional random jitter to escape flat regions ----------
        if rng.random() < 0.01:
            jitter = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)            # re‑normalise

    # ------------------------------------------------------------------
    # Final sanity checks and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = _objective_score(best_seq)
    print(f"[{time.time() - _start_time:.2f}s] Finished – final objective ≈ {final_score:.6f}")

    return [float(x) for x in best_seq]
