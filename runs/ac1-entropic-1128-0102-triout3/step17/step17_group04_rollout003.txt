def _smooth_max_and_grad_fixed(a: np.ndarray, beta: float):
    n = a.shape[0]
    conv = _conv_self(a)  # length 2n-1
    # compute weights w via softmax
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    Z = exp_scaled.sum()
    w = exp_scaled / Z
    # gradient = 2 * correlation of a and w
    # compute correlation via convolution with reversed w
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)  # length n + (2n-1) -1 = 3n-2
    start = len(w) - 1  # 2n-2
    grad = 2.0 * conv_rev[start : start + n]  # pick forward slice
    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, grad
```

def _random_hill_climb(a, best_score, deadline, step=1000):
    rng = np.random.default_rng()
    cur = a.copy()
    cur_score = best_score
    while time.time() < deadline:
        # sample two indices
        i, j = rng.integers(0, len(cur), size=2)
        if i == j:
            continue
        # amount to transfer
        delta = rng.random() * cur[i] * 0.1
        if delta == 0:
            continue
        cur[i] -= delta
        cur[j] += delta
        # ensure non-neg
        cur = np.clip(cur, 0, None)
        cur /= cur.sum()
        new_score = _objective_score(cur)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur.copy()
        else:
            # revert
            # simpler: continue to next iteration
            pass
    return best_seq, best_score

"""
Propose a non‑negative step function (list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search works in four phases (time‑limited, total budget ≤ 1000 s).

1️⃣  Simple parametric families (geometric, symmetric exponential, Gaussian,
    triangular, two‑block).  These cheap sweeps give a decent starter.

2️⃣  Random Dirichlet draws with a wide range of concentration parameters.

3️⃣  Smooth‑max mirror‑descent on the full simplex.
    We replace the non‑smooth max by a log‑sum‑exp surrogate, compute its
    exact gradient (correcting the sign error of the previous code) and
    perform exponential updates with a line‑search on the true objective.
    The smoothness parameter β grows from 0.5 to 80, while the step size η
    adapts automatically.

4️⃣  A short stochastic hill‑climb (pairwise mass transfers) to escape any
    remaining plateau.

The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (returns full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Objective 2·n·max(conv(seq, seq)) assuming sum(seq)=1.
    (If the sum deviates from 1 the denominator of the original
    evaluate_sequence would change, but we keep the normalisation.)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient) where gradient is d/d a.
    This version uses the correct gradient direction (no reversal).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)      # stability shift
    Z = exp_scaled.sum()
    w = exp_scaled / Z                            # soft‑max weights, sum to 1

    # gradient = 2 * correlation of a with w
    w_rev = w[::-1]                               # reversed for correlation
    conv_rev = _conv_fft(a, w_rev)                # length 3n‑2
    start = len(w) - 1                            # = 2n‑2, lag 0 position
    grad = 2.0 * conv_rev[start : start + n]      # forward slice = corr[a,w]

    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, grad


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponential (mirror‑descent) update on the simplex:
        a_i ← a_i·exp(−η·grad_i)   then renormalise.
    """
    # normalise gradient to avoid gigantic steps when grad is large
    max_abs = np.max(np.abs(grad))
    if max_abs > 0:
        grad = grad / max_abs
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)                # avoid overflow
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        # fall back to a uniform distribution
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total

# 1️⃣  Simple parametric families (very light sweep)

def _quick_sweeps(n, best_seq, best_score, deadline):
    """Run a few cheap parametric families and keep the best result."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # geometric (decay from the left)
    eps_grid = np.logspace(-5, -1, 80)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        sc = _objective_score(geo)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} score={sc:.6f}")

    # symmetric exponential (distance from centre)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 60):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")

    # Gaussian‑like bumps centred in the middle
    for sigma in np.logspace(0.5, 3.0, 45):   # sigma from ~3 to 1000
        if time.time() >= deadline:
            break
        seq = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Gauss σ={sigma:.1f} score={sc:.6f}")

    # triangular (piece‑wise linear, width controls flatness)
    for width in np.linspace(0.05, 0.5, 20):
        if time.time() >= deadline:
            break
        half = int(width * n / 2)
        seq = np.zeros(n)
        start = max(0, int(center) - half)
        end = min(n, int(center) + half + 1)
        if start < end:
            seq[start:end] = 1.0
            # make it truly triangular
            ramp = np.arange(1, end - start + 1)
            seq[start:end] = np.minimum(ramp, (end - start + 1) - ramp)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Tri width={width:.2f} score={sc:.6f}")

    # two‑block (mass at both ends)
    for L_frac in np.linspace(0.02, 0.25, 12):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in np.linspace(0.1, 0.9, 9):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")

    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=2000):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in roughly [0.01, 5]
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score

# 3️⃣  Smooth‑max mirror descent (core optimiser)
# ----------------------------------------------------------------------
def _smooth_descent(a0, deadline, best_seq, best_score):
    """
    Minimise a smooth surrogate of max(conv) with an exponential update.
    Corrects the gradient direction used in the previous implementation.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    beta_start, beta_end = 0.5, 80.0
    eta = 0.5                 # initial step size (will be scaled internally)
    min_eta = 1e-12
    stagnation = 0

    while time.time() < deadline:
        # linear schedule for β (more aggressive as we go)
        t = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + t * (beta_end - beta_start)

        smooth, grad = _smooth_max_and_grad(a, beta)

        # try an exponential update
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, grad, eta_try)
            new_score = _objective_score(a_new)
            if new_score < cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            stagnation = 0
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Smooth β={beta:.2f} η={eta_try:.2e} score={new_score:.6f}")
            # slowly increase η when things go well
            eta = min(1.0, eta * 1.02)
        else:
            # no improvement – inject a tiny random perturbation
            stagnation += 1
            rng = np.random.default_rng()
            eps = 1e-4 * rng.random(a.shape) * a
            a = np.clip(a + eps, 0.0, None)
            a /= a.sum()
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.5)

        # early exit if we have already crossed the target
        if best_score <= 1.503:
            break

        # occasional aggressive re‑normalisation to keep numeric health
        if stagnation >= 100:
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stagnation = 0

    return best_seq, best_score

def _random_hill_climb(a0, best_score, deadline, steps_per_sec=2000):
    """
    Very cheap local search: repeatedly move a small random amount of mass
    from one position to another.  Accept only improving moves.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_score = best_score
    n = cur.shape[0]

    # target number of proposals is roughly steps_per_sec * remaining_seconds
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a small fraction of the mass at i to j
        delta = rng.random() * cur[i] * 0.05  # up to 5 % of the source mass
        if delta <= 0:
            continue
        proposal = cur.copy()
        proposal[i] -= delta
        proposal[j] += delta
        # re‑normalise (numerically stable)
        proposal = np.clip(proposal, 0.0, None)
        proposal /= proposal.sum()
        new_score = _objective_score(proposal)
        if new_score < cur_score:
            cur = proposal
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={new_score:.6f}")

    return best_seq, best_score
