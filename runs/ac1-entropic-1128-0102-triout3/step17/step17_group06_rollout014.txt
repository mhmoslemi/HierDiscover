Pseudo:

```
def proj_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n = 4096
    L = _next_pow2(3*n - 2)  # for gradient steps
    L_conv = _next_pow2(2*n - 1)  # for scoring, maybe not needed

    uniform = np.full(n, 1.0/n)

    # initializer: try global if exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            ssum = starter.sum()
            if ssum > 0.0:
                starter = starter / ssum
                init_a = starter
    except Exception:
        init_a = uniform

    # also try beta shape as alternative candidate
    beta = _beta_shape(n, 0.5)
    beta_mix = 0.05 * uniform + 0.95 * beta
    beta_mix = beta_mix / beta_mix.sum()

    # pick better of init_a and beta_mix based on fast score
    def fast_score(a):
        b = np.fft.irfft(np.fft.rfft(a, n=L_conv) * np.fft.rfft(a, n=L_conv), n=L_conv)
        max_b = b[:2*n-1].max()
        return 2.0 * n * max_b

    score_init = fast_score(init_a)
    score_beta = fast_score(beta_mix)

    if score_beta < score_init:
        a = beta_mix.copy()
        best_score = score_beta
    else:
        a = init_a.copy()
        best_score = score_init

    # ensure a on simplex (should be)
    a = proj_simplex(a)

    best_a = a.copy()

    print(f"[{time.time() - start:.2f}s] start score = {best_score:.6f}")

    # parameters
    eta = 0.1
    eta_decay = 0.9995
    p = 5.0
    p_inc = 0.2
    p_max = 200.0

    # current score
    cur_score = best_score

    iter_count = 0
    next_print = start + 5.0   # print every ~5 sec

    while time.time() < deadline:
        # compute FFTs
        A = np.fft.rfft(a, n=L)
        # self-convolution
        b_full = np.fft.irfft(A * A, n=L)
        b = b_full[:2*n-1]
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # if current is best, update
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # optional print
            if time.time() >= next_print:
                print(f"[{time.time() - start:.2f}s] new best score = {best_score:.6f}")
                next_print = time.time() + 5.0

        # compute softmax weights
        shift = max_b
        # p may be large but safe
        exp_arg = p * (b - shift)
        # to avoid underflow for large negative values, we can set very low values to zero
        # but using np.exp is okay
        w = np.exp(exp_arg)
        w_sum = w.sum()
        w = w / w_sum

        # gradient via correlation
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(np.conj(A) * W, n=L)
        grad = 2.0 * grad_full[:n]   # first n entries

        # update a via projected gradient descent
        a_proposed = proj_simplex(a - eta * grad)

        # evaluate new candidate
        A_new = np.fft.rfft(a_proposed, n=L)
        b_full_new = np.fft.irfft(A_new * A_new, n=L)
        max_b_new = b_full_new[:2*n-1].max()
        new_score = 2.0 * n * max_b_new

        if new_score < cur_score:
            a = a_proposed
            cur_score = new_score

                if time.time() >= next_print:

                    next_print = time.time() + 5.0

        if p < p_max:

"""
Propose a step‑function that minimises
        score = 2·n·max(conv(a,a)) / (sum(a)²)

Algorithm
---------
* Initialise a distribution on the simplex (mostly the arcsine/Beta(0.5,0.5)
  law, optionally mixed with a little uniform mass).
* Run a projected‑gradient descent on a smooth surrogate of the
  L∞‑norm of the auto‑convolution:  J(a)= (1/p)·log∑_k exp(p·b_k)   with
  b = a∗a.  The gradient of J is 2·(w⋆a) where  w are the soft‑max weights.
  Correlation is evaluated by FFT (no O(n²) loops).
* The step size η decays slowly, while the temperature parameter p is
  increased so that the optimisation focuses more on the true maximum.
* After the gradient phase a cheap random pairwise “hill‑climb’’ finetunes the
  solution.
* The best vector found is projected onto the probability simplex and
  returned.  All heavy work is done with FFTs, so the 1000 s budget is
  easily respected.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (sum(a) is assumed to be 1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised Beta(p,p) density on the grid i=0,…,n‑1,
    i.e. a_i ∝ x_i^{p‑1} (1‑x_i)^{p‑1} with x_i = (i+0.5)/n.
    Returned vector is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex (non‑negative, sum=1)."""
    # based on the algorithm of Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and prints
    progress.  The returned list contains non‑negative entries that sum to 1.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate size keeps the final np.convolve cheap
    # --------------------------------------------------------------
    n = 4096
    L_corr = _next_pow2(3 * n - 2)          # length for correlation (grad)
    L_conv = _next_pow2(2 * n - 1)          # length used only for scoring

    # --------------------------------------------------------------
    # 0️⃣  Baseline initialisation
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)

    # try a global starter if present
    init_a = uniform.copy()
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            ss = starter.sum()
            if ss > 0.0:
                init_a = starter / ss
    except Exception:
        pass

    # also try the classic arcsine (Beta(0.5,0.5)) with a tiny uniform mix
    beta = _beta_shape(n, 0.5)
    beta_mix = 0.05 * uniform + 0.95 * beta
    beta_mix = beta_mix / beta_mix.sum()

    # pick the better of the two using the fast score
    def _fast_score(v):
        conv = _conv_self_fft(v, L_conv)
        return 2.0 * n * conv.max()

    if _fast_score(beta_mix) < _fast_score(init_a):
        a = beta_mix.copy()
        best_score = _fast_score(a)
    else:
        a = init_a.copy()
        best_score = _fast_score(a)

    a = _proj_simplex(a)                # make sure we are on the simplex
    best_a = a.copy()
    cur_score = best_score

    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Projected gradient descent on a smoothed max‑objective
    # --------------------------------------------------------------
    eta = 0.20                     # initial step size
    eta_decay = 0.9998             # slow decay
    p = 5.0                        # start with a modest smoothing parameter
    p_inc_factor = 1.006           # multiplicative increase
    p_max = 200.0

    it = 0
    next_report = start_time + 5.0

    while time.time() < deadline:
        # ----- convolution and current score -----
        A = np.fft.rfft(a, n=L_corr)
        b_full = np.fft.irfft(A * A, n=L_corr)          # linear a⋆a (zero‑padded)
        b = b_full[: 2 * n - 1]
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if time.time() >= next_report:
                print(f"[{time.time() - start_time:.2f}s] new best = {best_score:.6f}")
                next_report = time.time() + 5.0

        # ----- soft‑max weights (smooth max) -----
        # subtract max to avoid overflow
        w_raw = np.exp(p * (b - max_b))
        w = w_raw / w_raw.sum()          # this vector sums to 1

        # ----- gradient = 2·(w ⋆ a) -----
        W = np.fft.rfft(w, n=L_corr)
        grad_full = np.fft.irfft(np.conj(A) * W, n=L_corr)
        grad = 2.0 * grad_full[:n]       # correlation values for lags 0…n‑1

        # ----- projected gradient step -----
        a_candidate = _proj_simplex(a - eta * grad)

        # ----- evaluate candidate -----
        A_c = np.fft.rfft(a_candidate, n=L_corr)
        b_c_full = np.fft.irfft(A_c * A_c, n=L_corr)
        max_b_c = b_c_full[: 2 * n - 1].max()
        cand_score = 2.0 * n * max_b_c

        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                if time.time() >= next_report:
                    print(f"[{time.time() - start_time:.2f}s] improved after step = {best_score:.6f}")
                    next_report = time.time() + 5.0

        # ----- update hyper‑parameters -----
        eta *= eta_decay
        p = min(p_max, p * p_inc_factor)

        it += 1

        # occasional tiny random nudge to avoid stagnation
        if it % 2000 == 0:
            noise = np.random.rand(n) * 1e-4
            a = _proj_simplex(a + noise)

    # --------------------------------------------------------------
    # 2️⃣  Final cheap hill‑climb (pairwise mass transfers)
    # --------------------------------------------------------------
    a = best_a.copy()
    rng = np.random.default_rng(seed + 12345)
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.random() * min(a[i], 0.001)
        if delta == 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _proj_simplex(cand)

        # evaluate
        A_c = np.fft.rfft(cand, n=L_corr)
        b_c = np.fft.irfft(A_c * A_c, n=L_corr)[: 2 * n - 1]
        cand_score = 2.0 * n * b_c.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            if time.time() >= next_report:
                print(f"[{time.time() - start_time:.2f}s] hill‑climb improve → {best_score:.6f}")
                next_report = time.time() + 5.0

    # --------------------------------------------------------------
    # Normalisation & return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
