def _annealing_phase(a0, L_fft, deadline, start_time, best_seq, best_score, rng):

    a = a0.copy()
    n = len(a)
    T0 = 1e-2
    T_end = 1e-5
    total_time = deadline - start_time
    # temperature schedule: exponential decay based on fraction of elapsed time.
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T0 * (T_end/T0) ** frac
        # propose move
        src = rng.integers(n)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # epsilon magnitude based on temperature: scaling
        eps = a[src] * max(0.001, T) * rng.random()
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # ensure non-negative
        if a_prop[src] < 0:
            a_prop[src] = 0
        # renorm (should remain sum close to 1)
        # To avoid drift: after move sum unchanged, so we can leave as is.
        # But rounding errors may cause small drift; renorm anyway.
        a_prop = np.maximum(a_prop, 0.0)
        s = a_prop.sum()
        if s == 0:
            continue
        a_prop /= s
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - best_score
        if delta < 0 or rng.random() < np.exp(-delta/T):
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_seq = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal score={new_score:.6f}")
    return best_seq, best_score

"""
Optimisation of a non‑negative step function for the objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search is performed in several timed phases:

1️⃣  Baseline (uniform and optional global starter).
2️⃣  Fine grid scan of the geometric × power family (λ, p) on a wider range.
3️⃣  Scan of a Beta‑type family a_i ∝ (i+1)^p·(n‑i)^q.
4️⃣  Random sampling of mixed exponentials, Beta shapes and small jitter moves.
5️⃣  Projected sub‑gradient descent on the exact L∞‑objective.
6️⃣  Exponentiated‑gradient (multiplicative) refinement.
7️⃣  Targeted mass‑transfer hill‑climb guided by the sub‑gradient.
8️⃣  Final tiny random perturbations (simulated‑annealing style).

All updates keep the vector on the probability simplex.
The function stops just before the time budget expires and returns the best
sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – assumes the sum is ≈ 1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^p·exp(-lam·i)   (lam may be positive or negative).
    Result is normalised to unit L1 norm.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = -lam * i + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)          # stabilise exp
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_power(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^p·(n-i)^q  (Beta‑type polynomial family).
    Result is normalised to unit L1 norm.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_mixture_geo(n: int, lam1: float, lam2: float, w: float) -> np.ndarray:
    """Convex mixture of two pure exponentials (p=0)."""
    a1 = _make_geo_power(n, lam1, 0.0)
    a2 = _make_geo_power(n, lam2, 0.0)
    shape = w * a1 + (1.0 - w) * a2
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient for the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]               # includes the diagonal correctly
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on max(conv) with a simple back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.35
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # decreasing step length
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(10):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                      f"step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move (tiny chance)
            if rng.random() < 0.001:
                a = cand

        if it % 2000 == 0:
            # add a jitter to escape plateaus
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) updates on the exact L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.8
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # annealed step size
        eta = eta0 / (1.0 + 0.0008 * it)

        mult = np.exp(-eta * g)
        a = a * mult
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")

        # occasional jitter
        if it % 3000 == 0:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best_seq, best_score

def _random_sampling_phase(n: int,

# ----------------------------------------------------------------------
# Targeted mass‑transfer hill‑climb guided by sub‑gradient
# ----------------------------------------------------------------------


def _targeted_transfer_phase(a0: np.ndarray,
                             L_fft: int,
                             deadline: float,
                             start_time: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Move mass from indices with large sub‑gradient to those with small ones."""
    a = a0.copy()
    n = a.shape[0]
    while time.time() < deadline:
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # pick source among the largest sub‑gradient entries that still hold mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[g[src_candidates].argmax()]

        # destination is the smallest sub‑gradient entry (could be zero)
        dst = int(np.argmin(g))

        if src == dst:
            dst = rng.integers(n)

        # transfer a modest fraction of the source mass
        eps = a[src] * rng.random() * 0.08   # ≤8 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # renormalise (numerical drift)
        a_prop = np.maximum(a_prop, 0.0)
        s = a_prop.sum()
        if s == 0.0:
            continue
        a_prop /= s

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] targeted transfer score={new_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final tiny random perturbations (simulated‑annealing style)
# ----------------------------------------------------------------------


def _final_annealing_phase(a0: np.ndarray,
                           L_fft: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very small random moves with a temperature that cools towards zero."""
    a = a0.copy()
    n = a.shape[0]
    total_time = deadline - start_time
    if total_time <= 0:
        return best_seq, best_score
    T0 = 2e-2
    Tend = 5e-6
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T0 * (Tend / T0) ** frac

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * max(0.001, T)   # scale with temperature
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = np.maximum(a_prop, 0.0)
        s = a_prop.sum()
        if s == 0.0:
            continue
        a_prop /= s
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_seq = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] annealing score={new_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (the vector is L1‑normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate size that balances quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline candidates.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # optional global starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – fine grid sweep of geometric×power family (larger search space).
    # --------------------------------------------------------------
    sweep_deadline = start_time + 70.0
    min_lambda = -0.0015          # allow a small negative decay (growing tail)
    max_lambda = 0.0050           # relatively faster decay
    lam_grid = np.linspace(min_lambda, max_lambda, 500)   # 500 points
    p_grid = np.linspace(-10.0, 5.0, 151)                 # step 0.1

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * np.arange(n, dtype=np.float64)
        for p in p_grid:
            log_shape = base + p * np.log(np.arange(1, n + 1, dtype=np.float64))
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep geo λ={lam:.7f} "
                      f"p={p:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – sweep of Beta‑type polynomial family.
    # --------------------------------------------------------------
    beta_deadline = sweep_deadline + 40.0
    p_grid2 = np.linspace(-12.0, 8.0, 101)   # p ∈ [-12,8]
    q_grid2 = np.linspace(-12.0, 8.0, 101)   # q ∈ [-12,8]
    for p in p_grid2:
        if time.time() >= beta_deadline:
            break
        log_i = np.log(np.arange(1, n + 1, dtype=np.float64))
        log_ni = np.log(n - np.arange(0, n, dtype=np.float64))
        base = p * log_i
        for q in q_grid2:
            log_shape = base + q * log_ni
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep beta p={p:.2f} q={q:.2f} "
                      f"score={sc:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – random sampling of several families.
    # --------------------------------------------------------------
    rand_deadline = beta_deadline + 200.0
    best_seq, best_score = _random_sampling_phase(
        n, L_fft, rand_deadline, start_time, best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Phase 4 – Projected sub‑gradient descent.
    # --------------------------------------------------------------
    if time.time() < deadline - 260.0:
        sub_deadline = deadline - 260.0
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, sub_deadline, start_time,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Phase 5 – Exponentiated‑gradient refinement.
    # --------------------------------------------------------------
    if time.time() < deadline - 160.0:
        exp_deadline = deadline - 160.0
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, exp_deadline, start_time,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Phase 6 – Targeted mass‑transfer hill‑climb (guided by sub‑gradient).
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        targ_deadline = deadline - 80.0
        best_seq, best_score = _targeted_transfer_phase(
            best_seq, L_fft, targ_deadline, start_time,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Phase 7 – Final tiny random perturbations (simulated‑annealing style).
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 5e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 2e-3   # ≤0.2 % of source mass
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final tiny move score={new_sc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
