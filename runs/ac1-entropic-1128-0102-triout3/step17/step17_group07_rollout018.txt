def conv_max_geometric(r, n):
    if r <=0 or r>=1: handle.
    alpha = (1 - r)/(1 - r**n)
    # find k0 = -1/ln(r) - 1. Get integer k0 in 0..n-1.
    k_opt = int(math.floor(-1.0 / math.log(r) - 1.0))
    if k_opt < 0: k_opt = 0
    if k_opt >= n: k_opt = n-1
    # evaluate at k_opt and maybe k_opt+1 (if within n)
    conv_vals = []
    for k in [k_opt-1, k_opt, k_opt+1]:
        if 0 <= k < n:
            conv_vals.append((k+1) * (alpha**2) * (r**k))
    # also evaluate at k=n-1 just in case
    conv_vals.append(n * (alpha**2) * (r**(n-1)))
    max_conv = max(conv_vals)
    return max_conv

def compute_score_geom(eps):
    r = 1 - eps
    if r <= 0: return large.
    # compute log_term = n * np.log(r)
    log_r_n = n * np.log(r)

    denom = 1 - r_n
    if denom <= 0: denom small, maybe treat as large.
    alpha = eps / denom

    log_r = np.log(r)  # negative
    k_opt = int(np.floor(-1.0 / log_r - 1.0))

    k_opt = max(0, min(k_opt, n-1))
    # compute conv candidates:
    # compute r**k efficiently using exp(log_r * k)
    def conv_at(k):
        return (k+1) * (alpha**2) * np.exp(log_r * k)

    candidates.append(n * (alpha**2) * np.exp(log_r * (n-1)))

def golden_section_search(a, b, tol=1e-12, max_iter=200):
    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2  # 1/phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = compute_score_geom(c)
    fd = compute_score_geom(d)
    for i in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = compute_score_geom(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = compute_score_geom(d)
    eps_opt = (a + b) / 2
    return eps_opt

    # conv already computed.
    max_val = conv.max()
    exponent = np.exp(gamma * (conv - max_val))

"""
Enhanced search for a step‑function that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

Key ideas
---------
* Warm‑start with a mix of uniform, Dirichlet draws and a simple
  geometrically‑shaped family (U‑shaped “bell”).
* “Soft‑max” gradient descent – minimise a smooth approximation of the
  max‑convolution using an exponentiated‑gradient (mirror‑descent) update.
  The gradient of the smooth max is obtained analytically via a single
  FFT‑based convolution.
* Aggressive targeted hill‑climb that transfers mass from the index that
  contributes most to the current peak to the index that contributes the
  least.
* A short final refinement (mix with uniform + tiny random pair swaps).

All stages respect the global 1000 s budget; progress is printed.
The returned list is a plain Python list of non‑negative floats whose sum
is (approximately) one – exactly the format expected by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score (sum(a) is assumed to be 1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 0️⃣ Warm‑start helpers
# ----------------------------------------------------------------------
def _geometric_u_shape(n: int, eps: float) -> np.ndarray:
    """
    Symmetric “U” shape: a_i ∝ eps**|i-(n-1)/2|.
    eps ∈ (0,1) makes the ends larger than the centre.
    The vector is normalised to sum‑1.
    """
    centre = (n - 1) / 2.0
    dist = np.abs(np.arange(n) - centre)
    a = eps ** dist
    a_sum = a.sum()
    if a_sum > 0:
        a = a / a_sum
    return a


def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                L_fft: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """Dirichlet draws + a few deterministic geometric candidates."""
    rng = np.random.default_rng()
    # a few deterministic geometric “U” candidates
    for log_eps in np.linspace(np.log(0.9), np.log(0.9999), 6):
        eps = np.exp(log_eps)
        a = _geometric_u_shape(n, eps)
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] GeomU eps={eps:.6f} → score={sc:.6f}")

    # many random Dirichlet draws
    while time.time() < deadline:
        # concentration drawn log‑uniformly
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 1️⃣ Soft‑max (exponentiated) gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def _exp_grad_smoothing(a0: np.ndarray,
                        n: int,
                        L_fft: int,
                        L_corr: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth approximation of the max using an exponentiated‑gradient
    (mirror descent) update.  The smoothness is controlled by a temperature
    parameter γ that is increased linearly during the run.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    # schedule parameters
    gamma_start = 5.0
    gamma_end = 250.0          # fairly large – approaches the true max
    eta0 = 0.7                 # initial step size for mirror descent
    it = 0

    while time.time() < deadline:
        it += 1
        # ----- compute current convolution -----
        conv = _conv_self_fft(a, L_fft)                  # length 2n‑1
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGrad iter={it:,} score={cur_score:.6f}")

        # ----- soft‑max weights -----
        max_val = conv.max()
        # temperature grows linearly with elapsed proportion
        t = (time.time() - start_time) / (deadline - start_time)
        gamma = gamma_start + t * (gamma_end - gamma_start)
        # stable exponentiation
        exp_term = np.exp(gamma * (conv - max_val))
        w = exp_term / exp_term.sum()                     # size 2n‑1, non‑negative

        # ----- gradient of the smooth max -----
        # gradient_i = 2 * Σ_k w_k * a_{k-i}
        # computed as a correlation of a with w (via FFT)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_corr)
        Wf = np.fft.rfft(w_rev, n=L_corr)
        corr = np.fft.irfft(Af * Wf, n=L_corr)           # length L_corr
        # keep only the linear part (length 3n‑2)
        full_len = 3 * n - 2
        corr = corr[:full_len]
        idx = 2 * n - 2 - np.arange(n)                   # reverse index trick
        grad = 2.0 * corr[idx]

        # ----- mirror‑descent (exponentiated) update -----
        eta = eta0 / np.sqrt(it)                        # diminishing step
        a = a * np.exp(-eta * grad)                     # stay non‑negative
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a / a_sum
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

# 2️⃣ Targeted hill‑climb (more focused than the previous version)
# ----------------------------------------------------------------------
def _targeted_hill_climb(a: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move mass from the worst‑contributing index to the best‑improving one."""
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # contribution weights for this peak: w_i = a_{k_max-i}
        w = np.zeros(n, dtype=float)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            w[i_range] = best_a[k_max - i_range]

        src = int(np.argmax(w))
        dst = int(np.argmin(w))
        if src == dst:
            # fall back to a random move if the above fails
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue

        if best_a[src] < 1e-12:
            continue

        # move a modest fraction of the source weight (up to 10 %)
        delta = rng.uniform(0.0, min(best_a[src] * 0.10, best_a[src]))
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        new_score = _objective_score(cand, n, L_fft)
        if new_score < best_score:
            best_score, best_a = new_score, cand
            print(f"[{time.time() - start_time:.2f}s] Hill src={src}→{dst} δ={delta:.6e} → score={new_score:.6f}")

    return best_a, best_score

# 3️⃣ Final light mixing + random‑pair hill‑climb
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i] * 0.05, best_a[i]))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill‑climb i={i} j={j} δ={delta:.6e} → score={sc:.6f}")

    return best_a, best_score

            best_a, n, L_fft, L_corr, grad_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After exponentiated‑grad best≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Targeted hill‑climb on the maximal convolution entry
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + 0.20 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Targeted hill‑climb")
        best_a, best_score = _targeted_hill_climb(best_a, n, L_fft, hill_deadline)

    # ------------------------------------------------------------------
    # 4️⃣ Final mixing + light random‑pair hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement")
        best_a, best_score = _final_refinement(best_a, n, L_fft, deadline)

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
