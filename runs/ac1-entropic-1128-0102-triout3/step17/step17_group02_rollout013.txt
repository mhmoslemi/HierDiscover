    cur_score = score

    # we don't know iterations; define T0 = cur_score * 0.1 maybe.

    # We'll compute fraction elapsed = (time.time() - start) / total
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = (time.time() - start_time) / (deadline - start_time)
        # linearly decrease T
        T = T0 * (1 - frac) + T_end * frac

        src = rng.integers(n)
        dst = rng.integers(n-1)  # ensure may be same? handle.
        if src == dst:
            continue
        if cur_a[src] <= 1e-12:
            continue
        # choose delta fraction of src mass, say r ~ uniform(0, max_frac)
        max_frac = 0.3 if cur_score > 1.5 else 0.1
        delta = cur_a[src] * rng.random() * max_frac
        if delta <= 0:
            continue

        # backup small part of a maybe only src and dst values
        a_src = cur_a[src]
        a_dst = cur_a[dst]

        # compute linear update using cur_a copy (original values)
        # We'll compute conv update as described
        # Add to conv

        # Create temporary references
        conv_new = cur_conv.copy()
        # update conv_new slice
        conv_new[dst:dst+n] += 2*delta*cur_a
        conv_new[src:src+n] -= 2*delta*cur_a
        conv_new[2*src] += delta**2
        conv_new[2*dst] += delta**2
        conv_new[src+dst] -= 2*delta**2

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # compute acceptance probability
            if T > 0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            # commit move
            cur_a[src] = a_src - delta
            cur_a[dst] = a_dst + delta
            cur_conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()

                print(f"[{time.time() - start_time:.2f}s] SA new best {best_score:.6f}")

"""
Propose a non‑negative step function that minimizes
    2·n·max(conv(a,a)) / (sum(a)^2)

The implementation works in three stages:

1️⃣  **Initial pool** – a quick sweep of deterministic windows
    (sine, Hann, triangular, inverse‑triangular, β‑shaped) plus a
    modest number of random Dirichlet draws.  The best among them
    becomes the seed.

2️⃣  **Smooth projected‑gradient descent** – the non‑smooth max is
    replaced by a log‑sum‑exp (temperature β).  β is increased on a
    schedule so the optimisation first follows a gentle landscape
    and then focuses on the true maximum.  Several restarts with
    random jitter are performed until the time budget is exhausted.

3️⃣  **Simulated‑annealing fine‑tuning** – mass is transferred
    between two positions (source → destination).  The resulting
    change in the convolution is updated incrementally (O(n) per
    move).  Moves that improve the objective are always accepted;
    worsening moves are accepted with a Metropolis probability that
    cools down linearly.  The best vector found during this phase
    (and a final short greedy hill‑climb) is returned.

All vectors are kept on the probability simplex (non‑negative,
sum = 1) so the denominator of the original formula is fixed.
The final list is normalised once more before returning.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global timing (used by helper functions)
# ----------------------------------------------------------------------
_start_time = None


def _now() -> float:
    """seconds elapsed since start of optimisation."""
    return time.time() - _start_time


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = np.max(conv)
    exps = np.exp(beta * (conv - m))
    return m + np.log(np.sum(exps)) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Uses the identity  grad = 2·(a_rev * w)[center:centre+n],
    where w are the soft‑max weights over the convolution entries.
    """
    m = np.max(conv)
    exps = np.exp(beta * (conv - m))
    w = exps / np.sum(exps)               # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)   # length 3·n‑2
    n = a.shape[0]
    return 2.0 * rev_corr[n - 1 : n - 1 + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Random Dirichlet sampling (cheap, used for initial pool)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        conc_log = rng.uniform(-3.0, 0.7)   # concentration in [1e‑3,5]
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{_now():.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth projected‑gradient descent (β‑schedule)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    score0: float,
                    beta_vals,
                    deadline: float):
    """Projected gradient descent on the smooth max‑approximation."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    for beta in beta_vals:
        step = 0.5
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            improved = False
            for factor in (1.0, 0.5, 0.25, 0.125):
                cand = a - step * factor * grad
                cand = _project_simplex(np.maximum(cand, 0.0))
                cand_score = _objective_score(cand)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    improved = True
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()
                        print(f"[{_now():.2f}s] β={beta:.1f} new best {best_score:.6f}")
                    break
            if not improved:
                step = max(step * 0.5, 1e-12)
            else:
                step = min(step * 1.05, 1.0)

            if step < 1e-12:
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (incremental convolution updates)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         score0: float,
                         deadline: float):
    """
    Mass‑transfer SA where each move updates the self‑convolution
    incrementally (O(n)).  Temperature is lowered linearly.
    """
    n = a0.shape[0]
    rng = np.random.default_rng()
    cur_a = a0.copy()
    cur_conv = _conv_self(cur_a)
    cur_score = score0

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule (linear from T0 to T_end)
    T0 = cur_score * 0.5          # generous start
    T_end = cur_score * 1e-5

    # maximum fraction of source mass that can be moved in a single step
    max_frac_initial = 0.4
    max_frac_final = 0.02

    while time.time() < deadline:
        elapsed = _now()
        # linear cooling
        frac_time = elapsed / (deadline - _start_time)
        T = T0 * (1.0 - frac_time) + T_end * frac_time
        max_frac = max_frac_initial * (1.0 - frac_time) + max_frac_final * frac_time

        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or cur_a[src] <= 1e-12:
            continue

        # propose a transfer
        delta = cur_a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # keep a copy of the two affected entries (cheap)
        a_src = cur_a[src]
        a_dst = cur_a[dst]

        # Build a candidate convolution (copy + slice updates)
        conv_cand = cur_conv.copy()
        conv_cand[dst:dst + n] += 2.0 * delta * cur_a
        conv_cand[src:src + n] -= 2.0 * delta * cur_a
        conv_cand[2 * src] += delta ** 2
        conv_cand[2 * dst] += delta ** 2
        conv_cand[src + dst] -= 2.0 * delta ** 2

        new_max = conv_cand.max()
        new_score = 2.0 * n * new_max

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis step
            if T > 0.0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            # commit the move
            cur_a[src] = a_src - delta
            cur_a[dst] = a_dst + delta
            cur_conv = conv_cand
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{_now():.2f}s] SA new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (very small transfers)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                score0: float,
                deadline: float):
    """Deterministic greedy fine‑tuning (tiny moves)."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * 0.02   # ≤ 2 % of source mass
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{_now():.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            # occasional downhill move (tiny temperature ≈0)
            a = cand
            cur_score = cand_score
    return best_a, best_score
