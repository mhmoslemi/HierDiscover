Pseudo:

def proj_simplex(v):
    # ensure sum(v)=1 and v>=0
    # using algorithm by Wang & Carreira-Perpinán 2013.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def beta_shape(n, p, q):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

def gradient_descent(a, n, L_fft, time_deadline, initial_lr=0.01, min_lr=1e-6):
    best = a.copy()
    best_score, _ = _score_of(best,n,L_fft)
    lr = initial_lr
    while time.time() < time_deadline and lr >= min_lr:
        # compute conv and max conv index
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        # gradient
        g = np.zeros_like(a)
        # compute slice where k-i in [0,n-1]
        # i from max(0,k-(n-1)) to min(n-1, k)
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)  # inclusive

        idx_i = np.arange(start_i, end_i + 1)

            best = cand.copy()
            a = cand  # continue from improved
            # maybe increase lr?
            # if improvement, keep same lr
        else:
            # no improvement; maybe reduce lr
            lr *= 0.5
    return best, best_score

def hill_climb(a, n, L_fft, time_deadline):
    best = a.copy()
    best_score,_ = _score_of(best,n,L_fft)
    rng = np.random.default_rng()
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] <= 0:
            continue
        # propose moving delta up to some fraction of a[i]
        delta = rng.uniform(0.0, min(best[i], 0.02))  # maybe bigger
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)  # maintain sum=1
        sc,_ = _score_of(cand,n, L_fft)
        if sc < best_score - 1e-12:
            best = cand
            best_score = sc
    return best, best_score

"""
Propose a step‑function (non‑negative vector on the probability simplex) that
minimises  score = 2·n·max(conv(a,a))  (∑a = 1).

The search proceeds in three phases:

1. **Exploration on a medium sized grid (n=4096).**
   – Uniform baseline.
   – Optional globally supplied starter.
   – Random Dirichlet draws.
   – Systematic scan of the full Beta(p,q) family (including mixtures with
     the uniform vector) to locate a good shape.

2. **Local optimisation on the same grid.**
   – A projected sub‑gradient descent that reduces the maximal convolution
     entry (the gradient of max(conv) is known analytically).
   – Simple hill‑climbing moves (transfer a tiny amount of mass between two
     coordinates) to exploit local improvements.

3. **Upscaling and refinement (n=8192).**
   If time permits, the best medium‑size candidate is interpolated to a
   larger grid, and the gradient + hill‑climb steps are repeated.  A larger
   n reduces the discretisation error of the continuous arcsine law, giving
   a strictly lower score.

All heavy‑weight evaluations use an FFT based convolution, while the final
candidate is returned as a plain Python list (the evaluator will compute the
exact score with `np.convolve`).  The routine obeys the supplied time budget
and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> tuple[float, np.ndarray]:
    """
    Exact score used by the evaluator for a vector that already satisfies
    `np.sum(a) == 1`.  Returns (score, conv) where score = 2·n·max(conv).
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    score = float(2.0 * n * max_b)
    return score, conv


def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """
    Discretised Beta(p,q) density on the grid x_i = (i+0.5)/n.
    Returns a non‑negative vector normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w | w_i ≥ 0, Σ w_i = 1}.  Implements the classic O(n) algorithm.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_step(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of `max(conv(a,a))` at the index `k` where the max is attained.
    The gradient of conv[k] w.r.t. a_i equals 2·a_{k-i} whenever the index
    k-i lies in the valid range, otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)          # inclusive
    if i_start <= i_end:
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        g[idx_i] = 2.0 * a[idx_j]
    return g


def gradient_descent(a0: np.ndarray,
                     n: int,
                     L_fft: int,
                     deadline: float,
                     init_lr: float = 0.01,
                     min_lr: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent aimed at reducing `max(conv(a,a))`.
    Stops when the deadline is reached or the learning rate falls below
    `min_lr`.  Returns the best vector encountered together with its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score, _ = _score_of(best_a, n, L_fft)
    lr = init_lr

    while time.time() < deadline and lr >= min_lr:
        # compute convolution and the location of the current max
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))

        # sub‑gradient at that location
        g = _gradient_step(a, k)

        # single gradient step followed by projection onto the simplex
        cand = _project_simplex(a - lr * g)

        cand_score, _ = _score_of(cand, n, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand                         # continue from the improvement
            # keep the learning rate unchanged (the step was successful)
        else:
            # no improvement – reduce the step size
            lr *= 0.5

    return best_a, best_score


def hill_climb(a0: np.ndarray,
               n: int,
               L_fft: int,
               deadline: float,
               max_delta: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly move a tiny amount of mass from one
    coordinate to another, keeping the vector on the simplex.  Only strictly
    improving moves are kept.
    """
    a = a0.copy()
    best_score, _ = _score_of(a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 0.0:
            continue
        delta = rng.uniform(0.0, min(a[i], max_delta))
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)          # renormalise safely
        cand_score, _ = _score_of(cand, n, L_fft)
        if cand_score + 1e-12 < best_score:    # accept only genuine improvements
            a = cand
            best_score = cand_score
    return a, best_score


def upsample(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """
    Linear interpolation of a distribution from a small grid to a bigger one.
    The resulting vector is re‑normalised to sum 1.
    """
    n_small = a_small.shape[0]
    x_small = (np.arange(n_small, dtype=np.float64) + 0.5) / n_small
    x_big = (np.arange(n_big, dtype=np.float64) + 0.5) / n_big
    a_big = np.interp(x_big, x_small, a_small, left=0.0, right=0.0)
    a_big = np.maximum(a_big, 0.0)
    s = a_big.sum()
    if s > 0.0:
        a_big /= s
    else:
        a_big = np.full(n_big, 1.0 / n_big)
    return a_big
