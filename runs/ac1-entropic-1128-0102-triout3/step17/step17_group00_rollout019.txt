"""
Propose a step‐function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣  A dense parametric sweep over the 2‑parameter family
        a_i ∝ (i+1)^p · r^i      (geometric × power law)
    using a fine grid for the decay λ = –log r and the exponent p.
    The best vector from the sweep is kept.

2️⃣  Smoothed gradient descent.
    For a given β we replace   max(conv)   by a smooth
    approximation  (1/β)·log ∑exp(β·conv_k) .
    Its gradient can be written in closed form and computed by a
    single FFT‑based correlation.  We perform projected gradient steps
    (Euclidean projection onto the simplex) while slowly increasing β.
    The best vector found is stored.

3️⃣  Simulated‑annealing hill‑climbing.
    Small mass‑transfer moves (source → destination) are proposed;
    improvements are always accepted, uphill moves are accepted with a
    temperature that decays linearly to zero.
    This phase harvests any remaining tiny improvements.

All phases respect a 1000‑second wall‑time budget.  The final vector is
normalised to unit L1‑norm (the evaluation function is homogeneous) and
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (fast, works for unit L1‑norm)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Build a geometric‑power family vector and normalise it
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^p · exp(-lam·i)   (lam = -log r).
    Returned vector is normalised to unit L1‑norm.
    """
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)               # stabilise exponentiation
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smoothed gradient descent (β‑continuation)
# ----------------------------------------------------------------------


def _smooth_grad_descent(
    a0: np.ndarray,
    L_conv: int,
    L_corr: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed objective
        f_β(a) = (2·n/β)·log Σ_k exp(β·conv_k(a)),
    where conv = a*a.
    β is increased during the run (continuation).
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    beta = 1.0                     # start with a tiny smoothing
    while time.time() < deadline:
        it += 1
        # ----- occasional increase of the smoothing parameter -----
        if it % 2000 == 0:
            beta = min(beta * 1.6, 800.0)   # cap to avoid overflow

        # ----- compute conv and the soft‑max weights -----
        conv = _conv_self_fft(a, L_conv)
        max_c = conv.max()
        # stable soft‑max
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        w /= w_sum

        # ----- gradient = 4·n·correlation(w, a) -----
        a_rev = a[::-1]
        W_fft = np.fft.rfft(w, n=L_corr)
        A_rev_fft = np.fft.rfft(a_rev, n=L_corr)
        corr_full = np.fft.irfft(W_fft * A_rev_fft, n=L_corr)
        # valid part has length n, starts at offset n‑1
        grad = 4.0 * n * corr_full[n - 1 : 2 * n - 1]

        # ----- simple back‑tracking line‑search -----
        step0 = 0.3
        step = step0 / (it ** 0.55)          # shrinking step size
        improved = False
        for _ in range(8):
            cand = a - step * grad
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_conv)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(
                    f"[{time.time() - start_time:.2f}s] smooth‑grad it={it:,} "
                    f"β={beta:.1f} step={step:.3e} score={cand_score:.6f}"
                )
                break
            step *= 0.5
        if not improved:
            # occasional tiny random jitter to escape flat plateaus
            if it % 500 == 0:
                eps = 3e-5 * rng.random(n) * a
                a = np.maximum(a + eps, 0.0)
                a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a: np.ndarray,
    L_conv: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Simple SA that moves a small fraction of mass from one index to another."""
    n = a.shape[0]
    # initial temperature – a tiny fraction of the current objective
    T0 = 1e-3
    while time.time() < deadline:
        # linear cooling
        remaining = deadline - time.time()
        T = T0 * (remaining / (deadline - start_time))

        src = rng.integers(n)
        if a[src] <= 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # amount transferred – up to a few percent of the source mass,
        # scaled by the temperature so big moves happen early.
        max_frac = 0.05 * (T / T0 + 0.1)
        eps = a[src] * rng.random() * max_frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        score_prop = _eval_fast(a_prop, L_conv)
        delta = score_prop - best_score
        if delta < 0.0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            a = a_prop
            if score_prop < best_score:
                best_score = score_prop
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] SA move score={score_prop:.6f}"
                )
    return best_seq, best_score

    lam_grid = np.linspace(4e-05, 1.5e-03, 500)           # very fine λ‑grid
