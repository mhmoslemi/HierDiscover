```
def compute_gradient(a, conv, alpha):
    # conv = a*a (full) already
    # w = softmax
    max_val = conv.max()
    exp_scaled = np.exp(alpha * (conv - max_val))
    w = exp_scaled / exp_scaled.sum()
    # compute gradient
    a_rev = a[::-1]
    # conv via FFT
    L = _next_pow2(len(a_rev) + len(w) - 1)
    Ar = np.fft.rfft(a_rev, n=L)
    Wr = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(Ar * Wr, n=L)[: len(a_rev) + len(w) - 1]
    grad = 2 * corr[n-1 : n-1 + len(a)]
    return grad
```

```
def _smooth_descent(start_vec, deadline, rng, label=''):
    a = start_vec.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    alpha = 1.0
    max_alpha = 200.0
    # Compute next_pow2 for conv lengths
    L_conv = _next_pow2(2 * n - 1)
    while time.time() < deadline:
        t += 1
        # compute convolution via FFT
        Af = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(Af * Af, n=L_conv)[: 2 * n - 1]
        # loss and weights
        maxc = conv.max()
        # softmax weights
        exp_shift = np.exp(alpha * (conv - maxc))
        w = exp_shift / exp_shift.sum()
        # gradient via correlation
        a_rev = a[::-1]
        L_corr = _next_pow2(len(a_rev) + len(w) - 1)
        Ar = np.fft.rfft(a_rev, n=L_corr)
        Wr = np.fft.rfft(w, n=L_corr)
        corr = np.fft.irfft(Ar * Wr, n=L_corr)[: len(a_rev) + len(w) - 1]
        grad = 2 * corr[n-1 : n-1 + n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.5 / np.sqrt(t)  # decaying LR
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a = _project_simplex(np.maximum(a, 0.0))
        # evaluate objective
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] {label} smoothing new best {best_score:.6f}")
        # increase alpha gradually
        if alpha < max_alpha:
            alpha = min(alpha * 1.02, max_alpha)
    return best_a, best_score
```

"""
Continuation‑gradient descent for the step‑function optimisation.

The objective (with the sum of the vector fixed to 1) is
    F(a) = 2·n·max(conv(a,a))

where conv(a,a) is the full linear self‑convolution.
A smooth proxy for the max is the log‑sum‑exp (soft‑max):
    Lα(a) = (1/α)·log Σk exp(α·convk)

For increasing values of α the proxy approaches the true max.
We minimise Lα with projected Adam (simplex projection after each
update) while slowly raising α (continuation).  The gradient of Lα is

    ∂Lα/∂ai = 2·Σj aj·w(i+j) ,

where w is the normalised soft‑max weight vector
    wk = exp(α·convk) / Σℓ exp(α·convℓ) .

All heavy operations (convolutions and correlations) are carried out
with FFTs, so each iteration costs O(n log n).  The routine builds a
few diverse starting vectors (uniform, sparse, Dirichlet draws and the
optional global starter), runs the continuation optimiser from the
most promising ones and finishes with a short projected‑gradient
descent (as in the reference solution).  The final vector is projected
onto the probability simplex and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing (filled by propose_candidate)
# ----------------------------------------------------------------------
start_time = None  # will be overwritten at entry


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (the denominator is 1 because we keep the vector on the simplex)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at ``a`` (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Sparse starter generator
# ----------------------------------------------------------------------
def _sparse_initial(n: int, sparsity_ratio: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random sparse vector with roughly ``sparsity_ratio`` non‑zero entries.
    The non‑zero values are drawn from an exponential distribution and then
    normalised to sum to one.
    """
    k = max(2, int(sparsity_ratio * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projected Gradient Descent (as in the reference implementation)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray,
                                cur_score: float,
                                deadline: float) -> (np.ndarray, float):
    """Fine‑tune ``a0`` by moving opposite to the sub‑gradient of the active
    convolution entries and projecting back onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    step = 0.5

    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Continuation / Smooth gradient descent (new component)
# ----------------------------------------------------------------------
def _smooth_descent(start_vec: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator,
                    label: str = '') -> (np.ndarray, float):
    """
    Minimise a smooth approximation of the max‑convolution using Adam
    and a continuation on the soft‑max temperature α.
    """
    a = start_vec.copy()
    n = a.shape[0]

    # Pre‑compute FFT lengths (they stay constant)
    L_conv = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n - 2)          # for correlation a_rev * w

    best_a = a.copy()
    best_score = _objective_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # continuation schedule
    alpha = 1.0            # start with a very smooth proxy
    max_alpha = 200.0      # final temperature (sharp approximation)

    while time.time() < deadline:
        t += 1

        # ----- self‑convolution (exact) ---------------------------------
        Af = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(Af * Af, n=L_conv)[: 2 * n - 1]

        # ----- soft‑max weights -----------------------------------------
        maxc = conv.max()
        exp_shift = np.exp(alpha * (conv - maxc))
        w = exp_shift / exp_shift.sum()

        # ----- gradient of the smooth loss -------------------------------
        a_rev = a[::-1]
        # FFTs for the correlation a_rev * w
        Ar = np.fft.rfft(a_rev, n=L_corr)
        Wr = np.fft.rfft(w, n=L_corr)
        corr = np.fft.irfft(Ar * Wr, n=L_corr)[: len(a_rev) + len(w) - 1]
        # extract the part corresponding to indices 0 … n‑1
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # a learning‑rate that slowly decays
        lr = 0.5 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- simplex projection -----------------------------------------
        a = _project_simplex(np.maximum(a, 0.0))

        # ----- evaluate real objective ------------------------------------
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] {label} smooth new best {best_score:.6f}")

        # ----- increase the temperature α slowly ---------------------------
        if alpha < max_alpha:
            alpha = min(alpha * 1.02, max_alpha)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (probability vector) that minimises
    ``evaluate_sequence``.  The routine builds several diverse initial vectors,
    runs a continuation‑gradient descent from the most promising ones and finally
    refines the result with a short projected‑gradient descent.
    """
    global start_time
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we double the length used in the reference solution.
    # Larger n gives the optimiser more freedom to flatten the
    # self‑convolution.
    # ------------------------------------------------------------------
    n = 8192

    # ------------------------------------------------------------------
    # Assemble a pool of starting vectors (uniform, the optional global,
    # Dirichlet draws, sparse vectors).
    # ------------------------------------------------------------------
    starters = []

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=float)
    starters.append(('uniform', uniform))

    # optional global starter (padded / trimmed to the current length)
    try:
        global height_sequence_1  # may be provided by the host
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            arr = np.zeros(n, dtype=float)
            length = min(len(starter), n)
            arr[:length] = starter[:length]
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 0.0:
                arr /= s
                starters.append(('global', arr))
                print(f"[{time.time() - start_time:.2f}s] Loaded global starter")
    except Exception:
        pass

    # a handful of Dirichlet draws with a wide concentration range
    for i in range(12):
        conc_log = rng.uniform(-3.0, 0.7)        # log‑uniform in [10⁻³ , 10⁰·⁷]
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        starters.append((f'dirichlet_{i}', a))

    # a few sparse starts (≈2 % non‑zeros)
    for i in range(4):
        a = _sparse_initial(n, sparsity_ratio=0.02, rng=rng)
        starters.append((f'sparse_{i}', a))

    # ------------------------------------------------------------------
    # Keep the best among the raw starts (cheap objective evaluation).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    for name, vec in starters:
        sc = _objective_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.copy()
        print(f"[{time.time() - start_time:.2f}s] init {name:<12} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Continuation‑gradient descent from a couple of promising seeds.
    # ------------------------------------------------------------------
    # always try the current best
    candidates = [(best_seq, 'best_initial')]

    # add a few fresh Dirichlet starts for extra diversity (if time permits)
    for i in range(3):
        a = rng.dirichlet(np.ones(n))
        candidates.append((a, f'random_{i}'))

    for vec, label in candidates:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.2f}s] smooth descent from {label}")
        seq, sc = _smooth_descent(vec, deadline, rng, label)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] overall new best {best_score:.6f} from {label}")

    # ------------------------------------------------------------------
    # Final short projected‑gradient descent (helps polishing the last
    # sub‑gradient spikes).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] final PGD refinement")
        best_seq, best_score = _projected_gradient_descent(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise to guarantee strict feasibility for the evaluator.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
