"""
Propose a step‑function (non‑negative list) that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------
1. Choose a moderate length n (512) – large enough to get a good bound,
   small enough that each iteration is cheap.
2. Initialise the sequence from the supplied global `height_sequence_1`
   (down‑sampled) or from a uniform start.
3. **Phase 1 – cheap parametric & random families**  
   – geometric, exponential, 2 / 3‑block families, random Dirichlet draws.  
   The best candidate found becomes the seed for the next phase.
4. **Phase 2 – smooth‑max gradient descent**  
   The non‑smooth max is replaced by a log‑sum‑exp surrogate.
   We minimise this surrogate with a projected gradient step
   (gradient descent + Euclidean projection onto the probability simplex).
   The smoothing parameter β is increased linearly so that the surrogate
   becomes tighter while the optimisation proceeds.
5. **Phase 3 – convex LP refinement**  
   For the current f we solve the linear programme  
        maximise   1ᵀb
        subject to  f * b ≤ M   (element‑wise)
                    b ≥ 0
   where M = max(conv(f,f)).  If the optimum b has total mass >1 we can
   form a new distribution g = b / sum(b) and mix f←(1‑τ)f+τg
   (τ≈0.5).  The mixing strictly reduces the maximum self‑convolution.
   A few LP‑iterations are performed until no further improvement is
   possible or time runs out.
6. **Phase 4 – random hill‑climbing**  
   Small random mass transfers between two positions are tried;
   improving moves are kept (a simple simulated‑annealing style loop).

All steps respect the non‑negativity and normalisation constraints,
and the best sequence seen (according to the true objective
`evaluate_sequence`) is returned.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) for a sequence with sum=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient) where gradient is ∂/∂a of the surrogate.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # shift for stability
    Z = exp_scaled.sum()
    w = exp_scaled / Z                         # weights, sum to 1

    # gradient:  d/d a_i ( Σ_k w_k * conv_k ) = 2 * Σ_j a_j * w_{i+j}
    # implement via convolution of a with reversed w
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)          # length 3n‑2
    start = len(w) - 1                      # = 2n‑2
    G = 2.0 * conv_rev[start - np.arange(n)]

    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, G

def _exponential_family(n, best_seq, best_score, start, deadline):

            print(f"[{time.time() - start:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _two_block_family(n, best_seq, best_score, start, deadline):
    idx = np.arange(n, dtype=np.float64)
    for L_frac in np.linspace(0.02, 0.3, 20):

                print(f"[{time.time() - start:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _random_dirichlet_trials(n, best_seq, best_score, start, deadline, trials=2000):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:

        logc = rng.uniform(-2.0, 0.7)    # concentration in roughly [0.01,5]
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)

            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score

def _smooth_gradient_descent(a0, start, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    # schedule for β (smoothness) and learning rate η
    beta_start, beta_end = 0.5, 30.0
    eta = 0.1
    min_eta = 1e-6
    max_eta = 1.0

    while time.time() < deadline:
        # linear schedule for β
        elapsed = time.time() - start
        frac = (elapsed) / (deadline - start)
        beta = beta_start + frac * (beta_end - beta_start)

        smooth, grad = _smooth_max_and_grad(a, beta)

        # gradient step with projection onto simplex
        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        new_score = _objective_score(a_new)

        if new_score < cur_score - 1e-12:
            # improvement
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start:.2f}s] Grad β={beta:.2f} η={eta:.2e} score={new_score:.6f}")
            # gently increase step size
            eta = min(max_eta, eta * 1.02)
        else:
            # no improvement – shrink step and add a tiny jitter
            eta = max(min_eta, eta * 0.7)
            if np.random.rand() < 0.02:
                jitter = np.random.normal(scale=1e-3, size=n)
                a = a + jitter
                a = np.clip(a, 0, None)
                a = _project_simplex(a)
                cur_score = _objective_score(a)

        # early exit if we already passed a good bound
        if best_score <= 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 3: linear‑program refinement
# ----------------------------------------------------------------------
def _lp_refinement(a0, start, deadline, best_seq, best_score, max_iters=15):
    """
    For the current distribution f, solve
        maximise   1ᵀb
        subject to conv(f,b) ≤ M
                    b ≥ 0
    and mix the result back into f.  Perform a few iterations.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    # build index arrays that will be reused (row positions for each column)
    # row indices for column j : j ... j+n‑1
    col_idx = np.repeat(np.arange(n), n)
    row_idx = np.concatenate([np.arange(j, j + n) for j in range(n)])

    while time.time() < deadline and max_iters > 0:
        max_iters -= 1

        # current bound
        conv_f = _conv_self(a)
        M = conv_f.max()

        # data for the constraint matrix: each column repeats the current a
        data = np.tile(a, n)
        A_ub = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        # linear program: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A_ub, b_ub=ub, bounds=bounds, method='highs')
        if res.status != 0:
            # LP failed – give up
            break

        b = np.maximum(res.x, 0.0)
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # cannot increase total mass under the same M → stop
            break

        # construct g with the same total mass as a (which is 1)
        g = b / sum_b

        # mix a and g
        tau = 0.5
        a_new = (1.0 - tau) * a + tau * g
        a_new = _project_simplex(a_new)  # safety projection
        new_score = _objective_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start:.2f}s] LP mix new score={new_score:.6f}")
        else:
            # no improvement → stop early
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 4: simple random hill‑climbing
# ----------------------------------------------------------------------
def _hill_climb(a0, start, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    while time.time() < deadline:
        # propose a tiny mass transfer between two random indices
        i, j = np.random.choice(n, size=2, replace=False)
        eps = np.random.uniform(0, 0.02) * min(a[i], 0.5)  # up‑to 2 % of a[i]
        if eps == 0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        a_candidate = _project_simplex(a_candidate)
        cand_score = _objective_score(a_candidate)
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, a.copy()
                print(f"[{time.time() - start:.2f}s] Hill‑climb score={cand_score:.6f}")
        # occasional restart to avoid stagnation
        if np.random.rand() < 0.001:
            a = best_seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The routine combines cheap parametric sweeps, projected smooth‑max
    gradient descent, a few LP‑based refinement steps and a final
    random hill‑climb.  The best sequence found (normalised to sum = 1)
    is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # 0️⃣  Choose problem size
    # ------------------------------------------------------------------
    n = 512                       # moderate size – fast yet expressive

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from the global starter if possible
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            # down‑sample / pad to length n
            if starter.size >= n:
                factor = starter.size // n
                starter = starter[:factor * n].reshape(n, factor).mean(axis=1)
            else:
                starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
            else:
                starter = np.full(n, 1.0 / n)
        else:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)   # uniform fallback

    best_seq = starter.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric families & random Dirichlet draws
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.18 * budget_s
    best_seq, best_score = _geometric_family(
        n, best_seq, best_score, start_time, phase1_deadline
    )
    best_seq, best_score = _exponential_family(
        n, best_seq, best_score, start_time, phase1_deadline
    )
    best_seq, best_score = _two_block_family(
        n, best_seq, best_score, start_time, phase1_deadline
    )
    best_seq, best_score = _random_dirichlet_trials(
        n, best_seq, best_score, start_time, phase1_deadline, trials=1500
    )

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max gradient descent
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.5 * budget_s
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting gradient descent")
        best_seq, best_score = _smooth_gradient_descent(
            best_seq, start_time, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP‑based refinement
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.8 * budget_s
    if time.time() < phase3_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting LP refinement")
        try:
            import scipy.sparse as sp
        except Exception:
            sp = None
        if sp is not None:
            best_seq, best_score = _lp_refinement(
                best_seq, start_time, phase3_deadline, best_seq, best_score
            )
            print(f"[{time.time() - start_time:.2f}s] After LP best score={best_score:.6f}")
        else:
            print("Scipy sparse module not available – skipping LP phase.")

    # ------------------------------------------------------------------
    # Phase 4 – random hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise final answer and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    return [float(x) for x in best_seq]
