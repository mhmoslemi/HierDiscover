def subgradient_descent(a0, max_iters, deadline, best_seq, best_score):
    a = a0.copy()
    # optionally rescale to sum=1
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0/a.shape[0])  # uniform
    for t in range(max_iters):
        if time.time() >= deadline: break
        # compute conv and max index
        L = _next_pow2(2*a.shape[0] - 1)
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[:2*a.shape[0]-1]
        max_conv = conv.max()
        # compute objective value: 2 * n * max_conv
        obj = 2.0 * a.shape[0] * max_conv
        # update best
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(...)
        # find index of max (if tie pick any)
        k_max = int(np.argmax(conv))
        # compute subgradient g
        n = a.shape[0]
        g = np.zeros_like(a)
        # compute feasible range
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        # vector of indices
        idx = np.arange(i_start, i_end + 1)
        # g_i = 2 * a[k_max - i]
        g[idx] = 2.0 * a[k_max - idx]
        # option: multiply by n factor? We'll incorporate into step size.
        # step size schedule
        eta = 0.5 / np.sqrt(t+1)  # base coefficient
        a_new = a - eta * g
        # ensure non-negativity and project onto simplex
        a_new = _project_simplex(a_new)
        a = a_new
    return best_seq, best_score

def _sweep_product_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    # product term (i+1)*(n-i)
    prod = (idx + 1) * (n - idx)  # shape symmetric
    for p in np.linspace(-4, -0.2, 81):
        if time.time() >= deadline: break
        seq = np.power(prod, p)
        total = seq.sum()
        if total == 0:

        seq /= total
        sc = _objective_score(seq)

"""
A fast multi‑stage search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works on the probability simplex (∑a=1) because the
objective is scale invariant.  It consists of:

1️⃣  Light parametric sweeps (geometric, power, exponential, two‑block,
    and a product‑power family).
2️⃣  A large number of random Dirichlet draws.
3️⃣  Projected sub‑gradient descent on the true max‑objective with a
    simple back‑tracking line search.  Several random restarts are used
    and a tiny jitter is added when the procedure stalls.

All heavy work is done with FFT‑based convolutions (O(n log n)).
The routine respects the 1000 s time budget and returns the best
sequence found (normalised to sum = 1) as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a fast FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return the value of 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    (The denominator sum^2 equals 1 after normalisation.)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # implementation from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential decay from the centre (large at the edges)."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 120):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the ends with a tunable split."""
    for L_frac in (0.01, 0.02, 0.05, 0.1, 0.15, 0.2):
        L = max(1, int(L_frac * n))
        for w in np.linspace(0.0, 1.0, 11):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_product_power(n, best_seq, best_score, deadline):
    """
    Shape a_i ∝ ((i+1)*(n-i))^p with p negative.
    This family puts relatively more mass near the borders.
    """
    idx = np.arange(n, dtype=np.float64)
    prod = (idx + 1.0) * (n - idx)  # symmetric, maximal in the centre
    for p in np.linspace(-4.0, -0.2, 81):
        if time.time() >= deadline:
            break
        seq = np.power(prod, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] ProdPower p={p:.3f} score={sc:.6f}")
    return best_seq, best_score

        logc = rng.uniform(-2.0, 0.7)

            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score

def _subgradient_descent(a0, max_iters, deadline, best_seq, best_score):
    """
    Projected sub‑gradient descent for f(a)=2·n·max(conv(a,a)).
    Uses a simple back‑tracking line‑search and random choice among
    all maximising convolution indices.
    """
    rng = np.random.default_rng()
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / a.shape[0])

    n = a.shape[0]
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv

        if cur_obj < best_score - 1e-12:
            best_score, best_seq = cur_obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} obj={cur_obj:.6f}")

        # pick a random index among the (almost) maximisers
        max_idxs = np.where(conv >= max_conv - 1e-15)[0]
        k = int(rng.choice(max_idxs))

        # sub‑gradient of max(conv) at index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]

        # step size schedule – larger at the beginning, decays like 1/√t
        eta = 0.5 / np.sqrt(it)

        # back‑tracking line search
        step = 1.0
        improved = False
        while step > 1e-14:
            a_new = a - step * eta * g
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-13:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score - 1e-12:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} improved obj={obj_new:.6f}")
                break
            step *= 0.5

        if not improved:
            # tiny stochastic kick to escape a flat region
            eps = 1e-8
            a += rng.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

    return best_seq, best_score

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
