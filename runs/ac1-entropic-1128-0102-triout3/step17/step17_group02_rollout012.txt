def _hill_climb_fast(a0, score0, deadline):
    n = a0.shape[0]
    L = 2*n - 1
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_hc = time.time()
    max_frac_start = 0.2
    max_frac_end = 0.001
    start_temp = (best_score * 0.01) + 1e-9 # maybe scale to objective magnitude
    while time.time() < deadline:
        elapsed = time.time() - start_hc
        frac = elapsed / (deadline - start_hc)
        max_frac = max_frac_start * (1 - frac) + max_frac_end * frac
        temp = start_temp * (1 - frac) + 1e-9 * frac
        src = rng.integers(n)
        if a[src] == 0.0:
            continue
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1
        # delta fraction
        delta_frac = rng.random() * max_frac
        d = a[src] * delta_frac
        if d <= 0.0:
            continue
        # Compute new convolution via copy
        conv_new = conv.copy()
        # slice updates using original a (old)
        conv_new[src:src+n] -= 2.0 * d * a
        conv_new[dst:dst+n] += 2.0 * d * a
        conv_new[2*src] += d * d
        conv_new[2*dst] += d * d
        conv_new[src + dst] -= 2.0 * d * d
        # compute new score
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        # accept if better or with prob
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            # update
            a[src] -= d
            a[dst] += d
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] hill-climb new best {best_score:.6f}")
    return best_a, best_score

"""
Propose a non‑negative step function that minimises the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1️⃣  A quick global sweep – many Dirichlet draws and a set of hand‑crafted
    shapes – to obtain a good starting vector.

2️⃣  Smooth optimisation – the non‑smooth max is replaced by a log‑sum‑exp
    (temperature β).  The vector is parametrised by logits x (softmax(x) is the
    probability vector), and Adam is applied to the smooth objective.
    β is increased progressively; the best vector seen is kept.

3️⃣  Fast stochastic hill‑climbing – a mass‑transfer move between two
    positions is proposed; the linear‑time update of the self‑convolution is
    performed (no FFT) and the move is accepted if it improves the exact
    objective or with a temperature‑controlled uphill probability.
    This stage occupies the remaining time budget and typically yields the
    final improvement needed to beat the previous best bound.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (length = next power‑of‑2)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (v≥0, sum=1).
    O(n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Smooth approximation of max using log‑sum‑exp with temperature β."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max with respect to a.
    Uses the soft‑max weights over the convolution entries.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()                     # shape (2·n‑1,)
    # cross‑correlation a ⋆ w  (equivalently convolution of a_rev with w)
    rev_corr = _convolve_fft(a[::-1], w)      # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam‑based smooth optimisation (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth objective.
    Returns the best vector and its exact objective value.
    """
    global start_time
    n = a0.shape[0]
    eps = 1e-12
    # initialise logits (log‑probabilities)
    x = np.log(a0 + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.1
    t = 0

    best_a = a0.copy()
    best_score = _objective_score(a0)

    beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]

    while time.time() < deadline:
        for beta in beta_schedule:
            # a modest number of iterations per β; break when we run out of time
            for _ in range(500):
                if time.time() >= deadline:
                    break
                a = _softmax(x)
                conv = _conv_self(a)
                grad_a = _smooth_gradient(a, conv, beta)
                # chain rule for softmax parametrisation
                dot = np.dot(grad_a, a)
                grad_x = a * (grad_a - dot)

                t += 1
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x ** 2)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
                x -= step

                # exact objective for this a
                score = 2.0 * n * conv.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast hill‑climbing with O(n) convolution updates
# ----------------------------------------------------------------------
def _hill_climb_fast(a0: np.ndarray, score0: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer hill‑climb.
    Uses an O(n) update of the self‑convolution for each proposal.
    """
    global start_time
    n = a0.shape[0]
    L = 2 * n - 1
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_hc = time.time()

    # schedule for the transfer fraction and for the Metropolis temperature
    max_frac_start, max_frac_end = 0.20, 0.001
    temp_start = max(1e-2 * best_score, 1e-9)  # a tiny absolute scale

    while time.time() < deadline:
        elapsed = time.time() - start_hc
        frac = elapsed / (deadline - start_hc)  # 0→1
        max_frac = max_frac_start * (1.0 - frac) + max_frac_end * frac
        temp = temp_start * (1.0 - frac) + 1e-9 * frac

        src = rng.integers(n)
        if a[src] == 0.0:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # choose a small transfer amount
        delta_frac = rng.random() * max_frac
        d = a[src] * delta_frac
        if d <= 0.0:
            continue

        # --- compute new convolution by a rank‑2 update (O(n)) ---
        conv_new = conv.copy()
        # contributions from the linear part a*δ + δ*a
        conv_new[src : src + n] -= 2.0 * d * a
        conv_new[dst : dst + n] += 2.0 * d * a
        # self‑convolution of the transferred mass
        conv_new[2 * src] += d * d
        conv_new[2 * dst] += d * d
        conv_new[src + dst] -= 2.0 * d * d
        # numerical safety
        conv_new = np.maximum(conv_new, 0.0)

        new_score = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-18))
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= d
            a[dst] += d
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence (list of floats)
    found within the allotted time budget.
    """
    global start_time
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between expressiveness
    # and speed for the 1000 s budget.
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Use the optional global starter if it exists.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq = starter.copy()
                    best_score = sc
                    print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – random Dirichlet draws + a handful of deterministic shapes
    # --------------------------------------------------------------
    dir_deadline = min(deadline, start_time + 30.0)
    rng = np.random.default_rng(seed)
    trial = 0
    while time.time() < dir_deadline:
        conc_log = rng.uniform(-3.0, 0.7)          # log‑uniform concentration
        c = 10.0 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        total = a.sum()
        if total == 0.0:
            continue
        a /= total
        trial += 1
        sc = _objective_score(a)
        if sc < best_score:
            best_seq = a.copy()
            best_score = sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet trial {trial} c={c:.4f} new best {best_score:.6f}")

    # deterministic shapes (sine, Hann, triangular, exponential, power‑law, quadratic)
    idx = np.arange(n)
    shapes = []

    # half‑wave sine
    shapes.append(np.sin(np.pi * (idx + 1) / (n + 1)))
    # Hann (raised‑cosine) window
    shapes.append(0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1)))
    # symmetric triangular window
    shapes.append(np.minimum(idx, n - 1 - idx).astype(float))
    # exponential decay
    shapes.append(np.exp(-idx / (n / 10.0)))
    # power‑law (p‑=0.5)
    shapes.append((idx + 1) ** -0.5)
    # quadratic (parabolic) window, shifted to be non‑negative
    centre = (n - 1) / 2.0
    quad = 1.0 - ((idx - centre) / (n / 2.0)) ** 2
    shapes.append(np.maximum(quad, 0.0))

    for i, shp in enumerate(shapes):
        shp = np.maximum(shp, 0.0)
        shp = _project_simplex(shp)
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq = shp.copy()
            best_score = sc
            print(f"[{time.time() - start_time:.2f}s] deterministic shape {i} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation of the smoothed max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.70   # about 70 % of the rest
        restarts = 5
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break
            jitter = np.random.default_rng().normal(scale=0.02, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a_opt, sc_opt = _adam_optimize(a0, descent_deadline)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"[{time.time() - start_time:.2f}s] Adam restart {r} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – fast stochastic hill‑climbing (the final fine‑tuning)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting fast hill‑climbing")
        best_seq, best_score = _hill_climb_fast(best_seq, best_score, deadline)

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    final_obj = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {final_obj:.6f}")

    return [float(x) for x in best_seq]
