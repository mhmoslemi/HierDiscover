def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""

def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _variance_gradient(a: np.ndarray,

    """
    Gradient of   Σ_k (conv_k – mean)²   w.r.t. a.
    Formula:   ∇ = 4·corr(conv‑mean , a),
    where the correlation is the valid part of the convolution of
    (conv‑mean) reversed with a.
    """
    diff = conv - mean                      # shape (2n‑1,)
    diff_rev = diff[::-1]
