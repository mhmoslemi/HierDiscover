"""
Propose a candidate step‑function that beats the previous best (~1.5078).

Algorithm
---------
1. Choose a fairly large problem size (n = 8192).  The objective
   2·n·max(conv)/sum² is asymptotically independent of n, but a finer
   discretisation reduces discretisation error of the optimal continuous
   density.

2. Build a rich seed pool:
   * uniform distribution,
   * the symmetric “arcsine” shape a_i ∝ 1/√((i+½)(n‑i‑½)),
   * a one‑parameter symmetric family a_i ∝ ((i+½)(n‑i‑½))^p for p∈[-1.2,‑0.2],
     optionally mixed with the uniform distribution,
   * a handful of random Dirichlet draws.

   Keep the K = 5 best seeds (evaluated by a fast FFT based objective).

3. Refine every seed with Adam on a smooth‑max surrogate
   f_β(a)=log‑sum‑exp(β·(a*a)).  β is increased progressively
   (1 → … → 1024) and the number of Adam steps per seed is limited only
   by the remaining time budget.  After each step the vector is projected
   onto the probability simplex.

4. Polishing:
   * Exponentiated‑gradient descent on the true L∞‑objective,
   * Classical projected sub‑gradient descent with back‑tracking,
   * A targeted random hill‑climb that moves a (up to a few %) fraction of
     mass from indices that contribute heavily to the current maximal
     convolution entry to indices that contribute little.

   All phases respect the simplex constraint and constantly update the
   best sequence seen so far.

5. Return the best sequence (as a plain Python list) after normalising the
   sum to exactly 1.  The final result is then ready for evaluation by the
   provided `evaluate_sequence` function.

The code below is self‑contained, uses only NumPy and the standard library,
and conforms to the required API.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv_full


def _grad_softmax(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2 * Σ_j a_j p_{i+j}.
    Implemented as convolution of a_rev with p.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast objective (exact L∞ version, used during search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2).  a is assumed non‑negative."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (the arg‑max of the convolution)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Parametric families (used for seed generation)
# ----------------------------------------------------------------------


def _arcsine_shape(n: int) -> np.ndarray:
    """Symmetric arcsine density discretised on n points."""
    i = np.arange(n, dtype=np.float64)
    # (i+0.5)*(n-i-0.5) > 0
    dens = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    dens /= dens.sum()
    return dens


def _sym_power_shape(n: int, p: float) -> np.ndarray:
    """Shape a_i ∝ ((i+½)(n‑i‑½))^p, symmetric, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    base = (i + 0.5) * (n - i - 0.5)
    shape = np.power(base, p, where=base > 0, out=np.zeros_like(base))
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).  The routine uses
    a large problem size (n=8192), a rich seed pool (including the arcsine shape),
    Adam on a smooth‑max surrogate, multiplicative‑gradient polishing,
    projected sub‑gradient descent, and a targeted hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    # Problem size ------------------------------------------------------
    n = 8192                       # a good compromise between quality and speed
    L_fft = _next_pow2(2 * n - 1)  # FFT length for exact convolutions
    L_grad = _next_pow2(3 * n - 2)  # FFT length for smooth‑max gradients

    # Baseline -----------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # -------------------------------------------------------------------
    # 1️⃣  Build a diverse seed pool
    # -------------------------------------------------------------------
    seeds = [uniform]

    # arcsine (p = -0.5) – believed to be close to optimal
    arcsine = _arcsine_shape(n)
    seeds.append(arcsine)

    # symmetric power family (p in [-1.2, -0.2]) with optional uniform mixing
    p_vals = np.linspace(-1.2, -0.2, 11)
    lam_vals = [0.0, 0.2, 0.5]
    for p in p_vals:
        shape = _sym_power_shape(n, p)
        for lam in lam_vals:
            seeds.append(_mix_with_uniform(shape, lam))

    # random Dirichlet draws – give extra diversity
    for alpha in [0.3, 0.7, 1.0]:
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # Evaluate all seeds quickly (exact objective)
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 5
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start_time:.2f}s] initial pool best score={best_score:.6f}")

    # -------------------------------------------------------------------
    # 2️⃣  Adam on the smooth‑max surrogate
    # -------------------------------------------------------------------
    # Adam hyper‑parameters
    adam_lr = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # increasingly tight soft‑max temperature
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]

    for beta in beta_schedule:
        if time.time() >= deadline - 200:
            break
        max_iter_per_vec = 4000
        for idx in range(len(pool)):
            a = pool[idx]
            m = np.zeros_like(a)
            v_vec = np.zeros_like(a)
            cur_score = pool_scores[idx]
            t = 0
            # iterate while we still have some headroom for later phases
            while time.time() < deadline - 150:
                t += 1
                # convolution, soft‑max weights and gradient
                conv = _conv_self_fft(a, L_fft)
                p = _softmax_weights(conv, beta)
                g = _grad_softmax(a, p, L_grad)

                # Adam update (mirrored descent)
                m = beta1 * m + (1.0 - beta1) * g
                v_vec = beta2 * v_vec + (1.0 - beta2) * (g * g)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v_vec / (1.0 - beta2 ** t)
                step = adam_lr * m_hat / (np.sqrt(v_hat) + eps)

                a_candidate = a - step
                a_candidate = _project_simplex(a_candidate)

                cand_score = _eval_fast(a_candidate, L_fft)

                # Accept improvements; otherwise allow occasional uphill move
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                else:
                    prob = np.exp(-(cand_score - cur_score) * 5.0 / (np.sqrt(t) + 1.0))
                    if rng.random() < prob:
                        a = a_candidate
                        cur_score = cand_score

                # keep the best ever seen
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam β={beta} it={t} score={cand_score:.6f}")

                if cand_score < pool_scores[idx]:
                    pool_scores[idx] = cand_score
                    pool[idx] = a_candidate.copy()

                if t >= max_iter_per_vec:
                    break

    # -------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing (multiplicative updates)
    # -------------------------------------------------------------------
    if time.time() < deadline - 120:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 120, start_time,
            best_seq, best_score, rng)

    # -------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent (classic polishing)
    # -------------------------------------------------------------------
    if time.time() < deadline - 70:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 70, start_time,
            best_seq, best_score, rng)

    # -------------------------------------------------------------------
    # 5️⃣  Targeted random hill‑climb (larger mass moves guided by gradient)
    # -------------------------------------------------------------------
    while time.time() < deadline:
        # compute current convolution and its arg‑max
        conv = _conv_self_fft(best_seq, L_fft)
        k_max = int(np.argmax(conv))
        grad = _subgrad_max_conv(best_seq, k_max)

        # pick source from the top‑5 % of gradient entries, destination from bottom‑5 %
        n_top = max(1, n // 20)
        src_candidates = np.argpartition(-grad, n_top - 1)[:n_top]
        dst_candidates = np.argpartition(grad, n_top - 1)[:n_top]

        src = int(rng.choice(src_candidates))
        dst = int(rng.choice(dst_candidates))
        if src == dst or best_seq[src] < 1e-14:
            continue

        # move a modest (0.1 % – 2 %) fraction of the source mass
        eps = best_seq[src] * rng.random() * rng.uniform(0.001, 0.02)
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        # due to floating‑point drift re‑project
        prop = _project_simplex(prop)

        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={new_score:.6f}")

    # -------------------------------------------------------------------
    # Final normalisation and conversion to plain list
    # -------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]

# Helper polishing phases (kept unchanged from the reference implementation,
# still useful but with a few tweaks for stability)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking line search."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill jump to escape shallow plateaus
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score
