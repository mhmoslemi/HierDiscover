"""
Permutation‑annealing search for a low‑max‑self‑convolution step function.

The objective to minimise is

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

with a≥0 and n the length of the sequence.  Because the objective is
scale‑invariant we keep the vector on the probability simplex
(∑a_i = 1) during the whole search.

Algorithmic outline
-------------------
1. **Diverse stochastic starts** – many Dirichlet draws with a very
   wide range of concentration parameters (log10(c)∈[‑5, 0.5])
   plus occasional sparse random vectors.  For every draw we also
   apply a cheap “anti‑correlation” permutation that pairs the largest
   entry with the smallest one when the sequence is reversed – this is
   known to reduce the convolution peak.

2. **Permutation simulated‑annealing** – the best vector from step 1
   undergoes a long annealing that works only on the permutation of
   the entries (swaps).  Swaps preserve the ℓ₁‑norm, are cheap to apply,
   and allow the algorithm to escape the local minima that gradient‑based
   moves tend to get stuck in.  Acceptance follows a Metropolis rule with
   a slowly cooling temperature.

3. **Projected gradient descent (PGD)** – a short fine‑tuning phase
   that moves opposite to a sub‑gradient of the active convolution
   entries and projects back onto the simplex.

4. **Final polishing** – very cheap random pairwise transfers that
   accept only strict improvements.

All phases respect the global 1000 s wall‑time limit; the best sequence
found so far is returned as a plain list of float.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global timing
# ----------------------------------------------------------------------
start_time = None  # will be set in propose_candidate


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Helpers for the anti‑correlated re‑ordering
# ----------------------------------------------------------------------
def _anti_correlate(arr: np.ndarray) -> np.ndarray:
    """
    Produce a permutation that pairs large values with small ones
    when the sequence is reversed.  For i < n/2 we place the i‑th
    largest entry at position i and the i‑th smallest entry at position
    n‑1‑i.
    """
    n = arr.size
    asc = np.sort(arr)                # ascending
    desc = asc[::-1]                  # descending
    out = np.empty_like(arr)

    half = n // 2
    for i in range(half):
        out[i] = desc[i]              # large at the front
        out[n - 1 - i] = asc[i]       # tiny at the back
    if n % 2 == 1:                    # middle element for odd n
        out[half] = desc[half]
    return out


# ----------------------------------------------------------------------
# Sampling utilities
# ----------------------------------------------------------------------
def _sample_dirichlet(n: int, rng: np.random.Generator,
                      logc_range: tuple[float, float] = (-5.0, 0.5)) -> np.ndarray | None:
    """
    Draw a Dirichlet sample with concentration c = 10**logc
    where logc is uniform in *logc_range*.
    Returns a normalised probability vector or None if the draw fails.
    """
    logc = rng.uniform(*logc_range)
    c = 10.0 ** logc
    # Gamma shape must be >0; extremely small c gives many near‑zeros.
    a = rng.gamma(shape=c, scale=1.0, size=n)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def _sample_sparse(n: int, rng: np.random.Generator,
                   sparsity_range: tuple[float, float] = (0.001, 0.05)) -> np.ndarray:
    """
    Produce a random sparse vector with roughly *ratio*·n non‑zero entries.
    The non‑zero values are exponential.
    """
    ratio = rng.uniform(*sparsity_range)
    k = max(2, int(ratio * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 2️⃣  Permutation simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(start_seq: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the permutation of ``start_seq``.
    Swapping two entries preserves the simplex constraint.
    """
    n = start_seq.shape[0]
    cur = start_seq.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    # temperature initialisation – a few percent of the current value
    T = max(cur_score * 0.05, 1e-9)
    decay = 0.99996

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        # propose a swap
        cur[i], cur[j] = cur[j], cur[i]
        new_score = _objective_score(cur)
        diff = new_score - cur_score

        # Metropolis acceptance
        if diff <= 0.0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
                print(f"[{time.time() - start_time:.2f}s] PermSwap new best {best_score:.6f}")
        else:
            # revert the swap
            cur[i], cur[j] = cur[j], cur[i]

        T = max(T * decay, 1e-12)

    return best, best_score


# ----------------------------------------------------------------------
# 3️⃣  Projected gradient descent (same logic as the reference code)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at ``a`` (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _projected_gradient_descent(a0: np.ndarray,
                                cur_score: float,
                                deadline: float) -> tuple[np.ndarray, float]:
    """
    Fine‑tune ``a0`` by moving opposite to the sub‑gradient of the active
    convolution entries and projecting back onto the simplex.
    """
    a = a0.copy()
    best = a.copy()
    best_score = cur_score
    step = 0.5

    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best = a.copy()
                    best_score = cand_score
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)

    return best, best_score


# ----------------------------------------------------------------------
# 4️⃣  Tiny random local refinement
# ----------------------------------------------------------------------
def _random_local_refinement(a0: np.ndarray,
                             cur_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap random pairwise transfers (no temperature) – accept only
    strict improvements.  Acts as a final polishing step.
    """
    a = a0.copy()
    best = a.copy()
    best_score = cur_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        delta = a[src] * rng.random() * 0.005
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            a = cand
            best = cand.copy()
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine combines:
      • a huge pool of stochastic (Dirichlet/sparse) starts with an
        anti‑correlation permutation,
      • a long simulated‑annealing on the permutation space,
      • a short projected‑gradient fine‑tuning,
      • a final random local polish.
    The best vector found within the time budget is returned as a list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – we stay with n = 4096 (the same size used before)
    # --------------------------------------------------------------
    n = 4096

    # start with the uniform distribution
    best_seq = np.full(n, 1.0 / n, dtype=float)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – intensive random sampling + cheap anti‑correlation
    # --------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 150.0)
    while time.time() < phase1_deadline:
        a = _sample_dirichlet(n, rng)
        if a is None:
            continue
        # raw Dirichlet candidate
        sc_raw = _objective_score(a)
        if sc_raw < best_score:
            best_seq, best_score = a.copy(), sc_raw
            print(f"[{time.time() - start_time:.2f}s] Dirichlet raw best {best_score:.6f}")

        # anti‑correlated re‑ordering of the same sample
        a_anti = _anti_correlate(a)
        sc_anti = _objective_score(a_anti)
        if sc_anti < best_score:
            best_seq, best_score = a_anti.copy(), sc_anti
            print(f"[{time.time() - start_time:.2f}s] Dirichlet anti best {best_score:.6f}")

        # occasional sparse starter (5 % chance)
        if rng.random() < 0.05:
            a_sparse = _sample_sparse(n, rng)
            sc_sparse = _objective_score(a_sparse)
            if sc_sparse < best_score:
                best_seq, best_score = a_sparse.copy(), sc_sparse
                print(f"[{time.time() - start_time:.2f}s] Sparse start best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – permutation simulated annealing on the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = min(deadline, time.time() + 600.0)   # up to 10 min for this phase
        print(f"[{time.time() - start_time:.2f}s] Starting permutation annealing")
        best_seq, best_score = _permutation_annealing(best_seq, best_score,
                                                      perm_deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] Permutation annealing finished – best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected gradient descent (fine‑tuning)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting projected‑gradient descent")
        best_seq, best_score = _projected_gradient_descent(best_seq,
                                                           best_score,
                                                           deadline)
        print(f"[{time.time() - start_time:.2f}s] PGD finished – best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final tiny random local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = min(deadline, time.time() + 30.0)
        print(f"[{time.time() - start_time:.2f}s] Starting random local refinement")
        best_seq, best_score = _random_local_refinement(best_seq,
                                                        best_score,
                                                        local_deadline,
                                                        rng)
        print(f"[{time.time() - start_time:.2f}s] Local refinement finished – best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return (ensure strict feasibility)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
