def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Search for a non-negative step function minimizing the L∞ self-convolution bound.

    The algorithm uses a parametric family (geometric × power law), a local
    parametric refinement, two phases of sub‑gradient / exponentiated‑gradient
    descent on the exact objective, and a final simulated‑annealing hill‑climb.
    It works on a vector length n = 8192 (adjustable) and respects the 1000 s budget.
    """
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    # Problem size
    n = 8192
    L_fft = _next_pow2(2*n-1)

    # Baseline uniform
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time()-start:.2f}s] uniform baseline {best_score:.6f}")

    # If existing global sequence available, upsample
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # Upsample by repeating to match n
            factor = int(np.ceil(n / starter.size))
            up = np.repeat(starter, factor)[:n]
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                up /= up.sum()
                cur = _eval_fast(up, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = up.copy()
                    print(f"[{time.time()-start:.2f}s] global starter improved {cur:.6f}")
    except Exception:
        pass

    # ---- Stage 1: parametric sweep over (lam, p) -----------------
    sweep_end = start + 80.0   # 80 seconds for sweep + local refinement
    min_lambda = 2e-05
    max_lambda = 2e-03
    lam_grid = np.linspace(min_lambda, max_lambda, 400)
    p_grid = np.linspace(-8.0, 4.0, 121)  # step 0.1

    best_lam, best_p = None, None

    for lam in lam_grid:
        if time.time() >= sweep_end:
            break
        base = -lam * np.arange(n, dtype=np.float64)
        for p in p_grid:
            log_shape = base + p * np.log(np.arange(1, n+1, dtype=np.float64))
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                best_lam, best_p = lam, p
                print(f"[{time.time()-start:.2f}s] sweep lam={lam:.6e} p={p:.2f} score={sc:.6f}")

    # ---- Stage 2: local parametric refinement (lam,p) -------------
    if best_lam is None:
        # fallback to some default values
        best_lam = (min_lambda + max_lambda) / 2.0
        best_p = -2.0

    refine_end = start + 150.0   # allocate up to 150 sec for refinement and subgrad
    step_lam = (max_lambda - min_lambda) * 0.1
    step_p = (p_grid[-1] - p_grid[0]) * 0.1

    while time.time() < refine_end:
        # generate perturbation
        lam_cand = best_lam + step_lam * (rng.random() * 2 - 1)
        p_cand = best_p + step_p * (rng.random() * 2 - 1)
        # clip
        lam_cand = float(np.clip(lam_cand, min_lambda, max_lambda))
        p_cand = float(np.clip(p_cand, p_grid[0], p_grid[-1]))
        # build shape
        i = np.arange(n, dtype=np.float64)
        log_shape = -lam_cand * i + p_cand * np.log(i + 1.0)
        log_shape -= np.max(log_shape)
        cand = np.exp(log_shape)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_lam, best_p = lam_cand, p_cand
            print(f"[{time.time()-start:.2f}s] refine lam={lam_cand:.6e} p={p_cand:.2f} score={sc:.6f}")
        # decay steps
        step_lam *= 0.999
        step_p *= 0.999
        # slight random reset if step sizes become too small
        if step_lam < 1e-8 or step_p < 0.001:
            step_lam = (max_lambda - min_lambda) * 0.05
            step_p = (p_grid[-1] - p_grid[0]) * 0.05

    # ---- Stage 3: projected subgradient descent -------------------
    # Allocate time
    if time.time() < deadline - 300.0:
        sub_deadline = deadline - 300.0
        best_seq, best_score = _subgradient_descent(best_seq, L_fft,
                                                    sub_deadline, start,
                                                    best_seq, best_score, rng)

    # ---- Stage 4: exponentiated gradient --------------------------
    if time.time() < deadline - 200.0:
        exp_deadline = deadline - 200.0
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              exp_deadline, start,
                                              best_seq, best_score, rng)

    # ---- Stage 5: Simulated annealing final hill climb ----------
    # We'll use a temperature schedule from 1e-4 down to 1e-8
    cur_seq = best_seq.copy()
    cur_score = best_score
    T0 = 1e-4
    # annealing exponent
    while time.time() < deadline:
        # proportion of remaining time
        remaining = deadline - time.time()
        total = deadline - start
        fraction = remaining / total
        T = T0 * fraction**2  # quadratic cooling

        # pick source proportionally to its mass
        src = rng.choice(n, p=cur_seq)  # probability mass distribution
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to max frac of src
        eps = cur_seq[src] * rng.random() * 0.01   # up to 1% of source mass
        if eps <= 1e-15:
            continue
        prop = cur_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        # recompute score
        new_score = _eval_fast(prop, L_fft)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-18)):
            cur_seq = prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = prop.copy()
                print(f"[{time.time()-start:.2f}s] SA improvement {new_score:.6f}")

    # Final normalisation
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0/n, dtype=np.float64)

    print(f"[{time.time()-start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works on the full vector (default n = 8192) and consists of
five stages that together respect the 1000 s time budget:

1️⃣  **Parametric sweep** – exhaustive scan of the two‑parameter family
    a_i ∝ (i+1)^p · exp(-λ·i) (geometric × power law).  The best pair
    (λ, p) is remembered.

2️⃣  **Local (λ, p) refinement** – a simple stochastic hill‑climb in the
    (λ, p) space starting from the sweep optimum; this refines the shape
    beyond the coarse grid.

3️⃣  **Projected sub‑gradient descent** – optimisation of the exact
    L∞‑objective on the simplex using the sub‑gradient of the maximal
    convolution entry.

4️⃣  **Exponentiated‑gradient (multiplicative) updates** – a second
    simplex‑preserving descent that often polishes the solution further.

5️⃣  **Simulated‑annealing hill‑climb** – tiny mass‑transfer moves are
    accepted according to a temperature schedule, allowing the search to
    escape tiny plateaus in the final seconds.

All phases maintain non‑negative entries, normalise the vector to sum ≈1,
and keep the best sequence found.  The returned value is a plain Python
list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective (exact, assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (at the index attaining the max)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]  # diagonal terms appear twice, which is correct
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates a ← a·exp(-η·g) with annealed η."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.8
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # annealed step size
        eta = eta0 / (1.0 + 0.0008 * it)

        # protect against overflow in exp
        mult = np.exp(-eta * np.clip(g, -700, 700))
        a = a * mult
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")

        # occasional jitter to shake plateaus
        if it % 3000 == 0:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (back‑tracking line‑search)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on max(conv) with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.35
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # decreasing step
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(10):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time()-start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # very occasional uphill jump
            prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        if it % 2000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb on the full vector
# ----------------------------------------------------------------------


def _simulated_annealing(a_start: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves accepted with a temperature schedule.
    """
    cur_seq = a_start.copy()
    cur_score = best_score
    n = a_start.shape[0]

    # initial temperature (scale of typical objective changes)
    T0 = 1e-4

    while time.time() < deadline:
        # quadratic cooling
        remaining = deadline - time.time()
        total = deadline - start_time
        T = T0 * (remaining / total) ** 2

        # pick source index proportionally to its mass (more likely to move)
        src = rng.choice(n, p=cur_seq)
        dst = rng.integers(n)
        if src == dst:
            continue

        # move ≤1 % of the source mass
        eps = cur_seq[src] * rng.random() * 0.01
        if eps <= 1e-15:
            continue

        cand = cur_seq.copy()
        cand[src] -= eps
        cand[dst] += eps

        new_score = _eval_fast(cand, L_fft)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-18)):
            cur_seq = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cand.copy()
                print(f"[{time.time()-start_time:.2f}s] SA improvement {new_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the evaluation
    function.  Returns a plain list of floats (sum≈1).  The implementation
    follows the multi‑stage strategy described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # Small safety margin to avoid overrunning the budget
    deadline = start + budget_s - 0.12

    # ------------------------------------------------------------------
    # Problem size – 8192 gives a finer discretisation while still fast.
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time()-start:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try to initialise from the globally supplied sequence if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # Upsample by tiling/repeating to the required length.
            factor = int(np.ceil(n / starter.size))
            up = np.tile(starter, factor)[:n]
            up = np.maximum(up, 0.0)
            if up.sum() > 0.0:
                up = up / up.sum()
                cur = _eval_fast(up, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = up.copy()
                    print(f"[{time.time()-start:.2f}s] global starter improved score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep over (λ, p) – ≈80 s.
    # ------------------------------------------------------------------
    sweep_deadline = start + 80.0
    min_lambda = 2e-05
    max_lambda = 2e-03
    lam_grid = np.linspace(min_lambda, max_lambda, 400)               # 400 points
    p_grid = np.linspace(-8.0, 4.0, 121)                               # step ≈0.1

    best_lam = None
    best_p = None

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * np.arange(n, dtype=np.float64)
        for p in p_grid:
            log_shape = base + p * np.log(np.arange(1, n + 1, dtype=np.float64))
            log_shape -= np.max(log_shape)           # stabilise exponentials
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                best_lam, best_p = lam, p
                print(f"[{time.time()-start:.2f}s] sweep λ={lam:.6e} p={p:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Local stochastic refinement of (λ, p) – up to 70 s.
    # ------------------------------------------------------------------
    if best_lam is None:  # fallback if sweep didn’t hit anything
        best_lam = (min_lambda + max_lambda) * 0.5
        best_p = -2.0

    refine_deadline = start + 150.0
    step_lam = (max_lambda - min_lambda) * 0.1
    step_p = (p_grid[-1] - p_grid[0]) * 0.1

    while time.time() < refine_deadline:
        lam_cand = best_lam + step_lam * (rng.random() * 2 - 1)
        p_cand = best_p + step_p * (rng.random() * 2 - 1)

        lam_cand = float(np.clip(lam_cand, min_lambda, max_lambda))
        p_cand = float(np.clip(p_cand, p_grid[0], p_grid[-1]))

        i = np.arange(n, dtype=np.float64)
        log_shape = -lam_cand * i + p_cand * np.log(i + 1.0)
        log_shape -= np.max(log_shape)
        cand = np.exp(log_shape)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_lam, best_p = lam_cand, p_cand
            print(f"[{time.time()-start:.2f}s] refine λ={lam_cand:.6e} p={p_cand:.2f} score={sc:.6f}")

        # decay the perturbation radius
        step_lam *= 0.999
        step_p *= 0.999
        # occasional reset to keep exploration alive
        if step_lam < 1e-8 or step_p < 0.001:
            step_lam = (max_lambda - min_lambda) * 0.05
            step_p = (p_grid[-1] - p_grid[0]) * 0.05

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient descent (more intensive now we have a good start).
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        sub_deadline = deadline - 200.0
        best_seq, best_score = _subgradient_descent(best_seq, L_fft,
                                                    sub_deadline, start,
                                                    best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Exponentiated‑gradient phase.
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        exp_deadline = deadline - 100.0
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              exp_deadline, start,
                                              best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(best_seq, L_fft,
                                                   deadline, start,
                                                   best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time()-start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
