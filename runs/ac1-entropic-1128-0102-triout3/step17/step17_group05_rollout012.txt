"""
Propose a non‑negative step function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search uses several complementary heuristics within the 1000 s budget:

1️⃣   Fast parametric sweeps (geometric, power, exponential, linear,
     inverse, two‑block) covering a dense grid of shapes.
2️⃣   Massive random Dirichlet draws – the concentration is drawn
     log‑uniformly, producing both dense and highly sparse candidates.
3️⃣   A projected smooth‑max gradient descent (τ·log‑sum‑exp surrogate)
     with a slow exponential cooling schedule.  The descent is run
     from several independent restarts (the current best, a random
     Dirichlet and a perturbed version of the best) to escape local
     minima.
4️⃣   Simple hill‑climbing perturbations (transfer a little mass
     between two coordinates) that polish the result.

All helpers are top‑level, the simplex projection is O(n), and the
convolution uses a single FFT per evaluation.  The routine returns a
plain Python list of floats (sum normalised to 1, all entries ≥ 0).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Assume sum(seq)==1.  Return 2·n·max(conv(seq,seq))."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-6, -0.3, 180)      # more points than before
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 13):  # 0,0.1,…,1

            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[geo] eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score

# 3️⃣ Smooth‑max gradient descent (multiple restarts)
# ----------------------------------------------------------------------
def _smooth_max_descent(a0, deadline, best_seq, best_score, rng):
    """
    Projected gradient descent on the τ·log‑sum‑exp surrogate.
    τ is cooled geometrically (factor 0.9995) – a slow schedule that
    allows the algorithm to approach a true sub‑gradient solution.
    """
    a = a0.copy()
    n = a.shape[0]

    # pre‑compute once the current convolution to initialise τ
    cur_conv = _conv_self(a)
    tau = max(cur_conv.max() * 0.5, 1e-8)
    min_tau = 1e-12
    decay = 0.9995

    while time.time() < deadline and tau > min_tau:
        # --- current convex data -------------------------------------------------
        conv = _conv_self(a)
        max_conv = conv.max()

        # --- smoothed soft‑max weights -----------------------------------------
        shift = max_conv
        log_w = (conv - shift) / tau
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0.0:
            tau *= decay
            continue
        p = w / w_sum

        # --- gradient of the smooth surrogate ------------------------------------
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]

        # extract the n‑dimensional gradient slice
        grad = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # --- line search on the *real* objective --------------------------------
        cur_obj = 2.0 * n * max_conv
        step = 1.0
        improved = False
        while step > 1e-14:
            a_new = a - step * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-12:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[gd] it={int(step*1e6)} obj={obj_new:.6f} τ={tau:.2e}")
                break
            step *= 0.5

        if not improved:
            # small random jitter to escape flat plateaus
            eps = 1e-8
            a = a + rng.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

        tau *= decay

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣ Simple hill‑climbing local search
# ----------------------------------------------------------------------
def _local_hill_climb(seq, deadline, best_score):
    """Transfer a tiny amount of mass between two random indices."""
    n = seq.shape[0]
    rng = np.random.default_rng()
    cur_seq = seq.copy()
    cur_obj = best_score
    eps = 1e-3                     # absolute amount moved each trial
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < eps:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps
        # re‑normalise (total unchanged, but to remove tiny numerical drift)
        new_seq /= new_seq.sum()
        new_obj = _objective_score(new_seq)
        if new_obj < cur_obj:
            cur_seq = new_seq
            cur_obj = new_obj
            if new_obj < best_score:
                best_score = new_obj
                best_seq = new_seq.copy()
                print(f"[hill] improved to {best_score:.6f}")
        # slowly shrink the perturbation as time runs out
        if deadline - time.time() < 30.0:
            eps *= 0.99
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for the step‑function that minimises
        2·n·max(conv(a,a))/(sum(a)^2).

    Returns a list of Python floats (the sequence is normalised to sum = 1).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[init] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Use any globally supplied starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0:
                starter = starter / total
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[global] starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Parametric sweep (≈120 s)
    # --------------------------------------------------------------
    sweep_deadline = deadline - 250.0
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_linear(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_inverse(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, sweep_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈250 s)
    # --------------------------------------------------------------
    dirichlet_deadline = deadline - 200.0
    if time.time() < dirichlet_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, dirichlet_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣ Smooth‑max gradient descent (several restarts)
    # --------------------------------------------------------------
    # Warm‑start from the current best
    gd_deadline = deadline - 60.0
    rng = np.random.default_rng()
    restarts = 5
    for r in range(restarts):
        if time.time() >= gd_deadline:
            break
        # choose a different starting point for each restart
        if r == 0:
            a0 = best_seq.copy()
        elif r == 1:
            # a fresh Dirichlet sample (moderate concentration)
            a0 = rng.gamma(shape=0.5, scale=1.0, size=n)
            a0 /= a0.sum()
        else:
            # perturb current best slightly
            eps = 1e-3 * (1.0 + r)
            a0 = best_seq + rng.uniform(-eps, eps, size=n)
            a0 = np.clip(a0, 0.0, None)
            a0 = _project_simplex(a0)

        best_seq, best_score = _smooth_max_descent(
            a0, gd_deadline, best_seq, best_score, rng
        )
        print(f"[gd] restart {r+1}/{restarts} best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Local hill‑climbing polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _local_hill_climb(best_seq, deadline, best_score)

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[final] best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
