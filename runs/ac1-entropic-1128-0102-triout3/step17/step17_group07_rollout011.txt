"""
Softmax-gradient descent for minimizing max convolution of a step function.

The method:
- Initialise a probability vector a (sum=1, non‑negative) using uniform or a
  Dirichlet warm‑start.
- In each iteration compute the self‑convolution a∗a.
- Build a soft‑max weight vector w over the convolution entries whose
  temperature β grows linearly from β0 to βmax.
- The gradient of the soft‑max surrogate is obtained analytically via
  two FFT‑based convolutions.
- An Adam step is taken and the iterates are projected onto the probability
  simplex.
- The best exact score (2·n·max(conv)) seen so far is stored.
- When the time budget is exhausted the best vector is returned.
"""

import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT of length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    # truncate to needed length
    n = a.shape[0]
    return conv[: 2 * n - 1]

def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return soft‑max probabilities proportional to exp(beta*conv)."""
    # shift by max for numeric stability
    m = conv.max()
    exp = np.exp(beta * (conv - m))
    sum_exp = exp.sum()
    if sum_exp == 0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / sum_exp

def _gradient_softmax(a: np.ndarray, w: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of  (1/β)·log Σ exp(β·conv)   w.r.t. a,
    where w is the soft‑max probability vector over the convolution entries.
    The formula is  2·[ (a ∗ w)[0:n]  +  (a ∗ w_rev)[(2n‑2)‑(0:n)] ] .
    """
    n = a.shape[0]
    # FFT of a
    A = np.fft.rfft(a, n=L_grad)
    # FFT of w (zero‑padded to length L_grad)
    W = np.fft.rfft(w, n=L_grad)
    # forward convolution term: (a ∗ w)   → keep first n entries
    conv_fwd = np.fft.irfft(A * W, n=L_grad)[:n]

    # reversed w
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=L_grad)
    conv_rev_full = np.fft.irfft(A * W_rev, n=L_grad)
    # pick the slice that corresponds to Σ_j a_j w_{i+j}
    # index = (2n-2) - i   for i = 0,…,n‑1
    idx = (2 * n - 2) - np.arange(n)
    conv_rev = conv_rev_full[idx]

    grad = 2.0 * (conv_fwd + conv_rev)
    return grad

def _exact_score(a: np.ndarray, n: int, L_self: int) -> float:
    """Exact score 2·n·max(a∗a) assuming Σ a = 1."""
    conv = _conv_self_fft(a, L_self)
    return float(2.0 * n * conv.max())

    """

    # -------------------------------------------------

    # -------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)

    try:

        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0:
                starter /= total

"""
Soft‑max gradient descent for minimising the peak of a self‑convolution.

The algorithm works with a probability vector a (non‑negative, Σa=1).

1️⃣  Warm‑start – a few Dirichlet draws (≈5 % of the budget) plus an
    optional global starter (the variable ``height_sequence_1``).

2️⃣  Main optimisation – minimise the smooth surrogate
        Fβ(a)= (1/β)·log Σk exp(β·(a∗a)_k)
    by Adam steps.  The gradient of Fβ is obtained analytically with two
    FFT‑based convolutions.  After each step the iterate is projected onto
    the probability simplex.

    The temperature β is increased linearly from β_start to β_end,
    so the surrogate slowly turns into the true max‑convolution objective.

3️⃣  Final refinement – a short convex mixing with the uniform vector
    followed by a cheap random‑pair hill‑climb.

The routine respects a 1000 s wall‑clock budget, prints progress, and
returns a plain Python list of floats (already normalised to sum≈1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT of length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max probabilities proportional to exp(beta·conv)."""
    m = conv.max()
    exp = np.exp(beta * (conv - m))
    s = exp.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / s


def _gradient_softmax(a: np.ndarray, w: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of   (1/β)·log Σ exp(β·conv)   w.r.t. a.

    The gradient equals  2·[ (a∗w)_0:n  +  (a∗w_rev)[(2n‑2)‑(0:n)] ] .
    """
    n = a.shape[0]

    # FFT of a (zero‑padded to L_grad)
    A = np.fft.rfft(a, n=L_grad)

    # forward term  a∗w
    W = np.fft.rfft(w, n=L_grad)
    conv_fwd = np.fft.irfft(A * W, n=L_grad)[:n]

    # reverse‑w term
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=L_grad)
    conv_rev_full = np.fft.irfft(A * W_rev, n=L_grad)

    idx = (2 * n - 2) - np.arange(n)          # (2n‑2)‑i  for i=0…n‑1
    conv_rev = conv_rev_full[idx]

    grad = 2.0 * (conv_fwd + conv_rev)
    return grad


def _exact_score(a: np.ndarray, n: int, L_self: int) -> float:
    """Exact objective: 2·n·max(a∗a)  (sum(a) is assumed to be 1)."""
    conv = _conv_self_fft(a, L_self)
    return float(2.0 * n * conv.max())

# Main entry point
# ----------------------------------------------------------------------

    Heuristic optimisation (soft‑max + Adam + simplex projection) for a step
    function that minimises ``evaluate_sequence``.  Returns a plain Python list
    of non‑negative floats (approximately unit sum).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10          # tiny safety margin

    # -------------------------------------------------
    # Problem size – larger n gives a lower theoretical bound.
    # -------------------------------------------------
    n = 8192                         # 8192 works comfortably in the time budget
    L_self = _next_pow2(2 * n - 1)   # for a*a
    L_grad = _next_pow2(3 * n - 2)   # for the two convolutions needed in the gradient

    # -------------------------------------------------
    # Initialise a – try to use the global starter if present,
    # otherwise start from the uniform distribution.
    # -------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter /= s
                a = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter loaded, init score={_exact_score(a,n,L_self):.6f}")
    except Exception:
        pass

    # -------------------------------------------------
    # Warm‑start – a handful of Dirichlet draws.
    # -------------------------------------------------
    warm_end = min(deadline, start_time + 0.05 * (deadline - start_time))
    best_a = a.copy()
    best_score = _exact_score(best_a, n, L_self)

    rng = np.random.default_rng()
    while time.time() < warm_end:
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)

        sc = _exact_score(cand, n, L_self)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    # -------------------------------------------------

    beta1 = 0.9
    beta2 = 0.999

    m = np.zeros_like(a)
    v = np.zeros_like(a)

        beta = beta_start + (beta_end - beta_start) * progress

        w = _softmax_weights(conv, beta)

        grad = _gradient_softmax(a, w, L_grad)

    # -------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    for lam in np.linspace(0.0, 1.0, 21):

        cand /= cand.sum()
        sc = _exact_score(cand, n, L_self)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap stochastic hill‑climb (tiny mass transfers)
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _exact_score(cand, n, L_self)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill‑climb i={i}→{j} δ={delta:.5f} → score={sc:.6f}")

    # -------------------------------------------------
    # Normalise (safety) and return as plain Python list
    # -------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
