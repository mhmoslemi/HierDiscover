"""
Propose a high‑quality non‑negative step function by combining three
phases:

1️⃣ Smoothed‑max (soft‑max) optimisation.  The true max of the
   self‑convolution is approximated by a log‑sum‑exp with an annealed
   temperature β.  Gradients are computed analytically and an
   Adam‑style optimiser updates the logits of a soft‑max parametrisation
   (a_i = exp(x_i) / Σexp(x)).  This yields fast, smooth progress toward a
   flat convolution profile.

2️⃣ Exact sub‑gradient descent.  When β becomes large we switch to the
   true L∞‑objective and perform a projected sub‑gradient descent on the
   simplex, using the analytical sub‑gradient of the maximal convolution
   entry.

3️⃣ Final random hill‑climb.  Tiny mass‑transfer moves (simulated‑annealing
   style) finish the search.

All phases respect the 1000 s budget, keep the vector on the probability
simplex, and keep track of the best sequence seen (using the fast FFT
evaluation).  The returned list is normalised to sum ≈ 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _cross_corr_fft(a: np.ndarray, w_rev: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Cross‑correlation a ⋆ w (i.e. g_i = Σ_j a_j w_{i+j}) via FFT.
    Returns the full linear convolution a * w_rev (length n+m‑1).
    """
    n = a.shape[0]
    m = w_rev.shape[0]
    if L is None:
        L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to ≈1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # If a global starter exists we adopt its length, otherwise fallback to 4096.
    try:
        starter_raw = np.asarray(height_sequence_1, dtype=float)
        if starter_raw.ndim == 1 and starter_raw.size > 0:
            n = starter_raw.size
        else:
            raise ValueError
    except Exception:
        n = 4096

    L_conv = _next_pow2(2 * n - 1)          # FFT length for self‑conv
    L_cross = _next_pow2(3 * n - 2)         # FFT length for cross‑corr

    # ------------------------------------------------------------------
    # Initialise a (on the simplex) – use the global starter if possible,
    # otherwise start from uniform.
    # ------------------------------------------------------------------
    if 'starter_raw' in locals():
        a = np.maximum(starter_raw, 0.0)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a.sum()
        print(
            f"[{time.time() - start_time:.2f}s] Started from global starter (len={n})"
        )
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)
        print(f"[{time.time() - start_time:.2f}s] Started from uniform (len={n})")

    # keep track of the best sequence seen so far
    best_seq = a.copy()
    best_score = _eval_fast(best_seq, L_conv)
    print(f"[{time.time() - start_time:.2f}s] baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed‑max Adam optimisation
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.60 * (deadline - start_time)
    # Adam state
    x = np.log(a + 1e-30)                     # logits (log‑softmax parameters)
    m_adam = np.zeros_like(x)
    v_adam = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12                                 # base learning‑rate
    beta_start, beta_end = 1.0, 500.0          # temperature schedule
    it = 0
    eval_interval = 2000
    while time.time() < phase1_end:
        it += 1
        # recompute a from logits (soft‑max, numerically stable)
        x_shift = x - np.max(x)
        a = np.exp(x_shift)
        a = a / a.sum()

        # ----- smoothed max -----
        conv = _conv_self_fft(a, L_conv)          # self‑convolution
        # linearly anneal β on a log‑scale
        t_frac = (time.time() - start_time) / (phase1_end - start_time)
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # soft‑max weights over convolution entries
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()

        # ----- gradient w.r.t. a -----
        w_rev = w[::-1]
        cross = _cross_corr_fft(a, w_rev, L_cross)   # length = 3n‑2
        # extract the needed slice and reverse to obtain g_i
        grad_a = 2.0 * cross[n - 1 : 2 * n - 1][::-1]

        # ----- gradient w.r.t. logits x (soft‑max chain rule) -----
        grad_x = a * (grad_a - np.dot(a, grad_a))

        # ----- Adam update -----
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_x
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m_adam / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)
        lr = lr0 / (1.0 + 0.0004 * it)              # mild decay
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Occasionally evaluate the true (non‑smoothed) objective
        if it % eval_interval == 0:
            cur_score = _eval_fast(a, L_conv)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Phase‑1 it={it:,} β={beta:.1f} score={cur_score:.6f}"
                )
            # tiny random jitter to escape flat regions
            if it % (eval_interval * 5) == 0:
                jitter = 1e-5 * rng.random(n) * a
                a = np.maximum(a + jitter, 0.0)
                a /= a.sum()
                x = np.log(a + 1e-30)

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent on the true L∞‑objective
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.90 * (deadline - start_time)
    # start from the best point of phase 1
    a = best_seq.copy()
    step0 = 0.30
    it2 = 0
    sub_eval_interval = 500
    while time.time() < phase2_end:
        it2 += 1
        conv = _conv_self_fft(a, L_conv)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        step = step0 / (it2 ** 0.55)          # decaying step‑size
        cand = a - step * g
        cand = _project_simplex(cand)

        cand_score = _eval_fast(cand, L_conv)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(
                f"[{time.time() - start_time:.2f}s] Phase‑2 it={it2:,} step={step:.5e} score={cand_score:.6f}"
            )
        else:
            # occasional uphill move (tiny probability) to shake out of plateaus
            if rng.random() < 0.001:
                a = cand

        if it2 % sub_eval_interval == 0:
            cur_score = _eval_fast(a, L_conv)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Phase‑2 eval it={it2:,} score={cur_score:.6f}"
                )
        # jitter every few thousand steps
        if it2 % 4000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Phase 3 – final random hill‑climb (tiny mass transfers)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 5e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 2e-3  # ≤0.2 % of source mass
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_conv)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={new_sc:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (normalised to sum≈1)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
