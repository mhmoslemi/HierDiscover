Implementation of simplex projection:

```
def project_to_simplex(v):
    # v is np.ndarray
    # returns projection onto simplex {x >=0, sum(x)=1}
    # Use algorithm from Wang & Carreira-Perpinan 2013
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import cvxpy as cp
import scipy.sparse as sp

def solve_qcqp_small(n_small=256, max_iters=5000, reltol=1e-6):
    a = cp.Variable(n_small)
    M = cp.Variable()
    constraints = [a >= 0, cp.sum(a) == 1]

    mats = []
    for j in range(2*n_small - 1):
        # construct sparse matrix
        rows = []
        cols = []
        data = []
        i_start = max(0, j - (n_small - 1))
        i_end = min(n_small - 1, j)
        for i in range(i_start, i_end + 1):
            k = j - i
            if i < k:
                rows.append(i); cols.append(k); data.append(0.5)
                rows.append(k); cols.append(i); data.append(0.5)
            elif i == k:
                rows.append(i); cols.append(i); data.append(1.0)
        Mj = sp.csr_matrix((data, (rows, cols)), shape=(n_small, n_small))
        mats.append(Mj)

    for Mj in mats:
        constraints.append(cp.quad_form(a, Mj) <= M)

    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=cp.SCS, eps=1e-5, max_iters=max_iters, verbose=False) # maybe tune
    if a.value is None:
        return None
    a_opt = np.maximum(a.value, 0.0)
    a_opt /= a_opt.sum()
    return a_opt
```

"""
Propose a step‑function (non‑negative vector) that minimises
        score = 2·n·max(conv(a,a)) / (sum(a)²)

Algorithm
---------

1.  Baseline candidates – uniform, the arcsine (Beta(0.5,0.5)) shape and a tiny
    mixture with the uniform vector.  Any global starter stored in the variable
    ``height_sequence_1`` is also examined.

2.  Convex “QCQP” warm‑start – we solve the convex problem

        minimise   M
        subject to a_i ≥ 0 , Σ a_i = 1
                    aᵀ Q_j a ≤ M   for every shift j,
        where (Q_j)ₖₗ = ½ if k+l=j and k≠l, 1 if k=l=j/2, 0 otherwise.

    The problem is convex because each Q_j is PSD.  We solve it with
    cvxpy (SCS) for a modest size (n_small=256).  The optimal small‑scale
    distribution is interpolated to the full size (n=4096) and used as a
    promising start.

3.  Projected gradient descent on the *soft‑max* approximation of the
    Chebyshev norm of the convolution.
    - At each iteration we compute the full self‑convolution of the current
      vector a (FFT based, O(n log n)).
    - A temperature τ controls a soft‑max weight vector w that concentrates
      on the largest convolution entries.
    - The gradient of the weighted sum wᵀconv(a) equals
          g = 2·(w * a_rev)   (a_rev is a reversed copy of a)
      and is obtained again by an FFT‑based convolution.
    - A step of fixed Euclidean length (≈0.02) is taken opposite to g and
      projected onto the probability simplex (non‑negative, sum = 1).
    - τ is slowly increased so that the optimisation gradually focuses on the
      true maximum.  The best vector seen during this phase is stored.

4.  Final cheap hill‑climb – random pairwise transfers of a tiny amount of
    mass (accepting only improvements) are performed until the time budget
    expires.

All stages respect the total time budget (default 1000 s) and print short
progress messages.  The returned list contains non‑negative floats that sum
to ≈1, which is scale‑invariant for the evaluator.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> tuple[float, np.ndarray]:
    """Exact score used by the evaluator (sum(a) is assumed to be 1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    # sum(a) == 1 → score = 2·n·max(conv)
    return float(2.0 * n * max_b), conv


def _beta_shape(n: int, p: float, q: float | None = None) -> np.ndarray:
    """
    Discretised Beta(p,q) density on the grid i=0,…,n‑1
    (if q is None we use the symmetric case p==q).
    The result is normalised to sum 1.
    """
    if q is None:
        q = p
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # clip to avoid 0^negative
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _solve_qcqp_small(n_small: int, max_time: float) -> np.ndarray | None:
    """
    Solve the convex QCQP:
        minimise   M
        subject to a_i ≥ 0 , Σ a_i = 1
                    aᵀ Q_j a ≤ M   for each shift j = 0,…,2·n‑2 .
    Returns a feasible vector a (sum = 1) or None on failure.
    """
    start = time.time()
    a = cp.Variable(n_small)
    M = cp.Variable(nonneg=True)
    constraints = [a >= 0, cp.sum(a) == 1]

    # pre‑compute the PSD matrices Q_j (sparse)
    mats = []
    for j in range(2 * n_small - 1):
        rows = []
        cols = []
        data = []
        i_min = max(0, j - (n_small - 1))
        i_max = min(n_small - 1, j)
        for i in range(i_min, i_max + 1):
            k = j - i
            if i < k:
                rows.extend([i, k])
                cols.extend([k, i])
                data.extend([0.5, 0.5])
            elif i == k:
                rows.append(i)
                cols.append(i)
                data.append(1.0)
        Qj = sp.csr_matrix((data, (rows, cols)), shape=(n_small, n_small))
        mats.append(Qj)

    for Qj in mats:
        constraints.append(cp.quad_form(a, Qj) <= M)

    prob = cp.Problem(cp.Minimize(M), constraints)

    # use SCS – first‑order, works with many quadratic constraints
    # we limit the runtime loosely; the outer caller ensures enough time.
    try:
        prob.solve(solver=cp.SCS, eps=1e-5, max_iters=5000, verbose=False)
    except Exception as e:
        print(f"[QCQP] Solver exception: {e}")
        return None

    if a.value is None:
        print("[QCQP] No solution returned.")
        return None

    a_opt = np.asarray(a.value).flatten()
    a_opt = np.maximum(a_opt, 0.0)
    total = a_opt.sum()
    if total > 0.0:
        a_opt /= total
    else:
        a_opt = np.full(n_small, 1.0 / n_small)
    # make sure we respect the time limit
    if time.time() - start > max_time:
        print("[QCQP] Warning: solver hit the time budget.")
    return a_opt


def _upsample(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a_small (size n_small) to length n_big."""
    n_small = a_small.shape[0]
    x_small = (np.arange(n_small) + 0.5) / n_small
    x_big = (np.arange(n_big) + 0.5) / n_big
    a_big = np.interp(x_big, x_small, a_small, left=0.0, right=0.0)
    a_big = np.maximum(a_big, 0.0)
    total = a_big.sum()
    if total > 0.0:
        a_big /= total
    else:
        a_big = np.full(n_big, 1.0 / n_big)
    return a_big


def _gradient_descent(a_start: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate of max(conv).
    Returns the best vector found and its score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score, _ = _score_of(a, n, L_fft)

    # pre‑compute stuff for the gradient convolution
    Lgrad = _next_pow2(3 * n - 2)

    tau = 1.0                     # initial soft‑max temperature
    tau_max = 250.0
    iteration = 0
    target_step = 0.02            # Euclidean length of each GD step

    while time.time() < deadline:
        # 1) self‑convolution and max entry
        conv = _conv_self_fft(a, L_fft)
        max_conv = conv.max()

        # 2) soft‑max weights focusing on large entries
        shifted = conv - max_conv          # ≤ 0
        w = np.exp(tau * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 3) gradient of weighted sum (2·(w * a_rev))
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=Lgrad)
        Af = np.fft.rfft(a_rev, n=Lgrad)
        tmp = np.fft.irfft(Wf * Af, n=Lgrad)
        g = 2.0 * tmp[n - 1 : n - 1 + n]   # length n

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break

        step = target_step / norm_g
        a_new = a - step * g
        a_new = _project_to_simplex(a_new)

        # evaluate
        sc, _ = _score_of(a_new, n, L_fft)

        # accept if improvement (otherwise keep old a)
        if sc <= best_score:
            # we keep the improvement both as the current point and as the global best
            a = a_new
            best_score = sc
            best_a = a_new.copy()
            print(f"[{time.time() - start:.2f}s] GD improve → score = {sc:.6f}")

        # schedule temperature (slowly tighten)
        tau = min(tau_max, tau * 1.005)
        iteration += 1

    return best_a, best_score


def _hill_climb(a_start: np.ndarray,
                n: int,
                L_fft: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random pairwise mass transfers (tiny delta) that accept only
    improvements.  Returns the best vector seen and its score.
    """
    a = a_start.copy()
    best_score, _ = _score_of(a, n, L_fft)
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # move a tiny random amount; bound it to keep changes smooth
        max_delta = min(a[i], a[j], 0.001)
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑normalise (numerical drift)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s
        sc, _ = _score_of(cand, n, L_fft)
        if sc < best_score:
            best_score = sc
            a = cand
            print(f"[{time.time() - start:.2f}s] hill‑climb improve → score = {sc:.6f}")
    return a, best_score

        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start:.2f}s] arcsine λ={lam:.2f} → score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Optional small‑scale QCQP warm‑start (if time permits)
    # ------------------------------------------------------------------
    time_now = time.time()
    remaining = deadline - time_now
    # reserve at least 200 s for later stages
    if remaining > 250.0:
        # n_small can be tuned; 256 offers a good trade‑off between quality and speed
        n_small = 256
        max_qcqp_time = min(120.0, remaining * 0.4)   # cap the solver time
        print(f"[{time_now - start:.2f}s] Solving QCQP (n={n_small}) …")
        small_a = _solve_qcqp_small(n_small, max_qcqp_time)
        if small_a is not None:
            # interpolate to the full size
            big_a = _upsample(small_a, n_big)
            sc, _ = _score_of(big_a, n_big, L_fft_big)
            if sc < best_score:
                best_score, best_a = sc, big_a.copy()
                print(f"[{time.time() - start:.2f}s] QCQP up‑sampled candidate score = {sc:.6f}")
        else:
            print(f"[{time.time() - start:.2f}s] QCQP solver did not produce a solution.")
        # refresh remaining time
        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 3️⃣  Gradient‑descent refinement on the soft‑max of the convolution
    # ------------------------------------------------------------------
    gd_deadline = min(deadline, start + 0.75 * budget_s)
    print(f"[{time.time() - start:.2f}s] Starting gradient descent")
    gd_a, gd_score = _gradient_descent(best_a, n_big, L_fft_big, gd_deadline)
    if gd_score < best_score:
        best_score, best_a = gd_score, gd_a.copy()
        print(f"[{time.time() - start:.2f}s] Gradient descent improvement → score = {gd_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb (random pairwise transfers)
    # ------------------------------------------------------------------
    hill_deadline = deadline
    print(f"[{time.time() - start:.2f}s] Starting hill‑climb refinement")
    hc_a, hc_score = _hill_climb(best_a, n_big, L_fft_big, hill_deadline)
    if hc_score < best_score:
        best_score, best_a = hc_score, hc_a.copy()
        print(f"[{time.time() - start:.2f}s] Hill‑climb final improvement → score = {hc_score:.6f}")

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
