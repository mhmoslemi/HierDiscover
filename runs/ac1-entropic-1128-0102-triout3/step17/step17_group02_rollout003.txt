"""
Optimiser for the self‑convolution supremum problem.

The algorithm proceeds in three phases:

1️⃣  Initialise a diverse pool of candidates (uniform, sine, Hann,
    triangular, arcsine‑like) and a fast random Dirichlet sweep.
    The best of these seeds the optimiser.

2️⃣  Smooth optimisation with an Adam‑style optimiser on the soft‑max
    parametrisation of the probability vector.  The non‑smooth
    ``max(conv)`` is approximated by a log‑sum‑exp with a temperature
    β that is gradually increased (β‑schedule).  For each β we run a
    limited number of Adam steps, always keeping the best exact
    objective (2·n·max(conv)).

3️⃣  A very cheap stochastic hill‑climbing (mass‑transfer) phase for
    the last seconds, able to exploit remaining improvement opportunities.

All heavy convolutions use FFTs and the soft‑max implicitly enforces the
probability simplex, so we never need an explicit projection.  The
total running time respects the 1000 s budget; the function returns the
best sequence found as a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used by helper functions for logging)
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv) with temperature β."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    w are soft‑max weights on the convolution entries.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()                     # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)     # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Simple soft‑max parametrisation (ensures a≥0, sum(a)=1)
# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z = z - np.max(z)          # shift for stability
    e = np.exp(z)
    s = e.sum()
    if s == 0.0:
        # fallback – should never happen
        return np.full_like(z, 1.0 / z.size)
    return e / s


# ----------------------------------------------------------------------
# Random Dirichlet initialisation (time‑bounded)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """
    Draw up to *max_trials* Dirichlet vectors (log‑uniform concentration)
    and keep the best one w.r.t. the exact objective.
    """
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration factor uniformly in log‑space ∈[10⁻³, 10⁰·⁷] ≈ [0.001, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10.0 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s                     # normalise to sum = 1
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climbing (mass‑transfer) refinement
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float, deadline: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer moves for the last few seconds."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # transfer up to 5 % of the source mass
        delta = a[src] * rng.random() * 0.05
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # soft‑max style re‑normalisation (projection onto simplex)
        cand = cand / cand.sum()
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            # occasional downhill move to escape tiny plateaux
            a = cand
            cur_score = cand_score
    return best_a, best_score


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    # ------------------------------------------------------------------
    # Global timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (trade‑off between expressiveness and speed)
    # ------------------------------------------------------------------
    n = 4096
    # generic uniform starter (baseline)
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter exists (height_sequence_1) use it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – deterministic shapes + short Dirichlet sweep
    # ------------------------------------------------------------------
    shapes = []

    # sine (half‑wave, positive)
    shapes.append(np.sin(np.pi * (np.arange(n) + 1) / (n + 1)))

    # Hann window
    shapes.append(0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # symmetric triangular window
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    shapes.append(tri)

    # arcsine‑like (large at the borders, small in the centre)
    idx = np.arange(n, dtype=float) + 0.5
    arcsine = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    shapes.append(arcsine)

    # sqrt‑shape (peaked in the centre)
    sqrt_shape = np.sqrt(np.minimum(idx, n + 0.5 - idx))
    shapes.append(sqrt_shape)

    for shp in shapes:
        shp = np.maximum(shp, 0.0)
        shp = shp / shp.sum()
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq, best_score = shp.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] deterministic shape score {sc:.6f}")

    # Dirichlet draws (limited to 30 s)
    dirichlet_deadline = min(deadline, start_time + 30.0)
    if time.time() < dirichlet_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=800, deadline=dirichlet_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth Adam optimiser on the soft‑max parametrisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        opt_deadline = time.time() + remaining * 0.80   # 80 % of what's left for Adam

        # β‑schedule – start very smooth, finish very sharp
        beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0, 10000.0]

        # Adam hyper‑parameters
        adam_lr = 0.12          # base learning‑rate for the log‑its
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        # initialise log‑its from the current best probability vector
        z = np.log(best_seq + 1e-14)   # avoid log(0)

        for b_idx, beta in enumerate(beta_schedule):
            if time.time() >= opt_deadline:
                break

            # allocate a modest slice of the remaining optimiser budget to this β
            stage_end = min(opt_deadline,
                            time.time() + (opt_deadline - time.time()) / (len(beta_schedule) - b_idx))

            m = np.zeros_like(z)
            v = np.zeros_like(z)
            t = 0

            while time.time() < stage_end:
                a = _softmax(z)                     # probability vector (sum = 1)
                conv = _conv_self(a)                # full convolution
                grad_a = _smooth_gradient(a, conv, beta)   # ∂smooth_max/∂a

                # chain rule: ∂L/∂z_i = a_i (grad_a_i – ⟨a, grad_a⟩)
                inner = np.dot(a, grad_a)
                dz = a * (grad_a - inner)

                # Adam update
                t += 1
                m = beta1 * m + (1.0 - beta1) * dz
                v = beta2 * v + (1.0 - beta2) * (dz * dz)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = adam_lr * m_hat / (np.sqrt(v_hat) + eps)

                # limit the magnitude of a single update (prevents wild jumps)
                max_step = 1.0
                step = np.clip(step, -max_step, max_step)

                z -= step

                # Periodically evaluate the exact objective (every 15 it.)
                if t % 15 == 0:
                    cur_score = _objective_score(a)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} new best {best_score:.6f}")

                # early exit if the update has collapsed
                if np.max(np.abs(step)) < 1e-10:
                    break

    # ------------------------------------------------------------------
    # Phase 3 – short hill‑climbing tweak for the very end
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hc_deadline = deadline
        print(f"[{time.time() - start_time:.2f}s] starting hill‑climbing")
        best_seq, best_score = _hill_climb(best_seq, best_score, hc_deadline)

    # ------------------------------------------------------------------
    # Final normalisation (safety) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
