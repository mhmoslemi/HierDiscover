"""
Propose a step‑function (non‑negative sequence) that minimises
    score = 2·n·max(conv(a,a)) / (sum(a)²)

Algorithm
---------
1. Choose a relatively large problem size (n = 8192).  Larger n gives a
   finer discretisation of the optimal continuous solution, pushing the
   theoretical lower bound 1 closer to the discrete case.

2. Warm‑start with a few sensible candidates:
   * the continuous arcsine law (Beta(½,½)) discretised on the grid,
   * convex mixtures of the arcsine shape with the uniform vector,
   * a few Dirichlet draws – cheap random points on the simplex.

   All candidates are evaluated with an exact FFT‑based convolution, and
   the best one is kept.

3. Refine the best candidate by a smooth min‑max optimisation.
   The objective `max(conv(a,a))` is approximated by a soft‑max
   `f_α(a) = (1/α)·log Σ exp(α·c_k)` where `c = a*a`.
   The gradient of `f_α` is
        ∇f_α = 2 ·  ( w ⋆ a )        (⋆ = linear correlation)
   with `w_i = exp(α·c_i) / Σ exp(α·c)`.  Correlation is computed via a
   single FFT, giving an O(n log n) iteration.

   We start with a modest smoothing parameter `α≈5` and increase it
   geometrically (α←α·1.02) up to a large value (≈2000) so that the
   smoothed objective converges to the true max.  After each gradient
   step we project back onto the probability simplex using the
   Duchi‑et‑al. O(n) projection.

4. Final local optimisation (hill‑climb):
   Randomly pick two indices and transfer a tiny amount of mass
   `δ`.  Accept the move only if the exact score (computed with the
   fast FFT routine) improves.  This cleans up any remaining
   discretisation artefacts.

The whole routine respects the supplied time budget (default 1000 s)
and prints intermediate progress.  The returned sequence consists of
non‑negative numbers that sum to one, which is scale‑invariant for the
evaluator.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    # a is kept normalised, sum(a)=1 ⇒ score = 2·n·max_b
    return float(2.0 * n * max_b)


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised Beta(p,p) density on the grid i=0,…,n‑1,
    i.e. a_i ∝ x_i^{p‑1} (1‑x_i)^{p‑1} with x_i = (i+0.5)/n.
    Returned vector is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    # Based on Duchi et al., 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal or negative -> return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            if s > 0.0:

                sc = _score_of(starter, n, L_fft)

                    print(f"[{time.time() - start_time:.2f}s] Global starter improved score = {sc:.6f}")

        sc = _score_of(a, n, L_fft)
        if sc < best_score:
