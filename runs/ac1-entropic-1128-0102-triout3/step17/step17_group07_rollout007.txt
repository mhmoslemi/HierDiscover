```python
def _warm_start_dirichlet(n, best_a, best_score, L_fft, deadline):
```

```python
def _mirror_descent_max(a0, n, L_fft, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)
    rng = np.random.default_rng()

    it = 0
    eta0 = 3.0  # maybe larger
    while time.time() < deadline:
        it += 1
        # compute conv
        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] MirrorIter={it:,} score={cur_score:.6f}")

        # gradient subgradient
        k = int(conv.argmax())
        # compute gradient g_i = 2 * a[k - i] for valid indices
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        g = np.zeros_like(a)
        i_idx = np.arange(i_start, i_end + 1)
        g[i_idx] = 2.0 * a[k - i_idx]

        # step size schedule
        eta = eta0 / np.sqrt(it)
        # mirror descent update (exponential weights)
        a = a * np.exp(-eta * g)
        # renormalize
        a_sum = a.sum()
        if a_sum == 0.0:
            # reinitialize to uniform
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        # occasional random jitter to escape local minima
        if rng.random() < 0.01:
            eps = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a /= a.sum()
    return best_a, best_score
```

```python
def _final_mixing(a, n, L_fft, deadline):
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing
    for lam in np.linspace(0.0, 1.0, 21):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand = cand / cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # random pair hill climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill i={i}→{j} δ={delta:.5f} → score={sc:.6f}")
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative vector) that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of four stages:

1️⃣  Warm‑start – a collection of cheap candidates (global starter,
    uniform vector and Dirichlet draws).  The best of them is kept.

2️⃣  Sub‑gradient descent on the *maximum* of the self‑convolution.
    The sub‑gradient of `max(conv)` is elementary:
    for the current peak index `k*` the i‑th component equals
    `2·a_{k*‑i}` (if the index is in range, otherwise 0).
    After each gradient step the vector is projected onto the simplex
    (non‑negative, sum = 1) using the O(n log n) Duchi projection.
    A decreasing step‑size  η₀/√t  and occasional random jitter help
    escape plateaux.

3️⃣  Final refinement – convex mixing with the uniform distribution
    followed by a light random‑pair hill‑climb (as in the reference).

All FFT‑based convolutions are performed with a power‑of‑two length
to keep every iteration fast.  The routine respects the 1000 s budget,
prints timing information and finally returns a plain Python list of
floats whose sum is (approximately) 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """
    Exact score used by ``evaluate_sequence`` (the sum of a is assumed to be 1).
    """
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Warm‑start via Dirichlet draws (cheap candidates)
# ----------------------------------------------------------------------
def _warm_start_dirichlet(
    n: int,
    best_a: np.ndarray,
    best_score: float,
    L_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sample many Dirichlet‑like candidates and keep the best."""
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration parameter drawn log‑uniformly
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the maximum of the convolution
# ----------------------------------------------------------------------
def _subgrad_descent_max(
    a0: np.ndarray,
    n: int,
    L_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on  max(conv(a,a)).
    The sub‑gradient at the current peak index k* is
        g_i = 2·a_{k*‑i}   (if 0 ≤ k*‑i < n, else 0).
    After each step we project back onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    rng = np.random.default_rng()
    it = 0
    eta0 = 0.7  # initial stepsize; tuned experimentally

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] SubGradIter={it:,} score={cur_score:.6f}"
            )

        # ---- sub‑gradient of the max entry ----
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        g = np.zeros_like(a)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            g[idx] = 2.0 * a[k - idx]

        # ---- step size (1/√t) ----
        eta = eta0 / np.sqrt(it)

        # ---- projected gradient step ----
        a_new = a - eta * g
        a = _project_simplex(a_new)

        # ---- occasional random jitter to leave flat regions ----
        if rng.random() < 0.005:
            eps = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light final polishing (mixing + random‑pair hill‑climb)
# ----------------------------------------------------------------------
def _final_refinement(
    a: np.ndarray,
    n: int,
    L_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Mix with uniform and perform a cheap random‑pair hill‑climb."""
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform vector
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Light hill i={i}→{j} δ={delta:.5f} → score={sc:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a non‑negative step‑function that minimises
    the evaluation score.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # safety margin to avoid overshooting the budget
    deadline = start_time + budget_s - 0.15

    # problem size – a power of two gives fast FFTs; we use 8192 points
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Basic initialiser (uniform) and optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # if a global starter named ``height_sequence_1`` is present, use it
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size < n:
                # repeat until we reach the required length
                reps = int(np.ceil(n / starter.size))
                starter_up = np.tile(starter, reps)[:n]
            else:
                starter_up = starter[:n]
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0.0:
                starter_up /= s
                sc = _objective_score(starter_up, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter_up.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        # ignore if the variable does not exist or is malformed
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start: cheap Dirichlet draws (≈5 % of total budget)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * (deadline - start_time))
    if time.time() < warm_deadline:
        best_a, best_score = _warm_start_dirichlet(
            n, best_a, best_score, L_fft, warm_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent on the maximum convolution entry
    # ------------------------------------------------------------------
    if time.time() < deadline:
        desc_deadline = time.time() + 0.80 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_a, best_score = _subgrad_descent_max(
            best_a, n, L_fft, desc_deadline
        )
        print(
            f"[{time.time() - start_time:.2f}s] After descent best score≈{best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing (mixing + tiny hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement")
        best_a, best_score = _final_refinement(best_a, n, L_fft, deadline)

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
