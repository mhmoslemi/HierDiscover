def propose_candidate(seed:int=0, budget_s:int=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n=4096
    L_fft_self = _next_pow2(2*n-1)
    L_fft_corr = _next_pow2(3*n-2)  # for correlation

    # baseline uniform
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft_self)
    print(f"[{time.time() - start:.2f}s] uniform score {best_score:.6f}")

    # try global starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim==1 and starter.size>0:
            starter = np.maximum(starter,0)
            if starter.sum()==0:
                starter = uniform.copy()
            else:
                starter = starter/ starter.sum()
            cur = _eval_fast(starter, L_fft_self)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time()-start:.2f}s] global starter improved to {cur:.6f}")
    except Exception:
        pass

    # 1️⃣ Beta shape sweep
    beta_sweep_deadline = start + 40.0  # 40 secs for sweep
    p_vals = np.linspace(-4.5, 0.5, 121)  # step 0.04 approx
    # symmetrical shape: a_i = ((i+1)*(n-i))^p
    # Actually using exponent p (the same exponent for both sides)
    i = np.arange(n, dtype=np.float64)
    left = i + 1
    right = n - i
    for p in p_vals:
        if time.time() >= beta_sweep_deadline:
            break
        shape = (left * right) ** p
        shape_sum = shape.sum()
        if shape_sum == 0:
            continue
        cand = shape / shape_sum
        sc = _eval_fast(cand, L_fft_self)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time()-start:.2f}s] beta sweep p={p:.3f} score={sc:.6f}")

    # 2️⃣ Smooth gradient descent (SGD)
    smooth_deadline = deadline - 250.0  # reserve some for final phases
    a = best_seq.copy()
    eta0 = 0.5  # initial step size
    beta = 2.0  # initial smoothing parameter
    beta_max = 80.0
    beta_inc = (beta_max - beta) / 8000  # incremental step per iter approx
    it = 0
    while time.time() < smooth_deadline:
        it += 1
        # Compute convolution a*a
        conv = _conv_self_fft(a, L_fft_self)  # length 2n-1
        # Softmax weights for smoothed max
        # avoid overflow
        maxc = conv.max()
        # Compute beta scaling factor; we may increase gradually
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            # fall back to uniform
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum
        # Compute gradient: 2 * cross-correlation of a and w
        # cross-correlation = conv_rev[n-1:n-1+n] where conv_rev = conv_fft(reverse(a), w)
        a_rev = a[::-1]
        corr_full = conv_fft(a_rev, w, L_fft_corr)
        grad = 2.0 * corr_full[n-1:n-1+n]   # length n
        # step size schedule: decreasing with sqrt(it)
        step = eta0 / np.sqrt(it)
        # update via Euclidean projection onto simplex
        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        # evaluate exact objective
        cur_score = _eval_fast(a_new, L_fft_self)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[{time.time()-start:.2f}s] smooth SGD it={it} beta={beta:.2f} step={step:.3e} score={cur_score:.6f}")
        # accept update (sob basis)
        a = a_new
        # slowly increase beta
        if beta < beta_max:
            beta += beta_inc

        # occasional random jitter
        if it % 2000 == 0:
            # add small noise
            eps = 1e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    # 3️⃣ Subgradient descent refinement (reuse earlier)
    if time.time() < deadline - 100.0:
        sub_deadline = deadline - 100.0
        best_seq, best_score = _subgradient_descent(best_seq, L_fft_self,
                                                    sub_deadline, start,
                                                    best_seq, best_score, rng)
    # 4️⃣ Final random hill climb
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 5e-3   # up to 0.5% of source mass
        cand = best_seq.copy()
        cand[src] -= eps
        cand[dst] += eps
        cur_score = _eval_fast(cand, L_fft_self)
        if cur_score < best_score:
            best_seq = cand
            best_score = cur_score
            print(f"[{time.time()-start:.2f}s] final hill climb improved to {cur_score:.6f}")

    # Normalise final
    best_seq = np.maximum(best_seq, 0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time()-start:.2f}s] Finished best score {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The method consists of four stages:

1️⃣  **Beta‑family sweep** – we try symmetric Beta‑like shapes
    a_i ∝ (i+1)^α·(n‑i)^α for many α in a wide interval.
    This family is not covered by the previous “geometric×power’’ sweep
    and often yields a lower convolution peak.

2️⃣  **Smoothed gradient descent** – the objective  max(conv) is replaced
    by a smooth approximation
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]),
    whose gradient is a simple cross‑correlation:
        ∇f_β(a) = 2·(a ⋆ softmax_β(conv)).
    We start with a small β (very smooth) and increase it slowly,
    performing Euclidean‑projected gradient steps on the probability
    simplex.

3️⃣  **Projected sub‑gradient refinement** – a classic sub‑gradient descent
    on the true max‑convolution (as in the reference code) cleans up any
    remaining slack.

4️⃣  **Final random hill‑climb** – tiny mass‑transfer moves with a
    simulated‑annealing acceptance rule capture any last improvement.

All phases respect the 1000 s budget, keep the vector on the simplex,
and finally return the best sequence discovered as a plain list of
floats (already normalised to sum = 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear convolution x*y via FFT."""
    n1, n2 = x.shape[0], y.shape[0]
    if L is None:
        L = _next_pow2(n1 + n2 - 1)
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)[: n1 + n2 - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – assumes the sum ≈ 1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (as in the reference implementation)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on max(conv) with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.35
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(10):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                      f"step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # probabilistic uphill move (rare)
            prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        if it % 2000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.12   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 is a good trade‑off.
    # --------------------------------------------------------------
    n = 4096

    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft_self)

    try:

            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft_self)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter improved to {cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Symmetric Beta‑family sweep (α exponents).
    # --------------------------------------------------------------
    sweep_deadline = start + 40.0   # 40 s for the sweep
    # α in [-0.9, 0.3] with step ≈0.02 → ~61 candidates
    alpha_grid = np.linspace(-0.9, 0.3, 61)
    i = np.arange(n, dtype=np.float64)
    left = i + 1.0
    right = n - i
    for alpha in alpha_grid:
        if time.time() >= sweep_deadline:
            break
        # shape = (left * right) ** alpha   (α may be negative)
        # protect against NaNs when alpha < 0 and left*right = 0 (doesn't happen)
        shape = np.power(left * right, alpha, out=np.empty_like(left))
        shape_sum = shape.sum()
        if shape_sum == 0.0:
            continue
        cand = shape / shape_sum
        sc = _eval_fast(cand, L_fft_self)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start:.2f}s] beta sweep α={alpha:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent on the soft‑max‑approximation.
    # --------------------------------------------------------------
    # allocate most of the remaining time to this phase
    grad_deadline = deadline - 250.0    # reserve ≈250 s for later phases
    a = best_seq.copy()
    eta0 = 0.5          # initial step‑size
    beta = 2.0          # start smoothing parameter
    beta_max = 80.0
    # linear schedule for β (≈8000 iterations needed to reach beta_max)
    beta_inc = (beta_max - beta) / 8000.0
    it = 0
    while time.time() < grad_deadline:
        it += 1

        # ---- convolution a*a (full) ----
        conv = _conv_self_fft(a, L_fft_self)               # length 2n‑1

        # ---- soft‑max weights ----
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---- gradient = 2 * (a ⋆ w) (cross‑correlation) ----
        a_rev = a[::-1]
        corr_full = _conv_fft(a_rev, w, L_fft_corr)       # length 3n‑2
        grad = 2.0 * corr_full[n - 1:n - 1 + n]            # slice → length n

        # ---- step and projection ----
        step = eta0 / np.sqrt(it)                         # decreasing steps
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        # ---- evaluate ----
        cur_score = _eval_fast(a_new, L_fft_self)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[{time.time() - start:.2f}s] smooth it={it:,} beta={beta:.2f} "

        a = a_new

        # ---- increase smoothing gradually ----
        if beta < beta_max:
            beta += beta_inc

        # ---- occasional jitter to hop out of tiny plateaus ----
        if it % 2000 == 0:
            eps = 1e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    # --------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient refinement (as before).
    # --------------------------------------------------------------
    if time.time() < deadline - 100.0:

        best_seq, best_score = _subgradient_descent(best_seq, L_fft_self,

    # --------------------------------------------------------------
    # 4️⃣  Final random hill‑climb (tiny mass transfers).
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 5e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move at most 0.3 % of the source mass
        eps = best_seq[src] * rng.random() * 3e-3
        cand = best_seq.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand_score = _eval_fast(cand, L_fft_self)
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={cand_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
