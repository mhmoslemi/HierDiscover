assistantfinal**Solution Overview**

The algorithm consists of three phases  

1. **Warm‑start** – generate many random Dirichlet vectors (and optionally use the
   provided global starter) and keep the best one.  
2. **Soft‑max gradient descent** – minimise a smooth approximation of the
   `max(conv(a,a))` term.  
   * The convolution is obtained with an FFT.  
   * A soft‑max with a temperature β turns the non‑smooth max into a differentiable
     loss `L_β = (1/β) log Σ exp(β·conv)`.  
   * The gradient of `L_β` w.r.t. `a` is `∇L = 2·corr(w,a)` where the weights
     `w` are the soft‑max probabilities.  The correlation is computed with a
     second FFT.  
   * Mirror‑descent on the probability simplex (`a ← a·exp(−η∇L)`, renormalise)
     keeps all entries non‑negative and sums to 1.  
   * β is increased slowly to focus the optimisation onto the true maximum, and
     η is decreased with the iteration count.
3. **Fine‑grained hill‑climb** – after the smooth optimisation we perform an
   aggressive local search that moves a tiny amount of probability mass away
   from the pair that contributes most to the current peak of the convolution.
   Only improving moves are accepted.

The three phases respect the 1000 s budget, print progress, and finally return
the best sequence found.

```python
"""
Soft‑max gradient descent + targeted hill‑climb for minimising
    score = 2·n·max(conv(a,a)) / (sum(a)²)

The search works on a probability simplex (∑a = 1, a_i ≥ 0).

1️⃣ Warm‑start with Dirichlet draws (and optional global starter).
2️⃣ Smooth optimisation of the maximum convolution entry using a
   soft‑max surrogate.  Gradient is obtained by a single FFT‑based correlation
   and the iterate is updated by mirror‑descent on the simplex.
3️⃣ Targeted hill‑climbing that attacks the current maximal convolution entry
   by moving a tiny amount of mass from the larger partner of the dominant pair
   to a random destination.  This eliminates the remaining “spikes”.

All steps are time‑boxed; the best vector seen so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (assumes sum(a)=1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over convolution entries, stabilised by subtraction of max."""
    mx = conv.max()
    exp_shifted = np.exp(beta * (conv - mx))
    return exp_shifted / exp_shifted.sum()


def _gradient_softmax(a: np.ndarray, w: np.ndarray,
                     n: int, L_corr: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate L_β.
    g_i = 2 * Σ_k w_k * a_{k-i}
    which is a correlation of w with a.
    """
    a_rev = a[::-1]
    # correlation via FFT (length = len(a_rev)+len(w)-1 = 3n‑2)
    Aw = np.fft.rfft(w, n=L_corr)
    A_rev = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(Aw * A_rev, n=L_corr)
    # extract the valid centre part (indices n‑1 … n‑1+n‑1)
    return 2.0 * corr[n - 1 : n - 1 + n]


def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    For a given convolution index k, return the index pair (i,j) with the
    largest product a_i·a_j (i+j=k) together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    prod = a[idx] * a[k - idx]
    pos = int(prod.argmax())
    i = int(idx[pos])
    j = int(k - i)
    return i, j, float(prod[pos])

def _warm_start(n: int, best_a: np.ndarray, best_score: float,
                L_fft: int, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # draw concentration log‑uniformly
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Soft‑max gradient descent on the maximum convolution entry
# ----------------------------------------------------------------------
def _softmax_descent(a0: np.ndarray, n: int,
                     L_fft: int, L_corr: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the soft‑max surrogate of max(conv(a,a)).
    β is increased slowly so that the surrogate approaches the true max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    beta = 5.0                     # start with a smooth surrogate
    eta0 = 0.2                     # initial step size (mirror‑descent)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SD iter {it:,} score={cur_score:.6f}")

        # soft‑max weights and gradient
        w = _softmax_weights(conv, beta)
        g = _gradient_softmax(a, w, n, L_corr)

        # mirror‑descent update (multiplicative, stays on simplex)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        # schedule: slowly tighten the surrogate
        if it % 200 == 0 and beta < 50.0:
            beta = min(beta * 1.15, 50.0)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.003:
            eps = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a /= a.sum()

    return best_a, best_score

# 3️⃣ Targeted hill‑climb (aggressive local optimisation)
# ----------------------------------------------------------------------
def _targeted_hill_climb(a: np.ndarray, n: int,
                         L_fft: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly attacks the current maximal convolution entry:
    pick the pair (i,j) that contributes most, move a tiny amount of mass from
    the larger member of the pair to a random destination.  Accept only
    improving moves.
    """
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        k_max = int(conv.argmax())
        i, j, contrib = _max_contributions(best_a, k_max)
        if i == -1:
            continue

        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-12:
            continue
        dst = rng.integers(0, n)
        if dst == src:
            continue

        # move a small fraction of the source mass
        delta = rng.uniform(0.0, min(best_a[src], 0.004))
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()               # renormalise
        new_score = _objective_score(cand, n, L_fft)

        if new_score < best_score:
            best_score, best_a = new_score, cand
            print(f"[{time.time() - start_time:.2f}s] Hill src={src}→dst={dst} δ={delta:.5f} → score={new_score:.6f}")

    return best_a, best_score

    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # try to use a provided global starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start (≈5 % of the budget)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * (deadline - start_time))
    if time.time() < warm_deadline:
        best_a, best_score = _warm_start(n, best_a, best_score,
                                         L_fft, warm_deadline)

    # ------------------------------------------------------------------
    # 2️⃣ Soft‑max gradient descent (≈70 % of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        grad_deadline = time.time() + 0.70 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Starting soft‑max descent")
        best_a, best_score = _softmax_descent(best_a, n,
                                              L_fft, L_corr,
                                              grad_deadline)
        print(f"[{time.time() - start_time:.2f}s] After descent best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Targeted hill‑climb (≈20 % of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + 0.20 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Targeted hill‑climb")
        best_a, best_score = _targeted_hill_climb(best_a, n,
                                                  L_fft, hill_deadline)

    # ------------------------------------------------------------------
    # Normalisation & final return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
```
