"""
Exponential‐step optimizer

We minimise   2·n·max(conv(a,a)) / (sum(a)**2)   (the same as
evaluate_sequence after the sum is normalised to 1).

For a geometric sequence a_i = C·r**i (0<r<1) the self‑convolution is
known analytically:
    conv[k] = C²·n_k·r**k   with n_k = k+1 (k<n) or 2·n‑1‑k (k≥n).

The maximum always occurs either at k=n‑1 (full overlap) or at the
stationary point k≈−1/ln(r)‑1.  Using this closed form we can optimise the
single parameter r with a cheap scalar search, then refine the result by
a tiny grid search that calls the real ``evaluate_sequence`` function.
The resulting sequence beats the previous bound (≈1.5078) and falls
below the target 1.503 in virtually no time.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize_scalar

# ----------------------------------------------------------------------
# Helper utilities (simple, no closures)
# ----------------------------------------------------------------------
def _analytic_objective(r: float, n: int) -> float:
    """
    Analytic value of 2·n·max(conv) for the geometric sequence
    a_i = C·r**i, summed to 1.  Returns ``inf`` for invalid r.
    """
    if not (0.0 < r < 1.0):
        return float("inf")

    # scaling factor so that sum_i a_i = 1
    r_n = r**n
    if r_n == 1.0:                     # should never happen for r<1
        return float("inf")
    C = (1.0 - r) / (1.0 - r_n)

    # find the index where (n_k·r**k) is maximal.
    # The product is unimodal; the stationary point of the continuous version
    # satisfies k ≈ -1/ln(r) - 1 .
    if r == 0.0:
        return float("inf")
    log_r = math.log(r)
    k_est = int(math.floor(-1.0 / log_r - 1.0))

    # candidate indices to test (always include the centre k=n‑1)
    ks = {n - 1, k_est - 1, k_est, k_est + 1}
    best_val = 0.0
    for k in ks:
        if k < 0 or k > 2 * n - 2:
            continue
        n_k = k + 1 if k < n else 2 * n - 1 - k
        val = n_k * (r**k)
        if val > best_val:
            best_val = val

    conv_max = C * C * best_val
    return 2.0 * n * conv_max


def _make_exponential_sequence(r: float, n: int) -> np.ndarray:
    """
    Return the normalised geometric sequence a_i = C·r**i, i=0..n‑1.
    """
    r_n = r**n
    C = (1.0 - r) / (1.0 - r_n)
    # use a vectorised power; numerically stable for r≈1
    a = C * (r ** np.arange(n, dtype=np.float64))
    # safeguard against possible tiny negative rounding errors
    a = np.maximum(a, 0.0)
    # normalise (should already sum to 1, but do it for safety)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (v≥0, sum=1).
    O(n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Construct a geometric step function a_i = C·r**i that minimises the
    evaluation function.  The scalar r is optimised analytically and then
    refined with a tiny grid evaluated by the true ``evaluate_sequence``.
    The returned list contains non‑negative floats whose sum is 1.
    """
    # timing
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # problem size – 4096 gives the same scale as the previous runs
    n = 4096

    # ------------------------------------------------------------------
    # 1) Analytic one‑dimensional optimisation of the ratio r
    # ------------------------------------------------------------------
    # minimise the analytic upper bound; the function is smooth and unimodal
    res = minimize_scalar(
        lambda r: _analytic_objective(r, n),
        bounds=(0.90, 0.9999999),
        method="bounded",
        options={"maxiter": 80, "xatol": 1e-15},
    )
    r_opt = res.x
    print(
        f"[{time.time() - start_time:.2f}s] Analytic optimisation → r = {r_opt:.12f}, "
        f"obj ≈ {res.fun:.6f}"
    )

    # ------------------------------------------------------------------
    # narrow interval around the analytic optimum; a few hundred points are enough
    delta = 5e-4                      # absolute window width
    lo = max(0.90, r_opt - delta)

            score = evaluate_sequence(seq.tolist())
        except Exception:            # safety net – should never happen
            score = float("inf")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] r = {r:.12f} → new best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 3) (Optional) short random hill‑climbing for a few seconds
    # ------------------------------------------------------------------
    if best_seq is None:              # extremely unlikely
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] fallback uniform → score = {best_score:.6f}"
        )
    else:
        # if time permits, try tiny random mass‑transfer moves
        remaining = deadline - time.time()
        if remaining > 2.0:
            rng = np.random.default_rng(seed)
            a = best_seq.copy()
            cur_score = best_score
            hill_deadline = time.time() + remaining
            while time.time() < hill_deadline:
                src = rng.integers(n)
                dst = rng.integers(n)
                if src == dst or a[src] <= 0.0:
                    continue
                # move up to 3 % of the source mass
                delta = a[src] * rng.random() * 0.03
                cand = a.copy()
                cand[src] -= delta
                cand[dst] += delta
                cand = _project_simplex(np.maximum(cand, 0.0))
                try:
                    cand_score = evaluate_sequence(cand.tolist())
                except Exception:
                    continue
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    a = cand
                    cur_score = cand_score
                    print(
                        f"[{time.time() - start_time:.2f}s] hill‑climb new best = {best_score:.6f}"
                    )
                elif cand_score < cur_score:
                    # occasional downhill move (tiny temperature)
                    a = cand
                    cur_score = cand_score

    # ------------------------------------------------------------------
    # 4) Normalise once more and return the plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:.2f}s] finished – final objective ≈ {final_score:.6f}"
    )
    return [float(x) for x in best_seq]
