def _grad_smooth(a, beta):
    n = a.shape[0]
    conv = _conv_self(a)
    maxv = conv.max()
    shifted = conv - maxv
    exp_shifted = np.exp(beta * shifted)
    sumexp = np.sum(exp_shifted)
    w = exp_shifted / sumexp   # shape (2n-1)
    # compute cross correlation:
    w_rev = w[::-1]  # reversed
    # conv_rev = np.convolve(w_rev, a, mode='full')
    conv_rev = _conv_full(w_rev, a)  # uses FFT
    # w_rev index len(w)-1 = 2n-2 (since w length 2n-1)
    # compute grad_i = 4*n * conv_rev[ (len(w)-1) - i ]
    idx = (len(w)-1) - np.arange(n)
    grad = 4.0 * n * conv_rev[idx]
    return grad
```

def _conv_full(a,b):
    L = _next_pow2(len(a) + len(b) -1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[:len(a)+len(b)-1]
    return conv
```

    n = a.shape[0]

    beta2=0.999
    epsilon=1e-8
    lr=0.1  # maybe adapt to n scale
    # schedule for beta: start beta_small = 1.0, final beta_max = 200, increase linearly with time fraction.
    beta_start = 1.0
    beta_end = 200.0
    total_time = deadline - start_time
    while time.time() < deadline:
        t += 1
        # compute current beta based on elapsed fraction
        elapsed = time.time() - start_time
        frac = min(elapsed/total_time, 1.0)
        beta = beta_start + (beta_end - beta_start) * frac
        # compute gradient
        grad = _grad_smooth(a, beta)  # returns gradient w.r.t a
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

def _grad_smooth(a, beta):
    n = a.shape[0]
    conv = _conv_self(a)  # shape 2n-1
    maxv = conv.max()
    shifted = conv - maxv  # <=0
    exp_shifted = np.exp(beta * shifted)
    sumexp = exp_shifted.sum()
    w = exp_shifted / sumexp  # weights
    # cross-correlation w * a: compute conv_rev = conv_full(w[::-1], a)
    w_rev = w[::-1]
    conv_rev = _conv_full(w_rev, a)  # length 3n-2
    idx = (len(w) - 1) - np.arange(n)
    grad = 4.0 * n * conv_rev[idx]
    return grad

"""
Hybrid hierarchical optimizer for the self‑convolution bound.

The algorithm proceeds in four stages:

1️⃣ *Coarse search* – generate a diverse pool of short (N≈512) candidate
   vectors (uniform, triangular, cosine, Dirichlet draws, sparse, beta‑family,
   sine‑power).  Keep the best one.

2️⃣ *Smooth Adam optimisation* – minimise a smooth approximation of the
   objective 2·n·max(conv(a,a)) using an Adam‑style update on the
   soft‑max of the convolution entries.  The smoothing parameter β is
   increased linearly from 1 up to ~200, turning the problem from a smooth
   landscape into the true max‑objective.  After each step the vector is
   projected onto the probability simplex.

3️⃣ *Upsampling* – the refined short vector is interpolated to the target
   length (N=4096) and optionally combined with a global starter if one is
   supplied.

4️⃣ *Fine‑tuning* – the full‑size vector is further optimised with the same
   smooth Adam scheme, followed by a few seconds of projected‑gradient
   descent on the exact max‑convolution (sub‑gradient) and a cheap random
   local refinement.  The best feasible vector found before the time budget
   expires is returned.

All operations respect the non‑negativity and ℓ₁‑normalisation constraints,
so the returned list can be passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing
# ----------------------------------------------------------------------
start_time = None   # will be set inside ``propose_candidate``

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b (length len(a)+len(b)-1) using an FFT."""
    L = _next_pow2(len(a) + len(b) - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: len(a) + len(b) - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective and gradient (smooth approximation)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (assumes Σa = 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _grad_smooth(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth objective
        Oβ(a) = (2·n/β)·log Σk exp(β·conv_k(a))
    where conv = a*a (full self‑convolution).
    Returns ∇a Oβ.
    """
    n = a.shape[0]
    conv = _conv_self(a)                      # length 2·n‑1
    maxv = conv.max()
    shifted = conv - maxv                     # ≤0, prevents overflow
    exp_shifted = np.exp(beta * shifted)      # safe because ≤0
    sumexp = exp_shifted.sum()
    # soft‑max weights (sum to 1)
    w = exp_shifted / sumexp                  # shape (2·n‑1,)

    # cross‑correlation of w with a: grad_i = 4·n· Σk w_k·a_{k‑i}
    # implement via convolution of reversed w with a
    w_rev = w[::-1]
    conv_rev = _conv_full(w_rev, a)          # length 3·n‑2 (real)
    idx = (len(w) - 1) - np.arange(n)        # pick the right slice
    grad = 4.0 * n * conv_rev[idx]
    return grad


def _smooth_adam_optimize(a_start: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> (np.ndarray, float):
    """
    Adam optimisation on the smooth objective.  ``beta`` (soft‑max sharpness)
    grows linearly from 1 to 200 during the run.
    Returns the best vector encountered and its exact (non‑smooth) score.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective_score(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.15                    # modest step size (works well for n≈4k)

    beta_start = 1.0
    beta_end = 200.0
    total_budget = max(deadline - start_time, 1e-12)

    while time.time() < deadline:
        t += 1
        # schedule β
        elapsed = time.time() - start_time
        frac = min(elapsed / total_budget, 1.0)
        beta = beta_start + (beta_end - beta_start) * frac

        # gradient of the smooth surrogate
        grad = _grad_smooth(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(np.maximum(a_new, 0.0))

        cur_score = _objective_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Smooth‑Adam new best {best_score:.6f} (β={beta:.1f})")
        a = a_new
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient helpers (used for a final exact PGD phase)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at ``a`` (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _projected_gradient_descent(a0: np.ndarray,
                                cur_score: float,
                                deadline: float) -> (np.ndarray, float):
    """
    Fine‑tune ``a0`` by moving opposite to the sub‑gradient of the
    active convolution entries and projecting back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    step = 0.5

    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break
        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)
    return best_a, best_score


# ----------------------------------------------------------------------
# Random local refinement (very cheap polish)
# ----------------------------------------------------------------------
def _random_local_refinement(a0: np.ndarray,
                             cur_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> (np.ndarray, float):
    """
    Cheap random pairwise mass transfers (strictly improving only).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        delta = a[src] * rng.random() * 0.005
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Candidate generators (small N)
# ----------------------------------------------------------------------
def _sparse_initial(n: int, sparsity_ratio: float, rng: np.random.Generator) -> np.ndarray:
    """
    Random sparse vector with ~sparsity_ratio·n non‑zero entries,
    values drawn from an exponential distribution and ℓ₁‑normalised.
    """
    k = max(2, int(sparsity_ratio * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a /= a.sum()
    return a


def _generate_initial_candidates(N: int, rng: np.random.Generator):
    """Diversified pool of vectors of length N."""
    cand = []

    # uniform
    cand.append(np.full(N, 1.0 / N))

    # triangular (mountain shape)
    t = np.arange(N, dtype=float)
    tri = np.where(t <= (N - 1) / 2,
                   t + 1,
                   N - t)
    cand.append(tri / tri.sum())

    # cosine‑bump (non‑negative)
    cos = np.cos(np.pi * (t - (N - 1) / 2) / (N - 1))
    cos = cos - cos.min()
    cand.append(cos / cos.sum())

    # few random Dirichlet draws
    for _ in range(5):
        conc = 10 ** rng.uniform(-3.0, 0.5)
        a = rng.gamma(shape=conc, scale=1.0, size=N)
        if a.sum() > 0:
            cand.append(a / a.sum())

    # sparse starters
    for _ in range(3):
        a = _sparse_initial(N, sparsity_ratio=rng.uniform(0.01, 0.05), rng=rng)
        cand.append(a)

    # beta‑family shapes
    for _ in range(4):
        alpha = rng.exponential(scale=2.0) + 0.1
        beta = rng.exponential(scale=2.0) + 0.1
        x = np.linspace(0, 1, N)
        a = (x ** alpha) * ((1.0 - x) ** beta)
        cand.append(a / a.sum())

    # sine‑power shapes
    for _ in range(4):
        p = rng.uniform(0.2, 5.0)
        a = np.sin(np.pi * (np.arange(N) + 0.5) / N) ** p
        cand.append(a / a.sum())

    return cand
