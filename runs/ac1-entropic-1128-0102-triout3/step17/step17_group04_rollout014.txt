"""
Propose a step‑function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages.

1️⃣  Light parametric sweep (geometric, exponential, two‑block,
    sine‑powers and Beta‑like families).  These families are cheap to
    evaluate and often produce a decent start far below the uniform
    baseline.

2️⃣  A small number of Frank–Wolfe / LP steps.  For a fixed distribution
    `f` we solve

        maximise    Σ g_i
        subject to  (f * g)_k ≤ M   for all k,
                    g_i ≥ 0,

    where `M = max_k (f * f)_k`.  The LP is linear in the unknown `g`
    and can be solved with SciPy's HiGHS interior‑point method.
    The solution `g` is renormalised to unit sum and mixed with `f`
    (`f ← (1‑t)·f + t·g`).  Repeating a few iterations reduces the
    maximum convolution value.

3️⃣  Final tiny stochastic hill‑climb (random jitter + exponential
    mirror‑descent) to polish the best sequence found.

All heavy work (convolutions) is performed with FFTs, the whole routine
stops well before the 1000 s limit and returns the best sequence discovered.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, projection, score)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Objective `2·n·max(conv(seq,seq))` assuming `sum(seq)=1`.
    The factor 2·n is the exact expression used by the competition.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _build_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1) × n) representing the linear
    constraints (a * g)_k ≤ M for a fixed left factor `a`.
    Row k contains the slice a[:min(k+1,n)] placed at columns that
    contribute to shift k.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for col in range(n):
        A[col : col + n, col] = a
    return A


def _lp_step(f: np.ndarray, mix: float) -> np.ndarray | None:
    """
    One Frank–Wolfe step:
      * compute current bound M = max(conv(f,f))
      * solve   maximise Σ g_i   s.t. (f*g)_k ≤ M, g_i ≥ 0
      * renormalise g to sum 1
      * return the convex combination (1‑mix)·f + mix·g
    Returns ``None`` if the LP fails.
    """
    n = f.shape[0]
    M = _conv_self(f).max()

    # build constraints A·g ≤ M
    A = _build_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_i   →   minimise −Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # solve with HiGHS (fast, robust)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s <= 0.0 or not np.isfinite(s):
        return None
    g /= s                     # renormalise to unit sum

    new_f = (1.0 - mix) * f + mix * g
    # numerical drift – enforce the simplex again
    new_f = np.maximum(new_f, 0.0)
    new_f /= new_f.sum()
    return new_f

        seq[np.isinf(seq)] = 0.0
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0.0:
            continue
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SinPower p={p:.2f} score={sc:.6f}")

    # Beta‑like family: a_i ∝ (i+0.5)^{α‑1} (n‑i‑0.5)^{β‑1}
    pos = np.arange(n, dtype=np.float64) + 0.5
    for α in np.linspace(0.2, 5.0, 11):
        for β in np.linspace(0.2, 5.0, 11):
            if time.time() >= deadline:
                break
            seq = (pos ** (α - 1)) * ((n - pos) ** (β - 1))
            if np.any(np.isnan(seq)) or np.any(seq <= 0):
                continue
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={α:.2f} β={β:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Random Dirichlet draws
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=1500):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)      # concentration ∈ [0.01, 5] (log‑uniform)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe / LP refinement
# ----------------------------------------------------------------------
def _lp_refinement(seq, best_score, deadline, max_iters=20):
    """
    Perform a few Frank‑Wolfe style LP steps.  The mixing coefficient `t`
    starts at 0.5 and is gradually decreased.
    """
    a = seq.copy()
    t = 0.5
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        new_a = _lp_step(a, t)
        if new_a is None:
            # LP failed – fall back to a tiny random perturbation
            eps = 1e-6 * np.random.random(a.shape)
            a = np.clip(a + eps, 0.0, None)
            a /= a.sum()
            continue

        new_score = _objective_score(new_a)
        if new_score < best_score:
            best_score = new_score
            seq = new_a.copy()
            a = new_a
            print(f"[{time.time() - start_time:.2f}s] LP step {it+1} t={t:.3f} score={new_score:.6f}")
        else:
            # no improvement – shrink the step size
            t *= 0.7
            if t < 1e-3:
                break
    return seq, best_score


# ----------------------------------------------------------------------
# Very small stochastic polishing
# ----------------------------------------------------------------------
def _final_polish(a, best_score, deadline, steps=2000):
    """Random jitter + exponential mirror descent (very cheap)."""
    rng = np.random.default_rng()
    cur = a.copy()
    cur_score = _objective_score(cur)
    eta = 0.3
    for i in range(steps):
        if time.time() >= deadline:
            break
        # jitter: move a tiny random amount from one index to another
        i1, i2 = rng.integers(0, cur.shape[0], size=2)
        if cur[i1] == 0:
            continue
        delta = min(cur[i1], rng.random() * cur[i1] * 1e-3)
        cur[i1] -= delta
        cur[i2] += delta
        cur /= cur.sum()
        new_score = _objective_score(cur)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                a = cur.copy()
                print(f"[{time.time() - start_time:.2f}s] Polished step {i+1} score={new_score:.6f}")
        else:
            # revert if not accepted
            cur[i1] += delta
            cur[i2] -= delta
            # occasional mirror‑descent step
            if i % 50 == 0:
                grad = _smooth_max_and_grad(cur, beta=15.0)[1]  # re‑use helper from previous code
                cur = _exp_update(cur, grad, eta)
                cur /= cur.sum()
                cur_score = _objective_score(cur)

    return a, best_score


# ----------------------------------------------------------------------
# Helper for the very last polish (re‑using part of the previous code)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient) where gradient is wrt a.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # shift for stability
    Z = exp_scaled.sum()
    w = exp_scaled / Z                         # soft‑max weights, sum to 1

    # gradient: 2 * Σ_j a_j * w_{i+j}
    # perform convolution a * reverse(w)
    w_rev = w[::-1]
    # using FFT for the convolution
    L = _next_pow2(n + w_rev.shape[0] - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(Af * Bf, n=L)[: n + w_rev.shape[0] - 1]
    start = w.shape[0] - 1
    G = 2.0 * conv_rev[start - np.arange(n)]

    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, G


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponential (mirror‑descent) update: a_i ← a_i·exp(−η·grad_i),
    then renormalise onto the probability simplex.
    """
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine runs three phases:
      1. cheap parametric sweeps (including sine‑power and Beta families);
      2. a few random Dirichlet initialisers;
      3. an LP‑based Frank‑Wolfe refinement followed by a tiny polish.
    The best vector (normalised to sum = 1) is returned as a plain list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # Problem dimension – 1024 offers a good trade‑off between speed
    # and the theoretical lower bound (≈1.0002 for n=1024)
    # --------------------------------------------------------------
    n = 1024
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global starter (if provided by the evaluation environment)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric families (expanded sweep)
    # --------------------------------------------------------------
    phase_deadline = deadline - 500.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet draws
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline, trials=2500
        )

    # --------------------------------------------------------------
    # 3️⃣  LP‑based Frank‑Wolfe refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting LP refinement")
        best_seq, best_score = _lp_refinement(best_seq, best_score, deadline, max_iters=12)

    # --------------------------------------------------------------
    # 4️⃣  Tiny stochastic polish (optional, cheap)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final polishing")
        best_seq, best_score = _final_polish(best_seq, best_score, deadline, steps=1500)

    # --------------------------------------------------------------
    # Normalise and return a clean list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
