    def _subgradient_descent(a0, score0, deadline):
        a = a0.copy()
        best = a.copy()
        best_score = score0
        t = 0
        step0 = 1.0   # evaluate adapt
        while time.time() < deadline:
            conv = _conv_self(a)
            max_val = conv.max()
            # Determine indices within epsilon of max (maybe top few)
            # Use tolerance like rel 1e-12.
            max_mask = np.isclose(conv, max_val, atol=1e-12)  # bool array
            # Build weight vector p with weights normalized
            p = max_mask.astype(float)
            count = p.sum()
            if count == 0:
                break
            p /= count
            # Compute gradient = 2 * (a_rev * p) where * is convolution.
            grad = 2.0 * _convolve_fft(a[::-1], p)[n-1 : n-1+n]
            # step size schedule typical: step = step0 / sqrt(t+1)
            step = step0 / np.sqrt(t + 1)
            # candidate = a - step * grad
            cand = a - step * grad
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{time.time()-start_time:.2f}s] subgrad new best {best_score:.6f}")
            # Accept if improvement, else maybe still update a for next iteration.

"""
A fast two‑stage optimiser for the self‑convolution L∞‑objective.

Stage 1 – a large random Dirichlet sweep (and a few simple deterministic
shapes) to obtain a good starting probability vector.

Stage 2 – a pure sub‑gradient descent on the exact max‑convolution
(every iteration projects back onto the probability simplex).  The
sub‑gradient is obtained analytically from the indices attaining the
maximum of the convolution, avoiding the log‑sum‑exp smoothing used
before.

Finally a short directed hill‑climbing phase moves a small amount of mass
away from the “worst” convolution index, which often yields a final
tiny improvement.

All work is done on vectors normalised to sum = 1, so the final
objective reported by ``evaluate_sequence`` is exactly
``2·n·max(conv(a,a))``.  The routine runs within the 1000 s budget and
returns the best sequence found.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ *x*."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using an FFT of length a power of two."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Objective helpers (sum is forced to 1, denominator = 1)
# --------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution term at *a*.
    It is 2·(a_rev ⋆ w) where w is the normalised indicator of the
    maximal convolution entries.
    """
    conv = _conv_self(a)
    max_val = conv.max()
    # all indices that are (numerically) maximal
    mask = conv >= max_val - 1e-14
    cnt = mask.sum()
    if cnt == 0:               # fallback – should never happen
        mask = np.zeros_like(conv, dtype=bool)
        mask[np.argmax(conv)] = True
        cnt = 1
    w = mask.astype(float) / cnt
    # cross‑correlation a_rev ⋆ w (implemented as convolution)
    rev_corr = _convolve_fft(a[::-1], w)
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# --------------------------------------------------------------
# Random Dirichlet initialisation
# --------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Draw up to *max_trials* Dirichlet vectors, keep the best."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration sampled log‑uniformly in [1e‑3, 5]
        conc = 10 ** rng.uniform(-3.0, 0.7)
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet conc={conc:.4f} score={sc:.6f}")
    return best_seq, best_score


# --------------------------------------------------------------
# Sub‑gradient descent (no smoothing)
# --------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, score0: float, deadline: float):
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    step = 0.5  # initial step size
    while time.time() < deadline:
        grad = _subgradient(a)
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = a - step * factor * grad
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sub‑grad new best {best_score:.6f}")
                break
        if not improved:
            step = max(step * 0.5, 1e-12)
        else:
            step = min(step * 1.05, 1.0)
        if step < 1e-12:
            break
    return best_a, best_score


# --------------------------------------------------------------
# Directed hill‑climbing (mass transfer away from worst lag)
# --------------------------------------------------------------
def _directed_hill_climb(a0: np.ndarray, score0: float, deadline: float):
    """Mass‑transfer moves that explicitly target the worst convolution lag."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        k_max = int(conv.argmax())
        # partner index for each i under this lag
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)
        partner_vals = np.zeros_like(a)
        partner_vals[valid] = a[partner[valid]]
        contrib = a * partner_vals                     # a[i]·a[k_max‑i]
        src = int(np.argmax(contrib))                  # biggest offender
        if a[src] == 0.0:
            # nothing to move – break to avoid infinite loop
            break
        # choose a target where contribution is tiny (bottom 10 %)
        thresh = np.quantile(contrib, 0.10)
        low = np.where(contrib <= thresh)[0]
        if low.size == 0:
            tgt = rng.choice(np.delete(np.arange(n), src))
        else:
            # avoid picking src again
            low = low[low != src]
            if low.size == 0:
                tgt = rng.choice(np.delete(np.arange(n), src))
            else:
                tgt = int(rng.choice(low))
        delta = a[src] * rng.random() * 0.05  # up to 5 % of source mass
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] directed climb new best {best_score:.6f}")
        else:
            # occasional downhill move to escape tiny plateaux
            if cand_score < cur_score and rng.random() < 0.02:
                a = cand
                cur_score = cand_score
    return best_a, best_score


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence`` within the time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # small safety margin

    # ----------------------------------------------------------
    # Problem size – a trade‑off between expressiveness and speed
    # ----------------------------------------------------------
    n = 8192                     # a bit larger than the previous 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ----------------------------------------------------------
    # Global starter (height_sequence_1) if present
    # ----------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass  # ignore if the variable does not exist

    # ----------------------------------------------------------
    # Phase 1 – random Dirichlet draws + a few deterministic shapes
    # ----------------------------------------------------------
    dir_deadline = min(deadline, start_time + 30.0)
    if time.time() < dir_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=5000, deadline=dir_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

    # a small collection of simple monotone shapes
    shapes = []
    # increasing linear ramp
    ramp = np.arange(1, n + 1, dtype=float)
    shapes.append(ramp)
    # exponential rise
    exp_rise = np.exp(np.linspace(-5, 0, n))
    shapes.append(exp_rise)
    # decreasing cosine (half‑cosine window, monotone)
    half_cos = 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))
    shapes.append(half_cos)

    for shp in shapes:
        shp = np.maximum(shp, 0.0)
        shp = _project_simplex(shp)
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq, best_score = shp.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] deterministic shape score {sc:.6f}")

    # ----------------------------------------------------------
    # Phase 2 – pure sub‑gradient descent (multiple restarts)
    # ----------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.78   # keep ~22 % for final hill‑climb
        restarts = 6
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.02, size=n)          # modest random perturbation
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0_score = _objective_score(a0)

            sub_deadline = min(descent_deadline,
                               time.time() + (descent_deadline - time.time()) / (restarts - r))
            a_opt, sc_opt = _subgradient_descent(a0, a0_score, sub_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] restart {r} new best {best_score:.6f}")

    # ----------------------------------------------------------
    # Phase 3 – directed hill‑climbing for the last few seconds
    # ----------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting directed hill‑climbing")
        best_seq, best_score = _directed_hill_climb(best_seq, best_score, deadline)

    # ----------------------------------------------------------
    # Normalise and return
    # ----------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
