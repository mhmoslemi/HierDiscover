def _lp_improve(a, max_iter, deadline):
    a = np.copy(a)
    n = a.shape[0]
    best_score = _objective_score(a)
    best_a = a.copy()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = _conv_self(a)
        M = conv.max()
        # Build A_ub
        rows = []
        cols = []
        data = []
        for i in range(n):
            rows.extend(range(i, i + n))
            cols.extend([i] * n)
            data.extend([a[i]] * n)
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            # fallback to random perturbation
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # cannot increase sum - stagnated
            break
        g_norm = g / sum_g
        # line search for t
        for t in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            a_new = (1 - t) * a + t * g_norm
            sc = _objective_score(a_new)
            if sc < best_score - 1e-9:
                a = a_new
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] LP improve t={t:.4f} score={sc:.6f}")
                break
        else:
            # If no improvement, break loop
            break
    return best_a, best_score

"""
A fast, deterministic search for a step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  Generate many candidate vectors of modest size (n_opt = 1024) using
    simple parametric families and random Dirichlet draws.  The best
    candidate is kept.

2️⃣  Refine the best candidate with a linear‑programming “exchange”
    step (the classic Frank‑Wolfe style approach).  For a current vector
    a we compute M = max(conv(a,a)).  Solving

        maximise    Σ_j g_j
        subject to  a * g ≤ M          (convolution is linear in g)
                    g_j ≥ 0

    yields a vector g that carries more total mass while still respecting
    the upper‑bound M.  After normalising g to a unit‑sum we mix it with
    a (full replacement or a convex combination) and keep the mixture
    only if the objective improves.  Repeating these LP steps quickly
    drives the maximal convolution down.

3️⃣  The final vector of length n_opt is up‑sampled by simple repetition
    to the required length (4096).  Re‑sampling does not change the
    objective because every entry is divided by the repeat factor while
    the number of pairwise products grows exactly by the square of that
    factor.

All helper functions are top‑level, there are no closures, and the code
stays well within the 1000 s time budget.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Objective 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣  Simple parametric families (cheap sweep)
# ----------------------------------------------------------------------
def _quick_sweeps(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)

    # geometric decay
    for eps in np.logspace(-5, -1, 60):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        sc = _objective_score(geo)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} → {sc:.6f}")

    # symmetric exponential (distance from centre)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 50):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} → {sc:.6f}")

    # two‑block (mass at both ends)
    for L_frac in np.linspace(0.02, 0.3, 15):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in np.linspace(0.1, 0.9, 9):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} → {sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws
# ----------------------------------------------------------------------
def _dirichlet_trials(n, best_seq, best_score, deadline, trials=1500):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in [0.01, 5]
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  LP‑based improvement (Frank‑Wolfe style)
# ----------------------------------------------------------------------
def _lp_improve(a0, max_iter, deadline):
    """
    Repeatedly solve the LP
        maximise Σ g_j  subject to a * g ≤ M, g ≥ 0
    and mix the resulting g into a.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current peak
        M = _conv_self(a).max()

        # build sparse matrix for the convolution constraints a * g ≤ M
        rows = []
        cols = []
        data = []
        for i in range(n):
            rows.extend(range(i, i + n))
            cols.extend([i] * n)
            data.extend([a[i]] * n)
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # linear programme: maximise Σ g  ⇔ minimise −Σ g
        c = -np.ones(n)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            # LP failed – bail out
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # no mass increase possible → convergence
            break

        g_norm = g / sum_g   # renormalise to unit sum

        # try a few mixing rates, keep the first that improves the objective
        improved = False
        for t in (1.0, 0.5, 0.25, 0.125, 0.0625):
            a_new = (1.0 - t) * a + t * g_norm
            sc = _objective_score(a_new)
            if sc < best_score - 1e-9:
                a = a_new
                best_score = sc
                best_a = a.copy()
                improved = True
                print(f"[{time.time() - start_time:.2f}s] LP step t={t:.3f} → {sc:.6f}")
                break

        if not improved:
            # mixing did not help – stop
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises `evaluate_sequence`.  The routine runs a quick parametric
    sweep, many Dirichlet samples and finally a custom LP‑based
    refinement.  The best vector (normalised to sum = 1) is up‑sampled
    to length 4096 and returned.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # --------------------------------------------------------------
    # Work at a modest size – 1024 points are enough for a good shape.
    # --------------------------------------------------------------
    n_opt = 1024
    uniform = np.full(n_opt, 1.0 / n_opt)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional external starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            # truncate / pad to n_opt
            if starter.size > n_opt:
                starter = starter[:n_opt]
            elif starter.size < n_opt:
                # pad with zeros
                starter = np.concatenate([starter,
                                          np.zeros(n_opt - starter.size)])
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Quick parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 400.0   # keep plenty of time for LP step
    if time.time() < phase_deadline:
        best_seq, best_score = _quick_sweeps(n_opt, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Dirichlet sampling
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _dirichlet_trials(
            n_opt, best_seq, best_score, phase_deadline, trials=4000)

    # --------------------------------------------------------------
    # 3️⃣ LP‑based refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting LP refinement")
        # we allow up to 30 LP iterations (fast for n_opt=1024)
        best_seq, best_score = _lp_improve(best_seq, max_iter=30, deadline=deadline)
        print(f"[{time.time() - start_time:.2f}s] LP refinement finished – best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Upsample to the required length (4096) – simple repetition.
    # --------------------------------------------------------------
    target_len = 4096
    repeat_factor = target_len // n_opt
    # each entry is divided by repeat_factor to keep total sum = 1
    upsampled = np.repeat(best_seq / repeat_factor, repeat_factor)

    # final safety normalisation
    upsampled = np.maximum(upsampled, 0.0)
    total = upsampled.sum()
    if total > 0:
        upsampled = upsampled / total
    else:
        upsampled = np.full(target_len, 1.0 / target_len)

    final_score = _objective_score(upsampled)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score≈{final_score:.6f}")

    return [float(x) for x in upsampled]
