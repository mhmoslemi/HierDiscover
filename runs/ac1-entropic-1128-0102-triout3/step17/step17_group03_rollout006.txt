"""
Propose a candidate step‑function (non‑negative vector) that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The scale of a does not matter, so we keep the vector on the probability
simplex (sum = 1).  The search proceeds in three phases:

1️⃣  A fast diversified initialisation – uniform vector, simple analytic shapes
    (triangular, Hann window, exponential), a few sparse vectors and many
    Dirichlet draws.  The best among them seeds the optimisation.

2️⃣  Smoothed gradient descent on the soft‑max approximation of the ℓ∞‑norm of
    the self‑convolution.  At iteration *k* we form weights
    w ∝ exp(α·(conv−max(conv))) (α grows slowly) and compute the gradient
    via an FFT‑based convolution of the reversed vector with w.  The step size
    decays as 1/√k, the update is projected back onto the simplex by simple
    normalisation, and occasional random pairwise transfers help escape flat
    regions.

3️⃣  A cheap random local refinement that tries tiny mass transfers between
    random indices and keeps any strict improvement.

All heavy work (convolutions, FFTs) is done with NumPy; the routine respects
the 1000 s wall‑time budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (set in propose_candidate)
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑2 FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_rev_w(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of the reversed vector a_rev (length n) with a weight vector w
    (length 2·n−1).  Returns the gradient 2·(a_rev * w)[i+n−1] as a length‑n array.
    """
    n = a_rev.shape[0]
    L = _next_pow2(n + w.shape[0] - 1)          # ≥ 3·n‑2
    A = np.fft.rfft(a_rev, n=L)
    B = np.fft.rfft(w, n=L)
    conv = np.fft.irfft(A * B, n=L)[: n + w.shape[0] - 1]   # length 3·n‑2
    start = n - 1
    grad = 2.0 * conv[start : start + n]   # slice of length n
    return grad


def _objective_score(a: np.ndarray) -> float:
    """Objective value 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

    best_seq = np.full(n, 1.0 / n, dtype=float)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # deterministic analytic shapes
    # triangular (symmetric)
    tri = np.empty(n, dtype=float)
    half = (n + 1) // 2
    tri[:half] = np.arange(1, half + 1)
    tri[half:] = np.arange(half, 0, -1)[: n - half]
    tri /= tri.sum()
    for name, vec in [("Triangular", tri)]:
        sc = _objective_score(vec)
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] {name} start score = {sc:.6f}")

    # Hann (raised‑cosine) window – zeros at the ends are allowed
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    hann /= hann.sum()
    sc = _objective_score(hann)
    if sc < best_score:
        best_seq, best_score = hann.copy(), sc
        print(f"[{time.time() - start_time:.2f}s] Hann window start score = {sc:.6f}")

    # exponential decay
    lam = np.log(2.0) / (n / 10.0)      # half‑life at n/10
    expo = np.exp(-lam * np.arange(n))
    expo /= expo.sum()
    sc = _objective_score(expo)
    if sc < best_score:
        best_seq, best_score = expo.copy(), sc
        print(f"[{time.time() - start_time:.2f}s] Exponential start score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Random Dirichlet draws (≈30 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 30.0)
    max_dirichlet = 400
    while time.time() < dirichlet_deadline and max_dirichlet > 0:
        max_dirichlet -= 1
        conc_log = rng.uniform(-4.0, 0.5)          # from 10⁻⁴ to ≈3
        c = 10 ** conc_log
        if c <= 0:
            continue
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Sparse random start (≈1 % non‑zero)
    # ------------------------------------------------------------------
    sparsity_ratio = 0.01
    k = max(2, int(sparsity_ratio * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    sparse = np.zeros(n, dtype=float)
    sparse[idx] = vals
    if sparse.sum() > 0.0:
        sparse /= sparse.sum()
        sc = _objective_score(sparse)
        if sc < best_score:
            best_seq, best_score = sparse.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Sparse start score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent with soft‑max weights
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    best_a = a.copy()
    best_obj = best_score

    step_base = 0.5         # maximal per‑iteration change (relative)
    alpha = 5.0             # soft‑max temperature (will increase)
    iteration = 0

    while time.time() < deadline:
        # current convolution and objective
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # soft‑max weights focusing on the currently largest entries
        w = np.exp(alpha * (conv - max_conv))   # max element gets weight 1
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            break
        w /= w_sum

        # gradient of the smoothed objective
        a_rev = a[::-1]                         # reversed view
        grad = _conv_rev_w(a_rev, w)            # length n
        max_abs = np.max(np.abs(grad))
        if max_abs < 1e-15:
            break

        # step size decays as 1/√iteration, direction normalised by max_abs
        step = step_base / np.sqrt(iteration + 1.0)
        step_vec = step * (grad / max_abs)      # move opposite the gradient
        cand = a - step_vec
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = np.full(n, 1.0 / n)

        # evaluate candidate
        conv_cand = _conv_self(cand)
        cand_score = 2.0 * n * conv_cand.max()

        # accept only improvements (simple greedy descent)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_obj:
                best_obj = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] GD iter {iteration} new best {best_obj:.6f}")
        else:
            # no improvement → shrink the aggressive step base a bit
            step_base *= 0.9995

        # temperature annealing (gradually focus more on the true max)
        alpha = min(alpha * 1.001, 400.0)

        # occasional random perturbation to escape plateaus
        if iteration % 5000 == 0 and iteration > 0:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src != dst and a[src] > 1e-12:
                delta = a[src] * 0.01 * rng.random()
                pert = a.copy()
                pert[src] -= delta
                pert[dst] += delta
                conv_pert = _conv_self(pert)
                pert_score = 2.0 * n * conv_pert.max()
                if pert_score < best_obj:
                    best_obj = pert_score
                    best_a = pert.copy()
                    print(f"[{time.time() - start_time:.2f}s] Random perturb new best {best_obj:.6f}")
                if pert_score < cur_score:
                    a = pert
                    cur_score = pert_score

        iteration += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final cheap random local refinement (polishing)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        local_deadline = time.time() + time_left - 1.0
        a_ref = best_a.copy()
        best_ref_score = best_obj
        while time.time() < local_deadline:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a_ref[src] <= 1e-15:
                continue
            delta = a_ref[src] * 0.005 * rng.random()
            cand_ref = a_ref.copy()
            cand_ref[src] -= delta
            cand_ref[dst] += delta
            conv_cand_ref = _conv_self(cand_ref)
            cand_ref_score = 2.0 * n * conv_cand_ref.max()
            if cand_ref_score < best_ref_score:
                a_ref = cand_ref
                best_ref_score = cand_ref_score
                best_a = cand_ref.copy()
                best_obj = cand_ref_score
                print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_obj:.6f}")
    return [float(x) for x in best_a]
