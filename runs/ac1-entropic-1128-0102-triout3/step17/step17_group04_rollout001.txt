"""
Propose a step‑function (non‑negative list) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three scales:

1️⃣  **Coarse search on a smaller problem** (n≈1024).  
    •  A quick sweep of simple parametric families (geometric,
       exponential, two‑blocks, sinusoid, triangular).  
    •  Many random Dirichlet draws with a wide range of concentration
       parameters.  
    •  Two optimisation passes on the small problem:
        – a smooth‑max (log‑sum‑exp) descent with an exponential mirror‑step,
        – a hard‑max (sub‑gradient) descent that directly reduces the
          actual peak of the autocorrelation.  After every update the
          vector is forced to stay on the probability simplex and is
          symmetrised (a←(a+a[::-1])/2) because the optimal solution is
          expected to be symmetric.

2️⃣  **Upscaling** – the best small‑scale vector is repeated to the
   target length (4096).  The up‑scaled vector is normalised again.

3️⃣  **Refinement on the full size** – a short hard‑max descent on the
   large problem (the same mirror‑descent scheme) finishes the search.

The whole routine stops well before the 1000 s budget and returns the
best sequence found.  All steps use FFT‑based convolutions, so each
objective evaluation is cheap (≈O(n log n)).  Random jitter is added
whenever a step cannot improve the score, to escape plateaus.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing / RNG (will be initialised in propose_candidate)
# ----------------------------------------------------------------------
start_time = None          # set at the beginning of propose_candidate

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (returns full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Objective 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    The returned value matches evaluate_sequence(seq) for sum(seq)=1.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponential (mirror‑descent) update: a_i ← a_i·exp(−η·grad_i), then renormalise.
    Guarantees a_i ≥ 0 and Σ a_i = 1.
    """
    # work in log‑space
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)           # avoid overflow
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force the vector to be symmetric around the centre."""
    return (a + a[::-1]) * 0.5


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient) where gradient is ∂/∂a of the smooth value.
    """
    n = a.shape[0]
    conv = _conv_self(a)               # length 2n‑1
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # stability shift
    Z = exp_scaled.sum()
    w = exp_scaled / Z                         # soft‑max weights, sum to 1

    # Gradient: d / da ( Σ_k w_k conv_k ) = 2 * (a * w_rev) (convolution)
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)             # length 3n‑2
    start = len(w) - 1                         # = 2n‑2
    G = 2.0 * conv_rev[start - np.arange(n)]

    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, G

def _quick_sweeps(n, best_seq, best_score, deadline):
    """Cheap parametric families (geometric, exponential, 2‑block, sinusoid, triangular)."""
    idx = np.arange(n, dtype=np.float64)

    # 1. Geometric
    for eps in np.logspace(-5, -1, 80):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        sc = _objective_score(geo)
        if sc < best_score:

            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} score={sc:.6f}")

    # 2. Simple exponential (symmetric around centre)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 60):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")

    # 3. Two‑block at the ends
    for L_frac in np.linspace(0.02, 0.3, 20):

            seq[:L] = w / L

                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")

    sc = _objective_score(tri)
    if sc < best_score:
        best_score, best_seq = sc, tri.copy()
        print(f"[{time.time() - start_time:.2f}s] Triangular score={sc:.6f}")

    return best_seq, best_score


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=2000):
    """Draw many Dirichlet vectors with random concentration."""
    global _rng
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration parameter in roughly [0.01, 5] (log‑uniform)
        logc = _rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = _rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


def _smooth_descent(a0, deadline, best_seq, best_score):
    """Smooth‑max descent with exponential mirror‑updates (β grows to a large value)."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    beta_start, beta_end = 0.5, 120.0            # more aggressive schedule
    eta = 0.5
    min_eta = 1e-12

    while time.time() < deadline:
        # linear schedule for β
        t = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + t * (beta_end - beta_start)

        smooth, grad = _smooth_max_and_grad(a, beta)

        # backtracking line search on η
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, grad, eta_try)
            a_new = _symmetrize(a_new)
            a_new = a_new / a_new.sum()
            new_score = _objective_score(a_new)
            if new_score <= cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Smooth β={beta:.2f} η={eta_try:.2e} score={new_score:.6f}")
            eta = min(1.0, eta * 1.02)
        else:
            # small random jitter to escape stagnation
            eps = 1e-4 * _rng.random() * a
            a = a + eps
            a = np.clip(a, 0.0, None)
            a = _symmetrize(a)
            a = a / a.sum()
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.5)

        # early exit if we cross the target
        if best_score <= 1.503:
            break

    return best_seq, best_score


def _hard_descent(a0, deadline, best_seq, best_score):
    """Mirror‑descent on the true max‑convolution (sub‑gradient)."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    eta = 0.05
    min_eta = 1e-12

    while time.time() < deadline:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]          # derivative of the active peak

        # backtracking line search
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, grad, eta_try)
            a_new = _symmetrize(a_new)
            a_new = a_new / a_new.sum()
            new_score = _objective_score(a_new)
            if new_score <= cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hard k={kmax} η={eta_try:.2e} score={new_score:.6f}")
            eta = min(1.0, eta * 1.02)
        else:
            # jitter
            eps = 1e-4 * _rng.random() * a
            a = a + eps
            a = np.clip(a, 0.0, None)
            a = _symmetrize(a)
            a = a / a.sum()
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.5)

        if best_score <= 1.503:
            break

    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  The routine combines a cheap
    parametric sweep, massive random Dirichlet sampling, a smooth‑max
    descent, and a hard‑max sub‑gradient descent.  It first works on a
    reduced size (n≈1024) for speed, then upsamples to the target length
    (4096) and refines there.  The best sequence found before the time
    budget expires is returned.
    """
    global start_time, _rng
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin
    _rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we start on a smaller instance for speed
    # ------------------------------------------------------------------
    n_target = 4096
    n_small = 1024                     # small instance size (must divide n_target)

    # ------------------------------------------------------------------
    # Baseline: uniform distribution on the small instance
    # ------------------------------------------------------------------
    uniform_small = np.full(n_small, 1.0 / n_small)
    best_seq = uniform_small.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Small uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                # rescale to sum = 1 and to the small size (if needed)
                if starter.shape[0] != n_small:
                    # truncate or repeat to match n_small
                    if starter.shape[0] > n_small:
                        starter = starter[:n_small]
                    else:
                        # repeat to fill
                        repeats = n_small // starter.shape[0] + 1
                        starter = np.tile(starter, repeats)[:n_small]
                starter = starter / starter.sum()
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – quick parametric families
    # ------------------------------------------------------------------
    phase_deadline = deadline - 600.0   # reserve ~600 s for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _quick_sweeps(n_small, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – massive Dirichlet sampling
    # ------------------------------------------------------------------
    phase_deadline = deadline - 400.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n_small, best_seq, best_score, phase_deadline, trials=8000
        )

    # ------------------------------------------------------------------
    # Phase 3 – smooth‑max descent on the small instance
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent (small)")
        best_seq, best_score = _smooth_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent (small) best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – hard‑max (sub‑gradient) descent on the small instance
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hard‑max descent (small)")
        best_seq, best_score = _hard_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After hard descent (small) best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Upscale to the target size
    # ------------------------------------------------------------------
    repeat_factor = n_target // n_small
    seq_large = np.repeat(best_seq, repeat_factor)
    # If n_target is not a multiple (should be), pad or truncate
    if seq_large.shape[0] > n_target:
        seq_large = seq_large[:n_target]
    elif seq_large.shape[0] < n_target:
        seq_large = np.concatenate([seq_large, np.zeros(n_target - seq_large.shape[0])])
    seq_large = np.maximum(seq_large, 0.0)
    seq_large /= seq_large.sum()
    large_score = _objective_score(seq_large)
    if large_score < best_score:
        best_score, best_seq = large_score, seq_large.copy()
        print(f"[{time.time() - start_time:.2f}s] After up‑scaling score={large_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final hard‑max refinement on the full size
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hard‑max descent (large)")
        best_seq, best_score = _hard_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After final refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n_target, 1.0 / n_target)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
