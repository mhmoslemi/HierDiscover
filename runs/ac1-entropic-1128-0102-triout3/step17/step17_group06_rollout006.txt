"""
Propose a step‑function (non‑negative vector) that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases:

1. **Warm‑start** – sample a handful of Dirichlet draws and Beta(p,p)
   densities (including mixtures with the uniform vector) and keep the
   best one according to the exact score.

2. **Smoothed sub‑gradient descent** – the objective max(conv) is
   approximated by the smooth log‑sum‑exp trick:
        f(a) = ε·log Σₖ exp(convₖ/ε) .
   Its gradient is
        ∇f(a) = 2·corr(a, w)    with   wₖ ∝ exp(convₖ/ε) .
   Here `corr` is the (linear) cross‑correlation of the current vector
   with the weight vector `w`.  The descent step is performed with an
   exponentiated‑gradient (mirror‑descent with the KL‑divergence) which
   automatically preserves non‑negativity; after each step the vector is
   re‑normalised to sum to 1.

   An adaptive step size `η` and a decreasing smoothing parameter `ε`
   are used.  A candidate is accepted only if the true score
   `2·n·max(conv(a,a))` decreases; otherwise the step size is reduced.

3. **Finalisation** – the best vector found during the whole run is
   returned (clipped to non‑negative values and normalised).

All linear operations (convolutions and correlations) are carried out
with the FFT for speed.  The routine respects the supplied time budget
(~1000 s) and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised Beta(p,p) density on the grid x_i = (i+0.5)/n.
    Returns a normalised (sum = 1) non‑negative vector of length `n`.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_self_fft(a: np.ndarray, L: int, n: int) -> np.ndarray:
    """
    Full linear convolution a⋆a (length 2·n‑1) using FFT of length `L`.
    Returns the real array of length 2·n‑1.
    """
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)
    return conv[: 2 * n - 1]


def _corr_fft(a: np.ndarray, w: np.ndarray, L: int, n: int) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ w) for non‑negative lags 0…n‑1.
    Both inputs are zero‑padded to length `L` (L ≥ n + len(w) – 1).
    Returns an array of length `n` containing the correlation values
    for lags 0,…,n‑1.
    """
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(A * np.conj(W), n=L)
    return corr_full[:n]          # positive lags only


def _score_of(a: np.ndarray, n: int, L: int) -> float:
    """Exact score used by the evaluator (sum(a)=1)."""
    conv = _conv_self_fft(a, L, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    prints progress.  The returned list sums to ≈1 (scale‑invariant).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # --------------------------------------------------------------
    # Problem size (fixed for reproducibility)
    # --------------------------------------------------------------
    n = 8192                                 # a larger n gives a finer grid
    L_fft = _next_pow2(3 * n)                # ≥ (n) + (2n‑1) – 1 for correlation

    # --------------------------------------------------------------
    # 0️⃣ Baseline (uniform) and optional global starter
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_a = uniform.copy()
    best_score = _score_of(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score = {best_score:.6f}")

    # Try a globally supplied starter (height_sequence_1) if present
    try:

        grad = 2.0 * _corr_fft(a, w, L_fft, n)      # length n

        grad_centered = grad - grad.mean()

            cand_sum = cand.sum()
        cand = cand / cand_sum

            a = cand
            curr_score = cand_score
            accept_cnt += 1
            # record global best
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] New best score = {cand_score:.6f}")
            # modestly increase step‑size
            eta = min(eta * 1.05, eta_max)
        else:

            eta = max(eta * 0.5, eta_min)

        # ---- occasional adaptation of the smoothing factor ----
        it += 1
        if it % 1000 == 0:
            # If acceptance rate is low, increase smoothing; otherwise decrease
            if accept_cnt < 5:
                eps_factor = max(eps_factor * 0.7, 0.02)
            else:
                eps_factor = min(eps_factor * 1.1, 0.5)
            accept_cnt = 0

    # ------------------------------------------------------------------
    # 3️⃣ Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
