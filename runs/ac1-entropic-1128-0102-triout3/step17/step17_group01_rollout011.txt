    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n = 4096
    L_fft = _next_pow2(2*n-1)
    L_grad = _next_pow2(3*n-2)

    # baseline uniform
    uniform = np.full(n, 1.0/n)
    best_seq = uniform.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Prepare seeds: similarly as earlier: uniform, random dirichlet, parametric shapes etc.
    seeds = [...]

    # For each seed:
    for seed_idx, a0 in enumerate(seeds):
        if time.time() >= deadline - 5:
            break
        # Convert a0 to x0 via log
        eps = 1e-12
        x = np.log(a0 + eps)  # anyway shift later

        # Continue across betas
        for beta in [1,2,4,8,16,32,64,128,256,512]:
            if time.time() >= deadline - 5:
                break
            # define objective function closure that captures beta
            def f_and_grad(x_vec, beta=beta):
                # compute softmax a
                x_shift = x_vec - np.max(x_vec)
                exp_x = np.exp(x_shift)
                a = exp_x / exp_x.sum()
                conv = _conv_self_fft(a, L_fft)
                max_conv = conv.max()
                # compute softmax weights
                shift = max_conv
                exps = np.exp(beta * (conv - shift))
                sum_exps = exps.sum()
                p = exps / sum_exps
                # gradient w.r.t a (smooth)
                grad_a = 2.0 * n * _grad_softmax_new(a, p, L_grad)
                # gradient w.r.t x
                dot = np.dot(a, grad_a)
                grad_x = a * (grad_a - dot)
                f_val = 2.0 * n * (max_conv + np.log(sum_exps)/beta)
                return f_val, grad_x.astype(np.float64)

            # Use SciPy minimize L-BFGS-B
            options = {'maxiter': 200, 'disp': False}
            try:
                res = scipy.optimize.minimize(fun=lambda x: f_and_grad(x)[0],
                                              x0=x,
                                              jac=lambda x: f_and_grad(x)[1],
                                              method='L-BFGS-B',
                                              options=options)
            except Exception as e:
                # fallback: break
                break
            x = res.x
            # evaluate exact objective on candidate a = softmax(x)
            a_candidate = np.exp(x - np.max(x))
            a_candidate /= a_candidate.sum()
            score = _eval_fast(a_candidate, L_fft)
            if score < best_score:
                best_score = score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] seed {seed_idx} beta {beta} L-BFGS score {score:.6f}")

"""
Propose a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

by a three‑phase scheme:

1️⃣ **Diverse seeds** – uniform, global starter (if present), a few
   parametric families (geometric·power, beta‑shapes) and random
   Dirichlet draws.  The best K seeds are kept.

2️⃣ **Smooth‑max L‑BFGS continuation** – The objective is replaced by the
   smooth surrogate
        fβ(a)=2·n·β⁻¹·log Σ exp(β·conv(a,a)),
   whose gradient is computed analytically via FFTs.
   For a schedule of β∈{1,2,4,…,256} we run a short L‑BFGS‑B optimisation
   (max 80 iterations per β) starting from each seed.  The simplex constraint
   is enforced by the soft‑max re‑parameterisation a=softmax(x).

3️⃣ **Polish** – exponentiated‑gradient and projected sub‑gradient descent
   (as in the reference implementation) plus a final random hill‑climb.

All steps respect the 1000 s time budget and the best sequence found is
returned as a plain list (already normalised to sum = 1).
"""

import time
import numpy as np
import scipy.optimize

# Basic utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector.
    p : np.ndarray, shape (2·n‑1,)
        Soft‑max weights over the convolution entries.
    L : int
        Length of the zero‑padded FFT (power of two).

    Returns
    -------
    g : np.ndarray, shape (n,)
        Gradient ∂/∂a_i  Σ_k p_k conv_k .
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a positive vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑type shape: (i+1)^{α-1}·(n-i)^{β-1}, normalised.
    Allows symmetric or skewed U‑/∧‑shapes.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape

# Smooth‑max objective wrapped for L‑BFGS (soft‑max re‑parameterisation)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable object that evaluates the smooth‑max surrogate and its gradient.
    The variables are the unconstrained logits `x`; the probability vector is
    obtained by a soft‑max:  a = softmax(x).  The surrogate is

        fβ(a) = 2·n·β⁻¹·log Σ_k exp(β·conv_k(a))

    which is differentiable for any β>0.
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        """Stable soft‑max conversion."""
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        """
        Compute (fβ(a), ∇_x fβ(a)) at the point x.

        Returns
        -------
        f_val : float
            Smooth‑max surrogate value.
        grad_x : np.ndarray, shape (n,)
            Gradient w.r.t. the logits x.
        """
        a = self._softmax_logits(x)

        # convolution a*a (full linear)
        conv = _conv_self_fft(a, self.L_fft)

        # smooth‑max weights (soft‑max over β·conv)
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a of the smooth‑max term (unnormalised)
        # grad_a = 2·n·∇_a log‑sum‑exp = 2·n·_grad_softmax_new(a, p)
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule: ∂f/∂x_i = a_i·(∂f/∂a_i - Σ_j a_j ∂f/∂a_j)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # smooth‑max surrogate value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)

# Polishing phases (same as the reference implementation)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates (exponentiated‑gradient) on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10  # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 provides a good speed/quality trade‑off.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1 if present).
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Generate a rich pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [best_seq]  # already contains uniform / starter

    # Parametric geometric×power sweep
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9995), 9)
    p_vals = np.linspace(-4.0, 3.0, 15)
    lam_vals = [0.0, 0.2, 0.5, 0.8]
    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Discrete Beta‑type shapes (incl. U‑shapes and skewed shapes)
    alpha_vals = [0.5, 1.0, 2.0, 3.0]
    beta_vals = [0.5, 1.0, 2.0, 3.0]
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # Random Dirichlet draws with various concentrations
    for alpha in [0.2, 0.5, 1.0]:
        for _ in range(5):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # Keep the K best seeds after cheap evaluation.
    # ------------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 6
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Continuation with L‑BFGS on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256]

    # Number of L‑BFGS iterations per β (tuned for the time budget)
    max_iter_per_beta = 80

    for beta in beta_schedule:
        if time.time() >= deadline - 300:  # reserve time for polishing
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
        for idx in range(len(pool)):
            if time.time() >= deadline - 300:
                break
            a0 = pool[idx]
            # initialise logits from the current probability vector
            eps = 1e-12
            x0 = np.log(a0 + eps)

            # L‑BFGS‑B optimisation (joint objective+gradient)
            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per_beta,
                                                   'disp': False})
            # obtain the new probability vector
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()

            cand_score = _eval_fast(a_candidate, L_fft)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} score={cand_score:.6f}")

            # keep the updated seed for possible further β‑steps
            pool[idx] = a_candidate
            pool_scores[idx] = cand_score

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 120, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 70:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 70, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣  Final random hill‑climb (tiny mass transfers).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 4e-3  # ≤0.4 % of source mass
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
