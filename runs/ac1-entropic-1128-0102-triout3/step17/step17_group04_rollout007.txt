def _subgrad_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    # set learning rate schedule
    eta = 0.05

    while time.time() < deadline:
        conv = _conv_self(a)
        M = conv.max()
        # find max indices (within tolerance)
        eps = max(1e-12, M * 1e-12) # small relative tolerance
        K = np.where(conv >= M - eps)[0]
        # build indicator
        w = np.zeros_like(conv)
        w[K] = 1.0 / len(K)
        # compute gradient via convolution with reversed w
        rev_w = w[::-1]
        conv_rev = _conv_fft(a, rev_w) # length 3n-2
        shift = len(rev_w) - 1
        grad = 2.0 * conv_rev[shift : shift + n]  # shape n

        # option: we could also add regularization to gradients causing stable step
        # usual mirror descent IP update
        eta_try = eta
        improved = False
        while eta_try > 1e-6:
            # exponential update
            a_new = _exp_update(a, grad, eta_try)
            new_score = _objective_score(a_new)
            if new_score < cur_score:
                improved = True
                break
            eta_try *= 0.5
        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Subgrad descent improved score={new_score:.6f}")
            # increase eta gently
            eta = min(1.0, eta * 1.05)
        else:
            # no improvement, try random perturbation
            rng = np.random.default_rng()
            # choose two random indices i, j; move small amount
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = min(a[i], 0.01) * rng.random()
                a[i] -= delta
                a[j] += delta
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                cur_score = _objective_score(a)
            # reduce eta
            eta = max(1e-6, eta * 0.5)

        # optional early exit if best score < target
        if best_score <= 1.503:
            break
    return best_seq, best_score

"""
Hybrid parametric sweep + exact subgradient descent for the
max‑self‑convolution objective.

The algorithm:
1️⃣  Scan a few low‑dimensional families (geometric,
    exponential, triangular, cosine, two‑block) and keep the
    best shape.
2️⃣  Perform many random Dirichlet restarts.  Each restart is
    refined by a mirror‑descent step using the exact sub‑gradient
    of  f(a)=2·n·max(conv(a,a))/‖a‖₁² .  The sub‑gradient is
    obtained via a single FFT‑based convolution with the
    indicator of the active (max‑tight) constraints.
3️⃣  A final long descent from the best candidate.
All updates stay on the probability simplex, guaranteeing
non‑negativity and Σa_i=1.  The routine respects the 1000 s budget
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (returns full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Objective 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Mirror‑descent (exponential) update and renormalisation."""
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)  # shift for numerical stability
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total

# ----------------------------------------------------------------------
def _quick_sweeps(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)

    # geometric (single‑parameter)
    eps_grid = np.logspace(-6, -1, 100)
    for eps in eps_grid:

        geo = r ** idx
        geo /= geo.sum()
        sc = _objective_score(geo)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} score={sc:.6f}")

    # symmetric exponential (distance from centre)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-5, -0.1, 80):
        if time.time() >= deadline:

        if sc < best_score:

            print(f"[{time.time() - start_time:.2f}s] Exp lam={lam:.1e} score={sc:.6f}")

    # triangular family (base shape = 1 - |2x-1|)

# 2️⃣  Exact sub‑gradient descent (mirror‑descent on the simplex)
# ----------------------------------------------------------------------
def _subgrad_descent(a0, deadline, best_seq, best_score):
    """
    Mirror‑descent using the exact sub‑gradient of the max‑convolution
    objective.  a0 must be a probability vector (sum=1, non‑negative).
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    eta = 0.05  # initial step size
    min_eta = 1e-7

    while time.time() < deadline:
        # compute self‑convolution and identify active (max) indices
        conv = _conv_self(a)
        M = conv.max()
        # tolerance to collect all binding constraints
        tol = max(1e-12, M * 1e-12)
        K = np.where(conv >= M - tol)[0]
        # build indicator of active constraints (uniform weight)
        w = np.zeros_like(conv)
        w[K] = 1.0 / len(K)

        # gradient of the max term: 2 * (a * w) (cross‑correlation)
        rev_w = w[::-1]                     # reverse for correlation
        conv_rev = _conv_fft(a, rev_w)      # length 3n‑2
        shift = len(rev_w) - 1              # = 2n‑2
        grad = 2.0 * conv_rev[shift:shift + n]  # shape (n,)

        # try an exponential update with back‑tracking line search
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, grad, eta_try)
            new_score = _objective_score(a_new)
            if new_score < cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Subgrad improved score={new_score:.6f}")
            # slowly increase step size when progress is made
            eta = min(0.5, eta * 1.02)
        else:
            # no improvement – inject a tiny random jitter and keep trying
            rng = np.random.default_rng()
            eps = 1e-4 * rng.random() * a
            a = a + eps
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.5)

        # optional early stop
        if best_score <= 1.503:
            break

    return best_seq, best_score

        if best_score <= 1.503:

The algorithm:
1️⃣  Scan a few low‑dimensional families (geometric,

    best shape.
2️⃣  Perform many random Dirichlet restarts.  Each restart is

    obtained via a single FFT‑based convolution with the

3️⃣  A final long descent from the best candidate.
All updates stay on the probability simplex, guaranteeing
non‑negativity and Σa_i=1.  The routine respects the 1000 s budget
and prints progress information.

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (returns full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Objective 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Mirror‑descent (exponential) update and renormalisation."""
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)          # shift for stability
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total

def _subgrad_descent(a0, deadline, best_seq, best_score):
