"""
Hybrid local‑search for the step‑function bound

* start from the global starter (if present) or from a uniform vector;
* perform a light parametric sweep (monotone exponential / geometric) to
  get a good foothold;
* run many independent **sub‑gradient descent** runs on the exact
  max‑convolution (no smooth approximation).  The gradient of the
  active convolution mode is computed by a single FFT; a back‑tracking
  line search together with a Euclidean projection onto the probability
  simplex guarantees feasibility;
* when an iteration fails to improve, a tiny random jitter is added to
  escape flat plateaus;
* each run is limited by a short time‑budget (≈10 s) and the best
  sequence over all runs is kept.

The routine respects the 1000 s total budget, prints progress stamps,
and finally returns a plain list of non‑negative floats whose sum is
exactly 1.  The returned sequence is therefore ready for
``evaluate_sequence`` without further processing.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, simplex projection, …
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a fast FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Objective 2·n·max(conv(seq,seq)) for a sequence normalised to sum = 1.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    (Implementation of https://arxiv.org/abs/1309.1541)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# 1️⃣ Light parametric sweeps (monotone exponential / geometric)
# ----------------------------------------------------------------------
def _sweep_mono_exponential(n, best_seq, best_score, deadline):
    """a_i ∝ exp(-λ·i)  (strictly decreasing)."""
    idx = np.arange(n, dtype=np.float64)
    for lam in np.logspace(-4.5, -0.2, 120):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * idx)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] MonoExp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_mono_geometric(n, best_seq, best_score, deadline):
    """a_i ∝ (1-ε)^i  (geometric decay)."""
    idx = np.arange(n, dtype=np.float64)
    for eps in np.logspace(-5, -1, 120):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        seq = np.power(r, idx)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] MonoGeo ε={eps:.1e} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0, deadline, best_seq, best_score):
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Uses the true sub‑gradient (indicator of the active shift(s)),
    back‑tracking line search and a tiny random jitter when stuck.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1

        # current objective & active set
        conv = _conv_self(a)
        M = conv.max()
        cur_obj = 2.0 * n * M

        # record global best
        if cur_obj < best_score:
            best_score, best_seq = cur_obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} obj={cur_obj:.6f}")

        # build indicator of all maximising shifts (numerical tolerance)
        mask = conv >= M - 1e-14
        p = mask.astype(np.float64)               # shape (2n‑1,)

        # gradient = 2·(p * a_rev) (convolution with reversed a)
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        grad_full = np.fft.irfft(np.fft.rfft(p, n=L) *
                                 np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]   # central window, length n

        # ----- back‑tracking line search on the true objective -----
        step = 1.0
        improved = False
        while step > 1e-13:
            a_new = a - step * grad
            a_new = _project_simplex(a_new)       # stay on the simplex
            obj_new = _objective_score(a_new)
            if obj_new < cur_obj - 1e-12:
                a = a_new
                improved = True
                break
            step *= 0.5

        if not improved:
            # tiny random jiggle to leave potential plateaus
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

    return best_seq, best_score

    rng = np.random.default_rng(seed + 1)
    while time.time() < deadline:
        # pick a starting point:
        if rng.random() < 0.4:          # fresh random Dirichlet
            logc = rng.uniform(-2.0, 0.6)   # concentration between ~0.01 and ~4
            c = 10 ** logc
            a0 = rng.gamma(shape=c, scale=1.0, size=n)
            a0 /= a0.sum()
        else:                           # perturb the current best
            a0 = best_seq.copy()
            eps = rng.uniform(1e-4, 2e-3)
            a0 += rng.normal(scale=eps, size=n)
            a0 = np.clip(a0, 0.0, None)
            a0 = _project_simplex(a0)

        # each run gets up to 12 s (or whatever is left)
        sub_deadline = min(deadline, time.time() + 12.0)
        best_seq, best_score = _subgradient_descent(a0, sub_deadline,
                                                    best_seq, best_score)

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
