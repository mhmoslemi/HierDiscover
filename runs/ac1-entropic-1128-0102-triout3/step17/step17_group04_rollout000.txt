"""
Propose a step‑function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search routine works in four phases:

1️⃣  Very cheap parametric sweeps (geometric, exponential, two‑block).
2️⃣  Random Dirichlet draws – many candidates are generated and the best
    is kept.
3️⃣  Smooth‑max gradient descent on the simplex.  A log‑sum‑exp
    approximation of the max‐convolution is minimised; the gradient is
    obtained analytically via FFTs.  An exponential (mirror‑descent)
    update keeps the iterate on the simplex, a symmetric projection
    (a ↦ (a+a[::-1])/2) is applied after every successful step and the
    smoothing parameter β is increased from 0.5 up to 200.
4️⃣  Gradient‑guided hill‑climbing: using the current gradient we move a
    small amount of mass from the most “harmful’’ coordinate (largest
    gradient) to the most “helpful’’ one (smallest gradient).

Timing checks guarantee termination before the provided budget, and the
best sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_normed(a: np.ndarray) -> float:
    """
    Return the objective 2·n·max(conv(a,a)) assuming sum(a)=1.
    (The returned value equals what evaluate_sequence would give for a.)
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient) where gradient is with respect to a.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1

    # soft‑max weights over convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # shift for stability
    Z = exp_scaled.sum()
    w = exp_scaled / Z                         # weights, sum to 1

    # gradient: d/d a_i (∑_k w_k * conv_k) = 2 * Σ_j a_j * w_{i+j}
    # implement via convolution with reversed weights
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)          # length 3n‑2
    start = len(w) - 1                      # = 2n‑2
    G = 2.0 * conv_rev[start - np.arange(n)]

    # smooth max value = (log Σ exp(beta·conv)) / beta
    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, G


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponential (mirror‑descent) update: a_i ← a_i·exp(−η·grad_i), then renormalise.
    Guarantees a_i ≥ 0 and Σ a_i = 1.
    """
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)      # shift for numerical stability
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total


def _symmetrise(a: np.ndarray) -> np.ndarray:
    """Force mirror symmetry a_i = a_{n-1-i}."""
    return 0.5 * (a + a[::-1])


# ----------------------------------------------------------------------
# 1️⃣  Simple parametric sweeps (very light)
# ----------------------------------------------------------------------
def _quick_sweeps(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)

    # geometric (single‑parameter)
    for eps in np.logspace(-5, -1, 60):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        sc = _objective_normed(geo)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} score={sc:.6f}")

    # simple exponential (symmetric)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 50):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective_normed(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")

    # two‑block (ends)
    for L_frac in np.linspace(0.01, 0.25, 15):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in np.linspace(0.1, 0.9, 9):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            seq = _symmetrise(seq)
            sc = _objective_normed(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")

    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=800):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration in roughly [0.01, 5] (log‑uniform)
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        a = _symmetrise(a)
        sc = _objective_normed(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smooth‑max gradient descent (core optimiser)
# ----------------------------------------------------------------------
def _smooth_descent(a0, best_seq, best_score, deadline):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_normed(a)

    beta_start, beta_end = 0.5, 200.0
    eta = 5.0           # starting step size for exponential update
    min_eta = 1e-12
    max_eta = 50.0

    start_descent = time.time()
    total_descent = deadline - start_descent

    while time.time() < deadline:
        # linear schedule for β
        t = (time.time() - start_descent) / total_descent
        beta = beta_start + t * (beta_end - beta_start)

        smooth, grad = _smooth_max_and_grad(a, beta)

        # Try exponential (mirror‑descent) update with back‑tracking
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, grad, eta_try)
            a_new = _symmetrise(a_new)
            new_score = _objective_normed(a_new)
            if new_score <= cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.2f} η={eta_try:.2e} score={new_score:.6f}")
            # gently increase η for the next iteration
            eta = min(max_eta, eta * 1.02)
        else:
            # If no improvement, fall back to a tiny Euclidean step
            # (gradient descent with projection) – useful near plateaus.
            a_tmp = a - eta * grad
            a_tmp = _project_simplex(a_tmp)
            a_tmp = _symmetrise(a_tmp)
            tmp_score = _objective_normed(a_tmp)
            if tmp_score < cur_score:
                a = a_tmp
                cur_score = tmp_score
                if tmp_score < best_score:
                    best_score, best_seq = tmp_score, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Euclid step β={beta:.2f} η={eta:.2e} score={tmp_score:.6f}")
                eta = min(max_eta, eta * 1.02)
                continue

            # no improvement – inject a very small random jitter
            rng = np.random.default_rng()
            eps = 1e-5 * rng.random(size=n) * a
            a = a + eps
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            a = _symmetrise(a)
            cur_score = _objective_normed(a)
            eta = max(min_eta, eta * 0.5)

    return best_seq, best_score

# 4️⃣  Gradient‑guided hill‑climbing
# ----------------------------------------------------------------------
def _hill_climb(a0, best_seq, best_score, deadline):
    a = a0.copy()
    n = a.shape[0]

    while time.time() < deadline:
        # use smooth‑max at a very large β to approximate the true subgradient
        smooth, grad = _smooth_max_and_grad(a, beta=300.0)

        # indices with largest positive / negative gradient
        i_max = np.argmax(grad)
        i_min = np.argmin(grad)
        if i_max == i_min:
            break

        # move a small fraction of mass from i_max to i_min
        eps = min(a[i_max] * 0.05, 1e-5)
        if eps <= 0:
            break
        a_new = a.copy()
        a_new[i_max] -= eps
        a_new[i_min] += eps
        a_new = _symmetrise(a_new)

        new_score = _objective_normed(a_new)
        if new_score < best_score:
            best_score, best_seq = new_score, a_new.copy()
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb move {i_max}->{i_min} eps={eps:.2e} score={new_score:.6f}")
        else:
            # occasional random jitter to escape tiny plateaus
            rng = np.random.default_rng()
            jitter = 1e-6 * rng.random(size=n) * a
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            a = _symmetrise(a)

    return best_seq, best_score

    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
