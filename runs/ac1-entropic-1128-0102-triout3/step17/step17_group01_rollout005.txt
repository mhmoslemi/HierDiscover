"""
Evolutionary + gradient polishing search for a step function minimising
    2 * n * max(conv(a,a)) / (sum(a)**2).

The algorithm proceeds in three stages:
1️⃣  Generate a diverse seed pool (uniform, geometric×power families,
    sine, random Dirichlet).  Keep the K best.
2️⃣  Refine each seed with a short exponentiated‑gradient (multiplicative)
    phase and add them to a small population.
3️⃣  Run an evolutionary loop (log‑space mutation + occasional crossover)
    with a simple simulated‑annealing acceptance rule.  The current global
    best is periodically polished by a longer exponentiated‑gradient phase
    and a final projected sub‑gradient descent.

All calculations use FFT‑based convolution for speed; the objective is
evaluated with a custom fast routine (`_eval_fast`).  The final vector is
normalised to sum = 1 and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Fast objective
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families (seed generation)
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _sine_shape(n: int) -> np.ndarray:
    """Sine‑based shape a_i ∝ sin(π·(i+1)/(n+1)), normalised."""
    i = np.arange(n, dtype=np.float64) + 1.0
    shape = np.sin(np.pi * i / (n + 1.0))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Gradient / sub‑gradient helpers
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multiplicative updates (exponentiated gradient) on the true L∞ objective.
    Stops when deadline is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        # occasional jitter to avoid stagnation
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the L∞ objective with back‑tracking.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill acceptance (simulated‑annealing style)
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score

    K = 10
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Short exponentiated‑gradient polishing of each pool entry.
    # ------------------------------------------------------------------
    # Give each seed a few cheap multiplier steps (no more than ~30 s in total).
    polish_deadline = start + min(60.0, budget_s * 0.05)  # at most 5 % of total budget
    for idx in range(len(pool)):
        if time.time() > polish_deadline:
            break
        a = pool[idx]
        a, sc = _expgrad_phase(
            a, L_fft,
            deadline=time.time() + 5.0,   # tiny inner deadline per seed
            start_time=start,
            best_seq=a,
            best_score=_eval_fast(a, L_fft),
            rng=rng
        )
        pool[idx] = a
        pool_scores[idx] = sc
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start:.2f}s] post‑seed polish score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Evolutionary search (mutation + occasional crossover).
    # ------------------------------------------------------------------
    population = pool          # size = K (≈10)
    pop_scores = pool_scores   # parallel list of scores
    pop_size = len(population)

    # Simulated‑annealing style temperature for mutation magnitude.
    T0 = 0.20                # initial temperature (fraction of source mass)
    Tmin = 0.001
    total_evol_time = deadline - start - 180.0  # reserve ~3 min for final polishing
    evolution_deadline = start + total_evol_time
    it = 0
    while time.time() < evolution_deadline:
        it += 1
        # --------------------------------------------------------------
        # Select a parent via tournament (2‑way)
        # --------------------------------------------------------------
        i = rng.integers(pop_size)
        j = rng.integers(pop_size)
        parent = population[i] if pop_scores[i] < pop_scores[j] else population[j]

        # --------------------------------------------------------------
        # Mutation in log‑space (multiplicative noise)
        # --------------------------------------------------------------
        # Temperature decays roughly linearly with remaining time.
        frac = max(0.0, (evolution_deadline - time.time()) / total_evol_time)
        T = Tmin + (T0 - Tmin) * frac
        sigma = T            # std‑dev of Gaussian noise in log‑space

        log_parent = np.log(parent + 1e-12)   # avoid log(0)
        noise = rng.normal(scale=sigma, size=n)
        child = np.exp(log_parent + noise)
        child = child / child.sum()

        # --------------------------------------------------------------
        # Occasionally try a simple crossover (average of two parents)
        # --------------------------------------------------------------
        if rng.random() < 0.05:
            p2 = population[rng.integers(pop_size)]
            child = 0.5 * (parent + p2)
            # add mild log‑noise to avoid exact duplication
            child = child * np.exp(rng.normal(scale=sigma * 0.3, size=n))
            child = child / child.sum()

        # --------------------------------------------------------------
        # Evaluate child and possibly insert into population
        # --------------------------------------------------------------
        child_score = _eval_fast(child, L_fft)

        # Replace worst individual if child is better
        worst_idx = int(np.argmax(pop_scores))
        if child_score < pop_scores[worst_idx]:
            population[worst_idx] = child
            pop_scores[worst_idx] = child_score

        # Update global best
        if child_score < best_score:
            best_score = child_score
            best_seq = child.copy()
            print(f"[{time.time() - start:.2f}s] New global best it={it:,} score={best_score:.6f}")

        # --------------------------------------------------------------
        # Periodic intensive polishing of the current global best
        # --------------------------------------------------------------
        if it % 3000 == 0 and time.time() < deadline - 120:
            # Give the best a short exp‑grad burst (≈30 s)
            best_seq, best_score = _expgrad_phase(
                best_seq, L_fft,
                deadline=time.time() + 30.0,
                start_time=start,
                best_seq=best_seq,
                best_score=best_score,
                rng=rng
            )
            # Refresh population entry for the best (if present)
            # (ensures the best is not lost by later replacements)
            # No explicit insertion needed – the best is already saved.

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing: exponentiated‑gradient + sub‑gradient descent.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 80.0:
        # 60 s for a deeper exp‑grad phase
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft,
            deadline=time.time() + 60.0,
            start_time=start,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng
        )
        remaining = deadline - time.time()

    if remaining > 30.0:
        # remainder for projected sub‑gradient descent
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft,
            deadline=deadline,
            start_time=start,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final tiny random hill‑climb (mass transfers) – very cheap.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to 0.4 % of the source mass (as in the reference)
        eps = best_seq[src] * rng.random() * 4e-3
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
