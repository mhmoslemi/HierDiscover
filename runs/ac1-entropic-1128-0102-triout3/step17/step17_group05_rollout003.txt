"""
Propose a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three stages:

1️⃣  Very fast parametric sweeps (geometric, power, exponential, two‑block)
    to obtain decent starting points.

2️⃣  Random Dirichlet draws – cheap global exploration of the simplex.

3️⃣  Convex sub‑gradient descent on the true objective.
    The sub‑gradient of   max_i (a*a)_i   is obtained analytically,
    a simple mirror‑descent (multiplicative) update on the probability
    simplex is applied and the step‑size decays as η_k∝1/√k.
    This gives a genuine descent direction (no temperature smoothing) and
    converges to the global optimum of the convex problem.

All heavy work is performed with FFT‑based convolutions; the routine
automatically respects the 1000 s CPU budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a fast real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    (The true evaluate_sequence multiplies by 1/(sum(seq)**2) but the
    expression is homogeneous, so fixing sum=1 is sufficient.)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Adapted from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws (cheap global search)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration parameter: log‑uniform between 10⁻² and 5
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Exact sub‑gradient descent (mirror‑descent on the simplex)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float):
    """
    Projected mirror‑descent using the exact sub‑gradient of
    max(conv(a,a)).  Step size η_k = η0 / sqrt(k).
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.5                     # initial step‑size (tuned empirically)

    while time.time() < deadline:
        it += 1
        # current convolution and objective
        conv = _conv_self(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val

        # record improvement if any
        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} obj={obj:.6f} η={eta0/np.sqrt(it):.2e}")

        # compute exact sub‑gradient of the max‑convolution term
        # active indices (allow a tiny relative tolerance)
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]
        m = len(active)

        # gradient w.r.t. a (vectorised slice updates)
        grad = np.zeros_like(a)
        for i in active:
            lo = i - (n - 1)
            if lo < 0:
                lo = 0
            hi = i if i < n else n - 1
            # slice j = lo..hi ; corresponding k = i-j
            j = np.arange(lo, hi + 1)
            k = i - j
            grad[lo:hi + 1] += a[k]
        grad = 2.0 * grad / m    # average over the active set

        # mirror‑descent update (multiplicative, stays on the simplex after renorm)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        total = a.sum()
        if total == 0.0:                 # numerical safeguard
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises the
    evaluation function.  The routine runs a quick parametric sweep,
    a large random Dirichlet exploration and a deterministic sub‑gradient
    descent on the simplex.  The best sequence (normalised to sum = 1) is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Quick parametric sweep (~150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈200 s, as many as possible)
    # --------------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Exact sub‑gradient mirror descent (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting exact sub‑gradient descent")
        best_seq, best_score = _subgradient_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End of descent, best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
