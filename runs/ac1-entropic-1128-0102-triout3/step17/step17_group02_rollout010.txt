def build_conv_matrix(f):
    n = len(f)
    m = 2*n - 1
    A = np.zeros((m, n))
    # for each k, compute start and end indices
    for k in range(m):
        i_start = max(0, k - (n-1))       # minimal i such that k-i < n => i > k-(n-1)
        i_end = min(k, n-1)               # maximal i such that k-i >=0 => i <= k
        # indices j = i, corresponding coefficient f_{k-i}
        # so for j from i_start to i_end inclusive:
        A[k, i_start:i_end+1] = f[k - np.arange(i_start, i_end+1)]
```

def _lp_refinement(a0, deadline, max_iter=5, t=0.2):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    n = len(a)
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv_self = _conv_self(a)
        M = conv_self.max()
        # Build sparse matrix A
        offsets = np.arange(-n+1, 1)   # Actually we want -i for i from 0 to n-1 => offsets negative
        # Offsets list: -np.arange(n)
        offsets = -np.arange(n)
        data = [np.full(n, a[i]) for i in range(n)]
        A = scipy.sparse.diags(data, offsets, shape=(2*n-1, n), format='csr')
        # Solve LP maximize sum(g) s.t. A * g <= M, g >= 0
        c = -np.ones(n)  # minimize -sum(g)
        b = np.full(2*n-1, M)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("LP failed")
            break
        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 1e-12:
            break
        g_norm = g0 / sum_g
        # Mix
        new = (1 - t) * a + t * g_norm
        # Ensure non-negative, normalize
        new = np.maximum(new, 0.0)
        new /= new.sum()
        new_score = _objective_score(new)
        if new_score < best_score:
            best_score = new_score
            best_a = new.copy()
            a = new
            print(f"[{time.time()-start_time:.2f}s] LP mix improved to {best_score:.6f}")
        else:
            # if no improvement, reduce mixing fraction
            t *= 0.5
            if t < 1e-4:
                break
    return best_a, best_score
```

def _lp_refinement(a0, deadline, max_iter=4, mix_factor=0.2):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute current max conv
        conv_self = _conv_self(a)
        M = conv_self.max()

        # build sparse convolution matrix A (size (2n-1, n))
        # row indices: i + j ; column indices: j
        idx = np.arange(n, dtype=np.int64)
        row_idx = np.repeat(idx, n) + np.tile(idx, n)   # shape (n*n,)
        col_idx = np.tile(idx, n)
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2*n - 1, n))

        # solve LP: maximize sum(g) --> minimize -sum(g)
        c = -np.ones(n)
        b = np.full(2*n - 1, M)
        # use HiGHS
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[{time.time()-start_time:.2f}s] LP failed: {res.message}")
            break
        g0 = res.x
        S = g0.sum()
        if S <= 1e-12:
            print(f"[{time.time()-start_time:.2f}s] LP returned near-zero sum")
            break
        g_norm = g0 / S

        # Mixing
        new = (1 - mix_factor) * a + mix_factor * g_norm
        new = np.maximum(new, 0.0)
        new /= new.sum()
        new_score = _objective_score(new)
        if new_score < best_score:
            best_score = new_score
            best_a = new.copy()
            a = new
            print(f"[{time.time()-start_time:.2f}s] LP mix improved to {best_score:.6f}")
        else:
            # reduce mixing factor
            mix_factor *= 0.5
            if mix_factor < 1e-4:
                break
    return best_a, best_score

"""
Hybrid optimizer for the self‑convolution objective

Goal: minimise  evaluate_sequence(seq)  = 2·n·max(conv(seq,seq)) / (sum(seq)^2)
All vectors are normalised to sum = 1, so the denominator is fixed and the
objective reduces to 2·n·max(conv).

The search consists of four phases, each respecting a global 1000 s budget:

1️⃣ Load a known good construction (height_sequence_1) or fall back to a
   uniform start.  A quick sweep of random Dirichlet draws and a few
   deterministic windows (sine, Hann, triangular) is performed.

2️⃣ Smooth projected‑gradient descent using a log‑sum‑exp (temperature β)
   smoothing of the max.  The schedule β = [0.5,2,5,20,80,200,800,2000,5000]
   walks the landscape from gentle to sharp.  A Nesterov‑style momentum
   and an adaptive step size are employed.

3️⃣ Linear‑program refinement:  for the current vector f compute M = max(f*f).
   Solve the LP   maximise 1ᵀg   subject to   f * g ≤ M  and g ≥ 0.
   The solution g₀ typically has a sum > 1; after normalising
   g = g₀/∑g₀ we replace f ← (1‑t)·f + t·g (t≈0.3) and repeat a few times.
   The LP matrix is built as a sparse Toeplitz convolution matrix, so the
   solve scales to n ≈ 4096.

4️⃣ A short random “mass‑transfer’’ hill‑climb to polish the final answer.

All phases stop as soon as the time budget expires, and the best vector
found is returned as a plain list of floats.
"""

import scipy.sparse as sp

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (vectors are assumed to sum to 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation to max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Uses cross‑correlation of a_rev with the softmax weights w.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()
    rev_corr = _convolve_fft(a[::-1], w)          # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Random Dirichlet initialisation
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Sample Dirichlet vectors and keep the best (lowest objective)."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5] (≈10⁻³ … 10⁰·⁷)
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10.0 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth projected‑gradient descent (β‑schedule)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray, beta_vals, deadline: float):
    """
    Gradient descent on the smooth approximation of max‑convolution.
    Nesterov‑style momentum (µ=0.9) is used.
    """
    a = a0.copy()
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score

    for beta in beta_vals:
        step = 0.5
        momentum = np.zeros_like(a)
        mu = 0.9
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            # Nesterov update
            momentum = mu * momentum - step * grad
            cand = a + momentum
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)

            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time()-start_time:.2f}s] β={beta:.1f} new best {best_score:.6f}")

            if cand_score <= cur_score:
                a = cand
                cur_score = cand_score
                step = min(step * 1.05, 1.0)
            else:
                # reject candidate, shrink step and reset momentum
                momentum = np.zeros_like(a)
                step *= 0.5
                if step < 1e-12:
                    break
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program refinement
# ----------------------------------------------------------------------
def _lp_refinement(a0: np.ndarray, deadline: float, max_iter: int = 4, mix_factor: float = 0.3):
    """
    Solve  maximise ∑g  s.t. a0⋆g ≤ M, g ≥ 0  (M = max(a0⋆a0)).
    The solution is normalised and mixed back into a0.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current bound M
        M = _conv_self(a).max()

        # build sparse convolution matrix A (shape (2n‑1, n))
        idx = np.arange(n, dtype=np.int32)
        row_idx = np.repeat(idx, n) + np.tile(idx, n)       # i + j
        col_idx = np.tile(idx, n)                         # j
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        # linear program: maximise sum(g)  →  minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
        if not res.success:
            print(f"[{time.time()-start_time:.2f}s] LP failed: {res.message}")
            break

        g0 = res.x
        S = g0.sum()
        if S <= 1e-12:
            print(f"[{time.time()-start_time:.2f}s] LP returned near‑zero sum")
            break

        g_norm = g0 / S

        # convex combination
        new = (1.0 - mix_factor) * a + mix_factor * g_norm
        new = np.maximum(new, 0.0)
        new /= new.sum()
        new_score = _objective_score(new)

        if new_score < best_score:
            best_score = new_score
            best_a = new.copy()
            a = new
            print(f"[{time.time()-start_time:.2f}s] LP mix improved to {best_score:.6f}")
        else:
            # if no improvement, shrink mixing factor
            mix_factor *= 0.5
            if mix_factor < 1e-4:
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Final random hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float, deadline: float):
    """Cheap random mass‑transfer moves for the last few seconds."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = score0
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        delta = a[src] * rng.random() * 0.05        # move ≤5 % of source mass
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _objective_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time()-start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Locate a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain list) found within the time budget.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Load a global starter if available
    # ------------------------------------------------------------------
    height_seq = globals().get("height_sequence_1", None)
    if height_seq is not None:
        try:
            init = np.asarray(height_seq, dtype=float).copy()
            init = np.maximum(init, 0.0)
            if init.size == 0:
                init = None
        except Exception:
            init = None
    else:
        init = None

    # ------------------------------------------------------------------
    # Problem size (use the size of the starter if present)
    # ------------------------------------------------------------------
    if init is not None:
        n = init.size
        a = init.copy()
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)
    else:
        n = 4096
        a = np.full(n, 1.0 / n)

    best_seq = a.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time()-start_time:.2f}s] start objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet + deterministic shapes
    # ------------------------------------------------------------------
    dir_deadline = min(deadline, start_time + 30.0)
    if time.time() < dir_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=300, deadline=dir_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time()-start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

    # deterministic shapes (sine, Hann, triangular)
    shapes = [
        np.sin(np.pi * (np.arange(n) + 1) / (n + 1)),
        0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)),
        np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float),
    ]
    for shp in shapes:
        shp = np.maximum(shp, 0.0)
        shp = _project_simplex(shp)
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq, best_score = shp.copy(), sc
            print(f"[{time.time()-start_time:.2f}s] deterministic shape score {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth projected‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.75
        beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
        restarts = 5
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.005, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a_opt, sc_opt = _smooth_descent(a0, beta_schedule, descent_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time()-start_time:.2f}s] descent restart {r} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – linear‑program refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = deadline - 20.0   # reserve ~20 s for the final hill‑climb
        a_opt, sc_opt = _lp_refinement(best_seq, lp_deadline, max_iter=4, mix_factor=0.3)
        if sc_opt < best_score:
            best_seq, best_score = a_opt, sc_opt
            print(f"[{time.time()-start_time:.2f}s] after LP best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – very short hill‑climbing tweak for the last seconds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time()-start_time:.2f}s] start final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time()-start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
