def _grad_descent_smoothed_max(a0, n, L_fft, L_corr, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    rng = np.random.default_rng()
    
    p_vals = [2, 4, 8, 16, 32, 64]
    # allocate time slice per p according to remaining time.
    total_time_rem = deadline - time.time()
    # allocate equal fraction to each p
    time_per_p = total_time_rem / len(p_vals)
    for p in p_vals:
        sub_deadline = time.time() + time_per_p
        it = 0
        eta0 = 0.2
        while time.time() < sub_deadline:
            it += 1
            # compute conv and gradient
            conv = _conv_self_fft(a, L_fft)
            max_c = conv.max()
            # compute weight w
            diff = conv - max_c  # <=0
            # compute w=exp(p*diff)
            w = np.exp(p * diff)
            sum_w = w.sum()
            if sum_w == 0:
                # fallback
                w_norm = np.full_like(w, 1.0 / w.size)
            else:
                w_norm = w / sum_w
            # compute gradient via correlation
            # reversed w_norm
            w_rev = w_norm[::-1]
            # FFT
            A = np.fft.rfft(a, n=L_corr)
            W_rev = np.fft.rfft(w_rev, n=L_corr)
            full = np.fft.irfft(W_rev * A, n=L_corr)
            start_idx = n - 1
            grad = 2.0 * full[start_idx:start_idx + n]
            # evaluate actual score occasionally
            if it % 20 == 0:
                cur_score = evaluate_sequence(a.tolist())
                if cur_score < best_score:
                    best_score, best_a = cur_score, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] p={p} iter={it} score={cur_score:.6f}")
            # step size
            eta = eta0 / np.sqrt(it + 1)
            # step and project
            a_new = a - eta * grad
            a = _project_simplex(a_new)
            # small random jitter occasionally
            if np.random.rand() < 0.001:
                eps = np.random.normal(scale=0.001, size=n)
                a = _project_simplex(np.maximum(a + eps, 0.0))
    # after loops, compute final score
    final_score = evaluate_sequence(a.tolist())
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score

    total_rem = deadline - time.time()

        return best_a, best_score
    time_per_p = total_rem / len(p_vals)
    for p in p_vals:
        sub_deadline = time.time() + time_per_p
        it = 0
        eta0 = 0.2  # initial step size

            # compute gradient via smoothed max

            # exponential weights
            # compute w = exp(p * diff)
            # to avoid underflow for large values, we can compute in double; diff ~ -0.00012 minimal.

def propose_candidate(seed=0,budget_s=1000,**kwargs):
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    n = 4096
    L_fft = _next_pow2(2*n - 1)
    L_corr = _next_pow2(3*n - 2)

    uniform = np.full(n, 1.0/n)
    a = uniform.copy()
    conv = _conv_self_fft(a, L_fft)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # Warm start Dirichlet draws for 5% of budget
    warm_deadline = min(deadline, start_time + 0.05 * (deadline - start_time))
    if time.time() < warm_deadline:
        best_a, best_score = _warm_start_dirichlet(n, best_a, best_score, L_fft, warm_deadline)

    # Gradient descent smoothed max for ~80% remaining time

        grad_deadline = time.time() + 0.80 * (deadline - time.time())

        hill_deadline = time.time() + 0.15 * (deadline - time.time())

    # Final refinement mixing/uniform

        print(f"[{time.time() - start_time:.2f}s] Final refinement")
        best_a, best_score = _final_refinement(best_a, n, L_fft, deadline)

    # Normalize for safety
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total

        best_a = uniform.copy()

        print(f"Error evaluating final sequence: {e}")

"""
Improved search for a non‑negative step‑function that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of four stages:

0️⃣  Warm‑start – a few Dirichlet draws (log‑uniform concentration) to
    obtain a better than uniform start.

1️⃣  Smoothed‑max gradient descent – we minimise a smooth approximation of
    the L∞‑norm of the self‑convolution:
        Φp(a) = (1/p)·log Σk exp(p·(c_k – max(c)))   ,  c = a*a .
    For a given smoothing parameter p the gradient is
        ∇Φp = 2·corr(w , a)      where   w_i ∝ exp(p·(c_i–max(c))).
    We start with a small p and increase it (2,4,…,1024) so that the
    optimisation first flattens the whole convolution and then focuses
    on the current maximum.  After each step the iterate is projected onto
    the probability simplex (non‑negative, unit sum) with the Duchi
    projection.

2️⃣  Targeted hill‑climb – while time permits we repeatedly locate the
    index k where the convolution is maximal, pick the pair (i,j) that
    contributes most to this entry and move a tiny amount of mass from
    the larger of {i,j} to a random coordinate, accepting only improving
    moves.

3️⃣  Final refinement – a short convex mixing with the uniform vector
    followed by a cheap random‑pair hill‑climb, as in the reference
    implementation.

All heavy work (FFT‑based convolutions, gradients, simplex projection) is
implemented with NumPy; the total running time respects the 1000 s budget.
The returned list sums (approximately) to one, and all entries are
non‑negative.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    For a given convolution index k, return the pair (i, j) with the
    largest product a_i·a_j (where i+j=k) together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    max_pos = np.argmax(contrib)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(contrib[max_pos])


# ----------------------------------------------------------------------
# 0️⃣ Warm‑start with Dirichlet draws
# ----------------------------------------------------------------------
def _warm_start_dirichlet(
    n: int,
    best_a: np.ndarray,
    best_score: float,
    L_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # choose concentration log‑uniformly
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={cur_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# 1️⃣ Smoothed‑max gradient descent
# ----------------------------------------------------------------------
def _grad_descent_smoothed_max(
    a0: np.ndarray,
    n: int,
    L_fft: int,
    L_corr: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed surrogate of the L∞‑norm
    of the self‑convolution.  The surrogate is
        Φp(a) = (1/p)·log Σ exp(p·(c - max(c)))   (c = a*a).
    The gradient is 2·corr(w , a) where w_i ∝ exp(p·(c_i - max(c))).
    The smoothing parameter p is increased successively.
    """
    a = a0.copy()
    # current best (using the exact objective)
    conv = _conv_self_fft(a, L_fft)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    # schedule of smoothing parameters
    p_vals = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]

    # allocate roughly equal time to each p
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_a, best_score
    time_per_p = remaining / len(p_vals)

    for p in p_vals:
        sub_deadline = time.time() + time_per_p
        it = 0
        eta0 = 0.25  # initial stepsize (tuned empirically)

        while time.time() < sub_deadline:
            it += 1

            # current convolution and exact score
            conv = _conv_self_fft(a, L_fft)
            max_c = conv.max()
            cur_score = 2.0 * n * max_c
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] p={p}  iter={it:,}  score={cur_score:.6f}"
                )

            # ---------- smoothed‑max weights ----------
            diff = conv - max_c                # ≤ 0
            w = np.exp(p * diff)               # w_k = exp(p·(c_k – max_c))
            sum_w = w.sum()
            if sum_w == 0.0:
                w_norm = np.full_like(w, 1.0 / w.size)
            else:
                w_norm = w / sum_w

            # ---------- gradient (2·corr(w_norm , a)) ----------
            w_rev = w_norm[::-1]               # reversed for correlation
            A = np.fft.rfft(a, n=L_corr)
            W_rev = np.fft.rfft(w_rev, n=L_corr)
            full = np.fft.irfft(W_rev * A, n=L_corr)
            start_idx = n - 1
            grad = 2.0 * full[start_idx : start_idx + n]

            # ---------- projected step ----------
            eta = eta0 / np.sqrt(it + 1)       # classic 1/√t schedule
            a_new = a - eta * grad
            a = _project_simplex(a_new)

            # occasional tiny random jitter to escape plateaus
            if np.random.rand() < 0.001:
                eps = np.random.normal(scale=0.001, size=n)
                a = _project_simplex(np.maximum(a + eps, 0.0))

    # final check
    conv = _conv_self_fft(a, L_fft)
    cur_score = 2.0 * n * conv.max()
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score

# 2️⃣ Targeted hill‑climb on the maximum convolution entry
# ----------------------------------------------------------------------
def _targeted_hill_climb(
    a: np.ndarray,
    n: int,
    L_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Aggressive local search: locate the index k where conv(a,a) is maximal,
    pick the pair (i,j) contributing the most to that entry and shift a tiny
    amount of mass from the larger partner to a random coordinate.
    Accept only improving moves.
    """
    best_a = a.copy()
    conv = _conv_self_fft(best_a, L_fft)
    best_score = 2.0 * n * conv.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        i, j, contrib = _max_contributions(best_a, k_max)
        if i == -1:
            continue

        # move mass from the larger of the two contributing entries
        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-12:
            continue
        dst = rng.integers(0, n)
        if dst == src:
            continue

        delta = rng.uniform(0.0, min(best_a[src], 0.004))
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        new_score = 2.0 * n * _conv_self_fft(cand, L_fft).max()
        if new_score < best_score:
            best_score, best_a = new_score, cand
            print(
                f"[{time.time() - start_time:.2f}s] Hill i={src}→{dst} δ={delta:.5f} → score={new_score:.6f}"
            )
    return best_a, best_score

# 3️⃣ Final mixing + cheap random‑pair hill‑climb
# ----------------------------------------------------------------------
def _final_refinement(
    a: np.ndarray,
    n: int,
    L_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self_fft(best_a, L_fft).max()
    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = 2.0 * n * _conv_self_fft(cand, L_fft).max()
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = 2.0 * n * _conv_self_fft(cand, L_fft).max()
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Light hill i={i} j={j} δ={delta:.5f} → score={sc:.6f}"
            )
    return best_a, best_score
