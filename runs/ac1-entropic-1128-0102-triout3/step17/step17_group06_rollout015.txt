def proj_simplex(v):
    # returns projection of v onto simplex {x>=0, sum x=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    # Set random seed
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05

    # Determine n (size)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            n = starter.size
        else:
            n = 4096
    except Exception:
        n = 4096

    # Compute L = next_pow2(3*n-2)
    L = _next_pow2(3 * n - 2)

    # Baseline uniform
    uniform = np.full(n, 1.0 / n)
    # Compute score function using custom _score function to avoid evaluation overhead
    def _score(a):
        conv = _conv_self_fft(a, _next_pow2(2 * n - 1))
        maxb = conv.max()
        return float(2.0 * n * maxb)

    best_a = uniform.copy()
    best_score = _score(best_a)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score = {best_score:.6f}")

    # if global starter present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                # resample to size n via linear interpolation
                # use np.interp
                x_old = np.linspace(0, 1, starter.size)
                x_new = np.linspace(0, 1, n)
                starter_res = np.interp(x_new, x_old, starter)
                starter = starter_res
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _score(starter)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter improved score = {sc:.6f}")
    except Exception:
        pass

    # Warm start with Dirichlet draws (approx 3% time)
    warm_deadline = min(deadline, start_time + 0.03 * budget_s)  # 3% of budget
    while time.time() < warm_deadline:
        conc = np.exp(rng.uniform(np.log(0.03), np.log(30.0)))  # log-uniform
        sample = rng.gamma(shape=conc, scale=1.0, size=n)

        sc = _score(a)
        if sc < best_score:

            print(f"[{time.time() - start_time:.2f}s] Dirichlet warm start → score = {sc:.6f}")

    p_vals = np.linspace(0.1, 2.0, 191)  # step 0.01

"""
Annealed exponentiated‑gradient optimisation for the self‑convolution‐max
problem.

The routine

* scans a fine family of Beta(p,p) densities (including convex mixtures
  with the uniform vector) to obtain a strong deterministic warm‑start,
* then runs an exponentiated‑gradient descent on a smooth (soft‑max)
  approximation of the objective  max_k (a*a)_k.
  The temperature is slowly annealed so that the algorithm first moves
  using a smooth landscape and finally focuses on the true max.
* the update keeps the vector on the probability simplex (positivity is
  guaranteed by the exponential, normalisation restores the sum‑to‑one).

The whole search respects the supplied time budget (default 1000 s) and
prints progress.  The final list is a non‑negative step‑function that
minimises ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised Beta(p,p) density on the grid i=0,…,n‑1,
    i.e. a_i ∝ x_i^{p‑1} (1‑x_i)^{p‑1} with x_i = (i+0.5)/n.
    Result is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _conv_self_full(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Full linear convolution a⋆a (length 2·len(a)‑1) via FFT.
    The result is a real vector of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function (list of floats) that minimises
    evaluate_sequence.  The routine respects the time budget and prints
    progress.  The returned list sums to ~1 (scale‑invariant for the
    evaluator).

    rng = np.random.default_rng(seed)

    start_time = time.time()

    # otherwise fall back to a fairly large grid (8192 points) which gives
    # a finer discretisation of the continuous Beta family.
    # --------------------------------------------------------------

        else:
            raise ValueError
    except Exception:
        n = 8192                     # a size that still runs comfortably fast

    # FFT length large enough for both self‑convolution (2n‑1) and
    # correlation (n + 2n‑2 = 3n‑1).  Use a single length for all transforms.
    L_fft = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Helper: fast score for a probability vector (sum = 1)
    # --------------------------------------------------------------
    def _score(a: np.ndarray) -> float:
        """Return 2·n·max(a⋆a) for a vector already normalised to sum 1."""
        conv = _conv_self_full(a, L_fft)

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    eta = 0.5
    eta_decay = 0.9999

    # main optimisation loop (runs until the deadline)
    while time.time() < deadline:
        iteration += 1

        # ---- convolution a⋆a (full) ---------------------------------
        Af = np.fft.rfft(a, n=L_fft)
        conv_full = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
        max_b = conv_full.max()
        cur_score = float(2.0 * n * max_b)

        # keep the best ever seen

            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter {iteration} → best score = {cur_score:.6f}")

        # ---- soft‑max weights (smooth max) -------------------------
        # w_i ∝ exp(conv_i / τ)  (stabilised by subtracting the max)
        conv_shifted = conv_full - np.max(conv_full)

            a_sum = a.sum()

    best_a = np.maximum(best_a, 0.0)

    return [float(v) for v in best_a]
