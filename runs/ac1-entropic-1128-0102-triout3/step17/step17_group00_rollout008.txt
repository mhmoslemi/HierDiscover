def convolve_fft(x, y, L=None):
    n1 = len(x)
    n2 = len(y)
    L = L or _next_pow2(n1 + n2 - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n1 + n2 - 1]
    return conv

def _conv_fft(a, b, L_fft):
    Af = np.fft.rfft(a, n=L_fft)
    Bf = np.fft.rfft(b, n=L_fft)
    conv = np.fft.irfft(Af * Bf, n=L_fft)

    return conv[:out_len]

"""
Deterministic‑annealing + Adam optimisation of a step function.

The algorithm works with a probability‑simplex representation:
    a_i = exp(x_i) / sum_j exp(x_j)  (x ∈ ℝⁿ).

A smooth approximation of the max‑convolution
    f(a) = max_k (a * a)_k
is built with a soft‑max over the full self‑convolution:
    w_k ∝ exp(α·((a*a)_k / max(a*a) – 1))

For a fixed α the gradient of the smooth objective is
    ∇_a f ≈ 2·(w * a)            (convolution of w with the reversed a)
which is computed efficiently with FFTs.
The logits x are updated by the Adam optimiser,
while α is slowly increased (deterministic annealing) so that
the smooth objective approaches the true L∞‑norm.
After the annealing phase a lightweight random hill‑climb
performs final fine‑tuning.

The whole search runs under the 1000 s budget and returns the
best sequence found (normalised to sum = 1, all entries ≥0).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a * a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a * b via FFT (length len(a)+len(b)‑1)."""
    na, nb = a.shape[0], b.shape[0]
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: na + nb - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Deterministic‑annealing optimiser (Adam on logits)
# ----------------------------------------------------------------------


def _annealing_opt(a_start: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits of a probability vector.
    A smooth max‑convolution (soft‑max) is used with a temperature α
    that is slowly increased (deterministic annealing).
    Returns the best sequence and its objective value.
    """
    n = a_start.shape[0]
    # initialise logits
    x = np.log(a_start + 1e-30)

    # Adam buffers
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # annealing schedule
    alpha = 0.5            # start temperature
    max_alpha = 80.0       # final temperature
    alpha_mult = 1.02      # multiplicative increase

    it = 0
    best_seq = a_start.copy()
    best_score = _eval_fast(best_seq, L_fft)

    while time.time() < deadline:
        it += 1

        # ----- logits → probability vector (softmax) -----
        mx = x.max()
        expx = np.exp(x - mx)           # stabilise
        a = expx / expx.sum()           # stays on the simplex

        # ----- self‑convolution and soft‑max weights -----
        conv = _conv_self_fft(a, L_fft)          # length 2n‑1
        max_conv = conv.max()
        if max_conv == 0.0:                     # degenerate – should not happen
            norm_conv = conv
        else:
            norm_conv = conv / max_conv        # values in [0,1]
        w = np.exp(alpha * (norm_conv - 1.0))   # soft‑max (peak at max_conv)
        w_sum = w.sum()
        w_norm = w / w_sum

        # ----- gradient  ∇_a f ≈ 2 * (w * a) -----
        a_rev = a[::-1]                         # reversed a
        conv_wa = _conv_fft(w_norm, a_rev, L_fft)   # length 3n‑2
        start_idx = n - 1
        g_a = 2.0 * conv_wa[start_idx:start_idx + n]

        # ----- gradient w.r.t. logits (chain rule) -----
        dot = np.dot(a, g_a)
        g_x = a * (g_a - dot)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Simple learning‑rate schedule (decreases slowly)
        lr = 0.25 / (1.0 + 2e-4 * it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- evaluate (every few steps) -----
        if it % 5 == 0:
            # objective can be obtained directly from max_conv
            cur_score = 2.0 * n * max_conv
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} "
                      f"α={alpha:.2f} score={cur_score:.6f}")

        # ----- annealing update -----
        if it % 2000 == 0 and alpha < max_alpha:
            alpha = min(max_alpha, alpha * alpha_mult)

        # occasional jitter to avoid tiny plateaus
        if it % 10000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            x += jitter

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final random hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------


def _final_hill_climb(a_start: np.ndarray,
                      L_fft: int,
                      deadline: float,
                      start_time: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very local stochastic search: move a tiny amount of mass from a random
    source index to a random destination index.  Accept only improving moves.
    """
    n = a_start.shape[0]
    best_seq = a_start.copy()
    best_score = _eval_fast(best_seq, L_fft)

    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 5e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # up to 0.2 % of the source mass is transferred
        eps = best_seq[src] * rng.random() * 2e-3
        cand = best_seq.copy()
        cand[src] -= eps
        cand[dst] += eps

        cur_score = _eval_fast(cand, L_fft)
        if cur_score < best_score:
            best_seq = cand
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={cur_score:.6f}")

    return best_seq, best_score

    # --------------------------------------------------------------
    # Baseline – initial sequence.
    # --------------------------------------------------------------
    initial_score = _eval_fast(a0, L_fft)
    print(f"[{time.time() - start_time:.2f}s] initial score={initial_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Deterministic‑annealing optimisation (Adam on logits)
    # --------------------------------------------------------------
    # Reserve most of the budget for the annealing phase; leave ~120 s
    # for the final hill‑climb.
    anneal_deadline = deadline - 120.0
    if anneal_deadline > start_time + 10.0:  # need at least a few seconds
        a_opt, best_score = _annealing_opt(a0, L_fft, anneal_deadline,
                                           start_time, rng)
    else:
        a_opt, best_score = a0, initial_score

    # --------------------------------------------------------------
    # 2️⃣ Final random hill‑climb (very fine local search)
    # --------------------------------------------------------------
    a_final, best_score = _final_hill_climb(a_opt, L_fft,
                                            deadline, start_time, rng)

    # --------------------------------------------------------------
    # Normalise finally and return as a plain Python list.
    # --------------------------------------------------------------
    a_final = np.maximum(a_final, 0.0)
    total = a_final.sum()
    if total > 0.0:
        a_final = a_final / total
    else:
        a_final = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in a_final]
