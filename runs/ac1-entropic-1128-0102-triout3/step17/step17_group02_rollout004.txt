```python
"""
Multi‑phase optimizer for minimize 2·n·max(conv(a,a)).

Phases:
 1. Random Dirichlet draws + a few analytic shapes (Gaussian, exponential,
    Hann, triangular) to obtain a strong start.
 2. Symmetric projected‑gradient descent on a smooth approximation of
    the max‑convolution (log‑sum‑exp) with an increasing temperature schedule.
    Several restarts with jitter broaden the search.
 3. Simulated‑annealing “mass‑transfer’’ moves that keep the vector on the
    probability simplex and respect symmetry.
 4. (Optional) brief polishing gradient step.

All vectors are kept on the probability simplex (non‑negative, sum = 1) and
symmetrised after each update, because the objective is invariant under
reversal.  The algorithm is deliberately lightweight – the most expensive
operation is an FFT‑based convolution, which for n≈4096 costs ≈5 µs,
allowing many thousands of iterations within the 1000 s budget.
"""

import time
import numpy as np

# Global timing variable used by helper functions
start_time = None

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT of length power‑of‑2."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector a with sum=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Smooth max via log‑sum‑exp."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta

def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Uses the identity grad = 2 * (a_rev * w) where w are softmax weights.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()  # shape (2n-1,)
    rev_corr = _convolve_fft(a[::-1], w)  # length 3n-2 (but we need n entries)
    n = a.shape[0]
    # The relevant slice corresponds to offsets that line up with a's length
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad

def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of real vectors via FFT (no length restriction)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]

def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force a vector to be symmetric (v_i = v_{n-1-i})."""
    return (v + v[::-1]) / 2.0

def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best vector obtained from a limited number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration drawn log‑uniformly in [10⁻³, 10⁰·⁷] (≈[0.001,5])
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        # optional symmetry
        a = _symmetrize(a)
        a = _project_simplex(a)
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score

def _smooth_descent(a0: np.ndarray,
                    score0: float,
                    beta_vals,
                    deadline: float):
    """
    Projected gradient descent on a smooth approximation of max‑convolution.
    The vector is kept symmetric after each projection.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    for beta in beta_vals:
        step = 0.5
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)
            improved = False
            for factor in (1.0, 0.5, 0.25, 0.125):
                cand = a - step * factor * grad
                cand = np.maximum(cand, 0.0)
                cand = _project_simplex(cand)
                cand = _symmetrize(cand)
                # No need to re‑project after symmetrisation because sum is preserved.
                cand_score = _objective_score(cand)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    improved = True
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} new best {best_score:.6f}")
                    break
            if not improved:
                step = max(step * 0.5, 1e-12)
            else:
                step = min(step * 1.05, 1.0)
            if step < 1e-12:
                break
    return best_a, best_score

def _simulated_annealing(a0: np.ndarray,
                         score0: float,
                         deadline: float):
    """
    Mass‑transfer simulated‑annealing that respects the simplex and symmetry.
    """
    a = a0.copy()
    cur_score = score0
    best_seq = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    # temperature schedule (geometric decay)
    init_temp = best_score * 0.05   # start fairly warm
    final_temp = best_score * 1e-6

    # pre‑compute length for FFT (same as _conv_self)
    while time.time() < deadline:
        # fraction of the SA budget elapsed
        total = deadline - start_time
        elapsed = time.time() - start_time
        frac = (elapsed) / total
        # exponential cooling
        temp = init_temp * ((final_temp / init_temp) ** frac)
        # decreasing move amplitude
        move_factor = 0.5 * (1.0 - frac) + 0.02 * frac

        # propose a random mass transfer
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.random() * move_factor
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # enforce symmetry (preserves sum) and non‑negativity
        cand = _symmetrize(cand)
        # after symmetrisation, rounding errors may introduce tiny negatives
        cand = np.maximum(cand, 0.0)

        # evaluate
        conv = _conv_self(cand)
        cand_score = float(2.0 * n * conv.max())
        dE = cand_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / max(temp, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that minimises
    ``evaluate_sequence``.  The method combines random Dirichlet starts,
    deterministic analytic shapes, a symmetric projected‑gradient descent on a
    smooth approximation of the max‑convolution and a final simulated‑annealing
    phase.  The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1    # a small safety margin

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressive power and speed
    # ------------------------------------------------------------------
    n = 4096

    # initial uniform vector (already symmetric)
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter “height_sequence_1” if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            starter = _project_simplex(starter)
            starter = _symmetrize(starter)
            sc = _objective_score(starter)
            if sc < best_score:
                best_seq = starter.copy()
                best_score = sc
                print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet draws + deterministic analytic shapes
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 50.0)   # up to 50 s for this phase
    if time.time() < phase1_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=3000, deadline=phase1_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

    # deterministic shapes (Gaussian, exponential, Hann, triangular, sine)
    shapes = []

    # Gaussian‑like shape
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n / 6.0
    shapes.append(np.exp(-0.5 * (x / sigma) ** 2))

    # Exponential decay from the centre
    lam = 0.004
    shapes.append(np.exp(-lam * np.abs(x)))

    # Hann window (raised cosine)
    shapes.append(0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # Triangular (symmetric)
    triangular = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    shapes.append(triangular)

    # Half‑sine (positive version)
    shapes.append(np.sin(np.pi * (np.arange(n) + 1) / (n + 1)))

    for idx, shp in enumerate(shapes):
        shp = np.maximum(shp, 0.0)
        shp = _project_simplex(shp)
        shp = _symmetrize(shp)
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq, best_score = shp.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] deterministic shape {idx} score {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth projected‑gradient descent (symmetrised)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.65   # reserve ~35 % for SA
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0,
                         200.0, 500.0, 1000.0, 2000.0]
        restarts = 12
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break
            # jitter around current best (scale grows with restart index)
            rng = np.random.default_rng()
            jitter_scale = 0.02 + 0.03 * (r / (restarts - 1))
            jitter = rng.normal(scale=jitter_scale, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0 = _symmetrize(a0)
            a0_score = _objective_score(a0)

            sub_deadline = min(descent_deadline,
                               time.time() + (descent_deadline - time.time()) / (restarts - r))
            a_opt, sc_opt = _smooth_descent(a0, a0_score, beta_schedule, sub_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing mass‑transfer refinement
    # ------------------------------------------------------------------

Multi‑phase optimizer for minimise 2·n·max(conv(a,a))

The search works on a probability vector (non‑negative, sum = 1) of length n≈4096.
All vectors are kept symmetric (a[i]=a[n‑1‑i]) because the objective is invariant
under reversal.  The optimiser consists of four stages:

1️⃣  Random Dirichlet draws (with a few analytic shapes such as Gaussian,
    exponential, Hann, triangular, sine) to obtain a strong starting point.

2️⃣  Symmetric projected‑gradient descent on a smooth surrogate of the
    max‑convolution (log‑sum‑exp).  An increasing temperature (β) schedule
    and several jittered restarts broaden the search.

3️⃣  Simulated‑annealing mass‑transfer moves.  Each move conserves the total
    mass, respects non‑negativity and symmetry, and is accepted according to
    a temperature schedule that decays to near‑zero.

4️⃣  A quick polishing gradient step (if time permits).

All heavy convolutions are performed with FFTs; the projection onto the
simplex uses the O(n) algorithm of Duchi et al.  The function returns the best
sequence (as a plain Python list) found before the 1000 s budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing variable (used by helper functions for logging)
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry: v_i = v_{n‑1‑i}."""
    return (v + v[::-1]) / 2.0


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Smooth approximation of max(conv) using log‑sum‑exp with temperature β."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max w.r.t. a.
    Uses the identity grad = 2·(a_rev ∗ w) where w are softmax weights.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()                # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)   # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad

    best_score = cur_score
    n = a.shape[0]

        # move amplitude linearly shrinks from 0.5 → 0.02
