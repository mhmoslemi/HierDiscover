def subgradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    cur_score = _objective_score(a)
    n = a.shape[0]
    eta_initial = 0.1
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # compute gradient for max indices
        # find all indices within tolerance relative to max
        tol = max_val * 1e-12 + 1e-12
        idxs = np.where(conv >= max_val - tol)[0]
        if idxs.size == 1:
            k = idxs[0]
            # compute gradient by slicing
            grad = np.zeros_like(a)
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            # compute slice range
            i_range = np.arange(i_start, i_end+1, dtype=int)
            grad[i_range] = 2.0 * a[k - i_range]
        else:
            # average gradients for multiple max indices
            grad = np.zeros_like(a)
            for k in idxs:
                i_start = max(0, k - (n-1))
                i_end = min(n-1, k)
                i_range = np.arange(i_start, i_end+1, dtype=int)
                grad[i_range] += 2.0 * a[k - i_range]
            grad /= idxs.size
        # step size diminishing: eta = eta_initial / np.sqrt(iteration)
        eta = eta_initial / np.sqrt(iteration)
        a_new = a - eta * grad
        # project onto simplex
        a_new = _project_simplex(a_new)
        new_score = _objective_score(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Subgrad iteration {iteration} new best score={new_score:.6f}")
        else:
            # occasional random jitter to avoid stall
            if iteration % 100 == 0:
                rng = np.random.default_rng()
                eps = 1e-3 * rng.random(a.shape) * a
                a = np.clip(a + eps, 0, None)
                a = _project_simplex(a)
                cur_score = _objective_score(a)
    return best_seq, best_score

def random_hill_climb(a0, deadline, best_seq, best_score):
    a = a0.copy()
    cur_score = _objective_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # pick random index i with non-zero mass, random j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # propose delta up to a[i] (if a[i] > small)
        max_delta = a[i] * 0.5  # move at most half mass
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non negativity automatically
        new_score = _objective_score(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill climb iter {iteration} best={new_score:.6f}")
        # else occasionally accept with small probability to escape
        elif rng.random() < 0.001:
            a = a_new
            cur_score = new_score
    return best_seq, best_score

"""
Propose a step‑function (non‑negative list) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq,seq)) / (sum(seq)**2)

The function builds a good candidate by

1.   scanning a few simple analytic families (geometric, exponential,
     Hann, Gaussian, triangular, sine…) and many random Dirichlet draws;
2.   launching a smooth‑max mirror‑descent (exponential update) with a
     β‑schedule that slowly tightens the soft‑max approximation;
3.   refining the result with a true sub‑gradient descent on the exact
     ‑max objective;
4.   escaping residual plateaus with a cheap random‑jitter hill‑climb.

All updates stay on the probability simplex, guaranteeing the
non‑negativity and sum‑positive requirements of ``evaluate_sequence``.
The routine stops just before the 1000 s budget and returns the best
sequence found (as a plain Python list)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (full length) via FFT."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (sum(seq) is assumed to be 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient) where the gradient is wrt a.
    """
    n = a.shape[0]
    conv = _conv_self(a)              # length 2·n‑1
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # stability shift
    Z = exp_scaled.sum()
    w = exp_scaled / Z               # soft‑max weights, sum to 1

    # gradient of the weighted sum: 2 * a * w (convolution)
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)   # length 3·n‑2
    start = len(w) - 1                # = 2·n‑2   (zero‑shift of convolution)
    grad = 2.0 * conv_rev[start - np.arange(n)]

    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, grad


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent (exponential) step: a_i ← a_i·exp(−η·grad_i), then renormalise.
    Guarantees non‑negativity and Σa_i = 1.
    """
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)          # avoid overflow
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total


def _subgradient_max(a: np.ndarray, conv: np.ndarray):
    """
    Compute a (sub)gradient of max(conv) at a.
    For a single active index k the gradient is:
        g_i = 2·a_{k‑i}   if 0≤k‑i<n, else 0.
    If several indices tie for the maximum the gradients are averaged.
    Returns (k_indices, gradient).
    """
    n = a.shape[0]
    max_val = conv.max()
    # numerical tolerance – treat values indistinguishable from max as active
    tol = max_val * 1e-12 + 1e-12
    idxs = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)                 # inclusive
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]
    grad /= max(1, idxs.size)
    return idxs, grad

def _quick_families(n, best_seq, best_score, deadline, start_time):
    """A handful of cheap analytic families – often already beats uniform."""
    idx = np.arange(n, dtype=np.float64)

    # 1) Geometric decay (left‑to‑right)
    for eps in np.logspace(-5, -1, 80):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        sc = _objective_score(geo)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} score={sc:.6f}")

    # 2) Symmetric exponential from centre
    dist = np.abs(idx - (n - 1) / 2.0)
    for lam in np.logspace(-4, -0.2, 60):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SymExp λ={lam:.1e} score={sc:.6f}")

    # 3) Hann (raised cosine) window
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    hann /= hann.sum()
    sc = _objective_score(hann)
    if sc < best_score:
        best_score, best_seq = sc, hann.copy()
        print(f"[{time.time() - start_time:.2f}s] Hann window score={sc:.6f}")

    # 4) Gaussian centred at the middle
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((idx - (n - 1) / 2.0) / sigma) ** 2)
    gauss /= gauss.sum()
    sc = _objective_score(gauss)
    if sc < best_score:
        best_score, best_seq = sc, gauss.copy()
        print(f"[{time.time() - start_time:.2f}s] Gaussian σ={sigma:.1f} score={sc:.6f}")

    # 5) Triangular (peak in centre)
    tri = 1.0 - np.abs(idx - (n - 1) / 2.0) / ((n - 1) / 2.0)
    tri = np.clip(tri, 0.0, None)
    tri /= tri.sum()
    sc = _objective_score(tri)
    if sc < best_score:
        best_score, best_seq = sc, tri.copy()
        print(f"[{time.time() - start_time:.2f}s] Triangular score={sc:.6f}")

    # 6) Sine shape (positive on the whole interval)
    sine = np.sin(np.pi * (idx + 0.5) / n)   # never zero
    sine /= sine.sum()
    sc = _objective_score(sine)
    if sc < best_score:
        best_score, best_seq = sc, sine.copy()
        print(f"[{time.time() - start_time:.2f}s] Sine shape score={sc:.6f}")

    return best_seq, best_score


def _random_dirichlet_trials(n, best_seq, best_score, deadline, start_time, trials=3000):
    """Draw many random Dirichlet vectors with a wide range of concentrations."""
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)        # conc. in [0.01, 5] roughly
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


def _smooth_max_descent(a0, deadline, best_seq, best_score, start_time):
    """Mirror‑descent on the log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    cur_score = _objective_score(a)
    n = a.shape[0]

    # schedule parameters
    beta_start, beta_end = 0.2, 200.0      # very smooth → very tight
    eta = 0.8
    min_eta = 1e-12

    while time.time() < deadline:
        # linearised β schedule (more aggressive as we approach the deadline)
        t = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + t * (beta_end - beta_start)

        smooth, grad = _smooth_max_and_grad(a, beta)

        # back‑tracking line‑search on the true objective (max‑conv)
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, grad, eta_try)
            new_score = _objective_score(a_new)
            if new_score <= cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Smooth β={beta:.2f} η={eta_try:.2e} score={new_score:.6f}")
            # slowly raise η when progress is made
            eta = min(1.5, eta * 1.02)
        else:
            # no progress – inject a tiny random jitter and shrink η
            rng = np.random.default_rng()
            eps = 1e-4 * rng.random(a.shape) * a
            a = np.clip(a + eps, 0.0, None)
            a = _project_simplex(a)
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.7)

        # early stop if we already beat the target
        if best_score <= 1.503:
            break

    return best_seq, best_score


def _subgradient_descent(a0, deadline, best_seq, best_score, start_time):
    """Descent on the exact max‑convolution using sub‑gradients."""
    a = a0.copy()
    cur_score = _objective_score(a)
    n = a.shape[0]
    it = 0
    eta0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idxs, grad = _subgradient_max(a, conv)

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        new_score = _objective_score(a_new)

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Subgrad it={it} score={new_score:.6f}")

        # occasional random jitter to avoid getting stuck
        if it % 200 == 0:
            rng = np.random.default_rng()
            eps = 5e-4 * rng.random(a.shape) * a
            a = np.clip(a + eps, 0.0, None)
            a = _project_simplex(a)
            cur_score = _objective_score(a)

    return best_seq, best_score


def _hill_climb(a0, deadline, best_seq, best_score, start_time):
    """Very cheap random‑jitter hill‑climb (move a tiny amount between two entries)."""
    a = a0.copy()
    cur_score = _objective_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 1e-12:
            continue
        delta = rng.random() * 0.4 * a[i]      # move up to 40 % of a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _objective_score(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it} score={new_score:.6f}")
        # small probability to accept worsening moves (helps escape tiny plateaus)
        elif rng.random() < 1e-4:
            a = a_new
            cur_score = new_score

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a plain Python list of floats (already normalised to sum = 1).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2    # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer search‑space without harming the bound.
    # ------------------------------------------------------------------
    n = 4096

    # Uniform distribution as a baseline
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Quick analytic families
    # ------------------------------------------------------------------
    phase_deadline = deadline - 350.0        # keep ~350 s for the heavy optimisation
    if time.time() < phase_deadline:
        best_seq, best_score = _quick_families(
            n, best_seq, best_score, phase_deadline, start_time)

    # ------------------------------------------------------------------
    # Random Dirichlet initialisers
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline, start_time, trials=6000)

    # ------------------------------------------------------------------
    # Smooth‑max mirror‑descent (core optimiser)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_seq, best_score = _smooth_max_descent(
            best_seq, deadline, best_seq, best_score, start_time)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient descent targeting the true max
    # ------------------------------------------------------------------
    mid_deadline = start_time + 0.75 * (deadline - start_time)
    if time.time() < mid_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_seq, best_score = _subgradient_descent(
            best_seq, mid_deadline, best_seq, best_score, start_time)
        print(f"[{time.time() - start_time:.2f}s] After sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final random hill‑climb to polish any remaining plateaus
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb phase")
        best_seq, best_score = _hill_climb(
            best_seq, deadline, best_seq, best_score, start_time)
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a clean Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
