Implementation details for projection onto simplex: Use function:

def project_simplex(v):
    """Project onto probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all -ve, set uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (probability vector) that minimises
    score = 2·n·max(conv(a,a)) / (sum(a)²)

Approach
--------
1. **Exact Beta‑cell masses** – instead of evaluating the Beta(p,p)
   density at the middle of each bin we integrate it on the whole cell.
   For p=½ this gives the arcsine law whose continuous self‑convolution
   is uniform.  The exact integration (via the regularised incomplete
   beta function) removes the discretisation artefacts that limited the
   previous method.

2. **Warm‑start** – evaluate a few analytically derived shapes
   (uniform, integrated arcsine, several integrated Beta(p,p) families)
   and a handful of random Dirichlet draws.  Keep the best one.

3. **Projected gradient descent** – minimise a smooth surrogate
   of the ∞‑norm:
        f(a) = (1/α)·log∑_k exp(α·c_k)   with   c = a⋆a .
   The gradient of f can be written as a * (softmax(α·c)) + its
   reversed‑convolution and is obtained completely with FFTs.
   After a gradient step we project back onto the probability simplex.
   The step‑size η and the temperature α are adapted on‑the‑fly.

4. **Fine‑grained hill‑climbing** – random pairwise mass transfers
   (accepting only improvements) to squeeze out the remaining
   few × 10⁻³.

The whole routine respects the supplied time budget (default 1000 s)
and prints progress.  The returned list sums to ≈1, is non‑negative,
and can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.special import betainc

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> tuple[float, np.ndarray]:
    """Exact score used by the evaluator (sum(a)=1 ⇒ denominator =1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    score = float(2.0 * n * max_b)   # denominator = 1 because ∑a = 1
    return score, conv


def _beta_integrated(p: float, n: int) -> np.ndarray:
    """Return the discretised Beta(p,p) probabilities by exact cell integration.

    a_i = F((i+1)/n) – F(i/n)  where F is the regularised incomplete beta CDF.
    The result is normalised to sum 1.
    """
    # CDF values at the cell boundaries
    xs = np.linspace(0.0, 1.0, n + 1)
    cdf = betainc(p, p, xs)               # regularised incomplete beta
    a = np.diff(cdf)                       # mass of each cell
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are non‑positive → uniform vector
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and prints
    progress.  The returned list contains non‑negative entries that sum to ~1.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – keep it reproducible
    # --------------------------------------------------------------
    n = 4096
    L_self = _next_pow2(2 * n - 1)              # length for a⋆a
    L_lin = _next_pow2(3 * n)                   # length for convolutions with w (size 2n‑1)

    # --------------------------------------------------------------
    # 0️⃣  Baselines
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score, _ = _score_of(best_a, n, L_self)
    print(f"[0.00s] Uniform baseline score = {best_score:.6f}")

    # Optional global starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc, _ = _score_of(starter, n, L_self)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Exact Beta‑family warm‑start (integrated cells)
    # --------------------------------------------------------------
    # arcsine (Beta(0.5,0.5)) – the theoretically optimal continuous law
    arcsine = _beta_integrated(0.5, n)
    sc, _ = _score_of(arcsine, n, L_self)
    if sc < best_score:
        best_score, best_a = sc, arcsine.copy()
        print(f"[{time.time() - start_time:.2f}s] Integrated arcsine score = {sc:.6f}")

    # Mixtures of arcsine with uniform (coarse grid)
    for lam in np.linspace(0.0, 0.30, 7):          # λ∈{0,.05,…,.30}
        cand = lam * arcsine + (1.0 - lam) * uniform
        cand = cand / cand.sum()
        sc, _ = _score_of(cand, n, L_self)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] arcsine+uniform λ={lam:.2f} → score = {sc:.6f}")

    # A few other integrated Beta(p,p) shapes
    for p in (0.20, 0.25, 0.30, 0.80, 1.00, 1.50, 2.00):
        vec = _beta_integrated(p, n)
        sc, _ = _score_of(vec, n, L_self)
        if sc < best_score:
            best_score, best_a = sc, vec.copy()
            print(f"[{time.time() - start_time:.2f}s] Integrated Beta(p={p:.2f}) → score = {sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet warm‑starts (≈5 % of budget)
    # --------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * budget_s)
    rng = np.random.default_rng()
    while time.time() < warm_deadline:
        # concentration drawn log‑uniformly in [0.02, 20]
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc, _ = _score_of(a, n, L_self)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet warm‑start → score = {sc:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Projected gradient descent on a smooth surrogate of max(conv)
    # --------------------------------------------------------------
    grad_deadline = min(deadline, start_time + 0.80 * budget_s)
    a = best_a.copy()
    eta = 0.5                # initial step‑size
    alpha = 5.0              # temperature for the soft‑max approximation
    min_eta = 1e-6
    max_eta = 2.0
    iter_count = 0

    print(f"[{time.time() - start_time:.2f}s] Starting projected gradient descent")
    while time.time() < grad_deadline:
        iter_count += 1

        # ----- convolution a⋆a (full length 2n‑1) -----
        A_fft = np.fft.rfft(a, n=L_lin)                 # padded for linear conv later
        conv_full = np.fft.irfft(A_fft * A_fft, n=L_lin)[: 2 * n - 1]
        conv = conv_full                                 # shape (2n‑1,)

        # ----- soft‑max weights centred around the maximum -----
        max_c = conv.max()
        exp_scaled = np.exp(alpha * (conv - max_c))
        w = exp_scaled / exp_scaled.sum()                # w ≥0, sum 1, length 2n‑1

        # ----- gradients via FFT (two linear convolutions) -----
        #   g2 = (w * a)       (standard convolution)
        #   g1 = (reverse(w) * a) evaluated at shifted indices
        W_fft = np.fft.rfft(w, n=L_lin)
        conv_w_a = np.fft.irfft(A_fft * W_fft, n=L_lin)[: n + (2 * n - 1) - 1]   # length 3n‑2

        w_rev = w[::-1]
        W_rev_fft = np.fft.rfft(w_rev, n=L_lin)
        conv_rev = np.fft.irfft(A_fft * W_rev_fft, n=L_lin)[: n + (2 * n - 1) - 1]

        # extract the two contributions (both length n)
        g2 = conv_w_a[:n]                                    # (w * a)[i]
        g1 = conv_rev[(2 * n - 2) - np.arange(n)]           # (reverse(w) * a)[i]
        gradient = g1 + g2                                   # ∇f (non‑negative)

        # ----- attempt a projected gradient step -----
        accepted = False
        for _ in range(5):          # at most 5 back‑tracking attempts
            cand = a - eta * gradient
            cand = _project_to_simplex(cand)
            sc_cand, _ = _score_of(cand, n, L_self)
            if sc_cand < best_score - 1e-12:
                # improvement – keep the move
                best_score, best_a = sc_cand, cand.copy()
                a = cand
                accepted = True
                eta = min(max_eta, eta * 1.10)   # gently increase step‑size
                print(f"[{time.time() - start_time:.2f}s] GD iter {iter_count:04d}  "
                      f"improved → score={sc_cand:.6f}   η={eta:.5f}")
                break
            else:
                # no improvement – shrink step‑size
                eta = max(min_eta, eta * 0.5)

        if not accepted:
            # no progress – slowly increase α to focus more on the true max
            alpha = min(80.0, alpha * 1.02)

    # --------------------------------------------------------------
    # 4️⃣  Final hill‑climb (tiny random pairwise transfers)
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Hill‑climbing refinement")
    a = best_a.copy()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # move a small amount from i to j (max 0.001 · current mass)
        delta = rng.uniform(0.0, min(a[i], 0.001))
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand)      # protects against rounding noise
        sc_cand, _ = _score_of(cand, n, L_self)
        if sc_cand < best_score:
            best_score, best_a = sc_cand, cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improvement → score = {sc_cand:.6f}")

    # --------------------------------------------------------------
    # Normalise & return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
