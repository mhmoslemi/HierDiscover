def _subgradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        max_c = conv.max()
        # find max indices
        idxs = np.where(conv >= max_c - 1e-12)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            for i in range(i_start, i_end+1):
                j = k - i
                if i < j:
                    grad[i] += a[j]
                    grad[j] += a[i]
                elif i == j:
                    grad[i] += 2 * a[i]
        # average across max indices
        grad = grad / len(idxs)
        # step size schedule

    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0.0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0.0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential decay from the centre."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0.0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the ends."""
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_cosine(n, best_seq, best_score, deadline):
    """Half‑cosine window (zero at the edges, peak in the centre)."""
    idx = np.arange(n, dtype=np.float64)
    # shape = (1 - cos(pi * i/(n-1))) / 2  → zero at i=0,n-1, 1 at centre
    shape = 0.5 * (1.0 - np.cos(np.pi * idx / (n - 1)))
    for power in np.linspace(0.5, 3.0, 26):
        if time.time() >= deadline:
            break
        seq = np.power(shape, power)
        if seq.sum() == 0.0:
            continue
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Cosine power={power:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_triangular(n, best_seq, best_score, deadline):
    """Symmetric triangular shape."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    shape = 1.0 - np.abs(idx - centre) / centre
    shape = np.clip(shape, 0.0, None)
    for power in np.linspace(0.5, 3.0, 26):
        if time.time() >= deadline:
            break
        seq = np.power(shape, power)
        if seq.sum() == 0.0:
            continue
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Triangular power={power:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    # keep drawing until the deadline hits
    while time.time() < deadline:
        # concentration sampled log‑uniformly between 10⁻³ and 10¹
        logc = rng.uniform(-3.0, 1.0)
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Local refinement (sub‑gradient + simulated annealing)
# ----------------------------------------------------------------------
def _subgradient_refine(a0, deadline, best_seq, best_score):
    """
    Sub‑gradient descent on the true L∞ objective.
    After a few unsuccessful steps a small random jitter is added.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    iteration = 0
    # initial step size – will be adapted
    step = 0.5

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        max_c = conv.max()
        # indices attaining the maximum (allow tiny tolerance)
        idxs = np.where(conv >= max_c - 1e-14)[0]

        # ----- compute a sub‑gradient -----
        grad = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if i < j:
                    grad[i] += a[j]
                    grad[j] += a[i]
                else:          # i == j
                    grad[i] += 2.0 * a[i]

        # average over all maximal indices
        grad = grad / len(idxs)

        # ----- line‑search on the true objective -----
        cur_obj = 2.0 * n * max_c
        improved = False
        step_try = step
        for _ in range(12):          # backtrack up to 2⁻¹²
            a_new = a - step_try * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-12:        # strict improvement
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score - 1e-12:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SubGrad it={iteration:,} obj={obj_new:.6f}")
                break
            step_try *= 0.5

        if not improved:
            # add a tiny jitter to try to escape a flat region
            eps = 1e-8
            a = a + rng.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

        # slowly decay the step size
        step = max(step * 0.9995, 1e-8)

    return best_seq, best_score


def _simulated_annealing_refine(a0, deadline, best_seq, best_score):
    """
    Very cheap annealing: repeatedly move a tiny fraction of mass
    from a random source index to a random target index.
    Accept moves that improve the score or with a temperature‑dependent
    probability.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    temperature = 1e-2
    decay = 0.9997

    while time.time() < deadline:
        # propose a move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.uniform(0.0, min(0.5 * a[i], 1e-4))
        if delta == 0.0:
            continue
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta
        a_prop = _project_simplex(a_prop)

        sc_prop = _objective_score(a_prop)
        if sc_prop < best_score - 1e-12:
            best_score, best_seq = sc_prop, a_prop.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improve score={sc_prop:.6f}")
        # Metropolis acceptance
        cur_sc = _objective_score(a)
        if sc_prop < cur_sc or rng.random() < np.exp(-(sc_prop - cur_sc) / temperature):
            a = a_prop
        temperature = max(temperature * decay, 1e-6)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine performs cheap parametric sweeps, a massive
    Dirichlet random search and two local refinement stages (sub‑gradient
    descent followed by a light simulated‑annealing walk).  The best
    sequence (normalised to sum = 1) is returned as a list of Python floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096                      # a comfortable size for the time budget
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Quick parametric sweeps (≈150 s)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 400.0    # keep ~400 s for the heavy phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_cosine(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_triangular(n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈ 300 s)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After Dirichlet draws best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Local refinement on the simplex
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient refinement")
        best_seq, best_score = _subgradient_refine(best_seq, deadline, best_seq, best_score)

    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated‑annealing fine tune")
        best_seq, best_score = _simulated_annealing_refine(best_seq, deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
