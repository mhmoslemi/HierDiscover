"""
Propose a step‑function (non‑negative vector) that minimizes

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the full linear self‑convolution.
The method is completely different from the previous simulated‑annealing
approach:

1️⃣  **Random Dirichlet seeding** – many draws with a wide range of
    concentration parameters give diverse starting points.

2️⃣  **Smooth‐max gradient descent** – we replace the non‑differentiable
    max by a soft‑max (log‑sum‑exp) with an inverse temperature β that is
    slowly increased.  The gradient of the smoothed objective is computed
    via FFT‑based correlation, and an Adam‑style update (projected onto the
    probability simplex) drives the optimisation.  Symmetry (a[i]=a[n‑1‑i])
    is enforced after each step to reduce the search space.

3️⃣  **Final random local polishing** – cheap pairwise mass transfers that
    accept only strict improvements finish the run.

The whole routine respects the 1000 s wall‑time limit; whenever time runs
out the current best vector is returned.  All numbers are non‑negative,
normalised to sum to 1, and the final list is returned as plain Python
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing (set inside propose_candidate)
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (the denominator is 1 because we keep the vector on the simplex)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣ Random Dirichlet seeding
# ----------------------------------------------------------------------
def _random_dirichlet_one(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a single Dirichlet vector with a random concentration."""
    # concentration log‑uniform on [10⁻³, 10⁰·⁷] (≈0.001 … 5)
    conc_log = rng.uniform(-3.0, 0.7)
    c = 10.0 ** conc_log
    a = rng.gamma(shape=c, scale=1.0, size=n)
    if a.sum() == 0:
        return None
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 2️⃣ Smooth‑max Adam descent (projected onto the simplex, symmetric)
# ----------------------------------------------------------------------
def _smooth_max_adam(start_vec: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> (np.ndarray, float):
    """
    Perform Adam‑style projected gradient descent on the smoothed max
    objective.  β (inverse temperature) ramps from β₀≈0.5 to βₘₐₓ≈200,
    turning the smooth approximation into the true max.
    Symmetry (a[i]=a[n‑1‑i]) is enforced after every update.
    """
    a = start_vec.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05            # base learning rate

    # β schedule
    beta0 = 0.5
    beta_max = 200.0

    best_a = a.copy()
    best_score = _objective_score(a)

    iteration = 0
    total_budget = deadline - start_time

    while time.time() < deadline:
        # ---------- convolution ----------
        conv = _conv_self(a)                # length 2n‑1
        max_conv = conv.max()

        # ---------- smooth weights ----------
        # w ∝ exp(β (conv - max_conv))
        beta = beta0 + (beta_max - beta0) * ((time.time() - start_time) / total_budget)
        shifted = conv - max_conv
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient (via FFT‑based correlation) ----------
        # grad = 2 * correlation(w, a)   (valid part of length n)
        a_rev = a[::-1]
        L = _next_pow2(w.size + a_rev.size - 1)
        wf = np.fft.rfft(w, n=L)
        af = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(wf * af, n=L)          # length L
        grad = 2.0 * corr[n - 1:n - 1 + n]        # slice of length n

        # ---------- Adam update ----------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        # keep feasibility before projection
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        # enforce symmetry (a[i]=a[n‑1‑i])
        cand = (cand + cand[::-1]) * 0.5

        # ---------- evaluate ----------
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam new best {best_score:.6f}")

        a = cand

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣ Tiny random local refinement (polishing)
# ----------------------------------------------------------------------
def _random_local_refinement(a0: np.ndarray,
                             cur_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> (np.ndarray, float):
    """
    Very cheap random pairwise transfers (no temperature) – only accept
    strict improvements.  Acts as a final polishing step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        # move a tiny random fraction of the source mass
        delta = a[src] * rng.random() * 0.005
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (probability vector) that minimises

    """

    # ------------------------------------------------------------------
    n = 4096

    except Exception:

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_seq, best_score = cand, cand_score
                print(f"[{time.time() - start_time:.2f}s] Dirichlet seed new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam descent – repeatedly restart from best or random,

    # ------------------------------------------------------------------
    # Allocate about 80 % of the remaining time to the gradient stage.
    gradient_deadline = start_time + 0.80 * (deadline - start_time)
    run_id = 0
    while time.time() < gradient_deadline:
        # decide whether to start from current best or from a fresh random seed
        if rng.random() < 0.3:
            # fresh random start

            if init_vec is None:

        else:

        # reserve a small chunk for each run (≈150 s or whatever is left)

        run_deadline = time.time() + run_time
        print(f"[{time.time() - start_time:.2f}s] Adam run {run_id} (budget {run_time:.1f}s)")

        seq, sc = _smooth_max_adam(init_vec, run_deadline, rng)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] After Adam run {run_id} best = {best_score:.6f}")
        run_id += 1

    print(f"[{time.time() - start_time:.2f}s] Smooth‑max Adam finished – best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final random local polishing (≈30 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = min(deadline, time.time() + 30.0)
        print(f"[{time.time() - start_time:.2f}s] Starting random local refinement")
        best_seq, best_score = _random_local_refinement(best_seq, best_score,
                                                        local_deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] Local refinement finished – best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return (ensure strict feasibility for the evaluator)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:

    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
