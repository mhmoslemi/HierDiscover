def gradient_smooth(a, beta):
    n = a.shape[0]
    L = _next_pow2(3*n)  # length for correlation
    # compute FFT of a
    A = np.fft.rfft(a, n=L)
    # compute conv self for objective (not needed for grad)
    conv = np.fft.irfft(A * A, n=L)[: 2*n - 1]
    max_conv = conv.max()
    # compute softmax weights
    # To avoid overflow: shift
    exp_vals = np.exp(beta * (conv - max_conv))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    # compute correlation gradient part
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=L)
    # cross-correlation via conj(A) * W_rev
    corr = np.fft.irfft(np.conj(A) * W_rev, n=L)
    # extract g_j = corr[2*n - 2 - np.arange(n)]
    idx = 2*n - 2 - np.arange(n)
    g = corr[idx]   # this yields length n
    # gradient = 4 * n * g
    grad = 4.0 * n * g
    return conv, grad

def _gradient_smooth(a, beta):
    n = a.shape[0]
    L = _next_pow2(3 * n)  # for correlation
    # compute FFT of a (padded to L)
    A = np.fft.rfft(a, n=L)
    # compute self-convolution
    conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    max_conv = conv_full.max()
    # compute softmax weights
    # To avoid overflow: subtract max_conv
    # use double precision
    exp_vals = np.exp(beta * (conv_full - max_conv))
    w = exp_vals / exp_vals.sum()
    # compute gradient via cross-correlation
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=L)
    corr = np.fft.ifft(np.conj(A) * W_rev, n=L).real  # length L
    # extract needed part for indices 0..n-1
    idx = 2 * n - 2 - np.arange(n)  # integer array
    # j might index from start
    g = corr[idx]
    grad = 4.0 * n * g
    return conv_full, grad

def _smooth_projected_gradient_descent(a0, deadline, rng):
    a = a0.copy()
    # ensure normalized sum=1
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / a.sum()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    step = 0.01
    beta = 0.5
    beta_max = 500.0
    beta_mul = 1.02
    min_step = 1e-8
    no_improve_count = 0
    while time.time() < deadline:
        # compute conv and grad
        conv, grad = _gradient_smooth(a, beta)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve_count = 0
            print(f"[{time.time() - start_time:.2f}s] PGDSmooth new best {best_score:.6f} beta={beta:.2f}")
        else:
            no_improve_count += 1

        # try update
        # compute candidate
        cand = a - step * grad
        cand = _project_simplex(np.maximum(cand, 0.0))
        # compute objective of cand
        cand_score = _objective_score(cand)
        # acceptance
        if cand_score < cur_score:
            a = cand
            # increase step a bit
            step = min(step * 1.05, 1.0)
        else:
            # decrease step
            step = max(step * 0.5, min_step)
            # optional: random perturbation if step too small
            if step <= min_step:
                # add small random noise
                perturb = rng.normal(scale=0.001, size=n)
                a = a + perturb
                a = _project_simplex(np.maximum(a, 0.0))
                step = 0.01

        # increase beta
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        # early exit if no improve for many iterations and beta reached max
        if no_improve_count > 5000 and beta >= beta_max:
            # break out early
            break

    return best_a, best_score

"""
Optimisation via smoothed max‑gradient descent.

We minimise

    F(a) = 2·n·max(conv(a,a))

under a_i ≥ 0 and Σ a_i = 1 (the scale does not matter for the
objective).  The non‑differentiable max is replaced by a log‑sum‑exp
(soft‑max) with a temperature β.  For a given β the gradient of the
smoothed objective can be written in closed form and is evaluated with
FFT‑based cross‑correlation, which gives an O(n log n) step even for
n=4096.  Starting from a non‑uniform seed we iteratively:

1. compute the soft‑max weights of the self‑convolution,
2. obtain the gradient,
3. take a projected gradient step onto the probability simplex,
4. slowly increase β (so the smooth objective approaches the true max)
   and adapt the step size.

A few random Dirichlet/sparse seeds are tried (each refined for a short
time) and the best result is further refined for the remaining budget.
The algorithm finishes with a vector normalised to sum 1, which can be
evaluated directly by ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Global timing (set in propose_candidate)
# ----------------------------------------------------------------------
start_time: float = 0.0


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (v ≥ 0, Σv = 1).  The implementation follows the algorithm of
    Wang‑Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """
    Exact objective value (sum a = 1) : 2·n·max(conv(a,a)).
    Uses an FFT of power‑of‑2 length.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _gradient_smooth(a: np.ndarray, beta: float):
    """
    Compute (conv(a,a), ∇_a 2·n·log‑sum‑exp(β·conv(a,a))) .
    The gradient is obtained via FFT‑based cross‑correlation.
    Returns the full self‑convolution (length 2·n‑1) and the gradient.
    """
    n = a.shape[0]
    # length sufficient for both convolution (2n‑1) and correlation (3n‑2)
    L = _next_pow2(3 * n)

    # FFT of a (zero‑padded to length L)
    A = np.fft.rfft(a, n=L)

    # ==== self‑convolution ====
    conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

    # ==== soft‑max weights ====
    max_conv = conv_full.max()
    # shift by max to avoid overflow
    exp_vals = np.exp(beta * (conv_full - max_conv))
    w = exp_vals / exp_vals.sum()                # shape (2n‑1,)

    # ==== cross‑correlation a * w ====
    w_rev = w[::-1]                              # reverse for correlation
    W_rev = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(np.conj(A) * W_rev, n=L)  # length L, real‑valued

    # extract g_j = Σ_i a_i w_{i+j} for j=0..n‑1
    idx = 2 * n - 2 - np.arange(n)               # Python broadcasting
    g = corr[idx]

    # gradient of the smoothed objective (factor 4·n comes from derivation)
    grad = 4.0 * n * g
    return conv_full, grad


def _sparse_initial(n: int, sparsity_ratio: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random sparse vector with roughly ``sparsity_ratio`` non‑zero
    entries; non‑zero values are exponential and the vector is normalised.
    """
    k = max(2, int(sparsity_ratio * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed projected gradient descent (core optimiser)
# ----------------------------------------------------------------------
def _smooth_pgdescent(start_vec: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> (np.ndarray, float):
    """
    Refine a starting vector using the smoothed‑max gradient method.
    The routine returns the best vector found before the deadline.
    """
    a = np.maximum(start_vec, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s

    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)

    # algorithmic parameters
    step = 1e-4                      # initial step size
    beta = 5.0                       # start of the smoothing temperature
    beta_max = 500.0
    beta_mul = 1.015                 # how fast beta grows
    min_step = 1e-12
    no_improve = 0

    while time.time() < deadline:
        # ----- compute smoothed objective and its gradient -----
        conv, grad = _gradient_smooth(a, beta)
        cur_score = float(2.0 * n * conv.max())

        # ----- keep the best seen so far -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f} beta={beta:.2f}")
        else:
            no_improve += 1

        # ----- line‑search on a few step multipliers -----
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            step_sz = step * factor
            cand = a - step_sz * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score - 1e-12:
                a = cand
                step = step_sz * 1.05          # slightly enlarge after success
                improved = True
                break
        if not improved:
            step = max(step * 0.5, min_step)   # shrink step if no progress

        # ----- increase smoothing temperature slowly -----
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        # ----- occasional random jitter to escape plateaus -----
        if no_improve >= 2000:
            jitter = rng.normal(scale=0.001, size=n)
            a = _project_simplex(np.maximum(a + jitter, 0.0))
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Random jitter applied")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The method uses many random starts followed by
    a smoothed‑max projected gradient descent.  It respects the time budget
    and returns the best vector found (as a plain list of floats).
    """
    global start_time
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # safety margin

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # problem size – 4096 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4096

    # ---- start from the supplied global starter if it exists ----
    best_seq = np.full(n, 1.0 / n, dtype=float)    # uniform fallback
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – diversified random starts, each refined briefly
    # --------------------------------------------------------------
    init_deadline = start_time + 120.0   # spend ~2 min on broad exploration
    while time.time() < init_deadline:
        # Choose a random initial distribution
        if rng.random() < 0.7:
            # Dirichlet‑like draw with a wide concentration range
            conc_log = rng.uniform(-3.0, 0.7)
            c = 10.0 ** conc_log
            a = rng.gamma(shape=c, scale=1.0, size=n)
            if a.sum() == 0.0:
                continue
            a /= a.sum()
        else:
            # a sparse start
            a = _sparse_initial(n, sparsity_ratio=rng.uniform(0.005, 0.02), rng=rng)

        # short refinement (≈2 s per start)
        inner_deadline = time.time() + 2.0
        seq, sc = _smooth_pgdescent(a, inner_deadline, rng)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] Improved after random start: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long refinement from the best found vector
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting long refinement")
        best_seq, best_score = _smooth_pgdescent(best_seq, deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] Long refinement finished – best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation (protect against tiny negatives)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
