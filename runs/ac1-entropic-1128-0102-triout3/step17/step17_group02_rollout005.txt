Propose a step‑function that (hopefully) beats the previous bound 1.507818.

The algorithm combines three ideas that differ from the reference code:

1️⃣  **Soft‑max re‑parameterisation** – we optimise a log‑density `x`,
    transformed to a probability vector by a numerically stable soft‑max.
    This removes the need for an explicit projection onto the simplex.

2️⃣  **Adam optimisation on a smooth surrogate** – the non‑smooth

    temperature β.  A β‑schedule (0.5  →  …  → 5 000) lets the optimiser
    first see a gentle landscape and then converge to the true peak.
    Adam’s per‑coordinate step‑size adaption is far more robust than a
    plain line‑search.

3️⃣  **Guided hill‑climbing fine‑tuning** – after the Adam phase we perform
    cheap random mass‑transfer moves that are accepted only when they
    improve the exact (non‑smooth) objective.  This often pushes the

The whole search runs under the 1000 s budget, respects the non‑negative

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using an FFT of power‑of‑2 length."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    shift = np.max(z)
    e = np.exp(z - shift)
    return e / e.sum()


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max_and_weights(conv: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) via log‑sum‑exp.
    Returns (smooth_max, softmax_weights) where the weights have the same
    length as `conv` and sum to 1.
    """
    m = conv.max()
    # subtract the maximum for numerical stability
    exps = np.exp(beta * (conv - m))
    sum_exps = exps.sum()
    smooth = m + np.log(sum_exps) / beta
    w = exps / sum_exps
    return smooth, w


def _grad_a_from_w(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. the probability vector a.
    Implements  grad = 2 * (a_rev ⋆ w) restricted to the original length.
    """
    rev_corr = _convolve_fft(a[::-1], w)          # length 3·n‑2
    n = a.shape[0]
    # slice the centre part corresponding to the original n entries
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


def _grad_x_from_a(a: np.ndarray, grad_a: np.ndarray) -> np.ndarray:
    """
    Chain rule for the soft‑max re‑parameterisation:
        ∂f/∂x_j = a_j * (∂f/∂a_j - Σ_i a_i ∂f/∂a_i)
    """
    inner = np.dot(a, grad_a)            # scalar ⟨a,∇_a f⟩
    return a * (grad_a - inner)
