def _conv_fft(x, y):
    L = _next_pow2(len(x)+len(y)-1)
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)
    return conv[:len(x)+len(y)-1]

def _subgrad_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    step = 1.0
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_c = conv.max()
        obj = 2.0 * n * max_c
        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgradient it={it} obj={obj:.6f} step={step:.3e}")
        # compute mask for max conv
        eps = max_c * 1e-12
        mask = (conv >= max_c - eps).astype(float)
        count = mask.sum()
        if count == 0:
            mask = (conv == max_c).astype(float)
            count = mask.sum()
            if count == 0:
                # no mask, break
                break
        M = mask / count
        # gradient
        grad = _subgradient(a, M)
        # line search
        cur_obj = obj
        s_local = step
        improved = False
        while s_local > 1e-12:
            a_new = a - s_local * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-14:
                a = a_new
                step = s_local * 0.9  # slightly reduce step for next iteration
                improved = True
                break
            s_local *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-15:
                # jitter small random
                a = a + np.random.uniform(-1e-8, 1e-8, size=n)
                a = _project_simplex(a)
                step = 1.0
        it += 1
        # optional break condition based on iteration count
        if it % 2000 == 0:
            # maybe log
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter {it}, best obj {best_score:.6f}")
    return best_seq, best_score

def _smooth_max_descent_iter(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    cur_conv = _conv_self(a)
    max_conv = cur_conv.max()
    tau = max(max_conv * 5.0, 0.01)  # start with larger tau
    min_tau = 1e-9
    decay = 0.994
    step = 1.0
    it = 0

    while time.time() < deadline and tau >= min_tau:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        # compute smoothed p
        shift = max_conv
        log_w = (conv - shift) / tau
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0:
            tau *= decay
            continue
        p = w / w_sum
        # gradient of smooth max
        grad = _smooth_gradient(a, p) # compute 2 * conv(p, a_rev), length n
        # line search
        cur_obj = 2 * n * max_conv
        step_local = step
        improved = False
        while step_local > 1e-14:
            a_new = a - step_local * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2 * n * conv_new.max()
            if obj_new < cur_obj - 1e-12:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_seq, best_score = a.copy(), obj_new
                    print(f"[{time.time() - start_time:.2f}s] Smooth descent it={it} obj={obj_new:.6f} tau={tau:.2e}")
                break
            step_local *= 0.5
        if not improved:
            # jitter
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = _project_simplex(a)
        tau *= decay
        # adapt step size
        if improved:
            step = step_local * 1.2
        else:
            step = step * 0.8

    return best_seq, best_score

"""
A multi‑stage optimizer for the objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm proceeds as follows:

1️⃣  **Cheap parametric sweeps** – geometric, power, exponential,
    two‑block, triangular and cosine families.  The best few candidates
    are remembered.

2️⃣  **Random Dirichlet exploration** – draws from the simplex with a
    wide range of concentration parameters.

3️⃣  **Smooth‑max descent** – minimise the differentiable surrogate
    τ·log ∑exp(conv/τ).  The temperature τ starts large (≈0.1) so the
    landscape is very smooth, is slowly annealed (τ←0.995·τ) and the
    true objective is used for a back‑tracking line‑search.

4️⃣  **Multi‑restart** – the smooth‑max stage is re‑started from a set
    of random points (subject to the remaining time budget).

5️⃣  **Sub‑gradient polishing** – after the temperature becomes tiny the
    algorithm switches to a pure sub‑gradient descent on the true
    max‑convolution (average of the gradients at the active indices).

All iterates are projected onto the probability simplex after each
step, guaranteeing non‑negativity and unit sum.  The best sequence found
within the 1000 s budget is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    z = np.fft.irfft(Xf * Yf, n=L)
    return z[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Objective assuming the sequence is already normalised (sum=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient utilities for the smoothed objective
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of τ·log∑exp(conv/τ) with respect to a (τ already factored out).
    It equals 2·conv(p, a_rev) on the central window.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_full = _conv_fft(p, a_rev)          # length 3·n‑2
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]  # central n entries
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, M: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) given the weight vector M (sum M = 1,
    support only on indices attaining the current maximum).  Returns the
    gradient of the *max* term (without the outer factor 2·n).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv1_full = _conv_fft(M, a_rev)       # length 3·n‑2
    s1 = conv1_full[n - 1 : n - 1 + n]

    # sliding sum of M over a window of length n
    ones = np.ones(n)
    conv2_full = _conv_fft(M, ones)
    s = conv2_full[n - 1 : n - 1 + n]

    # derivative of conv_i w.r.t a_j = 2·a_{i‑j}  (when i‑j in range)
    # the factor 2 is applied later by the caller
    return s1 + a * s

# 1️⃣ Parametric sweeps
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for eps in np.logspace(-5, -1, 80):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        # mix with uniform – a few mixing ratios
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential decay from the centre."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the ends with a tunable split."""
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_triangle(n, best_seq, best_score, deadline):
    """Triangular “tent” shape centred at the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    base = 1.0 - np.abs(idx - centre) / centre            # 1 at centre, 0 at ends
    for p in np.linspace(0.0, 4.0, 41):
        if time.time() >= deadline:

        seq = np.power(base, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Triangle p={p:.2f} score={sc:.6f}")
    return best_seq, best_score

    idx = np.arange(n, dtype=np.float64)

    base = 0.5 * (1.0 + np.cos(np.pi * (idx - centre) / centre))  # max 1, zero at ends

def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 2000
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration c sampled log‑uniformly between 0.01 and 5
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)

        if s == 0:

        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Smooth‑max gradient descent (annealed temperature)
# ----------------------------------------------------------------------
def _smooth_max_descent(a0, sub_deadline, best_seq, best_score,
                        init_tau=0.1, decay=0.995, min_tau=1e-9):
    """
    Annealed smooth‑max descent.
    - init_tau : starting temperature (large → very smooth)
    - decay    : geometric factor τ←τ·decay each iteration
    - min_tau  : stop when τ drops below this value
    """
    a = a0.copy()
    n = a.shape[0]
    tau = max(init_tau, 0.1)            # ensure a decent starting temperature
    step = 1.0
    iteration = 0

    while time.time() < sub_deadline and tau > min_tau:
        iteration += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv

        # --- smoothed weights (softmax) ---
        shift = max_conv
        log_w = (conv - shift) / tau
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0.0:
            tau *= decay
            continue
        p = w / w_sum

        # --- gradient of the smooth surrogate ---
        grad = _smooth_gradient(a, p)
        # Scale to the true objective (outer factor 2·n)
        grad *= (2.0 * n)

        # --- back‑tracking line‑search on the true objective ---
        improved = False
        step_local = step
        while step_local > 1e-14:
            a_new = a - step_local * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-13:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Smooth τ={tau:.2e} it={iteration:,} obj={obj_new:.6f}")
                break
            step_local *= 0.5

        if not improved:
            # small jitter to escape plateaus
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = _project_simplex(a)
            step = max(step * 0.8, 1e-6)
        else:
            step = step_local * 1.2

        tau *= decay
    return best_seq, best_score

# 4️⃣ Pure sub‑gradient descent (used after τ becomes tiny)
# ----------------------------------------------------------------------
def _subgradient_descent(a0, sub_deadline, best_seq, best_score):
    """Sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    iteration = 0
    step = 1.0

    while time.time() < sub_deadline:
        iteration += 1
        conv = _conv_self(a)
        max_c = conv.max()
        cur_obj = 2.0 * n * max_c

        if cur_obj < best_score:
            best_score, best_seq = cur_obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad it={iteration:,} obj={cur_obj:.6f}")

        # identify active indices (within a tiny relative tolerance)
        tol = max_c * 1e-12
        mask = (conv >= max_c - tol).astype(float)
        cnt = mask.sum()
        if cnt == 0:
            mask = (conv == max_c).astype(float)
            cnt = mask.sum()
            if cnt == 0:
                break
        M = mask / cnt

        # gradient of max (without outer factor)
        g = _subgradient(a, M)
        # gradient of the full objective = 2·n·g
        grad = 2.0 * n * g

        # line‑search
        improved = False
        step_local = step
        while step_local > 1e-14:
            a_new = a - step_local * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-13:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Subgrad update it={iteration:,} obj={obj_new:.6f}")
                break
            step_local *= 0.5

        if not improved:
            step *= 0.7
            # occasional jitter
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = _project_simplex(a)
        else:
            step = step_local * 1.1

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine performs cheap parametric sweeps, random
    Dirichlet exploration, an annealed smooth‑max gradient descent with
    multiple restarts, and a final sub‑gradient polishing phase.
    The best sequence (normalised to sum = 1) is returned as a list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Parametric sweep (≈200 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0   # keep ~300 s for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_triangle(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_cosine(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 150.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Smooth‑max descent (large block, then multi‑restart)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # a) long descent starting from the current best
        sub_deadline = min(deadline, time.time() + 300.0)   # up to 5 min
        print(f"[{time.time() - start_time:.2f}s] Smooth‑max descent (initial) – "
              f"budget {sub_deadline - time.time():.1f}s")
        best_seq, best_score = _smooth_max_descent(best_seq, sub_deadline,
                                                   best_seq, best_score)

        # b) several short restarts from random points
        while time.time() < deadline - 60.0:   # keep at least a minute for polishing
            remaining = deadline - time.time()
            block = min(90.0, remaining)       # each restart gets ≤90 s
            # generate a fresh random Dirichlet start
            rng = np.random.default_rng()
            c = 10 ** rng.uniform(-2.0, 0.7)
            a0 = rng.gamma(shape=c, scale=1.0, size=n)
            a0 /= a0.sum()
            print(f"[{time.time() - start_time:.2f}s] Restarting smooth‑max (τ≈{0.1:.2e})")
            best_seq, best_score = _smooth_max_descent(a0, time.time() + block,
                                                       best_seq, best_score)

    # --------------------------------------------------------------
    # 4️⃣ Sub‑gradient polishing (use any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.2f}s] Sub‑gradient polishing")
        best_seq, best_score = _subgradient_descent(best_seq, deadline,
                                                    best_seq, best_score)

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
