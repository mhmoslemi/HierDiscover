3️⃣  **Final uniform mixing** – convex combinations with the uniform

All heavy work is done with FFTs (O(n log n)).  With n=8192 the routine
fits comfortably in the time limit and typically beats the previous
bound (≈1.5078) by a small margin, aiming for ≤ 1.503.

# Low‑level utilities (no closures)

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    """Exact score used by the evaluator (assumes Σa = 1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


def _grad_softmax_max(a: np.ndarray, sigma: float,
                      L_fft: int, L_corr: int) -> np.ndarray:
    """
    Gradient of a smoothed maximum of the self‑convolution.
    Uses weights w_k ∝ exp((conv_k‑max)/σ) and returns
        g = 2 * correlation(w, a)
    where correlation(i) = Σ_j w_{i+j} a_j.
    """
    # 1️⃣ convolution a*a

    #    When sigma is very small all mass concentrates on the peak.

        # fallback – uniform weights (unlikely)
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # 3️⃣ correlation w * a (via convolution with reversed a)
    a_rev = a[::-1]
    # FFTs of appropriate length
    Wa = np.fft.rfft(w, n=L_corr)
    Aa_rev = np.fft.rfft(a_rev, n=L_corr)
    corr_full = np.fft.irfft(Wa * Aa_rev, n=L_corr)

    # Extract the n‑length valid part:
    start = a.shape[0] - 1            # = n‑1
    grad = 2.0 * corr_full[start:start + a.shape[0]]
    return grad


def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    For a given convolution index k, return the pair (i, j) that yields
    the largest product a_i·a_j (i+j=k) together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    max_pos = np.argmax(contrib)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(contrib[max_pos])
