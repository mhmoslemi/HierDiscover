"""
Propose a step‑function (non‑negative vector) that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The procedure consists of four stages:

1️⃣  Warm‑start – a mixture of the public starter (if available), the uniform
    distribution and many Dirichlet samples.  The best cheap candidate is kept.

2️⃣  Smoothed‑max projected gradient descent – we minimise a smooth approximation
    of the max‑convolution:  τ·log(∑ exp(conv/τ)).  The gradient of this loss
    is obtained analytically via a single FFT‑based convolution.  After each
    gradient step we project back onto the probability simplex (non‑negative,
    unit sum) with the Duchi projection.  The smoothing parameter τ is reduced
    geometrically, driving the optimisation towards the true ∞‑norm.

3️⃣  Aggressive targeted hill‑climb – while time permits we repeatedly locate the
    index k where the convolution attains its current maximum, take the pair
    (i,j) that contributes most to this entry and relocate a tiny amount of
    mass from *both* i and j to two random other coordinates.  Only improving
    moves are kept.

4️⃣  Final refinement – convex mixing with the uniform vector and a light random‑pair
    hill‑climb, exactly as in the reference implementation.

All stages respect the 1000 s budget.  The returned list is normalised to a
unit sum (the score is scale‑invariant).  The algorithm focuses directly on the
∞‑norm of the convolution, which is the quantity the evaluator penalises,
and therefore typically improves on the previous variance‑flattening approach.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global time‑keeping (used by helper functions for printing)
# ----------------------------------------------------------------------
start_time = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # corner case – all entries zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """
    Exact evaluation used by ``evaluate_sequence`` for a unit‑sum vector.
    score = 2·n·max(conv(a,a))
    """
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    For a given convolution index k, return the pair (i, j) with the
    largest product a_i·a_j (where i+j=k) together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    max_pos = np.argmax(contrib)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(contrib[max_pos])

# ----------------------------------------------------------------------
def _warm_start(
    n: int,
    best_a: np.ndarray,
    best_score: float,
    L_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sample many Dirichlet‑type vectors (i.e. Gamma draws) and keep the best.
    """
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration drawn log‑uniformly in a sensible range
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(

            )
    return best_a, best_score


# ----------------------------------------------------------------------
# 1️⃣  Smoothed‑max projected gradient descent
# ----------------------------------------------------------------------
def _smoothed_max_descent(
    a0: np.ndarray,
    n: int,
    L_fft: int,
    L_corr: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv) using projected
    gradient descent.  The smoothing parameter τ is decreased geometrically.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    # smoothing schedule
    tau = 1.0e-3          # start a little above typical conv magnitude
    tau_min = 1.0e-8
    tau_decay = 0.9995    # multiplicative decay per iteration
    it = 0

    while time.time() < deadline:
        it += 1

        # convolution of the current iterate
        conv = _conv_self_fft(a, L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] MaxGD it={it:,} score={cur_score:.6f}")

        # decay τ
        tau = max(tau * tau_decay, tau_min)

        # softmax weights over the 2·n‑1 convolution entries
        shift = conv.max()
        # exponentiation in double precision; subtract shift to avoid overflow
        exp_arg = (conv - shift) / tau
        # clip to a safe range – extreme negatives just become zero
        exp_arg = np.clip(exp_arg, -700, 0)  # np.exp(-700) ≈ 5e-305
        w_raw = np.exp(exp_arg, dtype=np.float64)
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            # should not happen, but guard against numerical collapse
            p = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            p = w_raw / w_sum

        # gradient of the smoothed objective:
        #   ∇ = 2 * (a * p)   (full linear convolution, then take first n entries)
        Ar = np.fft.rfft(a, n=L_corr)
        Pr = np.fft.rfft(p, n=L_corr)
        grad_full = np.fft.irfft(Ar * Pr, n=L_corr)[:n] * 2.0

        # step size – 0.5 / ||grad|| gives a reasonably sized move
        grad_norm = np.linalg.norm(grad_full)
        if grad_norm == 0.0:
            # nothing to do – break early
            break
        step = 0.5 / grad_norm

        # simple back‑tracking line search (max 8 reductions)
        for _ in range(8):
            cand = a - step * grad_full
            cand = _project_simplex(cand)
            cand_score = _objective_score(cand, n, L_fft)
            if cand_score < cur_score:
                a = cand
                break
            step *= 0.5
        else:
            # no improvement – inject a tiny random jitter to escape flat region
            eps = np.random.normal(scale=1.0e-4, size=n)
            a = _project_simplex(a + eps)

        # occasional random jitter (helps avoid tiny plateaus)
        if np.random.rand() < 0.001:
            eps = np.random.normal(scale=5.0e-4, size=n)
            a = _project_simplex(a + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Aggressive targeted hill‑climb
# ----------------------------------------------------------------------
def _targeted_hill_climb(
    a: np.ndarray,
    n: int,
    L_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    While time permits we attack the current maximal convolution entry:
    locate the pair (i,j) that contributes most to that entry and move a tiny
    amount of probability mass from *both* i and j to two other random locations.
    Only strictly improving moves are accepted.
    """
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        i, j, _ = _max_contributions(best_a, k_max)
        if i == -1:
            continue

        # Choose two destination indices distinct from i and j
        dst_pool = np.delete(np.arange(n), [i, j])
        if dst_pool.size < 2:
            continue
        dst1, dst2 = rng.choice(dst_pool, size=2, replace=False)

        # Transfer a small amount from both sources to both destinations
        max_delta = min(best_a[i], best_a[j], 0.003)  # adaptively small
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] -= delta
        cand[dst1] += delta
        cand[dst2] += delta

        # Renormalise (project onto simplex is simply normalisation here)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        new_score = _objective_score(cand, n, L_fft)
        if new_score < best_score:
            best_score, best_a = new_score, cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Hill i={i},j={j} → d1={dst1},d2={dst2} δ={delta:.6f} score={new_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final mixing + light random‑pair hill‑climb
# ----------------------------------------------------------------------
def _final_refinement(
    a: np.ndarray,
    n: int,
    L_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Convex mixing with the uniform vector and a cheap random‑pair hill‑climb."""
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex interpolation with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}"
            )

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Light climb i={i} j={j} δ={delta:.6f} → score={sc:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget and
    returns a plain Python list of floats (unit sum).
    """
    global start_time
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 offers a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n - 2)         # for a * p (gradient step)

    # ------------------------------------------------------------------
    # 0️⃣  Baseline initialiser (uniform) and optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # try the provided global starter (height_sequence_1) if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        # no global starter available
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start with Dirichlet draws (≈5 % of the total budget)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * (deadline - start_time))
    if time.time() < warm_deadline:
        best_a, best_score = _warm_start(
            n, best_a, best_score, L_fft, warm_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed‑max projected gradient descent (~70 % of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        grad_deadline = time.time() + 0.70 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed‑max descent")
        best_a, best_score = _smoothed_max_descent(
            best_a, n, L_fft, L_corr, grad_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After descent best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Aggressive targeted hill‑climb (~20 % of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + 0.20 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Targeted hill‑climb")
        best_a, best_score = _targeted_hill_climb(
            best_a, n, L_fft, hill_deadline
        )

    # ------------------------------------------------------------------
    # 4️⃣  Final mixing + light random‑pair hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement")
        best_a, best_score = _final_refinement(
            best_a, n, L_fft, deadline
        )

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
