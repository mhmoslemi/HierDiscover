"""
Propose a non‑negative step‑function (probability vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  Initialise a pool of candidates from several parametric families
    (geometric, exponential, arcsine‑like, two‑block, random Dirichlet).
    The best candidate is kept.

2️⃣  Refine the best candidate with a smooth‑max (log‑sum‑exp) surrogate
    of the max‑convolution.  We use exponentiated‑gradient (mirror‑descent)
    on the probability simplex, slowly increasing the smoothness parameter
    β up to a large value (≈2000) so that the surrogate approaches the true
    max.  A back‑tracking line‑search adapts the step‑size η and occasional
    tiny random perturbations help to escape plateaus.

3️⃣  A cheap local hill‑climb: repeatedly move a tiny amount of mass from
    one coordinate to another and keep the move if the true objective
    improves.

All heavy work (self‑convolution and the gradient) is performed with FFTs.
The routine respects the 1000 s wall‑clock budget: the phases are allocated
dynamically and the function returns the best sequence found before the
deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Returns the exact objective 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient) where gradient is ∂/∂a of the smooth max.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)  # stability shift
    Z = exp_scaled.sum()
    w = exp_scaled / Z                         # soft‑max weights, sum to 1

    # Gradient of Σ w_k * conv_k w.r.t a is 2 * (a convolved with reversed w)
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)            # length 3n‑2
    start = len(w) - 1                         # = 2n-2
    G = 2.0 * conv_rev[start - np.arange(n)]

    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, G


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponential (mirror‑descent) update on the simplex:
        a_i ← a_i·exp(−η·grad_i), then renormalise.
    Guarantees a_i ≥ 0 and Σ a_i = 1.
    """
    log_factor = -eta * grad
    # shift to avoid overflow
    log_factor -= np.max(log_factor)
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total

# Initial families (cheap sweeps)
# ----------------------------------------------------------------------
def _geometric_family(n, best_seq, best_score, deadline, start_time):
    idx = np.arange(n, dtype=np.float64)
    for eps in np.logspace(-6, -0.5, 100):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        seq = np.power(r, idx)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} score={sc:.6f}")
    return best_seq, best_score


def _exponential_family(n, best_seq, best_score, deadline, start_time):
    idx = np.arange(n, dtype=np.float64)
    for lam in np.logspace(-4, -0.1, 120):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * idx)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _arcsine_family(n, best_seq, best_score, deadline, start_time):
    """Weights proportional to 1/√(x·(1‑x)) – heavy at the ends."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    seq = w / w.sum()
    sc = _objective_score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Arcsine shape score={sc:.6f}")
    return best_seq, best_score


def _two_block_family(n, best_seq, best_score, deadline, start_time):
    """Two equal‑size blocks at the extremes."""
    for L_frac in np.linspace(0.02, 0.30, 25):
        L = max(1, int(L_frac * n))
        if time.time() >= deadline:
            break
        for w in np.linspace(0.05, 0.95, 19):
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Two‑blocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _random_dirichlet_trials(n, best_seq, best_score, deadline, start_time, trials=6000):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # Log‑uniform concentration in [1e‑2, 5]
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score

# Smooth‑max gradient descent (core optimiser)
# ----------------------------------------------------------------------
def _smooth_descent(a0, deadline, start_time, best_seq, best_score):
    a = a0.copy()
    cur_score = _objective_score(a)
    n = a.shape[0]

    beta_start, beta_end = 0.5, 2000.0       # smoothness schedule (exponential)
    eta = 0.5                                 # initial step size
    min_eta = 1e-12
    stagnation = 0

    while time.time() < deadline:
        # exponential schedule for β (more aggressive later)
        t = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start * (beta_end / beta_start) ** t

        smooth, grad = _smooth_max_and_grad(a, beta)

        # back‑tracking line search on the true objective
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, grad, eta_try)
            new_score = _objective_score(a_new)
            if new_score < cur_score - 1e-9:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} η={eta_try:.2e} score={new_score:.6f}")
            eta = min(1.0, eta * 1.02)          # gently increase step size
            stagnation = 0
        else:
            # mild random jitter to escape flat regions
            rng = np.random.default_rng()
            eps = 1e-4 * rng.random(a.shape) * a
            a = a + eps
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.5)
            stagnation += 1

            # occasional random restart after many non‑improving steps
            if stagnation >= 100:
                a = rng.dirichlet(np.ones(n))
                a = a / a.sum()
                cur_score = _objective_score(a)
                stagnation = 0

        # optional early exit (once we are clearly below the target)
        if best_score <= 1.48:
            break

    return best_seq, best_score

def _local_hill_climb(seq, best_score, deadline, start_time):
    """Tiny mass‑transfer moves that improve the true objective."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = best_score
    rng = np.random.default_rng()
    attempts_without_gain = 0

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # move a small fraction of mass from i to j
        eps = 0.001 * a[i]
        if eps <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        new_score = _objective_score(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            attempts_without_gain = 0
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {new_score:.6f}")
        else:
            attempts_without_gain += 1
            if attempts_without_gain > 5000:
                break
    return a, best_score
