def hill_climb(a, max_iter, deadline):
    n = len(a)
    L = 2*n-1
    conv = np.convolve(a, a)
    max_conv = conv.max()
    best_seq = a.copy()
    best_score = 2*n*max_conv
    it=0
    while time.time() < deadline and it < max_iter:
        it+=1
        # pick random i with a_i>0
        i = rng.integers(0, n)
        if a[i] <= 1e-12: continue
        j = rng.integers(0, n)
        if i==j: continue
        # choose delta up to a[i]
        delta = a[i] * rng.uniform(0.01, 0.2) # 1% to 20% of a[i]
        # compute new conv
        conv_new = conv.copy()
        # conv[i:i+n] -= 2*delta*a
        conv_new[i:i+n] -= 2.0 * delta * a
        conv_new[j:j+n] += 2.0 * delta * a
        conv_new[2*i] += delta*delta
        conv_new[2*j] += delta*delta
        conv_new[i+j] -= 2.0 * delta*delta
        new_max = conv_new.max()
        if new_max < max_conv - 1e-12:
            # accept
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            max_conv = new_max
            score = 2*n*max_conv
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        # else reject (do nothing)
    return best_seq, best_score
```

"""
Proposes a non‑negative step‑function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of four phases (all respecting a 1000 s budget):

1️⃣  A rich set of parametric sweeps (geometric, reverse geometric,
    power, exponential, cosine‑shaped and two‑block patterns).

2️⃣  Massive random Dirichlet draws (different concentration parameters)
    to explore the simplex.

3️⃣  A multi‑restart smooth‑max gradient descent (τ·log‑sum‑exp surrogate)
    with a slow geometric cooling schedule and a simple line‑search.

4️⃣  Final discrete hill–climbing: small mass transfers between random
    coordinates are evaluated by an O(n) convolution‑update (no FFT).
    Only improving moves are kept (optional simulated‑annealing style
    jitter is injected when progress stalls).

The best sequence (normalised to sum = 1) found during the run is returned
as a plain Python list of floats.
"""
import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _smooth_max_gradient_descent(a0, deadline, best_seq, best_score):
    """
    Projected gradient descent on τ·log‑sum‑exp of the convolution.
    Temperature τ is geometrically cooled.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    tau = max(max_conv * 0.5, 1e-8)          # start a bit smoother than before
    min_tau = 1e-12
    decay = 0.999                         # slower cooling

    it = 0
    while time.time() < deadline and tau > min_tau:
        it += 1
        conv = _conv_self(a)               # fresh conv at current point
        max_conv = conv.max()
        # ---- smooth max weights (softmax) ----
        shift = max_conv                     # for stability
        log_w = (conv - shift) / tau
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0:
            tau *= decay
            continue
        p = w / w_sum

        # ---- gradient of smooth max ----
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]
        grad = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # ---- line search on the true objective ----
        cur_obj = 2.0 * n * max_conv
        step = 1.0
        improved = False
        while step > 1e-14:
            a_new = a - step * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-12:
                a = a_new
                conv = conv_new
                max_conv = conv_new.max()
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SmoothGD it={it:,} obj={obj_new:.6f} τ={tau:.2e}")
                break
            step *= 0.5

        if not improved:
            # small random jitter to escape flat plateaus
            eps = 1e-8
            a += np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

        tau *= decay
    return best_seq, best_score

def _hill_climb(a, max_iter, deadline):
    """
    Small random transfers of mass between two coordinates,
    updating the full convolution in O(n) using slice operations.
    Only strictly improving moves are accepted.
    """
    n = a.shape[0]
    L = 2 * n - 1
    conv = np.convolve(a, a)               # exact full convolution (numpy uses FFT internally for large sizes)
    max_conv = conv.max()
    best_seq = a.copy()
    best_score = 2.0 * n * max_conv

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # pick a source with some mass
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        # pick a destination (different)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # choose a transfer amount (1 %–20 % of source mass)
        delta = a[i] * rng.uniform(0.01, 0.20)
        # build the updated convolution (O(n) work)
        conv_new = conv.copy()
        # subtract the contribution of the moved mass at source
        conv_new[i:i + n] -= 2.0 * delta * a
        # add the contribution at destination
        conv_new[j:j + n] += 2.0 * delta * a
        # quadratic correction for the two self‑terms and the cross‑term
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        if new_max < max_conv - 1e-14:          # strictly better
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            max_conv = new_max
            cur_score = 2.0 * n * max_conv
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} score={cur_score:.6f}")
        # if not improving, simply continue – we keep the old state
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given evaluation
    function.  The routine performs parametric sweeps, massive random Dirichlet
    sampling, several restarts of a smooth‑max gradient descent and a final
    discrete hill‑climbing refinement.  The best sequence (normalised to sum = 1)
    is returned as a list of Python floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # safety margin

    # --------------------------------------------------------------
    # Problem size – use the length of the supplied global starter if
    # it exists, otherwise fall back to 4096.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
        else:
            raise Exception
    except Exception:
        n = 4096

    # --------------------------------------------------------------
    # Baseline – uniform distribution
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (if it beats uniform)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Rich parametric sweep (≈200 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_reverse_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_cosine(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 150.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Smooth‑max gradient descent – several restarts (remaining time)
    # --------------------------------------------------------------
    while time.time() < deadline - 30.0:      # keep ~30 s for the final hill‑climb
        # optionally perturb the current best to generate a fresh start
        a0 = best_seq.copy()
        if np.random.rand() < 0.15:
            # replace a fraction of the mass with a random Dirichlet
            rng = np.random.default_rng()
            c = rng.uniform(0.01, 5.0)
            a_rand = rng.gamma(shape=c, scale=1.0, size=n)
            a_rand /= a_rand.sum()
            a0 = 0.3 * a0 + 0.7 * a_rand
            a0 = _project_simplex(a0)
        best_seq, best_score = _smooth_max_gradient_descent(a0, deadline, best_seq, best_score)

    # --------------------------------------------------------------
    # 4️⃣ Final discrete hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climbing phase")
        best_seq, best_score = _hill_climb(best_seq, max_iter=10**7, deadline=deadline)
        print(f"[{time.time() - start_time:.2f}s] Hill‑climbing finished, best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
