"""
Propose a step‑function (list of non‑negative heights) that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages:

1️⃣  Warm‑start – a few Dirichlet draws and the optional global variable
    ``height_sequence_1`` (if present).

2️⃣  Structured search – we scan the symmetric Beta(p,p) family and a
    larger asymmetric Beta(α,β) family.  For each shape we pre‑compute the
    three convolutions needed to evaluate any convex mixture with the uniform
    vector:
        conv = λ²·BB + 2λ(1‑λ)·BU + (1‑λ)²·UU .
    A fine grid on λ (step 0.01) is examined, and the best candidate is kept.

3️⃣  Guided hill‑climb – from the current best sequence we repeatedly
    identify the shift(s) where the convolution attains its maximum,
    compute the gradient w.r.t. the mass at each index (the reversed slice of
    the peak shift), and move a small amount of probability from a high‑weight
    index to a low‑weight index.  Occasionally a tiny smoothing step is tried.
    All moves are accepted only if they improve the score.

The whole procedure respects the supplied time budget (default 1000 s) and
prints progress.  The returned list is normalised to sum ≈ 1, satisfies the
non‑negativity requirement and is ready for ``evaluate_sequence``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _conv_fft(a: np.ndarray, b: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆b (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    Bf = np.fft.rfft(b, n=L_fft)
    conv = np.fft.irfft(Af * Bf, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (sum(a) is assumed = 1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _beta_shape(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) density on the grid x_i=(i+0.5)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _beta_ab_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) density on the same grid (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

        sc, conv_arr = _score_of(a, n, L_fft)
        if sc < best_score - 1e-12:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improve → score = {sc:.6f}")

        # indices where the convolution is (practically) maximal
        max_val = conv_arr.max()
        active = np.where(conv_arr >= max_val - 1e-12)[0]

        # gradient weights w_i = Σ_k a_{k‑i} over all active shifts k
        w = np.zeros(n, dtype=np.float64)
        for k in active:
            if k < n:
                w[:k + 1] += a[k::-1]
            else:
                start = k - (n - 1)
                w[start:] += a[start:][::-1]

        i_high = int(np.argmax(w))
        i_low = int(np.argmin(w))
        if a[i_high] < 1e-12:
            # nothing to move – break to avoid endless loop
            break

        # move a modest fraction of the mass from high‑weight to low‑weight index
        delta = a[i_high] * rng.uniform(0.05, 0.2)   # 5‑20 % of the source mass
        cand = a.copy()
        cand[i_high] -= delta
        cand[i_low] += delta
        cand /= cand.sum()

        cand_score, _ = _score_of(cand, n, L_fft)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improve → score = {cand_score:.6f}")
            continue

        # occasional tiny smoothing move (probability ≈ 1 %)
        if rng.random() < 0.01:
            kernel = np.array([0.25, 0.5, 0.25])
            smooth = np.convolve(a, kernel, mode='same')
            smooth /= smooth.sum()
            smooth_score, _ = _score_of(smooth, n, L_fft)
            if smooth_score < best_score:
                best_score, best_a = smooth_score, smooth.copy()
                a = smooth
                print(f"[{time.time() - start_time:.2f}s] smoothing improve → score = {smooth_score:.6f}")

        # time guard
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Normalisation & return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
