def _conv_self(a):
    n = a.shape[0]
    L = _next_pow2(2*n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    return conv

def _grad_from_w(a, w):
    n = a.shape[0]
    # reverse w
    w_rev = w[::-1]
    L = _next_pow2(3*n - 2)
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(fa * fw, n=L)[:3*n - 2]  # length 3n-2
    idx = (2*n - 2) - np.arange(n)
    M_a = conv[idx]
    grad = 4.0 * n * M_a
    return grad

"""
Smooth‐max projected gradient descent for the step‑function problem.

The objective (with the sequence normalised ∑a_i = 1) is

    F(a) = 2·n·max(conv(a,a))

where conv(a,a) is the full linear self‑convolution.
The function max(·) is nonsmooth, but it can be smoothed with a
log‑sum‑exp (soft‑max) approximation

    Φ_ε(a) = 2·n·ε·log(∑_k exp(conv_k/ε))

∇Φ_ε(a) can be written in closed form:
let w_k = soft‑max weights, i.e. w_k = exp(conv_k/ε)/∑_j exp(conv_j/ε).
Define the (symmetric) matrix M with entries M_{i,j}=w_{i+j}.
Then ∇Φ_ε(a) = 4·n·M·a.
The product M·a is obtained efficiently by a single FFT‑based convolution
between a and the reversed weight vector w.

The algorithm proceeds as follows

1. Initialise a on the simplex (uniform vector and a few random Dirichlet draws).
2. For a decreasing schedule of ε values (from 1e‑2 down to 1e‑5) run a
   projected gradient descent:
        – compute the smoothed gradient,
        – take a step and project back onto the probability simplex,
        – keep the step only if the true objective 2·n·max(conv) improves.
   A simple multiplicative step‑size decay is used; the step is reduced
   whenever a trial does not improve the objective.
3. The best vector found (over all starts) is returned.

All heavy work is performed with FFTs (O(n log n) per gradient), so the
method can afford thousands of iterations within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing
# ----------------------------------------------------------------------
start_time = None  # set in propose_candidate


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_from_w(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective Φ_ε at ``a``.
    ``w`` must be the soft‑max weights (shape 2·n‑1, sum to 1).
    Returns 4·n·M·a where M_{i,j}=w_{i+j}.
    """
    n = a.shape[0]
    # reverse w for the convolution that yields (M·a)
    w_rev = w[::-1]
    L = _next_pow2(3 * n - 2)          # length needed for a (n) * w_rev (2n‑1)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(Af * Wf, n=L)[: 3 * n - 2]   # length 3n‑2
    # indices that give (M·a)_i = Σ_j w_{i+j} a_j
    idx = (2 * n - 2) - np.arange(n)
    Ma = conv[idx]
    return 4.0 * n * Ma


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x≥0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_real(a: np.ndarray) -> float:
    """True objective 2·n·max(conv(a,a)) (assumes Σa = 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation routine (smooth‑max projected gradient descent)
# ----------------------------------------------------------------------
def _smooth_descent(start_vec: np.ndarray,
                    deadline: float,
                    eps_schedule=None) -> (np.ndarray, float):
    """
    Perform a projected gradient descent on the smoothed objective
    Φ_ε for a decreasing schedule of ε.
    Returns the best vector (and its true objective) discovered.
    """
    if eps_schedule is None:
        # from coarse to fine; smallest epsilon should stay above 1e‑6
        eps_schedule = [1e-2, 5e-3, 2e-3, 1e-3, 5e-4,
                        2e-4, 1e-4, 5e-5, 2e-5, 1e-5]

    a = start_vec.copy()
    cur_obj = _objective_real(a)
    best_a = a.copy()
    best_obj = cur_obj

    n = a.shape[0]

    # main outer loop over the ε schedule
    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        # an adaptive step size; start fairly aggressive
        step = 0.5
        # inner loop: keep improving while time permits
        while time.time() < deadline:
            # ----- compute gradient of Φ_ε -----
            conv = _conv_self(a)                # length 2n‑1
            shift = conv.max()
            # stable soft‑max
            exp_vals = np.exp((conv - shift) / eps)
            w = exp_vals / exp_vals.sum()       # weights, sum to 1

            grad = _grad_from_w(a, w)           # shape (n,)

            # ----- simple back‑tracking line search -----
            improved = False
            trial_step = step
            for _ in range(10):                 # at most 10 shrinkings
                cand = a - trial_step * grad
                cand = _project_simplex(cand)

                cand_obj = _objective_real(cand)
                if cand_obj < cur_obj - 1e-12:
                    improved = True
                    break
                trial_step *= 0.5               # shrink until improvement

            if not improved:
                # no improvement possible with current step – reduce it and try again
                step *= 0.5
                if step < 1e-12:
                    # step is hopelessly small → move to next ε
                    break
                continue

            # ----- accept the candidate -----
            a = cand
            cur_obj = cand_obj
            if cand_obj < best_obj:
                best_obj = cand_obj
                best_a = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] eps {eps:.1e} "
                      f"new best {best_obj:.6f}")

            # slowly increase the step for the next iteration (but keep it bounded)
            step = min(trial_step * 1.2, 1.0)

            # early stop for this ε if progress stalls
            if trial_step * np.linalg.norm(grad) < 1e-12:
                break

    return best_a, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (probability vector) that
    minimises ``evaluate_sequence``.  The search consists of:

    * a uniform start,
    * a handful of random Dirichlet starts,
    * a smooth‑max projected gradient descent with a decreasing ε schedule.

    The best vector found before the time budget expires is returned.
    """
    global start_time
    start_time = time.time()
    # tiny safety margin (the judge kills the process if we overshoot)
    deadline = start_time + budget_s - 0.1

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we keep the same size as the reference implementation
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Baseline: the perfectly uniform vector
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = _objective_real(best_a)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (e.g. ``height_sequence_1`` supplied by the
    # environment).  The vector is normalised before being considered.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_real(starter)
                if sc < best_score:
                    best_a, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet initialisations (time‑boxed)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 30.0)
    while time.time() < dirichlet_deadline:
        # logarithmically spaced concentration parameter
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10.0 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        # short smooth‑descent from this random start (≈5 s)
        sub_deadline = time.time() + 5.0
        a_opt, sc_opt = _smooth_descent(a, sub_deadline)
        if sc_opt < best_score:
            best_a, best_score = a_opt, sc_opt
            print(f"[{time.time() - start_time:.2f}s] Dirichlet start improved best to {best_score:.6f}")

        if time.time() >= deadline - 20.0:   # reserve ~20 s for final polishing
            break

    # ------------------------------------------------------------------
    # 2️⃣  Main refinement from the best vector discovered so far
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting main smooth‑max descent")
        best_a, best_score = _smooth_descent(best_a, deadline)
        print(f"[{time.time() - start_time:.2f}s] Main descent best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Tiny random local polishing (optional, last ~10 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = min(deadline, time.time() + 10.0)
        print(f"[{time.time() - start_time:.2f}s] Starting final random polishing")
        a = best_a.copy()
        cur_score = best_score
        n = a.shape[0]
        while time.time() < local_deadline:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue
            delta = a[src] * rng.random() * 0.002   # very small transfer
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand_score = _objective_real(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_a, best_score = cand.copy(), cand_score
                    print(f"[{time.time() - start_time:.2f}s] Polishing new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return (ensure strict feasibility for the evaluator)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
