def _adam_smooth_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    # initial tau
    cur_conv = _conv_self(a)
    tau = max(cur_conv.max()*0.5, 1e-5)
    min_tau = 1e-9
    tau_decay = 0.99
    while time.time() < deadline and tau > min_tau:
        t += 1
        # compute conv
        conv = _conv_self(a)
        max_conv = conv.max()
        # compute softmax weights
        shift = max_conv
        exp_weights = np.exp((conv - shift) / tau)
        w_sum = exp_weights.sum()
        p = exp_weights / w_sum
        # gradient of smooth max: 2*conv(p, a_rev)
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]
        grad = 2.0 * conv_pa[(n-1):(n-1+n)]
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step_size = 0.1 / (np.sqrt(v_hat) + eps)  # adapt per-coordinate maybe
        a_new = a - step_size * m_hat
        a_new = _project_simplex(a_new)
        # evaluate objective
        conv_new = _conv_self(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam tau={tau:.2e} iter={t} best={obj_new:.6f}")
        # accept if objective improved else maybe keep old a
        if obj_new < 2.0 * n * max_conv:
            a = a_new
        else:
            # maybe occasional acceptance? but keep a unchanged
            pass
        tau *= tau_decay
    return best_seq, best_score
```

def _smooth_max_gradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    cur_conv = _conv_self(a)
    tau = max(cur_conv.max() * 5.0, 1e-5) # start larger smoothing
    min_tau = 1e-9
    decay = 0.995
    while time.time() < deadline and tau > min_tau:
        conv = _conv_self(a)
        max_conv = conv.max()
        # normal smoothing:
        shift = max_conv
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        p = w / w_sum
        # grad:
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]
        grad = 2.0 * conv_pa[(n-1):(n-1+n)]

        # line search
        step = 1.0
        cur_obj = 2.0 * n * max_conv
        improved = False
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-12:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] GD it={int(time.time()*1000)%100000} tau={tau:.2e} obj={obj_new:.6f}")
                break
            step *= 0.5
        if not improved:
            # add jitter
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
        tau *= decay
    return best_seq, best_score

"""
Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines several cheap parametric sweeps (including a beta‑family
of shapes), a large number of random Dirichlet draws (keeping the best few
candidates), multi‑start smoothed‑max gradient descent on the simplex and a
short random‑jitter local search.  All operations are performed on the full
length n (default 4096) but the code respects a 1000 s wall‑time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full a*a convolution using a fast FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """
    Exact objective for a sequence that already satisfies sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    # Implementation of https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (very light sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for eps in np.logspace(-5, -1, 100):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = r ** idx
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.1f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = (idx + 1.0) ** p
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.1f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_beta(n, best_seq, best_score, deadline):
    """Beta‑distribution shapes: a_i ∝ x^{p‑1}(1‑x)^{q‑1} with interior x."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # interior points, avoids 0/1
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    p_vals = np.logspace(-0.7, 0.7, 20)   # approx 0.2 … 5
    q_vals = np.logspace(-0.7, 0.7, 20)
    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline:
                return best_seq, best_score
            a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            # try a few mixes with the uniform distribution
            for lam in (0.0, 0.4, 0.8):
                seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * a
                seq /= seq.sum()
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] Beta p={p:.3f} q={q:.3f} lam={lam:.1f} score={sc:.6f}")
    return best_seq, best_score

# 2️⃣ Random Dirichlet draws (keep the best few)
# ----------------------------------------------------------------------
def _random_dirichlet_top_k(n, max_k, deadline):
    rng = np.random.default_rng()
    top = []   # list of (score, seq)
    while time.time() < deadline:
        # concentration between 1e-2 and 1e2 (log‑uniform)
        logc = rng.uniform(-2.0, 2.0)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if len(top) < max_k:
            top.append((sc, a.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if sc < top[-1][0]:
                top[-1] = (sc, a.copy())
                top.sort(key=lambda x: x[0])
        # occasional progress output
        if len(top) and (len(top) % 5 == 0):
            print(f"[{time.time() - start_time:.2f}s] Dirichlet best={top[0][0]:.6f}")
    return top


# ----------------------------------------------------------------------
# 3️⃣ Smoothed‑max gradient descent (multi‑start)
# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(a0, deadline, best_seq, best_score):
    """
    Gradient descent on τ·log‑sum‑exp(conv/τ).
    Starts with a relatively large τ and cools geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    # Initialise τ using a generous factor to obtain a smooth landscape
    cur_conv = _conv_self(a)
    tau = max(cur_conv.max() * 5.0, 1e-5)
    min_tau = 1e-9
    decay = 0.995

    while time.time() < deadline and tau > min_tau:
        conv = _conv_self(a)
        max_conv = conv.max()
        # smoothed weight vector (soft‑max over conv entries)
        shift = max_conv                       # for numerical stability
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            tau *= decay
            continue
        p = w / w_sum

        # gradient of the smooth max w.r.t. a:
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]
        grad = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # simple back‑tracking line‑search on the *true* objective
        cur_obj = 2.0 * n * max_conv
        step = 1.0
        improved = False
        while step > 1e-14:
            a_new = a - step * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-12:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] GD iter={int(time.time()*1000)%100000} τ={tau:.2e} obj={obj_new:.6f}")
                break
            step *= 0.5

        if not improved:
            # tiny jitter to shake off possible plateaus
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

        tau *= decay
    return best_seq, best_score


def _run_multi_start_gd(initials, deadline, best_seq, best_score):
    """Run the SM‑gradient descent from several starting points."""
    # allocate roughly equal time for each start
    remaining = deadline - time.time()
    m = len(initials)
    for i, seq in enumerate(initials):
        if time.time() >= deadline:
            break
        # give the remaining time proportionally to the number of runs left
        run_deadline = time.time() + (deadline - time.time()) / (m - i)
        best_seq, best_score = _smooth_max_gradient_descent(seq, run_deadline,
                                                            best_seq, best_score)
    return best_seq, best_score

# ----------------------------------------------------------------------
def _simulated_annealing(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    cur_obj = 2.0 * n * _conv_self(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        # temperature decays linearly with remaining time
        frac = (deadline - time.time()) / (deadline - start_time)
        eps = 0.01 * frac      # max perturbation amplitude
        perturb = np.random.uniform(-eps, eps, size=n)
        a_new = a + perturb
        a_new = np.clip(a_new, 0.0, None)
        a_new = _project_simplex(a_new)
        obj_new = 2.0 * n * _conv_self(a_new).max()
        if obj_new < cur_obj - 1e-12:
            a = a_new
            cur_obj = obj_new
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it} obj={obj_new:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine performs cheap parametric sweeps (including a beta
    family), massive Dirichlet sampling, several multi‑start smoothed‑max
    gradient descents and a final jitter‑based polishing step.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # safety margin so the caller does not cut us off abruptly
    deadline = start_time + budget_s - 0.1

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Parametric sweeps (≈20 % of the budget)
    # --------------------------------------------------------------
    sweep_deadline = start_time + (budget_s * 0.20)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_beta(n, best_seq, best_score, sweep_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈25 % of the budget)
    # --------------------------------------------------------------
    dirichlet_deadline = sweep_deadline + (budget_s * 0.25)
    top_k = 12                         # how many good candidates to retain
    if time.time() < dirichlet_deadline:
        top = _random_dirichlet_top_k(n, top_k, dirichlet_deadline)
        # update global best from the stored top‑k list
        for sc, seq in top:
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
        # also keep the top few as starting points for the next phase
        start_candidates = [seq for _, seq in top[:5]]
    else:
        start_candidates = []

    # --------------------------------------------------------------
    # 3️⃣ Multi‑start smoothed‑max gradient descent (≈40 % of the budget)
    # --------------------------------------------------------------
    gd_deadline = dirichlet_deadline + (budget_s * 0.40)
    # ensure we have at least one start (fallback to current best)
    if not start_candidates:
        start_candidates = [best_seq.copy()]
    # add the current best as an extra start
    start_candidates.append(best_seq.copy())
    if time.time() < gd_deadline:
        best_seq, best_score = _run_multi_start_gd(start_candidates,
                                                   gd_deadline,
                                                   best_seq, best_score)

    # --------------------------------------------------------------
    # 4️⃣ Final short jitter based local search (≈15 % of the budget)
    # --------------------------------------------------------------
    sa_deadline = deadline
    if time.time() < sa_deadline:
        best_seq, best_score = _simulated_annealing(best_seq,
                                                    sa_deadline,
                                                    best_seq, best_score)

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
