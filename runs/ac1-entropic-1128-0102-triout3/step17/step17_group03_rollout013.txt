def adam_optimize(a0, max_iter, deadline):
    n = len(a0)
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    t=0
    while time.time() < deadline and t<max_iter:
        t+=1
        # compute subgradient of max conv
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv) * (2 * a.shape[0]) # as F gradient
        # update moments
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = 0.01 / (np.sqrt(v_hat)+epsilon)  # learning rate scaling
        a = a - step * m_hat
        # projection onto simplex
        a = _project_simplex(a)
        # evaluate
        score = _objective_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam iter {t} new best {best_score:.6f}")

    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the full linear self‑convolution.
The algorithm is a hybrid of

* many random Dirichlet initialisations,
* a few short Adam‑style projected gradient runs (fast sub‑gradient updates),
* a final back‑tracking projected‑gradient descent (as in the reference code),
* a cheap random local refinement.

All vectors are kept on the probability simplex (sum = 1) so the denominator
cancels and the objective reduces to 2·n·max(conv).  The implementation
updates the convolution incrementally only when needed, respects the 1000 s
wall‑time limit and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing (updated inside propose_candidate)
# ----------------------------------------------------------------------
_start_time = None


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Assumes sum(a)=1 → returns 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at ``a`` (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected optimisation
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> (np.ndarray, float):
    """
    Fast projected Adam on the simplex using the sub‑gradient of max(conv).
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = a.copy()
    best_score = _objective_score(a)

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        # gradient of the full objective = 2*n * subgradient of max(conv)
        grad = _subgradient_maxconv(a, conv) * (2.0 * n)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # simple learning‑rate schedule
        step = 0.05 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Adam iter {t:,} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple random Dirichlet initialisation (collect a few good seeds)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int,
                             max_trials: int,
                             deadline: float,
                             rng: np.random.Generator) -> list:
    """Return a list of the best vectors found (up to 5) from Dirichlet draws."""
    top = []          # list of (score, vector) sorted increasingly
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        conc_log = rng.uniform(-3.0, 0.7)          # wide concentration range
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        # keep at most 5 best candidates
        if len(top) < 5 or sc < top[-1][0]:
            top.append((sc, a.copy()))
            top.sort(key=lambda x: x[0])
            if len(top) > 5:
                top.pop()
            print(f"[{time.time() - _start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return top


# ----------------------------------------------------------------------
# Small random “local‑search” refinement after PGD (unchanged)
# ----------------------------------------------------------------------
def _random_local_refinement(a0: np.ndarray,
                             cur_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> (np.ndarray, float):
    """Very cheap random pairwise transfers – only accept strict improvements."""
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        delta = a[src] * rng.random() * 0.005
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Local refine new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent (back‑tracking) – same logic as reference
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray,
                                cur_score: float,
                                deadline: float) -> (np.ndarray, float):
    """Fine‑tune by moving opposite to the sub‑gradient of the active shifts."""
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    step = 0.5

    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - _start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)

    return best_a, best_score
