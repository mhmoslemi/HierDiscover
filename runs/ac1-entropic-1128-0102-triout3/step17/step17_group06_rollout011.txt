def project_simplex(v):
    # ensure sums to 1, nonnegative.
    # if any negative values after projection allowed? algorithm works for any v.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    # projects onto probability simplex (sum=1, nonnegative)
    n = v.shape[0]

    rho = np.where(rho_arr)[0][-1]

"""
Propose a step‑function (non‑negative vector) that minimises

        score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three phases:

1️⃣  **Coarse warm‑start** –  Scan a few Beta(p,p) densities (including
    a uniform mixture) and a handful of Dirichlet samples; keep the
    best one.

2️⃣  **Projected sub‑gradient descent** –  For the current vector *a*
    compute the index `k_max` where the self‑convolution is maximal.
    The sub‑gradient of `max(conv)` w.r.t. `a` is `g_i = 2·a_{k_max‑i}`
    (zero outside the support).  Take a step opposite to `g`,
    then project back onto the probability simplex
    (non‑negative, sum = 1).  The step size decays as 1/√t.
    This deterministic descent quickly flattens the convolution
    and improves the score far beyond the simple hill‑climb used
    previously.

3️⃣  **Final return** –  The best vector ever seen is returned as a
    plain Python list.  All numbers are non‑negative, automatically
    satisfy the evaluator’s clipping limits and sum to one, so the
    score reported by `evaluate_sequence` equals the internally
    computed value `2·n·max(conv)`.
"""

import time
import numpy as np
import scipy.sparse

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised Beta(p,p) density on the grid x_i = (i+0.5)/n,
    normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    w >= 0 , sum(w) = 1.
    """
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            print(f"[{time.time() - start_time:.2f}s] iter {iteration:6d} → new best {cur_score:.6f}")
        else:
            it_without_improve += 1

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        g[mask] = 2.0 * a[j[mask]]

        # decreasing step size
        step = step0 / np.sqrt(iteration)

        # gradient step + projection onto the simplex
        a_next = a - step * g
        a = _project_simplex(a_next)

        # occasional random perturbation if we are stuck
        if it_without_improve >= 5000:
            perturb = rng.gamma(shape=0.3, scale=1.0, size=n)
            a = 0.9 * a + 0.1 * perturb
            a = a / a.sum()
            it_without_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Stagnation – random perturbation")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
