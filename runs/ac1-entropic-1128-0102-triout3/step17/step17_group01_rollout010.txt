def adam_optimize(a0, L_fft, L_grad, deadline, start_time, rng):
    a = a0.copy()
    best_a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 1.0
    beta_multiplier = 2.0
    max_beta = 8192.0
    while time.time() < deadline and beta <= max_beta:
        # adjust beta schedule
        # inner loop
        for _ in range(2000):
            if time.time() >= deadline:
                break
            t += 1
            conv = _conv_self_fft(a, L_fft)
            # compute softmax weights
            p = _softmax_weights(conv, beta)
            g = _grad_softmax_new(a, p, L_grad)  # returns gradient of smooth max
            # Adam update
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * (g * g)
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            step = 0.6 * m_hat / (np.sqrt(v_hat) + 1e-8)
            a_new = a - step
            a_new = _project_simplex(a_new)
            score = _eval_fast(a_new, L_fft)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve score {score:.6f}")
            # Accept or not - maybe always accept new? Use Metropolis
            # but we can accept if better else probabilistic
            if score < _eval_fast(a, L_fft):
                a = a_new
            else:
                # Simulated annealing acceptance
                delta = score - _eval_fast(a, L_fft)
                prob = np.exp(-delta * 5.0 / np.sqrt(t))
                if rng.random() < prob:
                    a = a_new
        beta *= beta_multiplier
    return best_a, best_score

"""
Propose a non‑negative step‑function that (hopefully) improves on the previous
best score.  The algorithm works as follows:

1️⃣  **Seed generation** – a very rich pool:
    * uniform distribution,
    * the optional global starter ``height_sequence_1`` (if it has the right size),
    * a dense parametric sweep of geometric‑×‑power shapes
      (``a_i ∝ r**i·(i+1)**p``) mixed with the uniform distribution,
    * several Dirichlet draws with different concentrations.

   After a cheap evaluation we keep the K best seeds.

2️⃣  **Flattening phase** – minimise the L2‑norm of the self‑convolution
    (i.e.  Σ (a*a)_k²).  Because Σ (a*a)_k = 1 this pushes the whole
    convolution towards a flat (≈ constant) shape, which is beneficial for the
    L∞‑objective.  We use Adam on this quadratic surrogate.

3️⃣  **Continuation on a smooth‑max surrogate** – for each seed we run
    Adam on the log‑sum‑exp of the convolution with an exponentially growing
    temperature β (β = 1,2,4,…,8192).  This progressively approximates the true
    max‑norm while keeping the optimisation smooth.

4️⃣  **Multiplicative (exponentiated‑gradient) polishing** – classic exp‑grad
    updates on the true L∞‑objective with a decreasing learning rate.

5️⃣  **Final random hill‑climb** – tiny mass transfers between random positions
    (up to 0.5 % of the source mass) to escape possible tiny local minima.

All phases respect the simplex constraint (∑a = 1, a_i ≥ 0) via Euclidean
projection.  The whole search stops a little before the 1000 s budget and
returns the best vector found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k·(a*a)_k w.r.t. a.
    Returns 2·Σ_j a_j·p_{i+j}.
    Implemented as a convolution of a_rev with p.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(β·conv_k)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact objective (used for bookkeeping)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a positive vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    # work in log domain for stability
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)               # prevent overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

# Optimisation utilities
# ----------------------------------------------------------------------


def _adam_flatten(a0: np.ndarray,
                  L_fft: int,
                  L_grad: int,
                  deadline: float,
                  start_time: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the quadratic surrogate   Σ (a*a)_k².
    The gradient is   g = 2·_grad_softmax_new(a, conv, L_grad).
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # Adam hyper‑parameters (standard)
    lr = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # we do a fixed maximum number of iterations; the outer driver will
    # stop earlier if the global deadline is reached.
    max_iter = 3000

    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self_fft(a, L_fft)           # a*a
        # gradient of Σ conv_k² = 2·_grad_softmax_new(a, conv, L_grad)
        g = 2.0 * _grad_softmax_new(a, conv, L_grad)

        # Adam accumulators
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        cand_score = _eval_fast(a_candidate, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best = a_candidate.copy()
            print(f"[{time.time() - start_time:.2f}s] flatten‑Adam it={t:,} score={cand_score:.6f}")

        # simple Metropolis acceptance to keep some exploration
        cur_score = _eval_fast(a, L_fft)
        if cand_score < cur_score:
            a = a_candidate
        else:
            prob = np.exp(-(cand_score - cur_score) * 4.0 / (np.sqrt(t) + 1.0))
            if rng.random() < prob:
                a = a_candidate

    return best, best_score


def _adam_maxsurrogate(a0: np.ndarray,
                       L_fft: int,
                       L_grad: int,
                       deadline: float,
                       start_time: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate
        f_β(a) = log‑sum‑exp(β·conv(a,a))
    with a continuation on β (geometric schedule up to 8192).
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(a, L_fft)

    # Adam hyper‑parameters
    lr = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # β schedule: 1 → 2 → 4 → … → 8192
    beta = 1.0
    while time.time() < deadline and beta <= 8192.0:
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        max_iter_beta = 2500                     # generous per‑β budget
        while time.time() < deadline and t < max_iter_beta:
            t += 1
            conv = _conv_self_fft(a, L_fft)
            p = _softmax_weights(conv, beta)     # weights sum to 1
            g = _grad_softmax_new(a, p, L_grad)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_candidate = a - step
            a_candidate = _project_simplex(a_candidate)

            cand_score = _eval_fast(a_candidate, L_fft)

            if cand_score < best_score:
                best_score = cand_score
                best = a_candidate.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] max‑Adam β={beta:.0f} it={t:,} score={cand_score:.6f}"
                )

            # accept if improvement, otherwise occasional uphill move
            cur_score = _eval_fast(a, L_fft)
            if cand_score < cur_score:
                a = a_candidate
            else:
                prob = np.exp(-(cand_score - cur_score) * 5.0 / (np.sqrt(t) + 1.0))
                if rng.random() < prob:
                    a = a_candidate

        beta *= 2.0                                   # increase temperature

    return best, best_score


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated‑gradient) updates directly on the
    true L∞‑objective 2·n·max(conv)/sum².
    """
    a = a0.copy()
    best = a.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))           # index of current max
        # sub‑gradient of max at index k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
        idx_j = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[idx_j]

        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best, best_score


def _random_hillclimb(a0: np.ndarray,
                      L_fft: int,
                      deadline: float,
                      start_time: float,
                      rng: np.random.Generator,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Very small random mass transfers between two positions.
    """
    a = a0.copy()
    n = a.shape[0]
    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 4e-3   # ≤0.4 % of source mass
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            a = prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={new_score:.6f}")
    return a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.15      # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 8192 gives a finer discretisation while staying fast.
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)          # needed for all gradient calculations

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter (if available) – only when it matches the current length.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # down‑sample / up‑sample by simple interpolation to the target size
                x_old = np.linspace(0, 1, starter.size)
                x_new = np.linspace(0, 1, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Generate a very rich pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [best_seq]            # already contains the uniform (or starter)

    # Parametric sweep (geometric × power) with a few mixing ratios.
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9999), 18)   # finer grid
    p_vals = np.linspace(-6.0, 6.0, 25)
    lam_vals = [0.0, 0.2, 0.5, 0.8]
    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Random Dirichlet draws with a handful of concentration parameters.
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(8):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # Keep the K best seeds after a cheap evaluation.
    # ------------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 12
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Flattening phase – minimise Σ (a*a)_k² (quadratic surrogate).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0:
        phase_deadline = time.time() + min(120.0, remaining * 0.3)   # allocate ~30 % of remaining time
        for idx in range(len(pool)):
            vec, sc = _adam_flatten(pool[idx], L_fft, L_grad, phase_deadline, start, rng)
            pool[idx] = vec
            pool_scores[idx] = sc
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time() - start:.2f}s] after flatten‑Adam best score={best_score:.6f}")
            if time.time() >= phase_deadline:
                break

    # ------------------------------------------------------------------
    # 3️⃣ Continuation on a smooth‑max surrogate (Adam + β‑schedule).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0:
        phase_deadline = time.time() + min(300.0, remaining * 0.5)   # allocate up to half the remaining budget
        for idx in range(len(pool)):
            vec, sc = _adam_maxsurrogate(pool[idx], L_fft, L_grad, phase_deadline, start, rng)
            pool[idx] = vec
            pool_scores[idx] = sc
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time() - start:.2f}s] after max‑Adam best score={best_score:.6f}")
            if time.time() >= phase_deadline:
                break

    # ------------------------------------------------------------------
    # 4️⃣ Exponentiated‑gradient polishing.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        exp_deadline = time.time() + min(80.0, remaining * 0.2)
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, exp_deadline, start, rng, best_score
        )

    # ------------------------------------------------------------------
    # 5️⃣ Final random hill‑climb (tiny mass transfers).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        best_seq, best_score = _random_hillclimb(
            best_seq, L_fft, deadline, start, rng, best_score
        )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
