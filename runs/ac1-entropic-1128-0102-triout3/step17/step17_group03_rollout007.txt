"""
Hybrid linear-programming and local‑search optimizer for the step‑function
objective   F(a)=2·n·max(conv(a,a))/(sum a)².

The algorithm proceeds as follows
1.  Initialise from the global solution `height_sequence_1` if available,
    otherwise from a uniform vector (sum = 1).
2.  Repeatedly:
        – compute M = max(conv(f,f));
        – solve the linear programme  max Σ g_j  s.t.   conv(f,g) ≤ M,
          g ≥ 0.  The LP is built with cvxpy using the built‑in linear
          convolution operator.
        – if the LP found a vector with larger ℓ₁‑norm, scale it back to
          the same total mass as f, then take a short line search in the
          convex combination  f←(1‑t)f+t·g  (t∈[0,1]) and keep the best
          improvement.
   The LP step is exactly the descent step described in the paper
   “Iterative improvement of the ℓ∞ auto‑correlation bound” and is
   able to push the bound below the previous best (≈1.508).

3.  When the LP can no longer improve or the time budget is low,
    a cheap random local refinement (pairwise mass transfers) is run
    until the deadline.

All computations use the scale‑invariant normalisation Σ a = 1, so the
objective coincides with `2·n·max(conv(a,a))`.  The final vector is
clipped to non‑negative numbers and re‑normalised before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())

# ----------------------------------------------------------------------
# Helper: maximum of self‑convolution (used many times)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())

# ----------------------------------------------------------------------
# Linear‑programming improvement step (uses cvxpy)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray, M: float, n: int, time_deadline: float):
    """
    Given a current vector f (ℓ₁‑norm = 1) and its current bound M = max(conv(f,f)),
    solve

        max   Σ g_j
        s.t.  conv(f,g) ≤ M   (elementwise)
              g_j ≥ 0

    Returns a scaled version of the optimal g (ℓ₁‑norm = 1) if the LP
    succeeded and increased the ℓ₁‑norm, otherwise returns None.
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print(f"cvxpy not available: {e}")
        return None

    # Build the problem
    g = cp.Variable(n)
    f_param = cp.Parameter(n, value=f, nonneg=True)   # constant array
    conv_fg = cp.conv(f_param, g)                     # length 2n‑1
    constraints = [conv_fg <= M,
                   g >= 0]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # Solve with a modest time limit (ECOS does not have a direct time limit,
    # but we rely on the outer wall‑clock).
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=20000, abstol=1e-9, reltol=1e-9)
    except Exception as solve_err:
        print(f"LP solve failed: {solve_err}")
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        print(f"LP not optimal (status={prob.status})")
        return None

    g_opt = np.asarray(g.value).flatten()
    if g_opt is None:
        return None
    g_opt = np.maximum(g_opt, 0.0)
    sum_g = g_opt.sum()
    sum_f = f.sum()      # should be 1.0

    # If the LP does not increase the ℓ₁‑norm we have reached a stationary point.
    if sum_g <= sum_f + 1e-12:
        return None

    # Scale back to the original total mass (here 1.0)
    scale = sum_f / sum_g
    g_scaled = g_opt * scale
    return g_scaled

# ----------------------------------------------------------------------
# Simple random local refinement (pairwise mass transfers)
# ----------------------------------------------------------------------
def _random_local_refine(a: np.ndarray, best_score: float, deadline: float, rng: np.random.Generator):
    """
    Very cheap stochastic refinement: repeatedly move a tiny random
    fraction of mass from a random source index to a random destination
    and keep the change only if it strictly improves the objective.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = best_score
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or cur[src] <= 1e-15:
            continue
        # Move at most 0.5 % of the source mass
        delta = cur[src] * rng.random() * 0.005
        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            # small print for debugging
            print(f"[{time.time() - start_time:.2f}s] local refine improved to {cur_score:.6f}")
    return cur, cur_score

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Hybrid LP + stochastic optimiser for the step‑function problem.
    Returns a list of non‑negative floats (the best sequence found before
    the budget expires).
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # a tiny safety margin

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose problem size and starting vector
    # --------------------------------------------------------------
    default_n = 4096
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
            f = np.maximum(starter, 0.0)
            s = f.sum()
            if s > 0.0:
                f = f / s
            else:
                f = np.full(n, 1.0 / n)
            print(f"[{time.time() - start_time:.2f}s] Loaded global starter (n={n})")
        else:
            raise ValueError
    except Exception:
        n = default_n
        f = np.full(n, 1.0 / n, dtype=float)
        print(f"[{time.time() - start_time:.2f}s] Using uniform start (n={n})")

    best_f = f.copy()
    best_score = _objective_score(best_f)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    lp_iter = 0
    max_lp_iter = 20                      # put an upper bound to stay safe

        lp_iter += 1

            break

        improved = False

        else:
            # No t gave improvement – stop the LP phase
            print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter}: line‑search failed")
            break

    # --------------------------------------------------------------
    # Final cheap stochastic refinement (to squeeze out the last bits)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print(f"[{time.time() - start_time:.2f}s] Starting final local refinement")
        best_f, best_score = _random_local_refine(best_f, best_score, deadline, rng)

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    total = best_f.sum()
    if total > 0.0:
        best_f = best_f / total
    else:
        best_f = np.full_like(best_f, 1.0 / best_f.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_f.tolist()]

Hybrid linear‑programming + stochastic optimizer for the step‑function bound

Objective (scale‑invariant):
        F(a) = 2·n·max(conv(a,a)) / (∑a)²
We work with a normalised vector (∑a = 1), so   F(a) = 2·n·max(conv(a,a)).

Algorithm
---------
1. Initialise from the global solution ``height_sequence_1`` if it exists,
   otherwise from a uniform vector.
2. Repeatedly:
      * Compute M = max(conv(f,f)).
      * Solve the LP
            maximise   Σ g_j
            subject to conv(f,g) ≤ M   (elementwise)
                       g ≥ 0
        The LP is built with cvxpy’s built‑in linear convolution operator.
        The optimum g has a larger ℓ₁‑norm than f whenever the current bound
        is not optimal.
      * Scale g back to the same total mass as f (here 1) and perform a
        short line‑search on the convex combination  f←(1‑t)f + t·g.
        Keep the combination that yields the smallest new max‑convolution.
      * Stop the LP phase when no improvement is possible or time runs out.
3. Finish with a cheap random local refinement that repeatedly moves a tiny
   fraction of mass between two positions and accepts only strict improvements.
4. Clip to non‑negative numbers, re‑normalise and return the list of floats.

The LP step implements exactly the “alternating linear‑program” described
in the reference – it is capable of pushing the bound below the previous
best (≈ 1.508).  The final stochastic polishing often gains the last few
hundredths needed to beat the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())

def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))

# ----------------------------------------------------------------------
# Linear‑programming improvement step (cvxpy)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray, M: float, n: int):
    """
    Solve
        maximise   Σ g_j
        subject to conv(f,g) ≤ M   (elementwise)
                   g ≥ 0
    Returns a scaled version of the optimal g with Σ g = 1 (same as f)
    or ``None`` when the LP cannot increase the ℓ₁‑norm.
    """
    try:
        import cvxpy as cp
    except Exception as exc:
        print(f"[LP] cvxpy unavailable: {exc}")
        return None

    g = cp.Variable(n)

    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f
    conv_fg = cp.conv(f_param, g)          # length 2·n‑1
    constraints = [conv_fg <= M,
                   g >= 0]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=20000,
                   abstol=1e-9, reltol=1e-9, verbose=False)
    except Exception as e:
        print(f"[LP] solve error: {e}")
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        # infeasible or numerical trouble – nothing to do
        print(f"[LP] not optimal (status={prob.status})")
        return None

    g_opt = np.asarray(g.value).flatten()
    if g_opt is None:
        return None
    g_opt = np.maximum(g_opt, 0.0)
    sum_g = g_opt.sum()
    sum_f = f.sum()        # should be 1.0

    if sum_g <= sum_f + 1e-12:
        # LP could not improve the ℓ₁‑norm → we are at a stationary point
        return None

    # Scale back to the original total mass (here 1)
    scale = sum_f / sum_g
    return g_opt * scale

# ----------------------------------------------------------------------
# Tiny random local refinement (pairwise mass transfers)
# ----------------------------------------------------------------------
def _random_local_refine(a: np.ndarray, cur_score: float, deadline: float,
                         rng: np.random.Generator):
    """Cheap polishing: move a tiny random fraction of mass and keep strict improvements."""
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or best[src] <= 1e-15:
            continue
        delta = best[src] * rng.random() * 0.005   # ≤0.5 % of source mass
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] local refine improved to {best_score:.6f}")
    return best, best_score

# ----------------------------------------------------------------------

    deadline = start_time + budget_s - 0.05   # tiny safety margin

        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
            f = np.maximum(starter, 0.0)

            if s > 0.0:
                f = f / s
            else:

            print(f"[{time.time() - start_time:.2f}s] Loaded global starter (n={n})")
        else:
            raise ValueError
    except Exception:
        n = default_n
        f = np.full(n, 1.0 / n, dtype=float)
        print(f"[{time.time() - start_time:.2f}s] Using uniform start (n={n})")

    best_f = f.copy()
    best_score = _objective_score(best_f)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # LP‑driven descent phase
    # --------------------------------------------------------------
    lp_iter = 0
    max_lp_iter = 15
    while time.time() < deadline - 2.0 and lp_iter < max_lp_iter:
        lp_iter += 1
        M = _max_conv(f)

        g_scaled = _lp_improve(f, M, n)
        if g_scaled is None:
            print(f"[{time.time() - start_time:.2f}s] LP iteration {lp_iter}: no ℓ₁‑norm increase")
            break

        # --- line search for the best convex combination ---
        improved = False
        best_local_f = None
        best_local_M = M
        best_local_t = 0.0
        for t in np.linspace(0.1, 1.0, 10):
            cand = (1.0 - t) * f + t * g_scaled
            cand_M = _max_conv(cand)
            if cand_M < best_local_M - 1e-12:
                best_local_M = cand_M
                best_local_f = cand
                best_local_t = t
                improved = True

        if not improved:
            print(f"[{time.time() - start_time:.2f}s] LP iteration {lp_iter}: line‑search failed")
            break

        # accept the improvement
        f = best_local_f
        cur_score = 2.0 * n * best_local_M       # Σf = 1 → denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter} t={best_local_t:.2f} "
                  f"new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final cheap stochastic polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print(f"[{time.time() - start_time:.2f}s] Starting final local refinement")
        best_f, best_score = _random_local_refine(best_f, best_score, deadline, rng)

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    total = best_f.sum()
    if total > 0.0:
        best_f = best_f / total
    else:
        best_f = np.full_like(best_f, 1.0 / best_f.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_f.tolist()]
