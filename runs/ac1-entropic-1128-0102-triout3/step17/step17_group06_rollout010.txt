"""
Search for a non‑negative step function (list of floats) that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages:

1️⃣  **Initialisation** – start from the Beta(½,½) (arcsine) density,
    which is known to give a uniform self‑convolution in the continuous
    limit.  The distribution is discretised on a uniform grid and
    normalised to sum 1.

2️⃣  **Fine Beta‑family sweep** – explore a refined grid of shape
    parameters `p` for the symmetric Beta(p,p) law and convex mixtures
    with the uniform vector.  All candidates are evaluated quickly with an
    FFT based convolution; the best candidate so far is retained.

3️⃣  **Directed hill‑climb** – repeatedly lower the current maximum of the
    self‑convolution.  For the index `k` where the convolution attains its
    maximum we compute the per‑coordinate contributions `a_i·a_{k‑i}`.
    Mass is transferred from an index with a large contribution to an
    index with a small contribution (chosen proportionally to the inverse
    of its value).  The move is accepted only if it improves the score.
    The loop runs until the time budget expires.

All vectors are kept on the probability simplex (∑a_i = 1) so the
evaluator’s clipping does not affect the result.  The routine respects
the supplied time budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (∑a = 1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    # because ∑a = 1, score = 2·n·max_b
    return float(2.0 * n * max_b)


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Return the discretised symmetric Beta(p,p) density on the grid
    i = 0,…,n‑1, i.e. a_i ∝ x_i^{p‑1} (1‑x_i)^{p‑1} with
    x_i = (i+0.5)/n.  The result is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)

        # propose the new candidate
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑normalise to avoid drift (should be negligible)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        sc = _score_of(cand, n, L_fft)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = cand.copy()
            a = cand                                   # continue from the improved point
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improve → score = {sc:.6f}")

    # --------------------------------------------------------------
    # Normalisation & return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
