def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find where u > cssv/ind
    cond = u > cssv / ind
    if not np.any(cond):
        # all <=, return uniform
        return np.full(n, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a, L):
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)
    return conv_full[:2 * a.shape[0] - 1]

def _beta_shape(n, p):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u > cssv / ind
    if not np.any(cond):
        # all values <= threshold, return uniform
        return np.full(n, 1.0 / n)

Gradient‑descent optimiser for the L∞ self‑convolution problem.

The score is   score = 2·n·max(conv(a,a)) / (∑a)² .
Because the evaluator normalises the input (∑a=1) we minimise the
maximum of the linear convolution a⋆a.

The algorithm:
  1. Initialise on a discretised arcsine law (Beta(½,½)) – the continuous
     optimiser for this problem.
  2. Optionally try a few nearby Beta(p,p) shapes and keep the best.
  3. minimise a smooth approximation of  max(conv)  using an Adam‑style
     update with a “soft‑max’’ of the convolution (temperature α that is
     increased during the run).  After each step we project onto the
     probability simplex (non‑negative entries that sum to 1).
  4. Keep the best sequence seen and return it when the time budget expires.

The routine respects the given time budget (default 1000 s) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised Beta(p,p) density on the grid i=0,…,n‑1,
    i.e. a_i ∝ x_i^{p‑1}(1‑x_i)^{p‑1} with x_i = (i+0.5)/n.
    The result is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u > cssv / ind
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        eta = 0.05                                         # base learning rate
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        # ---- evaluate candidate ----
        cand_score = _score(a_candidate)

        # always move (Adam works with the raw gradient);
        # we only keep the best ever seen for the final answer.
        a = a_candidate

        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
            print(f"[{time.time() - start_time:.2f}s] iter {t_iter:06d} → new best {best_score:.6f}")

            # early exit if we have already beaten the target
            if best_score <= 1.503:
                print(f"[{time.time() - start_time:.2f}s] target reached, stopping early.")
                break

        # ---- slowly increase the temperature to tighten the approximation ----
        if (t_iter % 200) == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_inc, alpha_max)

    # ------------------------------------------------------------------
    # 3️⃣  Normalise and return the best sequence found.
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
```
