"""
A multi‑phase optimizer for the step‑function objective

    2 * n * max(conv(a,a)) / (sum(a)^2)

The search works as follows

1️⃣  Extensive parametric sweep (exponential, double‑exponential,
    sinusoidal blocks, random mixtures, Dirichlet draws).  All candidates
    are evaluated with a fast FFT‑based convolution and the best one is
    kept.

2️⃣  Greedy hill‑climbing that repeatedly moves a small amount of mass
    from the index that most contributes to the current maximal
    convolution entry to an index that contributes the least.
    The convolution is recomputed after each trial; the move is kept
    only if it reduces the objective (or with a tiny Metropolis
    probability to escape plateaus).

3️⃣  A final smooth‑max gradient descent (log‑sum‑exp surrogate) that
    refines the best vector found in the previous stages.  The smoothness
    parameter β is increased up to 200 and an exponential (mirror‑descent)
    update keeps the iterate on the probability simplex.

The whole routine respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence as a plain Python list.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Exact evaluation of the objective (sum is not assumed to be 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

def _quick_sweep(n: int, best_seq: np.ndarray,
                 best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """Heavy parametric sweep that tries many analytic families."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    dist_to_end = np.minimum(idx, n - 1 - idx)

    # ---- 1. symmetric exponential (already used in the reference code) ----
    for lam in np.logspace(-6, 1, 300):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist_to_end)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SymExp λ={lam:.2e} score={sc:.6f}")

    # ---- 2. mixture of two one‑sided exponentials -----------------------
    lam_grid = np.logspace(-6, 1, 150)
    w_grid = np.linspace(0.0, 1.0, 21)
    for lam in lam_grid:
        exp_left = np.exp(-lam * idx)
        exp_right = np.exp(-lam * (n - 1 - idx))
        for w in w_grid:
            if time.time() >= deadline:
                break
            seq = w * exp_left + (1.0 - w) * exp_right
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] MixExp λ={lam:.2e} w={w:.2f} score={sc:.6f}")

    # ---- 3. sinusoidal block at the ends --------------------------------
    for frac in np.linspace(0.03, 0.45, 30):
        L = int(frac * n)
        if L < 2:
            continue
        # half‑sine on each side
        left = np.sin(np.pi * (np.arange(L) + 0.5) / (2 * L))
        seq = np.zeros(n)
        seq[:L] = left
        seq[-L:] = left[::-1]
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SineBlock L={L} score={sc:.6f}")

    # ---- 4. half‑cosine block -------------------------------------------
    for frac in np.linspace(0.03, 0.45, 30):
        L = int(frac * n)
        if L < 2:
            continue
        left = 0.5 * (1 - np.cos(np.pi * np.arange(L) / (L - 1)))  # Hann‑window like
        seq = np.zeros(n)
        seq[:L] = left
        seq[-L:] = left[::-1]
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] CosineBlock L={L} score={sc:.6f}")

    # ---- 5. quadratic decay block ---------------------------------------
    for frac in np.linspace(0.03, 0.45, 30):
        L = int(frac * n)
        if L < 2:
            continue
        x = np.arange(L) / (L - 1)
        left = 1.0 - x ** 2
        seq = np.zeros(n)
        seq[:L] = left
        seq[-L:] = left[::-1]
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] QuadBlock L={L} score={sc:.6f}")

    # ---- 6. random mixtures of a few bases ------------------------------
    # pre‑compute some bases
    base_exp = np.exp(-np.logspace(-3, 0, 100)[:, None] * dist_to_end[None, :])
    base_lin = np.maximum(0.0, 1.0 - (dist_to_end / n)[:, None].T)
    bases = [base_exp.mean(axis=0), base_lin.mean(axis=0)]
    for _ in range(2000):
        if time.time() >= deadline:
            break
        coeffs = rng.random(len(bases))
        seq = sum(c * b for c, b in zip(coeffs, bases))
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s <= 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] RandomMix score={sc:.6f}")

    # ---- 7. Dirichlet draws (different concentration) -------------------
    for _ in range(2500):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in [1e-3, 5]
        logc = rng.uniform(-3.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        total = a.sum()
        if total <= 0:
            continue
        a /= total
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Greedy hill‑climbing that moves mass away from the current peak
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                best_seq: np.ndarray,
                best_score: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small chunk of probability mass from the index that
    contributes the most to the current maximal convolution entry to an
    index that contributes the least.  Accept only improving moves (with a
    tiny Metropolis chance to escape flat plateaus).
    """
    seq = start_seq.copy()
    n = seq.shape[0]

    # helper to evaluate the current vector (sum stays 1)
    def eval_seq(v):
        return 2.0 * n * _conv_self(v).max()

    cur_score = eval_seq(seq)

    # counters for adaptive step size
    no_improve = 0
    while time.time() < deadline:
        conv = _conv_self(seq)
        cur_score = 2.0 * n * conv.max()
        k_max = int(np.argmax(conv))

        # contributions to the maximal entry
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(seq)
        contrib[valid] = seq[idx[valid]] * seq[partner[valid]]

        # choose source (largest contribution) and sink (smallest contribution)
        i = int(np.argmax(contrib))
        # avoid degenerate case where the biggest contribution is zero
        if contrib[i] == 0:
            break
        # pick a sink among the 5 % lowest contributors
        low_cut = max(1, n // 20)
        low_idxs = np.argpartition(contrib, low_cut)[:low_cut]
        j = int(np.random.choice(low_idxs))
        if i == j:
            continue

        # step size – a fraction of the source mass, shrinks when we stall
        max_move = seq[i] * 0.4
        if max_move <= 0:
            continue
        # adaptive factor
        factor = 0.02 + 0.28 * (1.0 - no_improve / 5000.0)  # between 0.02 and 0.3
        delta = max_move * factor
        if delta <= 0:
            continue

        # propose new vector
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # evaluate
        new_score = eval_seq(new_seq)

        # accept if better; otherwise allow a vanishing uphill prob.
        accept = False
        if new_score < cur_score - 1e-12:
            accept = True
        else:
            # tiny probability to accept worsening move (helps escaping)
            prob = math.exp(-(new_score - cur_score) / max(1e-6, 0.0005))
            if np.random.rand() < prob:
                accept = True

        if accept:
            seq = new_seq
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score, best_seq = new_score, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {best_score:.6f}")
            # early exit if we already beat the target
            if best_score <= 1.503:
                break
        else:
            no_improve += 1
            # occasional random jitter to escape a deep plateau
            if no_improve % 2000 == 0:
                jitter = np.random.normal(scale=0.001, size=n)
                jitter -= jitter.mean()
                cand = seq + jitter
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()
                cand_score = eval_seq(cand)
                if cand_score < cur_score:
                    seq = cand
                    cur_score = cand_score
                    no_improve = 0
                    if cand_score < best_score:
                        best_score, best_seq = cand_score, seq.copy()
                        print(f"[{time.time() - start_time:.2f}s] Jitter improved to {best_score:.6f}")

    return best_seq, best_score

# 3️⃣  Smooth‑max gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient) where gradient is ∂/∂a.
    """
    n = a.shape[0]
    conv = _conv_self(a)          # length 2n‑1
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # stabilise
    Z = exp_scaled.sum()
    w = exp_scaled / Z                         # softmax weights

    # gradient: 2 * (a *convolve* w_rev)
    w_rev = w[::-1]
    # convolution a * w_rev using FFT (length 3n‑2)
    m = w_rev.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    start = m - 1
    grad = 2.0 * conv_rev[start:start + n]

    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, grad


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Exponential mirror‑descent update and simplex projection."""
    log_factor = -eta * grad
    # normalise to avoid overflow
    log_factor -= np.max(log_factor)
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total


def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float) -> tuple[np.ndarray, float]:
    """Fine‑tune the solution using the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    beta_start, beta_end = 0.5, 200.0
    eta = 0.4
    min_eta = 1e-13

    while time.time() < deadline:
        t = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + t * (beta_end - beta_start)

        smooth, grad = _smooth_max_and_grad(a, beta)

        # backtracking line search for the exponential step
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, grad, eta_try)
            new_score = _objective_score(a_new)
            if new_score <= cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Smooth descent β={beta:.2f} η={eta_try:.2e} score={new_score:.6f}")
            # slowly increase learning rate when progress is good
            eta = min(1.0, eta * 1.02)
        else:
            # tiny random perturbation to get out of a stall
            eps = 1e-4 * np.random.rand() * a
            a = np.clip(a + eps, 0.0, None)
            a /= a.sum()
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.5)

        if best_score <= 1.503:
            break

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
        evaluate_sequence(sequence)
    The routine consists of a broad parametric sweep, a greedy hill‑climb,
    and a final smooth‑max gradient descent.  The best vector found is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # --------------------------------------------------------------
    # Determine problem size (use the length of the provided starter if possible)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        n = starter.size
        seq0 = np.maximum(starter, 0.0)
        if seq0.sum() <= 0:
            raise ValueError
        seq0 = seq0 / seq0.sum()
    except Exception:
        n = 4096
        seq0 = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Initial evaluation
    # --------------------------------------------------------------
    best_seq = seq0.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Extensive parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 600.0  # keep ~10 minutes for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _quick_sweep(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Greedy hill‑climbing
    # --------------------------------------------------------------
    phase_deadline = deadline - 180.0
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, best_seq, best_score, phase_deadline)
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max gradient descent (fine tuning)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_seq, best_score = _smooth_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a clean list (non‑negative, sum > 0)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
