def _project_weighted_simplex(v, w):
    """Project onto simplex {x >=0, w.dot(x) = 1}."""
    if np.dot(w, np.maximum(v, 0)) == 0:
        # fallback uniform distribution
        return np.full_like(v, 1.0 / np.sum(w))
    # Determine mu bounds
    mu_low = -np.max(v / w) - 1.0
    mu_high = np.max(v / w) + 1.0
    # Ensure sum at low > 1, sum at high < 1
    def sum_w(mu):
        x = np.maximum(v - mu * w, 0.0)
        return np.dot(w, x)
    # Expand bounds adaptively
    while sum_w(mu_low) < 1.0:
        mu_low -= (abs(mu_low) + 1) * 2
    while sum_w(mu_high) > 1.0:
        mu_high += (abs(mu_high) + 1) * 2
    # binary search
    for _ in range(60):
        mu_mid = (mu_low + mu_high) / 2.0
        s = sum_w(mu_mid)
        if s > 1.0:
            mu_low = mu_mid
        else:
            mu_high = mu_mid
    x = np.maximum(v - mu_high * w, 0.0)
    # rescale in case of small error
    s = np.dot(w, x)
    if s > 0:
        x *= 1.0 / s
    else:
        x = np.full_like(v, 1.0 / np.sum(w))
    return x
```

"""
Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works on a *block representation* (K blocks covering the
full length n).  Each block has a constant height h_j and a known length
ℓ_j, so the full vector a is obtained by repeating h_j ℓ_j times.
The optimisation proceeds as follows:

1.  Initialise a collection of promising starting blocks:
    * the uniform distribution,
    * the block‐averaged version of the global starter ``height_sequence_1``,
    * a few random Dirichlet draws (block‑masses → heights).

2.  For each start run a **smooth‑max gradient descent** on the simplex.
    The non‑smooth max in the objective is replaced by a τ·log‑sum‑exp
    surrogate, giving a differentiable gradient.  The gradient w.r.t. the
    block heights is obtained by expanding the blocks, computing the
    soft‑max weights, and convolving with the reversed sequence (the same
    trick as in the reference implementation).

    After each gradient step we project the block vector onto the
    *weighted simplex* {h ≥ 0, Σ ℓ_j h_j = 1} using a fast bisection
    projection (h_i = max(v_i – μ·ℓ_i, 0)).  A back‑tracking line‑search
    on the true (non‑smoothed) objective guarantees monotone improvement.
    The temperature τ is started relatively high and slowly cooled, which
    helps the optimiser escape shallow plateaux.

3.  All runs share a global best sequence; any improvement is printed.

The whole search respects the 1000 s budget, uses only NumPy and the
provided ``evaluate_sequence`` (indirectly via ``_objective_score``), and
returns a plain Python list of floats whose sum is 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (same as reference)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Assumes sum(seq)=1; returns 2·n·max(conv(seq,seq))."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Weighted simplex projection (h >= 0, Σ w_i h_i = 1)
# ----------------------------------------------------------------------
def _project_weighted_simplex(v: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {h≥0, w·h = 1}."""
    # If all entries are zero, fall back to uniform distribution.
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / np.sum(w))

    # Initialise bounds for the Lagrange multiplier μ.
    mu_low = -np.max(v / w) - 1.0    # guarantees all v - μ·w > 0
    mu_high = np.max(v / w) + 1.0   # guarantees all v - μ·w ≤ 0

    # Helper: sum of weighted positives for a given μ.
    def _weighted_sum(mu):
        x = np.maximum(v - mu * w, 0.0)
        return np.dot(w, x)

    # Binary search for μ such that Σ w_i * max(v_i - μ w_i,0) = 1.
    for _ in range(60):
        mu_mid = (mu_low + mu_high) / 2.0
        s = _weighted_sum(mu_mid)
        if s > 1.0:
            mu_low = mu_mid
        else:
            mu_high = mu_mid

    h = np.maximum(v - mu_high * w, 0.0)
    # Numerical correction – enforce exact Σ w·h = 1.
    total = np.dot(w, h)
    if total > 0:
        h *= (1.0 / total)
    else:
        h = np.full_like(v, 1.0 / np.sum(w))
    return h


# ----------------------------------------------------------------------
# Block utilities
# ----------------------------------------------------------------------
def _expand_blocks(h: np.ndarray, lengths: np.ndarray) -> np.ndarray:
    """Repeat each block height according to its length."""
    return np.repeat(h, lengths)


def _block_grads_from_full(grad_a: np.ndarray, block_idx: np.ndarray, K: int) -> np.ndarray:
    """Aggregate per‑index gradient to per‑block gradient."""
    return np.bincount(block_idx, weights=grad_a, minlength=K)


# ----------------------------------------------------------------------
# Smooth‑max gradient descent on the block representation
# ----------------------------------------------------------------------
def _block_smooth_max_gd(
    h0: np.ndarray,
    lengths: np.ndarray,
    block_idx: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the τ·log‑sum‑exp surrogate.
    Returns the best full sequence found (as an array) and its score.
    """
    K = h0.size
    w = lengths.astype(float)               # weights for the simplex
    h = h0.copy()
    n = int(np.sum(lengths))

    # Initialise temperature τ from the current maximum convolution.
    a = _expand_blocks(h, lengths)
    cur_conv = _conv_self(a)
    tau = max(cur_conv.max() * 0.4, 1e-8)
    min_tau = 1e-10
    decay = 0.997

    iteration = 0
    while time.time() < deadline and tau > min_tau:
        iteration += 1
        a = _expand_blocks(h, lengths)
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv

        # ----- smooth max weights (softmax over convolution entries) -----
        shift = max_conv
        log_w = (conv - shift) / tau
        w_vec = np.exp(log_w)
        w_sum = w_vec.sum()
        if w_sum == 0:
            tau *= decay
            continue
        p = w_vec / w_sum

        # ----- gradient w.r.t. the full vector -----
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]
        grad_a = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # ----- aggregate to block gradient -----
        grad_h = _block_grads_from_full(grad_a, block_idx, K)

        # ----- back‑tracking line search on the true objective -----
        step = 1.0
        improved = False
        while step > 1e-14:
            h_new = h - step * grad_h
            h_new = _project_weighted_simplex(h_new, w)
            a_new = _expand_blocks(h_new, lengths)
            a_new = a_new / a_new.sum()          # renormalise (tiny numeric drift)
            obj_new = _objective_score(a_new)
            if obj_new < cur_obj - 1e-12:
                h = h_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] "
                        f"Block‑GD it={iteration:,} obj={obj_new:.6f} τ={tau:.2e}"
                    )
                break
            step *= 0.5

        if not improved:
            # Small random jitter to leave flat regions.
            eps = 1e-8
            h = h + np.random.uniform(-eps, eps, size=K)
            h = np.maximum(h, 0.0)
            h = _project_weighted_simplex(h, w)

        tau *= decay

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative step‑function under the 1000 s budget.
    Returns a list of floats whose sum is 1.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # Problem size
    n = 4096                       # fixed length used in the reference runs
    block_sz = 32                  # each block will contain ~32 entries
    K = (n + block_sz - 1) // block_sz
    lengths = np.full(K, block_sz, dtype=int)
    lengths[-1] = n - block_sz * (K - 1)      # adjust last block

    # Mapping from full index to block index (used for gradient aggregation)
    block_idx = np.repeat(np.arange(K), lengths)

    # ------------------------------------------------------------------
    # Baseline uniform distribution
    # ------------------------------------------------------------------
    uniform_full = np.full(n, 1.0 / n)
    best_seq = uniform_full.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter exists, evaluate and possibly adopt it
    # ------------------------------------------------------------------
    try:

            starter = np.maximum(starter, 0.0)
            s = starter.sum()

                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter better: score={sc:.6f}")
                # also build a block‑average version for the block optimiser
                blk_h = np.empty(K, dtype=float)
                pos = 0
                for j, L in enumerate(lengths):
                    blk_h[j] = starter[pos : pos + L].mean()
                    pos += L
                starter_block = blk_h
    except Exception:
        starter_block = None

    # ------------------------------------------------------------------
    # Helper to create a random block start from a Dirichlet on block masses
    # ------------------------------------------------------------------
    rng = np.random.default_rng()

    def _random_block_start():
        # concentration parameter drawn log‑uniformly
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        masses = rng.gamma(shape=c, scale=1.0, size=K)
        masses /= masses.sum()                     # Σ masses = 1
        # Convert block masses → heights (mass = ℓ·h)
        h = masses / lengths
        return h

    # ------------------------------------------------------------------
    # Run a handful of deterministic starts (uniform + possibly global)
    # ------------------------------------------------------------------
    starts = [np.full(K, 1.0 / n)]  # uniform block heights (since each entry is 1/n)
    if starter_block is not None:
        starts.append(starter_block)
    # add a few simple parametric families on the block level
    idx = np.arange(K, dtype=float)

    # geometric decay on blocks
    for eps in np.logspace(-5, -1, 5):
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        starts.append(geo)

    # power law on blocks
    for p in np.linspace(-2.0, 2.0, 5):
        pw = (idx + 1.0) ** p
        pw /= pw.sum()
        starts.append(pw)

    # exponential from centre (symmetric)
    dist = np.minimum(idx, K - 1 - idx)
    for lam in np.logspace(-4, -0.5, 5):
        expc = np.exp(-lam * dist)
        expc /= expc.sum()
        starts.append(expc)

    # ------------------------------------------------------------------
    # Main optimisation loop – many random restarts with block‑GD
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:   # keep a few seconds for final wrap‑up

            lengths,
            block_idx,
            sub_deadline,
            best_seq,

            start_time,

    # ------------------------------------------------------------------

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
