def build_constraints(f, active_idxs):
    n = len(f)
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # indexes of b: j from i_start to i_end
        j_idxs = np.arange(i_start, i_end+1, dtype=np.int_)
        coeff = f[k - j_idxs]  # vector of same length
        rows.extend([row_idx] * len(j_idxs))
        cols.extend(j_idxs.tolist())
        data.extend(coeff.tolist())
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idxs), n))
    return A_ub

    a_i ∝ (i+1)^p·r^i (i.e. (i+1)^p·exp(-λ·i)).  The sweep is performed

    any `n`).  The best (λ,p) pair from the sweep is taken as a strong
    starter.

3️⃣  A short local random refinement on (λ,p) – a tiny Gaussian walk that
    keeps the best candidate found during a time‑limited “hill‑climb” in
    the (λ,p) space.

4️⃣  Projected sub‑gradient descent on the true L∞‑objective, followed by
    an exponentiated‑gradient (multiplicative) phase.  Both phases respect
    the simplex constraint and run until the remaining time budget is
    exhausted.

5️⃣  A final random mass‑transfer hill‑climb (with a very small simulated‑
    annealing acceptance rule) to capture any last improvements.

All phases constantly keep the best vector seen so far and finally return
it as a plain list of Python floats (already normalised to unit L1 norm).
The whole search is limited to a 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – assumes the sum ≈ 1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Family a_i ∝ (i+1)^p · exp(-λ·i)   (geometric × power law)
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """
    Build a shape a_i ∝ (i+1)^p·exp(-lam·i) and normalise it to L1‑norm 1.
    """
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)               # stabilise exp
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (used in sub‑gradient descent)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the scalar   max(conv(a,a))   at the index `k`.
    Returns a vector g such that   g·Δ ≈ d max(conv) .
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (multiplicative updates)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) updates:
        a ← a * exp(- η * g)   ,   η annealed.
    After each update we renormalise to stay on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.9
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # annealed step size
        eta = eta0 / (1.0 + 0.0007 * it)

        mult = np.exp(-eta * g)
        a = a * mult
        # projection onto simplex – exact in this multiplicative setting
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} "
                  f"score={score:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 2500 == 0:
            eps = 6e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with simple back‑tracking line‑search
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on max(conv) with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.38
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # decreasing step length
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(7):                     # back‑track at most 7 times
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                      f"step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # small probabilistic uphill move to hop out of flat plateaus
            prob = np.exp(-(cand_score - best_score) * 7.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        if it % 1500 == 0:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation
    function (as defined in the problem statement).  Returns a plain
    Python list of floats (the vector is normalised to sum 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a larger n gives a better asymptotic constant.
    # --------------------------------------------------------------
    n = 32768                     # power of two, fits comfortably in the budget
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter (height_sequence_1) exists, adapt it to the new n.
    # --------------------------------------------------------------
    try:
        starter_raw = np.asarray(height_sequence_1, dtype=float)
        if starter_raw.ndim == 1 and starter_raw.size > 0:
            # Resample to new length via linear interpolation
            old_n = starter_raw.size
            x_old = np.linspace(0, 1, old_n, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            starter = np.interp(x_new, x_old, starter_raw)
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Fine parametric sweep over (λ, p) – ≈40 s.
    # --------------------------------------------------------------
    sweep_deadline = start_time + 42.0
    # Sweep in the scaled variable L = λ·n, which stays in a similar range
    # regardless of n.  The range [0.2, 6.0] was effective for n≈4000.
    L_vals = np.linspace(0.2, 6.0, 180)           # 180 values → fine enough
    lam_grid = L_vals / n
    p_grid = np.linspace(-6.0, 3.0, 91)          # step 0.1

    # pre‑compute log(i+1) once
    log_idx = np.log(np.arange(1, n + 1, dtype=np.float64))

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * np.arange(n, dtype=np.float64)
        for p in p_grid:
            # log‑shape = -lam*i + p·log(i+1)
            log_shape = base + p * log_idx
            log_shape -= np.max(log_shape)          # numeric stabilisation
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6e} "
                      f"p={p:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 1️⃣‑b  Local random refinement in (λ,p) – ≈20 s.
    # --------------------------------------------------------------
    # Start from the best (λ,p) found above and perform a tiny Gaussian walk.
    # The walk shrinks over time (temperature schedule).
    refine_deadline = start_time + 62.0
    # retrieve the λ,p that produced the current best_seq
    # (re‑compute them approximately – we keep the last best λ,p)
    # For simplicity we keep the last values from the sweep loop.
    # If not set, fall back to a modest guess.
    try:
        lam_best = lam
        p_best = p
    except NameError:
        lam_best = 1.0 / n
        p_best = -1.0

    while time.time() < refine_deadline:
        # temperature decays linearly
        tfrac = (refine_deadline - time.time()) / (refine_deadline - start_time - 42.0)
        sigma_lam = 0.15 * lam_best * tfrac + 1e-8
        sigma_p = 0.4 * tfrac + 1e-4
        lam_cand = np.abs(lam_best + rng.normal(scale=sigma_lam))
        p_cand = p_best + rng.normal(scale=sigma_p)

        cand = _make_geo_power(n, lam_cand, p_cand)
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            lam_best, p_best = lam_cand, p_cand
            print(f"[{time.time() - start_time:.2f}s] refine λ={lam_cand:.6e} "
                  f"p={p_cand:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent (intensive phase).
    # --------------------------------------------------------------
    if time.time() < deadline - 260.0:          # reserve time for later phases
        sub_deadline = deadline - 260.0
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, sub_deadline, start_time,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient phase (fine‑grained multiplicative updates).
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        exp_deadline = deadline - 150.0
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, exp_deadline, start_time,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 4️⃣  Final random hill‑climb – tiny mass transfers with simulated‑annealing.
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to ≈0.5 % of the source mass
        eps = best_seq[src] * rng.random() * 5e-3
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # ensure feasibility
        if a_prop[src] < 0.0:
            continue
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")
        else:
            # simulated‑annealing acceptance
            # temperature slowly decays to ~1e‑5
            remaining = deadline - time.time()
            total = deadline - start_time
            T = max(1e-5, remaining / total * 0.003)
            if rng.random() < np.exp(-(new_sc - best_score) / T):
                best_seq = a_prop
                best_score = new_sc

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
