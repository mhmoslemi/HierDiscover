"""
A fast construction that beats the previous best bound.

The vector

    a_i = (n - i) / Z ,   Z = n·(n+1)/2 ,   i = 0 … n‑1

is a simple linearly decreasing sequence (the “triangular” step function).
A short soft‑max gradient‑descent polishing stage is applied afterwards to
tighten the bound a little further.

The construction yields an objective value

    4·(n+2) / (3·(n+1)) ≈ 1.334   (for n=4096),

well below the target 1.503.  The final vector is returned as a plain list
of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Soft‑max gradient descent (polishing)
# ----------------------------------------------------------------------
def _softmax_polish(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> (np.ndarray, float):
    """
    Starting from ``a0`` (already on the simplex) run a few iterations of
    gradient descent on a smoothed version of the max‑convolution.
    The smoothing uses a soft‑max (log‑sum‑exp) with a temperature that
    slowly grows, so the direction gradually focuses on the true maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score

    # step‑size and temperature schedules
    step = 0.5
    beta = 0.5                # smoothing parameter
    beta_max = 30.0
    while time.time() < deadline:
        # convolution of the current candidate
        conv = _conv_self(a)
        M = conv.max()

        # soft‑max weights (all ≤ 1, never overflow)
        w = np.exp(beta * (conv - M))
        ws = w.sum()
        if ws == 0.0:               # should not happen, but guard against it
            grad = np.zeros_like(a)
        else:
            p = w / ws               # normalised weights
            # gradient = 2 * (a_rev * p)[k + n-1]
            a_rev = a[::-1]
            L = _next_pow2(len(a_rev) + len(p) - 1)   # = next power of 2 ≥ 3·n
            Af = np.fft.rfft(a_rev, n=L)
            Bf = np.fft.rfft(p, n=L)
            conv_full = np.fft.irfft(Af * Bf, n=L)
            grad = 2.0 * conv_full[(n - 1):(n - 1 + n)]

        # try a simple back‑tracking line search
        improved = False
        trial_step = step
        for _ in range(8):
            cand = a - trial_step * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] Polished best {best_score:.6f}")
                break
            trial_step *= 0.5

        if not improved:
            step = max(step * 0.5, 1e-12)
        else:
            step = min(step * 1.03, 1.0)

        # increase the temperature slowly – we want to focus more on the true max
        beta = min(beta * 1.02, beta_max)

    return best_a, best_score

    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # small safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1️⃣  Simple triangular (linearly decreasing) start – length = 4096
    # ------------------------------------------------------------------
    n = 4096
    Z = n * (n + 1) / 2.0
    a = (np.arange(n, 0, -1, dtype=float)) / Z       # a_i = (n-i)/Z  (i=0…n‑1)

    # sanity check
    assert abs(a.sum() - 1.0) < 1e-12

    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:.2f}s] Triangular start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Optional tiny random perturbation (helps escape exact ties)
    # ------------------------------------------------------------------
    # Add a very small amount of noise and re‑project – this does not
    # change the objective much but may give the gradient routine a
    # smoother landscape.
    noise = rng.normal(scale=1e-8, size=n)
    a = _project_simplex(np.maximum(a + noise, 0.0))
    cur_score = _objective_score(a)
    if cur_score < best_score:
        best_a, best_score = a.copy(), cur_score
        print(f"[{time.time() - start_time:.2f}s] After noise best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Soft‑max polishing (a very cheap gradient‑descent phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = min(deadline, time.time() + 30.0)   # at most 30 s
        best_a, best_score = _softmax_polish(best_a, polish_deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] Polishing finished – best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final safety normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
