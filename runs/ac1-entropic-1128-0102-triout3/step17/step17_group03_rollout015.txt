"""
Gradient‑based smoothing optimizer for the self‑convolution max problem.

The objective (for a probability vector a, Σa = 1) is
    F(a) = 2·n·max(conv(a,a))

We minimise max(conv) by a smooth approximation:
    Mγ(a) = (1/γ)·log Σ_t exp(γ·conv[t])
Its gradient is a weighted sum of the gradients of the individual
convolution entries, where the weights are a soft‑max over conv.
A simple projected (or exponentiated) gradient descent with an
adaptive step size and a slowly increasing γ quickly flattens the
convolution and improves the bound.

The search consists of:
  – uniform start,
  – several random Dirichlet draws,
  – a sparse random start,
  – the smoothed gradient descent,
  – a cheap random local polishing step.

All vectors are projected onto the probability simplex after each
update, guaranteeing feasibility for `evaluate_sequence`.  The routine
honours the 1000 s wall‑time limit and always returns the best vector
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing
# ----------------------------------------------------------------------
start_time = None  # set in propose_candidate


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of u and v using FFT (real inputs)."""
    L = _next_pow2(len(u) + len(v) - 1)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)
    return conv[: len(u) + len(v) - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # should not happen, but fallback to uniform
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (vectors are kept on the simplex → denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Initialisation helpers
# ----------------------------------------------------------------------
def _random_dirichlet_initial(
    n: int,
    rng: np.random.Generator,
    deadline: float,
    max_trials: int = 500,
):
    """Draw many Dirichlet vectors with a wide range of concentrations and keep the best."""
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻⁴, 10⁰]  (very spiky → near‑deterministic)
        conc_log = rng.uniform(-4.0, 0.0)
        conc = 10.0 ** conc_log
        a = rng.dirichlet(np.full(n, conc))
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet α={conc:.4e} → score={sc:.6f}")
    return best_seq, best_score


def _sparse_initial(n: int, sparsity_ratio: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random sparse vector with roughly ``sparsity_ratio`` non‑zero entries.
    Non‑zeros are exponential and the vector is renormalised.
    """
    k = max(2, int(sparsity_ratio * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 2️⃣  Smoothed gradient descent (soft‑max over convolution entries)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a0: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
):
    """
    Minimise max(conv) using a soft‑max approximation.
    Uses an exponentiated‑gradient style update and projects onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    cur = cur_score

    # hyper‑parameters
    gamma = 5.0                # initial smoothing strength
    gamma_max = 150.0
    step = 1.0                 # base step size for the (additive) update
    step_min = 1e-6

    while time.time() < deadline:
        # ----- convolution and soft‑max weights -----
        conv = _conv_self(a)
        max_c = conv.max()
        # weights w_t = exp(gamma·(conv[t] – max_c)) ≤ 1
        w = np.exp(gamma * (conv - max_c))

        # ----- gradient of the smoothed max -----
        # grad_i = 2·Σ_t w_t·a_{t‑i}  (cross‑correlation of w with a)
        grad_full = _conv_fft(w, a[::-1])          # length = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice of length n

        # ----- line‑search style additive update (projected) -----
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            cand_score = _objective_score(cand)
            if cand_score < cur:
                a = cand
                cur = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] GD new best {best_score:.6f}")
                break

        # ----- adapt step size -----
        if not improved:
            step = max(step * 0.5, step_min)
        else:
            step = min(step * 1.05, 5.0)

        # ----- slowly increase smoothing -----
        if gamma < gamma_max:
            # linear increase with elapsed time
            elapsed = time.time() - start_time
            gamma = min(gamma_max, 5.0 + (gamma_max - 5.0) * elapsed / 800.0)

        # ----- occasional random mass transfer to escape flat regions -----
        if rng.random() < 0.01:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src != dst and a[src] > 1e-12:
                delta = a[src] * rng.random() * 0.005
                cand = a.copy()
                cand[src] -= delta
                cand[dst] += delta
                cand = cand / cand.sum()
                cand_score = _objective_score(cand)
                if cand_score < cur:
                    a = cand
                    cur = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()
                        print(
                            f"[{time.time() - start_time:.2f}s] GD random move improved to {best_score:.6f}"
                        )
    return best_a, best_score

# ----------------------------------------------------------------------
def _random_local_refinement(
    a0: np.ndarray,
    cur_score: float,
    deadline: float,
    rng: np.random.Generator,
):
    """
    Very cheap pairwise transfers – only accept strict improvements.
    Acts as a final polishing step.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:

        cand = a.copy()

        cand[dst] += delta

        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = a.copy()

    return best_a, best_score

    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – stay comparable to the reference code (n = 4096)
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 0️⃣  Uniform start
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=float)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet initialisation (≈30 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 30.0)
    if time.time() < dirichlet_deadline:
        seq, sc = _random_dirichlet_initial(
            n, rng=rng, deadline=dirichlet_deadline, max_trials=800
        )
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] After Dirichlet best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1b️⃣  Sparse random start (helps explore distinct basins)
    # ------------------------------------------------------------------
    if time.time() < dirichlet_deadline:
        sparse = _sparse_initial(n, sparsity_ratio=0.02, rng=rng)
        sc = _objective_score(sparse)
        if sc < best_score:
            best_seq, best_score = sparse, sc
            print(f"[{time.time() - start_time:.2f}s] Sparse start score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.75
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed gradient descent (until {gd_deadline - start_time:.1f}s)")
        best_seq, best_score = _smooth_gradient_descent(
            best_seq, best_score, gd_deadline, rng
        )
        print(f"[{time.time() - start_time:.2f}s] GD finished – best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final cheap random polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = min(deadline, time.time() + 30.0)
        print(f"[{time.time() - start_time:.2f}s] Starting final local refinement")
        best_seq, best_score = _random_local_refinement(
            best_seq, best_score, local_deadline, rng
        )
        print(f"[{time.time() - start_time:.2f}s] Local refinement finished – best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return (strict feasibility for the evaluator)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```

This implementation introduces a smoothed‑gradient descent phase based on a soft‑max over the convolution entries, which provides a smoother landscape than the plain sub‑gradient used previously.  The method projects onto the probability simplex after each update, explores the space with occasional random mass transfers, and ends with a short local‑search polishing stage.  It respects the 1000 s time budget and returns the best sequence found.
