"""
Propose a step‑function (non‑negative vector) that minimises
    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm improves on the reference implementation by
* adding a rich warm‑start (Dirichlet, Gaussian, cosine‑power, sine‑squared)
* performing projected sub‑gradient descent on the *maximum* convolution entry
  (instead of variance flattening)
* using a more aggressive targeted hill‑climb that tests several random
  destination positions per iteration
* keeping the final mixing / light random‑pair refinement.

All stages respect the 1000 s budget and the vector is kept on the
probability simplex (non‑negative, sum = 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    Returns a 1‑D array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (assumes Σ a = 1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


def _max_gradient(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of conv_k w.r.t. a when k = k_max.
    For each i: ∂conv_k/∂a_i = 2·a_{k‑i} (if the index is inside the array, else 0).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    if i_start > i_end:
        return grad
    idx = np.arange(i_start, i_end + 1, dtype=int)
    grad[idx] = 2.0 * a[k_max - idx]
    return grad


def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    Return the pair (i, j) with i+j = k that gives the largest product a_i·a_j.
    Also returns that product value.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    max_pos = np.argmax(contrib)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(contrib[max_pos])


# ----------------------------------------------------------------------
# 0️⃣  Warm‑start (diverse candidates)
# ----------------------------------------------------------------------


def _warm_start_combined(n: int,
                         best_a: np.ndarray,
                         best_score: float,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Generates a variety of candidates (Dirichlet, Gaussian, cosine‑power,
    sine‑squared) while time permits.  Keeps the best found.
    """
    rng = np.random.default_rng()
    while time.time() < deadline:
        choice = rng.integers(0, 5)
        if choice == 0:
            # Dirichlet draw with random concentration
            log_c = rng.uniform(np.log(0.05), np.log(20.0))
            c = np.exp(log_c)
            sample = rng.gamma(shape=c, scale=1.0, size=n)
        elif choice == 1:
            # Truncated Gaussian centred in the middle
            sigma = rng.uniform(0.03, 0.20) * n
            positions = np.arange(n)
            sample = np.exp(-0.5 * ((positions - n / 2) / sigma) ** 2)
            # add slight random modulation
            sample *= rng.uniform(0.8, 1.2, size=n)
        elif choice == 2:
            # Cosine‑power shape (non‑negative half‑wave)
            p = rng.uniform(0.5, 4.0)
            positions = np.arange(n)
            half = np.cos(np.pi * (positions - (n - 1) / 2) / n)
            half = np.maximum(half, 0.0)
            sample = half ** p
        elif choice == 3:
            # Sine‑squared shape – a classic smooth bump
            positions = np.arange(1, n + 1)
            sample = np.sin(np.pi * positions / (n + 1)) ** 2
        else:
            # Tiny random perturbation of the current best
            sample = best_a + rng.normal(scale=0.001, size=n)

        sample = np.maximum(sample, 0.0)
        s = sample.sum()
        if s < 1e-12:
            continue
        a = sample / s
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Warm‑candidate #{choice} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 1️⃣  Sub‑gradient descent on the *maximum* convolution entry
# ----------------------------------------------------------------------


def _max_subgrad_descent(a0: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that directly reduces the peak of the
    self‑convolution.  Uses a diminishing stepsize η = η₀ / √t.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    it = 0
    eta0 = 0.5  # initial stepsize (tuned experimentally)

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGradIter={it:,}  score={cur_score:.6f}")

        # gradient of the currently maximal entry
        k_max = int(conv.argmax())
        grad = _max_gradient(a, k_max)

        # classic 1/√t schedule
        eta = eta0 / np.sqrt(it + 1)
        a = _project_simplex(a - eta * grad)

        # occasional tiny jitter to avoid plateaus
        if np.random.rand() < 0.004:
            eps = np.random.normal(scale=0.001, size=n)
            a = _project_simplex(np.maximum(a + eps, 0.0))

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Aggressive targeted hill‑climb
# ----------------------------------------------------------------------


def _targeted_hill_climb(a: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly attacks the current maximal convolution entry.
    For each iteration we examine the pair (i,j) that contributes most to
    that entry, then try several random destinations for a tiny mass transfer.
    Only improving moves are kept.
    """
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        i, j, _ = _max_contributions(best_a, k_max)
        if i == -1:
            continue

        # source is the heavier partner of the dominant pair
        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-12:
            continue

        # try a handful of random destinations and keep the best improvement
        improved = False
        for _ in range(5):
            dst = rng.integers(0, n)
            if dst == src:
                continue
            delta = rng.uniform(0.0, min(best_a[src], 0.01))
            if delta <= 0.0:
                continue
            cand = best_a.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective_score(cand, n, L_fft)
            if sc < best_score:
                best_score = sc
                best_a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] Hill src={src}→{dst} δ={delta:.5f} → score={sc:.6f}")
                break  # accept first improvement
        if not improved:
            # occasional random jitter even when no improvement
            if np.random.rand() < 0.001:
                eps = np.random.normal(scale=0.0005, size=n)
                best_a = _project_simplex(np.maximum(best_a + eps, 0.0))
    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final mixing + light random‑pair hill‑climb (as in the reference)
# ----------------------------------------------------------------------


def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Convex mixing with the uniform vector followed by cheap random moves."""
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the runner
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of non‑negative floats (approximately unit sum).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # problem size – 4096 is a good compromise between quality and speed
    n = 4096
    L_fft = _next_pow2(2 * n - 1)   # length for a*a via FFT

    # ------------------------------------------------------------------
    # 0️⃣  Trivial initialiser (uniform) and optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # optional global starter (provided as a global variable by the harness)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start (≈5 % of the total budget)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * (deadline - start_time))
    if time.time() < warm_deadline:
        best_a, best_score = _warm_start_combined(
            n, best_a, best_score, L_fft, warm_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent on the max convolution entry (~60 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        grad_deadline = time.time() + 0.60 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_a, best_score = _max_subgrad_descent(best_a, n, L_fft, grad_deadline)
        print(f"[{time.time() - start_time:.2f}s] After descent best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Aggressive targeted hill‑climb (~30 % of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + 0.30 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Targeted hill‑climb")
        best_a, best_score = _targeted_hill_climb(best_a, n, L_fft, hill_deadline)

    # ------------------------------------------------------------------
    # 4️⃣  Final light mixing & random‑pair refinement (remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement")
        best_a, best_score = _final_refinement(best_a, n, L_fft, deadline)

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
