#!/usr/bin/env python
"""
propose_candidate
-----------------

A specialised search for a non‑negative step‑function (vector) that
minimises

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the linear convolution.  The implementation builds on
the previous heuristic but adds two new phases:

1. **Projected sub‑gradient descent on the ℓ∞‑norm** – the gradient of
   the maximal convolution entry is a simple shifted copy of the
   current vector; after a step we project back onto the probability
   simplex.

2. **Fine‑grained incremental hill‑climbing** – we maintain the full
   convolution array and, for the current maximal entry, move a tiny
   amount of probability mass from the larger contributor of the
   responsible pair to a randomly chosen destination.  The convolution
   is updated *incrementally* (O(n) per move) using the exact algebraic
   update formula, which allows many cheap trials.  Several random
   destinations are examined and the best improving move is kept.

Both phases are wrapped by a short warm‑start (Dirichlet draws) and a
final light mixing with the uniform distribution.  The algorithm runs
until the supplied time budget (default 1000 s) expires and returns the
best vector it has seen.  All numbers are non‑negative and the vector
is normalised to sum to one, satisfying the checks of ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by evaluate_sequence (assumes sum(a)=1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    For a given convolution index k, return the pair (i, j) with the
    largest product a_i·a_j (where i+j=k) together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    max_pos = np.argmax(contrib)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(contrib[max_pos])


def _apply_incremental_move(a: np.ndarray,
                            conv: np.ndarray,
                            src: int,
                            dst: int,
                            delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Exact convolution update after moving ``delta`` mass from ``src`` to ``dst``.
    Returns the new vector (copy) and the updated convolution (copy).
    Complexity O(n).
    """
    n = a.shape[0]
    # copy to avoid mutating the originals
    a_new = a.copy()
    conv_new = conv.copy()

    # linear part: 2 * (a * d) where d has -delta at src, +delta at dst
    #   effect for src  -> subtract 2*delta * a[k-src] on k∈[src, src+n-1]
    #   effect for dst  -> add    2*delta * a[k-dst] on k∈[dst, dst+n-1]
    # both use the *original* a (before the move)
    conv_new[src:src + n] -= 2.0 * delta * a[:n]
    conv_new[dst:dst + n] += 2.0 * delta * a[:n]

    # quadratic part d*d (only 3 entries)
    conv_new[2 * src] += delta * delta
    conv_new[2 * dst] += delta * delta
    if src != dst:
        conv_new[src + dst] -= 2.0 * delta * delta

    # finally update a
    a_new[src] -= delta
    a_new[dst] += delta
    # guard against negative rounding errors
    a_new = np.maximum(a_new, 0.0)
    # renormalise to keep sum exactly 1 (the move conserves mass,
    # but tiny floating errors may appear)
    a_new /= a_new.sum()
    return a_new, conv_new

            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score

def _subgrad_max_descent(a0: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on max(conv(a,a)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    it = 0
    eta0 = 0.6   # a bit smaller than for variance descent
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGradIter={it:,}  score={cur_score:.6f}")

        # index attaining the maximum
        k_max = int(conv.argmax())
        # sub‑gradient: g_i = 2 * a_{k_max-i} (if inside range)
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[k_max - idx[mask]]

        # step size schedule  η = η0 / sqrt(it)
        eta = eta0 / np.sqrt(it + 1)
        a_new = a - eta * g
        a = _project_simplex(a_new)

    return best_a, best_score

        if best_a[src] < 1e-12:
            # nothing to move
            continue

        # try a few candidate destinations
        best_local_score = best_score
        best_move = None
        for _ in range(max_trials_per_iter):
            dst = rng.integers(0, n)
            if dst == src:
                continue
            # pick a tiny delta, scaled to the source weight
            delta = rng.uniform(min_delta_frac, max_delta_frac) * best_a[src]
            if delta <= 0.0:
                continue
            # compute new a and conv by incremental update (exact)
            cand_a, cand_conv = _apply_incremental_move(best_a,
                                                        best_conv,
                                                        src, dst, delta)
            cand_score = float(2.0 * n * cand_conv.max())
            if cand_score < best_local_score - 1e-9:   # strict improvement
                best_local_score = cand_score
                best_move = (cand_a, cand_conv)

        # if we found an improving move, commit it
        if best_move is not None:
            best_a, best_conv = best_move
            best_score = best_local_score
            print(f"[{time.time() - start_time:.2f}s] HillClimb src={src}→dst={dst} δ={delta:.6f} → score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final gentle mixing with uniform & random‑pair hill‑climb
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] LightHill i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of non‑negative floats (approximately unit sum).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a well‑balanced trade‑off
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Initialiser (uniform) and optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # If a global starter is available, use it
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter /= s
                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start (Dirichlet draws, about 5 % of the budget)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * (deadline - start_time))
    if time.time() < warm_deadline:
        best_a, best_score = _warm_start(n, best_a, best_score, L_fft, warm_deadline)

    # ------------------------------------------------------------------
    # 2️⃣ Sub‑gradient descent on max(conv) (~45 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.45 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient max descent")
        best_a, best_score = _subgrad_max_descent(best_a, n, L_fft, sub_deadline)
        print(f"[{time.time() - start_time:.2f}s] After sub‑grad best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Fine‑grained hill‑climb (~45 % of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + 0.45 * (deadline - time.time())
        print(f"[{time.time() - start_time:.2f}s] Fine hill‑climb")
        best_a, best_score = _fine_hill_climb(best_a, n, L_fft, hill_deadline)

    # ------------------------------------------------------------------
    # 4️⃣ Final light mixing + random‑pair hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement")
        best_a, best_score = _final_refinement(best_a, n, L_fft, deadline)

    # ------------------------------------------------------------------
    # Normalise and sanitise before returning
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
