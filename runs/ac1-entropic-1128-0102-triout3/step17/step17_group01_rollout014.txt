"""
Propose a step‑function (non‑negative vector) that minimises  

        2 · n · max(conv(a,a)) / (sum(a)²)

using a large, diverse pool of seeds and a long, multi‑stage optimisation.
The stages are

1️⃣  Seed generation – uniform, the global starter, geometric×power shapes,
    beta‑shape (i+1)ᵅ·(n‑i)ᵝ, logistic shapes and several Dirichlet draws.
    The K best seeds (by the exact objective) are kept.

2️⃣  Adam on a smooth‑max surrogate.  The temperature β is increased
    gradually (β←β·β_factor every β_step iterations) up to a large value,
    thus tightening the surrogate to the true L∞‑objective.
    After each step the vector is projected onto the probability simplex.

3️⃣  Exponentiated‑gradient polishing (multiplicative updates on the true
    max‑convolution) with a decaying learning rate.

4️⃣  Projected sub‑gradient descent with back‑tracking.

5️⃣  Simulated‑annealing hill‑climbing that moves a tiny amount of mass
    between a few random indices; uphill moves are occasionally accepted.

All stages respect the non‑negativity / simplex constraint, constantly
track the best sequence found and stop just before the 1000 s budget.
The returned list is normalised to sum = 1 and can be fed to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  fβ(a)=log‑sum‑exp(β·conv(a,a)).
    Implements   g_i = 2 * Σ_j a_j p_{i+j}.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Fast exact objective (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact evaluation  2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape (already used in the reference)."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture with the uniform distribution."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Beta‑type shape:
        a_i ∝ (i+1)^{a} · (n-i)^{b}
    (a, b may be negative).  Returned vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    # add 1 to avoid zero to the power of negative
    left = (i + 1.0) ** a
    right = (n - i) ** b
    shape = left * right
    # replace NaNs (0**negative) by a very small positive number
    shape[np.isnan(shape)] = 0.0
    if shape.sum() == 0.0:
        # fallback to uniform
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= shape.sum()
    return shape


def _logistic_shape(n: int, s: float, p: float) -> np.ndarray:
    """
    Logistic‑type shape:
        a_i ∝ 1 / (1 + exp(s·(i‑mid)))^{p}
    where mid = (n‑1)/2.
    """
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    logistic = 1.0 / (1.0 + np.exp(s * (i - mid)))
    shape = logistic ** p
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (hard max)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (ordered‑pair version)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] += 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray, L_fft: int,
                   deadline: float, start: float,
                   best_seq: np.ndarray, best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates directly on the L∞-objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (hard max)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, L_fft: int,
                         deadline: float, start: float,
                         best_seq: np.ndarray, best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------


def _hill_climb(a: np.ndarray, L_fft: int,
                deadline: float, start: float,
                best_seq: np.ndarray, best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Random mass‑transfer moves with a simple temperature schedule."""
    n = a.shape[0]
    temperature = 1.0
    decay = 0.9995  # slowly cool down
    while time.time() < deadline:
        # select up to three source positions with non‑negligible mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        n_src = min(3, src_candidates.size)
        src = rng.choice(src_candidates, size=n_src, replace=False)
        # amount to move from each source (up to 0.4 % of its mass)
        eps = a[src] * rng.random(n_src) * 4e-3
        # choose destinations (allow repeats)
        dst = rng.integers(n, size=n_src)
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps
        # project back onto simplex (very cheap)
        prop = _project_simplex(prop)
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_seq = prop.copy()
            a = prop
            print(f"[{time.time() - start:.2f}s] hill‑climb score={new_score:.6f}")
            temperature = 1.0  # reset temperature after improvement
        else:
            # accept uphill with probability exp(-Δ/T)
            prob = np.exp(-(new_score - best_score) / temperature)
            if rng.random() < prob:
                a = prop
        temperature *= decay
    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises the evaluation function.  Returns the best sequence found
    as a plain list of floats (already normalised to sum = 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.12          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter (if provided by the environment).
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter → {cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Generate a large and diverse pool of seeds.
    # ------------------------------------------------------------------
    seeds = [best_seq]   # uniform (or starter) already present

    # Geometric × power family (already used in the reference)
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9995), 9)
    p_vals = np.linspace(-4.0, 3.0, 15)
    lam_vals = [0.0, 0.2, 0.5, 0.8]
    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Beta‑type shapes (edge‑heavy / centre‑light families)
    alphas = np.linspace(-3.0, 3.0, 9)
    betas = np.linspace(-3.0, 3.0, 9)
    for a in alphas:
        for b in betas:
            shape = _beta_shape(n, a, b)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Logistic families (smooth step‑like shapes)
    slopes = np.linspace(-0.02, 0.02, 9)   # small to moderate slope
    powers = np.linspace(0.5, 4.0, 7)
    for s in slopes:
        for p in powers:
            shape = _logistic_shape(n, s, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Dirichlet draws with a range of concentration parameters
    for alpha in [0.05, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # Keep only the K best seeds according to the exact objective.
    # ------------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 7
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimisation on a smooth‑max surrogate.
    # ------------------------------------------------------------------
    beta = 0.5
    max_beta = 4096.0
    beta_factor = 1.4
    beta_step = 800          # increase β every β_step iterations
    adam_lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    max_iter_per_vec = 12000

    # Continue as long as we have at least ~200 s for polishing.
    while time.time() < deadline - 250:
        for idx in range(len(pool)):
            a = pool[idx]
            m = np.zeros_like(a)
            v_vec = np.zeros_like(a)
            cur_score = pool_scores[idx]
            t = 0
            while time.time() < deadline - 250 and t < max_iter_per_vec:
                t += 1
                # --- surrogate (log‑sum‑exp) quantities ---
                conv = _conv_self_fft(a, L_fft)
                p = _softmax_weights(conv, beta)
                g = _grad_softmax(a, p, L_grad)

                # --- Adam update ---
                m = beta1 * m + (1.0 - beta1) * g
                v_vec = beta2 * v_vec + (1.0 - beta2) * (g * g)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v_vec / (1.0 - beta2 ** t)
                step = adam_lr * m_hat / (np.sqrt(v_hat) + eps)

                a_candidate = a - step
                a_candidate = _project_simplex(a_candidate)

                cand_score = _eval_fast(a_candidate, L_fft)

                # accept if improving, else occasional uphill move
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                else:
                    prob = np.exp(-(cand_score - cur_score) *
                                  5.0 / (np.sqrt(t) + 1.0))
                    if rng.random() < prob:
                        a = a_candidate
                        cur_score = cand_score

                # update pool entry and global best
                if cand_score < pool_scores[idx]:
                    pool_scores[idx] = cand_score
                    pool[idx] = a_candidate.copy()
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time() - start:.2f}s] Adam β={beta:.1f} it={t:,} → {cand_score:.6f}")

                # increase β periodically
                if t % beta_step == 0:
                    beta = min(beta * beta_factor, max_beta)

                # tiny random perturbation every few thousand steps
                if t % 3000 == 0:
                    eps_noise = 1e-4 * rng.random(n) * a
                    a = np.maximum(a + eps_noise, 0.0)
                    a = a / a.sum()

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing (hard max).
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft,
            deadline - 120, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent (hard max).
    # ------------------------------------------------------------------
    if time.time() < deadline - 70:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft,
            deadline - 70, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣  Simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        best_seq, best_score = _hill_climb(
            best_seq, L_fft,
            deadline - 5, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Final normalisation and return.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
