import time, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self(a):
    n=a.shape[0]
    L=_next_pow2(2*n-1)
    Af=np.fft.rfft(a, n=L)
    conv=np.fft.irfft(Af*Af, n=L)[:2*n-1]
    return conv

def _conv_fft(a,b):
    n=a.shape[0]
    m=b.shape[0]
    L=_next_pow2(n+m-1)
    Af=np.fft.rfft(a, n=L)
    Bf=np.fft.rfft(b, n=L)
    conv=np.fft.irfft(Af*Bf, n=L)[:n+m-1]
    return conv

def _objective_score(a):
    # assumes sum(a)==1
    return float(2.0 * a.shape[0] * _conv_self(a).max())

def _exp_update(a, grad, eta):
    log_factor = -eta * grad
    # shift so max is zero
    log_factor -= np.max(log_factor)
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        # fallback: no change
        return a
    return a_new / total

def _subgradient_step(a, eta):
    # compute conv and max
    conv = _conv_self(a)
    max_val = conv.max()
    # mask entries close to max (tolerance relative)
    eps = 1e-12
    mask = conv >= max_val - eps
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        w[:] = 1.0
        w_sum = w.size
    w /= w_sum
    w_rev = w[::-1]
    conv_w = _conv_fft(a, w_rev)
    start = len(w) - 1
    grad = 2.0 * conv_w[start - np.arange(a.shape[0])]
    a_new = _exp_update(a, grad, eta)
    return a_new, max_val

"""
Heuristic optimisation for the step‑function bound.

The algorithm works in three stages:

1️⃣  Random Dirichlet draws give a decent starting point (the best
    draw is kept).

2️⃣  Mirror‑descent on the *exact* max‑convolution.
    We keep a probability vector `a` (non‑negative, sum = 1) and
    repeatedly
        – find the maximal entry of the self‑convolution,
        – compute a sub‑gradient of that maximum (convolution with a
          weight that is uniform over all maximal indices),
        – perform an exponential update `a_i ← a_i·exp(−η·g_i)` and
          renormalise.
    A back‑tracking line‑search on the step size η ensures monotone
    improvement; if no progress is made a tiny random jitter is added.

3️⃣  If time permits a short hill‑climb (random pairwise mass transfers)
    is performed to squeeze a little more out of the current best.
    The best sequence found (normalised to sum = 1) is returned as a
    plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (returns full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Assumes Σ a = 1. Returns 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Mirror‑descent exponential update a ← a·exp(−η·grad) and renormalise."""
    # work in log‑space to stay numerically safe
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)               # avoid overflow
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        # fallback – keep the old vector
        return a.copy()
    return a_new / total


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution. Returns best (a, score)."""
    rng = np.random.default_rng()
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    eta0 = 0.7                     # initial step size
    min_eta = 1e-6
    eta = eta0
    no_improve = 0                # counter for stagnation
    it = 0

    while time.time() < deadline:
        it += 1
        # ------- current objective -------
        conv = _conv_self(a)
        max_val = conv.max()
        score = 2.0 * n * max_val

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] it={it} new best {score:.6f}")

        if best_score <= target_score:
            break

        # ------- sub‑gradient of the max -------
        # weight vector w is uniform over all maximal entries
        mask = conv >= max_val - 1e-12
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:                # should never happen, but be safe
            w[:] = 1.0
            w_sum = w.size
        w /= w_sum

        # gradient = 2·(a * w_rev)  (see derivation in the analysis)
        w_rev = w[::-1]
        conv_w = _conv_fft(a, w_rev)                 # length 3n‑2
        start = len(w) - 1                           # = 2n‑2
        grad = 2.0 * conv_w[start - np.arange(n)]

        # ------- line‑search on η -------
        eta_try = eta
        improved = False
        for _ in range(12):          # at most 12 halvings
            a_new = _exp_update(a, grad, eta_try)
            new_score = _objective_score(a_new)
            if new_score <= score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            # slowly increase η when we make progress
            eta = min(eta0, eta_try * 1.1)
            no_improve = 0
        else:
            # inject a tiny random jitter and shrink η
            eps = 1e-5 * rng.standard_normal(n) * a
            a = np.clip(a + eps, 0.0, None)
            a /= a.sum()
            eta = max(min_eta, eta * 0.5)
            no_improve += 1
            if no_improve >= 200:   # re‑start from the best seen so far
                a = best_a.copy()
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Small random hill‑climb (optional final polishing)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                deadline: float,
                best_score: float,
                target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfers; accepts only improvements."""
    rng = np.random.default_rng()
    n = a.shape[0]
    cur = a.copy()
    cur_score = best_score

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small fraction of mass from i to j
        delta = min(cur[i], 0.001 * cur[i])
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()
        new_score = _objective_score(cand)
        if new_score < cur_score:
            cur, cur_score = cand, new_score
            if new_score < best_score:
                best_score = new_score
                print(f"[{time.time() - start_time:.2f}s] hill‑climb score={new_score:.6f}")
            if best_score <= target_score:
                break
    return cur, best_score
