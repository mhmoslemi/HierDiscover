def _project_simplex(v):
    # v can be any shape (1D)
    # Algorithm from Duchi et al. "Efficient Projections onto the L1-ball ..."
    if np.all(v >= 0) and np.isclose(v.sum(), 1):

"""
Propose a step‑function (non‑negative vector) that minimizes

        score = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three phases:

1️⃣  Warm‑start – scan the Beta(p,p) family (including mixtures with the
    uniform distribution) and a few Dirichlet draws.  The best vector
    found becomes the seed for the next phase.

2️⃣  Smoothed‑max mirror descent – minimise a smooth approximation of the
    maximum of the self‑convolution.
    * Use the soft‑max weight w_k ∝ exp(β·(conv_k−max(conv)))  (β grows
      during the run, turning the smooth objective into the true max).
    * The gradient of  Σ_k w_k·conv_k w.r.t. a is
          ∇ = 2·(a_rev * w)[n‑1 : n‑1+n]               (linear convolution)
      which is computed efficiently with FFTs.
    * Update a on the probability simplex with an exponentiated‑gradient
      (mirror‑descent) step:
          log a ← log a − η·∇ ,   a ← softmax(log a).
      The step size η decays as  η0/√t .
    * After each iteration we evaluate the exact score and keep the best
      vector seen so far.

3️⃣  Final normalisation – clip any tiny negative values, renormalise to
    sum = 1 and return the list of floats.

The whole search respects the given time budget (default 1000 s) and
prints progress.  All heavy work is performed with NumPy + FFT, which
fits easily into the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (sum(a) is assumed to be 1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised Beta(p,p) density on the grid i=0,…,n‑1,
    a_i ∝ x_i^{p‑1} (1‑x_i)^{p‑1} with x_i=(i+0.5)/n.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _interpolate_to_n(arr: np.ndarray, n: int) -> np.ndarray:
    """Resize an arbitrary 1‑D array to length n by linear interpolation."""
    if arr.size == n:
        return arr.copy()
    # positions in [0, 1)
    old_pos = (np.arange(arr.size, dtype=np.float64) + 0.5) / arr.size
    new_pos = (np.arange(n, dtype=np.float64) + 0.5) / n
    interp = np.interp(new_pos, old_pos, arr, left=0.0, right=0.0)
    # ensure non‑negativity and normalisation
    interp = np.maximum(interp, 0.0)
    s = interp.sum()
    if s == 0.0:
        interp = np.full(n, 1.0 / n)
    else:
        interp /= s
    return interp


def _mirror_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror descent) step on the simplex:
        log a ← log a − η·grad,   a ← softmax(log a)
    """
    # avoid log(0)
    log_a = np.log(a + 1e-30)
    log_a -= eta * grad
    # exponentiate and normalise
    a_new = np.exp(log_a - np.max(log_a))   # subtract max for numerical stability
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and prints
    progress.  Returned list sums to ≈1, which is scale‑invariant for the
    evaluator.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so we exit before the system kills the process
    deadline = start_time + budget_s - 0.05

    # --------------------------------------------------------------
    # Problem size – a power of two for fast FFTs
    # --------------------------------------------------------------
    n = 4096                     # we keep the original size for fairness
    L_fft = _next_pow2(2 * n - 1)        # length for a⋆a
    L_fft_grad = _next_pow2(3 * n - 2)   # length for a_rev⋆w in the gradient

    # --------------------------------------------------------------
    # 0️⃣  Baseline (uniform) and optional global starter
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _score_of(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score = {best_score:.6f}")

    # try to use a user‑provided vector if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = _interpolate_to_n(starter, n)
        sc = _score_of(starter, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, starter.copy()
            print(f"[0.00s] Global starter improved score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Warm‑start with Dirichlet samples (≈5 % of the budget)
    # --------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * budget_s)
    rng = np.random.default_rng()
    while time.time() < warm_deadline:
        # draw concentration parameter log‑uniformly
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _score_of(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet warm‑start → score = {sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Scan the Beta(p,p) family (pure shapes + convex mixtures)
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Scanning Beta‑family")
    p_vals = np.linspace(0.05, 2.00, 40)        # extended range, finer grid
    for p in p_vals:
        shape = _beta_shape(n, p)

        # pure Beta shape
        sc = _score_of(shape, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, shape.copy()
            print(f"[{time.time() - start_time:.2f}s] pure p={p:.3f} → score = {sc:.6f}")

        # convex mixtures with the uniform vector (λ·shape + (1‑λ)·uniform)
        for lam in np.linspace(0.0, 1.0, 21):    # finer λ‑grid
            cand = lam * shape + (1.0 - lam) * uniform
            cand /= cand.sum()
            sc = _score_of(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] p={p:.3f} λ={lam:.2f} → score = {sc:.6f}")

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 3️⃣  Smoothed‑max mirror descent (the heavy lifting)
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Mirror‑descent refinement")
    a = best_a.copy()                     # current iterate
    # hyper‑parameters for the descent
    eta0 = 0.5                            # initial step‑size multiplier
    beta = 0.2                            # start with gentle smoothing
    beta_max = 8.0
    beta_inc = 0.2
    iter_cnt = 0

    while time.time() < deadline:
        # ----- evaluate current objective (exact max, not the smooth one) -----
        conv_full = _conv_self_fft(a, L_fft)          # length 2n‑1
        max_conv = conv_full.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] New best exact score = {cur_score:.6f}")

        # ----- smooth max via soft‑max weights -----
        # shift by max_conv for numerical stability
        exp_arg = beta * (conv_full - max_conv)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            # degenerate case – continue with uniform weights
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of Σ_k w_k·conv_k -----
        a_rev = a[::-1]
        Ar = np.fft.rfft(a_rev, n=L_fft_grad)
        Wr = np.fft.rfft(w, n=L_fft_grad)
        grad_full = np.fft.irfft(Ar * Wr, n=L_fft_grad)   # length L_fft_grad
        # slice corresponding to indices n‑1 … n‑1+n‑1
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- mirror (exponentiated‑gradient) update -----
        eta = eta0 / math.sqrt(iter_cnt + 1)
        a = _mirror_update(a, grad, eta)

        # ----- schedule updates -----
        iter_cnt += 1
        if iter_cnt % 3000 == 0 and beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

        # occasional tiny random perturbation to help escape shallow plateaux
        if iter_cnt % 15000 == 0:
            noise = rng.normal(scale=1e-4, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()

    # --------------------------------------------------------------
    # Normalisation & final return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # convert to plain Python floats (required by the outer harness)
    return [float(v) for v in best_a]
