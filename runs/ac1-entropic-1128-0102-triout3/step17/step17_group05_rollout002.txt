    left, right = 0.0, 1.0
    phi = (np.sqrt(5) - 1) / 2
    # initial points
    x1 = right - phi * (right - left)
    x2 = left + phi * (right - left)

def _random_dirichlet_search(n, deadline, best_seq, best_score, num_trials=2000):
    rng = np.random.default_rng()

"""
Propose a step function (list of non‑negative floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search works in three phases under a 1000 s time budget:

1️⃣  Random Dirichlet draws (exploratory, fast).
2️⃣  Simple parametric sweep of a symmetric “tent” shape.
3️⃣  A Frank‑Wolfe style optimisation on the simplex.
    – At each iteration the active index of the autocorrelation
      (where the max occurs) defines a linear sub‑gradient.
    – The linear‑minimisation oracle of Frank‑Wolfe is trivial:
      put all mass on the coordinate with the smallest sub‑gradient.
    – A cheap golden‑section line‑search finds the optimal step‑size
      (exact objective can be evaluated analytically because the
      update only changes f by a rank‑1 perturbation).
    – The iterate stays on the probability simplex, thus the
      denominator of the original evaluation stays constant (1).

The optimizer works on a moderate size (n = 1024) for speed, then the
result is up‑sampled (repeating each entry four times) to the required
length 4096 before returning.  All numbers are kept non‑negative and
normalised to sum = 1, so ``evaluate_sequence`` will never reject them.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a fast FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    The sequence is assumed to be non‑negative and already normalised.
    """
    n = seq.shape[0]
    conv = _conv_self_fft(seq)
    return float(2.0 * n * conv.max())


def _objective_gamma(conv_f: np.ndarray,
                     shift: np.ndarray,
                     i0: int,
                     gamma: float,
                     n: int) -> float:
    """
    Exact objective for the candidate f_new = (1‑γ) f + γ e_i0 .
    ``conv_f`` = f*f, ``shift`` = f * e_i0 (the shifted copy of f).
    """
    fac1 = (1.0 - gamma) ** 2
    fac2 = 2.0 * gamma * (1.0 - gamma)
    conv_new = fac1 * conv_f + fac2 * shift
    # the self‑convolution of the unit vector adds γ² at index 2·i0
    conv_new[2 * i0] += gamma * gamma
    return float(2.0 * n * conv_new.max())


def _line_search_gamma(conv_f: np.ndarray,
                       shift: np.ndarray,
                       i0: int,
                       n: int,
                       deadline: float) -> tuple[float, float]:
    """
    Golden‑section line search for the optimal γ∈[0,1] that minimises the
    true (non‑smoothed) objective.
    Returns (γ_opt, obj_opt).
    """
    left, right = 0.0, 1.0
    phi = (np.sqrt(5.0) - 1.0) / 2.0  # ≈0.618
    # initial interior points
    x1 = right - phi * (right - left)
    x2 = left + phi * (right - left)
    f1 = _objective_gamma(conv_f, shift, i0, x1, n)
    f2 = _objective_gamma(conv_f, shift, i0, x2, n)

    # tolerance on γ – 1e‑4 is plenty for our purpose
    while right - left > 1e-4 and time.time() < deadline:
        if f1 < f2:
            right = x2
            x2 = x1
            f2 = f1
            x1 = right - phi * (right - left)
            f1 = _objective_gamma(conv_f, shift, i0, x1, n)
        else:
            left = x1
            x1 = x2
            f1 = f2
            x2 = left + phi * (right - left)
            f2 = _objective_gamma(conv_f, shift, i0, x2, n)

    gamma_opt = (left + right) / 2.0
    obj_opt = _objective_gamma(conv_f, shift, i0, gamma_opt, n)
    return gamma_opt, obj_opt


# ----------------------------------------------------------------------
# 1️⃣  Random Dirichlet exploration
# ----------------------------------------------------------------------
def _random_dirichlet_search(n: int,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # number of draws is limited by the time budget
    while time.time() < deadline:
        # concentration parameter (log‑uniform between 10⁻² and 5)
        logc = rng.uniform(-2.0, 0.7)
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a = a / s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Simple “tent” (symmetric polynomial) sweep
# ----------------------------------------------------------------------
def _sweep_tent(n: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """
    Symmetric shape a_i ∝ (i+1)^p (n-i)^p, p∈[0.1,2].
    """
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(0.1, 2.0, 40):
        if time.time() >= deadline:
            break
        left = (idx + 1.0) ** p
        right = (n - idx) ** p
        seq = left * right
        s = seq.sum()
        if s == 0.0:
            continue
        seq = seq / s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Tent p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Frank‑Wolfe optimisation on the simplex
# ----------------------------------------------------------------------
def _frank_wolfe_optimize(a0: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float) -> tuple[np.ndarray, float]:
    """
    Classic Frank‑Wolfe (conditional gradient) method.
    The linear minimisation oracle is trivial – put all mass on the
    index with the smallest sub‑gradient (often a zero entry).
    A cheap golden‑section line search provides the optimal step size.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    # tolerance for stagnation
    tol_improve = 1e-7
    last_improve_iter = 0

    while time.time() < deadline - 0.1:
        it += 1
        conv_f = _conv_self_fft(a)
        max_val = conv_f.max()
        k_max = int(conv_f.argmax())

        # ----- sub‑gradient of the max‑convolution at k_max -----
        subgrad = np.zeros(n, dtype=np.float64)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        if i_end >= i_start:
            idxs = np.arange(i_start, i_end + 1)
            subgrad[idxs] = 2.0 * a[k_max - idxs]

        # index with smallest sub‑gradient (zero is best)
        i0 = int(np.argmin(subgrad))

        # ----- shift vector: (f * e_i0) -----
        shift = np.zeros(2 * n - 1, dtype=np.float64)
        shift[i0:i0 + n] = a

        # ----- line search for optimal step size -----
        gamma, obj_new = _line_search_gamma(conv_f, shift, i0, n, deadline)

        # ----- accept the move if it improves the objective -----
        if obj_new + 1e-12 < best_score:
            best_score = obj_new
            best_seq = (1.0 - gamma) * a + gamma * np.eye(1, n, i0).reshape(-1)
            print(
                f"[{time.time() - start_time:.2f}s] FW it={it:,} i0={i0} γ={gamma:.5f} score={obj_new:.6f}"
            )
            last_improve_iter = it

        # always update the iterate (the algorithm proceeds regardless)
        a = (1.0 - gamma) * a + gamma * np.eye(1, n, i0).reshape(-1)

        # stop if we have been idle for a while
        if it - last_improve_iter > 3000:
            print(f"[{time.time() - start_time:.2f}s] FW stagnated after {it} iterations.")
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: down‑sample a longer sequence to a shorter one (mean within blocks)
# ----------------------------------------------------------------------
def _downsample_to(seq_full: np.ndarray, target_n: int) -> np.ndarray:
    L = len(seq_full)
    factor = L // target_n
    if factor < 1:
        # fallback: uniform
        return np.full(target_n, 1.0 / target_n)
    trimmed = seq_full[: factor * target_n]
    reshaped = trimmed.reshape(target_n, factor)
    return reshaped.mean(axis=1)
