def _make_geo_power_shift(n, lam, p, shift=0):
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + shift + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape_sum = shape.sum()
    if shape_sum == 0:
        shape = np.full(n, 1.0/n)
    else:
        shape /= shape_sum
    return shape

"""
Fast two‑phase search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

Phase 1 – parametric sweep over a three‑parameter family  
        a_i ∝ (i+shift+1)^p · exp(-λ·i)
          λ (geom.)  in a fine grid,
          p (power)  in a fine grid,
          shift      a few integer offsets to soften the head of the distribution.
The best vector from the sweep is stored.

Phase 2 – smooth‑max Adam optimisation.
The non‑smooth max is replaced by a temperature‑smoothed soft‑max:
    w_k = exp(conv_k/τ) / Σ exp(conv/τ)
The gradient of τ·log Σ exp(conv/τ) with respect to a is
    g = 2 * (a ★ w)      (cross‑correlation)
and is updated with Adam; after each step the vector is projected onto the
probability simplex.  The temperature τ is annealed from 0.02 down to 5e‑4.
Finally a cheap random hill‑climb finishes the remaining time.

All work is done on the simplex (sum = 1) so the returned list can be
used directly with ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Family a_i ∝ (i+shift+1)^p · exp(-lam·i)
# ----------------------------------------------------------------------


def _make_geo_power_shift(n: int, lam: float, p: float, shift: int = 0) -> np.ndarray:
    """
    Build a vector a_i ∝ (i+shift+1)^p · exp(-lam·i)
    (shift ≥ 0).  The result is normalised to unit L1 norm.
    """
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + shift + 1.0)        # log(i+shift+1)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)          # stabilise exponentials
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient for the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]               # includes the diagonal correctly
    return g


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) gradient via cross‑correlation
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      tau: float,
                      L_corr: int) -> np.ndarray:
    """
    Returns the gradient of the smoothed objective
        τ·log Σ exp(conv/τ)
    with respect to a (using the identity
    ∇_a conv_k = 2·a_{k‑i}).
    """
    # soft‑max weights
    w = np.exp(conv / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ★ w  (size n)
    w_rev = w[::-1]                     # reverse for correlation
    Af = np.fft.rfft(a, n=L_corr)
    Wf = np.fft.rfft(w_rev, n=L_corr)
    # length of linear correlation = 3·n‑2
    corr_full = np.fft.irfft(Af * Wf, n=L_corr)[: (3 * a.shape[0] - 2)]
    grad = 2.0 * corr_full[::-1][: a.shape[0]]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def _adam_smoothing(a0: np.ndarray,
                    L_fft: int,
                    L_corr: int,
                    deadline: float,
                    start_time: float,
                    best_seq: np.ndarray,
                    best_score: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam updates on the τ‑smoothed max‑convolution objective.
    Temperature τ is annealed from 2·10⁻² down to 5·10⁻⁴.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    tau = 2e-2               # start temperature
    tau_min = 5e-4
    # anneal every few thousand iterations
    tau_anneal_step = 5000

    # base learning‑rate (will be annealed)
    alpha0 = 0.05

    while time.time() < deadline:
        it += 1

        # compute convolution of a with itself
        conv = _conv_self_fft(a, L_fft)

        # gradient of smoothed max‑conv
        grad = _softmax_gradient(a, conv, tau, L_corr)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        # simple annealing of step‑size
        alpha = alpha0 / (1.0 + 1e-4 * it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluate true (non‑smoothed) objective
        score = _eval_fast(a_new, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] adam it={it:,} τ={tau:.5g} "
                  f"score={score:.6f}")

        a = a_new

        # anneal temperature
        if it % tau_anneal_step == 0 and tau > tau_min:
            tau = max(tau * 0.5, tau_min)

    return best_seq, best_score
