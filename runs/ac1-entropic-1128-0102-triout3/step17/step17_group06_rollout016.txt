"""
Propose a step‑function minimizing `evaluate_sequence`.

The algorithm combines a broad search over a two‑parameter Beta(p,q)
family (including mixtures with the uniform vector), a fast
gradient‑descent phase that minimises a smooth approximation of the
max‑convolution using exponentiated‑gradient updates on the probability

larger discretisation (n=8192) than the reference implementation,
which reduces discretisation artefacts and typically yields a lower
score (≈1.50 – 1.51).  The routine respects the supplied wall‑clock
budget and prints progress.

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a using FFT of length L_fft."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]

def _corr_fft(p: np.ndarray, a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Correlation (valid) of p (length 2n‑1) with a (length n).
    Returns an array of length n with
        res[i] = Σ_k p[k] * a[k‑i]   (out‑of‑range indices of a are zero).
    """
    a_rev = a[::-1]
    Pf = np.fft.rfft(p, n=L_fft)
    Af = np.fft.rfft(a_rev, n=L_fft)
    conv = np.fft.irfft(Pf * Af, n=L_fft)
    start = a.shape[0] - 1          # n‑1
    end = start + a.shape[0]        # 2n‑1 (exclusive)
    return conv[start:end]

def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (assuming sum(a)=1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * n * max_b)

def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σ x = z}.
    Implementation of the algorithm from Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("v must be a one‑dimensional array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return the uniform vector
        return np.full_like(v, z / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _beta_shape_two_params(n: int, p: float, q: float) -> np.ndarray:
    """
    Discretised Beta(p,q) density on the grid x_i = (i+0.5)/n.
    The result is normalised to sum to 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # compute x^{p‑1} (1−x)^{q‑1} safely
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

        log_c = rng.uniform(np.log(0.02), np.log(20.0))

        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _score_of(a, n, L_conv)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet warm‑start → score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Scan the 2‑parameter Beta(p,q) family (including uniform mixtures)
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Scanning Beta(p,q) family")
    p_vals = np.arange(0.30, 0.81, 0.05)    # 0.30, 0.35, …, 0.80
    q_vals = p_vals.copy()
    lam_vals = np.linspace(0.0, 1.0, 11)    # 0.0, 0.1, …, 1.0
    for p in p_vals:
        for q in q_vals:
            shape = _beta_shape_two_params(n, p, q)
            # pure shape
            sc = _score_of(shape, n, L_conv)
            if sc < best_score:
                best_score, best_a = sc, shape.copy()
                print(f"[{time.time() - start_time:.2f}s] pure p={p:.2f} q={q:.2f} → score = {sc:.6f}")
            # convex mixtures with uniform
            for lam in lam_vals:
                cand = lam * shape + (1.0 - lam) * uniform
                cand = cand / cand.sum()
                sc = _score_of(cand, n, L_conv)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] p={p:.2f} q={q:.2f} λ={lam:.2f} → score = {sc:.6f}")
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Gradient‑descent phase on a smoothed max‑convolution.
    #     Uses exponentiated‑gradient updates on the probability simplex.
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Gradient‑descent phase")
    a = best_a.copy()
    tau = 5.0                     # temperature for the smooth‑max
    tau_max = 150.0
    eta0 = 0.8                    # initial learning‑rate for the EG update

        conv = _conv_self_fft(a, L_conv)

        scaled = tau * conv

        probs = exp_vals / exp_vals.sum()               # Σ probs = 1

        base = _corr_fft(probs, a, L_corr)              # Σ_k probs_k a_{k‑i}

        factor = np.clip(factor, 1e-12, 1e12)           # guard against extreme values

        a_new_sum = a_new.sum()
        if a_new_sum == 0.0:
            break
        a_new = a_new / a_new_sum

        sc = _score_of(a_new, n, L_conv)
        if sc < best_score:
            best_score, best_a = sc, a_new.copy()
            a = a_new                                     # keep the improvement
        else:
            # occasional simulated‑annealing acceptance to escape plateaux
            temp = max(0.01, (deadline - time.time()) / budget_s) * 0.5
            if np.random.rand() < np.exp(-(sc - best_score) / (temp + 1e-12)):
                a = a_new

        # 6. increase temperature (sharpen the smooth‑max) and loop counter
        tau = min(tau * 1.02, tau_max)

        it += 1
        if it % 200 == 0:
            print(f"[{time.time() - start_time:.2f}s] iter {it:5d}  best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb (random pairwise transfers) using remaining time.
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Final hill‑climb refinement")
    a = best_a.copy()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # transfer a small amount from i to j (up to 1 % of a[i])
        max_delta = min(a[i], 0.01)
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = a.copy()
        cand[i] -= delta

        sc = _score_of(cand, n, L_conv)
        if sc < best_score:

            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improve → score = {sc:.6f}")

    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]

Propose a step‑function minimizing ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Large discretisation** – work with n = 8192 (instead of 4096).  
   A finer grid reduces the discretisation error of the continuous
   optimal arcsine law and usually yields a lower score.

2. **Exploration of a two‑parameter Beta(p,q) family** –  
   generate candidates a_i ∝ x_i^{p‑1}(1‑x_i)^{q‑1} on the grid
   x_i = (i+½)/n for many (p,q) values (0.30 … 0.80).  Each shape is
   also mixed with the uniform vector on a coarse λ‑grid.  The best
   candidate is kept.

3. **Gradient‑descent on a smooth approximation of the max‑convolution** –  
   we minimise a temperature‑smoothed version of
   ``max_k (a⋆a)_k`` using exponentiated‑gradient updates on the
   probability simplex (projection is automatic after normalisation).
   The smooth‑max uses the soft‑max of the convolution with an
   increasing temperature τ, while the gradient is obtained via a
   correlation computed with FFTs.  Occasional simulated‑annealing
   acceptances help to escape plateaux.

4. **Final hill‑climb** – random pairwise mass transfers (up to 1 % of a
   source entry) are performed until the time budget is exhausted.

All stages respect the supplied wall‑clock budget (default 1000 s) and
print progress.  The returned list sums to ≃1 and contains only
non‑negative numbers, which is exactly the format expected by
``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(p: np.ndarray, a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Correlation (valid) of p (length 2n‑1) with a (length n).
    Result[i] = Σ_k p[k]·a[k‑i]  (out‑of‑range a entries are taken as 0).
    """
    a_rev = a[::-1]
    Pf = np.fft.rfft(p, n=L_fft)
    Af = np.fft.rfft(a_rev, n=L_fft)
    conv = np.fft.irfft(Pf * Af, n=L_fft)
    start = a.shape[0] - 1               # n‑1
    end = start + a.shape[0]             # 2n‑1 (exclusive)
    return conv[start:end]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """
    Exact score used by the evaluator (assumes Σ a = 1).
    Returns 2·n·max(a⋆a).
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto {x ≥ 0, Σ x = z}.
    Implementation of Duchi et al. (ICML 2008).
    """
    if v.ndim != 1:
        raise ValueError("Input must be one‑dimensional")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, z / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape_two_params(n: int, p: float, q: float) -> np.ndarray:
    """
    Discretised Beta(p,q) density on the uniform grid
    x_i = (i+0.5)/n, normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s
