    a = a0.copy()
    best = a0.copy()
    best_score = eval_fast(a, L_fft)
    step0 = 0.4
    it=0
    while time.time() < deadline:
        it+=1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)
        step = step0/(it**0.5)    # may try exponent 0.6

        cand = _project_simplex(cand)

            best = cand.copy()

"""
Propose a non‑negative step function (probability vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm proceeds in four stages:

1️⃣  Parametric sweep over a two‑parameter family 
    a_i ∝ (i+1)^p · exp(-λ·i)  (geometric × power) and over a pure
    Gaussian family a_i ∝ exp(-c·i²).  The search is much finer than before
    and adapts to a larger problem size (n = 16384).

2️⃣  Projected sub‑gradient descent on the exact L∞ objective, using a
    decreasing step‑size and occasional random jitter to escape plateaus.

3️⃣  Multiplicative (exponentiated‑gradient) updates on the simplex, with an
    annealed learning rate to fine‑tune the distribution.

4️⃣  Final random hill‑climb: tiny mass transfers between random indices,
    accepting only improvements.  The whole run respects the 1000 s budget.

The final vector is normalised to sum = 1 (a probability distribution) and
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – works for arbitrary sum
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Return normalized a_i ∝ (i+1)^p · exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    # log(i+1) can be reused
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)           # stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """Return normalized a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the index where the maximum is attained.
    For each i involved in the max sum (i+j=k) we get 2·a_j.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    # add contributions
    np.add.at(g, i, 2.0 * a[j])
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (phase 2)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent on the exact L∞ objective.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    step0 = 0.45
    it = 0
    # counters for stagnation detection
    last_improve = start_time

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        # decreasing step size (tuned for n=16384)
        step = step0 / (it ** 0.55)

        # candidate by moving opposite to sub‑gradient
        cand = a - step * g
        cand = _project_simplex(cand)
        cand_score = _eval_fast(cand, L_fft)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand                 # continue from improved point
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.3e} score={cand_score:.6f}")
        else:
            # stay on current point but occasionally jitter
            if it % 1200 == 0:
                eps = rng.random(a.shape) * a * 8e-5
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)

        # if we have been stuck for a while, restart from the current best plus jitter
        if time.time() - last_improve > 30.0 and it % 2000 == 0:
            eps = rng.random(best.shape) * best * 2e-4
            a = np.maximum(best + eps, 0.0)
            a = _project_simplex(a)
            last_improve = time.time()

    return best, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (phase 3)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multiplicative updates on the simplex:
        a ← a * exp(-η·g) ;  η slowly annealed.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    it = 0
    eta0 = 0.8

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        # annealed learning rate
        eta = eta0 / (1.0 + 0.001 * it)

        mult = np.exp(-eta * g)
        a = a * mult
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / total

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] expgrad it={it:,} "
                  f"score={score:.6f}")

        # occasional tiny jitter to avoid getting stuck in exact plateaus
        if it % 2500 == 0:
            eps = rng.random(a.shape) * a * 5e-5
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best, best_score


# ----------------------------------------------------------------------
# Final random hill‑climb (phase 4)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                L_fft: int,
                deadline: float,
                start_time: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves with a simple acceptance rule.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        # select a source index with non‑negligible mass
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to 0.2 % of the source mass
        eps = a[src] * rng.random() * 2e-3
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        # re‑normalise (numerical drift)
        cand = cand / cand.sum()
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={cand_score:.6f}")
        # occasional larger exploratory jump
        if rng.random() < 0.001:
            # pick a random index and sprinkle a tiny amount of mass
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[i] = max(cand[i] + jitter, 0.0)
            cand = cand / cand.sum()
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={cand_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12  # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 16384 balances quality and speed.
    # --------------------------------------------------------------
    n = 16384
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Try a globally supplied starter if present (height_sequence_1).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Fine parametric sweep over (λ, p) and Gaussian shapes – ≈80 s.
    # --------------------------------------------------------------
    sweep_end = start_time + 80.0
    i_array = np.arange(n, dtype=np.float64)
    log_idx = np.log(i_array + 1.0)

    # geometric × power family
    lam_grid = np.linspace(1e-05, 2.5e-03, 350)        # denser grid
    p_grid = np.linspace(-8.0, 5.0, 131)               # step ≈0.1

    for lam in lam_grid:
        if time.time() >= sweep_end:
            break
        base = -lam * i_array
        for p in p_grid:
            log_shape = base + p * log_idx
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6e} p={p:.2f} score={sc:.6f}")

    # pure Gaussian family (exp(-c·i²))
    c_grid = np.logspace(-8.0, -5.0, 48)               # 48 points
    for c in c_grid:
        if time.time() >= sweep_end:
            break
        log_shape = -c * (i_array * i_array)
        log_shape -= np.max(log_shape)
        cand = np.exp(log_shape)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] gaussian c={c:.2e} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Projected sub‑gradient descent – allocate ~55 % of the budget.
    # --------------------------------------------------------------
    sub_end = start_time + 0.55 * budget_s
    if time.time() < sub_end:
        best_seq, best_score = _subgradient_descent(best_seq, L_fft,
                                                    sub_end, start_time,
                                                    rng)

    # --------------------------------------------------------------
    # 3️⃣ Exponentiated‑gradient phase – allocate ~25 % of the budget.
    # --------------------------------------------------------------
    exp_end = start_time + 0.80 * budget_s
    if time.time() < exp_end:
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              exp_end, start_time,
                                              rng)

    # --------------------------------------------------------------
    # 4️⃣ Final hill‑climb – use whatever time is left.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, L_fft,
                                           deadline, start_time,
                                           rng)

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Report final bound using the provided evaluate_sequence (optional)
    try:
        from __main__ import evaluate_sequence  # may already be in globals
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
