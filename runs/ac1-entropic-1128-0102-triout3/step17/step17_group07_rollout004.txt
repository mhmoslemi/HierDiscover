"""
Improved heuristic for minimizing

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is based on three ideas that complement the previous
implementation:

1️⃣ **Smooth‑max projected gradient descent** – we minimise a smooth
   approximation of the peak of the self‑convolution:
   L(a) = log‑sum‑exp(β·conv(a,a))/β.
   Its gradient is obtained by a single FFT‑based correlation
   (O(n log n)).  After each gradient step we project back onto the
   probability simplex (non‑negative, unit sum).

2️⃣ **Variance‑flattening descent** – the same variance‑gradient step as
   before is applied for a while, helping to flatten the whole
   convolution once the peak has been reduced.

3️⃣ **Targeted hill‑climb** – finally we locally move a tiny amount of
   mass from the largest contributor of the current peak to a random
   coordinate, accepting only improving moves (as in the reference).

All stages share the same time budget (≈1000 s).  The routine starts
from the best‑known construction (`height_sequence_1`) if it is
available, otherwise from the uniform vector.  The returned Python list
contains non‑negative floats (approximately summing to 1) and respects
the evaluator’s limits.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries zero – return uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (assumes sum(a)=1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
def _initial_candidate(n: int, L_fft: int) -> tuple[np.ndarray, float]:
    """Return a starting vector (global starter if present, else uniform)."""
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a, n, L_fft)

    # try to use the supplied global starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            # pad / trim to length n
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size),
                                     constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improves initial score to {sc:.6f}")
    except Exception:
        pass
    return best_a, best_score


# ----------------------------------------------------------------------
# 1️⃣  Smooth‑max projected gradient descent
# ----------------------------------------------------------------------
def _smooth_max_grad_descent(a0: np.ndarray,
                             n: int,
                             L_fft: int,
                             L_corr: int,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv) using projected gradient descent.
    The smoothness parameter β is increased gradually (annealing).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    it = 0
    eta0 = 0.12            # base stepsize
    beta = 2.0             # start temperature for the soft‑max
    beta_inc = 0.02        # annealing rate

    while time.time() < deadline:
        it += 1
        # full self‑convolution
        conv = _conv_self_fft(a, L_fft)

        # current (exact) score for bookkeeping
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothMaxIter={it:,} "
                  f"β={beta:.2f} score={cur_score:.6f}")

        # ----- smooth‑max (log‑sum‑exp) -----
        # we use a numerically stable form
        conv_max = conv.max()
        scaled = beta * (conv - conv_max)          # ≤ 0
        exp_scaled = np.exp(scaled)
        Z = exp_scaled.sum()
        # soft‑max weights (they sum to 1)
        p = exp_scaled / Z

        # ----- gradient of the smooth‑max -----
        # g_i = Σ_k p_k * a_{k-i}   (correlation a ⋆ p)
        # correlation via FFT:  corr = ifft( conj(fft(a)) * fft(p) )
        Af = np.fft.rfft(a, n=L_corr)
        Pf = np.fft.rfft(p, n=L_corr)
        corr_full = np.fft.irfft(Af.conj() * Pf, n=L_corr)
        # corr_full[0:n] contains the needed values g_i for i=0..n-1
        g = corr_full[:n]

        # ----- projected gradient step -----
        eta = eta0 / np.sqrt(it)          # classic diminishing step
        a_new = a - eta * g
        a = _project_simplex(a_new)

        # anneal β slowly – larger β sharpens the approximation of the max
        beta = min(beta + beta_inc, 30.0)

        # occasional tiny random jitter to avoid plateaus
        if np.random.rand() < 0.004:
            eps = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Variance‑flattening gradient descent (as in the reference)
# ----------------------------------------------------------------------
def _variance_gradient(a: np.ndarray,
                       conv: np.ndarray,
                       mean: float,
                       L_corr: int) -> np.ndarray:
    """
    Gradient of Σ_k (conv_k – mean)² w.r.t. a.
    Formula:   ∇ = 4·corr(conv–mean , a)
    """
    diff = conv - mean
    diff_rev = diff[::-1]
    Dr = np.fft.rfft(diff_rev, n=L_corr)
    Ar = np.fft.rfft(a, n=L_corr)
    full = np.fft.irfft(Dr * Ar, n=L_corr)
    start = a.shape[0] - 1
    grad = 4.0 * full[start:start + a.shape[0]]
    return grad


def _variance_flattening_descent(a0: np.ndarray,
                                 n: int,
                                 L_fft: int,
                                 L_corr: int,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent minimising the variance of the self‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    mean = 1.0 / (2 * n - 1)      # total mass of the convolution (sum = 1)
    it = 0
    eta0 = 0.8

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] VarIter={it:,} score={cur_score:.6f}")

        grad = _variance_gradient(a, conv, mean, L_corr)
        eta = eta0 / np.sqrt(it + 1)
        a_new = a - eta * grad
        a = _project_simplex(a_new)

        if np.random.rand() < 0.004:
            eps = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Targeted hill‑climb on the maximum convolution entry
# ----------------------------------------------------------------------
def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    For a given convolution index k, return the pair (i, j) with the
    largest product a_i·a_j (i+j=k) together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    max_pos = np.argmax(contrib)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(contrib[max_pos])


def _targeted_hill_climb(a: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Aggressive local search that reduces the peak of the convolution.
    """
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        i, j, contrib = _max_contributions(best_a, k_max)
        if i == -1:
            continue

        # move a tiny amount of mass from the larger partner to a random index
        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-12:
            continue
        dst = rng.integers(0, n)
        if dst == src:
            continue
        delta = rng.uniform(0.0, min(best_a[src], 0.004))
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        new_score = _objective_score(cand, n, L_fft)
        if new_score < best_score:
            best_score, best_a = new_score, cand
            print(f"[{time.time() - start_time:.2f}s] Hill i={src}→{dst} δ={delta:.5f} → score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final mixing + light random‑pair hill‑climb (as in the reference)
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill‑climb i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s*
    seconds and returns a plain Python list of floats (approximately
    unit sum).  The internal optimisation consists of:
      • warm‑start (global starter + uniform)
      • smooth‑max projected gradient descent
      • variance‑flattening descent
      • targeted hill‑climb
      • final mixing & light hill‑climb
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # a safety margin to avoid overshooting the timeout
    deadline = start_time + budget_s - 0.15

    # problem size – 4096 is a good trade‑off; you may increase it if
    # you have extra time, but keep within the 1 CPU budget.
    n = 4096
    L_fft = _next_pow2(2 * n - 1)       # for a*a
    L_corr = _next_pow2(3 * n - 2)      # for correlations used later

    # ------------------------------------------------------------------
    # 0️⃣  Initial candidate (global starter if present)
    # ------------------------------------------------------------------
    best_a, best_score = _initial_candidate(n, L_fft)
    print(f"[0.00s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Smooth‑max projected gradient descent (≈55 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0:
        sm_deadline = time.time() + 0.55 * remaining
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_a, best_score = _smooth_max_grad_descent(best_a, n,
                                                      L_fft, L_corr,
                                                      sm_deadline)
        print(f"[{time.time() - start_time:.2f}s] After smooth‑max best ≈ {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Variance‑flattening gradient descent (~30 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0:
        var_deadline = time.time() + 0.30 * remaining
        print(f"[{time.time() - start_time:.2f}s] Starting variance flattening")
        best_a, best_score = _variance_flattening_descent(best_a, n,
                                                          L_fft, L_corr,
                                                          var_deadline)
        print(f"[{time.time() - start_time:.2f}s] After variance best ≈ {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Targeted hill‑climb (~15 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0:
        hill_deadline = time.time() + 0.15 * remaining
        print(f"[{time.time() - start_time:.2f}s] Starting targeted hill‑climb")
        best_a, best_score = _targeted_hill_climb(best_a, n, L_fft, hill_deadline)
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best ≈ {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final refinement (mixing + light random‑pair hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement")
        best_a, best_score = _final_refinement(best_a, n,
                                               L_fft, deadline)

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
