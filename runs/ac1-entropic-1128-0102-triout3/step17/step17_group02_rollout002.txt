def _directed_hill_climb(a0, score0, deadline):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    # set temperature schedule: start high then decay
    T0 = (best_score - 1.0) * 0.1 + 0.001  # maybe small
    T = T0
    cooling = 0.995  # per iteration
    # factor for delta move
    delta_factor = 0.05
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute conv
        conv = _conv_self(a)
        max_val = conv.max()
        # find max indices
        # tolerance for floating point equality
        tol = 1e-12
        max_idx = np.where(conv >= max_val - tol)[0]
        # weight array
        w = np.zeros(2*n-1, dtype=float)
        # disjoint set of max indices, assign equal weights
        w[max_idx] = 1.0 / len(max_idx)
        # gradient of max (weighted) w.r.t a
        rev_corr = _convolve_fft(a[::-1], w)
        grad = 2.0 * rev_corr[n-1 : n-1 + n]
        # find src with largest gradient (mass that contributes most to max)
        src = np.argmax(grad)
        dst = np.argmin(grad)
        # only move if src != dst and a[src] > 0
        if src == dst or a[src] <= 0.0:
            # fallback to random move
            src = np.random.randint(n)
            dst = np.random.randint(n)
            while src == dst or a[src] == 0.0:
                src = np.random.randint(n)
                dst = np.random.randint(n)
        delta = a[src] * delta_factor
        if delta <= 0.0:
            # very small; break
            break
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # ensure nonnegative due to rounding
        if cand[src] < 0:
            # adjust delta to avoid negativity
            delta = a[src]
            cand[src] = 0.0
            cand[dst] = a[dst] + delta
        cand_score = _objective_score(cand)
        delta_score = cand_score - cur_score
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # uphill with probability exp(-Δ/T)
            if T > 0:
                prob = math.exp(- (cand_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - global_start_time:.2f}s] directed hill new best {best_score:.6f}")
        # cool temperature and maybe adjust delta factor over time
        T = T * cooling
        delta_factor = max(0.001, delta_factor * 0.999)  # slowly reduce
    return best_a, best_score

"""
Propose a non‑negative step function (probability vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

because ``evaluate_sequence`` forces the sum to 1.  The algorithm runs three
phases within a 1000 s budget:

1️⃣  A quick sweep of random Dirichlet draws (plus a few deterministic windows)
    to obtain a decent starter.

2️⃣  Smooth projected‑gradient descent on a log‑sum‑exp approximation of the
    max‑convolution.  The temperature β is increased in stages; after each
    stage we project back onto the probability simplex.

3️⃣  A directed hill‑climbing phase that uses the exact sub‑gradient of the
    (true) maximum to move mass from the “most responsible’’ indices to the
    least responsible ones; an annealed Metropolis rule allows occasional
    uphill moves.

All heavy convolutions use an FFT of power‑of‑2 length, the simplex projection
uses the O(n) algorithm of Duchi et al., and the code respects the global time
budget.  The best vector (as a plain list of floats) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global timing variable (set inside ``propose_candidate``)
# ----------------------------------------------------------------------
global_start_time = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using an FFT of power‑of‑2 length."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # should not happen, but fallback to uniform
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1 → denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max approximation of max(conv) w.r.t. a.
    Uses softmax weights w_i ∝ exp(β·conv_i).
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()                 # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)  # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float, rng: np.random.Generator):
    """Draw several Dirichlet vectors (different concentrations) and keep the best."""
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration c drawn log‑uniformly on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)   # 10**0.7 ≈ 5
        c = 10.0 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - global_start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth projected‑gradient descent (β‑schedule)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    beta_vals,
                    deadline: float):
    """
    Projected gradient descent on the smooth log‑sum‑exp surrogate of max‑conv.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score

    for beta in beta_vals:
        step = 0.5
        # simple iteration counter to avoid endless loops when time permits
        iter_cnt = 0
        while time.time() < deadline:
            iter_cnt += 1
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            improved = False
            # geometric backtracking line search
            for factor in (1.0, 0.5, 0.25, 0.125):
                cand = a - step * factor * grad
                cand = _project_simplex(np.maximum(cand, 0.0))
                cand_score = _objective_score(cand)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    improved = True
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()
                        print(f"[{time.time() - global_start_time:.2f}s] β={beta:.1f} new best {best_score:.6f}")
                    break

            if not improved:
                step = max(step * 0.5, 1e-14)
            else:
                step = min(step * 1.05, 1.0)

            if step < 1e-14:
                break
            # safety break – we do not want to waste too many tiny steps
            if iter_cnt > 5000:
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Directed hill‑climbing using the true sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _directed_hill_climb(a0: np.ndarray,
                         score0: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Move mass from indices that most increase the current maximum to those that
    least affect it.  An annealed Metropolis rule permits occasional uphill
    moves to escape shallow local minima.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    # initial temperature (a small fraction of the current excess over the
    # theoretical lower bound 1.0) – tuned empirically.
    T = max((cur_score - 1.0) * 0.05, 1e-4)
    cooling = 0.9995          # geometric cooling per iteration
    delta_factor = 0.07       # start moving up to 7 % of source mass
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # indices where the maximum is attained (within numerical tolerance)
        tol = 1e-12
        max_idx = np.where(conv >= max_val - tol)[0]

        # construct weight vector w for the (weighted) sub‑gradient
        w = np.zeros(2 * n - 1, dtype=float)
        w[max_idx] = 1.0 / len(max_idx)

        # sub‑gradient of the (weighted) max wrt a
        rev_corr = _convolve_fft(a[::-1], w)
        grad = 2.0 * rev_corr[n - 1 : n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # if source is empty or source==dst, fall back to a random feasible move
        if src == dst or a[src] <= 0.0:
            src = rng.integers(n)
            dst = rng.integers(n)
            while src == dst or a[src] <= 0.0:
                src = rng.integers(n)
                dst = rng.integers(n)

        delta = a[src] * delta_factor
        if delta <= 0.0:
            break

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # safeguard against tiny negative due to round‑off
        if cand[src] < 0.0:
            cand[dst] += cand[src]   # subtract the tiny overshoot
            cand[src] = 0.0

        cand_score = _objective_score(cand)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(- (cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - global_start_time:.2f}s] directed hill new best {best_score:.6f}")

        # anneal temperature and slowly shrink the move size
        T *= cooling
        delta_factor = max(0.001, delta_factor * 0.999)

        # occasional safeguard: after many iterations without improvement,
        # reset delta_factor a little to keep the search alive
        if iter_cnt % 2000 == 0 and delta_factor < 0.02:
            delta_factor = 0.02

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function (probability vector) that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    global global_start_time
    global_start_time = time.time()
    deadline = global_start_time + budget_s - 0.1   # safety margin

    np.random.seed(seed)
    rng_main = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose problem size – a modest increase over the reference code
    # --------------------------------------------------------------
    n = 8192                     # 8 k entries balances quality and speed

    # --------------------------------------------------------------
    # Uniform start (objective = 2.0)
    # --------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - global_start_time:.2f}s] uniform start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (if the host defined ``height_sequence_1``)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq = starter.copy()
                    best_score = sc
                    print(f"[{time.time() - global_start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – random Dirichlet sweeps + a few deterministic windows
    # --------------------------------------------------------------
    dir_deadline = min(deadline, global_start_time + 30.0)
    if time.time() < dir_deadline:
        seq, sc = _random_dirichlet_trials(
            n, max_trials=1500, deadline=dir_deadline, rng=rng_main
        )
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - global_start_time:.2f}s] Dirichlet best = {best_score:.6f}")

    # a few classic windows
    shapes = []
    # sine window
    shapes.append(np.sin(np.pi * (np.arange(n) + 1) / (n + 1)))
    # Hann (raised‑cosine) window
    shapes.append(0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    # symmetric triangular window
    triangular = np.minimum(np.arange(n), n - 1 - np.arange(n))
    shapes.append(triangular.astype(float))
    # exponential decay from the left
    shapes.append(np.exp(-np.arange(n) / (n / 10.0)))
    # cosine‑power windows
    for p in (1.0, 2.0, 3.0):
        shapes.append(np.maximum(np.cos(np.pi * np.arange(n) / (2.0 * n)), 0.0) ** p)

    for shp in shapes:
        shp = np.maximum(shp, 0.0)
        shp = _project_simplex(shp)
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq, best_score = shp.copy(), sc
            print(f"[{time.time() - global_start_time:.2f}s] deterministic shape score {sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – smooth projected‑gradient descent (β‑schedule)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.85   # most time for descent
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
            50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0
        ]
        restarts = 8
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break
            # tiny jitter around the current best to explore a new basin
            jitter = rng_main.normal(scale=0.012, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a_opt, sc_opt = _smooth_descent(a0, beta_schedule, descent_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - global_start_time:.2f}s] restart {r} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – directed hill‑climbing (final fine‑tuning)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - global_start_time:.2f}s] starting directed hill climbing")
        best_seq, best_score = _directed_hill_climb(
            best_seq, best_score, deadline, rng_main
        )

    # --------------------------------------------------------------
    # Normalise to a proper probability vector and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - global_start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
