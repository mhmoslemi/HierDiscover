def _smoothmax_adam(a0, deadline, best_seq, best_score):
    n = a0.shape[0]
    logits = np.log(a0 + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t_iter = 0
    beta_start = 0.5
    beta_end = 200.0
    lr0 = 0.05
    while time.time() < deadline:
        t_iter += 1
        # time fraction for schedule
        fraction = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start * (beta_end / beta_start) ** fraction  # exponential growth
        # compute smooth max and grad wrt a
        smooth, G_a = _smooth_max_and_grad(_softmax(logits), beta)

        v = 0.999 * v + 0.001 * (grad_logits ** 2)
        m_hat = m / (1 - 0.9 ** t_iter)
        v_hat = v / (1 - 0.999 ** t_iter)

"""
Propose a step‑function (non‑negative list with sum = 1) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution in numpy.  The algorithm consists of
four phases and uses only NumPy + the supplied evaluate_sequence().

1️⃣  A quick sweep of random Dirichlet draws (different concentration
    parameters) to obtain a few promising starting points.

2️⃣  Mirror‑descent on a **smooth‑max** surrogate of the max‑convolution.
    The surrogate is the log‑sum‑exp of the convolution; its gradient is
    obtained analytically and an Adam‑style update on the *log‑probabilities*
    (the “logits”) keeps the iterate on the probability simplex
    (non‑negative, sum = 1).

3️⃣  A second Adam phase that minimises the L²‑distance of the convolution
    to the uniform distribution (the theoretical lower bound of the max
    is 1/(2n‑1)).  Flattening the whole convolution pushes the peak down.

4️⃣  A short random‑jitter hill‑climb that tries tiny mass transfers
    between coordinates and keeps any improvement.

The routine works with n = 8192 (≈ 8 k entries) – large enough to give a
theoretical bound of ≈ 1.0 but still cheap to convolve by FFT.  All phases
stop as soon as the global time budget (default 1000 s) is exhausted.
The best sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (full length) via FFT."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Exact objective, assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    max_log = np.max(logits)
    exps = np.exp(logits - max_log)
    return exps / np.sum(exps)


def _grad_logits_from_grad_a(a: np.ndarray, grad_a: np.ndarray) -> np.ndarray:
    """
    Convert a gradient w.r.t. a (which lives on the simplex) to a gradient
    w.r.t. the logits of a soft‑max parametrisation.
    """
    avg = np.dot(a, grad_a)
    return a * (grad_a - avg)


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient) where gradient is ∂/∂a of the smooth max.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    scaled = beta * conv
    max_scaled = np.max(scaled)
    # stability shift
    exp_scaled = np.exp(scaled - max_scaled)
    Z = exp_scaled.sum()
    w = exp_scaled / Z                         # soft‑max weights, sum to 1

    # gradient: d/d a_i ( Σ_k w_k conv_k ) = 2 * Σ_j a_j w_{i+j}
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)          # length 3n‑2
    start = len(w) - 1                      # = 2n‑2
    grad = 2.0 * conv_rev[start - np.arange(n)]

    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, grad


def _flatten_loss_and_grad(a: np.ndarray):
    """
    L2 loss between a*a and the uniform distribution on {0,…,2n‑2}.
    Returns (loss, gradient) where gradient is ∂L/∂a.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    target = 1.0 / (2 * n - 1)
    w = conv - target                         # residual
    loss = np.sum(w ** 2)

    # ∂L/∂a_i = 4 * Σ_j a_j * w_{i+j}
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)
    start = len(w) - 1
    grad = 4.0 * conv_rev[start - np.arange(n)]
    return loss, grad


def _adam_step(logits: np.ndarray,
               grad_logits: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float = 0.05,
               b1: float = 0.9,
               b2: float = 0.999,
               eps: float = 1e-8):
    """One Adam update on the logits."""
    m = b1 * m + (1.0 - b1) * grad_logits
    v = b2 * v + (1.0 - b2) * (grad_logits ** 2)

    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return logits - step, m, v

# Phase 1 – random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, top_k, deadline, trials=8000):
    """
    Return a list ``candidates`` of up to ``top_k`` best (lowest‑score)
    Dirichlet samples seen so far.
    """
    rng = np.random.default_rng()
    candidates = []        # list of (score, vector) tuples, kept sorted
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in roughly [0.01, 5]
        logc = rng.uniform(-2.0, 0.7)      # 10^{-2} .. 10^{0.7}
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if len(candidates) < top_k:
            candidates.append((sc, a.copy()))
            candidates.sort(key=lambda x: x[0])      # keep smallest first
        else:
            # replace worst only if strictly better
            if sc < candidates[-1][0]:
                candidates[-1] = (sc, a.copy())
                candidates.sort(key=lambda x: x[0])
    return candidates

# ----------------------------------------------------------------------
def _smoothmax_adam(a0,
                    deadline,
                    best_seq,
                    best_score,
                    beta_start=0.5,
                    beta_end=250.0,
                    lr0=0.07):
    """
    Adam optimiser on the smooth‑max surrogate.
    Keeps the best exact score found and returns the best sequence.
    """
    n = a0.shape[0]
    logits = np.log(np.maximum(a0, 1e-20))
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t_iter = 0

    # we will linearly increase β on a log‑scale (exponential schedule)
    while time.time() < deadline:
        t_iter += 1
        # fraction of the phase that has elapsed
        phase_frac = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start * (beta_end / beta_start) ** phase_frac

        a = _softmax(logits)
        _, grad_a = _smooth_max_and_grad(a, beta)

        # convert to logits gradient (mirror descent on simplex)
        grad_logit = _grad_logits_from_grad_a(a, grad_a)

        # learning‑rate decay (simple 1/√t)
        lr = lr0 / np.sqrt(t_iter)

        logits, m, v = _adam_step(logits, grad_logit, m, v,
                                  t_iter, lr=lr)

        a = _softmax(logits)
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Smooth‑max β={beta:.2f} score={cur_score:.6f}")

        # optional early stop
        if best_score <= 1.45:
            break
    return best_seq, best_score

def _flatten_adam(a0,

    logits = np.log(np.maximum(a0, 1e-20))
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

        grad_logit = _grad_logits_from_grad_a(a, grad_a)

            print(f"[{time.time() - start_time:.2f}s] "

def _jitter_hillclimb(a0,
                      deadline,
                      best_seq,
                      best_score,
                      max_trials=1200):
    """
    Randomly move a tiny amount of probability mass between two
    coordinates.  Keep any improvement.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline and max_trials > 0:
        max_trials -= 1
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        # transfer a random fraction of a[i] (≤1 % of its mass)
        delta = a[i] * rng.uniform(0.0, 0.01)
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur_score = _objective_score(a_new)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a_new.copy()
            a = a_new          # continue from the improved point
            # optional: tighten the step size a little
    return best_seq, best_score

            best_seq=best_seq,
            best_score=best_score,
        )
        print(f"[{time.time() - start_time:.2f}s] After jitter best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a clean Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
