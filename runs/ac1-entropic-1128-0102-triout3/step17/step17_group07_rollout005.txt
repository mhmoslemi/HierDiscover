def _smooth_max_objective_and_gradient(a, L_fft, L_corr, alpha):
    conv = _conv_self_fft(a, L_fft)
    # compute smooth max gradient weight
    max_c = conv.max()
    # subtract max to improve numerics
    exp_vals = np.exp(alpha * (conv - max_c))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp  # shape (2n-1,)
    # gradient = 2 * correlation of w with a

    # correlation via FFT
    Ar = np.fft.rfft(a, n=L_corr)
    Wr = np.fft.rfft(w_rev, n=L_corr)
    full = np.fft.irfft(Ar * Wr, n=L_corr)
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    # smooth max value: max_c + log_sum_exp / alpha
    smooth_val = max_c + np.log(sum_exp) / alpha
    return smooth_val, grad

def _adam_smooth_max_descent(a0, n, L_fft, L_corr, deadline):
    # start
    rng = np.random.default_rng()
    x = np.log(a0 + 1e-12)  # shape (n,)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    iteration = 0
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = np.exp(x - np.max(x))
    best_a /= best_a.sum()
    best_score = 2.0 * n * _conv_self_fft(best_a, L_fft).max()
    # compute alpha schedule parameters
    alpha_start = 5.0
    alpha_end = 150.0

    start_time = time.time()
    while time.time() < deadline:
        iteration += 1
        t = (time.time() - start_time) / (deadline - start_time)
        alpha = alpha_start + (alpha_end - alpha_start) * t
        # compute a from x
        x_max = np.max(x)
        a = np.exp(x - x_max)
        a_sum = a.sum()
        a /= a_sum

        # compute conv
        conv = _conv_self_fft(a, L_fft)
        conv_max = conv.max()
        # compute smooth weights
        shift = conv - conv_max
        # to avoid overflow, we multiply by alpha and exponentiate
        exp_vals = np.exp(alpha * shift)
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp  # length 2n-1

        # gradient wrt a
        w_rev = w[::-1]
        # correlation
        Ar = np.fft.rfft(a, n=L_corr)
        Wr = np.fft.rfft(w_rev, n=L_corr)
        full = np.fft.irfft(Ar * Wr, n=L_corr)
        start = n - 1
        grad_a = 2.0 * full[start:start + n]

        # gradient wrt x
        inner = np.dot(grad_a, a)
        grad_x = a * (grad_a - inner)

        # Adam
        beta1_pow *= beta1
        beta2_pow *= beta2

        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)

        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # compute actual score
        cur_score = 2.0 * n * conv_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamIter={iteration} α={alpha:.1f} score={cur_score:.6f}")

        # occasional random perturbation to escape plateau:
        if rng.random() < 0.002:
            # add small Gaussian noise to x
            x += rng.normal(scale=0.01, size=n)

    return best_a, best_score

def _targeted_hill_climb_improved(a, n, L_fft, deadline):
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())
        i, j, contrib = _max_contributions(best_a, k_max)
        if i < 0:
            continue
        # choose delta as fraction of smaller of a[i], a[j] and a small bound
        bound = min(best_a[i], best_a[j], 0.005)
        if bound <= 0:
            # fallback: move from larger only
            src = i if best_a[i] >= best_a[j] else j
            if best_a[src] < 1e-12:
                continue
            dst = rng.integers(0, n)
            if dst == src:
                continue
            delta = rng.uniform(0.0, min(best_a[src], 0.001))
            cand = best_a.copy()
            cand[src] -= delta
            cand[dst] += delta
        else:
            # move from both i and j
            # choose delta between 0 and bound
            delta = rng.uniform(0.0, bound)
            # choose destination distinct
            dst = rng.integers(0, n)
            while dst == i or dst == j:
                dst = rng.integers(0, n)
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] -= delta
            cand[dst] += 2 * delta
        # ensure nonneg and renormalize
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        new_score = _objective_score(cand, n, L_fft)

            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] HillEdge i={i} j={j} → dst={dst} δ={delta:.6f} score={new_score:.6f}")
    return best_a, best_score

"""
propose_candidate – heuristic optimizer for the “max self‑convolution” score.

The algorithm works in four stages (all respecting a 1000 s wall‑clock budget).

1️⃣ Warm‑start: a few Dirichlet draws (randomised step‑functions) are
   evaluated, the best one is kept.  The global starter (if provided via the
   global variable ``height_sequence_1``) is also considered.

2️⃣ Smoothed‑max Adam descent: the objective  max(conv(a,a)) is approximated
   by a smooth “log‑sum‑exp” version.
   a is represented by a soft‑max over free parameters x, guaranteeing
   non‑negativity and unit sum.
   The gradient of the smooth max is obtained analytically via FFT based
   correlation, and Adam is used for fast convergence.
   The smoothing parameter α is ramped from 5 to 150 during the phase,
   focusing the search on the true maximum as optimisation proceeds.

3️⃣ Targeted hill‑climb: while time permits the current maximal convolution
   entry is attacked.  The pair (i,j) that contributes most to this peak is
   identified and a tiny amount of mass is moved from the two indices to a
   random third index (or, if one of them is zero, from the larger index
   alone).  Only improving moves are accepted.

4️⃣ Final refinement: convex mixing with the uniform distribution and a
   cheap random‑pair hill‑climb, exactly as in the reference code.

All helper functions are defined at module level.  The routine prints progress
and finally returns a plain Python list of non‑negative floats (approximately
unit‑sum) that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (assumes sum(a)=1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


def _max_contributions(a: np.ndarray, k: int):
    """
    For a given convolution index k, return the pair (i, j) with the
    largest product a_i·a_j (where i+j=k) together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    max_pos = np.argmax(contrib)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(contrib[max_pos])


# ----------------------------------------------------------------------
# 0️⃣ Warm‑start (Dirichlet draws, uniform, optional global starter)
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                L_fft: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """Sample many Dirichlet‑like vectors and keep the best one."""
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration drawn log‑uniformly in a wide range
        c = np.exp(rng.uniform(np.log(0.05), np.log(20.0)))
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Warm‑draw c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 1️⃣ Smoothed‑max gradient descent (Adam) on the simplex
# ----------------------------------------------------------------------
def _adam_smooth_max_descent(a0: np.ndarray,
                             n: int,
                             L_fft: int,
                             L_corr: int,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smoothed version of max(conv(a,a)).
    a is represented by a soft‑max of free parameters x.
    """
    rng = np.random.default_rng()
    # initialise x from a0 (soft‑max parametrisation)
    x = np.log(a0 + 1e-12)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2                     # base learning rate
    it = 0
    beta1_pow = 1.0
    beta2_pow = 1.0

    # smoothing schedule
    alpha_start = 5.0
    alpha_end = 150.0

    # best‑so‑far bookkeeping
    a = np.exp(x - np.max(x))
    a = a / a.sum()
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    start = time.time()
    while time.time() < deadline:
        it += 1
        t = (time.time() - start) / (deadline - start)   # in [0,1]
        alpha = alpha_start + (alpha_end - alpha_start) * t

        # current a from x (stable soft‑max)
        x_max = np.max(x)
        a = np.exp(x - x_max)
        a_sum = a.sum()
        a = a / a_sum

        # full convolution and its peak
        conv = _conv_self_fft(a, L_fft)
        conv_max = conv.max()

        # smooth‑max weights (log‑sum‑exp trick)
        shift = conv - conv_max                     # max entry becomes 0
        exp_vals = np.exp(alpha * shift)
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp                      # length 2n‑1, sums to 1

        # gradient of the smooth max w.r.t. a
        w_rev = w[::-1]
        Ar = np.fft.rfft(a, n=L_corr)
        Wr = np.fft.rfft(w_rev, n=L_corr)
        full = np.fft.irfft(Ar * Wr, n=L_corr)
        start_idx = n - 1
        grad_a = 2.0 * full[start_idx:start_idx + n]

        # chain rule: gradient w.r.t. the soft‑max parameters x
        inner = np.dot(grad_a, a)
        grad_x = a * (grad_a - inner)

        # Adam update on x
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny jitter to escape flat plateaux
        if rng.random() < 0.002:
            x += rng.normal(scale=0.01, size=n)

        # evaluate the *true* objective (not the smooth surrogate)
        cur_score = 2.0 * n * conv_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamIter={it:,} α={alpha:.1f} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Targeted hill‑climb (improved version)
# ----------------------------------------------------------------------
def _targeted_hill_climb_improved(a: np.ndarray,
                                  n: int,
                                  L_fft: int,
                                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Aggressively reduces the current maximum convolution entry.
    For the peak index k, the contributing pair (i,j) with largest a_i·a_j
    is identified and a tiny amount of mass is taken from both i and j
    (or from the larger one if the other is too small) and transferred to
    a random third coordinate.  Only improving moves are accepted.
    """
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        i, j, contrib = _max_contributions(best_a, k_max)
        if i == -1:
            continue

        # decide how much mass to move
        bound = min(best_a[i], best_a[j], 0.005)
        if bound > 1e-12:
            delta = rng.uniform(0.0, bound)
            dst = rng.integers(0, n)
            while dst == i or dst == j:
                dst = rng.integers(0, n)
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] -= delta
            cand[dst] += 2.0 * delta
        else:
            # fallback – move from the larger of the two only
            src = i if best_a[i] >= best_a[j] else j
            if best_a[src] < 1e-12:
                continue
            dst = rng.integers(0, n)
            if dst == src:
                continue
            delta = rng.uniform(0.0, min(best_a[src], 0.004))
            if delta <= 0.0:
                continue
            cand = best_a.copy()
            cand[src] -= delta
            cand[dst] += delta

        # re‑normalise (numerical drift)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        new_score = _objective_score(cand, n, L_fft)

        if new_score < best_score:
            best_score, best_a = new_score, cand
            print(f"[{time.time() - start_time:.2f}s] Hill i={i} j={j} → dst={dst} δ={delta:.6f} score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣ Final mixing + very light random‑pair hill‑climb
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Convex mixing with the uniform vector and a cheap random‑pair hill‑climb."""
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing (lam from 0 to 1)
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # inexpensive random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill‑climb i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score
