assistantfinal**Solution Overview**

The score to minimise is  

\[
\text{score}(a)=\frac{2\,n\;\max(a\star a)}{(\sum_i a_i)^2}\;,
\]

where `a⋆a` is the linear convolution of the non‑negative vector `a`.  
The ratio is *scale‑invariant* – we can (and do) normalise `a` to
`sum(a)=1`.  After normalisation the score is simply  

\[
\text{score}=2\,n\;\max(a\star a)\; .
\]

The algorithm below follows three stages

1. **Warm‑start** – a few analytically motivated candidates (uniform,
   the arcsine law `Beta(0.5,0.5)`, mixtures with the uniform vector,
   and a handful of Dirichlet draws).

2. **Fine‑grid search** – around the best Beta‑shape we sweep a fine
   grid of the mixing weight `λ` (uniform ↔ Beta) and of the shape
   parameter `p`.  This already brings the score close to the target.

3. **Projected sub‑gradient descent** – the current best vector is
   repeatedly improved.
   * The index `k*` where the convolution attains its maximum is located.
   * The sub‑gradient of `max(a⋆a)` at this index is
     \[
     g_j =
        \begin{cases}
          2\,a_{k^*-j}&\text{if }0\le k^*-j<n,\\[2mm]
          0            &\text{otherwise}.
        \end{cases}
     \]
   * We move a small step opposite to the sub‑gradient and **project**
     the result back onto the probability simplex
     (`a_i≥0, ∑a_i=1`).  
   * If the step does not improve the score the step size is reduced;
     otherwise it is kept (or slightly increased).  
   * Occasionally a random “pair‑transfer’’ move (as in the previous
     hill‑climb) is tried to escape shallow plateaus.

All convolutions are performed by FFT (`O(n log n)`).  The routine
honours the 1000 s budget, prints progress and finally returns the best
sequence it found.

```python
"""
Project‑gradient + fine Beta‑grid search for the step‑function minimising

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm:
  • warm‑start with uniform, arcsine (Beta(0.5,0.5)), mixtures with uniform
    and a few Dirichlet draws;
  • scan a fine grid of Beta(p,p) shapes and mixing weights λ;
  • improve the best candidate by projected sub‑gradient descent on the
    active convolution index, supplemented by occasional random pair‑mass
    transfers.
The search stops after the allotted time (default 1000 s) and returns the
best vector (normalised to sum = 1).  All numbers are non‑negative.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (sum(a) assumed = 1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised symmetric Beta(p,p) density on the grid
    x_i = (i+0.5)/n, normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w | w_i ≥ 0, Σ w_i = 1}.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a⋆a) at the active index k.
    g_j = 2·a_{k-j} for indices where the term exists, otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    # vectorised assignment
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g[idx] = 2.0 * a[k - idx]
    return g

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and prints
    progress.  The returned list is normalised to sum ≈ 1.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # --------------------------------------------------------------
    # Problem size – a power of two makes the FFT cheap
    # --------------------------------------------------------------
    n = 4096                                          # can be increased if desired
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Helper to keep the best solution seen so far
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _score_of(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 0️⃣  Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(globals().get('height_sequence_1'), dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _score_of(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Warm‑start with a few Dirichlet draws (≈5 % of the budget)
    # --------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * budget_s)
    rng = np.random.default_rng(seed + 1)
    while time.time() < warm_deadline:
        # concentration drawn log‑uniformly in [0.02, 20]
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = math.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _score_of(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet warm‑start → score = {sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Fine‑grid search over Beta(p,p) and mixtures with uniform
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Scanning Beta‑family (coarse)")
    p_vals = np.linspace(0.30, 1.70, 29)          # coarse sweep
    for p in p_vals:
        shape = _beta_shape(n, p)
        sc = _score_of(shape, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, shape.copy()
            print(f"[{time.time() - start_time:.2f}s] pure p={p:.3f} → score = {sc:.6f}")

        for lam in np.linspace(0.0, 0.20, 11):   # only a small amount of uniform helps
            cand = lam * shape + (1.0 - lam) * uniform
            cand = cand / cand.sum()
            sc = _score_of(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] p={p:.3f} λ={lam:.2f} → score = {sc:.6f}")

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 3️⃣  Refined sweep around the best p (λ resolution 0.01)
    # --------------------------------------------------------------
    # Determine p that produced the current best_a (if it is a Beta‑shape)
    # For safety we just re‑explore a tight rectangle around p≈0.5…
    print(f"[{time.time() - start_time:.2f}s] Refined Beta sweep")
    p_center = 0.5
    for p in np.linspace(p_center - 0.10, p_center + 0.10, 41):
        shape = _beta_shape(n, p)
        for lam in np.linspace(0.0, 0.10, 21):
            cand = lam * shape + (1.0 - lam) * uniform
            cand = cand / cand.sum()
            sc = _score_of(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] fine p={p:.4f} λ={lam:.3f} → score = {sc:.6f}")
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent (main optimisation loop)
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Gradient‑descent refinement")
    a = best_a.copy()
    step = 0.1                      # initial step size
    min_step = 1e-8
    iter_cnt = 0
    rng = np.random.default_rng(seed + 42)

    while time.time() < deadline:
        # compute current convolution and its maximum
        conv = _conv_self_fft(a, L_fft)
        max_b = conv.max()
        sc = 2.0 * n * max_b
        if sc < best_score - 1e-12:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] new best (gd) score = {sc:.6f}")

        # sub‑gradient at the active index
        k_star = int(conv.argmax())
        g = _gradient_max_conv(a, k_star)

        # scale step adaptively (keep the update magnitude moderate)
        norm_g = np.linalg.norm(g)
        if norm_g == 0:
            # fall back to a random small perturbation
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(a[i], 0.001) * rng.random()
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # try a step; if it worsens, halve the step size
            alpha = step / norm_g
            cand = a - alpha * g
            cand = _project_to_simplex(cand)

        cand_sc = _score_of(cand, n, L_fft)

        if cand_sc < sc:
            # accept improvement, maybe increase step a bit
            a = cand
            step = min(step * 1.05, 1.0)
        else:
            # reject – reduce step size
            step = max(step * 0.5, min_step)

        # occasional random pair‑transfer to help escaping plateaus
        if rng.random() < 0.02:
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(0.0, min(a[i], 0.001))
            if delta > 0:
                cand2 = a.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = _project_to_simplex(cand2)
                cand2_sc = _score_of(cand2, n, L_fft)
                if cand2_sc < best_score:
                    best_score, best_a = cand2_sc, cand2.copy()
                    a = cand2
                    print(f"[{time.time() - start_time:.2f}s] random transfer improve → score = {cand2_sc:.6f}")

        iter_cnt += 1
        # tiny progress print every few thousand iterations
        if iter_cnt % 5000 == 0:
            print(f"[{time.time() - start_time:.2f}s] iter {iter_cnt}, best_score ≈ {best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final tiny uniform mixing (helps to smooth residual spikes)
    # --------------------------------------------------------------
    # try mixing a few percent of the uniform vector one last time
    for lam in np.linspace(0.0, 0.03, 7):
        cand = lam * best_a + (1.0 - lam) * uniform
        cand = cand / cand.sum()
        sc = _score_of(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] final uniform mix λ={lam:.3f} → score = {sc:.6f}")

    # --------------------------------------------------------------
    # Normalise once more and return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
```
