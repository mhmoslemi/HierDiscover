"""
Propose a step‑function (non‑negative vector) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines several cheap heuristics:

1️⃣  A short random Dirichlet sweep and a few deterministic windows
    (uniform, sine, Hann, triangular) to obtain a decent initial point.

2️⃣  A smooth‑gradient descent on the *soft‑max* (log‑sum‑exp) approximation
    of the convolution‑max.  The probabilities are represented by a
    soft‑max of unrestricted parameters `y`, so the simplex constraint is
    satisfied automatically.  The descent uses the Adam optimiser and a
    standard β‑schedule (0.5 → … → 2000) that gradually focuses the loss on
    the true maximum.

3️⃣  A very cheap random “mass‑transfer’’ hill‑climb for the last few seconds.

All heavy convolutions are performed with a power‑of‑2 FFT, and the code
monitors the 1000 s time budget.  The best sequence seen is returned as a
plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used only for printing)
# ----------------------------------------------------------------------
_start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using an FFT of power‑of‑2 length."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


# ----------------------------------------------------------------------
# Objective helpers (sequence is forced to sum to 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max  m_β(conv) = log‑sum‑exp_β(conv)
    w.r.t. a.  `beta` is the temperature.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()                         # shape (2·n‑1,)
    # gradient = 2 * (a_rev ⋆ w)  (the slice that corresponds to a)
    rev = _convolve_fft(a[::-1], w)
    n = a.shape[0]
    return 2.0 * rev[n - 1 : n - 1 + n]


def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Small helper to keep track of elapsed time
# ----------------------------------------------------------------------
def _elapsed() -> float:
    return time.time() - _start_time


# ----------------------------------------------------------------------
# Random Dirichlet draws – cheap way to get a few good candidates
# ----------------------------------------------------------------------
def _dirichlet_trials(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10.0 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
            print(f"[{_elapsed():.2f}s] Dirichlet c={c:.4f}  score={sc:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
# Deterministic starter windows (uniform and a few classic shapes)
# ----------------------------------------------------------------------
def _deterministic_starters(n: int):
    starters = []

    # uniform
    starters.append(np.full(n, 1.0 / n))

    # sine window
    starters.append(np.sin(np.pi * (np.arange(n) + 1) / (n + 1)))

    # Hann window
    starters.append(0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # symmetric triangular window
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    starters.append(tri)

    # normalise each starter to the simplex
    out = []
    for s in starters:
        s = np.maximum(s, 0.0)
        s_sum = s.sum()
        if s_sum > 0.0:
            s = s / s_sum
            out.append(s)
    return out


# ----------------------------------------------------------------------
# Adam optimiser (operates on the unrestricted vector `y` – softmax of `y`
# gives the probability vector `a`).
# ----------------------------------------------------------------------
class Adam:
    def __init__(self, dim: int, lr: float = 0.1,
                 beta1: float = 0.9, beta2: float = 0.999, eps: float = 1e-8):
        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps

    def step(self, grad: np.ndarray) -> np.ndarray:
        """Return the parameter update (to be subtracted from the parameters)."""
        self.t += 1
        self.m = self.beta1 * self.m + (1.0 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1.0 - self.beta2) * (grad * grad)

        m_hat = self.m / (1.0 - self.beta1 ** self.t)
        v_hat = self.v / (1.0 - self.beta2 ** self.t)

        update = self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
        return update


# ----------------------------------------------------------------------
# Very cheap random hill‑climbing (mass‑transfer) for the last seconds
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur = score0
    best = a0.copy()
    best_score = score0
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 0.0:
            continue
        # move up to 5 % of the source mass
        delta = a[src] * rng.random() * 0.05
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # soft‑project back onto the simplex (tiny numerical drift)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            cur = cand_score
            a = cand
            print(f"[{_elapsed():.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur:
            # occasional downhill move
            a = cand
            cur = cand_score
    return best, best_score

    """
    Search for a probability vector (step‑function) that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget
    and returns the best sequence found.
    """
    global _start_time
    np.random.seed(seed)
    _start_time = time.time()
    deadline = _start_time + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between expressiveness
    # and speed (the original solution used the same size).
    # ------------------------------------------------------------------
    n = 4096

    # keep the best vector seen so far
    best_a = np.full(n, 1.0 / n)
    best_score = _objective_score(best_a)
    print(f"[{_elapsed():.2f}s] start – uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter (height_sequence_1) if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            ss = starter.sum()
            if ss > 0.0:
                starter = starter / ss
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score = sc
                    best_a = starter.copy()
                    print(f"[{_elapsed():.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – a quick Dirichlet sweep and a few deterministic shapes
    # ------------------------------------------------------------------
    dir_deadline = min(deadline, _start_time + 20.0)   # 20 s for Dirichlet
    seq, sc = _dirichlet_trials(n, max_trials=500, deadline=dir_deadline)
    if seq is not None and sc < best_score:
        best_a, best_score = seq, sc
        print(f"[{_elapsed():.2f}s] after Dirichlet best = {best_score:.6f}")

    for shp in _deterministic_starters(n):
        sc = _objective_score(shp)
        if sc < best_score:
            best_a, best_score = shp.copy(), sc
            print(f"[{_elapsed():.2f}s] deterministic shape score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth Adam descent on the log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # initialise the unrestricted parameters y (soft‑max of y = a)
        eps = 1e-12
        y = np.log(best_a + eps)

        # Adam optimiser
        adam = Adam(dim=n, lr=0.12)

        # β‑schedule (slow early, aggressive later)
        beta_sched = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]

        # allocate ~85 % of the remaining time to this phase
        grad_deadline = time.time() + 0.85 * (deadline - time.time())

        for beta in beta_sched:
            if time.time() >= grad_deadline:
                break
            # run until either the beta‐stage timeout expires or we run out of budget
            stage_end = time.time() + (grad_deadline - time.time()) / max(1, len(beta_sched) - beta_sched.index(beta))
            while time.time() < stage_end and time.time() < deadline:
                a = _softmax(y)                     # current probability vector
                conv = _conv_self(a)
                M = conv.max()
                cur_score = float(2.0 * n * M)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{_elapsed():.2f}s] β={beta:.1f} new best {best_score:.6f}")

                # smooth‑max gradient (log‑sum‑exp with temperature β)
                grad_a = _smooth_gradient(a, conv, beta)

                # chain rule through the soft‑max
                grad_y = a * (grad_a - np.dot(grad_a, a))

                # Adam update (subtract because we minimise)
                upd = adam.step(grad_y)
                y -= upd

                # occasional tiny jitter to avoid stagnation (once every 2000 steps)
                if adam.t % 2000 == 0:
                    y += np.random.default_rng().normal(scale=1e-4, size=n)

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing tweak (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{_elapsed():.2f}s] starting final hill‑climb")
        best_a, best_score = _hill_climb(best_a, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise once more (tiny numerical drift may have occurred)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{_elapsed():.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
