"""
Flattened‐convolution optimizer
--------------------------------

1. Choose a moderately large dimension (default 8192).  A larger `n`
   lowers the theoretical lower bound (≈ 2 n/(2 n‑1) → 1).

2. Initialise several candidate vectors:
   * the uniform distribution,
   * an “arcsine’’ shape (high at the ends, low in the centre),
   * many random Dirichlet draws with a wide range of concentration
     parameters.

3. Refine each candidate with a two‑stage gradient method:
   – a smooth approximation of the maximum (log‑sum‑exp) is minimised
     using Adam (adaptive moments) with a schedule of increasing β
     (temperature).  The exact objective is monitored after each step
     and the best vector is kept.
   – a final sub‑gradient descent on the true max (no smoothing) fine‑tunes
     the solution.

4. End with a short stochastic hill‑climbing phase that transfers a small
   random amount of mass between two positions if it improves the exact
   objective.

All vectors are projected onto the probability simplex after every update,
so the denominator ``sum(a)**2`` stays constant (1) and the objective reduces
to ``2·n·max(conv(a,a))``.  The routine runs for at most ``budget_s`` seconds
and returns the best sequence found (as a plain Python list of non‑negative
floats).  The implementation uses only NumPy and the FFT for fast convolutions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer for progress printing
# ----------------------------------------------------------------------
_start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a power‑of‑2 FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (the vector is assumed to sum to 1 → denominator fixed)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp approximation of max(conv) with temperature β."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max with respect to a.
    Uses the identity   grad = 2 * (a_rev ⋆ w)   where w are the softmax weights.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()                     # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)     # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Random Dirichlet initialisation
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 0.7] (≈ [0.001, 5])
        conc = 10 ** rng.uniform(-3.0, 0.7)
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Dirichlet c={conc:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic “arcsine’’ shape
# ----------------------------------------------------------------------
def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine distribution on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    # shift by 0.5 to avoid division by zero at the boundaries
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    # normalise to a probability vector
    w = np.maximum(w, 0.0)
    w = _project_simplex(w)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  beta_vals,
                  lr: float,
                  deadline: float):
    """
    Adam optimiser on the smoothed max (log‑sum‑exp) of the self‑convolution.
    Returns the best vector and its exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    for beta in beta_vals:
        # a few passes at each temperature
        while time.time() < deadline:
            t += 1
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

            cur_score = _objective_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - _start_time:.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            # very early exit for this β if progress stalls
            if np.linalg.norm(step) < 1e-12:
                break
        # proceed to the next, larger β
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max (no smoothing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         lr: float,
                         max_iters: int,
                         deadline: float):
    """Plain sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        k_max = int(np.argmax(conv))

        # gradient of max(conv) wrt a  (sparse representation)
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k_max - idx
        grad[idx] = 2.0 * a[jdx]

        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Sub‑grad iter {it} new best {best_score:.6f}")
            # slow down after improvement
            lr *= 0.95
        else:
            # modestly shrink step size when no improvement
            lr *= 0.995
        # stop if step becomes negligible
        if lr < 1e-12:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Final short random hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                score0: float,
                deadline: float):
    """Random mass‑transfer moves for the last seconds."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # move up to 5 % of the source mass
        delta = a[src] * rng.random() * 0.05
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - _start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            # tiny uphill move to escape tiny plateaus
            a = cand
            cur_score = cand_score
    return best_a, best_score
