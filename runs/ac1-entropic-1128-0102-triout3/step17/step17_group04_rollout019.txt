"""
Hybrid optimizer for the convolution‐max problem.

The algorithm consists of four stages:

1️⃣  Very cheap parametric sweep (geometric, exponential, two‑block and
    three‑block families) on the full length `n=4096`.

2️⃣  A massive Dirichlet sampling phase – many random probability vectors
    with log‑uniform concentration parameters – to obtain a decent starter.

3️⃣  A “flattening” gradient descent that minimises the L2 deviation of the
    self‑convolution from the uniform distribution (the ideal flat spectrum).
    The gradient is obtained analytically via a single FFT, an additive step
    is taken and the vector is projected back onto the probability simplex.

4️⃣  A final smooth‑max refinement (log‑sum‑exp surrogate) identical to the
    previous solution but with a more aggressive β‑schedule, starting from the
    best vector found in stage 3.

All heavy work uses FFTs; the routine stops well before the 1000 s budget
and returns the best sequence discovered (as a plain Python list)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolutions)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (returns full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projections / updates on the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _add_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Simple additive update followed by simplex projection:
    a ← a - η·grad ,   then project onto {x≥0, Σx=1}.
    """
    a_new = a - eta * grad
    return _project_simplex(a_new)

# 3️⃣  Flattening L2‑gradient descent (push convolution towards flat)

def _l2_error_and_grad(a: np.ndarray):
    """
    Returns (error, gradient) where error = Σ (conv - avg)^2
    and gradient = ∂error/∂a (using the analytic convolution formula).
    """
    n = a.shape[0]
    conv = _conv_self(a)                         # length 2n‑1
    avg = 1.0 / (2 * n - 1)                      # ideal flat value
    e = conv - avg                               # deviation vector
    # gradient: 2· Σ_k e_k·a_{k‑i}  (i.e. 2· (e * a)_{i+offset})
    e_rev = e[::-1]
    conv_rev = _conv_fft(a, e_rev)               # length 3n‑2
    start = len(e) - 1                           # = 2n‑2
    grad = 2.0 * conv_rev[start - np.arange(n)]
    error = np.sum(e * e)
    return error, grad


def _flatten_descent(a0, deadline, best_seq, best_score):
    """L2‑error minimisation on the convolution, keeping the best discrete score."""
    a = a0.copy()
    cur_score = _objective_score(a)
    eta = 0.5
    min_eta = 1e-12
    while time.time() < deadline:
        err, grad = _l2_error_and_grad(a)

        # try a back‑tracking additive step
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _add_update(a, grad, eta_try)
            new_score = _objective_score(a_new)
            if new_score < cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Flatten descent score={new_score:.6f}")
            eta = min(1.0, eta * 1.02)          # slowly enlarge step when progress is good
        else:
            # no improvement – jitter a little and continue
            rng = np.random.default_rng()

            a = a + perturb
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / a.shape[0])

    return best_seq, best_score

def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient w.r.t a).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # stabilise
    Z = exp_scaled.sum()
    w = exp_scaled / Z                         # softmax weights
    # gradient  = 2 * Σ_j a_j * w_{i+j}
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)          # length 3n‑2
    start = len(w) - 1                      # = 2n‑2
    G = 2.0 * conv_rev[start - np.arange(n)]
    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, G

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine runs a quick parametric sweep,
    a large Dirichlet sampling phase, a flattening L2‑gradient descent,
    and finishes with a smooth‑max refinement.  The best vector found
    (normalised to sum = 1) is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Quick parametric sweep (expanded families)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 500.0
    if time.time() < phase_deadline:
        best_seq, best_score = _quick_sweeps(n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (large number)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline, trials=12000
        )

    # ------------------------------------------------------------------
    # 3️⃣ Flattening L2‑gradient descent (core optimiser)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting flattening descent")
        best_seq, best_score = _flatten_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After flattening best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final smooth‑max refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max refinement")
        best_seq, best_score = _smooth_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a clean list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
