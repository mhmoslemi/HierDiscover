"""
Propose a step‐function (non‑negative vector) that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the full linear self‑convolution.
The routine combines a diversified population of start points,
a lightweight Differential‑Evolution (DE) loop and a final
projected‑gradient polishing phase.  All vectors are kept on the
probability simplex (sum = 1) so the denominator cancels automatically.
The DE mutation uses only simple arithmetic and cheap clipping,
making it fast enough to explore thousands of candidates within the
1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing
# ----------------------------------------------------------------------
start_time = None  # will be initialised in propose_candidate


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (the denominator is 1 because we keep the vector on the simplex)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at ``a`` (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / float(active.size)
    return grad


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n, dtype=float)


def _make_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    """Draw a Dirichlet‑distributed vector with concentration ``conc``."""
    a = rng.gamma(shape=conc, scale=1.0, size=n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def _make_sparse(n: int, sparsity_ratio: float, rng: np.random.Generator) -> np.ndarray:
    """Random sparse vector with roughly ``sparsity_ratio`` non‑zero entries."""
    k = max(2, int(sparsity_ratio * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Projected gradient descent (fine‑tuning)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray,
                                cur_score: float,
                                deadline: float) -> (np.ndarray, float):
    """Simple PGD that follows the sub‑gradient of the active convolution entries."""
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    step = 0.5

    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random local “polish”
# ----------------------------------------------------------------------
def _random_local_refinement(a0: np.ndarray,
                             cur_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> (np.ndarray, float):
    """Very cheap pairwise transfers – only accept strict improvements."""
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        delta = a[src] * rng.random() * 0.005
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Differential Evolution core
# ----------------------------------------------------------------------
def _differential_evolution(pop: np.ndarray,
                            scores: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> (np.ndarray, float):
    """
    Very light DE (DE/best/1) on the simplex.
    *pop* – shape (pop_size, n) and already normalised.
    *scores* – current objective values.
    Returns the best vector found.
    """
    pop_size, n = pop.shape
    # keep a copy of the current best
    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    F = 0.5          # mutation factor
    CR = 0.9         # crossover probability

    while time.time() < deadline:
        for i in range(pop_size):
            # --- mutation: best/1 ---
            # choose two distinct indices a,b ≠ i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b = rng.choice(idxs, size=2, replace=False)
            donor = best_vec + F * (pop[a] - pop[b])

            # enforce non‑negativity and renormalise
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            if ds == 0.0:
                continue
            donor /= ds

            # --- crossover ---
            mask = rng.random(n) < CR
            trial = np.where(mask, donor, pop[i])
            ts = trial.sum()
            if ts == 0.0:
                continue
            trial /= ts

            # --- selection ---
            trial_score = _objective_score(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    print(f"[{time.time() - start_time:.2f}s] DE new best {best_score:.6f}")

        # keep loop running until the global deadline expires
    return best_vec, best_score

    # ------------------------------------------------------------------

    try:

    # ------------------------------------------------------------------
    # Tiny random local refinement (polishing)
    # ------------------------------------------------------------------
