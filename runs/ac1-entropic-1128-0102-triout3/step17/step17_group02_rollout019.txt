        mask = (j >= 0) & (j < n)
        # product contributions
        p = np.zeros(n)
        p[mask] = cur_a[i[mask]] * cur_a[j[mask]]
        # find source index i_max with largest contribution
        i_max = np.argmax(p)

        # compute epsilon to move
        eps = cur_a[i_max] * step_factor
        if eps <= 0:
            # break if cannot move
            break
        # ensure not exceed dest capacity - not needed
        cand = cur_a.copy()
        cand[i_max] -= eps

        # ensure non-negative
        if cand[i_max] < 0:
            # adjust eps
            eps = cur_a[i_max]
            cand[i_max] = 0.0

        cand_score = _objective_score(cand)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            # update best
            if cand_score < best_score:

"""
Improved 4096‑dimensional step‐function optimiser.

The algorithm consists of the following stages (all work on a probability
vector, i.e. sum = 1, so the original evaluation
2·n·max(conv(a,a)) is the exact objective of ``evaluate_sequence``).

1️⃣  A quick wide‑range random Dirichlet sweep plus a few deterministic
    shapes (uniform, sine, Hann, linear ramp, exponential).  The best
    of these becomes the seed for the remaining phases.

2️⃣  Smooth projected‑gradient descent (the same log‑sum‑exp smoothing as
    the reference code, but with a richer β‑schedule and more restarts).

3️⃣  Targeted “max‑convolution” reduction – we repeatedly locate the shift
    *t* that gives the current maximum of the self‑convolution and move a
    small amount of mass away from the index that contributes the largest
    product a[i]·a[t‑i] towards an index that contributes almost nothing.
    This directly attacks the bottleneck term and steadily lowers the
    objective.

4️⃣  A short random hill‑climbing (mass‑transfer) phase for final polish.

All heavy convolutions are performed with an FFT of length a power‑of‑2.
The routine respects a 1000 s time budget – the outer loop checks the
deadline and stops early if necessary.  The returned sequence is a plain
Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing – used by the helper functions for progress prints
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using an FFT of length a power‑of‑2."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (vectors are normalised to sum = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Smooth approximation of max(conv) via log‑sum‑exp."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    w are the soft‑max weights of the convolution entries.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()               # shape (2·n‑1,)
    # cross‑correlation a ⋆ w  (equivalently convolution of a_rev with w)
    rev_corr = _convolve_fft(a[::-1], w)   # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Random Dirichlet initialisation
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth projected‑gradient descent (β‑schedule)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    score0: float,
                    beta_vals,
                    deadline: float):
    """
    Gradient descent on the smooth max‑approximation.
    β grows gradually; after each stage we project onto the simplex.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    for beta in beta_vals:
        step = 0.5
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            improved = False
            for factor in (1.0, 0.5, 0.25, 0.125):
                cand = a - step * factor * grad
                cand = _project_simplex(np.maximum(cand, 0.0))
                cand_score = _objective_score(cand)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    improved = True
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} new best {best_score:.6f}")
                    break
            if not improved:
                step = max(step * 0.5, 1e-12)
            else:
                step = min(step * 1.05, 1.0)

            if step < 1e-12:
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted max‑convolution reduction (new component)
# ----------------------------------------------------------------------
def _targeted_reduce(a0: np.ndarray,
                     score0: float,
                     deadline: float):
    """
    Greedy reduction of the current maximal convolution entry.
    Each iteration identifies the shift t* with the largest conv[t*],
    picks the contributing index i* with the largest product a[i]*a[t*‑i],
    and moves a small fraction of its mass to an index that contributes
    almost nothing to the same shift.  The move is accepted if it does not
    increase the objective; otherwise it is rejected (or accepted with a
    tiny annealing probability).  The step size decays slowly.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    # initial fraction of source mass to move
    step_frac = 0.1

    while time.time() < deadline:
        conv = _conv_self(a)
        t_max = int(np.argmax(conv))

        # product contributions for this shift
        i = np.arange(n)
        j = t_max - i
        mask = (j >= 0) & (j < n)
        prod = np.zeros(n)
        prod[mask] = a[i[mask]] * a[j[mask]]

        # source index = one with largest contribution
        i_src = int(np.argmax(prod))

        # destination candidates: indices where the product for the same shift is tiny
        j2 = t_max - i
        mask2 = (j2 >= 0) & (j2 < n)
        prod_dest = np.full(n, np.inf)
        prod_dest[mask2] = a[i[mask2]] * a[j2[mask2]]
        # avoid picking the source itself
        prod_dest[i_src] = np.inf
        # also avoid moving to positions whose partner is out of range (product=0)
        # – those already have value 0 and are ideal.
        i_dst = int(np.argmin(prod_dest))

        # amount to move
        eps = step_frac * a[i_src]
        if eps <= 0.0:
            break
        cand = a.copy()
        cand[i_src] -= eps
        cand[i_dst] += eps

        # safeguard from negative due to rounding
        if cand[i_src] < 0.0:
            eps += cand[i_src]   # make it zero
            cand[i_src] = 0.0

        cand_score = _objective_score(cand)

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] targeted new best {best_score:.6f}")
        else:
            # tiny Metropolis step to occasionally escape plateaus
            T = max(1e-6, (deadline - time.time()) / (deadline - start_time))
            if rng.random() < np.exp(-(cand_score - cur_score) / T):
                a = cand
                cur_score = cand_score

        # slowly shrink the step fraction
        step_frac = max(0.001, step_frac * 0.9995)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final random hill‑climbing refinement (same as before)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                score0: float,
                deadline: float):
    """Cheap random mass‑transfer moves for the final few seconds."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        delta = a[src] * rng.random() * 0.05
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score
    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) discovered within the
    allotted time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1           # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between expressiveness
    # and speed.
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq = starter.copy()
                    best_score = sc
                    print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet + deterministic shapes
    # ------------------------------------------------------------------
    dir_deadline = min(deadline, start_time + 35.0)
    if time.time() < dir_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=8000, deadline=dir_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

    # deterministic shapes (sine, Hann, linear ramp, exponential)
    shapes = []

    # sine shape
    shapes.append(np.sin(np.pi * (np.arange(n) + 1) / (n + 1)))

    # Hann (raised‑cosine) window
    shapes.append(0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # symmetric triangular window
    triangular = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    shapes.append(triangular)

    # monotone linear ramp (large → small)
    shapes.append(np.linspace(2.0, 0.0, n))

    # exponential decay
    shapes.append(np.exp(-0.001 * np.arange(n)))

    for shp in shapes:
        shp = np.maximum(shp, 0.0)
        shp = _project_simplex(shp)
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq, best_score = shp.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] deterministic shape score {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth projected‑gradient descent with a rich β‑schedule
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.70
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                         50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0]

        restarts = 10
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.01, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0_score = _objective_score(a0)

            sub_deadline = min(descent_deadline,
                               time.time() + (descent_deadline - time.time()) /
                                            max(1, restarts - r))
            a_opt, sc_opt = _smooth_descent(a0, a0_score, beta_schedule, sub_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted reduction of the bottleneck convolution entry
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting targeted reduction")
        best_seq, best_score = _targeted_reduce(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Phase 4 – short hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise (protect against accidental drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
