"""
Propose a step‑function (non‑negative list) with a low value of  

    2 * n * max(convolution(seq, seq)) / (sum(seq)**2)

The search consists of several cheap sweeps of simple symmetric families,
many random Dirichlet samples (symmetrised), and a dedicated
mirror‑descent optimiser that works on the exact max‑convolution
(using a sub‑gradient of the ℓ∞‑norm).  The optimiser uses an exponential
update (guaranteeing non‑negativity and unit sum) and a simple line‑search.
Occasional random jitter is injected to escape plateaus.  All work is done
with FFT‑based convolutions for speed.  The routine stops before the 1000 s
budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Returns the objective 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    The function works for any non‑negative seq; the caller should normalise.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent (exponential) update: a_i ← a_i·exp(−η·grad_i), then renormalise.
    Guarantees a_i ≥ 0 and Σ a_i = 1.
    """
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)        # avoid overflow
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        # fall back to uniform if something goes wrong
        n = a.shape[0]
        return np.full_like(a, 1.0 / n)
    return a_new / total


def _max_grad(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Uses the exact active set (entries attaining the maximum).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    M = conv.max()
    # active mask (could be more than one index due to ties)
    mask = (conv >= M - 1e-12).astype(float)
    mask_sum = mask.sum()
    if mask_sum == 0:
        return np.zeros_like(a)

    w = mask / mask_sum               # distribute weight uniformly over active ks
    # gradient of Σ_k w_k·conv_k is 2 * a *⊗ w  (⊗ = cross‑correlation)
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)    # length 3n‑2
    start = len(w) - 1                # = 2n‑2
    grad = 2.0 * conv_rev[start - np.arange(n)]
    return grad


def _symmetrize(vec: np.ndarray) -> np.ndarray:
    """
    Produce a symmetric (mirror) ordering that places the largest values at the ends
    and the smallest near the centre.  The operation preserves the multiset of
    entries and the total sum.
    """
    vals = np.sort(vec)[::-1]          # descending
    n = len(vals)
    out = np.empty_like(vals)
    left, right = 0, n - 1
    i = 0
    while left <= right:
        out[left] = vals[i]
        i += 1
        if left != right:
            out[right] = vals[i]
            i += 1
        left += 1
        right -= 1
    return out

def _parametric_sweeps(n, best_seq, best_score, deadline):
    """Explore a few low‑dimensional families of symmetric shapes."""
    idx = np.arange(n, dtype=np.float64)

    # --------------------------------------------------------------
    # 1. Power‑law on distance from centre (U‑shapes and hill‑shapes)
    # --------------------------------------------------------------
    centre = (n - 1) / 2.0
    dist = np.abs(idx - centre) + 1e-12      # never zero
    for p in np.linspace(-2.0, 2.0, 41):
        if time.time() >= deadline:
            break
        shape = dist ** p
        shape /= shape.sum()
        shape = _symmetrize(shape)          # enforce the “big‑at‑ends” ordering
        sc = _objective_score(shape)
        if sc < best_score:
            best_score, best_seq = sc, shape.copy()
            print(f"[{time.time() - start_time:.2f}s] pow‑law p={p:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2. Beta‑like family a_i ∝ x^α (1‑x)^β   (x∈[0,1])
    # --------------------------------------------------------------
    x = (idx + 0.5) / n
    alphas = np.linspace(-0.4, 2.0, 13)
    betas = np.linspace(-0.4, 2.0, 13)
    for α in alphas:
        for β in betas:
            if time.time() >= deadline:
                break
            # avoid infinities when α or β is negative
            a = np.power(x, np.clip(α, -0.4, 10.0)) * np.power(1.0 - x, np.clip(β, -0.4, 10.0))
            # small epsilon to keep numerics stable
            a += 1e-12
            a /= a.sum()
            a = _symmetrize(a)
            sc = _objective_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] beta α={α:.2f},β={β:.2f} score={sc:.6f}")

    return best_seq, best_score


def _random_dirichlet_trials(n, best_seq, best_score, deadline, max_trials=4000):
    """Generate many symmetric Dirichlet draws with various concentrations."""
    rng = np.random.default_rng()
    half = (n + 1) // 2           # we generate only half and mirror
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in [0.01, 10]
        logc = rng.uniform(-2.0, 1.0)
        conc = 10 ** logc
        # sample only the first half; the second half will be its mirror
        half_vec = rng.gamma(shape=conc, scale=1.0, size=half)
        # enforce symmetry
        if n % 2 == 0:
            full = np.concatenate([half_vec, half_vec[::-1]])
        else:
            full = np.concatenate([half_vec[:-1], half_vec[::-1]])
        full /= full.sum()
        full = _symmetrize(full)          # extra safety (keeps ordering)
        sc = _objective_score(full)
        if sc < best_score:
            best_score, best_seq = sc, full.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet conc={conc:.3f} score={sc:.6f}")
    return best_seq, best_score


def _max_gradient_descent(a0, deadline, best_seq, best_score):
    """
    Mirror‑descent minimisation of the true max‑convolution.
    The update follows a_i ← a_i·exp(−η·g_i) with a line‑search on η.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    eta = 0.4               # initial step size
    min_eta = 1e-12
    stagnation = 0

    while time.time() < deadline:
        grad = _max_grad(a)

        # try an exponential update; backtrack if no improvement
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, grad, eta_try)
            new_score = _objective_score(a_new)
            if new_score <= cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            stagnation = 0
            # record global best
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Descent η={eta_try:.2e} score={new_score:.6f}")
            # slowly increase η after a success
            eta = min(1.0, eta * 1.02)
        else:
            # no improvement – inject a modest random jitter
            stagnation += 1
            rng = np.random.default_rng()
            # jitter size grows with stagnation (to help escape plateaus)
            scale = min(0.2, 0.01 * stagnation)
            eps = scale * rng.random(n) * a
            a = a + eps
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.6)

        # early exit if we already beat the target comfortably
        if best_score <= 1.45:
            break

    return best_seq, best_score

    """

    """
    np.random.seed(seed)

    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # --------------------------------------------------------------
    # Problem size (length of the step function)
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()

    # --------------------------------------------------------------

            if s > 0:

                sc = _objective_score(starter)

                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric sweeps (few seconds)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
