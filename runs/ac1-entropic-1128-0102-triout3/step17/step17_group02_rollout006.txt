"""
A fast multi‑phase optimizer for the self‑convolution objective.

The objective (with sum = 1) is 2·n·max(conv(a,a)).
We search for a probability vector a that makes the convolution as flat as possible,
hence minimising the maximum entry.
The algorithm consists of

1️⃣  a cheap sweep over several parametric families (exponential, Gaussian,
    power‑law, cosine‑raised) – this often finds a good starting shape;

2️⃣  a larger random‐Dirichlet exploration (still cheap thanks to FFT‑based convolutions);

3️⃣  an Adam‑style projected gradient descent on a smooth approximation of the max,
    with a temperature β that is increased continuously (log‑schedule);
    the step is projected onto the simplex after each update;

4️⃣  a short stochastic hill‑climbing phase that moves tiny amounts of mass
    between random coordinates; occasional “smooth‑move’’ steps (local averaging)
    help to flatten the convolution further.

All heavy convolutions use FFTs (O(n log n)) and the Euclidean projection onto the
probability simplex (O(n)).  The routine respects the 1000 s time budget and
returns the best sequence found as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing variable – set in propose_candidate
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a power‑of‑2 FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum forced to 1 → denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Uses the identity grad = 2 * (a_rev * w)[offset] where w are the softmax weights.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()
    rev_corr = _convolve_fft(a[::-1], w)                      # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad

def _explore_parametric_shapes(n: int, deadline: float):
    """Search over a few analytic families and return the best vector found."""
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # Helper to evaluate & possibly update the best solution
    def try_seq(seq):
        nonlocal best_seq, best_score
        sc = _objective_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] new parametric best {best_score:.6f}")

    # 1. Exponential decay (monotone)
    for r in np.linspace(0.80, 0.9999, 30):
        if time.time() >= deadline:
            break
        a = np.power(r, np.arange(n, dtype=float))
        a /= a.sum()
        try_seq(a)

    # 2. Symmetric exponential (peak in the centre)
    centre = (n - 1) / 2.0
    for r in np.linspace(0.80, 0.9999, 30):
        if time.time() >= deadline:
            break
        idx = np.abs(np.arange(n, dtype=float) - centre)
        a = np.power(r, idx)
        a /= a.sum()
        try_seq(a)

    # 3. Gaussian centred
    mu = centre
    for sigma in np.linspace(n / 16.0, n / 2.0, 30):
        if time.time() >= deadline:
            break
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
        a /= a.sum()
        try_seq(a)

    # 4. Power‑law (heavy tail)
    for alpha in np.linspace(0.5, 3.0, 30):
        if time.time() >= deadline:
            break
        a = 1.0 / np.power(np.arange(1, n + 1, dtype=float), alpha)
        a /= a.sum()
        try_seq(a)

    # 5. Raised cosine (Hann) with an exponent
    x = np.arange(n, dtype=float) / (n - 1)
    base = 0.5 - 0.5 * np.cos(2.0 * np.pi * x)   # standard Hann window
    for p in np.linspace(0.2, 2.0, 20):
        if time.time() >= deadline:
            break
        a = np.power(base, p)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        try_seq(a)

    # 6. Mixed two‑Gaussian bumps (symmetric)
    for sigma in np.linspace(n / 32.0, n / 8.0, 12):
        if time.time() >= deadline:
            break
        shift = n // 4
        g1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - shift) / sigma) ** 2)
        g2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - (n - 1 - shift)) / sigma) ** 2)
        a = g1 + g2
        a = np.maximum(a, 0.0)
        a /= a.sum()
        try_seq(a)

    return best_seq, best_score

def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Draw many Dirichlet vectors (different concentration) and keep the best."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration parameter c in [10⁻³, 5] (log‑uniform)
        c = 10 ** rng.uniform(-3.0, 0.7)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on smooth max
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  beta_max: float,
                  lr: float,
                  deadline: float):
    """
    Adam optimisation of the smooth‑max objective.
    β grows exponentially from 0.5 to beta_max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # exponential schedule for β (log‑space)
    beta_start = 0.5
    log_beta_start = np.log(beta_start)
    log_beta_end = np.log(beta_max)

    while time.time() < deadline:
        it += 1
        # current β based on elapsed fraction of the remaining budget
        fraction = (time.time() - start_time) / (deadline - start_time)
        beta = np.exp(log_beta_start + fraction * (log_beta_end - log_beta_start))

        conv = _conv_self(a)
        grad = _smooth_gradient(a, conv, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = _project_simplex(np.maximum(a_candidate, 0.0))

        cand_score = _objective_score(a_candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter {it} new best {best_score:.6f}")

        # simple acceptance rule (always accept if improvement)
        if cand_score <= _objective_score(a) or np.random.rand() < 0.01:
            a = a_candidate

    return best_a, best_score

# Simple hill‑climbing with local smoothing moves
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                score0: float,
                deadline: float):
    """Tiny random mass‑transfer moves plus occasional local averaging."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1
        # random mass transfer
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        delta = a[src] * rng.random() * 0.05   # move up to 5 % of source
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}")
            continue
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score
            continue

        # every 2000 iterations try a tiny smoothing move
        if iter_cnt % 2000 == 0:
            smooth = (a +
                      np.roll(a, 1) +
                      np.roll(a, -1)) / 3.0
            smooth = _project_simplex(np.maximum(smooth, 0.0))
            smooth_score = _objective_score(smooth)
            if smooth_score < best_score:
                best_score = smooth_score
                best_a = smooth.copy()
                a = smooth
                cur_score = smooth_score
                print(f"[{time.time() - start_time:.2f}s] smoothing move improved to {best_score:.6f}")

    return best_a, best_score

        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
