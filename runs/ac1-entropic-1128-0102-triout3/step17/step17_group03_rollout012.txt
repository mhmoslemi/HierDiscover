        vals = rng.exponential(scale=1.0, size=k)
        a = np.zeros(n); a[idx] = vals; a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
    
    # Build Fourier basis
    K = 12  # number of harmonic pairs
    i = np.arange(n)
    basis = []
    for k in range(1, K+1):
        basis.append(np.cos(2*np.pi * k * i / n))
        basis.append(np.sin(2*np.pi * k * i / n))
    basis = np.array(basis)  # shape (2*K, n)
    d = basis.shape[0]

    # Compute initial w via least squares onto best_seq - uniform
    uniform = np.full(n, 1.0/n, dtype=float)
    diff = best_seq - uniform
    # Solve least squares: minimize ||basis.T @ w - diff||^2
    # Equivalent to w = (basis @ basis.T)^{-1} @ (basis @ diff)
    # Use np.linalg.lstsq for better stability
    w, *_ = np.linalg.lstsq(basis.T, diff, rcond=None)

    # Ensure w is a 1D array with shape (d,)
    w = w.astype(float)

    # Gradient descent loop
    step = 0.1
    while time.time() < deadline:
        # compute candidate a from current w
        a_raw = uniform + basis.T @ w
        a = _project_simplex(a_raw)
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time() - start_time:.2f}s] New best via basis {best_score:.6f}")
        # compute gradient wrt a
        grad_a = _subgradient_maxconv(a, conv)  # shape (n,)
        # compute gradient wrt w
        grad_w = basis @ grad_a  # shape (d,) (note sign? want to reduce max, so direction = -grad_w)
        # If grad_w is zero (unlikely), break
        # Perform line search
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            step_try = step * factor
            w_new = w - step_try * grad_w
            # optional random jitter: add small noise occasionally
            a_raw_new = uniform + basis.T @ w_new
            a_new = _project_simplex(a_raw_new)
            new_score = _objective_score(a_new)
            if new_score < cur_score - 1e-12:

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:

    else:
        best_seq = uniform

    return [float(x) for x in best_seq]

"""
Improved optimisation of the step‑function objective

F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

Key ideas
---------
* Start from many cheap random Dirichlet draws (including a very sparse one) and keep
  the best feasible vector (sum = 1) as a warm‑start.
* Perform a fast “aggressive” coordinate descent:
   – compute the sub‑gradient of the active (maximal) convolution entries,
   – pick the index with largest gradient as source and the smallest as destination,
   – estimate a promising transfer size using the linearised effect on the maximal
     entry and back‑track (halving) until an improvement is obtained.
   – after each accepted move recompute the gradient.  This deterministic move
     reduces the maximal convolution far quicker than the small random moves used
     in the previous solution.
* Occasionally inject a tiny random perturbation to escape possible plateaux.
* Finish with a cheap random local‑refinement (pairwise transfers) for polishing.

All operations respect the non‑negativity and normalisation constraints,
and the total runtime respects the 1000 s wall‑time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, power‑of‑2 length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sequence is kept on the simplex → denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at ``a`` (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]          # contribution of the second factor
    if active.size:
        grad = 2.0 * grad / active.size   # factor 2 for ordered pairs, then average
    return grad


# ----------------------------------------------------------------------
# Random initialisers (Dirichlet + a very sparse one)
# ----------------------------------------------------------------------
def _random_dirichlet_initialisers(n: int, rng: np.random.Generator,
                                   deadline: float, max_trials: int = 300):
    """Return the best vector seen among a batch of Dirichlet draws."""
    best = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in [10⁻⁴, 10⁰·⁴]  (very spiky → near‑uniform)
        conc_log = rng.uniform(-4.0, 0.4)
        conc = 10.0 ** conc_log
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
    return best, best_score


def _sparse_initialiser(n: int, sparsity: float, rng: np.random.Generator) -> np.ndarray:
    """Create a vector with roughly ``sparsity``·n non‑zero entries."""
    k = max(2, int(sparsity * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Aggressive deterministic coordinate descent
# ----------------------------------------------------------------------
def _aggressive_descent(start_a: np.ndarray,
                        start_score: float,
                        deadline: float,
                        rng: np.random.Generator) -> (np.ndarray, float):
    """
    Fast coordinate descent that repeatedly transfers mass from the index with
    the largest sub‑gradient to the one with the smallest, using a line‑search
    based on the estimated reduction of the currently maximal convolution entry.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = start_score
    no_improve = 0

    while time.time() < deadline:
        # current convolution & maximal entry
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        # store best found so far
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] New best (aggressive) {best_score:.6f}")
        else:
            no_improve += 1

        # compute sub‑gradient
        grad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # If src and dst are the same the gradient is flat – inject noise and continue
        if src == dst or a[src] <= 0.0:
            # tiny random jitter of the whole vector
            a += rng.normal(scale=1e-4, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            continue

        # linear estimate of the reduction we can obtain
        diff = grad[src] - grad[dst]          # positive if moving reduces the max
        if diff <= 0.0:
            # cannot improve deterministically – make a random small move
            delta = a[src] * rng.random() * 0.01
        else:
            # second‑largest convolution entry (target after reduction)
            tol = 1e-12
            mask = conv < conv.max() - tol
            if np.any(mask):
                second_best = conv[mask].max()
            else:
                second_best = conv.max() - 1e-9
            needed = conv.max() - second_best   # how much we need to lower the peak
            delta_target = needed / diff        # linear estimate
            # move at most the whole source mass, but stay a bit conservative
            delta = min(a[src], max(delta_target * 0.9, 0.0))
            if delta <= 0.0:
                delta = a[src] * 0.1

        # back‑track line‑search (halving) until an improvement is observed
        accepted = False
        for _ in range(30):
            if delta <= 0.0:
                break
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
            # safety: numerical overflow may create a tiny negative entry
            if cand[src] < -1e-15:
                delta *= 0.5
                continue
            cand_score = _objective_score(cand)
            if cand_score < cur_score - 1e-15:
                a = cand
                accepted = True
                break
            delta *= 0.5

        if not accepted:
            # if nothing helped for a while, perform a tiny random perturbation
            if no_improve > 5000:
                a += rng.normal(scale=1e-4, size=n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                no_improve = 0
            else:
                # simple random pair transfer (much smaller) – always accepted only if it helps
                s = rng.integers(n)
                d = rng.integers(n)
                while d == s or a[s] <= 0.0:
                    d = rng.integers(n)
                delta = a[s] * rng.random() * 0.005
                cand = a.copy()
                cand[s] -= delta
                cand[d] += delta
                cand_score = _objective_score(cand)
                if cand_score < cur_score:
                    a = cand
                    # no need to set accepted flag – we already have a better state

    return best_a, best_score


# ----------------------------------------------------------------------
# Final cheap random local refinement
# ----------------------------------------------------------------------
def _random_local_refinement(a0: np.ndarray,
                             cur_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> (np.ndarray, float):
    """Very cheap random pairwise transfers (strict improvements only)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        delta = a[src] * rng.random() * 0.005
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score - 1e-12:
            a = cand
            best_score = cand_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function (probability vector) minimising
    evaluate_sequence().  The routine uses:
      • diversified random Dirichlet initialisation,
      • a fast aggressive coordinate descent,
      • a tiny random polishing phase.
    It respects the wall‑time limit and returns the best sequence found.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep the same dimension as the reference solution.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Initial uniformly distributed vector (baseline)
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=float)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet start (≈ 20 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 20.0)
    if time.time() < dirichlet_deadline:
        cand_seq, cand_score = _random_dirichlet_initialisers(
            n, rng, dirichlet_deadline, max_trials=300
        )
        if cand_seq is not None and cand_score < best_score:
            best_seq, best_score = cand_seq, cand_score
            print(f"[{time.time() - start_time:.2f}s] Dirichlet start best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1b️⃣  Very sparse start (helpful for exploring different basins)
    # ------------------------------------------------------------------
    if time.time() < dirichlet_deadline:
        sparse = _sparse_initialiser(n, sparsity=0.015, rng=rng)   # ~1.5 % non‑zeros
        sc = _objective_score(sparse)
        if sc < best_score:
            best_seq, best_score = sparse, sc
            print(f"[{time.time() - start_time:.2f}s] Sparse start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Aggressive coordinate descent (the heavy lifting)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting aggressive descent")
        best_seq, best_score = _aggressive_descent(best_seq, best_score,
                                                   deadline=deadline,
                                                   rng=rng)
        print(f"[{time.time() - start_time:.2f}s] Aggressive descent finished – best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final random local polishing (≈ 30 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polishing_deadline = min(deadline, time.time() + 30.0)
        print(f"[{time.time() - start_time:.2f}s] Starting final polishing")
        best_seq, best_score = _random_local_refinement(best_seq,
                                                        best_score,
                                                        polishing_deadline,
                                                        rng=rng)
        print(f"[{time.time() - start_time:.2f}s] Polishing finished – best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety normalisation before returning
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
