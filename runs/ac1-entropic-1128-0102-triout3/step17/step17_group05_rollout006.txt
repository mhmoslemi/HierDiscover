"""
Propose a non‑negative step‑function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search proceeds in several phases under the 1000 s budget:

1️⃣  Fast parametric sweeps (geometric, power, exponential, two‑block) to
    obtain a decent starting shape.

2️⃣  Random Dirichlet draws and a *spectral* construction (random Fourier
    phases, absolute value, projection onto the simplex) to explore the
    simplex more diversely.

3️⃣  A simulated‑annealing style perturbation loop.  The current best
    sequence is randomly jittered (Gaussian noise whose amplitude shrinks
    with the remaining time), projected onto the simplex and kept if it
    improves the objective.  After every improvement a short run of the
    smooth‑max gradient descent (see 4) is performed to fall into a nearby
    local optimum.

4️⃣  Smooth‑max gradient descent on the simplex.  The non‑smooth `max`
    in the objective is replaced by a τ·log‑sum‑exp surrogate, yielding a
    differentiable gradient.  A simple back‑tracking line‑search guarantees
    descent, the iterate is projected back onto the probability simplex,
    and τ is geometrically cooled.  The routine also keeps track of the
    overall best sequence seen.

The whole pipeline finishes by normalising the best vector to sum = 1 and
returning it as a plain Python list.  All heavy work is performed with
FFT‑based convolutions (O(n log n)).  The implementation uses only NumPy
and the standard library, runs on a single CPU and respects the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a fast FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# 1️⃣  Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo = geo / s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq = seq / s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet / spectral generators
# ----------------------------------------------------------------------
def _rand_dirichlet(n, rng):
    """Draw a random point on the simplex using a Gamma Dirichlet."""
    logc = rng.uniform(-2.0, 0.7)          # concentration between ≈0.01 and 5
    c = 10 ** logc
    a = rng.gamma(shape=c, scale=1.0, size=n)
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s


def _rand_spectral(n, rng):
    """
    Random Fourier‑phase construction.
    Generate a length‑L real signal with flat magnitude spectrum,
    take its absolute value, truncate to n entries and normalise.
    """
    L = _next_pow2(2 * n)          # enough zero‑padding for a smooth signal
    # rfft length = L//2 + 1
    freqs = L // 2 + 1
    phases = rng.uniform(0.0, 2.0 * np.pi, size=freqs)
    # unit magnitude for all non‑DC components
    mags = np.ones_like(phases, dtype=np.complex128)
    mags[0] = 0.0                    # start with zero DC – the absolute step will introduce a bias later
    A = mags * np.exp(1j * phases)
    a = np.fft.irfft(A, n=L)
    a = np.abs(a)                   # enforce non‑negativity
    seq = a[:n]
    s = seq.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return seq / s

# 3️⃣  Smooth‑max gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(a0,
                                 deadline,
                                 best_seq,
                                 best_score,
                                 max_iters=None,
                                 init_factor=0.5,
                                 min_tau=1e-9,
                                 decay=0.998):
    """
    Projected gradient descent on τ·log‑sum‑exp(conv/τ).
    `init_factor` controls the initial temperature (τ ≈ init_factor·max_conv).
    The routine updates the global best sequence whenever a new record is hit.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0

    # initial temperature
    cur_conv = _conv_self(a)
    tau = max(cur_conv.max() * init_factor, 1e-6)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # cooling schedule (first iteration keeps the initial τ)
        if it > 1:
            tau = max(tau * decay, min_tau)

        # soft‑max weights over the convolution entries
        shift = max_conv
        log_w = (conv - shift) / tau
        # subtraction of the shift already stabilises the exponentials
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerically degenerate – reset temperature and continue
            tau = max(max_conv * init_factor, 1e-6)
            continue
        p = w / w_sum

        # gradient of the smooth‑max (see analysis)
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]
        grad = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # line‑search on the true objective
        cur_obj = 2.0 * n * max_conv
        step = 1.0
        improved = False
        while step > 1e-14:
            a_new = a - step * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-12:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] GD it={it:,} obj={obj_new:.6f} τ={tau:.2e}")
                break
            step *= 0.5

        if not improved:
            # tiny random jitter to escape flat regions
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

    return best_seq, best_score

            init_factor=0.5,
            decay=0.998,
        )
        print(f"[{time.time() - start_time:.2f}s] End of final descent, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
