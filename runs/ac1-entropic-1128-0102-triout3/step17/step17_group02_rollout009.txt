```
def _beta_shape(alpha, beta, n):
    i = np.arange(n)
    # midpoints for better sampling
    x = (i + 0.5) / n
    a = np.power(x, alpha-1) * np.power(1-x, beta-1)
    # replace nan/infinite values: when alpha<1 and i=0, x**(alpha-1) with 0**negative => inf. Use x+eps.
    eps = 1e-12
    a = np.power(np.clip(x, eps, 1-eps), alpha-1) * np.power(np.clip(1-x, eps, 1-eps), beta-1)
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0/n)
    a /= a_sum
    return a
```

The objective to minimise is  

\[
J(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum a_i\bigr)^2},
\]

where `*` denotes the linear convolution and all entries of the vector `a`
must be non‑negative.  Because the objective is homogeneous we may restrict
ourselves to probability vectors (`∑a_i = 1`).  The problem is then

\[
\min_{a\ge 0,\;\sum a_i = 1}\; 2\,n\;\max\bigl(a*a\bigr).
\]

The code below builds a candidate in four stages, each designed to explore a
different part of the search space while respecting the 1000 s time limit.

1. **Low‑dimensional parametric sweep** –  
   a symmetric Beta density (including the very spiky arcsine case) is sampled
   on a grid of shape parameters `α=β`.  These 1‑D families already give
   considerably better peaks than the uniform distribution.

2. **Random Dirichlet starts** –  
   a large number of random probability vectors are drawn (concentration
   parameter logarithmically sampled).  The best of them is kept as a second
   seed.

3. **Smooth Adam‑descent** –  
   a log‑sum‑exp smooth approximation of the max is used so that the gradient
   is cheap to compute.  An Adam optimiser (with projection onto the simplex)
   is run for a schedule of increasing smoothness parameters `β`.  Several
   restarts (tiny jitter around the current best) increase the chance of
   escaping local minima.

4. **Greedy hill‑climbing** –  
   the shift `p` where the current convolution attains its maximum is found.
   The gradient of the *exact* max (which is `2·a[p‑i]`) tells which indices
   contribute most to the peak.  Mass is transferred from high‑contribution
   indices to low‑contribution ones; the move is kept if it reduces the
   objective.  This very cheap local search finishes the optimisation.

All heavy convolutions are performed with an FFT of power‑of‑2 length, and the
projection onto the probability simplex uses the linear‑time algorithm of
Duchi et al.  Timing checks guarantee that the function always returns before
the supplied budget expires.

```python
"""
step‑function optimiser

* parametric Beta sweep (α=β) – fast low‑dimensional search
* many random Dirichlet draws
* smooth Adam descent on the log‑sum‑exp approximation of max(conv)
* greedy mass‑transfer hill climbing based on the exact max‑gradient

All vectors are kept on the probability simplex (non‑negative, sum = 1).
The code respects the 1000 s time budget and returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing variable (used for progress messages)
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using an FFT of power‑of‑2 length."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv) with temperature β."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Uses soft‑max weights over the convolution entries.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()                      # shape (2·n‑1,)
    # cross‑correlation a ⋆ w  (= convolution of a_rev with w)
    rev_corr = _convolve_fft(a[::-1], w)       # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Parametric Beta shape
# ----------------------------------------------------------------------
def _beta_shape(alpha: float, beta_param: float, n: int) -> np.ndarray:
    """
    Discretised symmetric (or asymmetric) Beta density on n points.
    Returns a probability vector (sum = 1, non‑negative).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                # mid‑point rule
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)   # avoid 0^neg
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta_param - 1.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Random Dirichlet initialisation
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration on [10⁻³, 10⁰·⁵]  (≈[0.001, 5])
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimisation on the smooth max
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  score0: float,
                  beta_vals,
                  deadline: float):
    """Adam optimiser on the smooth max, with simplex projection."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    lr_base = 0.05                # base learning rate

    for beta in beta_vals:
        # at most 150 updates per β (fast enough and still gives progress)
        for _ in range(150):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)          # d/d a of smooth max
            # objective = 2·n·smooth_max  → scale gradient accordingly
            grad = 2.0 * a.shape[0] * grad

            # Adam moments
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # update (vectorised Adam step)
            update = lr_base * m_hat / (np.sqrt(v_hat) + eps)
            cand = a - update
            cand = _project_simplex(np.maximum(cand, 0.0))

            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            # stop early if the step becomes negligible
            if np.max(np.abs(update)) < 1e-12:
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climbing using the exact max‑gradient
# ----------------------------------------------------------------------
def _greedy_hill_climb(a0: np.ndarray,
                       score0: float,
                       deadline: float):
    """
    Mass‑transfer moves guided by the exact max‑gradient.
    Source indices are chosen proportionally to their contribution to the peak;
    destination indices are chosen where the contribution is small.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a)
        p_max = int(conv.argmax())                # shift with the current peak

        # gradient of conv[p_max] w.r.t. a: g[i] = 2 * a[p_max - i] (if index valid)
        g = np.zeros_like(a)
        i_min = max(0, p_max - (n - 1))
        i_max = min(n - 1, p_max)
        idx = np.arange(i_min, i_max + 1)
        g[idx] = 2.0 * a[p_max - idx]

        # contribution of each index to the peak (product a[i] * a[p_max-i])
        contrib = a * g                     # non‑negative
        if contrib.sum() == 0.0:
            src = rng.integers(n)
        else:
            probs_src = contrib / contrib.sum()
            src = rng.choice(n, p=probs_src)

        # destination – preferably where the gradient is tiny
        inv_g = 1.0 / (g + 1e-12)
        probs_dst = inv_g / inv_g.sum()
        dst = rng.choice(n, p=probs_dst)

        if src == dst or a[src] == 0.0:
            continue

        # move a small random fraction of the source mass (≤5 %)
        delta = a[src] * rng.random() * 0.05
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] greedy hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            # occasional downhill move
            a = cand
            cur_score = cand_score
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain list of floats) discovered within the time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between expressiveness and speed
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] start (uniform) score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1) if present
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – deterministic Beta sweep (≈30 s)
    # --------------------------------------------------------------
    sweep_deadline = min(deadline, start_time + 30.0)
    alphas = np.concatenate((
        np.linspace(0.2, 2.0, 20),
        np.linspace(2.0, 5.0, 10)
    ))
    for alpha in alphas:
        if time.time() >= sweep_deadline:
            break
        a = _beta_shape(alpha, alpha, n)                # symmetric Beta(α,α)
        sc = _objective_score(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Beta α={alpha:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – random Dirichlet draws (≈120 s)
    # --------------------------------------------------------------
    dir_deadline = min(deadline, start_time + 150.0)
    if time.time() < dir_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=3000, deadline=dir_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – smooth Adam descent with β‑schedule
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.80   # most of what is left
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                         50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0]

        restarts = 6
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.02, size=n)          # modest random perturbation
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0_score = _objective_score(a0)

            # allocate a shrinking slice of the remaining time for each restart
            sub_deadline = min(descent_deadline,
                               time.time() + (descent_deadline - time.time()) / (restarts - r))

            a_opt, sc_opt = _adam_descent(a0, a0_score, beta_schedule, sub_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] restart {r} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – greedy hill‑climbing (final seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting greedy hill‑climb")
        best_seq, best_score = _greedy_hill_climb(best_seq, best_score, deadline)

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
