def _dirichlet_random(n, rng):
    conc_log = rng.uniform(-3.0, 0.7)
    c = 10 ** conc_log
    a = rng.gamma(shape=c, scale=1.0, size=n)
    a_sum = a.sum()
    if a_sum==0: return uniform

    a /= a_sum
    return a

def _sparse_random(n, ratio, rng):
    k = max(2, int(ratio * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n)
    a[idx] = vals
    a /= a.sum()
    return a

def _adam_smooth_optimize(start_vec, best_score_initial, deadline, rng):

    a = start_vec.copy()
    n = a.shape[0]
    # initial best record
    best_a = a.copy()
    best_score = best_score_initial

    # Adam parameters
    lr = 0.5

    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothness param
    alpha = 2.0
    alpha_max = 5000.0
    alpha_factor = 1.005  # slower growth

    while time.time() < deadline:

        smooth_max, grad, conv = _smoothmax_and_grad(a, alpha)

        # compute actual objective
        actual_max = conv.max()

    return best_a, best_score

"""
Propose a step‑function (probability vector) that minimizes

    F(a) = 2·n·max(conv(a,a)) / (sum(a))²

with all entries non‑negative.  The algorithm uses a deterministic
smooth‑max approximation of the L∞‑norm together with an Adam optimiser
and a simple projection onto the probability simplex.  Random
initialisations (Dirichlet draws and sparse starts) are repeatedly
optimised; the best vector found within the 1000 s budget is returned.
A final cheap random local polishing step is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing (filled in propose_candidate)
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # should not happen, but fall back to uniform
        w = np.full_like(v, 1.0 / n)
    return w


def _correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Return the *valid* (non‑zero) part of the cross‑correlation
    of w and a, i.e.  c[i] = Σ_j w[i+j]·a[j]  for i=0..n‑1.
    Implemented via FFT for speed.
    """
    n = a.shape[0]
    Lw = w.shape[0]                     # = 2·n‑1
    N = _next_pow2(Lw + n - 1)          # length for FFT
    Wf = np.fft.rfft(w, n=N)
    Af_rev = np.fft.rfft(a[::-1], n=N)
    corr_full = np.fft.irfft(Wf * Af_rev, n=N)
    start = n - 1                        # first index where a fully overlaps w
    return corr_full[start: start + n]   # length n


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smoothmax_and_grad(a: np.ndarray, alpha: float):
    """
    Smooth approximation of max(conv) together with its gradient.
    Returns (smooth_max, grad, full_conv) where grad is the (sub)gradient
    of the smoothed objective w.r.t. a (ignoring the constant factor 2·n).
    """
    n = a.shape[0]

    # full self‑convolution
    conv = _conv_self(a)

    # soft‑max weights over the convolution entries
    max_val = conv.max()
    shifted = conv - max_val                 # all ≤ 0
    exp_alpha = np.exp(alpha * shifted)     # stability ensured
    sum_exp = exp_alpha.sum()
    w = exp_alpha / sum_exp                 # Σ w = 1, w ≥ 0

    # smooth max = weighted average (identical to log‑sum‑exp up to a constant)
    smooth_max = np.dot(w, conv)

    # gradient of the smoothed max:
    #   g_i = 2· Σ_k w_k a_{k-i}  -  w_{2i}·a_i   (the diagonal term appears only once)
    corr = _correlate_valid(w, a)           # length n, = Σ_k w_{i+k}·a_k
    w_diag = w[2 * np.arange(n)]            # elements of w on even indices 0,2,4,...
    grad = 2.0 * corr - w_diag * a

    return smooth_max, grad, conv


# ----------------------------------------------------------------------
# Optimisation core (Adam on the smooth‑max objective)
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start_vec: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> (np.ndarray, float):
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector (according to the true objective) found before the deadline.
    """
    a = start_vec.copy()
    n = a.shape[0]

    # best‑so‑far (according to the real objective)
    best_a = a.copy()
    best_score = start_score

    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smooth‑max schedule
    alpha = 2.0          # start very smooth
    alpha_max = 5000.0
    alpha_factor = 1.004

    while time.time() < deadline:
        # compute smooth max and its gradient
        smooth, grad, conv = _smoothmax_and_grad(a, alpha)

        # evaluate the true objective (no smoothing)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Adam new best {best_score:.6f}   (α={alpha:.1f})")

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a + step
        # keep feasibility (non‑negativity+simplex)
        a = _project_simplex(np.maximum(a, 0.0))

        # anneal the smoothness
        if alpha < alpha_max:
            alpha = min(alpha * alpha_factor, alpha_max)

    return best_a, best_score

# Tiny random local refinement (polishing)
# ----------------------------------------------------------------------
def _random_local_refinement(a0: np.ndarray,
                             cur_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> (np.ndarray, float):
    """Cheap pairwise mass transfers that accept only strict improvements."""
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        # move a tiny fruit‑fly fraction
        delta = a[src] * rng.random() * 0.004
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_score:.6f}")

    return best_a, best_score
