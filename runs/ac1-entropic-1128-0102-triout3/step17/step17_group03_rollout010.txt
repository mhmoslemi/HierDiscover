def _grad_smooth(a, softmax):

    n = a.shape[0]
    rev_a = a[::-1]
    # Convolution length: len(softmax) + n -1 = (2n-1) + n -1 = 3n-2

def _smooth_objective_and_gradient(a, eps):
    # Input a (np.ndarray length n)
    # Output smoothed objective f_eps and gradient (np.ndarray length n)
    conv = _conv_self(a) # length 2n-1
    max_conv = conv.max()
    # compute softmax over conv, with stability shift
    shifted = conv - max_conv
    exp_vals = np.exp(eps * shifted)
    sum_exp = exp_vals.sum()
    softmax = exp_vals / sum_exp
    # smooth objective: 2*n * (log(sum_exp)/eps + max_conv)
    # Because log-sum-exp = max_conv + log(sum_exp)/eps.
    smooth_val = 2.0 * a.shape[0] * (max_conv + np.log(sum_exp) / eps)
    # compute gradient: 2 * correlation a ⋆ softmax
    grad = _gradient_correlation(a, softmax) # shape n
    return smooth_val, grad

def _gradient_correlation(a, w):
    # compute grad = 2 * (a ⋆ w) [size n]
    n = a.shape[0]
    rev_a = a[::-1]
    # convolution length = len(w)+n-1
    L = _next_pow2(len(w) + n - 1)
    Sf = np.fft.rfft(w, n=L)
    Af = np.fft.rfft(rev_a, n=L)
    conv = np.fft.irfft(Sf * Af, n=L)  # length L
    # conv length is at least len(w)+n-1; we only need first len(w)+n-1
    conv = conv[: len(w) + n - 1]
    # slice middle part
    start = n - 1
    grad = 2.0 * conv[start : start + n]
    return grad

"""
Smooth‐gradient optimisation of the step‑function objective.

The objective (with a normalised sequence Σa_i = 1) is

    F(a) = 2·n·max_k (a∗a)[k]                     (1)

which is non‑differentiable because of the max.  We replace the max by a
log‑sum‑exp smoothing

    F_ε(a) = 2·n·( log Σ_k e^{ε·(a∗a)[k]} ) / ε      (2)

which converges to (1) for ε → ∞.  The gradient of (2) is smooth and can be
computed efficiently with a single FFT–based correlation.  A projected‐gradient
descent with an adaptive step size is performed for a sequence of increasing
ε (2,4,…,128).  Between the ε‑phases we add a tiny random perturbation to help
escape shallow basins.  Finally a cheap pairwise‑transfer refinement is run.
The whole search respects the 1000 s wall‑time limit and returns the best
vector found (as a plain list of floats).

All helper functions are top‑level; no closures or file I/O are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing (set in propose_candidate)
# ----------------------------------------------------------------------
start_time = None  # will be initialised by propose_candidate


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σv=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective utilities (the vector is kept on the simplex, Σa = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised a."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_objective_and_grad(a: np.ndarray, eps: float):
    """
    Smooth surrogate (2) and its gradient.

    Returns
        smooth_val : float
        grad       : np.ndarray, same shape as a
    """
    n = a.shape[0]
    # 1) convolution a*a
    conv = _conv_self(a)                     # length 2n‑1
    max_conv = conv.max()

    # 2) soft‑max over the convolution entries
    shifted = conv - max_conv                # for numerical stability
    exp_vals = np.exp(eps * shifted)
    sum_exp = exp_vals.sum()
    softmax = exp_vals / sum_exp

    # smooth objective value (2)
    smooth_val = 2.0 * n * (max_conv + np.log(sum_exp) / eps)

    # 3) gradient:  ∇F_ε = 4·n·corr(a, softmax)
    #    correlation (a ⋆ softmax)[i] = Σ_j a_j·softmax_{i+j}
    rev_a = a[::-1]                          # reversed a
    L = _next_pow2(len(softmax) + n - 1)
    Af = np.fft.rfft(rev_a, n=L)
    Sf = np.fft.rfft(softmax, n=L)
    corr_full = np.fft.irfft(Af * Sf, n=L)[: len(softmax) + n - 1]   # length 3n‑2
    core = corr_full[n - 1 : n - 1 + n]          # the required n entries
    grad = 4.0 * n * core                       # full gradient of the smooth objective
    return smooth_val, grad


# ----------------------------------------------------------------------
# Smooth projected gradient descent for a fixed smoothing parameter ε
# ----------------------------------------------------------------------
def _smooth_pgd(start_a: np.ndarray,
                eps: float,
                sub_deadline: float,
                rng: np.random.Generator):
    """
    Projected gradient descent on the smooth surrogate (2) for a given ε.

    Returns the best vector found (on the simplex) and its exact objective.
    """
    a = start_a.copy()
    smooth_val, grad = _smooth_objective_and_grad(a, eps)
    exact_best = a.copy()
    exact_best_score = _objective_score(a)

    # step size expressed as a fraction of (4·n) – see the comment in the
    # docstring above.  Start with a modest value and adapt.
    n = a.shape[0]
    step = 0.5 / (4.0 * n)      # about 6·10⁻⁶ for n=4096

    it = 0
    while time.time() < sub_deadline:
        it += 1

        # candidate step
        cand = a - step * grad
        cand = _project_simplex(np.maximum(cand, 0.0))

        cand_smooth, cand_grad = _smooth_objective_and_grad(cand, eps)

        # accept only if the smooth objective improves
        if cand_smooth < smooth_val:
            a = cand
            grad = cand_grad
            smooth_val = cand_smooth
            step = min(step * 1.05, 0.5)          # slowly relax the step
        else:
            step *= 0.5                           # back‑off
            if step < 1e-12:
                break

        # every 200 iterations evaluate the true (non‑smoothed) objective
        if it % 200 == 0:
            cur_score = _objective_score(a)
            if cur_score < exact_best_score:
                exact_best_score = cur_score
                exact_best = a.copy()
                print(f"[{time.time() - start_time:.2f}s] eps={eps} new best {exact_best_score:.6f}")

    return exact_best, exact_best_score


# ----------------------------------------------------------------------
# Cheap random local refinement (identical to the one used previously)
# ----------------------------------------------------------------------
def _random_local_refinement(a0: np.ndarray,
                             cur_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> (np.ndarray, float):
    """Very cheap pairwise‑transfer moves, accepting only strict improvements."""
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        delta = a[src] * rng.random() * 0.005   # tiny transfer
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (probability vector) that minimises
    ``evaluate_sequence``.  The algorithm uses a continuation of smooth
    projected gradient descent with several ε‑levels, occasional random
    perturbations and a final cheap local refinement.  It respects the wall‑time
    budget and returns the best vector found (as a plain list of floats).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1      # tiny safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep the same dimension used by the reference code.
    # ------------------------------------------------------------------
    n = 4096

    # ----- start from the uniform distribution ---------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=float)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # ----- use the global starter if it exists ---------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter, sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    #  Smooth‑gradient phase (continuation over ε)
    # ------------------------------------------------------------------
    epsilons = [2, 4, 8, 16, 32, 64, 128]
    cur_a = best_seq.copy()

    for eps in epsilons:
        if time.time() >= deadline:
            break

        # allocate a modest slice of the remaining budget for this ε
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(remaining * 0.15, 30.0)   # ≤30 s per ε

        print(f"[{time.time() - start_time:.2f}s] ε={eps} phase (budget {(sub_deadline - time.time()):.1f}s)")

        cur_a, cur_best_score = _smooth_pgd(cur_a, eps, sub_deadline, rng)

        if cur_best_score < best_score:
            best_seq, best_score = cur_a.copy(), cur_best_score
            print(f"[{time.time() - start_time:.2f}s] After ε={eps} best = {best_score:.6f}")

        # small random perturbation before the next ε‑level (helps escape plateaus)
        perturb = rng.normal(scale=0.01, size=n)      # about 1 % noise
        cur_a = cur_a + perturb
        cur_a = np.clip(cur_a, 0.0, None)
        cur_a_sum = cur_a.sum()
        if cur_a_sum > 0.0:
            cur_a /= cur_a_sum
        else:
            cur_a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    #  Final cheap random local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = min(deadline, time.time() + 30.0)
        print(f"[{time.time() - start_time:.2f}s] Starting final local refinement")
        best_seq, best_score = _random_local_refinement(best_seq, best_score,
                                                        local_deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] Local refinement finished – best = {best_score:.6f}")

    # ------------------------------------------------------------------
    #  Normalise and return (ensure strict feasibility for the evaluator)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {final_score:.6f}")
    return [float(x) for x in best_seq]
