def _lp_improvement(a, L_fft, deadline, start_time, rng):
    n = a.shape[0]
    # compute M
    conv = _conv_self_fft(a, L_fft)
    M = conv.max()
    # Build matrix C:
    offsets = np.arange(n)
    diag_data = [np.full(n, a[d]) for d in range(n)]
    C = sp.diags(diag_data, offsets, shape=(2*n-1, n), format='csc')
    # Upper bound vector
    ub = np.full(2*n-1, M, dtype=np.float64)
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    # bounds for g: (0, None)
    bounds = [(0, None) for _ in range(n)]
    # time left for LP
    time_left = deadline - time.time()
    # Use default solver; might take some seconds
    res = opt.linprog(c, A_ub=C, b_ub=ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return a, _eval_fast(a, L_fft)
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a, _eval_fast(a, L_fft)
    g_norm = g / sum_g
    # Evaluate g_norm alone
    score_g = _eval_fast(g_norm, L_fft)
    best_a = a
    best_score = _eval_fast(a, L_fft)
    # compute blend candidates
    for t in [0.2, 0.5, 0.8]:
        cand = (1 - t) * a + t * g_norm
        cand = _project_simplex(cand)  # ensure simplex
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_a = cand
            best_score = cand_score
    # also consider g_norm alone if better
    if score_g < best_score:
        best_a = g_norm
        best_score = score_g
    return best_a, best_score

"""
Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines several complementary phases:

1️⃣  A dense parametric sweep over geometric × power‑law families
    a_i ∝ (i+1)^p · r^i (r = exp(−λ)).
2️⃣  Projected sub‑gradient descent on the exact L∞ objective.
3️⃣  Multiplicative (exponentiated‑gradient) updates.
4️⃣  A single linear‑programming step that maximises the allowed total
    mass while keeping the convolution under the current worst value.
5️⃣  A simulated‑annealing hill‑climb that accepts occasional worsening
    moves to escape plateaus.

All phases respect a 1000 s time budget, work on the probability simplex,
and finally return the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – assumes the sum is ≈ 1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Family a_i ∝ (i+1)^p · r^i  (geometric × power)
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """
    Build a vector a_i ∝ (i+1)^p * exp(-lam*i)   (lam = -log r).
    The result is normalised to unit L1 norm.
    """
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)          # stabilise exp
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient for the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]               # includes the diagonal correctly
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (back‑tracking)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on max(conv) with simple back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.35
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        step = step0 / (it ** 0.55)          # decreasing step length
        improved = False
        for _ in range(10):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                      f"step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional gentle uphill to escape tiny plateaus
            prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (multiplicative updates)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) updates:
        a ← a * exp(- η * g)   ,   η annealed.
    After each update we renormalise to stay on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.8
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        eta = eta0 / (1.0 + 0.0008 * it)

        mult = np.exp(-eta * g)
        a = a * mult
        tot = a.sum()
        if tot == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / tot

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} "
                  f"score={score:.6f}")

        if it % 3000 == 0:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def _lp_improvement(a0: np.ndarray,
                    L_fft: int,
                    deadline: float,
                    start_time: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Solve   max 1ᵀg   s.t.  a * g ≤ M (element‑wise),  g ≥ 0.
    Blend the resulting distribution with the current one.
    """
    a = a0.copy()
    n = a.shape[0]

    # current worst convolution value
    conv = _conv_self_fft(a, L_fft)
    M = conv.max()

    # Build the (2n‑1) × n constraint matrix C where (C g)_k = (a * g)_k
    offsets = np.arange(n, dtype=np.int64)        # 0 … n‑1
    # each diagonal = a[d] repeated n times
    diag_data = np.tile(a, (n, 1)).T              # shape (n, n)
    C = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csc')

    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None) for _ in range(n)]

    # run the LP (HiGHS is fast for this size)
    res = opt.linprog(c, A_ub=C, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    if not res.success:
        # fallback – keep the original vector
        return a, _eval_fast(a, L_fft)

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return a, _eval_fast(a, L_fft)

    g_norm = g / sum_g

    # try a few convex combinations of a and g_norm
    best_local = a
    best_score = _eval_fast(a, L_fft)
    for tau in (0.2, 0.5, 0.8):
        cand = (1.0 - tau) * a + tau * g_norm
        cand = _project_simplex(cand)
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best_local = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] LP blend τ={tau:.2f} score={cand_score:.6f}")

    # also test g_norm alone
    g_score = _eval_fast(g_norm, L_fft)
    if g_score < best_score:
        best_score = g_score
        best_local = g_norm.copy()
        print(f"[{time.time() - start_time:.2f}s] LP alone score={g_score:.6f}")

    return best_local, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final refinement)
# ----------------------------------------------------------------------


def _sa_hill_climb(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA moves that accept occasional worsening steps."""
    n = a0.shape[0]
    cur = a0.copy()
    cur_score = _eval_fast(cur, L_fft)
    best_seq = cur.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.integers(n)
        if cur[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # up to 5 % of the source mass is moved
        eps = cur[src] * rng.random() * 0.05
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score

        if delta < 0:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}")
        else:
            # linear temperature schedule
            T = max(0.0, (deadline - time.time()) / (deadline - start_time))
            prob = np.exp(-delta / (0.001 + 0.01 * T))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        if it % 2000 == 0:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur /= cur.sum()
            cur_score = _eval_fast(cur, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (L1‑normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # small safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 offers a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter (height_sequence_1) exists, try it (and resize if needed).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # simple linear interpolation to the required length
                x_old = np.linspace(0, 1, starter.size)
                x_new = np.linspace(0, 1, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0:
                starter = starter / total
                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep over (λ, p) – ≈70 s (or less if time runs out).
    # --------------------------------------------------------------
    sweep_deadline = start_time + 70.0
    min_lambda = 4e-05          # tiny exponential decay
    max_lambda = 1.5e-03        # moderate decay
    lam_grid = np.linspace(min_lambda, max_lambda, 300)   # 300 points
    p_grid = np.linspace(-6.0, 3.0, 91)                   # step 0.1

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * np.arange(n, dtype=np.float64)
        for p in p_grid:
            log_shape = base + p * np.log(np.arange(1, n + 1, dtype=np.float64))
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6f} p={p:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent (more intensive now that we have a good start).
    # --------------------------------------------------------------
    if time.time() < deadline - 350.0:                     # reserve time for later phases
        sub_deadline = deadline - 350.0
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, sub_deadline, start_time,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient phase.
    # --------------------------------------------------------------
    if time.time() < deadline - 250.0:
        exp_deadline = deadline - 250.0
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, exp_deadline, start_time,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 4️⃣  Linear‑programming improvement step.
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        lp_deadline = deadline - 150.0
        best_seq, best_score = _lp_improvement(
            best_seq, L_fft, lp_deadline, start_time, rng)

    # --------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing hill‑climb.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _sa_hill_climb(
            best_seq, L_fft, deadline, start_time, rng)

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
