def project_simplex(v):
    # v: np.ndarray
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimisation of a step‑function to minimise

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three phases:

* **Warm‑start** – try a few Dirichlet draws and, if present,
  the global variable ``height_sequence_1``.
* **Fine scanning of the Beta(p,p) family** – for many values of the
  shape parameter p and for several modest smoothing kernels we build
  candidates a = λ·smooth(Beta(p,p)) + (1‑λ)·uniform and keep the best.
  The scan is deliberately fine (Δp≈0.01, Δλ≈0.02) because the improvement
  target is tiny.
* **Gradient‑descent refinement** – starting from the best candidate,
  we minimise a smooth approximation of the ``max`` using a
  soft‑max (log‑sum‑exp) on the self‑convolution.  The gradient is obtained
  analytically via FFT‑based correlation, the iterate is projected onto the
  probability simplex and an adaptive step‑size/back‑tracking line search
  is used.  Occasionally the step size is reduced and a small random
  perturbation is added to escape possible plateaus.

All operations keep the vector non‑negative and normalised, so the
evaluator’s clipping does not affect the result.  The routine respects the
time budget (default 1000 s) and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(Af * Af, n=L_fft)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> tuple[float, np.ndarray]:
    """
    Exact score used by the evaluator (sum(a)=1 → score = 2·n·max(conv)).
    Returns (score, full convolution).
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    score = float(2.0 * n * max_b)   # sum(a) is 1 after normalisation
    return score, conv


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised symmetric Beta(p,p) density on the grid x_i=(i+0.5)/n.
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _smooth_shape(shape: np.ndarray, width: int) -> np.ndarray:
    """Convolve ``shape`` with a centred uniform kernel of size ``width``."""
    if width <= 1:
        return shape.copy()
    kernel = np.full(width, 1.0 / width, dtype=np.float64)
    smooth = np.convolve(shape, kernel, mode='same')
    smooth = np.maximum(smooth, 0.0)
    total = smooth.sum()
    if total == 0.0:
        return shape.copy()
    return smooth / total


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {w | w_i ≥ 0, Σ w_i = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # Quick return if already feasible
    if v.min() >= 0 and np.isclose(v.sum(), 1.0):
        return v

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive → uniform distribution
        return np.full(n, 1.0 / n)

    rho = ind[cond].max() - 1  # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------

            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand)
            a = cand
            eta = 0.2
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Random jitter injected")

        # safety break if step size collapsed
        if eta <= min_eta and beta >= 500.0:
            break

    # --------------------------------------------------------------
    # Normalisation & return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
