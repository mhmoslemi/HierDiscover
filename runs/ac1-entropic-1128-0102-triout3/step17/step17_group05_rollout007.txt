"""
Propose a low‑autocorrelation step function.

The search proceeds in four independent stages, each respecting the
global 1000 s budget.

1️⃣  Broad parametric families (geometric, two‑sided exponential,
    power law) are swept on a fine grid.
2️⃣  Hundreds of random Dirichlet draws explore the simplex.
3️⃣  A deterministic sub‑gradient descent on the true (non‑smooth)
    objective uses the exact gradient at the currently maximal
    convolution index and a back‑tracking line search.
4️⃣  A final simulated‑annealing hill‑climb makes tiny mass transfers
    between two positions; improvements are kept, otherwise moves are
    occasionally accepted to escape plateaus.

All candidates are projected onto the probability simplex after each
update, guaranteeing non‑negativity and unit sum.  The objective that
is minimised is

        2·n·max(conv(a,a))          (sum(a)=1)

which is exactly the value returned by ``evaluate_sequence`` after the
normalisation step.  The best sequence found is returned as a plain
list of Python floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Assumes sum(seq) == 1.  Returns 2·n·max(conv(seq,seq))."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Reference:  Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# 1️⃣ Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    """a_i ∝ r^i  (monotone geometric) with optional uniform mixing."""
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-6, -0.5, 120)          # r = 1 - eps
    lam_grid = np.linspace(0.0, 1.0, 11)          # mixing weight of uniform
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = max(0.0, 1.0 - eps)
        geo = np.power(r, idx)
        for lam in lam_grid:
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_sided_exponential(n, best_seq, best_score, deadline):
    """a_i ∝ exp(-λ·i) + exp(-λ·(n-1-i)), optionally mixed with uniform."""
    idx = np.arange(n, dtype=np.float64)
    lam_vals = np.logspace(-5, -0.3, 100)   # decay constants
    mix_grid = np.linspace(0.0, 1.0, 11)    # uniform mixing weight
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        left = np.exp(-lam * idx)
        right = np.exp(-lam * (n - 1 - idx))
        base = left + right
        for mix in mix_grid:
            seq = mix * np.full(n, 1.0 / n) + (1.0 - mix) * base
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoSideExp λ={lam:.1e} mix={mix:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    """a_i ∝ (i+1)^p  with optional uniform mixing."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    p_grid = np.linspace(-2.5, 2.5, 101)
    mix_grid = np.linspace(0.0, 1.0, 11)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx, p)
        for mix in mix_grid:
            seq = mix * np.full(n, 1.0 / n) + (1.0 - mix) * pow_seq
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} mix={mix:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the two ends (mass split w)."""
    for L_frac in (0.01, 0.02, 0.05, 0.1):
        L = max(1, int(L_frac * n))
        for w in (0.0, 0.2, 0.5, 0.8, 1.0):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=2500):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration between 1e‑2 and 5
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score

# 3️⃣ Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0, deadline, best_seq, best_score):
    """
    Deterministic sub‑gradient descent:
        - compute the index k where convolution is maximal,
        - gradient = 2 * a_rev shifted by k,
        - back‑tracking line‑search,
        - Euclidean projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        k = int(conv.argmax())                     # max‑convolution index
        # ---- gradient for this k ---------------------------------
        grad = np.zeros_like(a)
        if k < n:
            # overlap length = k+1
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            shift = k - (n - 1)                    # shift > 0
            grad[shift:] = 2.0 * a[:n - shift][::-1]

        cur_obj = 2.0 * n * max_val
        # ---- line search -----------------------------------------
        step = 1.0
        improved = False
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-15:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} obj={obj_new:.6f} step={step:.1e}")
                break
            step *= 0.5

        if not improved:
            # small random perturbation to escape flat regions
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

        # optional early stop when progress stalls
        if it % 200 == 0 and time.time() >= deadline:
            break
    return best_seq, best_score

# 4️⃣ Hill‑climbing simulated annealing
# ----------------------------------------------------------------------
def _hill_climb(start_seq, deadline, best_seq, best_score):
    """
    Tiny mass transfers with an annealed acceptance rule.
    """
    a = start_seq.copy()
    n = a.shape[0]
    cur_obj = _objective_score(a)
    it = 0
    # simple temperature schedule (linear decay)
    T0 = 1e-3 * cur_obj
    while time.time() < deadline:
        it += 1
        # pick source (large mass) and destination (any)
        src = np.random.choice(n, p=a / a.sum())
        if a[src] < 1e-12:
            continue
        dst = np.random.randint(0, n)
        if src == dst:
            continue
        # transfer a tiny fraction of mass
        delta = min(a[src], 1e-4) * np.random.rand()
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = _project_simplex(a_new)    # projection keeps sum = 1
        obj_new = _objective_score(a_new)

        accept = False
        if obj_new < cur_obj:
            accept = True
        else:
            # annealed acceptance
            elapsed = time.time() - start_time
            remaining = max(0.0, deadline - time.time())
            # temperature decays with remaining budget
            T = T0 * (remaining / (deadline - start_time))
            if T > 0 and np.random.rand() < np.exp(-(obj_new - cur_obj) / T):
                accept = True

        if accept:
            a = a_new
            cur_obj = obj_new
            if obj_new < best_score:
                best_score, best_seq = obj_new, a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} obj={obj_new:.6f}")

        # occasional aggressive jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = np.random.uniform(-1e-6, 1e-6, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
            cur_obj = _objective_score(a)

    return best_seq, best_score
