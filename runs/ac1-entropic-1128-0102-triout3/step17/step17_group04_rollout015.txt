def _conv_self_fft(a):
    n=len(a)
    L = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    return conv

Now exponent update:

def exp_update(a, grad, eta):
   log_factor = -eta * grad
   log_factor -= np.max(log_factor)  # shift
   factor = np.exp(log_factor)
   a_new = a * factor
   total = a_new.sum()
   if total <= 0 or not np.isfinite(total):
       return np.full_like(a, 1.0 / a.shape[0])
   a_new /= total
   return a_new

def subgrad_descent(a0, deadline, best_seq, best_score):
   a = a0.copy()
   n = a.shape[0]
   cur_score = objective(a)
   eta = 0.5
   min_eta = 1e-8
   no_improve_iters = 0
   while time.time() < deadline:
       conv = _conv_self_fft(a)
       max_val = conv.max()
       idx = conv.argmax()
       # gradient
       grad = np.zeros_like(a)
       low = max(0, idx - (n - 1))
       high = min(n - 1, idx)
       i_range = np.arange(low, high + 1)
       j_range = idx - i_range
       grad[i_range] = 2.0 * a[j_range]
       # update
       a_new = exp_update(a, grad, eta)
       new_score = objective(a_new)
       if new_score < best_score:
           best_score, best_seq = new_score, a_new.copy()
           print(f"[{time.time() - start_time:.2f}s] Subgrad improvement score={new_score:.6f}")
       # acceptance
       if new_score <= cur_score:
           a = a_new
           cur_score = new_score
           eta = min(1.0, eta * 1.01)
           no_improve_iters = 0
       else:
           eta = max(min_eta, eta * 0.5)
           no_improve_iters += 1
           if no_improve_iters % 50 == 0:
               # random jitter
               rng = np.random.default_rng()
               i = rng.integers(0, n)
               j = rng.integers(0, n)
               delta = 0.01 * a[i] * (rng.random() - 0.5)  # small random shift
               new_a_i = max(a[i] - delta, 0.0)
               a[i] = new_a_i
               a[j] += a[i] - new_a_i
               a /= a.sum()
               cur_score = objective(a)
   return best_seq, best_score

"""
Search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a mixture of:
  • cheap parametric sweeps,
  • many random Dirichlet / sparse initialisers,
  • a simple sub‑gradient (mirror) descent that directly targets the
    maximal convolution entry,
  • short random‑jitter hill‑climbing phases.

The routine is deliberately time‑aware: it stops well before the 1000 s
budget and always returns the best sequence found.  The implementation
uses FFT‑based convolutions for speed and works entirely on the
probability simplex (weights sum to 1, are ≥0).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Objective 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent (exponential) update:
        a_i ← a_i·exp(−η·grad_i)  ,   then renormalise to sum = 1.
    Guarantees a_i≥0.
    """
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)               # avoid overflow
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        # fall back to uniform distribution
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total


def _max_conv_gradient(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    For the shift `idx = argmax conv`, the gradient is
        ∂ (a*a)[idx] / ∂ a_i = 2·a_{idx-i}   (if the index exists, else 0).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    low = max(0, idx - (n - 1))
    high = min(n - 1, idx)
    i_range = np.arange(low, high + 1)
    j_range = idx - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         start_time: float):
    """
    Simple (exponential) mirror‑descent on the exact max‑convolution
    surrogate.  The routine runs until `deadline` and returns the best
    sequence encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    eta = 0.5                     # initial step size
    min_eta = 1e-8
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        idx = int(conv.argmax())
        grad = _max_conv_gradient(a, idx)

        a_new = _exp_update(a, grad, eta)
        new_score = _objective(a_new)

        if new_score < best_score:
            best_score, best_seq = new_score, a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Subgrad improve → score={new_score:.6f}")

        # accept if we made progress, otherwise shrink step size
        if new_score <= cur_score:
            a = a_new
            cur_score = new_score
            eta = min(1.0, eta * 1.02)   # slowly grow step size when moving downhill
            no_improve = 0
        else:
            eta = max(min_eta, eta * 0.5)
            no_improve += 1

            # occasional random jitter to escape plateaus
            if no_improve % 50 == 0:
                rng = np.random.default_rng()
                i, j = rng.integers(0, n, size=2)
                delta = 1e-4 * rng.random() * a[i]
                a[i] = max(a[i] - delta, 0.0)
                a[j] += delta
                a /= a.sum()
                cur_score = _objective(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (as far as the allotted
    time permits) minimises the evaluation function.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n, dtype=float)

    best_seq = uniform.copy()
    best_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Simple parametric sweeps (extended a little)
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric families
    for eps in np.logspace(-5, -1, 80):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        sc = _objective(geo)
        if sc < best_score:
            best_seq, best_score = geo.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} → score={sc:.6f}")

    # exponential (symmetric) families
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 60):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective(seq)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} → score={sc:.6f}")

    # two‑block families (now allowing up to half the length)
    for L_frac in np.linspace(0.02, 0.5, 30):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in np.linspace(0.1, 0.9, 9):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet draws
    # --------------------------------------------------------------
    dirichlet_trials = 3000
    for _ in range(dirichlet_trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-3.0, 0.5)          # concentration in roughly [0.001, 3]
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Random sparse vectors (fixed support, random weights)
    # --------------------------------------------------------------
    for K in (50, 100, 150, 200, 300, 400, 500):
        for _ in range(120):
            if time.time() >= deadline:
                break
            idxs = rng.choice(n, size=K, replace=False)
            weights = rng.exponential(scale=1.0, size=K)
            weights /= weights.sum()
            a = np.zeros(n, dtype=float)
            a[idxs] = weights
            sc = _objective(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] Sparse K={K} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Sub‑gradient descent from several random starts
    # --------------------------------------------------------------
    # We keep a pool of promising starting points (incl. the current best)
    start_pool = [best_seq]
    # add a few fresh random Dirichlet vectors to the pool
    for _ in range(5):
        logc = rng.uniform(-1.0, 0.2)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        start_pool.append(a)

    # main descent loop: keep launching short runs until time runs out
    while time.time() < deadline - 8.0:      # reserve a few seconds for polishing
        a0 = start_pool[rng.integers(0, len(start_pool))]
        phase_deadline = min(deadline - 3.0, time.time() + 30.0)   # max 30 s per run
        best_seq, best_score = _subgradient_descent(
            a0, phase_deadline, best_seq, best_score, start_time
        )
        # occasionally refresh the pool with the current best
        if rng.random() < 0.3:
            start_pool.append(best_seq.copy())

    # --------------------------------------------------------------
    # 5️⃣  Final short random‑jitter hill‑climb
    # --------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        delta = 1e-4 * rng.random() * a[i]
        a[i] = max(a[i] - delta, 0.0)
        a[j] += delta
        a /= a.sum()
        new_score = _objective(a)
        if new_score < best_score:
            best_seq, best_score = a.copy(), new_score
            print(f"[{time.time() - start_time:.2f}s] Jitter improve → score={new_score:.6f}")
        if new_score > cur_score:
            # revert the worsening move
            a[i] += delta
            a[j] -= delta
            a /= a.sum()
        else:
            cur_score = new_score

    # --------------------------------------------------------------
    # Normalise and return a clean Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
