def subgradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    step = 0.1
    decay_factor = 0.9995
    min_step = 1e-8
    while time.time() < deadline and step > min_step:
        conv = _conv_self(a)
        max_conv = conv.max()
        # compute mask (binary) where conv close to max (tolerance)
        eps = 1e-12 # to avoid floating error
        mask = (conv >= max_conv - 1e-12).astype(np.float64)  # maybe use tolerance relative
        # gradient = 2 * convolution(mask, a_rev) (full conv)
        a_rev = a[::-1]
        L = _next_pow2(len(mask) + len(a_rev) - 1)
        Mf = np.fft.rfft(mask, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(Mf * Ar, n=L)
        # grad_full length = len(mask) + len(a_rev) - 1 = (2n-1)+n-1 = 3n-2?
        # Actually mask length = 2n-1. So grad_full length = (2n-1)+(n)-1 = 3n-2.
        # But we need gradient entries for a of length n. According to earlier, they extracted slice (n-1):(n-1+n).
        grad = 2.0 * grad_full[(n - 1) : (n - 1 + n)]

        # Take step
        a_new = a - step * grad
        # Ensure nonnegativity might produce negative; we then project onto simplex.
        a_new = _project_simplex(a_new)
        # Compute new objective:
        new_conv = _conv_self(a_new)
        new_score = 2.0 * n * new_conv.max()
        # Accept if improved:
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iteration step={step:.3e} score={new_score:.6f}")
            a = a_new
        else:
            # Maybe still move a a little: if not improvement, maybe keep a as a_new with some probability (like simulated annealing)
            # Or reject: keep old a
            # We'll implement simple acceptance: if new_score < cur_score (not best) accept else revert.
            cur_score = 2.0 * n * max_conv
            if new_score < cur_score:
                a = a_new

    starter = np.maximum(starter, 0)

    if sum_s > 0:

def _subgradient_descent(a0, best_score, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    step = 1.0
    min_step = 1e-6
    decay = 0.9995
    while time.time() < deadline and step > min_step:
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv
        # compute mask weight vector
        tol = max_conv * 1e-12 + 1e-14
        mask = (conv >= max_conv - tol).astype(np.float64)
        mask_sum = mask.sum()
        if mask_sum == 0:
            # numeric path - treat all as equal?
            mask = np.ones_like(conv)
            mask_sum = mask.size
        w = mask / mask_sum
        # gradient
        a_rev = a[::-1]
        L = _next_pow2(len(w) + len(a_rev) - 1)
        Wf = np.fft.rfft(w, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(Wf * Ar, n=L)
        grad = 2.0 * grad_full[(n - 1):(n - 1 + n)]
        # line search
        step_try = step
        improved = False
        while step_try > min_step:
            a_new = a - step_try * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-14:
                # accept
                a = a_new
                cur_obj = obj_new
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"[{time.time() - start_time:.2f}s] Subgrad step_len={step_try:.3e} score={obj_new:.6f}")
                improved = True
                break
            else:
                step_try *= 0.5
        if not improved:
            # add jitter to escape
            eps = step * 0.01
            a = a + rng.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
        step *= decay
    return best_seq, best_score

"""
search_for_minimum.py

Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The evaluation is scale‑invariant, so we always work with sequences whose
sum equals 1 (i.e. points on the probability simplex).

Algorithm
---------
1. Initialise from the global variable ``height_sequence_1`` if it exists,
   otherwise fall back to the uniform distribution.
2. **Projected sub‑gradient descent on the true max‑convolution**.
   * The indices attaining the maximal value of the self‑convolution form
     the active set.
   * A (convex) sub‑gradient of the objective is
         g = 2 * (a_rev  *  w)[n‑1:n‑1+n]      (convolution)
     where ``a_rev`` is ``a`` reversed and ``w`` are the normalized
     weights of the active set.
   * We take a step opposite to this sub‑gradient, then project the
     result onto the simplex (O(n) Euclidean projection).
   * A simple back‑tracking line search ensures monotone improvement.
3. **Local random perturbations**.
   When the sub‑gradient phase has plateaued we repeatedly draw a fresh
   Dirichlet vector, mix it with the current best sequence using a small
   mixing coefficient (drawn uniformly), project back onto the simplex
   and keep any improvement.
4. The whole procedure respects a 1000 s wall‑clock budget; the best
   sequence found is returned as a plain Python list of floats.

All helper functions are top‑level; no closures or file I/O are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Objective for a sequence that already sums to 1:
        2 * n * max(conv(seq, seq))
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpinán, 2013 (https://arxiv.org/abs/1309.1541)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Stage 1 – projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         best_score: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Sub‑gradient method that uses the *hard* max (no temperature).
    Returns the best sequence and the associated score discovered during
    the descent.
    """
    a = a0.copy()
    n = a.shape[0]

    # best values seen so far
    best_seq = a.copy()
    best_obj = best_score

    # step‑size schedule
    step = 1.0          # initial trial step length
    min_step = 1e-6
    decay = 0.9995

    while time.time() < deadline and step > min_step:
        # current convolution and objective
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv

        # --- sub‑gradient of the max (active‑set weighting) ---
        # tolerance to capture all (near) maximisers
        tol = max_conv * 1e-12 + 1e-14
        active = (conv >= max_conv - tol).astype(np.float64)
        w_sum = active.sum()
        if w_sum == 0:                     # numerical fallback
            active = np.ones_like(conv)
            w_sum = active.size
        weights = active / w_sum             # convex combination → sum 1

        # gradient = 2 * (weights * a_rev) restricted to original n entries
        a_rev = a[::-1]
        L = _next_pow2(len(weights) + len(a_rev) - 1)
        Wf = np.fft.rfft(weights, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(Wf * Ar, n=L)
        grad = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # --- back‑tracking line‑search on the true objective ---
        trial_step = step
        improved = False
        while trial_step > min_step:
            a_trial = a - trial_step * grad
            a_trial = _project_simplex(a_trial)
            conv_trial = _conv_self(a_trial)
            obj_trial = 2.0 * n * conv_trial.max()
            if obj_trial < cur_obj - 1e-12:
                # Accept the step
                a = a_trial
                cur_obj = obj_trial
                improved = True
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_seq = a_trial.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Subgrad step={trial_step:.2e} score={obj_trial:.6f}")
                break
            else:
                trial_step *= 0.5

        if not improved:
            # add a tiny random jitter to escape flat regions
            eps = step * 0.01
            a = a + rng.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

        # decay the trial step for the next iteration
        step *= decay

    return best_seq, best_obj


# ----------------------------------------------------------------------
# Stage 2 – cheap random local perturbations
# ----------------------------------------------------------------------
def _local_random_search(best_seq: np.ndarray,
                         best_score: float,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Repeatedly mix the current best vector with a fresh Dirichlet draw.
    The mixing coefficient is drawn uniformly from a modest interval so
    that the search stays local but still explores the neighbourhood.
    """
    n = best_seq.shape[0]
    while time.time() < deadline:
        eps = rng.uniform(0.0005, 0.05)          # how much we deviate
        # Dirichlet sample (concentration 1 → uniform on simplex)
        cand = rng.gamma(shape=1.0, scale=1.0, size=n)
        cand /= cand.sum()
        candidate = (1.0 - eps) * best_seq + eps * cand
        candidate = _project_simplex(candidate)   # keep exactly on simplex
        obj = _objective_score(candidate)
        if obj < best_score:
            best_score = obj
            best_seq = candidate.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Random mix eps={eps:.4f} score={obj:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for the step‑function that minimises the evaluation
    function.  The routine works entirely on the probability simplex,
    uses a projected sub‑gradient method on the exact max‑convolution,
    and finishes with a fast stochastic local search.
    The best sequence (as a list of Python floats) is returned.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise – use the provided global starter if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        # enforce non‑negativity and normalise
        starter = np.maximum(starter, 0.0)
        total = starter.sum()
        if total > 0:
            starter /= total
        else:
            raise ValueError
    except Exception:
        # fallback: uniform distribution of length 4096
        n = 4096
        starter = np.full(n, 1.0 / n)

    n = starter.shape[0]
    best_seq = starter.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – projected sub‑gradient descent (majority of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _subgradient_descent(
            best_seq, best_score, deadline, rng
        )
        print(f"[{time.time() - start_time:.2f}s] End sub‑gradient, "
              f"best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – local random mixing search (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _local_random_search(
            best_seq, best_score, deadline, rng
        )
        print(f"[{time.time() - start_time:.2f}s] End random search, "
              f"best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
