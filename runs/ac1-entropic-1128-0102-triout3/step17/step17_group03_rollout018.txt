"""
Improved search for a step‑function that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣  Parameter sweep – a handful of simple analytic families (uniform,
    triangular, Beta‑shaped, edge‑exponential) are normalised and the
    best one is kept as a start point.

2️⃣  Deterministic “active‑set” descent – at every iteration the active
    convolution indices (those attaining the current maximum) are
    identified.  The sub‑gradient tells from which entry mass should be
    removed (largest gradient) and to which entry it should be added
    (smallest gradient).  Using the linearised effect of a transfer on
    the active gradients we compute the optimal amount of mass to move
    (closed‑form line search) and accept the move only if it reduces the
    true objective.  This yields far larger improvement per step than a
    plain simulated‑annealing move.

3️⃣  Final cheap random polishing – after the deterministic phase a
    short random local‑search finishes the job.

All heavy work (convolution, gradient) is performed with FFTs,
so many iterations fit easily into the 1000 s budget.  The returned
vector is normalised to ℓ₁‑norm 1, non‑negative and therefore valid for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing
# ----------------------------------------------------------------------
_start_time = None  # will be set in propose_candidate


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Objective for a normalised to sum 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at ``a`` (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _initial_candidates(n: int, rng: np.random.Generator):
    """Generate a handful of analytically‑parameterised candidates."""
    best_seq = np.full(n, 1.0 / n, dtype=float)   # uniform
    best_score = _objective_score(best_seq)

    print(f"[{time.time() - _start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # triangular (piecewise linear, symmetric)
    half = n // 2
    tri = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)], dtype=float)
    if tri.shape[0] != n:
        # odd length – append the middle element
        tri = np.concatenate([tri, [half + 1]], axis=0)[:n]
    tri /= tri.sum()
    sc = _objective_score(tri)
    if sc < best_score:
        best_seq, best_score = tri, sc
        print(f"[{time.time() - _start_time:.2f}s] Triangular score = {sc:.6f}")

    # Beta‑style shapes a_i ∝ ((i+1)*(n-i))**p  (p can be negative)
    idx = np.arange(n, dtype=float)
    base = (idx + 1) * (n - idx)  # always >0
    eps = 1e-12
    exponents = np.linspace(-1.5, 2.0, 15)          # coarse sweep
    for p in exponents:
        shape = np.power(base + eps, p)
        shape_sum = shape.sum()
        if shape_sum == 0:
            continue
        shape /= shape_sum
        sc = _objective_score(shape)
        if sc < best_score:
            best_seq, best_score = shape, sc
            print(f"[{time.time() - _start_time:.2f}s] Beta p={p:.3f} score = {sc:.6f}")

        # mixtures with the uniform vector
        for lam in (0.2, 0.4, 0.6, 0.8):
            mix = lam * np.full(n, 1.0 / n) + (1.0 - lam) * shape
            mix /= mix.sum()
            sc = _objective_score(mix)
            if sc < best_score:
                best_seq, best_score = mix, sc
                print(f"[{time.time() - _start_time:.2f}s] Mix λ={lam:.1f}, p={p:.3f} score = {sc:.6f}")

    # Edge‑exponential families
    cs = np.logspace(-6, -1, 12)   # decay rates
    for c in cs:
        shape = np.exp(-c * idx ** 2) + np.exp(-c * (n - 1 - idx) ** 2)
        shape /= shape.sum()
        sc = _objective_score(shape)
        if sc < best_score:
            best_seq, best_score = shape, sc
            print(f"[{time.time() - _start_time:.2f}s] Edge exp c={c:.1e} score = {sc:.6f}")

    # A few random Dirichlet draws (light‑weight)
    max_dir = 200
    for _ in range(max_dir):
        if time.time() >= _deadline:
            break
        a = rng.gamma(shape=1.0, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
            print(f"[{time.time() - _start_time:.2f}s] Random Dirichlet score = {sc:.6f}")

    return best_seq, best_score

def _active_set_opt_move(a: np.ndarray, conv: np.ndarray, grad: np.ndarray,
                         active: np.ndarray) -> (np.ndarray, float):
    """
    For the given vector a compute the optimal transfer of mass from the
    index with largest gradient (src) to the one with smallest gradient (dst).
    The transfer amount is chosen analytically (linearised active‑set model)
    and the resulting candidate is accepted only if the true objective
    improves.
    Returns the new vector and its objective (or the original one if no
    improvement is found).
    """
    n = a.shape[0]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst or a[src] <= 0.0:
        return a, _objective_score(a)

    # Build the c‑vector (effect of moving mass on each gradient component)
    c = np.zeros(n, dtype=int)
    src_set = set()
    dst_set = set()
    for k in active:
        i = k - src
        if 0 <= i < n:
            src_set.add(i)
        j = k - dst
        if 0 <= j < n:
            dst_set.add(j)
    for i in src_set:
        c[i] -= 1
    for i in dst_set:
        c[i] += 1

    # constant‑gradient indices (c == 0)
    const_idx = np.where(c == 0)[0]
    if const_idx.size:
        min_const = grad[const_idx].min()
        max_const = grad[const_idx].max()
    else:
        min_const = max_const = None

    n_active = max(1, active.size)          # avoid division by zero
    alpha_max = 2.0 * a[src] / n_active      # α = 2·δ / |A|
    alphas = [0.0, alpha_max]

    # generate a small set of promising α values (closed‑form intersections)
    for i in src_set:
        gi = grad[i]
        if const_idx.size:
            alphas.append(max(gi - min_const, 0.0))
            alphas.append(max(gi - max_const, 0.0))
        for j in dst_set:
            gj = grad[j]
            alphas.append(max((gi - gj) / 2.0, 0.0))
    for i in dst_set:
        gi = grad[i]
        if const_idx.size:
            alphas.append(max(min_const - gi, 0.0))
            alphas.append(max(max_const - gi, 0.0))

    # prune and deduplicate
    alphas = [a for a in alphas if 0.0 < a < alpha_max]
    if not alphas:
        # fall back to a modest random transfer
        delta = a[src] * 0.05
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        return cand, _objective_score(cand)

    best_a = a
    best_score = _objective_score(a)
    # evaluate each candidate α
    for alpha in alphas:
        delta = alpha * n_active / 2.0
        if delta <= 0 or delta > a[src]:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # safety for numerical noise
        if cand[src] < -1e-12:
            continue
        # evaluate true objective
        sc = _objective_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand

    return best_a, best_score


def _deterministic_descent(a: np.ndarray, best_score: float,
                          deadline: float, rng: np.random.Generator):
    """Iteratively apply the active‑set optimal move until time runs out."""
    n = a.shape[0]
    best_a = a.copy()
    cur_score = best_score
    no_improve = 0
    while time.time() < deadline - 5.0:      # keep ~5 s for polishing
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] New best (deterministic) {best_score:.6f}")

        # active set and sub‑gradient
        tol = 1e-12
        active = np.where(conv >= max_b - tol)[0]
        grad = _subgradient_maxconv(a, conv)

        # try the analytical optimal move
        cand, cand_score = _active_set_opt_move(a, conv, grad, active)

        if cand_score < cur_score - 1e-12:
            a = cand
            no_improve = 0
            continue

        # if that fails, perform a tiny random transfer (helps escape)
        src = rng.integers(n)
        dst = rng.integers(n)
        if src != dst and a[src] > 1e-15:
            delta = a[src] * rng.random() * 0.02   # 2 % of source mass
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                no_improve = 0
                continue

        no_improve += 1
        if no_improve > 2000:                 # stagnation → re‑randomise a little
            a = _project_simplex(a + 0.01 * rng.normal(size=n))
            a /= a.sum()
            no_improve = 0
    return best_a, best_score

def _random_local_refinement(a: np.ndarray, cur_score: float,
                             deadline: float, rng: np.random.Generator):
    """Cheap random pairwise transfers accepting only strict improvements."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        delta = a[src] * rng.random() * 0.005
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - _start_time:.2f}s] Polishing new best {best_score:.6f}")
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    global _start_time, _deadline
    np.random.seed(seed)
    _start_time = time.time()
    _deadline = _start_time + budget_s - 0.1  # tiny safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem dimension – a few thousand entries gives a good trade‑off
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Phase 1 – fast analytic candidates
    # --------------------------------------------------------------
    best_seq, best_score = _initial_candidates(n, rng)

    # --------------------------------------------------------------
    # Phase 2 – deterministic active‑set descent
    # --------------------------------------------------------------
    if time.time() < _deadline:
        print(f"[{time.time() - _start_time:.2f}s] Starting deterministic descent")
        best_seq, best_score = _deterministic_descent(best_seq, best_score,
                                                      _deadline, rng)
        print(f"[{time.time() - _start_time:.2f}s] Deterministic descent finished – best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final cheap random polishing (≈30 s)
    # --------------------------------------------------------------
    if time.time() < _deadline:
        local_deadline = min(_deadline, time.time() + 30.0)
        print(f"[{time.time() - _start_time:.2f}s] Starting final polishing")
        best_seq, best_score = _random_local_refinement(best_seq,
                                                        best_score,
                                                        local_deadline,
                                                        rng)
        print(f"[{time.time() - _start_time:.2f}s] Polishing finished – best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return (ensuring feasibility for the evaluator)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - _start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
