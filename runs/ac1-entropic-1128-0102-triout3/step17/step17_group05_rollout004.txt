```python
def _lp_step(f, M, deadline):
    # returns new g normalized or None if fails or time out
    import cvxpy as cp
    n = f.shape[0]
    g = cp.Variable(n)
    f_const = cp.Parameter(n, value=f, name="f")
    M_const = cp.Parameter(nonneg=True, value=M, name="M")
    constraints = [cp.conv(f_const, g) <= M_const,
                   g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # solve
    try:
        # Use short time limit maybe using solver options?
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        if prob.status not in ("optimal", "optimal_inaccurate"):
            # fallback: use SCS
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=7500)
        if prob.status not in ("optimal", "optimal_inaccurate"):
            return None
        g_val = np.asarray(g.value).flatten()
        # ensure nonneg
        g_val = np.maximum(g_val, 0.0)
        s = g_val.sum()
        if s == 0:
            return None
        g_norm = g_val / s
        return g_norm
    except Exception as e:
        # print error
        print("LP solve error", e)
        return None
```

"""
Propose a non‑negative step‑function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search runs under a ~1000 s budget and consists of five phases:

1️⃣  Light parametric sweeps (geometric, power, exponential, two blocks,
    plus a new Gaussian family) – inexpensive global exploration.

2️⃣  A large batch of random Dirichlet draws – stochastic coverage of the
    probability simplex.

3️⃣  Multi‑start smooth‑max gradient descent – the non‑smooth max is
    replaced by a τ·log‑sum‑exp surrogate; a line‑search on the true
    objective ensures monotone improvement.

4️⃣  Projected sub‑gradient descent on the true max‑objective – cheap,
    fast, and able to escape plateaus.

5️⃣  Random jitter + short sub‑gradient refinements – a simple simulated‑
    annealing style loop that repeatedly perturbs the current best and
    runs a few sub‑gradient steps, keeping any improvement.

All phases respect the global time‑budget; the best sequence (normalised
to sum = 1) is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using FFT (real‑valued)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Assuming sum(seq)=1, returns 2·n·max(conv(seq,seq))."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# 1️⃣ Parametric sweeps
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_gaussian(n, best_seq, best_score, deadline):
    """Symmetric Gaussian centred in the middle. Mix optionally with uniform."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    # sigma ranging from very narrow to very wide
    sigma_grid = np.linspace(n / 200.0, n / 2.0, 40)
    for sigma in sigma_grid:
        if time.time() >= deadline:
            break
        seq = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        for lam in (0.0, 0.4, 0.8):
            seq_mix = lam * np.full(n, 1.0 / n) + (1.0 - lam) * seq
            seq_mix /= seq_mix.sum()
            sc = _objective_score(seq_mix)
            if sc < best_score:
                best_score, best_seq = sc, seq_mix.copy()
                print(f"[{time.time() - start_time:.2f}s] Gaussian σ={sigma:.1f} λ={lam:.2f} score={sc:.6f}")
    return best_seq, best_score

def _random_dirichlet_trials(
    n,
    best_seq,
    best_score,
    deadline,
    trials=3000,
    rng=None,
):
    if rng is None:
        rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)   # concentration exponent 10**logc
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score

# 3️⃣ Smooth‑max gradient descent (same as earlier but slightly tuned)
# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(a0, deadline, best_seq, best_score):
    """
    Gradient descent on the surrogate   τ·log∑exp(conv/τ) .
    τ is reduced geometrically. A back‑tracking line‑search on the true
    objective guarantees monotone decrease.
    """
    a = a0.copy()
    n = a.shape[0]
    # initialise temperature from current peak
    cur_conv = _conv_self(a)
    tau = max(cur_conv.max() * 0.5, 1e-6)
    min_tau = 1e-9
    decay = 0.996

    while time.time() < deadline and tau > min_tau:
        conv = _conv_self(a)
        max_conv = conv.max()
        # ---- smoothed weights (softmax) ----
        shift = max_conv
        log_w = (conv - shift) / tau
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0:
            tau *= decay
            continue
        p = w / w_sum

        # ---- gradient of the smooth max ----
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]
        grad = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # ---- line‑search on true objective ----
        cur_obj = 2.0 * n * max_conv
        step = 1.0
        improved = False
        while step > 1e-14:
            a_new = a - step * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-12:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SmoothGD obj={obj_new:.6f} τ={tau:.2e}")
                break
            step *= 0.5

        if not improved:
            # small jitter to escape a flat region
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

        tau *= decay
    return best_seq, best_score

# 4️⃣ Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0, deadline, best_seq, best_score):
    """
    Classic sub‑gradient method on   2·n·max(conv(a,a)).
    Uses a back‑tracking line‑search on the true objective at each step.
    """
    a = a0.copy()
    n = a.shape[0]
    step = 1.0
    decay = 0.99
    while time.time() < deadline and step > 1e-10:
        conv = _conv_self(a)
        m = int(conv.argmax())

        # gradient of max‑conv (sub‑gradient) : g_i = 2 * a_{m-i} if index in range
        lo = max(0, m - (n - 1))
        hi = min(n - 1, m)
        idxs = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idxs] = 2.0 * a[m - idxs]

        cur_obj = 2.0 * n * conv.max()
        improved = False
        trial = step
        while trial > 1e-14:
            a_new = a - trial * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-12:
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SubGrad obj={obj_new:.6f} step={trial:.2e}")
                break
            trial *= 0.5

        if not improved:
            # tiny random jitter to shake out of a plateau
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

        step *= decay
    return best_seq, best_score

    # --------------------------------------------------------------

            break

        if sc < best_score:
            best_score, best_seq = sc, seq.copy()

    # --------------------------------------------------------------
    # 4️⃣ Sub‑gradient descent refinement (≈100 s)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _subgradient_descent(best_seq, deadline, best_seq, best_score)

    # --------------------------------------------------------------
    # 5️⃣ Random jitter + short sub‑gradient bursts (use remaining time)
    # --------------------------------------------------------------
    while time.time() < deadline - 0.5:

    best_seq = np.maximum(best_seq, 0.0)
