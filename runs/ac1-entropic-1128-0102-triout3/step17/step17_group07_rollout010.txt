    beta2=0.999

            t += 1
            # compute convolution
            conv = _conv_self_fft(a, L_fft)

            scaled = alpha * conv
            max_scaled = scaled.max()
            exp_scaled = np.exp(scaled - max_scaled)  # subtract max for numerical stability
            sum_exp = exp_scaled.sum()

"""
Heuristic optimizer for the max‑self‑convolution objective.

The algorithm proceeds in four stages:

0️⃣  Trivial (uniform) and optional global starter.
1️⃣  Warm‑start: a few deterministic “U‑shaped’’ seeds and many Dirichlet draws.
2️⃣  Smooth‑max gradient descent (Adam) with a temperature (α) schedule.
   The objective  max(conv) is approximated by a soft‑max, whose gradient is
   obtained analytically via FFT.  After each step we project back onto the
   probability simplex.
3️⃣  Targeted hill‑climb that attacks the current maximal convolution entry.
4️⃣  Final uniform‑mixing + cheap random‑pair hill‑climb.

All phases respect the global time budget (default 1000 s) and stop early
if the target score 1.503 is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx_i=1, x_i≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (assumes Σa_i = 1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Warm‑start (deterministic shapes + Dirichlet draws)
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                L_fft: int,
                deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # deterministic “U‑shaped’’ seeds (a_i ∝ |i‑mid|^p )
    for p in (0.2, 0.5, 0.8, 1.0, 1.2):
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=float)
        mid = (n - 1) / 2.0
        shape = np.abs(idx - mid) + 1e-12
        a = shape ** p
        a = a / a.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] U‑shape p={p:.2f} → score={sc:.6f}")

    # random Dirichlet draws
    while time.time() < deadline:
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smooth‑max (soft‑max) gradient descent with Adam
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                        n: int,
                        L_fft: int,
                        L_corr: int,
                        deadline: float,
                        alphas) -> tuple[np.ndarray, float]:
    """
    Minimise the maximum self‑convolution by optimising a soft‑max surrogate.
    Uses Adam (first‑order) with projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    for alpha in alphas:
        # (re)initialise Adam for each temperature – helps escape plateaux
        m.fill(0.0)
        v.fill(0.0)

        # learning‑rate scaled with 1/α (higher temperature → larger steps)
        base_lr = 0.3 / max(alpha, 1.0)

        while time.time() < deadline:
            t += 1
            # ----- convolution (used later for scoring) -----
            conv = _conv_self_fft(a, L_fft)

            # ----- soft‑max over the convolution entries -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            exp_scaled = np.exp(scaled - max_scaled)      # stability
            p = exp_scaled / exp_scaled.sum()            # shape (2n‑1,)

            # ----- gradient of the soft‑max surrogate -----
            a_rev = a[::-1]
            Pr = np.fft.rfft(p, n=L_corr)
            Ar = np.fft.rfft(a_rev, n=L_corr)
            full = np.fft.irfft(Pr * Ar, n=L_corr)
            start = n - 1
            grad = alpha * full[start:start + n]          # ∇  (size n)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1**t)
            v_hat = v / (1.0 - beta2**t)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_simplex(a)

            # ----- occasional evaluation & tiny random jitter -----
            if t % 150 == 0:
                cur_score = _objective_score(a, n, L_fft)
                if cur_score < best_score:
                    best_score, best_a = cur_score, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] α={alpha:.1f}  it={t:,}  score={cur_score:.6f}")
                    if best_score <= 1.503:
                        print("[Early exit] Target score reached during smooth‑max descent.")
                        return best_a, best_score
                if np.random.rand() < 0.005:  # tiny jitter to escape plateaux
                    eps_noise = np.random.normal(scale=0.001, size=n)
                    a = np.maximum(a + eps_noise, 0.0)
                    a = _project_simplex(a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Helper for hill‑climb: largest product contributing to a convolution entry
# ----------------------------------------------------------------------
def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """Return (i, j, a_i·a_j) for the pair that contributes most to conv[k]."""
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    prod = a[idx] * a[k - idx]
    pos = np.argmax(prod)
    i = int(idx[pos])
    j = int(k - i)
    return i, j, float(prod[pos])


# ----------------------------------------------------------------------
# 3️⃣  Targeted hill‑climb on the current maximal convolution entry
# ----------------------------------------------------------------------
def _targeted_hill_climb(a: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        i, j, contrib = _max_contributions(best_a, k_max)
        if i == -1:
            continue

        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-12:
            continue
        dst = rng.integers(0, n)
        if dst == src:
            continue

        delta = rng.uniform(0.0, min(best_a[src], 0.004))
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        new_score = _objective_score(cand, n, L_fft)
        if new_score < best_score:
            best_score, best_a = new_score, cand
            print(f"[{time.time() - start_time:.2f}s] Hill i={src}→{dst} δ={delta:.5f} → score={new_score:.6f}")
            if best_score <= 1.503:
                print("[Early exit] Target score reached during hill‑climb.")
                return best_a, best_score
    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final mixing + light random‑pair hill‑climb (as in the reference)
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform vector
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")
            if best_score <= 1.503:
                return best_a, best_score

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill‑climb i={i} j={j} δ={delta:.5f} → score={sc:.6f}")
            if best_score <= 1.503:
                return best_a, best_score
    return best_a, best_score
