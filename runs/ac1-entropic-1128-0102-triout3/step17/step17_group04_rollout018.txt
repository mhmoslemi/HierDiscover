def _subgradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2*n-1)
    cur_score = _objective(a)
    eta = 0.5
    min_eta = 1e-8
    while time.time() < deadline:
        # compute convolution via FFT
        FA = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(FA * FA, n=L)[:2*n-1]
        k = conv.argmax()
        # subgradient: for i where 0<=i<n and 0<=k-i<n, grad_i = 2 * a[k-i]
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        grad = np.zeros_like(a)
        idx = np.arange(lo, hi+1)
        grad[idx] = a[k - idx] * 2.0
        # propose update
        a_new = _exp_update(a, grad, eta)
        new_score = _objective(a_new)
        if new_score <= cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time()-start_time:.2f}s] Subgrad new best score={new_score:.6f}")
            eta = min(1.0, eta * 1.02)
        else:
            # reduce step size
            eta = max(min_eta, eta * 0.5)
            # occasional random perturb if stuck
            if np.random.rand() < 0.01:
                # jitter
                eps = np.random.normal(scale=0.01, size=n)
                a = np.maximum(a + eps, 0.0)
                a /= a.sum()
                cur_score = _objective(a)

        if best_score <= 1.503:
            break
    return best_seq, best_score
```

"""
Propose a step‑function (non‑negative list) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The search consists of the following phases

1️⃣  Very cheap parametric sweeps (geometric, power‑law, inverse‑power,
    polynomial‑decay).  The best candidate found so far is kept.

2️⃣  A large number of random Dirichlet draws – useful for finding diverse
    starting points.

3️⃣  Mirror‑descent on the *exact* max‑convolution (a sub‑gradient step).
    The gradient of the max‑convolution is simply
        g_i = 2·a_{k*‑i}   (where k* is the index of the current maximum)
    and the exponential update
        a_i ← a_i·exp(−η·g_i)    followed by a projection onto the simplex,
    guarantees non‑negativity and Σa=1.

4️⃣  Periodic small random jitter to escape possible plateaus.

All convolutions are performed with a single FFT (length is a power of two
≥ 2·n−1) – an O(n log n) operation that is very fast for n≈4 000.
The routine stops well before the 1000 s budget and returns the best
sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a using a single FFT of length L (L≥2·n‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray, L: int) -> float:
    """Objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    conv = _conv_self_fft(a, L)
    return float(2.0 * a.shape[0] * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent (exponential) update:
        a_i ← a_i·exp(−η·grad_i)   then re‑normalise to Σa=1.
    """
    log_factor = -eta * grad
    # shift to avoid overflow
    log_factor -= np.max(log_factor)
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total

# Helper to try a candidate and keep the best one seen so far
# ----------------------------------------------------------------------
def _try_candidate(seq: np.ndarray, L: int,
                   best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Normalise seq, compute its score and update the best‑so‑far if needed."""
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s <= 0:
        return best_seq, best_score
    seq = seq / s
    sc = _objective(seq, L)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] New best score={sc:.6f}")
    return best_seq, best_score

# 1️⃣  Very cheap parametric sweeps
# ----------------------------------------------------------------------
def _parametric_sweeps(n, L, best_seq, best_score, deadline):
    """Explore a handful of simple families (geometric, power‑law, etc.)."""
    idx = np.arange(n, dtype=np.float64)

    # ----- geometric (a_i ∝ r^i) -----
    eps_grid = np.logspace(-5, -1, 80)            # r = 1‑ε
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        best_seq, best_score = _try_candidate(geo, L, best_seq, best_score)

    # ----- power‑law decreasing a_i ∝ (n‑i)^p
    for p in np.linspace(0.0, 5.0, 31):
        if time.time() >= deadline:
            break
        seq = (n - idx) ** p
        best_seq, best_score = _try_candidate(seq, L, best_seq, best_score)

    # ----- inverse power‑law a_i ∝ (i+1)^‑p
    for p in np.linspace(0.0, 5.0, 31):
        if time.time() >= deadline:
            break
        seq = (idx + 1) ** (-p)
        best_seq, best_score = _try_candidate(seq, L, best_seq, best_score)

    # ----- polynomial decay a_i ∝ (1‑i/(n‑1))^p
    for p in np.linspace(0.0, 5.0, 31):
        if time.time() >= deadline:
            break
        seq = (1.0 - idx / (n - 1)) ** p
        best_seq, best_score = _try_candidate(seq, L, best_seq, best_score)

    return best_seq, best_score

def _dirichlet_trials(n, L, best_seq, best_score, deadline, trials=2000):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration parameter in [0.01, 5]
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        best_seq, best_score = _try_candidate(a, L, best_seq, best_score)
    return best_seq, best_score

# 3️⃣  Sub‑gradient mirror descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, L: int,
                         deadline, best_seq, best_score):
    """
    Mirror‑descent using the true sub‑gradient of max(conv(a,a)).
    Keeps the best discrete objective seen so far.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a, L)
    eta = 0.5
    min_eta = 1e-8
    rng = np.random.default_rng()

    while time.time() < deadline:
        # full convolution and index of the current maximum
        conv = _conv_self_fft(a, L)
        k = int(conv.argmax())

        # sub‑gradient of the max‑convolution w.r.t. a:
        #   ∂/∂a_i max(conv) = 2·a_{k‑i}  (if   0 ≤ k‑i < n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        grad = np.zeros_like(a)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]

        # exponential (mirror‑descent) update
        a_new = _exp_update(a, grad, eta)
        new_score = _objective(a_new, L)

        if new_score <= cur_score:
            a, cur_score = a_new, new_score
            eta = min(1.0, eta * 1.02)          # gently increase step‑size
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Subgrad improved to {new_score:.6f}")
        else:
            # no improvement – shrink step‑size and occasionally jitter
            eta = max(min_eta, eta * 0.5)
            if rng.random() < 0.01:              # occasional random shake‑up
                eps = rng.normal(scale=0.01, size=n)
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)
                cur_score = _objective(a, L)

        if best_score <= 1.503:                  # early stop if target reached
            break

    return best_seq, best_score
