"""
propose_candidate
=================

A fast, low‑memory search for a non‑negative step function that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three phases:

1. **Large problem size** –  Choose a moderately large length (default
   12 000).  Because the score is scale‑invariant, increasing *n* brings the
   discrete optimum closer to the continuous lower bound (score → 1).

2. **Exploration of analytic families** –  Initialise candidates from
   several well‑behaved families on the probability simplex:
   * Beta(p,p) (including the arcsine law p=½),
   * a simple sine density,
   * the triangular density.
   Each family is also mixed with the uniform vector.  All scores are
   evaluated with an FFT‑based convolution, i.e.  O(n log n).

3. **Simple stochastic hill‑climb** –  Starting from the best candidate,
   repeatedly move a tiny amount of mass from a random index *i* to another
   index *j*.  If the move lowers the score it is kept.  The step size
   shrinks slowly so that the search finishes before the allocated budget
   (default 1000 s).

The best sequence found is returned as a plain Python list of floats.
All numbers are non‑negative and sum to one, so `evaluate_sequence` will
accept them without clipping.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Full linear self‑convolution a⋆a (length 2·len(a)‑1) using FFT.
    The result is a real array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_fft(a: np.ndarray, n: int, L_fft: int) -> float:
    """
    Exact score used by the evaluator for a *probability* vector ``a``.
    The vector is assumed to be normalised (sum = 1), therefore
    score = 2·n·max(conv(a,a)).
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised Beta(p,p) density on the grid x_i = (i+0.5)/n.
    Returns a normalised probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # p‑1 can be negative → use power with safety against 0⁻
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _sine_shape(n: int) -> np.ndarray:
    """Simple sinusoidal density sin(π·x) on (0,1) (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = np.sin(np.pi * x)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _triangular_shape(n: int) -> np.ndarray:
    """Triangular (tent) shape – linear rise then fall, normalised."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget,
    prints progress information and returns the best sequence it discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # a tiny safety margin so we never exceed the budget
    deadline = start + budget_s - 0.05

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical bound.
    # ------------------------------------------------------------------
    n = 12000                      # feel‑free to adjust (kept moderate)
    L_fft = _next_pow2(2 * n - 1)  # length for FFT‑based convolution

    # ------------------------------------------------------------------
    # 0️⃣  Baseline: uniform (only for reference)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _score_fft(best_a, n, L_fft)
    print(f"[0.00s] uniform baseline → score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣+  Global starter (if the host provides one)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter,
                                     (0, n - starter.size),
                                     constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _score_fft(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] global starter improved → score = {sc:.6f}")
    except Exception:
        # No global starter available – ignore silently
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start with a handful of Dirichlet draws (≈5 % of budget)
    # ------------------------------------------------------------------
    warm_end = min(deadline, start + 0.05 * budget_s)
    while time.time() < warm_end:
        # random concentration between 0.02 and 20 (log‑uniform)
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _score_fft(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start:.2f}s] Dirichlet warm‑start → score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Scan analytic families (Beta, sine, triangular) and their mixtures
    # ------------------------------------------------------------------
    print(f"[{time.time() - start:.2f}s] scanning analytic families")
    # common mixture grid
    lam_vals = np.linspace(0.0, 1.0, 11)   # 0, 0.1, …, 1

    # ---- Beta(p,p) ----------------------------------------------------
    p_vals = np.linspace(0.20, 2.00, 61)   # fine grid, includes p=0.5
    for p in p_vals:
        shape = _beta_shape(n, p)

        # pure shape
        sc = _score_fft(shape, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, shape.copy()
            print(f"[{time.time() - start:.2f}s] pure Beta(p={p:.3f}) → score = {sc:.6f}")

        # mixtures with uniform
        for lam in lam_vals:
            cand = lam * shape + (1.0 - lam) * uniform
            # renormalise just in case of rounding noise
            cand = cand / cand.sum()
            sc = _score_fft(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start:.2f}s] Beta(p={p:.3f}) λ={lam:.2f} → score = {sc:.6f}")

        if time.time() >= deadline:
            break

    # ---- Sine shape ---------------------------------------------------
    if time.time() < deadline:
        sine = _sine_shape(n)
        sc = _score_fft(sine, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, sine.copy()
            print(f"[{time.time() - start:.2f}s] sine shape → score = {sc:.6f}")
        for lam in lam_vals:
            cand = lam * sine + (1.0 - lam) * uniform
            cand = cand / cand.sum()
            sc = _score_fft(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start:.2f}s] sine λ={lam:.2f} → score = {sc:.6f}")

    # ---- Triangular shape ----------------------------------------------
    if time.time() < deadline:
        tri = _triangular_shape(n)
        sc = _score_fft(tri, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, tri.copy()
            print(f"[{time.time() - start:.2f}s] triangular shape → score = {sc:.6f}")
        for lam in lam_vals:
            cand = lam * tri + (1.0 - lam) * uniform
            cand = cand / cand.sum()
            sc = _score_fft(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start:.2f}s] triangular λ={lam:.2f} → score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Stochastic hill‑climb refinement (remaining time)
    # ------------------------------------------------------------------
    print(f"[{time.time() - start:.2f}s] hill‑climb refinement")
    a = best_a.copy()
    # start with a modest step size; it will gradually shrink
    step_scale = 0.02   # absolute amount transferred at most
    # safety: never move more than 1 % of the total mass in one step
    max_step = 0.01

    # batch‑wise time check to keep overhead low
    check_interval = 500
    iterations = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # pick a delta no larger than step_scale and also no larger
        # than a fraction of the source mass
        allowed = min(a[i], step_scale, max_step)
        if allowed <= 0.0:
            continue
        delta = rng.random() * allowed
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum is preserved → no renormalisation needed
        sc = _score_fft(cand, n, L_fft)
        if sc + 1e-12 < best_score:      # accept only strict improvements
            best_score, best_a = sc, cand.copy()
            a = cand
            # occasionally enlarge the step size a little (helps escape very flat plateaus)
            step_scale = min(0.05, step_scale * 1.02)
            print(f"[{time.time() - start:.2f}s] hill‑climb improve → score = {sc:.6f}")
        else:
            # slowly decay step size when moves are rejected
            step_scale = max(0.001, step_scale * 0.999)

        iterations += 1
        if iterations % check_interval == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Normalisation & final return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
