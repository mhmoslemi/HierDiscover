def _smgd(a0, n, L_fft, L_corr, deadline, best_score, rng):
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score

    alpha = 10.0
    alpha_max = 500.0
    eta0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution
        Af = np.fft.rfft(a, n=L_corr)
        conv = np.fft.irfft(Af * Af, n=L_corr)[:2*n - 1]
        # compute score
        max_conv = conv.max()
        cur_sc = 2 * n * max_conv
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SMGD iter={it} score={cur_sc:.6f}")

        # compute softmax weights
        maxc = conv.max()  # for stability
        # weight = exp(alpha * (conv - maxc))
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # gradient
        w_rev = w[::-1]
        Wf = np.fft.rfft(w_rev, n=L_corr)
        grad = 2.0 * np.fft.irfft(Af * Wf, n=L_corr)
        # extract relevant slice for grad of length n
        grad = grad[(2*n - 2) - np.arange(n)] * 1.0  # using earlier derived slice
        # step size
        eta = eta0 / np.sqrt(it + 1)
        a_new = a - eta * grad
        # project onto simplex
        a = _project_simplex(a_new)

        # occasional random jitter
        if rng.random() < 0.005:
            eps = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

        # increase alpha gradually
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.01)

    return best_a, best_sc

"""
Propose a step‑function (non‑negative vector) that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The method consists of four stages:

0️⃣  initialise from a global starter / uniform / Dirichlet draws.
1️⃣  **Soft‑max gradient descent** – minimise a smooth approximation of the
    maximum of the self‑convolution (soft‑max with a temperature that is
    gradually lowered).  The gradient is obtained analytically via FFT and
    the iterate is projected onto the probability simplex.
2️⃣  **Direct max sub‑gradient descent** – once the temperature is high,
    descend on the true max using its sub‑gradient (the terms that realise
    the current peak).  A small stepsize schedule and occasional jitter
    help to escape plateaux.
3️⃣  **Final refinement** – convex mixing with the uniform distribution and
    a cheap random‑pair hill‑climb.

All stages respect a 1000 s budget; intermediate progress and scores are
printed.  The returned list is normalised to sum ≈ 1 (the scale does not
affect the evaluation score because it cancels out).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv_full(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (assumes Σa = 1)."""
    conv = _self_conv_full(a, L_fft)
    return float(2.0 * n * conv.max())


def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                L_fft: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Random Dirichlet draws (as a cheap global search)."""
    while time.time() < deadline:
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


def _smgd(a0: np.ndarray,
          n: int,
          L_corr: int,
          deadline: float,
          best_score: float,
          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Soft‑max (smoothed‑max) gradient descent.
    The temperature (α) is increased slowly so that the objective
    converges to the true maximum of the convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score

    # schedule parameters
    alpha = 10.0          # initial inverse temperature
    alpha_max = 500.0
    eta0 = 0.3            # initial stepsize
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- convolution ----------
        Af = np.fft.rfft(a, n=L_corr)
        conv_full = np.fft.irfft(Af * Af, n=L_corr)[: 2 * n - 1]

        max_conv = conv_full.max()
        cur_sc = 2.0 * n * max_conv
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SMGD iter={it:,}  score={cur_sc:.6f}")

        # ---------- soft‑max weights ----------
        # stable exponentiation
        shift = max_conv
        exp_vals = np.exp(alpha * (conv_full - shift))
        w = exp_vals / exp_vals.sum()

        # ---------- gradient of the smooth max ----------
        # gradient = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]
        Wf = np.fft.rfft(w_rev, n=L_corr)
        grad_full = np.fft.irfft(Af * Wf, n=L_corr)

        # pick the required slice: grad_i = 2 * grad_full[(2n‑2)‑i]
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---------- update ----------
        eta = eta0 / np.sqrt(it + 1.0)
        a_new = a - eta * grad
        a = _project_simplex(a_new)

        # occasional random jitter to escape plateaux
        if rng.random() < 0.004:
            eps = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

        # raise temperature slowly (more emphasis on the true max)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.01)

    return best_a, best_sc


def _max_subgradient_descent(a0: np.ndarray,
                             n: int,
                             L_fft: int,
                             deadline: float,
                             best_score: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Direct sub‑gradient descent on the true maximum of the convolution.
    The sub‑gradient is obtained from the active convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score
    eta0 = 0.04
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _self_conv_full(a, L_fft)
        max_conv = conv.max()
        cur_sc = 2.0 * n * max_conv
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] MaxSubGrad iter={it:,}  score={cur_sc:.6f}")

        # locate the active index k_max and construct its sub‑gradient
        k_max = int(conv.argmax())
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)

        grad = np.zeros_like(a)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]          # ∂/∂a_i (conv_kmax)

        # stepsize schedule
        eta = eta0 / np.sqrt(it + 1.0)
        a_new = a - eta * grad
        a = _project_simplex(a_new)

        # tiny random perturbation every now and then
        if rng.random() < 0.01:
            eps = rng.normal(scale=0.0008, size=n)
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_sc


def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Light post‑processing: convex mixing with the uniform vector and a
    cheap random‑pair hill‑climb.
    """
    best_a = a.copy()
    best_sc = best_score
    uniform = np.full(n, 1.0 / n)

    # convex mixture with uniform – sweep λ from 0 to 1
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill‑climb i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_sc

    """

    """
    global start_time

    n = 4096

    # optional global starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
