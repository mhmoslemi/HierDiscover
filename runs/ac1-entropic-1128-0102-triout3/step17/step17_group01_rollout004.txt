"""
Propose a step‑function (non‑negative vector summing to 1) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

using a multi‑stage optimisation:

1️⃣  Diverse seeds – uniform, asymmetric parametric families and many
    random Dirichlet draws.  The K‑best seeds are kept.

2️⃣  Smooth‑max Adam – minimise a log‑sum‑exp surrogate of the
    convolution while gradually raising the temperature β.
    Adam updates are projected onto the simplex.

3️⃣  Exponentiated‑gradient polishing on the true L∞‑objective.

4️⃣  Projected sub‑gradient descent (classic polishing).

5️⃣  Randomised hill‑climbing:
    * swap two positions,
    * transfer a fraction of mass between two positions,
    * occasional random perturbations.

All phases respect the simplex constraint and stop before the
time‑budget expires (default 1000 s).  The best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2· Σ_j a_j p_{i+j}.
    Implemented as convolution of a_rev with p.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast objective (exact L∞‑objective)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (used by exp‑grad / sub‑grad)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (the argmax)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Parametric families (used to generate seeds)
# ----------------------------------------------------------------------


def _make_asymmetric_shape(n: int, r: float, pL: float, pR: float) -> np.ndarray:
    """Geometric × power shape with independent left/right exponents,
       normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0               # 1 … n
    rev = n - i                 # n … 1
    log_shape = i * np.log(r) + pL * np.log(idx) + pR * np.log(rev)
    log_shape -= np.max(log_shape)       # for numerical stability
    shape = np.exp(log_shape)
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (true L∞‑objective)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates on the exact L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        # occasional jitter to escape plateaux
        if it % 4000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (classic polishing)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the L∞‑objective with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # small probability to accept a worse move (simulated annealing)
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score

                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v_vec / (1.0 - beta2 ** t)
                step = adam_lr * m_hat / (np.sqrt(v_hat) + eps)

                a_candidate = a - step
                a_candidate = _project_simplex(a_candidate)

                cand_score = _eval_fast(a_candidate, L_fft)

                # accept if improving, otherwise occasional uphill move
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                else:
                    prob = np.exp(-(cand_score - cur_score) * 5.0 / (np.sqrt(t) + 1.0))
                    if rng.random() < prob:
                        a = a_candidate
                        cur_score = cand_score

                # update pool entry and global best
                if cand_score < pool_scores[idx]:
                    pool_scores[idx] = cand_score
                    pool[idx] = a_candidate.copy()
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time() - start:.2f}s] Adam β={beta} it={t} score={cand_score:.6f}")

                if t >= max_iter_per_vec:
                    break

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 120, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent (classic polishing).
    # ------------------------------------------------------------------
    if time.time() < deadline - 70:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 70, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣  Randomised hill‑climbing (swaps + mass transfers).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 5a – swap two positions
        i, j = rng.integers(n, size=2)
        if i != j:
            cand = best_seq.copy()
            cand[i], cand[j] = cand[j], cand[i]
            new_score = _eval_fast(cand, L_fft)
            if new_score < best_score:
                best_seq = cand
                best_score = new_score
                print(f"[{time.time() - start:.2f}s] swap hill‑climb score={new_score:.6f}")
                continue

        # 5b – transfer a fraction of mass from src to dst
        src = rng.integers(n)
        if best_seq[src] > 1e-15:
            dst = rng.integers(n)
            if src != dst:
                # allow up to 1 % of the source mass to move
                eps = best_seq[src] * rng.random() * 0.01
                cand = best_seq.copy()
                cand[src] -= eps
                cand[dst] += eps
                new_score = _eval_fast(cand, L_fft)
                if new_score < best_score:
                    best_seq = cand
                    best_score = new_score
                    print(f"[{time.time() - start:.2f}s] transfer hill‑climb score={new_score:.6f}")
                    continue

        # 5c – occasional jitter to shake the vector
        if rng.random() < 0.002:
            eps = 1e-4 * rng.random(n) * best_seq
            cand = np.maximum(best_seq + eps, 0.0)
            cand = cand / cand.sum()
            new_score = _eval_fast(cand, L_fft)
            if new_score < best_score:
                best_seq = cand
                best_score = new_score
                print(f"[{time.time() - start:.2f}s] jitter hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
