"""
Propose a step‑function (probability vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣  **Parametric sweep** – explore the two‑parameter family
    a_i ∝ (i+1)^p · r^i (geometric × power).  The best candidate
    from the sweep is kept as a warm start.

2️⃣  **Smoothed‑gradient descent** – minimise a smooth approximation
    of the max‑convolution using the soft‑max (log‑sum‑exp) trick.
    The gradient of the smoothed objective can be written in closed
    form and evaluated with a pair of FFTs per iteration.  The
    smoothing parameter α is increased gradually (continuation
    method) while a projected gradient step (onto the simplex) is
    performed.  This often escapes the plateaus that trap a pure
    sub‑gradient method.

3️⃣  **Final tiny‑mass hill‑climb** – ordinary simulated‑annealing‑style
    moves that transfer a few‑percent of the mass from one index to
    another are applied until the time budget is exhausted.

All phases respect a 1000 s budget and finally return a plain Python
list of floats (the vector is normalised to sum 1).  The code uses only
NumPy and the standard library.
"""

import time
import numpy as np
from scipy import sparse

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    L = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: n + m - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (fast) – assumes the sum is ≈ 1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Family a_i ∝ (i+1)^p · r^i  (geometric × power)
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """
    Build a vector a_i ∝ (i+1)^p · exp(-lam·i)   (lam = -log r).
    The result is normalised to unit L1 norm.
    """
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)          # numeric stabilisation
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smoothed‑gradient descent (continuation on α)
# ----------------------------------------------------------------------


def _smooth_grad_descent(a0: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the L∞‑objective.
    Uses soft‑max weights w_k ∝ exp(α·conv_k) and a projected gradient
    step on the simplex.  α is increased slowly (continuation).
    """
    a = a0.copy()
    step0 = 0.5
    alpha = 0.5           # start very smooth
    alpha_max = 30.0
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑grad it={it:,}  "
                  f"score={cur_score:.6f}  alpha={alpha:.3f}")

        # ----- smooth max via soft‑max -----
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)      # prevent overflow
        w /= w.sum()                         # probabilities, sum=1

        # ----- gradient of the smooth objective -----
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w)       # length 3n‑2
        g = 2.0 * conv_rev[n - 1: n - 1 + n]  # slice of length n

        # ----- step size (decreasing) -----
        step = step0 / np.sqrt(it)

        # ----- projected gradient update -----
        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new

        # ----- increase smoothness -----
        if alpha < alpha_max:
            alpha = min(alpha * 1.01, alpha_max)

        # ----- occasional jitter to jump out of plateaus -----
        if it % 500 == 0:
            eps = 1e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final random hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------


def _final_hill_climb(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very small random mass‑transfer moves with a simple Metropolis rule."""
    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 5e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move ≤0.2 % of the source mass
        eps = a[src] * rng.random() * 2e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={new_score:.6f}")
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter (height_sequence_1) exists, try it.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep over (λ, p) – ≈70 s.
    # --------------------------------------------------------------
    sweep_deadline = start_time + 70.0
    min_lambda = 4e-05          # r≈0.99996
    max_lambda = 1.5e-03        # r≈0.9985
    lam_grid = np.linspace(min_lambda, max_lambda, 300)   # 300 points
    p_grid = np.linspace(-6.0, 3.0, 91)                    # step 0.1

    i_arr = np.arange(n, dtype=np.float64)
    log_idx = np.log(i_arr + 1.0)

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * i_arr
        for p in p_grid:
            log_shape = base + p * log_idx
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6f} "
                      f"p={p:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smoothed‑gradient descent (continuation on α).
    # --------------------------------------------------------------
    if time.time() < deadline - 250.0:
        grad_deadline = deadline - 250.0
        best_seq, best_score = _smooth_grad_descent(best_seq, n, L_fft,

                                                    best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 3️⃣  Final tiny random hill‑climb.
    # --------------------------------------------------------------
    if time.time() < deadline:

                                                best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Normalisation and return.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
