"""
Optimisation of a non‑negative step‑function for the objective

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm works directly on the convex formulation
f(a)=max_i (a*a)_i.  Instead of the non‑smooth max we use a
smooth soft‑max (log‑sum‑exp) with a temperature parameter α.
For a fixed α the objective is a convex combination of the
individual convolution entries, whose gradient can be written
as a simple correlation:

    ∇_a  Σ_i w_i (a*a)_i = 2·corr(a, w) ,

where w_i = exp(α·conv_i)/∑_j exp(α·conv_j) are the soft‑max
weights.  The correlation is computed by a single FFT‑based
convolution with the reversed weight vector.

We start from a feasible point (the previous best solution,
or a uniform distribution), run a long Adam‑style projected
gradient descent while slowly increasing α (so the smoothed
objective approaches the true max), and finish with a tiny
random hill‑climb to capture any remaining improvement.

The whole search respects a 1000 s budget and returns the best
sequence found as a plain Python list of floats (normalised to
sum = 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)  # sum(a) is one after projection


def _weighted_gradient(a: np.ndarray, w: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Gradient of Σ_i w_i·conv_i(a)  = 2·corr(a, w) .
    w has length 2·n‑1, a has length n.
    The correlation is obtained via a convolution with the reversed w.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reverse the weight vector
    Af = np.fft.rfft(a, n=L_fft)
    Wf = np.fft.rfft(w_rev, n=L_fft)
    full = np.fft.irfft(Af * Wf, n=L_fft)          # length = 3·n‑2 (plus padding)
    # The needed slice is indices n‑1 … 2·n‑2 (inclusive) → Python slice n‑1:2·n‑1
    grad = 2.0 * full[(n - 1) : (2 * n - 1)]
    return grad

# Main optimisation entry point
# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12          # a tiny safety margin

    # --------------------------------------------------------------
    # Problem size – use the length of the global starter if available,
    # otherwise fall back to 4096 (a good trade‑off).
    # --------------------------------------------------------------

        n = starter.size

    except Exception:
        n = 4096

    # FFT length that can accommodate both 2·n‑1 and 3·n‑2 convolutions.
    L_fft = _next_pow2(3 * n)

    # --------------------------------------------------------------
    # Baseline – evaluate the starting point.
    # --------------------------------------------------------------
    best_seq = a0.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] start baseline  score={best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999

    # Initialise moments.

    # Soft‑max temperature schedule.

    alpha_mul = 1.0025

    # --------------------------------------------------------------

        # ----- increase temperature slowly (approaches true max) -----
        alpha = min(alpha_max, alpha * alpha_mul)

    # --------------------------------------------------------------
    # Final random hill‑climb (tiny mass‑transfer moves).
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move at most 0.2 % of the source mass
        eps_move = best_seq[src] * rng.random() * 2e-3
        a_prop = best_seq.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move
        # re‑normalise (projection onto simplex) – the move already preserves the sum,
        # but projection also removes any tiny negative drift caused by rounding.
        a_prop = _project_simplex(a_prop)
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            best_seq = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={new_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
