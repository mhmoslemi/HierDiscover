def _resample_to_length(seq, n):
    m = len(seq)
    if m == n:
        return seq.copy()
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, seq)
```

"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm uses a much larger search space (n=8192) and several
diverse optimisation stages:

1️⃣  Generate many diverse seeds:
    – uniform, a possibly provided global starter (resampled),
    – random 3‑parameter geometric × power shapes a_i ∝ r^i·(i+1)^p·(n‑i)^q,
      optionally mixed with the uniform distribution,
    – Dirichlet draws with several concentration parameters,
    – random mixtures of a few exponentials.
   Keep the K best seeds (K=10).

2️⃣  Refine each seed with Adam on a smooth‑max surrogate
    f_β(a)=log‑sum‑exp(β·conv(a,a)).
    The temperature β is increased from 1 up to 2048.
    After each Adam step the vector is projected onto the simplex.

3️⃣  Polishing with an exponentiated‑gradient phase that works directly
    on the true L∞ objective, followed by a classic projected sub‑gradient
    descent.

4️⃣  A simulated‑annealing hill‑climb that repeatedly redistributes the
    total mass of three randomly chosen positions; it accepts uphill
    moves with a temperature schedule.

All stages respect the simplex constraint, stop before the 1000 s
budget expires and finally return the best sequence (normalised to sum = 1)
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max (soft‑max) surrogate.
    Returns g_i = 2 * Σ_j a_j p_{i+j}.
    Implemented as convolution of a_rev with p.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Fast exact objective (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape3(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Geometric × power × tail‑power shape:
        a_i ∝ r**i * (i+1)**p * (n-i)**q
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)                 # avoid overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Convex combination of `shape` with the uniform distribution.
    `lam` ∈ [0,1] is the weight of the uniform part.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _resample_to_length(seq: np.ndarray, n: int) -> np.ndarray:
    """
    Linear interpolation of a 1‑D sequence to a new length `n`.
    The result is normalised to sum = 1.
    """
    m = seq.shape[0]
    if m == n:
        out = seq.copy()
    else:
        x_old = np.linspace(0.0, 1.0, m)
        x_new = np.linspace(0.0, 1.0, n)
        out = np.interp(x_new, x_old, seq)
    out = np.maximum(out, 0.0)
    total = out.sum()
    if total > 0.0:
        out /= total
    else:
        out = np.full(n, 1.0 / n, dtype=np.float64)
    return out


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (same as reference)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated‑gradient) updates on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

                    cur_score = cand_score
                else:
                    prob = np.exp(-(cand_score - cur_score) * 5.0 / (np.sqrt(t) + 1.0))
                    if rng.random() < prob:
                        a = a_candidate
                        cur_score = cand_score

                # update the pool entry
                if cand_score < pool_scores[idx]:
                    pool_scores[idx] = cand_score
                    pool[idx] = a_candidate.copy()

                # update global best
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time() - start:.2f}s] Adam β={beta} it={t} score={cand_score:.6f}")

                if t >= max_iter_per_vec:
                    break

                # occasional jitter to escape plateaus
                if t % 2000 == 0:
                    eps_arr = 1e-4 * rng.random(n) * a
                    a = np.maximum(a + eps_arr, 0.0)
                    a = a / a.sum()

    # ------------------------------------------------------------------
    # 3️⃣ Exponentiated‑gradient polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 120, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣ Projected sub‑gradient descent polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 70:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 70, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣ Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    curr = best_seq.copy()
    curr_score = best_score
    T = 0.001 * best_score   # initial temperature
    decay = 0.9999
    while time.time() < deadline:
        # pick three distinct indices and reshuffle their total mass
        idxs = rng.choice(n, size=3, replace=False)
        total = curr[idxs].sum()
        if total == 0.0:
            continue
        new_props = rng.dirichlet(np.ones(3)) * total
        prop = curr.copy()
        prop[idxs] = new_props
        new_score = _eval_fast(prop, L_fft)

        if new_score < curr_score:
            curr, curr_score = prop, new_score
            if new_score < best_score:
                best_seq, best_score = prop.copy(), new_score
                print(f"[{time.time() - start:.2f}s] anneal improved to {new_score:.6f}")
        else:
            prob = np.exp(-(new_score - curr_score) / max(T, 1e-12))
            if rng.random() < prob:
                curr, curr_score = prop, new_score

        T *= decay

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
