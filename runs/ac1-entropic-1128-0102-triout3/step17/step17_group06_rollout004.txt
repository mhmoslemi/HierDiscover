    n = a.shape[0]
    i0 = max(0, max_idx - n + 1)
    i1 = min(n-1, max_idx)
    # compute g for i in range i0..i1

def compute_gradient(a: np.ndarray, max_idx: int) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)

    if i0 < 0:

    j = max_idx - idx
    g[idx] = 2.0 * a[j]

"""
Propose a non‑negative step function that minimises

        score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases:

1️⃣ Warm‑start  
   – A uniform baseline, optional global starter, a few Dirichlet draws,
     and a fine scan of the Beta(p,p) family (including small mixtures with
     the uniform vector).  
   The best candidate of this cheap search is taken as the start point.

2️⃣ Projected gradient descent on the simplex  
   – The current convolution is examined, the index `k*` where it attains its
     maximum is found, and the sub‑gradient of `conv_k*` w.r.t. the
     probability vector is `g_i = 2·a_{k*‑i}` (zero otherwise).  
   – A step `Δ = β·median(a_i/g_i)` is taken in the direction `-g`,
     the result is clipped, mixed a tiny amount with the uniform vector
     (`ε`–smoothing) and projected back onto the simplex.
   – If the step improves the score we keep it, otherwise we shrink the
     learning rate `β`.  This loop runs until only a small fraction of the
     total time budget remains.

3️⃣ Final random hill‑climb  
   – Random pairwise mass transfers (up to 1 % of the source bin) are tried,
     always keeping the vector on the simplex.  Accept only improving moves.
   – This cheap local search is able to squeeze a little more out of the
     distribution found by the gradient phase.

All arithmetic is performed on a normalised vector (∑a = 1); the evaluator
is scale‑invariant, so the final list can be returned directly.  The routine
stops well before the 1000 s budget expires and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator for a normalised vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _beta_shape(n: int, p: float) -> np.ndarray:
    """
    Discretised Beta(p,p) density on the grid i = 0,…,n‑1
    (x_i = (i+0.5)/n), normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid 0^negative – the grid never hits 0 or 1 exactly
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative, sum = 1).  Standard O(n) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_gradient(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry at `max_idx`:
    g_i = 2·a_{max_idx‑i} for indices where both indices lie in [0,n).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i0 = max(0, max_idx - n + 1)
    i1 = min(n - 1, max_idx)
    idx = np.arange(i0, i1 + 1)
    g[idx] = 2.0 * a[max_idx - idx]
    return g


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    progress.  The returned list sums to roughly one (scale does not matter
    for the evaluator).
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.05

    # --------------------------------------------------------------
    # Problem size – a power‑of‑two length gives fast FFTs.
    # --------------------------------------------------------------
    n = 8192                     # larger than before → finer discretisation
    L_fft = _next_pow2(2 * n - 1)

    # Pre‑computed uniform vector (used many times)
    uniform = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 0️⃣  Baseline (uniform) and optional global starter
    # --------------------------------------------------------------
    best_a = uniform.copy()
    best_score = _score_of(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score = {best_score:.6f}")

    # Optional globally supplied starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter,
                                     (0, n - starter.size),
                                     constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _score_of(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Warm‑start (Dirichlet draws + Beta‑family scan)
    # --------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.07 * budget_s)  # ~7 % of total time
    rng = np.random.default_rng()
    # Dirichlet draws
    while time.time() < warm_deadline:
        # concentration drawn log‑uniformly in [0.01, 50]
        log_c = rng.uniform(np.log(0.01), np.log(50.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _score_of(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet warm‑start → score = {sc:.6f}")

    # Beta(p,p) scan – focus on the neighbourhood of the arcsine law (p≈0.5)
    print(f"[{time.time() - start_time:.2f}s] Scanning Beta‑family")
    p_vals = np.linspace(0.30, 1.20, 35)          # 0.30 … 1.20  (35 values)
    lam_vals = np.linspace(0.0, 0.20, 11)        # 0.0 … 0.20  (step 0.02)
    for p in p_vals:
        shape = _beta_shape(n, p)

        # pure shape
        sc = _score_of(shape, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, shape.copy()
            print(f"[{time.time() - start_time:.2f}s] pure p={p:.3f} → score = {sc:.6f}")

        # mixtures with uniform (tiny uniform mass smoothes discretisation artefacts)
        for lam in lam_vals:
            cand = lam * shape + (1.0 - lam) * uniform
            cand = cand / cand.sum()
            sc = _score_of(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] p={p:.3f} λ={lam:.2f} → score = {sc:.6f}")

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2️⃣  Projected gradient descent on the simplex
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Gradient‑descent phase")
    a = best_a.copy()
    beta = 0.9            # initial step‑fraction (applied to a typical a/g ratio)
    eps = 0.001           # uniform smoothing weight
    gd_deadline = min(deadline, start_time + 0.85 * budget_s)  # keep ~15 % for final hill‑climb

    while time.time() < gd_deadline:
        # compute self‑convolution and locate the worst index
        conv = _conv_self_fft(a, L_fft)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = _compute_gradient(a, max_idx)

        mask = g > 0.0
        if not np.any(mask):
            break  # nothing to move

        # typical admissible step size (median of a_i / g_i)
        ratios = a[mask] / g[mask]
        step = beta * np.median(ratios)

        # safeguard – avoid a step that would erase all mass from an index
        if step <= 0.0:
            break

        cand = a.copy()
        cand[mask] -= step * g[mask]
        cand = np.maximum(cand, 0.0)

        # a tiny uniform mixing prevents the support from collapsing completely
        cand = (1.0 - eps) * cand + eps * uniform
        total = cand.sum()
        if total == 0.0:
            break
        cand /= total

        sc = _score_of(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] GD improve → score = {sc:.6f}")
            # keep the learning rate unchanged on success
        else:
            # no improvement → shrink the step size a little
            beta *= 0.7
            if beta < 0.02:
                beta = 0.02
            # occasionally increase smoothing to escape flat regions
            eps = min(eps * 1.02, 0.005)

        # occasionally re‑inject a dash of uniform mass to keep diversity
        if time.time() < gd_deadline and rng.random() < 0.005:
            a = (0.99 * a) + (0.01 * uniform)
            a /= a.sum()

    # --------------------------------------------------------------
    # 3️⃣  Final hill‑climb (random pairwise mass transfers)
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Hill‑climb refinement")
    a = best_a.copy()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # Transfer at most 1 % of the source bin (or the whole bin if tiny)
        max_delta = min(a[i] * 0.5, 0.01)   # keep steps modest but not too small
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        total = cand.sum()
        if total == 0.0:
            continue
        cand /= total

        sc = _score_of(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improve → score = {sc:.6f}")

    # --------------------------------------------------------------
    # Normalisation & return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
