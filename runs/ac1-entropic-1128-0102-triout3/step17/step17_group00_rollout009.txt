    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta2 = 0.999
    eps = 1e-8
    it = 0
    lr0 = 1.0  # maybe initial learning rate scaled
    # alpha schedule:
    alpha0 = 5.0

        conv = _conv_self_fft(a, L_fft)

        alpha = min(alpha_max, alpha0 + alpha_inc_per_it * it)

        # to avoid overflow, subtract max

        exp_term = np.exp(alpha * conv - maxv)
        w = exp_term / np.sum(exp_term)

"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in several stages:

1️⃣  **Initialisation** – start from a good known construction (if available),
    otherwise use a linear interpolation of the previous best sequence and
    also try a quick parametric sweep over the family
        a_i ∝ (i+1)^p · exp(-λ·i)
    on a finer grid for a larger problem size (n = 8192).

2️⃣  **Projected sub‑gradient descent** – classic sub‑gradient updates on the
    exact L∞ objective with a back‑tracking line‑search.  This phase runs for
    the first half of the remaining time budget.

3️⃣  **Smoothed Adam optimisation** – the hard max is replaced by a
    temperature‑controlled soft‑max (log‑sum‑exp).  An Adam update is applied
    to the resulting smooth objective; the temperature (α) is slowly increased
    so that the optimisation tracks the true L∞ problem more closely.

4️⃣  **Final random hill‑climb** – tiny mass‑transfer moves are accepted when
    they improve the exact objective.  This extracts any last gain before the
    deadline.

All vectors are kept on the probability simplex (non‑negative, sum = 1) and
the best vector seen throughout the run is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT (length x+y‑1)."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: x.shape[0] + y.shape[0] - 1]


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (assumes sum ≈ 1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution at a specific index
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at convolution index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]               # diagonal handled automatically
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (original implementation)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.35
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # step diminishes slowly – this helps fine tuning
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(10):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                      f"step={step:.5e} score={cand_score:.7f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        if it % 2000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed Adam optimisation (soft‑max approximation of max)
# ----------------------------------------------------------------------


def _adam_smooth_phase(a0: np.ndarray,
                       L_fft: int,
                       deadline: float,
                       start_time: float,
                       best_seq: np.ndarray,
                       best_score: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam on a smooth approximation of max(conv).  α (temperature) increases."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # convolution lengths for the auxiliary term
    L_aux = _next_pow2(3 * n - 2)          # length for a * w_rev
    max_alpha = 200.0
    alpha_start = 5.0
    # increase α linearly – the exact schedule is not critical
    alpha_inc = (max_alpha - alpha_start) / max(1, int((deadline - start_time) * 1000))

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution a*a
        conv = _conv_self_fft(a, L_fft)

        # temperature schedule
        alpha = min(max_alpha, alpha_start + alpha_inc * it)

        # soft‑max weights (stable)
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w /= w.sum()                     # w has length 2·n‑1

        # ---- gradient of the smoothed objective ----
        # term1 = sum_k w_{j+k} * a_k   (valid convolution of a with reversed w)
        w_rev = w[::-1]
        full = _conv_full_fft(a, w_rev, L_aux)          # length 3·n‑2
        # extract term1_j = full[(2*n-2) - j] for j=0..n‑1
        term1 = full[(2 * n - 2) - np.arange(n)]

        # term2 = a_j * sum_{i=j}^{j+n-1} w_i  (sliding window sum of w)
        cs = np.concatenate([[0.0], np.cumsum(w)])      # length 2·n
        sliding = cs[n:] - cs[:n]                        # length n
        term2 = a * sliding

        g = term1 + term2

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate decays like 1/√t
        lr = 0.5 / (np.sqrt(it) + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # evaluate exact objective
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam it={it:,} "
                  f"α={alpha:.2f} score={score:.7f}")

        # tiny random perturbation every few thousand steps
        if it % 5000 == 0:
            eps_noise = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps_noise, 0.0)
            a = a / a.sum()

    return best_seq, best_score

                                                    best_seq, best_score, rng)

    # --------------------------------------------------------------

    else:

    # print final result with the *official* evaluator for transparency
    try:
        final_val = evaluate_sequence(best_seq.tolist())
    except Exception:
        final_val = None
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.7f}"
          f"{' (validated)' if final_val is not None else ''}")

    return [float(v) for v in best_seq]
