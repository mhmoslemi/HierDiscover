Implementation of projection onto simplex (standard algorithm):

def project_onto_simplex(v):
    # ensures components sum to 1 and are nonnegative.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    """Project vector v onto the probability simplex."""
    # Implementation from https://gist.github.com/...
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _beta_bin_distribution(n, p):
    import scipy.special as sp
    x = np.arange(n+1, dtype=np.float64) / n  # points from 0/n to n/n
    cdf = sp.betainc(p, p, x)  # shape (n+1,)
    a = np.diff(cdf)  # length n
    # ensure nonneg
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total

"""
Step‑function optimizer.

The algorithm proceeds in three stages:

1. **Exact Beta‑bin discretisation** –  For a symmetric Beta(p,p) distribution
   we compute the exact mass of each bin by using the regularised incomplete
   beta function (scipy.special.betainc).  This removes the discretisation
   error of the previous code that evaluated the density at mid‑points.
   A fine grid of shape parameters p≈0.5 (including a small mixture with the
   uniform vector) is scanned; the best‑scoring vector is kept.

2. **Projected sub‑gradient descent on the L∞‑norm** –  With the current best
   vector `a` (∑a=1) we repeatedly compute its self‑convolution, locate the
   index `k` of the maximal entry, and take a sub‑gradient
   `g_i = 2·a_{k‑i}` (zero outside the support).  A step `a←a−η·g` is taken
   and the result is projected back onto the probability simplex.  The step
   size η decays slowly; this directly minimises `max(a*a)` while respecting
   the non‑negativity and normalisation constraints.

3. **Deterministic hill‑climb fine‑tuning** –  Using the current best vector,
   we repeatedly move a tiny amount of mass from the index with the largest
   contribution to the offending convolution peak to the index with the
   smallest contribution, accepting only improving moves.  This cheap local
   search pushes the score a little further when we are already close to a
   discrete optimum.

All operations stay on the simplex, therefore the evaluator’s clipping does
not affect the result.  The routine respects the supplied time budget (default
1000 s) and prints progress.  The returned list contains non‑negative entries
that sum to ≈1.
"""

import time
import numpy as np
import scipy.special as sp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a⋆a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán, "Projection onto the probability simplex:
    #   An efficient algorithm with a simple proof, and an application", 2013.
    n = v.shape[0]
    u = np.sort(v)[::-1]                      # sorted descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_bin_shape(n: int, p: float) -> np.ndarray:
    """
    Exact discretisation of the symmetric Beta(p,p) on n equal bins.
    a_i = CDF((i+1)/n) – CDF(i/n), where the CDF is the regularised
    incomplete beta function.  The result is normalised to sum 1.
    """
    # points at bin boundaries 0/n, 1/n, …, n/n
    x = np.arange(n + 1, dtype=np.float64) / n
    # regularised incomplete beta = CDF for Beta(p,p)
    cdf = sp.betainc(p, p, x)
    a = np.diff(cdf)                     # probability mass of each bin
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator for a normalised vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and prints
    progress.  The returned list contains non‑negative entries that sum to ~1.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # --------------------------------------------------------------
    # Problem size – a power of two is convenient for FFTs
    # --------------------------------------------------------------
    n = 8192                      # larger than the previous 4096 to reduce the theoretical bound
    L_fft = _next_pow2(2 * n - 1)   # length for a*a convolution

    # --------------------------------------------------------------
    # 0️⃣  Baseline (uniform) and optional global starter
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _score_of(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score = {best_score:.6f}")

    # Global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _score_of(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Scan the exact Beta‑bin family (fine grid around p=0.5)
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Scanning exact Beta-bin family")
    # dense grid near the optimal p=0.5
    p_vals = np.linspace(0.30, 0.70, 81)   # step = 0.005
    for p in p_vals:
        shape = _beta_bin_shape(n, p)

        sc = _score_of(shape, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, shape.copy()
            print(f"[{time.time() - start_time:.2f}s] pure p={p:.3f} → score = {sc:.6f}")

        # convex mixtures with the uniform vector – only very small λ are useful
        for lam in np.linspace(0.0, 0.20, 21):   # 0.0 … 0.20 in steps of 0.01
            cand = lam * shape + (1.0 - lam) * uniform
            cand = cand / cand.sum()
            sc = _score_of(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] p={p:.3f} λ={lam:.2f} → score = {sc:.6f}")

        # early exit if time runs out
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent on the maximum convolution entry
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Sub‑gradient descent")
    a = best_a.copy()
    conv = _conv_self_fft(a, L_fft)               # current convolution
    max_b = conv.max()
    best_score = 2.0 * n * max_b
    iteration = 0
    max_iters = 500000   # upper bound – the loop will stop earlier by the deadline

    while time.time() < deadline and iteration < max_iters:
        # location of the current worst peak
        k = int(np.argmax(conv))

        # sub‑gradient for the max‑norm (g_i = 2·a_{k-i} where defined)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = a[idx[mask]]
        g *= 2.0

        # step size schedule – slowly decreasing
        frac = 1.0 - iteration / max_iters
        step = 0.01 * frac + 1e-6   # ensures a tiny positive step even at the end

        # gradient step + projection onto the simplex
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        # evaluate candidate
        conv_cand = _conv_self_fft(a_candidate, L_fft)
        score_cand = 2.0 * n * conv_cand.max()

        if score_cand < best_score - 1e-12:       # accept only strict improvements
            best_score = score_cand
            best_a = a_candidate.copy()
            a = a_candidate
            conv = conv_cand
            if iteration % 500 == 0:
                print(f"[{time.time() - start_time:.2f}s] iter {iteration:5d} → score = {best_score:.6f}")

        # otherwise keep the old a (no update)
        iteration += 1

    # --------------------------------------------------------------
    # 3️⃣  Deterministic hill‑climb (pairwise mass transfers)
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Hill‑climb refinement")
    a = best_a.copy()
    rng = np.random.default_rng()
    while time.time() < deadline:
        # recompute convolution for the current a
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))

        # contributions of each index to the offending peak
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]]

        # choose donor (largest contribution) and receiver (smallest contribution)
        donor = int(np.argmax(contrib))
        receiver = int(np.argmin(contrib))

        if a[donor] <= 1e-12:
            # nothing to move – break to avoid endless loop
            break

        delta = min(a[donor], 5e-4)               # tiny transfer
        cand = a.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        # projection not needed – mass is conserved and non‑negativity is kept

        conv_cand = _conv_self_fft(cand, L_fft)
        score_cand = 2.0 * n * conv_cand.max()
        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best_a = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improve → score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalisation & return
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
