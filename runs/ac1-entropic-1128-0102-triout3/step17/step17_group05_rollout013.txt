"""
A fast, multi‑stage search for a step‑function that minimises

        2 * n * max(conv(a,a)) / (sum(a)**2)

The code works entirely on the probability simplex (sum = 1, a_i ≥ 0) because
the objective is homogeneous.  The search consists of

1️⃣  Very dense parametric sweeps (geometric, power, symmetric exponential,
    forward / reversed geometric mixes).  The grids are deliberately fine
    (hundreds of points) – the cost is still tiny because each evaluation
    uses a single FFT of length ≈ 8 k.

2️⃣  A large number of random Dirichlet draws (up to 150 k samples).  The
    concentration parameter is chosen log‑uniformly; we also draw from
    “biased” Dirichlets where the α‑parameters decay exponentially.

3️⃣  A projected sub‑gradient descent that works directly on the non‑smooth
    objective.  At every iteration the gradient is the (averaged) gradient
    of the currently active convolution entry (or entries); a simple
    diminishing step‑size together with a small random jitter lets the
    method escape shallow plateaus.  The descent is started from the best
    few candidates found in the previous stages.

All heavy work is timed – the routine stops automatically before the
deadline (default 1000 s) and returns the best sequence it has seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a (length 2·n‑1) using a fast FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    #   from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# 1️⃣  Parametric families (dense sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)

    # extremely fine log‑grid for the decay factor (eps = 1‑r)
    eps_grid = np.logspace(-6, -0.2, 400)          # 400 points
    lam_grid = np.linspace(0.0, 1.0, 21)           # mixing with uniform
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s

        # also test mixing a forward‑/reversed‑geometric pair (high at both ends)
        rev_geo = geo[::-1]

        for lam in lam_grid:
            # forward only
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.2e} lam={lam:.2f} score={sc:.6f}")

            # symmetric ends (mix forward & reverse)
            seq2 = lam * np.full(n, 1.0 / n) + (1.0 - lam) * (0.5 * (geo + rev_geo))
            seq2 /= seq2.sum()

                print(f"[{time.time() - start_time:.2f}s] GeoSym eps={eps:.2e} lam={lam:.2f} score={sc:.6f}")

    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(1, n + 1, dtype=np.float64)  # 1‑based to avoid zero^p
    p_grid = np.linspace(-4.0, 4.0, 401)        # 401 points
    for p in p_grid:
        if time.time() >= deadline:
            break
        seq = np.power(idx, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential decay from the centre (distance‑based)."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    lam_grid = np.logspace(-6, -0.2, 250)      # 250 points
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.2e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two (possibly unequal) uniform blocks at the ends."""
    for L_frac in np.logspace(-3, -0.3, 15):        # block length as a fraction
        L = int(L_frac * n)
        if L < 1 or 2 * L > n:
            continue
        for w in np.linspace(0.05, 0.95, 19):        # how the total mass is split
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_triangular(n, best_seq, best_score, deadline):
    """Symmetric triangular shape (peak in the centre)."""
    idx = np.arange(n, dtype=np.float64)
    tri = np.minimum(idx + 1, n - idx)
    seq = tri.astype(float)
    seq /= seq.sum()
    sc = _objective_score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Triangular score={sc:.6f}")

    # also a reversed triangle (peak at the edges)
    rev = seq[::-1]
    seq2 = 0.5 * (seq + rev)          # smooth edges
    seq2 /= seq2.sum()
    sc = _objective_score(seq2)
    if sc < best_score:
        best_score, best_seq = sc, seq2.copy()
        print(f"[{time.time() - start_time:.2f}s] TriangularRev score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws (heavy‑weight sampling)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    # number of trials is bounded by remaining time
    max_trials = 200_000
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # 50 % of draws are plain Dirichlet, 50 % are “biased” Dirichlet
        if rng.random() < 0.5:
            # plain Dirichlet, concentration between 0.01 and 5 (log‑uniform)
            logc = rng.uniform(-2.0, 0.7)   # 10^logc ∈ [0.01, 5]
            c = 10 ** logc
            a = rng.gamma(shape=c, scale=1.0, size=n)
        else:
            # biased Dirichlet – decreasing α_i = exp(-β·i)
            beta = rng.uniform(0.0, 4.0)    # controls steepness
            alphas = np.exp(-beta * np.arange(n, dtype=np.float64))
            # add a global concentration factor
            scale = rng.uniform(0.02, 2.0)
            a = rng.gamma(shape=alphas * scale + 1e-6, scale=1.0)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet draw score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Projected sub‑gradient descent on the true non‑smooth objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0, best_seq, best_score, deadline, max_iters=800):
    """
    Projected sub‑gradient descent on
        f(a) = 2·n·max(conv(a,a))      (a lives on the simplex)
    The method uses the exact sub‑gradient (average over all active shifts)
    and a simple diminishing step‑size.  Small random jitter helps to leave
    flat plateaus.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # collect all indices that are (almost) maximal
        active = np.where(conv >= max_val - 1e-14)[0]
        p = np.zeros_like(conv)
        p[active] = 1.0 / len(active)          # uniform weight over the active set

        # gradient = 2 * conv(p , a_rev) (slice that yields length n)
        a_rev = a[::-1]
        L = _next_pow2(len(p) + n - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + n - 1]
        grad = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # diminishing step size (≈ 1/√it)
        step = 0.8 / np.sqrt(it)

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        # line‑search like acceptance – keep the move only if it improves
        new_obj = _objective_score(a_new)
        cur_obj = _objective_score(a)
        if new_obj < cur_obj - 1e-12:
            a = a_new
            if new_obj < best_score:
                best_score, best_seq = new_obj, a.copy()
                print(f"[{time.time() - start_time:.2f}s] GD it={it:,} obj={new_obj:.6f}")
        else:
            # add a tiny random jitter (helps escape flat regions)
            eps = 1e-8
            a = a + rng.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine runs three stages (dense parametric sweeps,
    massive random Dirichlet sampling, and a few runs of projected
    sub‑gradient descent) and returns the best sequence found (normalised
    to sum = 1) as a plain Python list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1) – if present
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Dense parametric sweeps (≈250 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 350.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_triangular(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Very many Dirichlet draws (≈300 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Sub‑gradient descent from the top few candidates
    # --------------------------------------------------------------
    # keep the top‑5 sequences seen so far (including the current best)
    top_k = 5
    candidates = [best_seq.copy()]
    scores = [best_score]
    # we already stored the best, now add a few extra random but promising ones
    # (sample 20 random Dirichlet draws quickly – they are cheap)
    rng = np.random.default_rng()
    for _ in range(20):
        a = rng.gamma(shape=0.3, scale=1.0, size=n)
        a /= a.sum()
        candidates.append(a)
        scores.append(_objective_score(a))

    # sort by score and keep the best `top_k`
    order = np.argsort(scores)
    candidates = [candidates[i] for i in order[:top_k]]
    scores = [scores[i] for i in order[:top_k]]

    # run a short gradient descent from each of them
    for i, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.2f}s] Starting GD from candidate #{i+1} (score={scores[i]:.6f})")
        best_seq, best_score = _subgradient_descent(cand, best_seq, best_score, deadline)

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
