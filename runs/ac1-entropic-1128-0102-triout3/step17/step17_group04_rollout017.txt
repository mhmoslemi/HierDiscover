"""
Improved search for a low‑value step‑function.

The objective used in the competition is

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

which is scale‑invariant, so we can enforce Σ a_i = 1 and simply minimise
`2 * n * max(conv)`.  The code below combines several ideas that helped to
beat the previous best (≈1.5078):

1. **Rich parametric families** – geometric, symmetric exponential,
   Gaussian‑like and power‑law shapes, together with a simple two‑block
   construction.  All families are scanned on a fine grid and the best
   candidate is kept.

2. **Random Dirichlet draws** – a large number of draws with
   concentration parameters ranging from 0.01 to 5 (log‑uniform).  This
   gives a diverse set of starting points.

3. **Smooth‑max mirror descent** – minimise a smooth upper‑bound of the
   max‑convolution using the log‑sum‑exp surrogate.  The gradient is
   obtained analytically (a single convolution).  We use an exponential
   (mirror‑descent) update, keep the iterate on the probability simplex
   and employ a momentum term to accelerate convergence.  The smoothness
   parameter β is increased from 0.2 to 200 during the run.

4. **Targeted hill‑climbing** – after the gradient phase we repeatedly
   look at the index `k` that attains the current maximum, locate the
   pair (i, j) with the largest product a_i·a_j contributing to that
   entry and move a tiny amount of probability mass from the heavy
   pair to a very light coordinate.  This local move strictly reduces
   the maximal convolution value and is accepted only when it improves
   the score.

All heavy work (convolutions) is performed with FFTs.  The routine
stops well before the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, very fast)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (uses the same formula as evaluate_sequence,
# but works on a normalised vector Σa = 1, i.e. 2·n·max(conv) )
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _max_conv_and_location(seq: np.ndarray) -> tuple[float, int]:
    """Maximum value of conv(seq,seq) and the corresponding index."""
    conv = _conv_self(seq)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Project onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient w.r.t. the probability vector
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of the max of the self‑convolution using
    log‑sum‑exp.  Returns (smooth_max, gradient) where gradient is a vector
    of the same size as `a`.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # shift for stability
    Z = exp_scaled.sum()
    w = exp_scaled / Z                         # soft‑max weights (sum to 1)

    # gradient of Σ_k w_k conv_k  w.r.t a_i :
    #   d/da_i ( Σ_k w_k Σ_j a_j a_{k-j} ) = 2 Σ_j a_j w_{i+j}
    #   (indices wrap only inside the valid range)
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)          # length 3n‑2
    start = len(w) - 1                      # = 2n‑2
    grad = 2.0 * conv_rev[start - np.arange(n)]

    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, grad


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponential (mirror‑descent) update on the simplex:
        a_i ← a_i·exp(−η·grad_i)   then renormalise.
    Guarantees a_i ≥ 0 and Σ a_i = 1.
    """
    log_factor = -eta * grad
    log_factor -= np.max(log_factor)            # stabilise
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total

def _scan_families(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)

    # ---- geometric (decreasing from the left) -----------------------
    for eps in np.logspace(-5, -1, 70):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        sc = _objective_score(geo)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] geo eps={eps:.1e} → {sc:.6f}")

    # ---- symmetric exponential (U‑shaped) ---------------------------
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] exp λ={lam:.1e} → {sc:.6f}")

    # ---- Gaussian‑like (bell) ---------------------------------------
    for sigma in np.linspace(200, 2000, 20):
        if time.time() >= deadline:
            break
        center = (n - 1) / 2.0
        seq = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] gauss σ={sigma:.0f} → {sc:.6f}")

    # ---- power‑law (both ends heavy) --------------------------------
    for p in np.linspace(0.5, 4.0, 15):
        if time.time() >= deadline:
            break
        seq = (np.sqrt(idx + 1) * np.sqrt(n - idx)) ** p
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] pow p={p:.2f} → {sc:.6f}")

    # ---- two‑block (ends) -------------------------------------------
    for L_frac in np.linspace(0.02, 0.30, 15):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in np.linspace(0.05, 0.95, 13):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] 2‑block L={L} w={w:.2f} → {sc:.6f}")

    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=2000):

    for _ in range(trials):
        if time.time() >= deadline:

        logc = rng.uniform(-2.0, 0.7)           # concentration ≈ 0.01 … 5

        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smooth‑max mirror descent with momentum
# ----------------------------------------------------------------------
def _smooth_descent(a0, deadline, best_seq, best_score):
    """Mirror‑descent on the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    beta_start, beta_end = 0.2, 200.0      # smoothness schedule
    eta = 0.5                               # nominal step size
    min_eta = 1e-12
    mom = np.zeros_like(a)                  # momentum buffer
    mom_decay = 0.9

    while time.time() < deadline:
        # linear schedule for β
        t = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + t * (beta_end - beta_start)

        smooth, grad = _smooth_max_and_grad(a, beta)
        # apply a little momentum to the gradient
        mom = mom_decay * mom + (1.0 - mom_decay) * grad

        # back‑tracking line search for the exponential update
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, mom, eta_try)
            new_score = _objective_score(a_new)
            if new_score <= cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.2f} η={eta_try:.2e} → {new_score:.6f}")
            eta = min(1.0, eta * 1.01)        # slowly increase step size
        else:
            # minor random shake – helps escape tiny plateaus
            rng = np.random.default_rng()
            eps = 5e-5 * rng.random() * a
            a = a + eps
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.5)

        # safety early stop – we keep working until the whole budget expires
    return best_seq, best_score

# 4️⃣  Targeted hill‑climbing (pair‑wise mass moves)
# ----------------------------------------------------------------------
def _hill_climb(a0, deadline, best_seq, best_score, max_iters=20000):
    """
    Repeatedly identify the pair (i,j) responsible for the current peak
    of the convolution and shift a tiny amount of probability from that
    pair to the currently lightest coordinate.  Accept only improving
    moves.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    no_improve = 0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current max value & its index
        cur_max, k_max = _max_conv_and_location(a)

        # indices that contribute to this entry
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        i_range = np.arange(i_low, i_high + 1)
        j_range = k_max - i_range
        prods = a[i_range] * a[j_range]

        # the pair with the biggest product – the “culprit”
        worst_idx = int(prods.argmax())
        i_star = int(i_range[worst_idx])
        j_star = int(j_range[worst_idx])

        # amount we are willing to move (a few percent of the smaller of the two)
        epsilon = 0.02 * min(a[i_star], a[j_star])
        if epsilon < 1e-12:
            # nothing to move – give up on this iteration
            no_improve += 1
            if no_improve > 500:
                break
            continue

        # destination index – a currently light coordinate
        # (choose the smallest entry, but avoid i_star/j_star)
        candidates = np.where((np.arange(n) != i_star) & (np.arange(n) != j_star))[0]
        dest = int(candidates[a[candidates].argmin()])

        a_new = a.copy()
        if i_star == j_star:
            # we are reducing a single entry
            a_new[i_star] -= epsilon
            a_new[dest] += epsilon
        else:
            a_new[i_star] -= epsilon
            a_new[j_star] -= epsilon
            a_new[dest] += 2.0 * epsilon

        # re‑normalise (tiny numerical drift)
        a_new = _project_simplex(a_new)

        new_score = _objective_score(a_new)

        if new_score < best_score - 1e-9:        # strict improvement
            a = a_new
            best_score, best_seq = new_score, a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it} → {new_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2000:
                # after a long stretch without progress we stop
                break

    return best_seq, best_score

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max mirror descent (core optimiser)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_seq, best_score = _smooth_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a clean list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
