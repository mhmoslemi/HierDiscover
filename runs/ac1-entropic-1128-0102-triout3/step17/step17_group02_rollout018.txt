The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the objective is invariant under scaling we can restrict the search
to probability vectors (`sum(a)==1`).  
`max(a*a)` is the largest entry of the a‑periodic self‑convolution of `a`.
The problem is non‑convex, but the maximum can be approximated smoothly with a
log‑sum‑exp (temperature β).  A smooth approximation is differentiable, so we
can apply a fast first‑order optimiser.  We use **Adam** on the *soft‑max*
parameterisation

\[
a_i = \frac{e^{x_i}}{\sum_j e^{x_j}}, \qquad x\in\mathbb R^n ,
\]

which automatically enforces non‑negativity and the simplex constraint.
The optimiser works in three phases

1. **initialisation** – a few deterministic shapes and many Dirichlet draws;
2. **Adam optimisation** – a β‑schedule (0.5 → 2000) and Adam updates;
   several random restarts are performed, each receiving a share of the
   remaining budget;
3. **final hill‑climbing** – cheap random mass‑transfer moves to polish the
   best vector.

All heavy convolutions are performed with FFTs, and the whole search respects
the 1000 s budget.

```python
"""
adam_optimizer.py

Searches for a non‑negative step function that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm:
* many random Dirichlet starts + a few deterministic shapes,
* Adam optimisation on the soft‑max parametrisation using a smooth
  log‑sum‑exp approximation of the max (β‑schedule),
* short random hill‑climbing for final fine‑tuning.

The code stays within the 1000 s time budget and returns the best sequence
found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing handle (set by propose_candidate)
# ----------------------------------------------------------------------
_start_time = None


# ----------------------------------------------------------------------
# Low‑level numerical helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using an FFT of power‑of‑2 length."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


# ----------------------------------------------------------------------
# Objective (sum is forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max utilities
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max (log‑sum‑exp) weights w_i = exp(β·(c_i-m))/∑exp(...),
    where m = max(conv).  The weights sum to 1 and are used for the smooth
    gradient of the max.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return exps / exps.sum()


def _smooth_gradient_wrt_a(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of  β‑smooth max(conv) w.r.t. a.
    Implements  grad = 2 * (a_rev * w)[n-1 : n-1+n],
    where w are the smooth‑max weights.
    """
    w = _smooth_weights(conv, beta)                # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)          # length 3·n‑2
    n = a.shape[0]
    return 2.0 * rev_corr[n - 1 : n - 1 + n]       # shape (n,)


# ----------------------------------------------------------------------
# Initial seeds
# ----------------------------------------------------------------------
def _deterministic_shapes(n: int) -> list[np.ndarray]:
    """
    A few handcrafted shapes that are known to be useful.
    Returned vectors are already normalised to sum = 1.
    """
    shapes = []

    # sine window (positive half‑wave)
    s = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    shapes.append(s)

    # Hann (raised‑cosine) window
    h = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    shapes.append(h)

    # symmetric triangular window (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    shapes.append(tri)

    # decreasing power‑law (edges larger, centre small)
    d = 1.0 / np.sqrt(np.arange(1, n + 1) * (n - np.arange(0, n)))
    shapes.append(d)

    # normalise all
    out = []
    for s in shapes:
        s = np.maximum(s, 0.0)
        s_sum = s.sum()
        if s_sum > 0:
            out.append(s / s_sum)
    return out


def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """
    Draw many Dirichlet vectors, keep the best.
    The concentration parameter is sampled log‑uniformly.
    """
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        conc_log = rng.uniform(-3.0, 0.7)          # ≈ 1e‑3 … 5
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max parametrisation
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   beta_schedule,
                   deadline: float):
    """
    Adam optimisation of the smooth‑max objective.
    a0 must already be a probability vector.
    Returns the best vector (as a probability vector) seen before the deadline.
    """
    n = a0.shape[0]
    # initialise logits (soft‑max parameters)
    eps = 1e-12
    x = np.log(a0 + eps)           # shape (n,)

    # Adam hyper‑parameters
    lr = 0.02
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = a0.copy()
    best_score = _objective_score(best_a)

    # allocate roughly equal time to each β‑stage
    remaining = deadline - time.time()
    n_beta = len(beta_schedule)
    for idx, beta in enumerate(beta_schedule):
        stage_deadline = time.time() + remaining / (n_beta - idx)
        while time.time() < stage_deadline:
            # probability vector from logits
            a = np.exp(x)
            a /= a.sum()

            # convolution & smooth gradient
            conv = _conv_self(a)
            grad_a = _smooth_gradient_wrt_a(a, conv, beta)

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_x = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
            t = np.arange(1, len(m) + 1)  # dummy, we use bias‑correction scalars
            m_hat = m / (1.0 - beta1)
            v_hat = v / (1.0 - beta2)
            x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # evaluate the *exact* (non‑smoothed) objective
            a_norm = np.exp(x)
            a_norm /= a_norm.sum()
            cur_score = _objective_score(a_norm)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_norm.copy()
                print(f"[{time.time() - _start_time:.2f}s] β={beta:.0f} new best {best_score:.6f}")

        # recompute remaining time after this β‑stage
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                score0: float,
                deadline: float):
    """Tiny random mass‑transfer moves for the last seconds."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # move a small random fraction (up to 5 % of the source mass)
        delta = a[src] * rng.random() * 0.05
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - _start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            # occasional downhill move (tiny metropolis step)
            a = cand
            cur_score = cand_score
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Locate a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain list) found inside the time budget.
    """
    np.random.seed(seed)
    global _start_time
    _start_time = time.time()
    deadline = _start_time + budget_s - 0.1   # safety margin

    # problem size – 4096 provides a good trade‑off between expressive power
    # and speed (full FFT length ≈ 8192)
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - _start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - _start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet + deterministic shapes
    # ------------------------------------------------------------------
    dir_deadline = min(deadline, _start_time + 30.0)
    if time.time() < dir_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=800, deadline=dir_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - _start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

    for shp in _deterministic_shapes(n):
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq, best_score = shp.copy(), sc
            print(f"[{time.time() - _start_time:.2f}s] deterministic shape score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation with β‑schedule (multiple restarts)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.85
        beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]

        # number of independent restarts (each gets a fair share of the time)
        restarts = 12
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break

            # start from the current best with a small random jitter
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.015, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()

            sub_deadline = min(descent_deadline,
                               time.time() + (descent_deadline - time.time()) / (restarts - r))
            a_opt, sc_opt = _adam_optimize(a0, beta_schedule, sub_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - _start_time:.2f}s] restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing (last seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - _start_time:.2f}s] starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - _start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
