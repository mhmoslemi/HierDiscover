        A = np.fft.rfft(a, L_full)
        conv_full = np.fft.irfft(A * A, n=L_full)
        conv = conv_full[:L]  # length 2n-1
        max_conv = conv.max()
        score = 2.0 * n * max_conv  # sum=1
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter {it} gamma={gamma:.2f} score={score:.6f}")

        # compute softmax weights
        maxv = conv.max()
        scaled = gamma * (conv - maxv)
        # to avoid overflow, subtract max of scaled? Already subtract max conv; but scaled may still be large (negative)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / L)
        else:
            w /= w_sum

        # FFT of w
        W = np.fft.rfft(w, L_full)

        # linear conv w * a (full) via FFT

"""
A fast “smoothed‑max” optimiser for the step‑function problem.

The evaluation metric is
    score = 2·n·max(conv(a,a)) / (sum(a)²) .
Because the metric is scale‑invariant we may (and do) keep
∑a_i = 1, i.e. a lies on the probability simplex.

The algorithm consists of four stages

1️⃣  Warm‑start – a few Dirichlet draws and the optional global starter.
2️⃣  Smoothed‑max gradient descent – minimise the smooth approximation
    g_γ(a)= (1/γ)·log∑_k exp(γ·conv_k).  Its gradient has a very simple
    form:  ∇g_γ(a)_p = 2·∑_j a_j·w_{p+j},
    where w are the soft‑max weights of the convolution.
    The gradient is obtained by a single FFT‑based correlation.
    An Adam‑style update together with a Euclidean projection onto the simplex
    drives the iterates.  The smoothing parameter γ is increased slowly so that
    the optimisation finally targets the true max.
3️⃣  Targeted hill‑climb – while time permits we repeatedly move a tiny
    amount of mass from the larger member of the pair that contributes most to
    the current maximal convolution entry.
4️⃣  Light final mixing – convex mixture with the uniform distribution and a
    few random‑pair hill‑climbs.

The whole routine respects the 1000 s budget, prints progress, and finally
returns a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int) -> float:
    """
    Exact score used by the evaluator (the sum of a is assumed to be 1).
    """
    # full linear convolution a*a via FFT
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _softmax(x: np.ndarray, gamma: float) -> np.ndarray:
    """
    Numerically stable softmax: w_i ∝ exp(gamma·x_i).
    """
    mx = np.max(x)
    shifted = gamma * (x - mx)
    # very large negative values can under‑flow to 0 – that's fine
    e = np.exp(shifted)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform weights (should not happen)
        return np.full_like(x, 1.0 / x.size)
    return e / s


# ----------------------------------------------------------------------
# 1️⃣  Warm‑start (Dirichlet draws + optional global starter)
# ----------------------------------------------------------------------
def _warm_start(n, best_a, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration drawn log‑uniformly
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        cand = sample / sample.sum()
        sc = _objective_score(cand, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smoothed‑max gradient descent (Adam + projection)
# ----------------------------------------------------------------------
def _smoothed_max_descent(a0: np.ndarray, n: int, deadline: float):
    """
    Minimise the smooth approximation of max(conv) using Adam updates.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(best_a, n)

    # FFT length sufficient for the correlation a⋆w (size L+n‑1)
    L = 2 * n - 1
    L_full = _next_pow2(L + n - 1)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    gamma = 0.8               # initial smoothing
    gamma_max = 80.0
    gamma_inc = 1.015          # geometric increase per iteration

    while time.time() < deadline:
        it += 1

        # ----- convolution a*a -------------------------------------------------
        A = np.fft.rfft(a, n=L_full)
        conv_full = np.fft.irfft(A * A, n=L_full)[:L]            # length 2n‑1
        max_conv = conv_full.max()
        score = 2.0 * n * max_conv        # sum(a)=1 → denominator = 1

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter {it:5d} γ={gamma:5.2f} → score={score:.6f}")

        # ----- soft‑max weights -------------------------------------------------
        w = _softmax(conv_full, gamma)

        # ----- gradient = 2 * correlation(a, w) --------------------------------
        W = np.fft.rfft(w, n=L_full)
        # linear correlation a ⋆ w  (real part only)
        corr_full = np.fft.irfft(A * np.conj(W), n=L_full)
        offset = L - 1                         # zero‑lag index
        grad = 2.0 * corr_full[offset: offset + n]   # shape (n,)

        # ----- Adam step -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.03 * m_hat / (np.sqrt(v_hat) + eps)   # learning‑rate 0.03 works well
        a_new = a - step
        a = _project_simplex(a_new)

        # increase smoothing (approach true max)
        gamma = min(gamma_max, gamma * gamma_inc)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Targeted hill‑climb on the current maximum of the convolution
# ----------------------------------------------------------------------
def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    For convolution index k, return the pair (i,j) with the largest product
    a_i·a_j (i+j=k) together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    prod = a[idx] * a[k - idx]
    max_pos = np.argmax(prod)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(prod[max_pos])


def _targeted_hill_climb(a: np.ndarray, n: int, deadline: float):
    best_a = a.copy()
    best_score = _objective_score(best_a, n)
    rng = np.random.default_rng()

    L = 2 * n - 1
    while time.time() < deadline:
        # convolution (fast FFT version)
        A = np.fft.rfft(best_a, n=_next_pow2(L))
        conv = np.fft.irfft(A * A, n=_next_pow2(L))[:L]
        k_max = int(conv.argmax())
        i, j, _ = _max_contributions(best_a, k_max)

        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-12:
            continue
        dst = rng.integers(0, n)
        if dst == src:
            continue
        delta = rng.uniform(0.0, min(best_a[src], 0.004))
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)   # keep sum = 1

        new_score = _objective_score(cand, n)
        if new_score < best_score:
            best_score, best_a = new_score, cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb i={src}→{dst} δ={delta:.5e} → {new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final light mixing + random‑pair hill‑climb
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray, n: int, deadline: float):
    best_a = a.copy()
    best_score = _objective_score(best_a, n)
    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand = _project_simplex(cand)
        sc = _objective_score(cand, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] mix λ={lam:.3f} → {sc:.6f}")

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        sc = _objective_score(cand, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] light hill‑climb i={i} j={j} δ={delta:.5e} → {sc:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of non‑negative floats (approximately unit sum).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    # problem size – 4096 provides a good quality / speed trade‑off
    n = 4096

    # 0️⃣  Trivial starter (uniform) and optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # if a global starter called ``height_sequence_1`` is present we use it
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:

            starter = np.maximum(starter, 0.0)

            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter, n)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()

    except Exception:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    if time.time() < deadline:

    # ------------------------------------------------------------------
    if time.time() < deadline:

    # return plain Python list
