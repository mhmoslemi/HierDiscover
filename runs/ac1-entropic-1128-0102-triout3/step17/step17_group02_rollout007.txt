"""
Gradient‑guided mass‑transfer optimizer

This implementation searches the simplex of non‑negative vectors (sum = 1)
for a step‑function that minimises

    2·n·max(conv(a,a))

where conv is the linear (zero‑padded) convolution.  The core idea is a
simple sub‑gradient step on the *exact* max‑convolution: at the current
maximal shift k we move a small fraction of mass from the index that
contributes most to this shift (largest entry of the reversed, shifted
vector) to the index that contributes least.  The move is accepted only
if it lowers the objective, otherwise the step size is reduced.  Random
jitter is injected after a streak of failed moves so the algorithm can
escape flat regions.  The search is seeded, respects the 1000 s budget,
and returns the best sequence found.
"""

# Low‑level utilities (FFT convolution and helpers)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using an FFT of power‑of‑2 length."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient for the *exact* max‑convolution
# ----------------------------------------------------------------------
def _max_conv_and_gradient(a: np.ndarray):
    """
    Returns (max_value, argmax_index, gradient) where gradient is
    g_i = a_{k‑i} (zero when the index is out of range).  The factor 2
    from the true sub‑gradient is omitted because only the ordering of
    entries matters for the source/destination choice.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    k = int(conv.argmax())
    max_val = conv[k]

    # build the reversed‑shift vector efficiently
    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1
        grad[: k + 1] = a[k::-1]
    else:
        # overlap length = 2*n-1-k
        offset = k - (n - 1)               # first index in `grad` that is non‑zero
        length = 2 * n - 1 - k
        grad[offset:] = a[::-1][:length]

    return max_val, k, grad


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

        if sc < best_score:
            best_a, best_score = starter.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] starter improves to {best_score:.6f}")

    print(f"[{time.time() - start_time:.2f}s] initial best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap random Dirichlet / deterministic shapes (≈5 s)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 1)
    dir_end = min(deadline, start_time + 5.0)
    while time.time() < dir_end:
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        cand = rng.gamma(shape=c, scale=1.0, size=n)
        tot = cand.sum()
        if tot == 0.0:
            continue
        cand = cand / tot
        sc = _objective_score(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet new best {best_score:.6f}")

    # deterministic test shapes (sine, Hann, triangular)
    shapes = []
    shapes.append(np.sin(np.pi * (np.arange(n) + 1) / (n + 1)))               # half‑sine
    shapes.append(0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))) # Hann
    triangular = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    shapes.append(triangular)

    for shp in shapes:
        shp = np.maximum(shp, 0.0)
        shp = _project_simplex(shp)
        sc = _objective_score(shp)
        if sc < best_score:
            best_a, best_score = shp.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] shape improves to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – gradient‑guided mass‑transfer loop
    # ------------------------------------------------------------------
    # parameters controlling the transfer magnitude
    transfer_frac = 0.20                     # initial fraction of source mass to move
    min_frac = 1e-4
    max_frac = 0.5
    fail_streak = 0
    max_fail_before_jitter = 200

    a = best_a.copy()
    cur_score = best_score

    while time.time() < deadline:
        # compute the current max convolution and its gradient
        max_val, k_max, grad = _max_conv_and_gradient(a)

        # source = index with largest gradient among positions that actually hold mass
        mask = a > 1e-12
        if not np.any(mask):
            break
        grad_src = np.where(mask, grad, -np.inf)
        src = int(np.argmax(grad_src))

        # destination = index with smallest gradient (could be zero)
        dst = int(np.argmin(grad))

        # If no useful direction, fall back to jitter after a while
        if src == dst or grad[src] <= grad[dst] + 1e-15:
            fail_streak += 1
            if fail_streak >= max_fail_before_jitter:
                # inject a small random perturbation and reset the streak
                jitter = rng.normal(scale=0.01, size=n)
                a = a + jitter
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                cur_score = _objective_score(a)
                fail_streak = 0
                continue
            else:
                # shrink the step size and try again
                transfer_frac = max(transfer_frac * 0.7, min_frac)
                continue

        # amount of mass to transfer
        delta = a[src] * transfer_frac
        if delta <= 0.0:
            # nothing to move – treat as failure
            fail_streak += 1
            transfer_frac = max(transfer_frac * 0.7, min_frac)
            continue

        # propose the move
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # numerical safety
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()                     # re‑normalise to avoid drift

        cand_score = _objective_score(cand)

        if cand_score < cur_score - 1e-14:
            # accept the improvement
            a = cand
            cur_score = cand_score
            fail_streak = 0
            # keep the best ever found
            if cand_score < best_score - 1e-14:
                best_a, best_score = cand.copy(), cand_score
                print(f"[{time.time() - start_time:.2f}s] new global best {best_score:.6f}")

            # slowly increase step size on success
            transfer_frac = min(transfer_frac * 1.05, max_frac)
        else:
            # reject – make the step smaller and count a failure
            fail_streak += 1
            transfer_frac = max(transfer_frac * 0.7, min_frac)

        # gradually shrink the allowed step fraction as we approach the deadline
        elapsed = time.time() - start_time
        fraction_budget = max(min_frac,
                              0.20 * (1.0 - elapsed / budget_s))
        transfer_frac = min(transfer_frac, fraction_budget)

    # ------------------------------------------------------------------
    # Normalise the best vector and return it as a plain list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    tot = best_a.sum()
    if tot > 0.0:
        best_a = best_a / tot
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
