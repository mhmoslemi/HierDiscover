def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, math, random
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1
    n = 4096
    L_fft = 1 << ((2*n-2).bit_length())  # 8192
    best_score = float('inf')
    best_a = None

    # baseline: uniform
    uniform = np.full(n, 1.0 / n)
    best_score = 2 * n * max(np.convolve(uniform, uniform))
    best_a = uniform.copy()
    print(f"Uniform baseline score {best_score}")

    # try global starter if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = 2 * n * np.max(np.convolve(starter, starter))
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"Global starter score {sc}")
    except Exception:
        pass

    # warm start: random Dirichlet etc (maybe skip)
    
    # random mask search
    rng = np.random.default_rng(seed)
    # Pre-allocate zero array to avoid reallocation inside loop?
    while time.time() < deadline:
        # choose sparsity fraction p
        p = rng.uniform(0.2, 0.95)  # adjust range
        m = max(2, int(p * n))
        # random unique indices
        idx = rng.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0 / m
        # compute conv via FFT
        Af = np.fft.rfft(a, n=L_fft)
        conv = np.fft.irfft(Af * Af, n=L_fft)[:2*n-1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time()-start_time:.2f}s] Random mask p={p:.3f} m={m} score={cur_score:.6f}")

    # after random search, use local refinement: gradient subgradient descent on max
    # limited time: maybe 10% remainder
    time_left = deadline - time.time()
    if time_left > 0 and best_a is not None:
        # perform subgradient descent
        a = best_a.copy()
        # parameters
        max_iters = 20000
        it = 0
        while time.time() < deadline and it < max_iters:
            it += 1
            # compute conv
            Af = np.fft.rfft(a, n=L_fft)
            conv = np.fft.irfft(Af * Af, n=L_fft)[:2*n-1]
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[{time.time()-start_time:.2f}s] Subgrad iter={it} score={cur_score:.6f}")
            # subgradient at max
            k_max = int(conv.argmax())
            # compute gradient g (size n)
            # compute j = k_max - i
            idx_i = np.arange(n)
            j = k_max - idx_i
            mask = (j >= 0) & (j < n)
            g = np.zeros_like(a)
            g[mask] = a[j[mask]]
            # double for self-pair
            if k_max % 2 == 0:
                i0 = k_max // 2
                if 0 <= i0 < n:
                    g[i0] += a[i0]  # extra additive
            # compute dot = a·g
            dot = np.dot(a, g)
            # gradient wrt x (softmax param) not needed; we directly adjust a with projection.
            # subgradient step on a: a = a - η * g
            eta = 0.1 / np.sqrt(it)  # step schedule
            a_new = a - eta * g
            # project onto simplex
            a = _project_simplex(a_new)   # need to define function at top-level (we can copy).
        # final best store

    # Return best sequence (list)
    if best_a is None:
        # fallback uniform
        best_a = uniform.copy()
        best_score = 2.0 * n * np.max(np.convolve(best_a, best_a))
    # Normalise again (ensure sum=1)
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    print(f"[{time.time()-start_time:.2f}s] Finished best score {best_score}")
    return [float(v) for v in best_a]

"""
A fast, simple stochastic search for a non‑negative step function that
minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is deliberately different from the reference
implementation:

  1️⃣  Random *sparse* masks – choose a random subset of indices,
      give each a uniform weight 1/|S| (so the sum is 1).  For a random
      set the self‑convolution is almost flat, giving scores close to
      the theoretical optimum (~1.0).  Millions of such masks can be
      evaluated cheaply with an FFT based convolution.

  2️⃣  Sub‑gradient descent on the exact max‑convolution objective.
      The sub‑gradient at the current maximal entry `k` is simply
      `a_{k‑i}` (with a doubled contribution when `i = k/2`).  After a
      small step we project back onto the probability simplex.

  3️⃣  A short “uniform mixing + random pair hill‑climb” phase (identical
      to the reference) to polish the final candidate.

The whole routine respects the 1000 s budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities – must be top level (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _score_of(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (assumes sum(a)=1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Random mask search
# ----------------------------------------------------------------------
def _random_mask_search(n: int,
                        L_fft: int,
                        deadline: float,
                        rng: np.random.Generator,
                        best_a: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Generate random subsets of indices, give each the same weight,
    evaluate the score and keep the best.
    """
    while time.time() < deadline:
        # pick a sparsity fraction; avoid too small m (self‑terms would dominate)
        p = rng.uniform(0.2, 0.95)          # fraction of non‑zeros
        m = max(2, int(p * n))              # at least two entries
        idx = rng.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0 / m                     # normalised sum = 1

        # convolution via FFT (fast)
        conv = _conv_self_fft(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time()-start_time:.2f}s] "
                  f"Mask p={p:.3f} m={m} → score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float,
                         best_a: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    it = 0
    while time.time() < deadline and it < 20000:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time()-start_time:.2f}s] "
                  f"SubGrad iter={it:,} score={cur_score:.6f}")

        # index of the current maximal convolution entry
        k = int(conv.argmax())

        # sub‑gradient g_i = a_{k-i} (doubling when i==k-i)
        i_arr = np.arange(n, dtype=np.int64)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        g = np.zeros_like(a)
        g[mask] = a[j_arr[mask]]
        if k % 2 == 0:                # add the extra self‑pair term
            i0 = k // 2
            if 0 <= i0 < n:
                g[i0] += a[i0]

        # diminishing stepsize
        eta = 0.5 / np.sqrt(it)

        a_new = a - eta * g
        a = _project_simplex(a_new)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final mixing + cheap random‑pair hill‑climb
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Convex mixing with the uniform vector and a tiny random‑pair climb."""
    best_a = a.copy()
    best_score = _score_of(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _score_of(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time()-start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # very cheap random‑pair hill‑climb (identical to reference)
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _score_of(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time()-start_time:.2f}s] Light hill‑climb i={i} j={j} "
                  f"δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Stochastic construction of a step‑function with a low max‑convolution.
    The routine spends most of the budget generating random sparse masks,
    then runs a few sub‑gradient steps and finishes with a light mixing/
    hill‑climb phase.
    Returns a plain Python list of non‑negative floats (sum≈1).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    # ------------------------------------------------------------------
    # problem size – 4096 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)   # 8192 for n=4096

    # ------------------------------------------------------------------
    # 0️⃣  Trivial initialiser (uniform) and optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _score_of(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    try:
        # height_sequence_1 may be supplied globally
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _score_of(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random mask search (≈70 % of the overall budget)
    # ------------------------------------------------------------------
    rand_deadline = time.time() + 0.70 * (deadline - time.time())
    rng = np.random.default_rng(seed + 12345)
    best_a, best_score = _random_mask_search(
        n, L_fft, rand_deadline, rng, best_a, best_score
    )
    print(f"[{time.time()-start_time:.2f}s] After random masks best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent on the exact max‑convolution (≈20 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = time.time() + 0.20 * (deadline - time.time())
        print(f"[{time.time()-start_time:.2f}s] Sub‑gradient descent")
        best_a, best_score = _subgradient_descent(
            best_a, n, L_fft, subgrad_deadline, best_a, best_score
        )
        print(f"[{time.time()-start_time:.2f}s] After sub‑grad best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final refinement (mixing + tiny hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time()-start_time:.2f}s] Final refinement")
        best_a, best_score = _final_refinement(best_a, n, L_fft, deadline)

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
