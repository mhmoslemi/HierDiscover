def _adam_smooth_max(a0, deadline, best_seq, best_score):

    v = np.log(a0 + 1e-12)  # initialize v such that softmax is a0.

    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1=0.9; beta2=0.999; eps=1e-8

    tau = max(_conv_self(a0).max() * 0.2, 1e-6)

    iteration = 0

    while time.time()<deadline and tau>1e-9:
        iteration+=1

        # compute a = softmax(v)
        e_v = np.exp(v - np.max(v))
        a = e_v / e_v.sum()

        conv = _conv_self(a)
        max_conv = conv.max()

        # compute softmax weights vector w over conv entries:
        shift = max_conv
        log_w = (conv - shift) / tau
        w = np.exp(log_w)
        w_sum = w.sum()
        w = w / w_sum

        # compute gradient wrt a (smooth objective)
        a_rev = a[::-1]
        L = _next_pow2(len(w) + len(a_rev) - 1)
        Pf = np.fft.rfft(w, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(w) + len(a_rev) - 1]
        grad_a = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # gradient wrt v via softmax Jacobian
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam updates:
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat_corr = v_hat / (1 - beta2 ** iteration)
        # step size
        step = 0.05 / (np.sqrt(v_hat_corr) + eps)
        v = v - step * m_hat

        # evaluate true objective: conv, obj = 2*n*max_conv
        obj = 2.0 * n * max_conv
        if obj < best_score:
            best_score = obj; best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam step={iteration}, obj={obj:.6f}, tau={tau:.2e}")

        # temperature decay
        tau *= 0.9995

    return best_seq, best_score

"""
Propose a non‑negative step function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search runs for ≈1000 s and consists of four stages:

1️⃣  Rich parametric sweeps (geometric, power, exponential, two‑block,
    triangular, Gaussian, Beta‑like).  The best candidate from these
    sweeps seeds the later phases.

2️⃣  Random draws from Dirichlet distributions (different concentrations)
    to explore the simplex.

3️⃣  Adam‑based smooth‑max gradient descent.
    The non‑smooth max is approximated by a temperature‑controlled
    log‑sum‑exp, gradients are computed analytically in O(n log n) by FFT.
    The softmax is expressed via an unconstrained vector `v` and a
    soft‑max map `a = softmax(v)`, which automatically stays on the
    probability simplex.

4️⃣  Simple local hill‑climbing (mass transfer between two random bins)
    for a final fine‑tuning pass.

The current best score (around 1.503) is usually reached in stage 3–4.
All helpers are top‑level; no closures, file I/O or external binaries are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):

# 3️⃣ Adam‑based smooth‑max gradient descent on the simplex
# ----------------------------------------------------------------------
def _adam_smooth_max(initial_a, deadline, best_seq, best_score):
    """
    Gradient descent on τ·log∑exp(conv/τ) with Adam steps,
    the optimisation variable is an unconstrained vector `v`
    which is mapped to the simplex by softmax.
    """
    n = initial_a.shape[0]
    # initialise v so that softmax(v) = initial_a (add tiny offset)
    v = np.log(initial_a + 1e-30)

    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.07                # base learning rate

    # temperature schedule
    cur_conv = _conv_self(initial_a)
    tau = max(cur_conv.max() * 0.4, 1e-6)
    min_tau = 1e-9
    decay = 0.9988           # very slow cooling

    it = 0
    while time.time() < deadline and tau > min_tau:
        it += 1

        # softmax to stay on the simplex
        max_v = np.max(v)
        e_v = np.exp(v - max_v)
        a = e_v / e_v.sum()

        # convolution & smooth‑max weights
        conv = _conv_self(a)
        max_conv = conv.max()
        # soft‑max (log‑sum‑exp) weights
        log_w = (conv - max_conv) / tau
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0:
            tau *= decay
            continue
        w = w / w_sum

        # gradient w.r.t. a of the smooth max
        a_rev = a[::-1]
        L = _next_pow2(len(w) + len(a_rev) - 1)
        Pf = np.fft.rfft(w, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(w) + len(a_rev) - 1]
        grad_a = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # back‑prop through the softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on the unconstrained v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat_corr) + eps)
        v = v - step * m_hat

        # evaluate true objective (non‑smooth)
        obj = 2.0 * n * max_conv
        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} obj={obj:.6f} τ={tau:.2e}")

        # cooling
        tau *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣ Simple hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(a0, deadline, best_seq, best_score):
    """Mass‑transfer hill climbing on the simplex (few thousand moves)."""
    n = a0.shape[0]
    a = a0.copy()
    rng = np.random.default_rng()
    move_counter = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] == 0:
            continue
        # transfer up to 30 % of the source mass
        delta = rng.random() * 0.3 * a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        obj = _objective_score(a_new)
        move_counter += 1
        if obj < best_score:
            best_score, best_seq = obj, a_new.copy()
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] HillClimb move={move_counter} obj={obj:.6f}")
        # occasional random jitter to shake the process
        if move_counter % 2000 == 0:
            eps = 1e-8
            a += rng.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    evaluation function.  The routine performs parametric sweeps,
    Dirichlet sampling, Adam‑based smooth‑max descent and a final
    hill‑climbing pass.  The best sequence (normalised to sum = 1)
    is returned as a list of Python floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # tiny safety buffer

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1, if present)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Parametric sweeps (≈250 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 500.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_triangular(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_gaussian(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_beta(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Adam smooth‑max descent (≈350 s)
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam smooth‑max descent")
        best_seq, best_score = _adam_smooth_max(best_seq, deadline - 150.0, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End of Adam descent, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Final hill‑climbing fine‑tune (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climbing refinement")
        best_seq, best_score = _hill_climb(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Hill‑climbing finished, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
