"""
Improved multi‑stage search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of:

1️⃣  Refined parametric sweeps (geometric·power family and symmetric Beta family)
    with a coarse grid followed by a local coordinate descent around the best
    lam‑p pair.

2️⃣  Multi‑peak projected sub‑gradient descent:
    at each iteration the true gradient of a *soft‑max* of the convolution
    peaks (top‑K entries) is used, then a Euclidean projection onto the simplex.
    This reduces all large convolution entries simultaneously instead of only the
    single maximum.

3️⃣  Final simulated‑annealing hill‑climbing:
    tiny random mass‑transfer moves are accepted with a temperature schedule,
    allowing occasional uphill steps to escape shallow local minima.

All stages respect the 1000 s time budget and keep the vector on the
probability simplex (sum = 1, non‑negative).  The best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – assumes the sum ≈ 1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families for the parametric sweep
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """
    Construct a_i ∝ (i+1)^p * exp(-lam*i) and normalise to L1‑norm = 1.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = -lam * i + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)        # numerical stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta‑type shape on the grid {0,…,n‑1}.
    a_i ∝ (x)^{α‑1} (1‑x)^{β‑1}   where x = (i+0.5)/n.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid infinities at the boundaries
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    if shape.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient for a single convolution component
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of conv_k = (a*a)[k] (the k‑th entry of the self‑conv)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Multi‑peak gradient descent (soft‑max of the top‑K convolution entries)
# ----------------------------------------------------------------------


def _multi_peak_descent(a0: np.ndarray,
                        L_fft: int,
                        deadline: float,
                        start_time: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        rng: np.random.Generator,
                        K: int = 5,
                        beta: float = 12.0) -> tuple[np.ndarray, float]:
    """Projected gradient descent using a weighted sum of the top‑K peaks."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)

        # --- pick the K largest convolution values
        if K < conv.size:
            top_idx = np.argpartition(-conv, K - 1)[:K]
        else:
            top_idx = np.arange(conv.size)
        top_vals = conv[top_idx]
        # soft‑max weighting (focus on the very largest entries)
        max_val = top_vals.max()
        w = np.exp(beta * (top_vals - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- weighted sub‑gradient
        g = np.zeros_like(a)
        for weight, k_idx in zip(w, top_idx):
            g += weight * _subgrad_max_conv(a, int(k_idx))

        # --- line‑search / back‑tracking
        step = step0 / np.sqrt(it)
        cand = a - step * g
        cand = _project_simplex(cand)
        cand_score = _eval_fast(cand, L_fft)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] multi‑peak it={it:,} "
                  f"score={cand_score:.6f}")
            continue

        # simple back‑track if no improvement
        improved = False
        for _ in range(4):
            step *= 0.5
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] multi‑peak bt it={it:,} "
                      f"step={step:.5e} score={cand_score:.6f}")
                break
        if not improved and it % 1500 == 0:
            # occasional jitter to escape tiny plateaus
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def _final_sa(best_seq: np.ndarray,
              L_fft: int,
              deadline: float,
              start_time: float,
              best_score: float,
              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny random mass‑transfer moves with a linear temperature schedule."""
    a = best_seq.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move at most 0.5 % of the source mass
        eps = a[src] * rng.random() * 0.005
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)
        # linear temperature: 1 → 0 over the remaining time
        total = deadline - start_time
        remain = deadline - time.time()
        temp = max(1e-6, remain / total)

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / temp):
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_seq = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} "
                      f"new best={new_score:.6f}")
    return best_seq, best_score

                    lam2 = lam_cur + dl
                    p2 = p_cur + dp
                    if lam2 <= 0.0:
                        continue
                    cand = _make_geo_power(n, lam2, p2)
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_score, best_seq = sc, cand.copy()
                        lam_cur, p_cur = lam2, p2
                        improved = True
                        print(f"[{time.time() - start_time:.2f}s] refine λ={lam2:.6e} "
                              f"p={p2:.2f} score={sc:.6f}")
            if not improved:
                step_lam *= 0.5
                step_p *= 0.5
                if step_lam < 1e-9 and step_p < 1e-3:
                    break

    # ------------------------------------------------------------------
    # 3️⃣  Sweep over a symmetric Beta‑type family (captures unimodal shapes).
    # ------------------------------------------------------------------
    beta_deadline = deadline - 250.0   # keep some time for the descent phases
    alphas = np.linspace(0.3, 5.0, 25)
    betas = np.linspace(0.3, 5.0, 25)

    for a_val in alphas:
        for b_val in betas:
            if time.time() >= beta_deadline:
                break
            cand = _make_beta_shape(n, a_val, b_val)
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] beta α={a_val:.2f} "
                      f"β={b_val:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Multi‑peak projected sub‑gradient descent (soft‑max of top‑K peaks).
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        descent_deadline = deadline - 150.0
        best_seq, best_score = _multi_peak_descent(
            best_seq, L_fft, descent_deadline, start_time,
            best_seq, best_score, rng, K=7, beta=12.0)

    # ------------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing hill‑climb.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        best_seq, best_score = _final_sa(
            best_seq, L_fft, deadline, start_time, best_score, rng)

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
