"""
Propose a step‑function (non‑negative sequence) that minimizes
`evaluate_sequence`.  

The method combines three ideas:

1. **Smoothing of the max‑convolution** – we minimise a smooth
   approximation of `max (a*a)` using the log‑sum‑exp trick.  The
   gradient of this objective can be written as a convolution and is
   computed efficiently with FFTs.

2. **Adam optimisation on the probability simplex** – the variables live
   on the simplex (they are non‑negative and sum to one).  After each
   Adam step we project back onto the simplex with the O(n) algorithm
   of Duchi et al.

3. **Multi‑scale search** – we start with a modest length, optimise,
   then up‑sample the best sequence for a larger length.  After the
   smooth optimisation a short projected‑gradient phase and a tiny
   simulated‑annealing refinement are applied.

All time is bounded by the supplied budget (`budget_s`).  The code prints
progress messages and finally returns the best sequence found as a plain
Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – projection onto the probability simplex {x_i >= 0, sum x_i = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return the centre of the simplex
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear / full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full, length 2*len(a)-1) via FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _next_pow2(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y via FFT."""
    m = x.shape[0] + y.shape[0] - 1
    fft_len = _next_pow2(m)
    Xf = np.fft.rfft(x, n=fft_len)
    Yf = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(Xf * Yf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Smooth optimisation (Adam on the simplex)
# ----------------------------------------------------------------------
def _run_adam_smooth(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    max_iter: int = 20000,
    lr: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 5.0,
    alpha_end: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate
        f_alpha(a) = (1/alpha) log sum_i exp(alpha * (a*a)[i])
    on the simplex.  Returns the best sequence encountered (and its true
    score 2*n*max(a*a)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    # iteration counter for the α schedule
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # 1. current convolution and true score (for bookkeeping)
        # --------------------------------------------------------------
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"smooth Adam improve: n={n}, iter={it}, score={cur_score:.6f}"
            )

        # --------------------------------------------------------------
        # 2. smoothing weight vector (softmax over the convolution entries)
        # --------------------------------------------------------------
        # α grows from α_start to α_end (exponential schedule)
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max_iter)

        # stable softmax: subtract the maximum before exponentiation
        shift = conv - max_conv            # ≤ 0
        w = np.exp(alpha * shift)          # w_i = exp(alpha*(conv_i-max))
        w_sum = w.sum()
        if w_sum == 0.0:                   # numerical safe‑guard
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3. gradient of the smooth surrogate (derived via convolution)
        # --------------------------------------------------------------
        a_rev = a[::-1]
        # full convolution of w with a_rev, then pick the centre slice
        conv_wr = _conv_full(w, a_rev)               # length 3n-2
        grad = 2.0 * conv_wr[n - 1 : n - 1 + n]       # shape (n,)

        # --------------------------------------------------------------
        # 4. Adam update + projection onto simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = _project_simplex(a_new)

    # final evaluation on the last a (in case loop ended without an update)
    conv = _conv_self(a)
    final_score = 2.0 * n * conv.max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent (sub‑gradient of the true max, as fallback)
# ----------------------------------------------------------------------
def _run_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    K: int = 5,
    beta: float = 0.9,
    step0: float = 0.6,
    max_iter: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent on the true max‑convolution.
    Uses the average sub‑gradient of the K largest convolution entries.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"proj‑grad improve: n={n}, iter={it}, score={cur_score:.6f}"
            )

        # active set (indices of K largest entries)
        if K >= conv.size:
            act = np.arange(conv.size, dtype=int)
        else:
            act = np.argpartition(-conv, K - 1)[:K]

        # sub‑gradient: sum of gradients of the selected entries
        g = np.zeros_like(a)
        for k in act:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += 2.0 * a[j]

        if act.size > 0:
            g /= act.size

        # momentum & diminishing step size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    # final check
    conv = _conv_self(a)
    final_score = 2.0 * n * conv.max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing local refinement (tiny hill‑climber)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 20000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"annealing improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from one length to another
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    if up.sum() > 0:
        up /= up.sum()
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

        # --------------------------------------------------------------
        # initialise a candidate for this length
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # add a tiny random jitter to help escape flat regions
        a0 += 0.01 * np.random.rand(n)
        a0 = _project_simplex(a0)

        # allocate a slice of the remaining budget (at least 8 s)
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        slice_deadline = time.time() + max(8.0, 0.15 * remaining)

        # --------------------------------------------------------------
        # Phase 1 – smooth Adam optimisation
        # --------------------------------------------------------------
        a_opt, score_opt = _run_adam_smooth(
            a0,
            n,
            slice_deadline,
            start_time,
            max_iter=30000,
            lr=0.5,
            beta1=0.9,
            beta2=0.999,
            alpha_start=5.0,
            alpha_end=250.0,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"global improve after Adam: n={n}, score={score_opt:.6f}"
            )

        # --------------------------------------------------------------
        # Phase 2 – short projected sub‑gradient descent (fine tune)
        # --------------------------------------------------------------
        # give a small extra budget if time permits
        if time.time() < slice_deadline:
            gd_deadline = time.time() + max(4.0, 0.07 * (slice_deadline - time.time()))
            a_ref, score_ref = _run_gradient_descent(
                a_opt,
                n,
                gd_deadline,
                start_time,
                K=7,
                beta=0.85,
                step0=0.4,
                max_iter=20000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] "
                    f"global improve after GD: n={n}, score={score_ref:.6f}"
                )
        else:
            a_ref = a_opt

        # --------------------------------------------------------------
        # Phase 3 – simulated‑annealing refinement (cheap local search)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(5.0, 0.05 * (deadline - time.time()))
            a_final, score_final = _run_annealing(
                a_ref,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=20000,
            )
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] "
                    f"global improve after annealing: n={n}, score={score_final:.6f}"
                )

        # --------------------------------------------------------------
        # remember the best for the next scale
        # --------------------------------------------------------------
        if best_seq is not None:
            global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed but guarantees a valid output
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # return plain Python list
    return [float(x) for x in best_seq]
