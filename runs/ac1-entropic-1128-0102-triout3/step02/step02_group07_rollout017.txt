```python
"""
Propose candidate sequence for evaluate_sequence.

Algorithm:
1) Scan a few parametric families (power‑law, exponential, arcsine) on a
   fixed length n≈2500.  Each candidate is normalised to sum 1 and the
   evaluation score 2·n·max(conv) is recorded.  The best candidate
   becomes the starting point.
2) Iteratively improve the candidate by solving a linear programme:
   maximise ∑ g_i   subject to   g_i≥0  and   (f*g)_k ≤ M   for all k,
   where M is the current maximum of the self‑convolution of f.
   The solution g concentrates mass where f is small.
   The new point is either g (if its own convolution is already smaller)
   or a convex combination (mixing) of f and g.
   A few (≤20) LP steps are performed – each step reduces the peak of the
   convolution.
3) A short projected‑gradient descent (sub‑gradient of the active
   convolution entry) refines the result further.
4) The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
def conv_fft(x: np.ndarray) -> np.ndarray:
    """Self‑convolution using a real FFT (length 2·len(x)‑1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv

def score_of(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – the numerator of the evaluation function."""
    n = a.shape[0]
    max_conv = conv_fft(a).max()
    return 2.0 * n * max_conv

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    a /= a.sum()
    return a

def exponential_sequence(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a

def arcsine_sequence(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on (0,1)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a /= a.sum()
    return a

def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix T_f such that T_f @ g = conv(f,g)
    (zero‑padded convolution).  Uses a Toeplitz representation.
    """
    n = f.shape[0]
    first_col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    first_row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    T = scipy.linalg.toeplitz(first_col, first_row)
    # keep only the first 2·n‑1 rows (the rest are zeros)
    return T[: 2 * n - 1, :]

def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max sum(g)   s.t.   T_f @ g ≤ M·1,   g ≥ 0.
    Returns the optimal g (not normalised) or None on failure.
    """
    T = build_conv_matrix(f)
    n = f.shape[0]
    c = -np.ones(n)  # linprog minimises, so use -sum(g)
    b = M * np.ones(2 * n - 1, dtype=np.float64)
    bounds = [(0, None)] * n
    # “highs” is fast for dense problems of this size
    res = scipy.optimize.linprog(c, A_ub=T, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Exact sub‑gradient of the active convolution entry."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 0. configuration
    n = 2500                       # length of the searched sequence
    best_a = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1. parametric sweep (power‑law, exponential, arcsine)
    print("[search] parametric sweep (n = %d)" % n)

    # power‑law grid
    offsets = [0.5, 1.0, 2.0]
    p_vals = np.arange(0.45, 0.81, 0.02)   # coarse grid
    for offset in offsets:
        for p in p_vals:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=offset)
            sc = score_of(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[grid] powerlaw p={p:.3f} offset={offset:.1f} → score={sc:.6f}")

    # exponential grid (lam in log‑space)
    lam_vals = np.logspace(-4, -1, 15)   # from 1e‑4 to 1e‑1
    for lam in lam_vals:
        if time.time() > deadline:
            break
        a = exponential_sequence(n, lam)
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[grid] exponential λ={lam:.5f} → score={sc:.6f}")

    # arcsine (single candidate)
    if time.time() < deadline:
        a = arcsine_sequence(n)
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[grid] arcsine → score={sc:.6f}")

    # ------------------------------------------------------------------
    # If nothing found (should never happen) fall back to the known good
    if best_a is None:
        best_a = power_law_sequence(n, 0.503, offset=0.5)
        best_score = score_of(best_a)
        print("[fallback] using default power‑law")

    print(f"[init] best initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. linear‑programming refinement
    #   Perform a handful of LP steps, each time mixing the solution.
    max_lp_iter = 15
    mix_factor = 0.2                # step size for mixing
    lp_iter = 0

    print("[lp] starting LP refinement")
    while time.time() < deadline and lp_iter < max_lp_iter:
        cur_score = score_of(best_a)
        conv = conv_fft(best_a)
        M = conv.max()
        g = solve_lp_max_sum(best_a, M)
        if g is None:
            print("[lp] LP failed – stopping")
            break

        # normalise the LP solution to a probability vector
        if g.sum() == 0.0:
            # pathological case – skip this iteration
            break
        g_norm = g / g.sum()
        s_g = score_of(g_norm)

        if s_g < best_score:
            best_score = s_g
            best_a = g_norm.copy()
            print(f"[lp iter {lp_iter}] new global best score = {best_score:.6f}")

        # decide how to update the working vector
        # use the LP solution directly if it already improves the peak,
        # otherwise mix a little with the current vector.
        if s_g < cur_score:
            best_a = g_norm
        else:
            best_a = (1.0 - mix_factor) * best_a + mix_factor * g_norm
            # re‑project to the simplex to avoid numerical drift
            best_a = project_onto_simplex(best_a)

        lp_iter += 1

    # ------------------------------------------------------------------
    # 3. short projected gradient descent on the active peak
    print("[grad] projected gradient descent")
    a = best_a.copy()
    cur_score = score_of(a)
    eta = 0.08
    decay_factor = 0.93
    decay_every = 1500
    temperature = 0.03
    max_iter = 500_000   # safety bound, we’ll also respect the deadline

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = conv_fft(a)
        max_val = conv.max()
        peak_idx = int(conv.argmax())

        grad = gradient_of_peak(a, peak_idx)

        # step opposite to gradient, limited to avoid negatives
        step = eta * grad
        step = np.minimum(step, a * 0.99)   # keep at least 1 % of each entry
        a_trial = a - step

        # occasional jitter to escape flat regions
        if rng.random() < 0.004:
            a_trial += (rng.random(n) - 0.5) * 1e-5

        a_trial = project_onto_simplex(a_trial)

        trial_score = score_of(a_trial)
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            print(f"[iter {it:07d}] new global best = {best_score:.6f}")

        # acceptance rule (deterministic improvement or Metropolis)
        if trial_score <= cur_score:
            a = a_trial
            cur_score = trial_score
        else:
            delta = trial_score - cur_score
            if temperature > 0.0 and rng.random() < np.exp(-delta / temperature):
                a = a_trial
                cur_score = trial_score
            # else reject

        # schedule decay
        if (it + 1) % decay_every == 0:
            eta = max(eta * decay_factor, 1e-6)
            temperature = max(temperature * decay_factor, 1e-8)

        it += 1

    # ------------------------------------------------------------------
    # 4. final sanity check with the provided evaluator
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, final reported score={final_score:.6f}")

    return best_a.tolist()

"""
Propose candidate sequence for evaluate_sequence.

Algorithm
---------
1. **Parametric sweep** –  Generate candidate vectors from a few
   simple families (power‑law, exponential, arcsine) for a fixed length
   n≈2500.  All vectors are normalised to sum = 1 and their score
   ``2·n·max(conv)`` is recorded.  The best vector becomes the starting
   point.

2. **Linear‑programming refinement** –  Given the current vector *f*,
   let M = max(f∗f).  Solve the LP

        maximise   ∑ g_i
        subject to  g_i ≥ 0
                    (f∗g)_k ≤ M   for every k = 0,…,2n‑2 .

   The solution puts mass where *f* is small, thereby reducing the
   peak of the self‑convolution.  The new point is either the LP
   solution (if it already improves the peak) or a convex combination
   ``(1‑t)·f + t·g`` with a small mixing factor *t*.  A handful of LP
   steps are performed – each step generally lowers the peak.

3. **Projected gradient descent** –  A short sub‑gradient descent on the
   active peak of the convolution further refines the result.
   After each step the vector is projected onto the probability simplex.

4. **Return the best vector** –  The globally best sequence found before
   the time budget expires is returned (as a plain ``list``).  A final
   call to ``evaluate_sequence`` checks that the reported score matches
   the official evaluator.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
def conv_fft(x: np.ndarray) -> np.ndarray:
    """Self‑convolution using a real FFT (length 2·len(x)‑1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def score_of(a: np.ndarray) -> float:
    """Return the numerator of the evaluation function: 2·n·max(conv)."""
    n = a.shape[0]
    return 2.0 * n * conv_fft(a).max()


def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    a /= a.sum()
    return a


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def arcsine_sequence(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on (0, 1)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a /= a.sum()
    return a


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n Toeplitz matrix T_f such that
    T_f @ g == conv(f, g) (zero‑padded convolution).
    """
    n = f.shape[0]
    first_col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    first_row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    T = scipy.linalg.toeplitz(first_col, first_row)
    return T[: 2 * n - 1, :]


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max sum(g)   s.t.   T_f @ g ≤ M·1 ,   g ≥ 0.
    Returns the optimal (unnormalised) vector g, or None on failure.
    """
    T = build_conv_matrix(f)
    n = f.shape[0]
    c = -np.ones(n)                     # linprog minimises → maximise sum(g)
    b = M * np.ones(2 * n - 1, dtype=np.float64)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(
        c, A_ub=T, b_ub=b, bounds=bounds, method='highs', options={'presolve': True}
    )
    if not res.success:
        return None
    return res.x


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Exact sub‑gradient of the active convolution entry (the peak)."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0           # keep a safety margin

    # ---------------------------- configuration -------------------------
    n = 2500                                    # length of the sequence
    best_a = None
    best_score = np.inf
    best_p = None
    best_offset = None

    # -------------------------- 1. parametric sweep --------------------
    print("[search] start parametric sweep (n=%d)" % n)

    # --- power‑law ----------------------------------------------------
    offsets = [0.5, 1.0, 2.0]
    p_vals = np.arange(0.45, 0.81, 0.02)        # coarse grid
    for offset in offsets:
        for p in p_vals:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=offset)
            sc = score_of(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_p = p
                best_offset = offset
                print(f"[grid] powerlaw p={p:.3f} offset={offset:.1f} → score={sc:.6f}")

    # --- exponential --------------------------------------------------
    lam_vals = np.logspace(-4, -1, 15)          # 1e‑4 … 1e‑1
    for lam in lam_vals:
        if time.time() > deadline:
            break
        a = exponential_sequence(n, lam)
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_p = None
            best_offset = None
            print(f"[grid] exponential λ={lam:.5e} → score={sc:.6f}")

    # --- arcsine -------------------------------------------------------
    if time.time() < deadline:
        a = arcsine_sequence(n)
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_p = None
            best_offset = None
            print(f"[grid] arcsine → score={sc:.6f}")

    # --- fine refinement around best power‑law ------------------------
    if best_a is not None and best_p is not None and time.time() < deadline:
        p_low = max(0.01, best_p - 0.01)
        p_high = min(0.99, best_p + 0.01)
        fine_ps = np.arange(p_low, p_high + 0.001, 0.001)
        print(f"[search] fine grid around p={best_p:.4f}")
        for offset in offsets:
            for p in fine_ps:
                if time.time() > deadline:
                    break
                a = power_law_sequence(n, p, offset=offset)
                sc = score_of(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    best_p = p
                    best_offset = offset
                    print(f"[fine] p={p:.4f} offset={offset:.1f} → score={sc:.6f}")

    # fall back to a known good seed if everything failed (should not happen)
    if best_a is None:
        best_a = power_law_sequence(n, 0.503, offset=0.5)
        best_score = score_of(best_a)
        print("[fallback] use default power‑law seed")

    print(f"[init] best initial score = {best_score:.6f}")

    # -------------------------- 2. LP refinement -----------------------
    max_lp_iter = 15
    mix_factor = 0.2
    lp_iter = 0
    print("[lp] starting LP refinement")
    while time.time() < deadline and lp_iter < max_lp_iter:
        cur_score = score_of(best_a)
        M = conv_fft(best_a).max()
        g = solve_lp_max_sum(best_a, M)
        if g is None:
            print("[lp] LP failed – aborting LP phase")
            break
        if g.sum() == 0.0:
            break
        g_norm = g / g.sum()
        s_g = score_of(g_norm)

        if s_g < best_score:
            best_score = s_g
            best_a = g_norm.copy()
            print(f"[lp iter {lp_iter}] new global best score = {best_score:.6f}")

        # update the working vector
        if s_g < cur_score:
            best_a = g_norm
        else:
            best_a = (1.0 - mix_factor) * best_a + mix_factor * g_norm
            best_a = project_onto_simplex(best_a)

        lp_iter += 1

    # --------------------------- 3. gradient descent -------------------
    print("[grad] projected gradient descent")
    a = best_a.copy()
    cur_score = score_of(a)
    eta = 0.08
    decay_factor = 0.93
    decay_every = 1500
    temperature = 0.03
    max_iter = 500_000

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = conv_fft(a)
        peak_idx = int(conv.argmax())
        grad = gradient_of_peak(a, peak_idx)

        step = eta * grad
        step = np.minimum(step, a * 0.99)        # keep entries non‑negative
        a_trial = a - step

        # occasional tiny jitter to avoid stagnation
        if rng.random() < 0.004:
            a_trial += (rng.random(n) - 0.5) * 1e-5

        a_trial = project_onto_simplex(a_trial)
        trial_score = score_of(a_trial)

        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            print(f"[iter {it:07d}] new global best = {best_score:.6f}")

        if trial_score <= cur_score:
            a = a_trial
            cur_score = trial_score
        else:
            delta = trial_score - cur_score
            if temperature > 0.0 and rng.random() < np.exp(-delta / temperature):
                a = a_trial
                cur_score = trial_score
            # otherwise reject

        if (it + 1) % decay_every == 0:
            eta = max(eta * decay_factor, 1e-6)
            temperature = max(temperature * decay_factor, 1e-8)

        it += 1

    # --------------------------- final sanity check --------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, final reported score={final_score:.6f}")

    return best_a.tolist()
