"""
Propose a candidate step‑function sequence for the evaluation function.

Algorithm
---------
1. Initialise a probability vector (non‑negative, sum = 1) using a few
   deterministic seeds (power‑law, sqrt‑difference, half‑binomial) and
   keep the best one according to a fast proxy score
   (2·n·max(self‑convolution)).

2. Refine the best seed by a *smooth‑max* projected gradient descent.
   - The true objective is 2·n·max(conv(a,a)).  Instead of the non‑smooth
     max we minimise the Lₚ‑norm of the self‑convolution,
      (∑ convᵖ)^{1/p}, which approaches the max when p→∞.
   - For a given p, the gradient of the Lₚ‑norm can be written as a
     weighted cross‑correlation; it is computed efficiently with real FFTs.
   - After each gradient step the iterate is projected onto the probability
     simplex (non‑negative, sum = 1) using the O(n log n) algorithm of
     Duchi et al.
   - The smoothing parameter p is increased from a modest value (≈8) to a
     large one (≈200) in stages; the step size decays slowly.
   - Global best (according to the true evaluation) is recorded every few
     hundred iterations; occasional tiny random jitter helps escape plateaus.

3. Return the best vector found before the time budget expires.

The routine respects the supplied time budget (default 950 s, leaving a
5‑second safety margin).  All helper functions are defined at module level
as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """
    Discrete self‑convolution a * a (real FFT version).
    Returns an array of length 2·len(a) – 1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


def conv_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Convolution of two real vectors using FFT.
    Result length = len(x) + len(y) – 1.
    """
    n = x.shape[0]
    m = y.shape[0]
    L = n + m - 1
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv


def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """
    a_i = (i + offset)^{-p}, i = 0 … n‑1, normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a.fill(1.0 / n)
    else:
        a /= total
    return a


def sqrt_diff_sequence(n: int, offset: float = 0.5) -> np.ndarray:
    """
    a_i = sqrt(i+1+offset) – sqrt(i+offset),
    normalised to sum = 1.  This is the discrete derivative of sqrt().
    """
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0 + offset) - np.sqrt(i + offset)
    a /= a.sum()
    return a


def half_binomial_sequence(n: int) -> np.ndarray:
    """
    Classical half‑binomial (central binomial coefficients) normalised.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    provided evaluation function.  The routine respects the given time
    budget (seconds) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # configuration
    n = 4000                     # length of the candidate sequence
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – quick scan of a few deterministic seeds
    print("[init] generating deterministic seeds")
    seeds = [
        ("power", power_law_sequence(n, p=0.5, offset=0.5)),
        ("sqrt", sqrt_diff_sequence(n, offset=0.5)),
        ("half", half_binomial_sequence(n)),
    ]

    best_a = None
    best_score = np.inf
    for name, a in seeds:
        conv = conv_self(a)
        score_proxy = 2.0 * n * conv.max()          # same as true score because sum(a)=1
        print(f"  seed {name:5s} proxy score = {score_proxy:.6f}")
        if score_proxy < best_score:
            best_score = score_proxy
            best_a = a.copy()
    print("[init] best proxy score after seeding:", best_score)

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max projected gradient descent
    a = best_a.copy()                           # current iterate
    cur_score = 2.0 * n * conv_self(a).max()    # true objective (sum=1)
    best_score = cur_score
    best_a = a.copy()
    print("[grad] start gradient descent, initial true score =", cur_score)

    # schedule of smoothing exponent p and step size
    p = 8.0                                      # start with moderate smoothness
    p_target = 200.0
    p_step = 5.0
    lr = 0.12                                    # step size (in L2‑norm of a)
    lr_decay = 0.93
    decay_every = 8000
    it = 0
    last_improve_it = 0

    # small jitter probability (helps escape flat regions)
    jitter_prob = 0.004
    jitter_scale = 1e-5

    while time.time() < deadline and p <= p_target:
        # run a block of iterations at the current smoothing level
        block_iters = 2500
        for _ in range(block_iters):
            if time.time() >= deadline:
                break

            # ---- compute self‑convolution (needed for weighting) ----
            conv = conv_self(a)                     # length 2n‑1
            # ---- weighting for the Lp‑norm gradient -----------------
            eps = 1e-30
            w = np.maximum(conv, eps) ** (p - 1.0)   # non‑negative weights
            # ---- cross‑correlation (gradient of weighted sum) -----
            a_rev = a[::-1]                         # reversed a
            conv_wa = conv_general(w, a_rev)       # length 3n‑2
            # slice the cross‑correlation part that corresponds to ∂/∂a
            start = n - 1
            grad = 2.0 * conv_wa[start:start + n] * p   # scaling by p (optional)

            # ---- normalise gradient direction -----------------------
            gnorm = np.linalg.norm(grad)
            if gnorm > 0:
                direction = grad / (gnorm + 1e-12)
            else:
                direction = grad

            # optional tiny random jitter before the step
            if rng.random() < jitter_prob:
                direction += jitter_scale * (rng.random(n) - 0.5)

            # ---- gradient step (projected) -------------------------
            a_trial = a - lr * direction
            a_trial = project_onto_simplex(a_trial)

            # ---- occasional evaluation of true objective ------------
            if it % 200 == 0:
                trial_score = 2.0 * n * conv_self(a_trial).max()
                if trial_score < best_score - 1e-12:
                    best_score = trial_score
                    best_a = a_trial.copy()
                    last_improve_it = it
                    print(f"[iter {it:07d}] new best true score = {best_score:.6f}")

                # accept the trial if it does not worsen the objective
                # (simple monotone acceptance; Metropolis logic omitted for speed)
                cur_score = trial_score
                a = a_trial
            else:
                # no evaluation – just move on
                a = a_trial

            it += 1

            # ---- learning‑rate decay --------------------------------
            if (it > 0) and (it % decay_every == 0):
                lr = max(lr * lr_decay, 1e-6)

        # ---- increase smoothing exponent p ---------------------------
        p = min(p + p_step, p_target)
        print(f"[schedule] increased smoothing exponent to p={p:.1f}, lr={lr:.6f}, it={it}")

        # early stop if no improvement for a long time
        if it - last_improve_it > 50000:
            print("[early stop] no improvement for many iterations")
            break

    # ------------------------------------------------------------------
    # Final sanity check (the evaluator also clamps values, but our vector
    # is already non‑negative and sum≈1)
    final_score = evaluate_sequence(best_a.tolist())
    print("[done] total iterations =", it)
    print("[done] final reported score =", final_score)
    return best_a.tolist()
