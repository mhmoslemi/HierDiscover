def _conv_fft_general(x, y):
    n = len(x)
    m = len(y)
    N = _next_pow2(n + m - 1)
    X = np.fft.rfft(np.r_[x, np.zeros(N - n)])
    Y = np.fft.rfft(np.r_[y, np.zeros(N - m)])
    conv = np.fft.irfft(X * Y, n=N)[:n+m-1]
    return conv

"""
Propose a candidate step‑function sequence.

Algorithm
---------
* Parameterise the step heights with a soft‑max (logits → a ≥ 0, Σ a = 1).
* Optimise the logits with Adam using a *smooth* approximation of the objective
  2·n·max_k (a*a)_k :
      J(a) = 2·n·log‑sum‑exp(α·(a*a)) / α .
  The smoothing parameter α starts small (≈0.5) and is increased
  linearly to a larger value (≈30) during the run – this yields a
  differentiable surrogate that initially guides the search globally
  and later focuses on the true max.
* The gradient of J w.r.t. a is obtained analytically:
      ∂J/∂a = Σ_k w_k·∂(a*a)_k/∂a ,   w_k = exp(α·conv_k)/Σ exp .
  Using FFT‑based convolution we compute the weighted gradient in
  O(N log N) time.
* Convert the gradient w.r.t. a to a gradient w.r.t. the logits
  (soft‑max chain rule) and apply Adam.
* Run several independent restarts for a handful of sequence lengths
  (1200 … 6000).  The best sequence seen (by the exact
  ``evaluate_sequence`` score) is returned.  The implementation stays
  well within the 1000 s budget.
"""

import time
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _conv_fft_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the full convolution (len(x)+len(y)-1).
    """
    n = x.shape[0]
    m = y.shape[0]
    N = _next_pow2(n + m - 1)
    x_pad = np.zeros(N, dtype=np.float64)
    y_pad = np.zeros(N, dtype=np.float64)
    x_pad[:n] = x
    y_pad[:m] = y
    X = np.fft.rfft(x_pad)
    Y = np.fft.rfft(y_pad)
    conv = np.fft.irfft(X * Y, n=N)[: n + m - 1]
    return conv


def _grad_weighted_conv(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∂/∂a Σ_k w_k·(a*a)_k efficiently via FFT.
    Returns a vector of length len(a) containing the gradient.
    """
    n = a.shape[0]
    L = w.shape[0]          # = 2·n‑1
    conv_len = n + L - 1    # = 3·n‑2
    N = _next_pow2(conv_len)

    a_pad = np.zeros(N, dtype=np.float64)
    w_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    w_pad[:L] = w

    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)

    # a * w
    conv_aw = np.fft.irfft(A * W, n=N)[:conv_len]

    # a * w_rev
    w_rev_pad = np.zeros(N, dtype=np.float64)
    w_rev_pad[:L] = w[::-1]
    W_rev = np.fft.rfft(w_rev_pad)
    conv_aw_rev = np.fft.irfft(A * W_rev, n=N)[:conv_len]

    # grad_i = conv_aw[i] + conv_aw_rev[(L‑1)‑i]
    idx_rev = (L - 1) - np.arange(n)   # shape (n,)
    grad = conv_aw[:n] + conv_aw_rev[idx_rev]
    return grad

def _run_optimizer(
    n: int,
    deadline: float,
    init_z: Optional[np.ndarray] = None,
    max_iters: int = 50000,
) -> Tuple[Optional[List[float]], float]:
    """
    Run Adam on a sequence of length n until the deadline.
    Returns the best sequence (as a plain list) found and its score.
    """
    # hyper‑parameters
    lr0 = 0.05 / np.sqrt(n)          # base learning rate
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    alpha_start, alpha_end = 0.5, 30.0

    # initialise logits
    if init_z is not None:
        if init_z.shape[0] != n:
            if init_z.shape[0] < n:
                pad = np.random.randn(n - init_z.shape[0]) * 0.01
                z = np.concatenate([init_z, pad])
            else:
                z = init_z[:n].copy()
        else:
            z = init_z.copy()
    else:
        z = np.random.randn(n) * 0.02

    a = _logits_to_a(z)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # keep the best seen locally
    best_seq = a.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:   # very unlikely
        best_score = float("inf")

    t = 0
    while time.time() < deadline and t < max_iters:
        # --------------------------------------------------------------
        # 1) self‑convolution a*a (full length 2·n‑1)
        # --------------------------------------------------------------
        conv = _conv_fft_full(a, a)        # length 2·n‑1

        # --------------------------------------------------------------
        # 2) smooth max via log‑sum‑exp, weight vector w
        # --------------------------------------------------------------
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / (max_iters // 2))
        x = alpha * conv
        x_max = np.max(x)
        exp_x = np.exp(x - x_max)          # stable
        sum_exp = np.sum(exp_x)
        w = exp_x / sum_exp                # weights, sum to 1

        # --------------------------------------------------------------
        # 3) gradient w.r.t. a (weighted sum of the individual gradients)
        # --------------------------------------------------------------
        grad_a = _grad_weighted_conv(a, w)   # shape (n,)

        # --------------------------------------------------------------
        # 4) chain rule for the soft‑max (logits → a)
        # --------------------------------------------------------------
        dot = a.dot(grad_a)
        grad_z = a * (grad_a - dot)

        # --------------------------------------------------------------
        # 5) Adam update in log‑space
        # --------------------------------------------------------------
        t += 1
        lr = lr0 / (1.0 + t * 1e-5)        # slow decay

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # occasional jitter to escape plateaux
        if t % 3000 == 0:
            z += np.random.randn(n) * 1e-3

        a = _logits_to_a(z)

        # --------------------------------------------------------------
        # 6) occasional exact evaluation (much slower than the FFT version)
        # --------------------------------------------------------------
        if t % 500 == 0:
            try:
                cur_score = float(evaluate_sequence(a.tolist()))
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[n={n}] iter={t}  new local best {best_score:.6f}")
            except Exception:
                pass

    # final check (in case we never entered the evaluation branch)
    try:
        final_score = float(evaluate_sequence(a.tolist()))
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
    except Exception:
        pass

    return best_seq.tolist(), best_score

                else:
                    init_z = init_logits[:n].copy()

            seq, score = _run_optimizer(n, deadline, init_z=init_z)
            if seq is not None and score < best_score:
                best_score = score
                best_seq = seq
                print(f"[global] New overall best {best_score:.6f} "
                      f"at length {n} (attempt {attempt})")

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = float(evaluate_sequence(best_seq))
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (official evaluation)
    # ------------------------------------------------------------------
    try:
        final_score = float(evaluate_sequence(best_seq))
        print(f"Final best official score {final_score:.6f} "
              f"(length {len(best_seq)})")
    except Exception:
        print("Warning: final evaluation failed.")

    return best_seq
