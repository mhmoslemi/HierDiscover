def compute_subgradient(a, conv, max_val, eps=1e-9, max_active=200):
    act = np.where(conv >= max_val - eps)[0]
    if act.size > max_active:
        # random subset
        act = np.random.choice(act, max_active, replace=False)
    g = np.zeros_like(a)
    n = a.shape[0]
    for k in act:
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # i from lo..hi, j=k-i
        i = np.arange(lo, hi+1)
        j = k - i
        g[i] += 2.0*a[j]
    if act.size > 0:
        g = g / act.size
    return g

"""
Propose a step‐function (non‑negative list) that minimizes
`evaluate_sequence`.  The optimizer works on the probability simplex
(sum = 1) – the objective is scale invariant – and consists of

* a projected gradient descent on the simplex, where the sub‑gradient
  is taken from the (approximately) active set of the self‑convolution
  (all entries within a small relative tolerance of the maximum);
* an aggressive momentum schedule;
* a final simulated‑annealing refinement that transfers a small amount
  of mass between two positions;
* a multi‑scale schedule: after optimizing a length `n` we up‑sample
  the best solution and continue with a larger `n`.

The routine respects the 1000 s time budget and prints intermediate
scores.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – O(n) algorithm (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x | x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT (real)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (size 2*len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Upsample (linear interpolation) while preserving the simplex constraint
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate sequence to length new_n and renormalise to sum‑1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up[:] = 1.0 / new_n
    return up


# ----------------------------------------------------------------------
# Gradient descent on the simplex (using active‑set sub‑gradient)
# ----------------------------------------------------------------------
def _run_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.8,
    beta: float = 0.9,
    max_iter: int = 100000,
    eps_rel: float = 1e-5,
    max_active: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with momentum.
    Returns the best sequence found and its (objective) score = 2*n*max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] GD improve: n={n}, iter={it}, score={cur_score:.6f}"
            )

        # ---- sub‑gradient from (approximately) active set ----
        # all entries within eps_rel * max_val of the maximum are considered active
        eps = max(eps_rel * max_val, 1e-12)
        act_idx = np.where(conv >= max_val - eps)[0]

        # cap the active set size for speed
        if act_idx.size > max_active:
            act_idx = np.random.choice(act_idx, max_active, replace=False)

        # compute sub‑gradient: for each active k add contribution 2*a[j] to coordinates i
        g = np.zeros_like(a)
        for k in act_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1)
            j = k - i
            g[i] += 2.0 * a[j]
        if act_idx.size > 0:
            g /= act_idx.size

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # diminishing step size (sqrt schedule)
        step = step0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing local refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Tiny annealer that repeatedly moves a small amount of probability mass
    from one coordinate to another.  Accepts worsening moves with a
    Metropolis criterion.
    Returns possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move: transfer a small epsilon from i to j
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        # global best tracking
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score
