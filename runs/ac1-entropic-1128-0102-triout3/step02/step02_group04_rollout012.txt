"""
Propose a step‑function sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a projected Adam optimizer on the probability simplex,
using a *soft‑max* smoothing of the max‑operator to obtain a useful
gradient.  At each iteration we

1. compute the self‑convolution a*a,
2. form the probability vector p_i ∝ exp(α·(a*a)_i)  (α grows from 1 to ≈30),
3. compute the exact gradient of the smoothed objective:
       ∇L = 2·conv(a , p_rev)   (p_rev = p reversed),
   and extract the central n entries (reversed again) to get a vector of length n,
4. take an Adam step with a diminishing learning rate,
5. project back onto the probability simplex {x≥0, Σx=1},
6. occasionally inject tiny random noise and, if no improvement has been made
   for a long time, restart from a random Dirichlet draw.

The loop runs until the supplied time budget expires and the best sequence
found is returned (as a plain Python list).  If a previous construction
`height_sequence_1` exists in the global namespace it is used as the starting
point; otherwise we fall back to the classic half‑binomial series.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of the half‑binomial series a_i = C(2i,i)/4**i,
    normalised so that Σ a_i = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(x: np.ndarray, y: np.ndarray, length: int) -> np.ndarray:
    """
    Full convolution of real vectors x and y using a real FFT.
    The result length is len(x) + len(y) - 1.
    The parameter `length` must be a power of two ≥ that length and is
    used for zero‑padding.
    """
    X = np.fft.rfft(x, n=length)
    Y = np.fft.rfft(y, n=length)
    conv = np.fft.irfft(X * Y, n=length)
    return conv[: x.shape[0] + y.shape[0] - 1]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that (approximately) minimises the
    evaluation function.  The routine runs a projected Adam optimiser with a
    smoothed max objective; it returns the best sequence encountered
    (a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)

    # ---------- configuration ----------
    n = 2500                     # length of the sequence
    max_iter = 10_000_000        # safety bound
    eta_start = 0.10             # initial Adam learning rate
    eta_min = 5e-5               # floor for the learning rate
    decay_rate = 0.99            # geometric decay per decay_interval
    decay_interval = 2000        # iterations between learning‑rate decays
    temperature_start = 0.015    # start temperature for occasional uphill moves
    temp_min = 1e-6
    temp_decay = 0.98
    temp_decay_interval = 3000
    restart_threshold = 100_000  # restart after this many iterations without improvement
    noise_prob = 0.006           # probability of adding a random perturbation
    noise_scale = 2e-4           # magnitude of random perturbation
    alpha_start = 1.0            # start smoothing factor (soft‑max temperature inverse)
    alpha_target = 30.0          # final smoothing factor
    alpha_ramp_iters = 200_000   # number of iterations over which to increase α

    # ---------- pre‑computed FFT lengths ----------
    # Length required for a*a (size 2n‑1) and for a * p_rev (size 3n‑2)
    L_self = next_pow2(2 * n - 1)
    L_full = next_pow2(3 * n - 2)

    # ---------- initial sequence ----------
    # Use the previously discovered construction if it exists; otherwise fall back
    # to the half‑binomial series.
    try:
        global height_sequence_1
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.size >= n:
            a = init[:n].copy()
        else:
            raise ValueError
    except Exception:
        a = sqrt_series(n)

    # Normalise to the simplex (sum = 1)
    a = project_onto_simplex(a)

    # ---------- Adam state ----------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta1_pow = 1.0
    beta2_pow = 1.0

    # ---------- bookkeeping ----------
    cur_conv = convolve_fft(a, a, L_self)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max               # sum(a)=1 ⇒ denominator = 1
    best_a = a.copy()
    best_score = cur_score
    last_improve = 0
    eta = eta_start
    temperature = temperature_start
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # leave a safety margin
    it = 0

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    while time.time() < deadline and it < max_iter:
        # --------------------------------------------------------------
        # 1) self‑convolution and smoothed max gradient
        X = np.fft.rfft(a, n=L_full)          # FFT of current a (zero‑padded)
        conv_self_full = np.fft.irfft(X * X, n=L_full)
        conv_self = conv_self_full[: 2 * n - 1]   # length 2n‑1

        # 2) soft‑max over the convolution values
        #    α is increased linearly from α_start to α_target
        alpha = min(alpha_target,
                    alpha_start + (alpha_target - alpha_start) * it / alpha_ramp_iters)

        max_c = conv_self.max()
        exp_scaled = np.exp(alpha * (conv_self - max_c))
        sum_exp = exp_scaled.sum()
        p = exp_scaled / sum_exp                 # probabilities, length 2n‑1

        # 3) gradient of the smoothed objective:
        #    g = 2 * conv(a , reverse(p))  over the central n entries,
        #    then reversed again to obtain the correct ordering.
        p_rev = p[::-1]
        Y = np.fft.rfft(p_rev, n=L_full)
        conv_rev_full = np.fft.irfft(X * Y, n=L_full)
        conv_rev = conv_rev_full[: 3 * n - 2]     # length 3n‑2
        # slice the middle n elements (indices n‑1 … 2n‑2) and reverse
        grad_slice = conv_rev[n - 1 : n - 1 + n][::-1]
        grad = 2.0 * grad_slice                    # gradient wrt a

        # --------------------------------------------------------------
        # 4) Adam update (with bias correction)
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # Prevent a step that would make any entry negative before projection
        step = np.minimum(step, a * 0.95)

        a_trial = a - step

        # occasional random jitter to escape flat regions
        if rng.random() < noise_prob:
            a_trial += (rng.random(n) - 0.5) * noise_scale

        # projection back onto the simplex
        a_trial = project_onto_simplex(a_trial)

        # --------------------------------------------------------------
        # 5) evaluate candidate (fast: we only need the max of the auto‑conv.)
        X_trial = np.fft.rfft(a_trial, n=L_self)
        conv_trial_full = np.fft.irfft(X_trial * X_trial, n=L_self)
        conv_trial = conv_trial_full[: 2 * n - 1]
        max_trial = conv_trial.max()
        trial_score = 2.0 * n * max_trial

        # keep the global best
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            last_improve = it
            print(f"[iter {it:08d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 6) acceptance rule (Metropolis for occasional uphill moves)
        if trial_score <= cur_score:
            a = a_trial
            cur_score = trial_score
        else:
            delta = trial_score - cur_score
            if temperature > 0.0 and rng.random() < np.exp(-delta / temperature):
                a = a_trial
                cur_score = trial_score
            # else: reject, keep current a

        # --------------------------------------------------------------
        # 7) schedule updates
        if it % decay_interval == 0:
            eta = max(eta * decay_rate, eta_min)

        if it % temp_decay_interval == 0:
            temperature = max(temperature * temp_decay, temp_min)

        # --------------------------------------------------------------
        # 8) restart if we have been stuck for a while
        if it - last_improve > restart_threshold:
            a = rng.dirichlet(np.full(n, 0.6))
            a = project_onto_simplex(a)
            cur_score = 2.0 * n * convolve_fft(a, a, L_self)[: 2 * n - 1].max()
            last_improve = it
            # reset Adam statistics to avoid being biased by the old trajectory
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            print(f"[iter {it:08d}] restart triggered – new random start, score={cur_score:.6f}")

        # occasional progress report
        if it % 20000 == 0:
            elapsed = time.time() - start_time
            print(
                f"[status] iter={it:,}, best={best_score:.6f}, cur={cur_score:.6f}, "
                f"elapsed={elapsed:.1f}s"
            )

    # ----------------------------------------------------------------------
    # final safety evaluation using the provided routine
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
