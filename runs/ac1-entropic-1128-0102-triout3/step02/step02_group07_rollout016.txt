def solve_lp(f, M):
    n = len(f)
    # Build sparse constraint matrix A_ub (shape (2*n -1, n))
    # We'll use coo_matrix.
    # Rows and cols arrays.
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    idx = 0
    for j in range(n):
        # for each column j, assign f_i to rows i+j
        rows[idx: idx+n] = np.arange(j, j+n)
        cols[idx: idx+n] = j
        data[idx: idx+n] = f
        idx += n
    A_ub = coo_matrix((data, (rows, cols)), shape=(2*n -1, n)).tocsc()
    b_ub = np.full(2*n -1, M)
    # bounds = (0, None) for each variable
    bounds = [(0, None)] * n
    # objective: maximize sum(b) => linprog solves minimization, so objective c = -1
    c = -np.ones(n)
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return current f
        return f.copy()
    b = res.x
    # Normalize to sum 1
    if b.sum() <= 0:
        return f.copy()
    b = b / b.sum()
    return b

"""
Propose a candidate step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a))**2

where `conv` is the discrete self‑convolution.  The algorithm consists of

1. **Initial seeding** –  a short power‑law sweep (p≈0.5) gives a good
   starting point.  If the global variable ``height_sequence_1`` is
   present, it is used (re‑normalised).

2. **Smoothed‑gradient descent** –  instead of moving only against the
   single active convolution entry, a soft‑max over the *K* largest
   entries is taken (weights = exp(ε·(c‑c_max))).  The resulting
   weighted gradient is inexpensive (O(K·n)) and pushes the whole
   convolution profile down.  A simple momentum term and an exponential
   schedule for the step‑size (η) and the smoothing factor (ε) are used.

3. **Fine‑grained peak descent** –  after the smoothed phase the usual
   projected‑gradient descent on the *single* peak (as in the reference
   implementation) is run for a few thousand iterations to polish the
   result.

All updates are projected onto the probability simplex (non‑negative,
unit‑sum) to keep the evaluation function unchanged.  The routine respects
the 1000 s time limit (a safety margin of 5 s is kept) and prints a few
progress messages.  The final list is passed through ``evaluate_sequence``
once before returning to guarantee that the reported score matches the
competition evaluator.
"""

import time
import numpy as np
from scipy.optimize import linprog  # only imported if needed later

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i = (i+offset)^(-p), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def half_binomial_sequence(n: int) -> np.ndarray:
    """Classic half‑binomial (square‑root of central binomial coefficients)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Self‑convolution using a real FFT (length 2·len(x)‑1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w | w_i ≥ 0 , Σ w_i = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Sub‑gradient of conv[a,a][peak_idx] with respect to a.
    g_i = 2·a_{peak_idx-i} whenever the index is in range, else 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


def compute_score_from_conv(conv: np.ndarray, n: int) -> float:
    """Score = 2·n·max(conv)   (used while sum(a)=1)."""
    return 2.0 * n * conv.max()

    n = 3000                     # length of the candidate (adjustable)
    max_iter_smooth = 150_000    # upper bound for the smoothed phase
    max_iter_fine = 30_000       # iterations for the final peak‑only phase

    # hyper‑parameters for the smoothed phase
    eta_start = 0.15             # initial step size
    eta_decay = 0.9995
    eps_start = 30.0             # initial smoothing factor (large ⇒ focus on the peak)
    eps_decay = 0.999            # slowly broaden the smoothing
    K = 7                        # number of top convolution entries that receive weight
    momentum_beta = 0.9          # momentum coefficient

    # hyper‑parameters for the fine phase (same as the reference implementation)
    eta_fine_start = 0.08
    decay_factor = 0.93
    decay_every = 1500
    temperature_start = 0.03

    # --------------------------------------------------------------
    # time management
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

        # record best
