"""
Propose a step‑function sequence that minimizes the max‑self‑convolution
objective used by `evaluate_sequence`.

Algorithm
---------
1. Initialise a probability vector `a` (non‑negative, sum = 1) by scanning a
   family of power‑law sequences and, if present, the previously found
   `height_sequence_1`.  The best among these is kept as the start point.

2. Optimise a *smooth* approximation of the max‑convolution:
        F(a) = (1/λ)·log Σ_k exp(λ·(a∗a)_k) .
   Its gradient is
        ∇F(a) = 2·(w∗a) ,
   where w is the soft‑max weights:
        w_k = exp(λ·(a∗a)_k) / Σ_j exp(λ·(a∗a)_j).

   Using λ that slowly grows (annealing) yields a differentiable
   objective that starts with a gentle landscape and ends up approximating
   the original max.  At each iteration we compute `w` and the gradient via
   FFT‑based convolutions.

3. Perform a mirror‑descent step on the probability simplex:
        a ← a * exp(−η·∇F(a))   (component‑wise),
        a ← a / Σ a.
   This keeps the vector non‑negative and normalised without an explicit
   projection.

4. Occasionally perform a random “mass‑transfer” move between two
   coordinates to escape shallow minima and accept occasional uphill
   moves with a very small probability.

5. Every few thousand iterations we evaluate the exact objective using
   the fast version of `evaluate_sequence` and keep the best sequence
   seen so far.

The loop runs until the allotted time budget (default 1000 s) expires.
The final best sequence is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast computation of the exact objective (without the safety checks)."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # length needed for a∗a is 2n‑1; use next power of two for FFT efficiency
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def build_initial_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Power‑law start: a_i ∝ (i+offset)^{-p}, normalised to ℓ¹‑norm = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a
