        max_val = conv.max()
        score = 2.0*n*max_val

            best_score = score
            best_a = a.copy()

        # lambda schedule
        lam = lambdas[0] + (lambdas[1] - lambdas[0]) * (t/steps)**0.5
        # compute softmax p
        m = conv.max()
        exp_vals = np.exp(lam*(conv - m))
        p = exp_vals / exp_vals.sum()
        # gradient:
        # sum_window

        prefix = np.cumsum(p)
        # sum_window_i = prefix[i + n -1] - (prefix[i-1] if i>0 else 0)

def _conv_fft(x, y):
    n = len(x)
    m = len(y)
    L = 1 << ((n + m - 1).bit_length())
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    conv = np.fft.irfft(X * Y, L)[: n + m - 1]
    return conv

"""
Smooth deterministic‑annealing gradient descent on the simplex
with a final greedy hill‑climb.  The algorithm works on the
probability simplex (∑a_i = 1) – the evaluation score is
scale‑invariant,  score = 2·n·max(a*a).  A continuation
schedule slowly sharpens a soft‑max approximation of the max,
which provides a smooth gradient that is easier to optimise
than the raw sub‑gradient.  After the smooth phase a cheap
local move‑transfer hill‑climber cleans up any remaining
spikes.

The code respects the 1000 s budget, works with any length
(up‑sampling the current best when the length grows) and
always returns a non‑negative list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – O(n) algorithm (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution via FFT (any two lengths)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of x and y (real FFT)."""
    n = x.shape[0]
    m = y.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    conv = np.fft.irfft(X * Y, L)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Self‑convolution (a * a) – convenience wrapper
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself (linear, not circular)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Upsample a probability vector to a new length by linear interpolation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_smooth_gradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    lam_min: float = 0.1,
    lam_max: float = 8.0,
    beta: float = 0.7,
    step0: float = 0.8,
    max_iter: int = 50000,
):
    """Return a locally optimal probability vector and its score."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    # pre‑allocate for speed
    ones_n = np.ones(n, dtype=float)

    while time.time() < deadline and it < max_iter:
        it += 1
        # convolution and current objective
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"smooth‑grad improve: n={n}, iter={it}, score={cur_score:.6f}"
            )
            # early stop if we already beat the target
            if best_score <= 1.503:
                break

        # continuation schedule for λ (smooth‑max sharpening)
        lam = lam_min + (lam_max - lam_min) * (it / max_iter) ** 0.5

        # soft‑max weights over the convolution entries
        m = conv.max()
        exp_vals = np.exp(lam * (conv - m))
        p = exp_vals / exp_vals.sum()                # shape (2n‑1,)

        # ------- gradient of the smooth max  ---------------------------------
        # 1) window sums Σ_{k=i}^{i+n‑1} p_k  (size n)
        prefix = np.cumsum(p)
        # end indices are i+n‑1 ranging from n‑1 … 2n‑2
        end = prefix[n - 1 : 2 * n - 1]               # length n
        # start indices i‑1, with 0 for i=0
        start = np.concatenate(([0.0], prefix[: n - 1]))
        win_sum = end - start                         # shape (n,)

        # 2) second part = convolution of p with a reversed
        a_rev = a[::-1]
        conv_rev = _conv_fft(p, a_rev)               # length 3n‑2
        second = conv_rev[n - 1 : n - 1 + n]          # centre slice, shape (n,)

        # full gradient of the smoothed objective (to be minimized)
        grad = a * win_sum + second

        # ------- momentum + step size ---------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb: tiny ε‑mass transfers that are only accepted
# if they improve the true max‑convolution objective.
# ----------------------------------------------------------------------
def _run_hill_climb(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    eps_factor: float = 0.15,
    attempts: int = 200000,
):
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    it = 0

    while time.time() < deadline and it < attempts:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 1e-12:
            continue
        # move a fraction of a[i] to a[j]
        eps = a[i] * np.random.rand() * eps_factor
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv = _conv_self(a_prop)
        new_score = 2.0 * n * conv.max()
        if new_score < best_score:
            best_score = new_score
            best_seq = a_prop.copy()
            a = a_prop
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"hill‑climb improve: n={n}, iter={it}, score={new_score:.6f}"
            )
            if best_score <= 1.503:
                break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative list) that minimises
    `evaluate_sequence`.  A smooth‑max gradient descent on the simplex
    is combined with a final greedy hill‑climb and a multi‑scale
    up‑sampling schedule.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # optional starter from a previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            s = starter.sum()
            if s > 0:
                starter = starter / s
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # length schedule – we grow the support geometrically.
    # ------------------------------------------------------------------
    lengths = [800, 1200, 2400, 3600, 4800, 6000, 7200]
    np.random.shuffle(lengths)        # avoid bias if time runs out early

    # keep the current best distribution (always on the simplex)
    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise a0: up‑sample the current best or start random
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # allocate a time slice for this length
        remain = deadline - time.time()
        if remain <= 0:
            break
        slice_deadline = time.time() + max(10.0, 0.18 * remain)

        # ------------------------------------------------------------------
        # smooth‑max gradient phase
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_smooth_gradient(
            a0,
            n,
            slice_deadline,
            start_time,
            lam_min=0.1,
            lam_max=9.0,
            beta=0.7,
            step0=0.8,
            max_iter=80000,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"global improve after gradient: n={n}, score={score_opt:.6f}"
            )

        # ------------------------------------------------------------------
        # greedy hill‑climb refinement (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            hill_deadline = time.time() + max(6.0, 0.07 * (deadline - time.time()))
            a_ref, score_ref = _run_hill_climb(
                a_opt,
                n,
                hill_deadline,
                start_time,
                best_score,
                best_seq,
                eps_factor=0.12,
                attempts=30000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] "
                    f"global improve after hill‑climb: n={n}, score={score_ref:.6f}"
                )

        # store the best so far for the next up‑sampling step
        global_a = best_seq.copy()

        # early exit if we already beat the target bound
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # safeguard: if nothing was found produce a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # convert to plain Python list for the evaluator
    return [float(x) for x in best_seq]
