import time, numpy as np, math

def _project_simplex_simple(v):
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total

def _auto_conv(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    return conv

def _subgradient(a, eps_rel):
    n = a.shape[0]
    conv = _auto_conv(a)   # returns full conv of length 2n-1
    max_val = conv.max()
    eps = max_val * eps_rel
    K = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g, max_val

"""
propose_candidate
~~~~~~~~~~~~~~~~~

Search for a non‑negative step‑function (a probability vector) that
minimises

    score = 2 * n * max(conv(a, a))

where ``conv`` is the full linear auto‑convolution.
The score is exactly the value returned by ``evaluate_sequence`` when the
sequence is normalised to sum = 1, because the evaluator rescales by the
square of the sum.

The optimiser is a projected Adam (adaptive‑momentum) method
applied to a sub‑gradient of the ``max``‑operator.
A progressive up‑sampling scheme (≈1.5× increase) is used so that a good
shape found for a small length is refined for larger lengths.
During optimisation the active set of “near‑maximal’’ convolution entries
is enlarged (ε is decreased from 5 % to ≈0.8 % during an inner run) which
provides a smoother descent direction than a single worst entry.

The implementation is deliberately lightweight:
* the simplex projection is performed by clipping negatives and renormalising
  (sufficient for this non‑convex problem and much faster than the
  Euclidean projection);
* FFT‑based convolution gives O(n log n) per iteration;
* a dynamic iteration budget respects the global time limit (default 1000 s);
* the best sequence seen so far is recorded and returned at the end.

The routine works with the previously discovered construction
(`height_sequence_1`) if it exists, otherwise it starts from a random
Dirichlet point.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _simplex_proj(v: np.ndarray) -> np.ndarray:
    """
    Simple simplex projection used in this optimiser:
    – clip negative entries,
    – renormalise to sum = 1 (uniform fallback if the vector becomes zero).
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _auto_conv(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Full linear auto‑convolution of a real vector `a` using FFT.
    Returns a vector of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _score_of_seq(a: np.ndarray) -> float:
    """
    Score of a probability vector (sum = 1):
        score = 2 * n * max(conv(a, a))
    where `conv` is the full auto‑convolution.
    """
    n = a.shape[0]
    conv = _auto_conv(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) achieves a lower
    ``evaluate_sequence`` score than the previous best (≈1.519272).
    The routine respects a wall‑clock budget (default 1000 seconds) and
    prints intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – leave a couple of seconds for clean‑up
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Initial sequence (use the previous best if available)
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
    except Exception:                     # no previous construction in globals
        init_len = 800
        init = np.random.rand(init_len)

    init = _simplex_proj(init)
    best_seq = init.copy()
    best_score = _score_of_seq(best_seq)
    print(f"[{time.time() - start_time:6.2f}s] Initial sequence: n={len(best_seq)}, "
          f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling + Adam optimisation
    # ------------------------------------------------------------------
    current = best_seq.copy()
    cur_n = current.shape[0]

    MAX_LEN = 10000            # upper bound for the length (adjustable)
    GROW_FACTOR = 1.5          # ~50 % increase per stage

    # Adam hyper‑parameters (shared across all sizes)
    beta1, beta2 = 0.9, 0.999

        if next_n > MAX_LEN:

        # ----- up‑sample current shape ----------------------------------
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a = np.interp(x_new, x_old, current)
        a = _simplex_proj(a)

        n = next_n
