def compute_gradient(a, eps_rel=0.01, alpha=10.0):
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    max_val = conv.max()
    # get indices within tolerance
    eps = max_val * eps_rel
    K = np.where(conv >= max_val - eps)[0]
    # compute softmax weights
    rel = conv[K] - max_val  # <= 0
    w = np.exp(alpha * rel)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / len(w))
    else:
        w = w / w_sum
    # compute gradient
    g = np.zeros_like(a)
    for idx, k in enumerate(K):
        wk = w[idx]
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * wk * a[j_idx]
    return g, max_val

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
