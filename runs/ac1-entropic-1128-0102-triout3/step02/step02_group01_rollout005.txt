linearly up‑sample the shape to larger lengths (≈1.5× growth, up to a
user‑defined MAX_LEN).  At each scale the step size η decays like
η₀/√t and the smoothing parameter α is increased from a gentle value
to a sharp one.  Symmetry (a ≈ a[::-1]) and occasional random jitter
help to avoid poor local minima.  The best sequence seen across all
scales is returned.

The routine respects the 1000 s time budget, prints intermediate
improvements, and never returns an empty or invalid list.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _score_of_seq(a: np.ndarray) -> float:
    """
    Compute the exact evaluation score (same as evaluate_sequence for a
    normalized to sum 1) without the safety clipping.
    """
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    Af = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(Af * Af, n=N)[:L]
    return float(2.0 * n * conv.max())


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return soft‑max probabilities proportional to exp(alpha*conv).
    Numerical stability is ensured by subtracting the max.
    """
    max_c = conv.max()
    exp_vals = np.exp(alpha * (conv - max_c))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        # fallback to uniform distribution (should never happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp


def _mirror_descent_step(a: np.ndarray,
                         grad: np.ndarray,
                         eta: float,
                         eps: float = 1e-12) -> np.ndarray:
    """
    Perform one exponentiated‑gradient (mirror descent) step and project onto
    the probability simplex.
    """
    # multiplicative update (ensure positivity)
    a = a * np.exp(-eta * grad)
    # avoid complete collapse to zero
    a = np.maximum(a, eps)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    return a


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (a step function) with a low
    `evaluate_sequence` score.  The routine runs for at most `budget_s`
    seconds and prints intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 0. Helper: read an optional initial construction from the global
    #    namespace.  If unavailable, start from a random simplex point.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 0:
            init = init / s
        else:
            raise ValueError
        print(f"Initialised from height_sequence_1: n={len(init)}")
    except Exception:
        init_len = 800                     # a modest start size
        init = np.random.rand(init_len)
        init = init / init.sum()
        print(f"Random initialisation: n={init_len}")

    best_seq = init.copy()
    best_score = _score_of_seq(best_seq)
    print(f"Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Progressive up‑sampling loop.
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_n = cur_seq.shape[0]

    MAX_LEN = 5000          # upper bound for the length (adjustable)

    while True:
        # decide next length (≈1.5× bigger, never exceeding MAX_LEN)
        nxt_n = int(cur_n * 1.5)
        if nxt_n > MAX_LEN:
            nxt_n = MAX_LEN
        if nxt_n <= cur_n:
            break

        # ----- up‑sample shape linearly -----
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, nxt_n)
        a = np.interp(x_new, x_old, cur_seq)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(nxt_n, 1.0 / nxt_n)
        else:
            a = a / a_sum

        # ----- optimisation parameters for this size -----
        n = nxt_n
        L = 2 * n - 1
        N = _next_pow2(3 * n - 2)               # length big enough for both conv & corr
        # iteration budget ~ proportional to 1/n (more iterations for small problems)
        base_iters = 80000
        max_iters = int(base_iters * 1200 / n) + 20000
        max_iters = max(max_iters, 15000)

        eta0 = 0.5                               # initial step size for mirror descent
        alpha_start = 5.0
        alpha_end = 200.0

        no_improve_cnt = 0
        stagnation_limit = 3000

        it = 0
        while time.time() < deadline and it < max_iters:
            it += 1

            # ----- auto‑convolution via FFT -----
            Af = np.fft.rfft(a, n=N)
            conv = np.fft.irfft(Af * Af, n=N)[:L]
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # record improvement
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                best_n = n
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"Improved: n={n}, iter={it}, score={cur_score:.6f}")
                no_improve_cnt = 0
            else:
                no_improve_cnt += 1

            # ----- soft‑max smoothing of the max -----
            alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / max_iters)
            p = _softmax_weights(conv, alpha)

            # ----- gradient (cross‑correlation a ⋆ p) -----
            Pf = np.fft.rfft(p, n=N)
            corr = np.fft.irfft(Af * np.conj(Pf), n=N)[:n]   # linear correlation
            grad = 2.0 * np.real(corr)

            # ----- mirror‑descent step -----
            step = eta0 / math.sqrt(it)
            a = _mirror_descent_step(a, grad, step)

            # enforce (optional) mirror symmetry to reduce the search space
            a = (a + a[::-1]) * 0.5

            # ----- random jitter if we appear stuck -----
            if no_improve_cnt >= stagnation_limit:
                jitter = 0.02
                a = a * (1.0 + jitter * np.random.randn(n))
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum == 0.0:
                    a = np.full(n, 1.0 / n)
                else:
                    a = a / a_sum
                no_improve_cnt = 0

        # prepare for the next size
        cur_seq = a
        cur_n = n
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2. Final refinement on the best length found (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        L = 2 * n - 1
        N = _next_pow2(3 * n - 2)

        eta0 = 0.3
        alpha_start = 20.0
        alpha_end = 500.0
        max_refine = 25000
        it = 0
        a = best_seq.copy()
        while time.time() < deadline and it < max_refine:
            it += 1
            Af = np.fft.rfft(a, n=N)
            conv = np.fft.irfft(Af * Af, n=N)[:L]
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Final refine: "
                      f"n={n}, iter={it}, score={cur_score:.6f}")

            alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / max_refine)
            p = _softmax_weights(conv, alpha)
            Pf = np.fft.rfft(p, n=N)
            corr = np.fft.irfft(Af * np.conj(Pf), n=N)[:n]
            grad = 2.0 * np.real(corr)

            step = eta0 / math.sqrt(it)
            a = _mirror_descent_step(a, grad, step)
            a = (a + a[::-1]) * 0.5   # keep symmetry
        # one final check
        if _score_of_seq(a) < best_score:
            best_seq = a.copy()
            best_score = _score_of_seq(best_seq)

    # ------------------------------------------------------------------
    # 3. Return the best sequence (as plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        # fallback: uniform sequence of length 100
        best_seq = np.full(100, 1.0 / 100)

    print(f"Best score achieved: {best_score:.6f} (length {best_seq.shape[0]})")
    return [float(v) for v in best_seq]
