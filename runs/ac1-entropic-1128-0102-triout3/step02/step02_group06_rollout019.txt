def greedy_sidon(n):
    A = []
    sums = set()
    for i in range(n):
        # check if adding i would cause collision
        collision = False
        for a in A:
            s = a + i
            if s in sums:
                collision = True
                break
        if not collision:
            # also check i+i
            if (i+i) in sums:
                continue
            # add
            for a in A:
                sums.add(a + i)
            sums.add(i + i)
            A.append(i)
        # stop if reached large size
        if len(A) >= int(np.sqrt(2*n)):
            break
    return A

import time
import numpy as np

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a low‑max self‑convolution sequence using Sidon sets.

    The algorithm builds a set of indices whose pairwise sums are all
    distinct (a Sidon / B2 set).  When the same weight is placed on those
    indices the self‑convolution contains at most one non‑zero term per
    index, so the maximal entry equals the square of that weight.
    The score is therefore `2·n / m²` where `m` is the size of the Sidon
    set.  The routine repeatedly draws random orderings of the indices
    and greedily constructs a Sidon set, keeping the largest one found
    within the allotted time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # length of the sequence – reuse a previous construction if present
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)          # use the same length
    else:
        n = 2401

    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # theoretical upper bound on the size of a Sidon set in {0,…,n‑1}
    max_m_theoretical = int((-1 + np.sqrt(16 * n - 7)) // 2)

    best_m = 0
    best_seq = None
    best_score = float('inf')
    iteration = 0

    # Helper: greedy Sidon set construction from a shuffled list of indices
    def greedy_sidon(order):
        selected = []
        sums = set()
        for idx in order:
            # quick reject if the self‑sum already used
            if (idx + idx) in sums:
                continue
            # check sums with already selected elements
            conflict = False
            for a in selected:
                if (a + idx) in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # accept the index
            for a in selected:
                sums.add(a + idx)
            sums.add(idx + idx)
            selected.append(idx)
        return selected

    # main search loop – keep trying until time expires
    while time.time() < deadline:
        iteration += 1

        # random ordering of the positions
        order = np.random.permutation(n)
        sidon = greedy_sidon(order)

        m = len(sidon)
        if m == 0:
            continue

        # compute the theoretical score for this size
        cur_score = 2.0 * n / (m * m)

        if cur_score < best_score:
            # construct the explicit sequence (uniform weight on the Sidon positions)
            weight = S / m
            seq = np.zeros(n, dtype=np.float64)
            seq[sidon] = weight
            # sanity check using the provided evaluator (cheap)
            eval_score = evaluate_sequence(seq.tolist())
            if eval_score < best_score:
                best_score = eval_score
                best_m = m
                best_seq = seq.copy()
                print(f"[{iteration:06d}] New best: m={best_m}, score={best_score:.6f}")

        # stop early if we have reached the theoretical optimum for the current m
        # (i.e. no better score can be found with the same m)
        if best_m == max_m_theoretical:
            # we cannot exceed the max possible size, break
            break

    if best_seq is None:
        # fallback – uniform sequence (should never happen)
        best_seq = np.full(n, S / n, dtype=np.float64)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Finished after {iteration} iterations. Best size m={best_m}, final score={final_score:.6f}")
    return best_seq.tolist()

import time
import numpy as np

# ----------------------------------------------------------------------
#  Sidon‑set based construction
# ----------------------------------------------------------------------
def _greedy_sidon(order, n):
    """
    Build a B2 (Sidon) set from a given ordering of the integers 0..n‑1.
    The set has the property that all pairwise sums i+j (i≤j) are distinct.
    """
    selected = []
    sums = set()
    for idx in order:
        # reject if the sum with itself is already taken
        if (idx + idx) in sums:
            continue
        # reject if any sum with an already selected element collides
        collision = False
        for a in selected:
            if (a + idx) in sums:
                collision = True
                break
        if collision:
            continue
        # accept this index
        for a in selected:
            sums.add(a + idx)
        sums.add(idx + idx)
        selected.append(idx)
    return selected


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence`` using
    Sidon‑set constructions.

    The idea: choose a set of positions whose pairwise sums are all
    different (a Sidon/B2 set).  Putting the same weight on those
    positions makes every entry of the self‑convolution either 0 or that
    weight squared – the maximum entry is therefore the square of the
    weight.  For a set of size ``m`` the score equals ``2·n / m²``,
    so the algorithm repeatedly draws random orderings of the indices,
    greedily builds a Sidon set and keeps the largest one found within
    the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)          # reuse length if known
    else:
        n = 2401

    target_sum = np.sqrt(2.0 * n)            # required ℓ₁‑norm

    # theoretical maximum size of a Sidon set in {0,…,n‑1}
    max_m_theoretical = int((-1 + np.sqrt(16 * n - 7)) // 2)

    best_m = 0
    best_seq = None
    best_score = float('inf')
    it = 0

    while time.time() < deadline:
        it += 1

        # random permutation of the indices 0 … n‑1
        order = np.random.permutation(n)
        sidon = _greedy_sidon(order, n)
        m = len(sidon)
        if m < 2:                # need at least two points to be useful
            continue

        # theoretical score for a Sidon set of this size
        # (max entry = (target_sum/m)², score = 2·n·max/(target_sum²) = 2·n/m²)
        cur_score = 2.0 * n / (m * m)

        if cur_score < best_score:
            # build the actual sequence (constant weight on the chosen indices)
            weight = target_sum / m
            seq = np.zeros(n, dtype=np.float64)
            seq[sidon] = weight

            # verify with the provided evaluator (cheap – O(n log n))
            eval_score = evaluate_sequence(seq.tolist())
            if eval_score < best_score:
                best_score = eval_score
                best_m = m
                best_seq = seq.copy()
                print(f"[{it:06d}] New best: |A|={best_m}, score={best_score:.6f}")

        # early exit if we have already reached the provable optimum for this n
        if best_m == max_m_theoretical:
            break

    # ------------------------------------------------------------------
    # final sanity: if nothing succeeded fall back to uniform distribution
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        best_m = n

    print(f"Finished after {it} iterations. Best |A|={best_m}, final score={best_score:.6f}")
    return best_seq.tolist()
