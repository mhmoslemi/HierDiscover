def propose_candidate(...):
    rng = np.random.default_rng(seed)
    n = 2500
    a = sqrt_series(n)  # sums to 1

    best_a = a.copy()
    best_score = 2 * n * np.convolve(a,a).max()

    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    eta0 = 0.4  # step size start
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 0
    last_improve_iter = 0
    restart_threshold = 50000
    smoothing_factor = 0.03  # mu = smoothing_factor * M
    noise_prob = 0.01
    noise_scale = 1e-4

    while time.time() < deadline and iteration < max_iter:
        conv = conv_fft(a)  # length 2n-1
        M = conv.max()
        # compute smoothed gradient
        mu = max(smoothing_factor * M, 1e-12)
        # compute weights
        w = np.exp(conv / mu)
        w_sum = w.sum()
        w = w / w_sum

        # gradient via correlation
        # compute correlation = a * w correlation
        L = 1 << ( (2*n-1 + n - 1).bit_length() )  # next pow2 >= 3n-2
        # but we can compute via np.fft.rfft with size L
        X = np.fft.rfft(a, n=L)
        W = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(np.conj(X) * W, n=L)  # length L
        offset = len(w) - 1  # start index for zero lag correlation
        grad = 2.0 * corr[offset:offset + n]

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (iteration+1))
        v_hat = v / (1 - beta2 ** (iteration+1))
        step = eta0 / (np.sqrt(v_hat) + eps) * m_hat

        # step size maybe adaptively decayed
        a_new = a - step
        # clip to non-negative
        a_new = np.maximum(a_new, 0.0)

        # random noise occasional
        if rng.random() < noise_prob:
            a_new += (rng.random(n) - 0.5) * noise_scale

        # project onto simplex
        a_new = project_onto_simplex(a_new)

        # evaluate
        cur_score = 2 * n * M  # current
        conv_new = conv_fft(a_new)
        new_M = conv_new.max()
        new_score = 2 * n * new_M

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            last_improve_iter = iteration
            print(f"[iter {iteration:08d}] new best score {best_score:.6f}")

        # acceptance rule: accept if improved, else sometimes accept
        if new_score <= cur_score:
            a = a_new
        else:
            # if not improved, maybe accept with low probability
            if rng.random() < 0.001:
                a = a_new

        # restart if stuck
        if iteration - last_improve_iter > restart_threshold:
            # restart from random Dirichlet (smooth)
            a = rng.dirichlet(np.full(n, 0.5))
            a = project_onto_simplex(a)  # should already sum to 1
            # optionally reset Adam moments
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            last_improve_iter = iteration
            print(f"[iter {iteration}] restart triggered, score approx {2*n*conv_fft(a).max():.6f}")

        iteration += 1
    final_score = evaluate_sequence(best_a.tolist())
    print("[done] best reported", final_score)
    return best_a.tolist()

"""
Propose a step‑function sequence that minimizes the evaluation function.

Algorithm
---------
* The feasible set is the probability simplex (a_i ≥ 0, Σ a_i = 1).  The
  objective is  f(a) = 2·n·max_k (a * a)_k , which is convex but
  non‑smooth (the “max” creates a kink).
* We approximate the max with a smooth log‑sum‑exp:
        f_μ(a) = 2·n· μ·log( Σ_k exp( (a*a)_k / μ ) )
  The gradient of this surrogate is a weighted sum of the sub‑gradients,
  where the weights are softmax‑like probabilities over the convolution
  values.
* The gradient of a single convolution entry (k) is
        ∂(a*a)_k / ∂a_i = 2·a_{k‑i}      (zero when the index is out of range).
  Hence the gradient of the smoothed objective can be obtained
  efficiently by a single correlation a ↔ w, where w is the softmax
  weight vector.
* We use an Adam‑style adaptive step with Euclidean projection onto the
  simplex after each update (the projection code implements the Duchi et
  al. algorithm).  The steps are kept small, the learning‑rate decays
  slowly, and a tiny amount of random noise is added occasionally to
  escape plateaus.
* If no improvement is seen for many iterations we restart from a fresh
  Dirichlet draw.  The best sequence found during the whole run is
  returned.

The whole loop runs under the time budget (≈ 950 s) and typically
performs several hundred thousand cheap FFT‑based iterations, often
pushing the score well below 1.54 for n = 2500.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of a_i = C(2i,i) / 4**i (half‑binomial series),
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Fast convolution x * x using a real FFT.
    Length of result is 2*len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation `corr[i] = Σ_j a[j] * w[i+j]` for i = 0 … len(a)-1.
    Implemented via FFT using a size that avoids circular wrap‑around.
    """
    n = a.size
    m = w.size
    L = 1 << ((n + m - 1).bit_length())   # next power of two
    # FFT of a and w
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    # correlation = ifft( conj(A) * W )
    corr_full = np.fft.irfft(np.conj(A) * W, n=L)
    offset = m - 1                     # zero‑lag position in the full correlation
    return corr_full[offset:offset + n]


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: float = 950.0,
    n: int = 2500,
    **kwargs,
) -> list[float]:
    """
    Search for a sequence a ∈ Δ_n (non‑negative, sum = 1) that minimizes
    evaluate_sequence(a).  The optimizer uses a smoothed version of the
    max‑convolution objective together with Adam‑style adaptive steps and
    simplex projection.
    """
    rng = np.random.default_rng(seed)

    # ----- initialise ----------------------------------------------------
    a = sqrt_series(n)                     # half‑binomial start, Σa = 1
    # keep a copy of the best solution seen so far
    best_a = a.copy()
    best_score = 2.0 * n * conv_fft(a).max()

    # ----- Adam state ----------------------------------------------------
    m = np.zeros_like(a)                   # first moment
    v = np.zeros_like(a)                   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # ----- hyper‑parameters -----------------------------------------------
    eta0 = 0.5                # base learning‑rate for Adam
    smooth_start = 0.25       # initial smoothing factor μ = smooth * M
    smooth_end = 0.02         # final smoothing factor (lower → sharper)
    decay_smooth_iters = 400000   # iterations over which smoothing decays
    noise_prob = 0.007
    noise_scale = 1e-4
    restart_thresh = 150000   # iterations without improvement before restart

    # ----- bookkeeping ----------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    it = 0
    last_improve = 0

    # pre‑compute FFT length for correlation (constant because n is fixed)
    corr_fft_len = 1 << ((3 * n - 2).bit_length())  # enough for a (n) and w (2n‑1)

    while time.time() < deadline:
        # ---- current objective ------------------------------------------------
        conv = conv_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M

        # ---- smoothing factor (linear decay towards smooth_end) -------------
        if it < decay_smooth_iters:
            smooth = smooth_start + (smooth_end - smooth_start) * (it / decay_smooth_iters)
        else:
            smooth = smooth_end
        mu = max(smooth * M, 1e-14)               # avoid division by zero

        # ---- soft‑max weights over the convolution values --------------------
        # w_k ∝ exp(conv_k / μ)
        # to keep numbers stable we subtract the max before exponentiation
        exp_arg = conv / mu
        max_exp = exp_arg.max()
        w = np.exp(exp_arg - max_exp)
        w_sum = w.sum()
        w /= w_sum                               # now Σ w = 1

        # ---- gradient of the smoothed objective (via correlation) ----------
        # g_i = 2 * Σ_k w_k * a_{k-i}
        grad = 2.0 * correlation_fft(a, w)      # length n

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        bias_correction1 = 1.0 - beta1 ** (it + 1)
        bias_correction2 = 1.0 - beta2 ** (it + 1)

        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        eta = eta0 / (np.sqrt(v_hat) + eps)
        step = eta * m_hat

        # ---- tentative new point (additive) ---------------------------------
        a_trial = a - step
        a_trial = np.maximum(a_trial, 0.0)       # keep non‑negative

        # ---- occasional random perturbation ---------------------------------
        if rng.random() < noise_prob:
            a_trial += (rng.random(n) - 0.5) * noise_scale
            a_trial = np.maximum(a_trial, 0.0)

        # ---- project back onto the simplex ------------------------------------
        a_trial = project_onto_simplex(a_trial)

        # ---- evaluate trial --------------------------------------------------
        conv_trial = conv_fft(a_trial)
        new_score = 2.0 * n * conv_trial.max()

        # ---- keep track of best ever -----------------------------------------
        if new_score < best_score:
            best_score = new_score
            best_a = a_trial.copy()
            last_improve = it
            print(f"[iter {it:08d}] new best score = {best_score:.6f}")

        # ---- acceptance rule (always accept if better) -----------------------
        if new_score <= cur_score:
            a = a_trial
        else:
            # rarely accept uphill moves (tiny probability) to add stochasticity
            if rng.random() < 1e-4:
                a = a_trial

        # ---- restart logic (to avoid getting stuck) --------------------------
        if it - last_improve > restart_thresh:
            a = rng.dirichlet(np.full(n, 0.6))
            a = project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"[iter {it:08d}] restart triggered – new random start, score≈{2.0*n*conv_fft(a).max():.6f}")

        # ---- progress report every 50 000 iterations -------------------------
        if (it + 1) % 50000 == 0:
            elapsed = time.time() - start_time
            print(f"[iter {it+1:08d}] elapsed {elapsed:.1f}s, cur_score={cur_score:.6f}, best_score={best_score:.6f}")

        it += 1

    # ----- final safety evaluation -------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
