"""
Propose a candidate step‐function sequence.

Algorithm
---------
* Treat the coefficients as a probability vector (soft‑max of logits).
* Minimise the smooth approximation
        L(α) = (1/α)·log∑_k exp(α·(a*a)_k)
  of the true objective max_k (a*a)_k.
  The gradient of L w.r.t. the logits can be written using two FFT‑convolutions,
  so each optimisation step costs only three FFTs.
* α (inverse temperature) is increased linearly from 1 up to a large value
  (≈200) – early iterations move the whole shape, late iterations focus on the
  true maximum.
* Adam is used on the logits with a learning‑rate that decays slowly.
* Several sequence lengths are tried (up to ~8000).  The best sequence found
  over all lengths and random restarts is kept.
* After the gradient phase a cheap random‑perturbation hill‑climb is applied
  for the remaining time.

The function returns the best list of non‑negative heights found before the
deadline.  Only numpy and the supplied evaluate_sequence are used.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the next power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns an array of length len(x) + len(y) - 1.
    """
    size = len(x) + len(y) - 1
    N = _next_pow2(size)
    X = np.fft.rfft(np.pad(x, (0, N - len(x))))
    Y = np.fft.rfft(np.pad(y, (0, N - len(y))))
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()

# Main optimisation routine
# ----------------------------------------------------------------------
