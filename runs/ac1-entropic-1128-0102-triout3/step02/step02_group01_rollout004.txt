def subgradient(a):
    n = len(a)
    # compute convolution a*a
    conv = np.fft.irfft(np.fft.rfft(a, n=fft_len) ** 2, n=fft_len)[:2*n-1]
    max_val = conv.max()
    eps = max_val * eps_rel  # maybe 0.01 * max_val
    K = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1)
        j_idx = k - i_idx
        # g[i] += 2*a[j]
        g[i_idx] += 2.0 * a[j_idx]
    return g, max_val

"""
Propose a candidate step‑function using a multi‑stage projected Adam optimizer.

The algorithm works as follows:
1. Start from the provided `height_sequence_1` (if available) or a random
   simplex point.
2. Repeatedly up‑sample the current best shape (≈1.5× larger) up to a
   configurable maximum length.
3. For each length run a projected Adam gradient descent on the
   objective  `score = 2·n·max(conv(a,a))` while keeping `a` on the
   probability simplex (non‑negative, sum = 1).  The sub‑gradient of the
   `max`‑convolution is computed from all entries whose value is within a
   relative tolerance of the current maximum.
4. Keep the globally best sequence seen during the whole run.
5. Perform a final refinement on the best length with a tighter tolerance.

The implementation respects the supplied time budget (default 1000 s) and
prints progress messages with the current score.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries, sum = 1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """
    Fast evaluation of the score for a sequence that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _subgradient(a: np.ndarray, conv: np.ndarray, eps_rel: float):
    """
    Compute a sub‑gradient of the function max(conv(a,a)).
    Only entries of `conv` within `eps_rel` of the maximum are used.
    Returns the gradient vector and the current maximum value.
    """
    n = a.shape[0]
    max_val = conv.max()
    eps = max_val * eps_rel
    K = np.where(conv >= max_val - eps)[0]

    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g, max_val


def _adam_optimize(a0: np.ndarray,
                   max_iters: int,
                   eps_rel: float,
                   lr_initial: float,
                   deadline: float,
                   start_time: float,
                   stage_name: str = ""):
    """
    Projected Adam optimisation for a fixed length.
    Returns the best sequence found and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution via FFT
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] {stage_name} n={n}  iter={t}  score={cur_score:.6f}")

        # sub‑gradient of max‑convolution
        g, _ = _subgradient(a, conv, eps_rel)

        # Adam update (with diminishing effective step size)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr_initial * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_score

        # iteration budget inversely proportional to length (so total time stays bounded)
        base_iters = 200_000
        max_iters = int(base_iters * 1200 / next_n) + 10_000
        max_iters = max(max_iters, 30_000)   # never too few

        # learning rate slightly larger for smaller problems
        if next_n <= 1800:
            lr = 0.12
            eps_rel = 0.018
        elif next_n <= 3000:
            lr = 0.09
            eps_rel = 0.015
        else:
            lr = 0.07
            eps_rel = 0.012

        # --- run Adam optimisation -----------------------------------
        a_opt, stage_score = _adam_optimize(
            a0=a,
            max_iters=max_iters,
            eps_rel=eps_rel,
            lr_initial=lr,
            deadline=deadline,
            start_time=start_time,
            stage_name=f"Stage{stage_id}"
        )

        # keep global best
        if stage_score < best_score:
            best_score = stage_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] **New global best** "
                  f"after stage {stage_id}: n={next_n}  score={stage_score:.6f}")

        # prepare for the next up‑sampling step
        cur_seq = a_opt
        cur_n = next_n
        stage_id += 1

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3. Final refinement on the best length found.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        final_iters = int(250_000 * 1200 / n)
        final_iters = max(final_iters, 80_000)

        a_refined, refine_score = _adam_optimize(
            a0=best_seq,
            max_iters=final_iters,
            eps_rel=0.006,          # tighter tolerance
            lr_initial=0.04,
            deadline=deadline,
            start_time=start_time,
            stage_name="FinalRefine"
        )

        if refine_score < best_score:
            best_score = refine_score
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final refinement improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence as plain Python floats.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback (should never happen)
        best_seq = np.full(100, 1.0 / 100)

    return [float(v) for v in best_seq]
