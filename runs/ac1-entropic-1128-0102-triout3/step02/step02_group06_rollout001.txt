"""
Propose a step function by alternating linear‑programming directions
with an exact line‑search on the mixing factor.
We work on the simplex (Σ a_i = √(2 n)), so the evaluation coincides
with the maximum entry of the self‑convolution.  At each iteration we
solve the LP
    maximise   Σ g_i
    subject to g_i ≥ 0 , (f ∗ g)_k ≤ M   ∀k,
where M = max(f ∗ f).  The solution g pushes mass towards the
boundary of the feasible region.  The new candidate is
    h = (1‑t) f + t g,
with t chosen by a tiny ternary search that minimises
max(h ∗ h).  The best sequence seen before the time budget expires
is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _convolve_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returning the full length."""
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]

def _convolve_two_fft(a: np.ndarray, b: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*b using FFT."""
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    conv = np.fft.irfft(A * B, n=nfft)
    return conv[: 2 * a.size - 1]

def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) = f ∗ g.
    Each column i of A contains a copy of f shifted by i positions.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = lil_matrix((m, n), dtype=np.float64)
    # column i contributes f to rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f
    return A.tocsr()

def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP
        max Σ g_i
        s.t. g_i ≥ 0 , (f ∗ g)_k ≤ M   (k = 0 … 2n‑2)
    Returns the primal solution g (dense array) or None if the solver fails.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimise -Σg_i  ⟹ maximise Σg_i
    bounds = [(0, None)] * n
    # Using the high‑performance HiGHS interior‑point/simplex implementation
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

def _optimal_mixing_t(conv_f: np.ndarray, conv_fg: np.ndarray,
                      conv_g: np.ndarray) -> float:
    """
    Return the t∈[0,1] that minimises  max( (1‑t)²·conv_f
                                          + 2 t(1‑t)·conv_fg
                                          + t²·conv_g ).
    The function is convex in t, so a ternary search works.
    """
    lo, hi = 0.0, 1.0
    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        h1 = (1.0 - t1) ** 2 * conv_f + 2.0 * t1 * (1.0 - t1) * conv_fg + t1 ** 2 * conv_g
        h2 = (1.0 - t2) ** 2 * conv_f + 2.0 * t2 * (1.0 - t2) * conv_fg + t2 ** 2 * conv_g
        max1 = h1.max()
        max2 = h2.max()
        if max1 < max2:
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """

    ``evaluate_sequence``.  The method alternates a linear‑programming

    exact line‑search on the mixing factor.  The best sequence found
    before the time limit is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # initialise the sequence (use the previous construction if present)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():

    else:

        a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64)

    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    a = np.maximum(a, 0.0)  # just in case
    # ------------------------------------------------------------------
    # bookkeeping of the best solution
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial score (eval): {best_score:.6f}")

    nfft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    iter_cnt = 0
    no_improve = 0
    restart_limit = 3000          # iterations without progress before a restart

    while time.time() < deadline:
        iter_cnt += 1

        # current convolution and bound
        conv_f = _convolve_fft(a, nfft)
        M = conv_f.max()

        # solve LP to obtain a feasible direction g
        g0 = _solve_lp(a, M)
        if g0 is None:
            # LP failed – fall back to a tiny random jitter and continue
            jitter = np.random.randn(n) * 0.01 * a
            a = np.maximum(a + jitter, 0.0)
            a = a * (S / a.sum())
            continue

        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:

            continue
        g = g0 * (S / sum_g0)

        conv_g = _convolve_fft(g, nfft)

        h = (1.0 - t_opt) * a + t_opt * g

        # final evaluation of the candidate
        cur_score = evaluate_sequence(h.tolist())

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = h.copy()
            no_improve = 0
            print(f"[{iter_cnt:06d}] New best: {best_score:.6f} (t={t_opt:.4f})")
        else:
            no_improve += 1

        # early stop if we have beaten the target
        if best_score < 1.503:
            print(f"Target reached after {iter_cnt} iterations.")
            break

        # replace the current iterate with the new point (even if not strictly better)
        a = h

        # occasional random restart to escape possible stagnation
        if no_improve >= restart_limit:

            no_improve = 0

    best_a = np.maximum(best_a, 0.0)
    best_a = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iter_cnt} iterations. Best score = {final_score:.6f}")
    return best_a.tolist()

"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

# ----------------------------------------------------------------------
# FFT based convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returning the full (2·len(a)‑1) array."""
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _convolve_two_fft(a: np.ndarray, b: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*b using FFT, returning the full (2·len(a)‑1) array."""
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    conv = np.fft.irfft(A * B, n=nfft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Linear‑programming utilities
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Construct the (2n‑1)×n matrix A such that (A @ g) = f ∗ g,
    where ∗ denotes linear convolution.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = lil_matrix((m, n), dtype=np.float64)
    for i in range(n):
        # column i contributes f to rows i … i+n‑1
        A[i:i + n, i] = f
    return A.tocsr()


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise   Σ g_i
        subject to g_i ≥ 0  and  (f ∗ g)_k ≤ M   (k = 0 … 2n‑2).

    Returns the optimal g (as a dense NumPy array) or None if the solver fails.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -Σg_i  → maximise Σg_i
    bounds = [(0, None)] * n

    # HiGHS (default for method='highs') is fast for large sparse LPs.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Exact line‑search for the optimal mixing coefficient
# ----------------------------------------------------------------------


def _optimal_mixing_t(conv_f: np.ndarray, conv_fg: np.ndarray,
                      conv_g: np.ndarray) -> float:
    """
    For fixed sequences f and g compute the convex mixing parameter t∈[0,1]
    that minimises the maximal entry of the self‑convolution of
        h = (1‑t) f + t g .
    The objective is convex in t, therefore a ternary search suffices.
    """
    lo, hi = 0.0, 1.0
    for _ in range(30):                         # enough for double precision
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0

        h1 = (1.0 - t1) ** 2 * conv_f + 2.0 * t1 * (1.0 - t1) * conv_fg + t1 ** 2 * conv_g
        h2 = (1.0 - t2) ** 2 * conv_f + 2.0 * t2 * (1.0 - t2) * conv_fg + t2 ** 2 * conv_g

        max1 = h1.max()
        max2 = h2.max()
        if max1 < max2:
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low value of
    ``evaluate_sequence``.  The method alternates a linear‑programming
    direction (Frank–Wolfe step) with an exact line‑search on the mixing
    factor.  The best sequence seen before the time limit expires is
    returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # initialise the sequence (use the previous construction if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        n = a.size
    else:
        n = 2401
        a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64)

    S = np.sqrt(2.0 * n)                        # required ℓ₁‑norm
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())                       # project onto the simplex

    # ------------------------------------------------------------------
    # bookkeeping of the best solution
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial score (eval): {best_score:.6f}")

    nfft = _next_pow2(2 * n - 1)

    iter_cnt = 0
    no_improve = 0
    restart_limit = 3000                        # after this many non‑improvements we restart

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iter_cnt += 1

        # current self‑convolution and bound M
        conv_f = _convolve_fft(a, nfft)
        M = conv_f.max()

        # ---- linear‑programming direction --------------------------------
        g0 = _solve_lp(a, M)
        if g0 is None:
            # LP failed – inject a tiny random jitter and continue
            jitter = np.random.randn(n) * 0.01 * a
            a = np.maximum(a + jitter, 0.0)
            a = a * (S / a.sum())
            continue

        # rescale to the same ℓ₁‑norm as a
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            # numerically degenerate – skip this iteration
            continue
        g = g0 * (S / sum_g0)

        # ---- exact line‑search for the best mixing coefficient ----------
        conv_g = _convolve_fft(g, nfft)
        conv_fg = _convolve_two_fft(a, g, nfft)
        t_opt = _optimal_mixing_t(conv_f, conv_fg, conv_g)

        # convex combination (ℓ₁‑norm stays equal to S automatically)
        h = (1.0 - t_opt) * a + t_opt * g

        # evaluate the candidate (calls the protected function)
        cur_score = evaluate_sequence(h.tolist())

        # bookkeeping
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = h.copy()
            no_improve = 0
            print(f"[{iter_cnt:06d}] New best: {best_score:.6f} (t={t_opt:.4f})")
        else:
            no_improve += 1

        # early exit if the target is already reached
        if best_score < 1.503:
            print(f"Target reached after {iter_cnt} iterations.")
            break

        # move to the new point (even if not strictly better)
        a = h

        # occasional restart to avoid getting stuck in a flat region
        if no_improve >= restart_limit:
            print(f"[{iter_cnt:06d}] Restarting after {restart_limit} non‑improving steps.")
            a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64) * S
            no_improve = 0

    # ------------------------------------------------------------------
    # final sanity projection (should already satisfy the simplex)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    best_a = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iter_cnt} iterations. Best score = {final_score:.6f}")

    return best_a.tolist()
