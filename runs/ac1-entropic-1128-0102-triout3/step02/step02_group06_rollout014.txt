"""
LP‑based alternating optimisation for the max‑self‑convolution problem.

The algorithm repeatedly solves a linear programme that, for the current
vector `a`, finds a non‑negative vector `g` of maximal ℓ₁‑norm while
respecting the bound

    (a * g)[k] ≤ M   for all k,

where `M = max(a*a)`.  The LP solution can be scaled to the required
ℓ₁‑norm `S = sqrt(2·n)` and mixed with `a`.  Each mixing step is
guaranteed not to increase the maximum convolution entry, and the
process converges to a strictly better point unless `a` is already
optimal.  The procedure is cheap (a few seconds per LP for n≈1000) and
fits easily into the 1000 s budget.  Several random restarts are used
to escape local minima; the best sequence found is returned.

No closures or lambda expressions are used – all helpers are top‑level.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns the full (2·len(a)‑1) vector.
    """
    n = a.size
    nfft = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
    return conv


def _precompute_ub_indices(n: int):
    """
    Return row and column index arrays that describe the
    (2·n‑1)×n matrix A with entries

        A[k, i] = a[i]   if 0 ≤ k-i < n
        A[k, i] = 0       otherwise

    The row/col pattern does not depend on a, only on n.
    """
    # each column i appears in exactly n rows: i … i+n‑1
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    rows = np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)
    rows = rows.ravel()
    return rows, cols


def _solve_lp_for_g(a: np.ndarray, M: float, rows: np.ndarray,
                    cols: np.ndarray) -> np.ndarray:
    """
    Solve the LP:
        maximise    sum(g_i)
        subject to  (a * g)[k] ≤ M   for all k
                    g_i ≥ 0

    Returns the optimal g (as a dense numpy array).
    """
    n = a.size
    # data of the constraint matrix = a[col] (same a_i for every
    # occurrence of column i)
    data = a[cols]
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises c·x; use -1 to maximise sum(g)
    c = -np.ones(n, dtype=np.float64)

    # HiGHS interior‑point method is fast for this size
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=(0, None), method='highs', options={'presolve': True})
    if not res.success:
        # In rare pathological cases the LP can fail; return zeros
        return np.zeros_like(a)
    return res.x


# ----------------------------------------------------------------------
# main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step function that minimises `evaluate_sequence`.
    The method uses a sequence of LP‑based mixing steps starting from
    several random initial points.  The best sequence found before the
    time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size – we keep it modest to make the LP cheap.
    # If a large pre‑computed sequence exists, down‑sample it.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
        # target length (odd, ≈1000) – enough to approach the lower bound 1
        target_n = 1001
        if init_full.size > target_n:
            # simple uniform down‑sampling
            step = init_full.size // target_n
            a0 = init_full[::step][:target_n]
        else:
            a0 = init_full.copy()
    else:
        target_n = 1001
        a0 = np.random.dirichlet(np.full(target_n, 0.5)).astype(np.float64)

    n = a0.size
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm
    a0 = a0 * (S / a0.sum())      # enforce the normalisation

    # ------------------------------------------------------------------
    # pre‑compute the constant index pattern for the LP constraints
    # ------------------------------------------------------------------
    ub_rows, ub_cols = _precompute_ub_indices(n)

    # bookkeeping of the best solution ever seen
    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[INIT] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # parameters controlling the outer random‑restart loop
    # ------------------------------------------------------------------
    max_restarts = 30               # safe upper bound (time‑controlled)
    max_mix_iters = 30              # number of LP‑mixing steps per restart
    mix_step = 0.5                  # initial mixing coefficient
    min_mix_step = 0.02
    mix_decay = 0.92                # geometric decay of mix_step

    restarts_done = 0
    while time.time() < deadline and restarts_done < max_restarts:
        restarts_done += 1

        # start from a fresh random point – use a low‑α Dirichlet for
        # a spiky initial distribution (helps the LP make progress)
        a = np.random.dirichlet(np.full(n, 0.1)).astype(np.float64)
        a = a * (S / a.sum())

        current_best_score = best_score
        t = mix_step

        for it in range(max_mix_iters):
            # --------------------------------------------------------------
            # compute the current maximum entry of a*a
            # --------------------------------------------------------------
            conv = _conv_self_fft(a)
            M = conv.max()

            # --------------------------------------------------------------
            # solve the LP that tries to increase the ℓ₁‑norm while keeping
            # the convolution bounded by M
            # --------------------------------------------------------------
            g = _solve_lp_for_g(a, M, ub_rows, ub_cols)
            sum_g = g.sum()
            if sum_g <= 0.0:
                # no feasible direction – abort this restart
                break

            # --------------------------------------------------------------
            # scale g to have the required ℓ₁‑norm S
            # --------------------------------------------------------------
            g_scaled = g * (S / sum_g)

            # --------------------------------------------------------------
            # mix old and new vectors, renormalise (to guard against tiny drift)
            # --------------------------------------------------------------
            a_new = (1.0 - t) * a + t * g_scaled
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new * (S / a_new.sum())

            # --------------------------------------------------------------
            # evaluate the new candidate
            # --------------------------------------------------------------
            score = evaluate_sequence(a_new.tolist())
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                print(f"[{restarts_done:02d}/{it:02d}] New global best = {best_score:.6f}")

            # accept the new point for the next iteration (even if not improving)
            a = a_new

            # adjust mixing weight – smaller steps as we get closer to a local optimum
            t = max(t * mix_decay, min_mix_step)

            # early stop if progress stalls
            if abs(best_score - current_best_score) < 1e-12:
                break
            current_best_score = best_score

            # check time budget
            if time.time() >= deadline:
                break

        # give a quick report after each restart
        print(f"[RESTART {restarts_done}] best score so far = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity check and return
    # ------------------------------------------------------------------
    final_seq = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {restarts_done} restarts. Best score = {final_score:.6f}")
    return final_seq.tolist()
