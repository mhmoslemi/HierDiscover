"""
Hybrid optimisation for the self‑convolution L∞ problem.

The algorithm proceeds in four stages:

1. **Broad parametric sweep** –  generate candidates of the form
       a_i ∝ (i+offset)^(‑p)·exp(‑q·i)
   with three tunable parameters (p, offset, q).  The best candidate
   (lowest 2·n·max(conv)) is kept.

2. **Fine sweep** –  zoom into a small neighbourhood of the best
   parameters found in stage 1 and explore a denser grid.

3. **Smoothed projected gradient descent** –  minimise the
   log‑sum‑exp of the convolution values (a smooth approximation of the
   max).  The gradient is computed efficiently via FFT, the step is
   projected onto the probability simplex and a simple learning‑rate
   schedule is applied.  The current best vector is updated whenever a
   lower score is observed.

4. **Light simulated‑annealing polishing** –  starting from the best
   vector, repeatedly move a random fraction of mass from one coordinate
   to another and accept the move with a Metropolis rule.  This helps
   escape shallow local minima.

All heavy work uses FFT‑based convolutions (≈ O(n log n)) and the
routine respects a 1000 s wall‑time budget (default 950 s for safety).
The final sequence is verified with the target ``evaluate_sequence``
before being returned.
"""
import time

# Helper functions (module level – no closures)

def power_exp_sequence(n: int, p: float, offset: float, q: float) -> np.ndarray:
    """Generate a non‑negative sequence a_i = (i+offset)^{-p}·exp(-q·i)
    and normalise it to have ℓ₁‑norm 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) * np.exp(-q * i)
    total = a.sum()
    if total == 0.0:                      # should never happen in the search ranges
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full self‑convolution x*x using a real FFT.  Length = 2·len(x)‑1."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x*y using a real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    X = np.fft.rfft(x, n=n)
    Y = np.fft.rfft(y, n=n)
    conv = np.fft.irfft(X * Y, n=n)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w | w_i ≥ 0, Σ w_i = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Fast 2·n·max(self‑convolution) for a vector with ℓ₁‑norm 1."""
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * conv.max())


def weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """∇_a Σ_k weights_k·(a*a)_k  (the factor 2 is already included)."""
    # gradient_i = 2 * Σ_j a_j·weights_{i+j}
    w_rev = weights[::-1]
    conv = conv_fft_general(a, w_rev)               # length = 3·n‑2
    idx = 2 * a.shape[0] - 2 - np.arange(a.shape[0])
    return 2.0 * conv[idx]

                a = a_trial
                cur_score = trial_score

        # learning‑rate and temperature decay
        if (it + 1) % decay_every == 0:
            eta = max(eta * decay_factor, 1e-6)
            temperature = max(temperature * decay_factor, 1e-9)

        it += 1

    # ----------------- phase 4: light simulated annealing ------------
    print("[optim] Phase 4 – simulated‑annealing polishing")
    cur_a = best_a.copy()
    cur_score = best_score
    temp = 0.01
    temp_decay = 0.99996
    sa_iters = 0
    max_sa_iters = 300_000

    while time.time() < deadline and sa_iters < max_sa_iters:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random fraction (up to 10 %) of mass from i to j
        delta = cur_a[i] * rng.uniform(0.0, 0.10)
        if delta <= 0.0:
            continue
        a_trial = cur_a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta

        trial_score = compute_score(a_trial)

        if trial_score < cur_score or rng.random() < np.exp(-(trial_score - cur_score) / temp):
            cur_a = a_trial
            cur_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_a = a_trial.copy()
                print(f"[sa {sa_iters:06d}] new global best – score={best_score:.6f}")

        temp = max(temp * temp_decay, 1e-9)
        sa_iters += 1

    # ------------------- final sanity check -------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] best reported score = {final_score:.6f}")
    print(f"[info] final parameters – p={best_params['p']:.5f}, "
          f"off={best_params['off']:.5f}, q={best_params['q']:.5f}")

    # fallback to a known good construction if something went wrong
    if not np.isfinite(final_score):
        if 'height_sequence_1' in globals():
            candidate = list(height_sequence_1)
            cand_score = evaluate_sequence(candidate)
            if cand_score < final_score:
                print("[fallback] using provided height_sequence_1")
                return candidate
    return best_a.tolist()
