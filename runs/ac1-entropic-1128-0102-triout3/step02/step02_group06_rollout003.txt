def local_search(a, S, deadline, current_best):
    n = len(a)
    nfft = _next_pow2(2*n - 1)
    best = current_best
    best_a = a.copy()
    eps_factor = 0.02 # fraction of S, initially

    while time.time() < deadline:
        # compute conv
        conv = _convolve_self_fft(a, nfft)
        M = conv.max()
        if M < best - 1e-12:
            best = M
            best_a = a.copy()
            print(f"Local search new best: {best:.6f}")

        # find k_max
        k_max = int(conv.argmax())
        # support indices
        i_low = max(0, k_max - (n-1))
        i_high = min(n-1, k_max)
        i_range = np.arange(i_low, i_high+1)
        # compute product contributions
        products = a[i_range] * a[k_max - i_range]
        # choose i that maximizes product and has enough value to move
        # we also need to avoid moving from zero.
        # we can sort descending; pick first with a_i > 0
        sorted_idx = np.argsort(-products)  # descending
        moved = False
        for idx in sorted_idx:
            i = i_range[idx]
            if a[i] < 1e-12:
                continue
            # pick j not in support ideally
            # we could pick random index j where a[j] is small
            # to reduce product contributions elsewhere.
            # Choose j random among indices with a[j] < median perhaps.
            # Ensure j != i.
            # try multiple attempts
            for _ in range(5):
                j = np.random.randint(0, n)
                if j == i:
                    continue
                # ensure moving doesn't increase max drastically? We'll just evaluate.
                # Determine epsilon: some fraction of a[i] but not too large
                eps = min(a[i] * 0.5, S * eps_factor)
                if eps < 1e-12:
                    continue
                # propose new a
                a_new = a.copy()
                a_new[i] -= eps
                a_new[j] += eps
                # maintain non-negative, sum unchanged
                # compute new max via conv
                conv_new = _convolve_self_fft(a_new, nfft)
                M_new = conv_new.max()
                if M_new < M - 1e-8:  # improvement
                    a = a_new
                    moved = True
                    break
            if moved:
                break
        if not moved:
            # no improvement possible: reduce eps_factor
            eps_factor *= 0.5
            if eps_factor < 1e-6:
                break
            # optionally random jitter
            # apply small random jitter for exploration
            jitter = np.random.randn(n) * (S * eps_factor * 0.1)
            a = np.maximum(a + jitter, 0)
            a = a * (S / a.sum())
    return best_a, best

"""
Hybrid Adam + focused local search for minimizing the maximum entry of a
non‑negative step function’s self‑convolution.

The algorithm proceeds in two phases:

1. **Adam on the simplex** – we optimise in the log‑space of the
   coefficients (ensuring positivity) and re‑normalise after each step
   to satisfy Σ a_i = √(2·n).  A soft‑max smoothing of the max‑convolution
   is used early on; once the temperature is low we switch to the exact
   sub‑gradient of the max‑entry.  Adam’s adaptive steps, occasional
   jitter and a simple restart policy keep the search from stagnating.

2. **Targeted local improvement** – starting from the best sequence found
   by Adam we repeatedly move a small amount of mass from the coefficient
   that contributes most to the current maximal convolution entry to a
   random location.  The move is accepted only if it strictly lowers the
   maximal entry; otherwise we shrink the step size and eventually inject
   a tiny random jitter.  This phase quickly squeezes out the remaining
   slack and often pushes the score below the target of 1.503.

The whole routine respects the 1000 s time budget, prints progress and
returns the best sequence (as a plain list of floats) found before the
deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using the FFT.
    Returns the first (2·len(a)‑1) real entries.
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    Uses a numerically stable soft‑max.
    """
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:                # safeguard
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the index k_max.
    The derivative of (a*a)[k] w.r.t. a_i equals 2·a_{k‑i}
    whenever the index lies inside the support.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad

# main search routine
# ----------------------------------------------------------------------

                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step function that minimises `evaluate_sequence`.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5        # tiny safety margin

    # --------------------------------------------------------------
    # initialise sequence length and ℓ₁‑norm (the standard normalisation)
    # --------------------------------------------------------------

        init_seq = np.asarray(height_sequence_1, dtype=np.float64)

    else:
        n = 2401                                   # default size
        init_seq = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64)

    # normalise to the required ℓ₁‑norm Σ a = √(2·n)
    S = np.sqrt(2.0 * n)
    init_seq = np.maximum(init_seq, 0.0)
    init_seq = init_seq * (S / init_seq.sum())

    # log‑variables for mirror descent on the simplex
    eps = 1e-12
    log_theta = np.log(init_seq + eps)

    # bookkeeping of the best solution
    best_a = init_seq.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[Init] Score = {best_score:.6f}")

    # pre‑compute FFT length (power of two for speed)
    nfft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters (operating on log‑variables)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta = 0.15                     # base learning rate in log‑space

    m = np.zeros_like(log_theta)
    v = np.zeros_like(log_theta)

    # soft‑max temperature schedule (annealed)
    tau = 0.5
    tau_min = 0.01
    # we do not know the exact number of iterations; we decay slowly
    tau_decay = 0.9999

    iteration = 0
    # --------------------------------------------------------------
    # ADAM phase – run until we have ~20 s left for the local phase
    # --------------------------------------------------------------
    while time.time() < deadline - 20.0:
        iteration += 1

        # ---- reconstruct a and enforce Σa = S ----
        a = np.exp(log_theta)
        sum_a = a.sum()
        if sum_a == 0.0:                     # numerical safety net
            a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64) * S
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            continue

        # ---- evaluate current max‑convolution ----
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()                 # = evaluate_sequence after normalisation

        # ---- keep the best seen so far ----
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{iteration:08d}] New best (Adam) = {best_score:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:
            print(f"Target reached (score={best_score:.6f}) at iteration {iteration}")
            break

        # ---- gradient of the (smoothed) max‑convolution ----
        if tau > tau_min * 2:
            grad_a = _grad_softmax(a, conv, tau, nfft)
        else:
            k_max = int(conv.argmax())
            grad_a = _grad_max(a, conv, k_max)

        # chain rule to get gradient w.r.t. log‑variables
        grad_log = grad_a * a

        # ---- Adam update (descent) ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step                     # descent step

        # occasional random jitter to shake off plateaus
        if np.random.rand() < 0.005:
            jitter = np.random.randn(n) * (eta * 0.05)
            log_theta = log_theta + jitter

        # temperature annealing
        tau = max(tau * tau_decay, tau_min)

    print(f"[Adam] Finished after {iteration} iterations. Best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # LOCAL SEARCH phase – fine‑tune the best sequence
    # --------------------------------------------------------------
    a = best_a.copy()
    a = a * (S / a.sum())          # re‑normalise just in case
    eps_factor = 0.02              # initial relative step size
    stagnation = 0

    while time.time() < deadline:
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            stagnation = 0
            print(f"[Local] New best = {best_score:.6f}")
        else:
            stagnation += 1

        # stop local search if we have been unable to improve for a while
        if stagnation > 5000:
            eps_factor *= 0.5
            stagnation = 0
            if eps_factor < 1e-6:
                break

        # --- identify the most “dangerous” coefficient pair ---
        k_max = int(conv.argmax())
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idxs = np.arange(i_low, i_high + 1)
        products = a[idxs] * a[k_max - idxs]

        # try to move mass away from the index that contributes most
        order = np.argsort(-products)          # descending
        moved = False
        for idx in order:
            i = idxs[idx]
            if a[i] <= 1e-12:
                continue
            # propose a target index j (preferably outside the current support)
            for _ in range(5):
                j = np.random.randint(0, n)
                if j == i:
                    continue
                eps_move = min(a[i] * 0.5, S * eps_factor)
                if eps_move <= 1e-12:
                    continue
                a_candidate = a.copy()
                a_candidate[i] -= eps_move
                a_candidate[j] += eps_move
                conv_cand = _convolve_self_fft(a_candidate, nfft)
                max_cand = conv_cand.max()
                if max_cand < cur_max - 1e-9:
                    a = a_candidate
                    moved = True
                    break
            if moved:
                break

        if not moved:
            # no improving move found – inject a tiny random jitter
            jitter = np.random.randn(n) * (S * eps_factor * 0.1)
            a = np.maximum(a + jitter, 0.0)
            a = a * (S / a.sum())

    print(f"[Local] Done. Best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # final sanity projection and verification
    # --------------------------------------------------------------
    final_seq = best_a * (np.sqrt(2.0 * len(best_a)) / best_a.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Final sequence length {len(final_seq)} – score = {final_score:.6f}")

    return final_seq.tolist()
