"""
Improved search for a non‑negative step function minimizing the
`evaluate_sequence` bound.

* Works on the probability simplex (sum = 1) – the objective is scale‑invariant.
* Uses a smoothed ℓ∞ objective via a soft‑max (temperature α) which gives a
  well‑behaved gradient of the maximal convolution entry.
* Gradient is computed by FFT‑based cross‑correlation (O(n log n) per step).
* Momentum, diminishing step size and a final short simulated‑annealing
  refinement are employed.
* Multi‑scale: optimise a small length, up‑sample the best distribution and
  continue with larger lengths.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution a * a  (full, not circular) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Cross‑correlation needed for the smooth gradient:
# h_i = Σ_j a_j * p_{j+i}
# (p has length 2n‑1)
# ----------------------------------------------------------------------
def _cross_corr(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = p.shape[0]               # must be 2n‑1
    L = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    P = np.fft.rfft(p[::-1], n=L)   # reverse p before FFT
    conv = np.fft.irfft(A * P, n=L)[: n + m - 1]   # length 3n‑2
    # slice that gives h_i = conv[(m‑1)‑i] for i=0…n‑1
    start = m - 1
    h = conv[start - np.arange(n)]
    return h


# ----------------------------------------------------------------------
# A single optimisation run for a fixed length n.
# Uses a soft‑max approximation of the ℓ∞ norm.
# ----------------------------------------------------------------------
def _run_smooth_gradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alpha_start: float = 0.5,
    alpha_end: float = 8.0,
    alpha_ramp: int = 800,          # iterations over which α is increased
    beta: float = 0.9,
    step0: float = 0.6,
    max_iter: int = 50000,
):
    """Return best sequence (and its score) for length n."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- convolution and current score -----
        conv = _conv_self(a)                     # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"smooth‑grad improve: n={n}, iter={it}, score={cur_score:.6f}"
            )

        # ----- soft‑max weights for the active entries -----
        # α grows linearly during the first `alpha_ramp` iterations.
        if it < alpha_ramp:
            alpha = alpha_start + (alpha_end - alpha_start) * it / alpha_ramp
        else:
            alpha = alpha_end
        # Shift by max for numerical stability: all entries ≤ 0.
        exp_vals = np.exp(alpha * (conv - max_conv))
        p = exp_vals / exp_vals.sum()            # length 2n‑1, sums to 1

        # ----- gradient of the smoothed objective -----
        # g_i = Σ_k p_k * a_{k-i}  + a_i   (since Σ p_k = 1)
        h = _cross_corr(p, a)                    # shape (n,)
        grad = h + a

        # ----- momentum + step -----
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        # ----- simplex projection -----
        a = _project_simplex(a - step * momentum)

        # occasional tiny random jitter to escape plateaus
        if it % 500 == 0 and time.time() < deadline:
            eps = 1e-4 * np.random.randn(n)
            a = _project_simplex(a + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (identical to the previous approach)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 20000,
):
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"annealing improve: n={n}, iter={it}, score={score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Up‑sample an existing distribution to a larger length by linear interpolation.
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Deterministic “arcsine‑like’’ start – puts more mass near the ends.
# ----------------------------------------------------------------------
def _initial_arcsine(n: int) -> np.ndarray:
    i = np.arange(n) + 0.5
    w = 1.0 / np.sqrt(i * (n - i))
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds,
    combining a smooth‑gradient phase, up‑sampling across scales and a final
    annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    best_score = float("inf")
    best_seq = None

    # if a previous construction is present, use it; otherwise start from arcsine
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # geometric length schedule – the larger the length the lower the theoretical bound
    lengths = [800, 1600, 3200, 6400, 9600]
    np.random.shuffle(lengths)

    # keep the current best (always on the simplex)
    global_a = starter if starter is not None else None

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- initialise -----
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = _initial_arcsine(n)            # arcsine‑like start
        # small random perturbation to avoid exact symmetry pitfalls
        a0 += 1e-6 * np.random.rand(n)
        a0 = _project_simplex(a0)

        # allocate a time slice for this length
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        slice_deadline = time.time() + max(8.0, 0.15 * remaining)

        # ----- smooth gradient descent -----
        a_opt, score_opt = _run_smooth_gradient(
            a0,
            n,
            slice_deadline,
            start_time,
            alpha_start=0.5,
            alpha_end=10.0,
            alpha_ramp=800,
            beta=0.9,
            step0=0.6,
            max_iter=60000,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"global improve after smooth‑grad: n={n}, score={score_opt:.6f}"
            )

        # ----- optional annealing refinement -----
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, 0.05 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=25000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] "
                    f"global improve after annealing: n={n}, score={score_ref:.6f}"
                )

        # remember the best candidate for the next scale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # convert to plain Python list for the evaluator
    return [float(x) for x in best_seq]
