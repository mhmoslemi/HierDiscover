"""
Propose a candidate step‑function sequence.

The algorithm works in three stages:

1. **Soft‑max gradient descent** – a fast Adam optimiser in the log‑space
   (logits → soft‑max → probabilities).  The objective 2·n·max(conv) is
   approximated by a sum over the *k* largest convolution entries.
   This quickly finds a decent low‑autocorrelation shape.

2. **Linear‑programming refinement** – given the current shape `f`,
   we solve a linear programme  
        maximise   Σ g_i
        subject to conv(f,g) ≤ M   (element‑wise) ,   g_i ≥ 0,
   where `M` is the current maximum convolution value.
   The solution `g` can have a larger ℓ₁‑norm; after normalising
   it to sum 1 we either replace `f` or take a small convex combination.
   Repeating this step monotonically drives the bound down.

3. **Best‑track** – any improvement (either from the gradient phase or from
   the LP phase) is recorded.  The routine runs for at most ``budget_s``
   seconds and finally returns the best sequence found.
"""

import time
from typing import List

import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (full) using FFT.
    Returns a vector of length 2·n‑1.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A                     # |A|²
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv_full


def _score_from_a(a: np.ndarray) -> float:
    """
    Compute the official evaluation score for a probability vector ``a``.
    The vector is assumed non‑negative and its sum is ignored (the score
    is homogeneous; we keep the sum for safety).
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _adam_update(z: np.ndarray,
                 grad_z: np.ndarray,
                 m: np.ndarray,
                 v: np.ndarray,
                 t: int,
                 lr: float,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, tuple[np.ndarray, np.ndarray]]:
    """Classic Adam update."""
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, (m, v)


def _gradient_wrt_a(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (soft) maximum of the convolution.
    For each selected offset ``k`` we add ∂(a*a)_k / ∂a_i = a_{k-i}+a_{i-k},
    which reduces to 2·a_{k-i} for admissible ``i``.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    for k in top_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start … i_end (inclusive)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]

    return grad


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix A such that A @ g = conv(f,g) (full).
    Row ``k`` corresponds to the linear combination of ``g`` that produces
    the k‑th entry of the full convolution.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The function stops after ``budget_s`` seconds and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq: List[float] | None = None
    best_score = float("inf")

    def _maybe_update(seq: np.ndarray) -> None:
        """Update the global best if the new sequence is better."""
        nonlocal best_seq, best_score
        try:
            sc = evaluate_sequence(seq.tolist())
        except Exception:                     # fallback to our own scorer
            sc = _score_from_a(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.tolist()
            print(f"[global] New best score {best_score:.6f} (len {len(seq)})")

    # ------------------------------------------------------------------
    # 0) Warm‑start: use a previously discovered sequence if available.
    # ------------------------------------------------------------------
    init_seq = None
    if "height_sequence_1" in globals():
        try:
            cand = np.array(list(height_sequence_1), dtype=float)
            cand = np.maximum(0.0, cand)      # protect against negatives
            s = cand.sum()
            if s < 1e-12:
                cand = np.full_like(cand, 1.0 / len(cand))
            else:
                cand = cand / s
            init_seq = cand
            _maybe_update(cand)
        except Exception:
            init_seq = None

    # ------------------------------------------------------------------
    # Lengths to try – modest sizes keep the LP tractable.
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000]

    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # --------------------------------------------------------------
        # 1) Initialise logits (soft‑max parameters)
        # --------------------------------------------------------------
        if init_seq is not None and init_seq.shape[0] == n:
            a = init_seq.copy()
            eps = 1e-12
            z = np.log(a + eps)                # invert soft‑max
        else:
            # random logits – small Gaussian noise around zero
            z = np.random.randn(n) * 0.1
            a = _logits_to_a(z)

        # Adam state for the log‑space optimiser
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t_iter = 0

        # --------------------------------------------------------------
        # 2) Gradient‑descent phase (soft‑max + Adam)
        # --------------------------------------------------------------
        top_k = max(1, n // 500)                 # a few largest offsets
        top_k_decay_iter = 20000                 # after this we focus on the true max

        while time.time() < deadline and t_iter < 50000:
            conv = _conv_fft(a)
            cur_score = float(2.0 * n * conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[{n}] Grad new best {best_score:.6f}")

            # choose the offsets that dominate the max
            if top_k == 1:
                ks = np.array([int(np.argmax(conv))], dtype=int)
            else:
                ks = np.argpartition(conv, -top_k)[-top_k:]

            # sub‑gradient w.r.t. the probabilities
            grad_a = _gradient_wrt_a(a, ks)

            # chain rule through the soft‑max (logits → probabilities)
            dot = a.dot(grad_a)
            grad_z = a * (grad_a - dot)

            # Adam update
            t_iter += 1
            lr = 0.05 / np.sqrt(n) / (1.0 + t_iter * 1e-5)
            z, (m, v) = _adam_update(z, grad_z, m, v, t_iter, lr)

            # re‑project to the simplex
            a = _logits_to_a(z)

            # occasional jitter to escape plateaus
            if t_iter % 2000 == 0:
                z += np.random.randn(n) * 1e-3

            # tighten the smoothing schedule
            if t_iter == top_k_decay_iter:
                top_k = 1

        # ensure the vector sums to one (numerical drift)
        a = a / a.sum()

        # --------------------------------------------------------------
        # 3) Linear‑programming refinement
        # --------------------------------------------------------------
        while time.time() < deadline:
            conv = _conv_fft(a)
            M = conv.max()
            cur_score = float(2.0 * n * M)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[{n}] LP new best {best_score:.6f}")

            # build the convolution matrix for the current ``a``
            A_mat = _build_conv_matrix(a)

            # LP: maximise Σ g_i  subject to A_mat @ g ≤ M, g ≥ 0
            c = -np.ones(n, dtype=np.float64)               # maximise sum(g)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            bounds = [(0.0, None)] * n

            res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})

            if not res.success:
                print(f"[{n}] LP failed: {res.message}")
                break

            g = res.x
            sum_g = g.sum()
            if sum_g <= 0:
                break

            # Normalise to a probability vector
            g_norm = g / sum_g

            # ----- try pure replacement (if it already improves) -----
            cand_score = _score_from_a(g_norm)
            if cand_score < cur_score:
                a = g_norm
                continue

            # ----- otherwise take a small convex combination -----
            alpha = 0.1                                   # step size
            a_new = (1.0 - alpha) * a + alpha * g_norm
            a_new = a_new / a_new.sum()                   # renormalise

            new_score = _score_from_a(a_new)
            if new_score < cur_score:
                a = a_new
                # continue the LP loop with the improved vector
                continue
            else:
                # no improvement any more → exit the LP loop for this length
                break

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"Final best score (official): {final_score:.6f} "
              f"(length {len(best_seq)})")
    except Exception:
        print("Warning: final evaluation failed.")

    return best_seq
