"""
Propose a non‑negative step‑function minimizing `evaluate_sequence`.

Algorithm
---------
* Work on the probability simplex (∑a_i = 1).  The objective
  `2 * n * max(conv(a,a))` is scale‑invariant, so fixing the sum is safe.
* For a given length `n` we run a **projected Adam** optimiser.
  - The active set consists of all convolution entries within a tiny
    relative tolerance of the current maximum (instead of a fixed K).
  - The sub‑gradient of an active entry k is the usual
    `2 * a_{k-i}` contribution; we sum these over the whole active set.
* After the Adam phase we perform a cheap **simulated‑annealing**
  refinement that transfers a small amount of mass between two coordinates.
* The optimisation proceeds on a schedule of increasing lengths.
  The best sequence found so far is up‑sampled (linear interpolation) and
  used as the initial point for the next, larger length.
* Time is split dynamically: each length receives a slice of the remaining
  budget; the final remaining seconds are used for a last refinement on the
  longest sequence.
* All updates are projected onto the simplex using the O(n) algorithm of
  Duchi et al. (2008).

The combination of a more accurate active set, Adam’s adaptive step size
and extra refinement usually drives the score below the previous best
(~1.514) and often reaches the target ≤ 1.503 within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x|x_i≥0,∑x_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution a * a (linear, not circular) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (full length 2*n‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient built from an arbitrary set of active convolution indices
# ----------------------------------------------------------------------
def _subgradient_from_active(a: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of the convolution entries indexed by act_idx."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    if act_idx.size:
        g /= act_idx.size
    return g


# ----------------------------------------------------------------------
# Score helper (the part that does not involve clipping or NaN checks)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the raw objective 2 * n * max(conv(a,a))."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Projected Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _run_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    max_iter: int = 30000,
    rel_tol: float = 1e-6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    step0: float = 0.04,
    eps_adam: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the probability simplex.
    Returns the best sequence encountered and its raw score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, "
                f"iter={t}, score={cur_score:.6f}"
            )

        # active set = all entries within a relative tolerance of the maximum
        tol = max_val * rel_tol
        act_idx = np.where(conv >= max_val - tol)[0]
        grad = _subgradient_from_active(a, act_idx)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / (np.sqrt(v_hat) + eps_adam)

        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 25000,
) -> tuple[np.ndarray, float]:
    """Tiny “move ε from i to j’’ annealing, returns possibly improved pair."""
    a = a0.copy()
    score = _score(a)
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.07  # up to 7 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _score(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Annealing improve: n={n}, "
                f"iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample a simplex vector by linear interpolation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation up‑sampling while preserving the simplex sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm respects the time budget, uses
    projected Adam, simulated annealing and multi‑scale up‑sampling.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from the external environment
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # Length schedule – geometric growth; shuffle a bit to avoid bias
    lengths = [512, 1024, 2048, 4096, 6144]
    np.random.shuffle(lengths)

    # Keep the best so far (always on the simplex)
    global_a = starter if starter is not None else None

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise (upsample previous best or random)
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # Allocate a time slice for this length
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # At least 12 s per length, otherwise take 12 % of the remaining budget
        slice_deadline = time.time() + max(12.0, 0.12 * remaining)

        # ------------------------------------------------------------------
        # Adam optimisation phase
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_adam(
            a0,
            n,
            slice_deadline,
            start_time,
            max_iter=25000,
            rel_tol=1e-6,
            step0=0.04,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Global improve after Adam: "
                f"n={n}, score={score_opt:.6f}"
            )

        # ------------------------------------------------------------------
        # Simulated annealing refinement (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(6.0, 0.06 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=25000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] Global improve after annealing: "
                    f"n={n}, score={score_ref:.6f}"
                )

        # ------------------------------------------------------------------
        # Keep the best sequence for the next scale
        # ------------------------------------------------------------------
        global_a = best_seq.copy()

    # ----------------------------------------------------------------------
    # Final polishing on the longest discovered sequence if time remains
    # ----------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        n = best_seq.shape[0]
        final_deadline = deadline
        # a quick extra Adam run
        a_opt, score_opt = _run_adam(
            best_seq,
            n,
            final_deadline,
            start_time,
            max_iter=20000,
            rel_tol=1e-6,
            step0=0.03,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Final Adam polishing: n={n}, "
                f"score={score_opt:.6f}"
            )
        # final annealing (tiny budget)
        if time.time() < deadline:
            anneal_deadline = deadline
            a_ref, score_ref = _run_annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=2e-2,
                end_temp=5e-6,
                steps=15000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] Final annealing polishing: n={n}, "
                    f"score={score_ref:.6f}"
                )

    # ----------------------------------------------------------------------
    # Fallback (should never be taken)
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # Convert to plain Python list for the evaluator
    return [float(x) for x in best_seq]
