"""
Soft‑max Projected Gradient Descent (with Adam) for minimizing the
max‑convolution of a non‑negative step function.
The algorithm works on the ℓ₁‑simplex {x ≥ 0, Σx = √(2n)} so that the
objective 2·n·max(conv)/(Σx)² reduces to the maximal entry of the
self‑convolution.  At each iteration we

* compute the exact self‑convolution by FFT,
* build a smooth approximation of the max via a soft‑max (log‑sum‑exp)
  with a temperature α,
* obtain the gradient of this smooth objective (a linear combination of
  the convolution kernels) also by FFT,
* take an Adam step and project back onto the simplex,
* occasionally perform a random mass‑transfer move or a full restart to
  escape plateaus,
* keep the best sequence seen so far (evaluated with the exact
  evaluate_sequence).

The temperature α is increased slowly so that the optimisation
progressively focuses on the true maximum.  The routine respects the
budget of 1000 s and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full (2·len(a)‑1) result (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_softmax(a: np.ndarray, w: np.ndarray, nfft: int) -> np.ndarray:
    """
    Gradient of the soft‑max (log‑sum‑exp) approximation of max(conv)
    with respect to a.  w must be a probability vector of length
    2·len(a)‑1 (the soft‑max weights).  Returns a gradient of length
    len(a).
    """
    # a_rev is a reversed copy of a
    a_rev = a[::-1]
    # FFTs (real)
    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    # convolution w * a_rev (linear part) – truncated to length 2·n‑1
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]
    n = a.size
    # the slice that corresponds to the gradient entries
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Soft‑max Adam optimiser for the max‑convolution problem.
    Returns the best non‑negative step‑function (list of floats) found
    before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size and simplex target
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=np.float64)
        n = a0.shape[0]
    else:
        # a moderate size that still allows many degrees of freedom
        n = 5000
        a0 = None

    S = np.sqrt(2.0 * n)                     # ℓ₁‑norm we keep fixed
    nfft = _next_pow2(2 * n - 1)             # FFT length sufficient for
                                             # self‑convolution

    # ------------------------------------------------------------------
    # initialise a
    # ------------------------------------------------------------------
    if a0 is not None:
        # keep the same support size but re‑project to the correct ℓ₁‑norm
        a = _project_to_simplex(a0, S)
    else:
        # start from a smooth sinusoidal shape (positive everywhere)
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 0.5) / n)          # strictly positive
        a = a / a.sum() * S
        a = _project_to_simplex(a, S)

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    base_lr = 0.25 * S / n                  # base learning‑rate scale
    beta1, beta2 = 0.9, 0.999               # Adam moments
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 5.0                             # soft‑max temperature
    alpha_max = 30.0
    alpha_update_every = 5000               # increase α every this many iters

    p_random_move = 0.03                    # probability of a mass‑transfer step
    rand_transfer = 0.07                    # fraction of source mass to move

    no_improve = 0
    max_no_improve = 20000                  # restart if stuck too long

    iteration = 0
    last_best_iter = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1) self‑convolution and exact max
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = cur_max                     # because Σa = √(2n)

        # keep global best (exact evaluation)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_best_iter = iteration
            no_improve = 0
            print(f"[{iteration:07d}] New best: {best_score:.6f}")
        else:
            no_improve += 1

        # early exit if target reached
        if best_score < 1.503:
            print(f"Target reached after {iteration} iterations.")
            break

        # 2) stochastic mass‑transfer move (occasionally)
        if np.random.rand() < p_random_move:
            # pick a source proportional to its mass
            src_probs = a / a.sum()
            i = np.random.choice(n, p=src_probs)
            # pick a destination uniformly (different from source)
            j = i
            while j == i:
                j = np.random.randint(n)
            delta = rand_transfer * a[i]
            if delta <= 0:
                delta = rand_transfer * S / n
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = _project_to_simplex(a_candidate, S)
        else:
            # 3) gradient step on the smooth approximation of max
            # --- soft‑max weights ---
            max_c = cur_max
            # scale to avoid overflow
            exp_arg = alpha * (conv - max_c)
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum == 0:                     # numerical safeguard
                w = np.ones_like(conv) / conv.size
            else:
                w /= w_sum                       # now a probability vector

            # --- gradient ---
            grad = _grad_softmax(a, w, nfft)

            # --- Adam update ---
            iteration += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
            a_candidate = a - step
            a_candidate = _project_to_simplex(a_candidate, S)

        # evaluate candidate (only the smooth objective needed for acceptance)
        # compute its exact max for a Metropolis‑style acceptance test
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()

        # acceptance rule: greedy + occasional Metropolis (temperature ≈ 0)
        # we use a tiny temperature to allow occasional uphill moves
        accept = cand_max < cur_score or np.random.rand() < 0.001
        if accept:
            a = a_candidate
        # else keep current a unchanged

        # temperature‑like schedule for the soft‑max temperature α
        if iteration % alpha_update_every == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.15, alpha_max)

        # stagnation handling
        if no_improve >= max_no_improve:
            # full restart from a Dirichlet draw (more exploration)
            a = np.random.dirichlet(np.full(n, 0.4)) * S
            m[:] = 0.0
            v[:] = 0.0
            no_improve = 0
            print(f"[{iteration:07d}] Restart due to stagnation.")
            continue

    # ------------------------------------------------------------------
    # final verification and return
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)    # ensure feasibility
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()
