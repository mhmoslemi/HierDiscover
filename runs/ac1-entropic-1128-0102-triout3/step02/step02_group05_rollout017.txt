"""
Propose a step‑function minimizing the “max‑self‑convolution’’ objective.

The algorithm combines three ideas:

1. **Smooth max (soft‑max) surrogate** – instead of using the
   non‑differentiable max we minimise
        Φτ(a)= (1/τ)·log Σ_k exp(τ·(a∗a)_k) .
   For τ→∞ the surrogate tends to the true max, while for small τ the
   landscape is smoother and easier to optimise.

2. **Projected Adam on the probability simplex** – the gradient of Φτ is a
   weighted sum of all convolution shifts.  Adam’s adaptive step‑sizes,
   together with a Euclidean projection onto the simplex (non‑negative,
   sum‑to‑one), give fast, stable convergence.

3. **Exploration & structure enforcement** – occasional random
   mass‑transfer moves (hill‑climbing) allow the method to escape plateaus,
   and after each update the vector is sorted decreasingly.
   For a fixed multiset of values this ordering minimises the
   centre convolution entry, thus never hurts the objective.

The routine runs until the time budget (default 1000 s) expires and returns
the best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Fast evaluation – same formula as `evaluate_sequence` but without the
# security checks.  Used internally for speed.
def fast_score(a: np.ndarray) -> float:
    n = a.size
    s = a.sum()
    if s < 1e-12:                     # should never happen (simplex)
        return np.inf
    # convolution length = 2*n‑1, pad to next power of two
    size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(a, size)
    conv = np.fft.irfft(ft * ft, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def build_initial_sequence(n: int,
                           p: float = 0.5,
                           offset: float = 0.0) -> np.ndarray:
    """Power‑law decreasing initialisation, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

        # gradient_i = 2· Σ_k w_k·a_{k‑i}
        # this is 2·(a cross‑correlated with w).  Use FFT for speed.
        w_rev = w[::-1]                         # reverse for cross‑correlation
        ft_w = np.fft.rfft(w_rev, size_fft)
        grad_full = np.fft.irfft(ft_a * ft_w, size_fft)
        start = (2 * n - 2)                     # L‑1 where L = 2n‑1
        grad = 2.0 * grad_full[start:start + n]

        # ----- 4.4 Adam update + projection onto the simplex ------------
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # enforce non‑negativity and the simplex constraint
        a = proj_simplex(np.maximum(a_new, 0.0))

        # keep decreasing order – this is never detrimental for the max‑conv.
        a = np.sort(a)[::-1]

        # ----- 4.5 occasional random mass‑transfer move (exploration) ----
        if np.random.rand() < move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = proj_simplex(cand)
                cand = np.sort(cand)[::-1]
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_seq = cand_val, cand.copy()
                    a = cand
                else:
                    # tiny uphill acceptance to maintain diversity
                    if np.random.rand() < 0.001:
                        a = cand

        # ----- 4.6 temperature increase (focus more on the true max) -----
        tau = min(tau_max, tau + tau_inc)

        # ----- 4.7 periodic evaluation of the true objective ------------
        if it % eval_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
                print(f"[iter {it:7d}] new best {best_val:.6f}")
            else:
                print(f"[iter {it:7d}] cur {cur_val:.6f} best {best_val:.6f}")

        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    # convert back to plain list; evaluate_sequence will perform final checks
    return best_seq.tolist()
