"""
Propose a step‑function sequence that minimizes the evaluation function.

The algorithm is a projected (scaled‑simplex) gradient descent on a
smooth approximation of the L∞‑norm of the self‑convolution.
Instead of using the sub‑gradient of only the maximal entry,
we minimise the p‑norm  (∑ (a*a)^p)^(1/p) for a growing p.
For p → ∞ this p‑norm converges to the L∞‑norm, i.e. the quantity
appearing in ``evaluate_sequence``.  The gradient of the p‑norm can be
computed efficiently with two FFT‑based convolutions, and after each
step we project back onto the set {x_i≥0, ∑x_i = √(2 n)} (the same
normalisation used in the reference implementation).  Occasional
random “mass‑transfer’’ moves are added to escape shallow plateaus.

The routine respects the 1000 s time budget, prints progress
and finally returns the best sequence found (scaled to the required
norm).  The code runs on a single CPU and makes only use of NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast (unchecked) evaluation – used for internal speed‑ups.
def fast_score(a: np.ndarray) -> float:
    """Return the same value as ``evaluate_sequence`` but without safety checks."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def grad_p_norm(a: np.ndarray, p: int,
                conv_size_c: int,
                conv_size_g: int) -> np.ndarray:
    """
    Gradient of  ∑_k (c_k)^p   where  c = a*a   (linear convolution).
    The returned vector has the same length as ``a``.
    """
    n = a.shape[0]

    # --- 1) compute self‑convolution c = a*a -------------------------
    fft_a = np.fft.rfft(a, conv_size_c)
    c = np.fft.irfft(fft_a * fft_a, conv_size_c)[: 2 * n - 1]

    # --- 2) u = c^{p‑1}   (element‑wise) ----------------------------
    # c is non‑negative; raising to a power is safe.
    u = np.power(np.maximum(c, 0.0), p - 1)

    # --- 3) gradient = 2·p·conv(a_rev , u) ---------------------------
    a_rev = a[::-1]
    fft_a_rev = np.fft.rfft(a_rev, conv_size_g)
    fft_u = np.fft.rfft(u, conv_size_g)
    conv_au = np.fft.irfft(fft_a_rev * fft_u, conv_size_g)[: 3 * n - 2]

    # slice the part that corresponds to g_i = 2p·∑_j a_j·u_{i+j}
    grad = 2.0 * p * conv_au[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The algorithm performs projected gradient descent on a smooth
    p‑norm surrogate of the objective, with a schedule that raises p to
    sharpen the approximation.  Random mass‑transfer moves are mixed in
    for extra exploration.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size and normalisation.
    n = 3000
    S_target = np.sqrt(2.0 * n)                     # sum = √(2n)  →  evaluation = max convolution

    # ------------------------------------------------------------------
    # 2. Initial sequence (spread over many indices).
    #    We start from an almost‑Sidon‑like set with equal weights.
    m_init = 90                                      # target support size (≈ sqrt(4·n/1.5))
    idx = np.linspace(0, n - 1, m_init, dtype=int) # roughly equally spaced positions
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m_init
    a = a / a.sum() * S_target                       # enforce the required sum

    # 2b. Blend with the previously found construction if available.
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                cand = prev[:n].copy()
            else:
                cand = np.zeros(n, dtype=np.float64)
                cand[:prev.size] = prev
            if cand.sum() > 0:
                cand = np.maximum(cand, 0.0)
                cand = cand / cand.sum() * S_target
                # simple averaging (50‑50) and re‑normalisation
                a = 0.5 * a + 0.5 * cand
                a = a / a.sum() * S_target
        except Exception:
            pass

    best_seq = a.copy()
    best_val = fast_score(a)
    print(f"[init] initial value {best_val:.6f}")

    # 3. Pre‑compute FFT sizes (powers of two for speed).
    conv_size_c = 1 << ((2 * n - 1).bit_length())   # for a*a

    # 6. Return the best sequence (as a plain Python list).
    # No further scaling is required because the sequence already has
