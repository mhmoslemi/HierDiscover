def _smooth_max_gradient(a, conv, beta, nfft):
    # conv is self-convolution (a*a)
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    a_rev = a[::-1]
    # convolution w * a_rev
    W = np.fft.rfft(w, n=nfft)
    A_rev = np.fft.rfft(a_rev, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[:len(w) + len(a_rev) - 1]
    # extract indices n-1 to n-1+n-1 (i.e., n size)
    start = len(a) - 1
    grad = 2.0 * grad_full[start:start + len(a)]
    return grad
```

"""
Annealed soft‑max gradient descent with simplex projection and occasional
random restarts.  The objective 2·n·max(conv)/(∑a)² reduces (with the fixed
ℓ¹‑norm S=√(2n)) to simply the maximal entry of the self‑convolution.
We minimise max(a*a) by optimiser iterations that
  * compute a smooth approximation of the max via a soft‑max (temperature β),
  * obtain its exact gradient using FFT‑based convolution,
  * take a projected (Euclidean) gradient step on the simplex,
  * occasionally perform random mass‑transfer moves and full Dirichlet restarts
    to avoid local minima,
  * stop as soon as the target bound 1.503 is reached or the time budget expires.
The routine returns the best sequence discovered.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_fft(u: np.ndarray, v: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution of real vectors u and v using FFT.
    Returns the full result of length len(u)+len(v)-1.
    """
    U = rfft(u, n=nfft)
    V = rfft(v, n=nfft)
    conv = irfft(U * V, n=nfft)[: len(u) + len(v) - 1]
    return conv


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return _convolve_fft(a, a, nfft)


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = target_sum}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_gradient(a: np.ndarray,
                         conv: np.ndarray,
                         beta: float,
                         nfft: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max_k (a*a)[k].
    The soft‑max weights are w_k ∝ exp(beta*(conv_k - max_conv)).
    The returned gradient has the same length as a.
    """
    cur_max = conv.max()
    # avoid overflow – shift by the maximum
    w = np.exp(beta * (conv - cur_max))
    w_sum = w.sum()
    if w_sum == 0:
        # fallback: uniform weights (should never happen)
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    a_rev = a[::-1]                     # reversed a
    # convolution w * a_rev (full length 3n‑2)
    grad_full = _convolve_fft(w, a_rev, nfft)
    n = a.size
    start = n - 1                       # index of a_rev aligned with a[0]
    grad = 2.0 * grad_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Annealed soft‑max gradient descent on the maximal self‑convolution.
    Returns the best non‑negative sequence (list of floats) found before the
    time budget expires.  The sequence is kept on the ℓ¹‑simplex
    Σa = √(2n) so that the evaluation function coincides with max(a*a).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------- problem size & initial point -------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)            # reuse the best known length
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2400                              # fallback size
        a = np.full(n, 1.0, dtype=np.float64)

    S = np.sqrt(2.0 * n)                     # target ℓ¹‑norm (kept fixed)
    a = _project_to_simplex(a, S)            # ensure feasibility

    # FFT lengths
    nfft_self = _next_pow2(2 * n - 1)         # for a*a (length 2n‑1)
    nfft_grad = _next_pow2(3 * n - 2)         # for w*a_rev (length 3n‑2)

    # ------------------- hyper‑parameters -------------------
    step_base = 0.5 * S / n                   # base step size for gradient moves
    beta = 0.5                                # initial soft‑max temperature
    beta_factor = 1.001                       # geometric increase per iteration
    beta_max = 30.0                           # cap to keep numbers stable

    temperature = 0.12                        # Metropolis temperature for occasional uphill moves
    temp_decay = 0.9997                       # decay per iteration
    min_temp = 1e-6

    p_grad = 0.70                             # probability of a gradient step
    p_random = 0.25                           # probability of a random mass‑transfer move
    # remaining probability → full restart

    stagnation = 0
    max_stagnation = 3000                     # trigger a restart after this many non‑improving iterations

    iteration = 0
    best_a = a.copy()
    # initial evaluation
    conv = _convolve_self_fft(a, nfft_self)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (a.sum() ** 2)   # equals cur_max for the chosen norm
    best_score = cur_score

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # recompute convolution and objective for the current a
        conv = _convolve_self_fft(a, nfft_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # keep the best seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:06d}] New best: {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # early exit if target attained
        if best_score < 1.503:
            print(f"Target reached after {iteration} iterations (best={best_score:.6f})")
            break

        # --------------------------------------------------------------
        # choose a move
        # --------------------------------------------------------------
        r = np.random.rand()
        if r < p_grad:
            # ----- gradient descent step on the smooth max -----
            grad = _smooth_max_gradient(a, conv, beta, nfft_grad)
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:
                eta = step_base / np.sqrt(iteration + 1)   # diminishing step size
                step = eta * grad / grad_norm
                a_candidate = _project_to_simplex(a - step, S)
            else:
                a_candidate = a.copy()

        elif r < p_grad + p_random:
            # ----- simple random mass‑transfer move -----
            if a.sum() > 0:
                src = np.random.choice(n, p=a / a.sum())
            else:
                src = np.random.randint(n)
            dst = src
            while dst == src:
                dst = np.random.randint(n)
            eps_transfer = 0.1
            delta = eps_transfer * a[src]
            if delta <= 0:
                delta = eps_transfer * S / n
            a_candidate = a.copy()
            a_candidate[src] -= delta
            a_candidate[dst] += delta
            a_candidate = _project_to_simplex(a_candidate, S)

        else:
            # ----- full restart from a Dirichlet draw -----
            alpha = np.full(n, 0.4)           # concentration (smaller → sparser)
            a_candidate = np.random.dirichlet(alpha) * S
            print(f"[{iteration:06d}] Full restart performed.")

        # --------------------------------------------------------------
        # evaluate the candidate (full objective)
        # --------------------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft_self)
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max / (a_candidate.sum() ** 2)

        # acceptance rule (greedy + Metropolis)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
            cur_max = cand_max
        else:
            # occasional jitter to escape shallow plateaux
            if np.random.rand() < 0.02:
                jitter = np.random.randn(n) * (step_base * 0.5)
                a = _project_to_simplex(a + jitter, S)

        # --------------------------------------------------------------
        # schedule updates
        # --------------------------------------------------------------
        temperature = max(temperature * temp_decay, min_temp)
        beta = min(beta * beta_factor, beta_max)

        # restart if we have been stuck for a long time
        if stagnation >= max_stagnation:
            a = np.random.dirichlet(np.full(n, 0.4)) * S
            stagnation = 0
            print(f"[{iteration:06d}] Restart due to stagnation.")

        iteration += 1

    # ------------------------------------------------------------------
    # final verification and return
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()
