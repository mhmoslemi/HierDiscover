def _build_conv_matrix(f):
    import scipy.sparse as sp
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for i, fi in enumerate(f):
        if fi == 0:

        rows.append(np.arange(i, i+n))
        cols.append(np.arange(n))
        data.append(np.full(n, fi))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    return A

from scipy.optimize import linprog

def _lp_step(f, M, S):
    n = len(f)
    A = _build_conv_matrix(f)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(b)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None
    scaling = S / sum_b
    g = scaling * b
    return g

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    if 'height_sequence_1' in globals():
        init_seq = np.array(height_sequence_1, dtype=float)
        n = init_seq.shape[0]
    else:
        n = 2000
        init_seq = None

    S = np.sqrt(2.0 * n)
    nfft = _next_pow2(2 * n - 1)

    if init_seq is not None:
        a = _project_to_simplex(init_seq, S)
    else:
        a = np.full(n, S / n, dtype=float)  # uniform

    # initial best
    conv = _convolve_self_fft(a, nfft)
    cur_max = conv.max()
    cur_score = 2 * n * cur_max / (a.sum() ** 2)
    best_a = a.copy()
    best_score = cur_score

    # hyperparameters
    step_base = 0.5 * S / n
    temperature = 0.1
    temp_decay = 0.9994
    min_temp = 1e-7

    p_subgrad = 0.55
    p_random = 0.25
    p_lp = 0.2
    eps_mask = 1e-12

    iteration = 0
    stale = 0
    max_stale = 3000

    while time.time() < deadline:
        # recompute conv, max, score for a
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = 2 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:06d}] New best: {best_score:.6f} (max={cur_max:.6f})")
            stale = 0
        else:
            stale += 1

        if best_score < 1.503:
            print(f"Target reached after {iteration} iterations, best {best_score:.6f}")
            break

        r = np.random.rand()
        if r < p_subgrad:
            # subgradient
            mask = (conv >= cur_max - eps_mask)
            grad = _grad_max_subgrad(a, mask, nfft)
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:
                eta = step_base / np.sqrt(iteration + 1)
                step = eta * grad / grad_norm
                a_candidate = _project_to_simplex(a - step, S)
            else:
                a_candidate = a.copy()
        elif r < p_subgrad + p_random:
            # random mass transfer
            src_probs = a / a.sum()
            i = np.random.choice(n, p=src_probs)
            j = np.random.choice([idx for idx in range(n) if idx != i])
            delta = 0.07 * a[i]
            if delta <= 0:
                delta = 0.07 * S / n
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = _project_to_simplex(a_candidate, S)
        else:
            # LP step
            M = cur_max
            g = _lp_step(a, M, S)
            if g is not None:
                # mix candidates
                candidate_f, candidate_score = _best_mix(a, g, M, S, nfft)
                a_candidate = candidate_f
            else:
                # fallback: jitter
                jitter = np.random.randn(n) * (step_base * 0.4)
                a_candidate = _project_to_simplex(a + jitter, S)

        # evaluate candidate using own conv
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        cand_score = 2 * n * cand_max / (np.sum(a_candidate) ** 2)

        # acceptance rule with temperature
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
            cur_max = cand_max
        else:
            # occasional jitter
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * (step_base * 0.5)
                a = _project_to_simplex(a + jitter, S)

            a = np.random.dirichlet(alpha) * S
            stale = 0
            print(f"[{iteration:06d}] Restart after stagnation.")

    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")

"""
Hybrid optimizer for the step‑function problem.

The objective after projecting onto the ℓ₁‑simplex {x≥0, Σx = √(2n)} is
simply the maximal entry of the self‑convolution (max a∗a).  We therefore
minimise that quantity.

Algorithm (single‑process, 1 CPU):

1. Initialise a on the simplex – either from the global `height_sequence_1`
   or from a uniform / Dirichlet start.
2. Main loop (time‑bounded):
   * Compute the self‑convolution via an FFT (O(n log n)).
   * With probability p_subgrad perform a projected sub‑gradient step
     on the active set of convolution entries attaining the current maximum.
   * With probability p_random perform a cheap random mass‑transfer move.
   * With probability p_lp solve a linear program

         maximise    Σ b_j
         subject to  (a * b)_k ≤ M   for all k,
                     b_j ≥ 0

     where M is the present maximal convolution value.
     The LP is linear in b (convolution is a linear operator in the second
     argument) and is solved with `scipy.optimize.linprog` (HiGHS).  The
     solution b is scaled to the required ℓ₁‑norm S and then mixed with the
     current a.  A tiny line‑search over mixing coefficients t∈{0,¼,½,¾,1}
     selects the best mixture.
   * Accept the candidate greedily; occasionally accept uphill moves
     according to a slowly cooling Metropolis temperature.
   * Restart from a Dirichlet draw after a long stagnation period.

The mix of deterministic sub‑gradient descent and global LP moves
explores the non‑convex landscape more effectively than the previous
pure sub‑gradient / simulated‑annealing code, and the exact linear‑program
step enforces the convolution bound tightly.

The routine returns the best sequence found before the time budget expires.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear self‑convolution a∗a using FFT, returns full length (2·len(a)-1)."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_max_subgrad(a: np.ndarray, mask: np.ndarray, nfft: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)[k] at the active set given by `mask`.
    Returns a gradient vector of the same length as `a`.
    """
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros_like(a)

    w = w / w_sum                     # turn into a probability vector
    a_rev = a[::-1]

    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    # slice of length n centred at index n‑1 gives the gradient entries
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ b) = f ∗ b (linear convolution).
    Sparse representation is used to keep memory reasonable.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # For each i (position in f) the contribution goes to rows i+j, column j.
    j_all = np.arange(n, dtype=np.int64)
    for i, fi in enumerate(f):
        if fi == 0.0:
            continue
        rows.append(i + j_all)          # shape (n,)
        cols.append(j_all)              # shape (n,)
        data.append(np.full(n, fi, dtype=np.float64))

    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_step(f: np.ndarray, M: float, S: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise    Σ b_j
        subject to  (f ∗ b)_k ≤ M   (for all k)
                    b_j ≥ 0.
    Returns the scaled vector g (with ℓ₁‑norm S) or None if the LP fails.
    """
    n = f.shape[0]

    # Build convolution matrix once per LP call.
    A = _build_conv_matrix(f)                     # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Maximise Σ b_j  →  minimise –Σ b_j
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n

    # HiGHS works with sparse A_ub directly.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None

    # Scale to the required ℓ₁‑norm S.
    scaling = S / sum_b
    g = scaling * b
    return g


def _best_mix(a: np.ndarray, g: np.ndarray, nfft: int) -> tuple[np.ndarray, float]:
    """
    Given current point a and a promising point g, try a short line‑search
    over mixing coefficients t ∈ {0,¼,½,¾,1} and return the mixture with
    minimal maximal convolution value (i.e. minimal objective).
    """
    t_vals = (0.0, 0.25, 0.5, 0.75, 1.0)
    best_f = a
    best_score = np.inf

    for t in t_vals:
        if t == 0.0:
            cand = a
        elif t == 1.0:
            cand = g
        else:
            cand = (1.0 - t) * a + t * g

        conv = _convolve_self_fft(cand, nfft)
        cur_max = conv.max()
        # Because Σcand = √(2n) by construction, the objective equals cur_max.
        if cur_max < best_score:
            best_score = cur_max
            best_f = cand

    return best_f, best_score


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Hybrid sub‑gradient / LP / simulated‑annealing optimiser.
    Returns the best non‑negative sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin for finalisation

    # ------------------------------------------------------------------
    # problem size, target ℓ₁‑norm and FFT length
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
    else:
        n = 2000
        init = None

    S = math.sqrt(2.0 * n)                 # fixed ℓ₁‑norm
    nfft = _next_pow2(2 * n - 1)           # FFT length ≥ convolution length

    # initialise a on the simplex
    if init is not None:
        a = _project_to_simplex(init, S)
    else:
        a = np.full(n, S / n, dtype=np.float64)   # uniform start

    # bookkeeping of the global best
    conv = _convolve_self_fft(a, nfft)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    # ------------------------------------------------------------------
    # hyper‑parameters
    # ------------------------------------------------------------------
    step_base = 0.5 * S / n               # base step size for sub‑gradient moves
    temperature = 0.12                    # Metropolis temperature
    temp_decay = 0.9995
    min_temp = 1e-7

    p_subgrad = 0.55
    p_random = 0.30
    p_lp = 0.15                          # probability of invoking the LP step

    eps_mask = 1e-12
    stale = 0
    max_stale = 3000                     # restart after this many non‑improving iters

    iteration = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # recompute convolution and maximal entry for the current a
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = cur_max               # after projection Σa = S the objective = max

        # update global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:07d}] New best: {best_score:.6f}")
            stale = 0
        else:
            stale += 1

        # early stop if we already beat the target
        if best_score < 1.503:
            print(f"Target reached after {iteration} iterations (best={best_score:.6f})")
            break

        # ------------------------------------------------------------------
        # decide which move to make
        # ------------------------------------------------------------------
        r = np.random.rand()
        if r < p_subgrad:
            # ---------- sub‑gradient descent on the active max set ----------
            mask = (conv >= cur_max - eps_mask)
            grad = _grad_max_subgrad(a, mask, nfft)
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:
                eta = step_base / math.sqrt(iteration + 1)
                step = eta * grad / grad_norm
                a_candidate = _project_to_simplex(a - step, S)
            else:
                a_candidate = a.copy()

        elif r < p_subgrad + p_random:
            # ---------- cheap random mass‑transfer move ----------
            if a.sum() > 0:
                src_probs = a / a.sum()
                i = np.random.choice(n, p=src_probs)
            else:
                i = np.random.randint(n)
            # pick a destination different from i uniformly
            j = i
            while j == i:
                j = np.random.randint(n)
            # transfer a modest fraction of the source mass
            eps_transfer = 0.07
            delta = eps_transfer * a[i]
            if delta <= 0:
                delta = eps_transfer * S / n
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = _project_to_simplex(a_candidate, S)

        else:
            # ---------- LP‑driven move ----------
            g = _lp_step(a, cur_max, S)
            if g is not None:
                # mix a and g, choose best mixing proportion
                a_candidate, _ = _best_mix(a, g, nfft)
            else:
                # fallback: small random jitter
                jitter = np.random.randn(n) * (step_base * 0.5)
                a_candidate = _project_to_simplex(a + jitter, S)

        # ------------------------------------------------------------------
        # evaluate the candidate (full objective)
        # ------------------------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        cand_score = cand_max

        # acceptance rule (greedy + Metropolis)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
            cur_max = cand_max
        else:
            # occasional jitter to shake out of shallow plateaus
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * (step_base * 0.4)
                a = _project_to_simplex(a + jitter, S)

        # temperature decay
        temperature = max(temperature * temp_decay, min_temp)

        # restart if stagnating
        if stale >= max_stale:
            alpha = np.full(n, 0.5)            # Dirichlet concentration
            a = np.random.dirichlet(alpha) * S
            stale = 0
            print(f"[{iteration:07d}] Restart after stagnation.")

        iteration += 1

    # ------------------------------------------------------------------
    # final verification and return
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()
