"""
Search for a non‑negative step‑function (vector of heights) that minimizes

    2·n·max(conv(a,a))   (the denominator Σa = 1 is enforced by a simplex projection)

The algorithm consists of three stages:

1. **Power‑law seeding** – a short grid search over a one‑parameter family
   a_i ∝ (i+offset)^(–p).  The best vector from this cheap sweep is taken as
   a starting point for the next stage.

2. **Smoothed‑max projected Adam** – the non‑smooth max is replaced by a
   temperature‑controlled soft‑max:
        Lτ(a) = τ·log Σ_k exp(conv_k/τ) .
   Its gradient is a weighted sum of the individual convolution‑gradients.
   We descend this smooth objective with Adam, projecting after each step
   onto the probability simplex (non‑negative, sum = 1).  The temperature τ
   is gradually reduced so that the algorithm finally focuses on the true
   max.

3. **Multi‑restart & final check** – a few random Dirichlet starts are also
   refined for a short budget; the overall best vector is returned after a
   final call to ``evaluate_sequence`` (the official black‑box evaluator).

All heavy work is done with FFT‑based convolutions, so the routine easily
fits into the 1000 s time limit (default 950 s internal margin).  Progress
messages are printed to help the user monitor the search.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Return a non‑negative vector a_i = (i+offset)^(-p), i=0..n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:                 # safety – should never happen
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def uniform_sequence(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using Gamma(1,1) (flat Dirichlet)."""
    a = rng.gamma(1.0, 1.0, size=n)
    a /= a.sum()
    return a.astype(np.float64)


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Self‑convolution x*x using a real FFT (length 2·len(x)‑1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w ≥0, Σw = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Return the objective 2·n·max(conv(a,a)) (assumes Σa = 1)."""
    n = a.shape[0]
    conv = conv_fft(a)
    return 2.0 * n * conv.max()


def smoothed_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the temperature‑smoothed max:
        ∂/∂a  τ·log Σ_k exp(conv_k/τ)
    Using the identity grad_k[i] = 2·a_{k‑i} (when indices are valid).
    The result is computed via a single FFT‑based convolution.
    """
    n = a.shape[0]
    conv = conv_fft(a)                     # length 2n‑1
    max_c = conv.max()
    if tau < 1e-12:                         # effectively a hard max
        w = np.zeros_like(conv)
        w[conv == max_c] = 1.0 / np.sum(conv == max_c)
    else:
        # numerical‑stable softmax
        shifted = (conv - max_c) / tau
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

    # Weighted sum of individual gradients:
    # g_i = 2 * Σ_j a_j w_{i+j}
    w_rev = w[::-1]                        # reverse for convolution
    L = n + w_rev.size - 1                  # = 3n‑2
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)
    tmp = np.fft.irfft(A * W, n=L)         # length L
    idx = (2 * n - 2) - np.arange(n)       # pick required entries
    grad = 2.0 * tmp[idx]
    return grad


def adam_optimize(
    a0: np.ndarray,
    n: int,
    start_tau: float,
    tau_decay: float,
    decay_every: int,
    eta0: float,
    max_iter: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the smoothed max objective.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    tau = start_tau
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        # ----- gradient of smoothed max -----
        grad = smoothed_gradient(a, tau)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size scales with η₀ and the current temperature (optional heuristic)
        step = eta0 * (tau / start_tau) * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_onto_simplex(a_new)

        # ----- evaluate -----
        cur_score = compute_score(a)
        new_score = compute_score(a_new)

        # keep global best
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[adam iter {it:08d}] new global best = {best_score:.6f}")

        # accept or reject (simple monotone acceptance)
        if new_score <= cur_score:
            a = a_new
        else:
            # occasional uphill acceptance to escape flat plateaus
            if rng.random() < 0.001:
                a = a_new

        # temperature schedule
        if it % decay_every == 0:
            tau = max(tau * tau_decay, 1e-8)

    return best_a, best_score

            for p in fine_ps:
                if time.time() > deadline:
                    break
                a = power_law_sequence(n, p, offset=off)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    best_p = p
                    best_off = off
                    print(f"[fine] new best  p={p:.4f}, off={off}, score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Phase 3 – multi‑restart Adam refinement
    # ------------------------------------------------------------------
    print("[refine] starting Adam refinement")
    # primary refinement from the best power‑law seed
    best_a, best_score = adam_optimize(
        a0=best_a,
        n=n,
        start_tau=start_tau,
        tau_decay=tau_decay,
        decay_every=decay_every,
        eta0=eta0,
        max_iter=max_total_iters,
        rng=rng,
        deadline=deadline,
    )

    # a few quick random restarts (each limited to a modest number of steps)
    num_restarts = 3
    restart_iters = 200_000  # per restart
    for r in range(num_restarts):
        if time.time() > deadline:
            break
        a0 = random_dirichlet(n, rng)
        a_refined, sc_refined = adam_optimize(
            a0=a0,
            n=n,
            start_tau=start_tau,
            tau_decay=tau_decay,
            decay_every=decay_every,
            eta0=eta0,
            max_iter=restart_iters,
            rng=rng,
            deadline=deadline,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[restart {r}] improved global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final sanity check using the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] best reported score = {final_score:.6f}")
    print(f"[info] final parameters: power‑law p={best_p}, offset={best_off}, length={n}")

    return best_a.tolist()
