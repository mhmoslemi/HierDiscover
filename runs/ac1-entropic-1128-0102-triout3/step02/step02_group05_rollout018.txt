def cross_corr(a,w):
    # a length n, w length 2n-1
    n = a.shape[0]
    size = 1 << ((3*n - 2).bit_length())
    fft_a = np.fft.rfft(a, size)
    fft_w = np.fft.rfft(w, size)
    # convolution a_rev * w as before - incorporate reversal
    a_rev = a[::-1]

    conv = np.fft.irfft(fft_ar * fft_w, size)
    # conv length 3n-2
    start = n-1
    grad = conv[start:start+n]
    return grad

        val = fast_score(a)

        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[:2*n-1]

        # compute softmax weights
        maxc = conv.max()
        exps = np.exp(lambda_val * (conv - maxc))
        w = exps / exps.sum()

        # gradient via correlation a_rev * w
        a_rev = a[::-1]
        # FFT size for correlation
        size_corr = 1 << ((3*n-2).bit_length())
        fft_rev = np.fft.rfft(a_rev, size_corr)
        fft_w = np.fft.rfft(w, size_corr)
        corr = np.fft.irfft(fft_rev * fft_w, size_corr)
        grad = 2.0 * corr[n-1: n-1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - step_size * m_hat / (np.sqrt(v_hat) + eps)

                print(f"[iter {t}] new best {best_val:.6f}")

        lambda_val = min(200.0, lambda_val + lambda_inc)
        t += 1

    print(f"Final best {best_val:.6f}")
    return best_seq.tolist()

    * Uniform distribution.

The whole search stops after the allotted time (default 1000 s) and
returns the best sequence seen.  All values are automatically clipped
to the allowed range by ``evaluate_sequence``; the algorithm itself
keeps the sequence inside [0,1] and therefore never triggers the safety
filters.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Fast internal evaluation (identical to the public `evaluate_sequence`
# but without the safety checks).  The returned value equals the
# objective that `evaluate_sequence` later computes.
def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) / (sum(a)²) for a non‑negative vector a."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of the vector v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def beta_u_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape on a uniform grid.
    The density is proportional to x^{α‑1}·(1‑x)^{α‑1}, where
    x = (i+½)/n, i = 0,…,n‑1.
    The result is normalised to ℓ¹‑norm 1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5          # centre of each bin
    x = i / n
    # avoid NaNs when α is exactly 0 – we never call with α ≤ 0.
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a[np.isnan(a)] = 0.0
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def symmetric_power_law(n: int, p: float) -> np.ndarray:
    """
    Symmetric decreasing power‑law:
        a_i ∝ (i+1)^{-p} + (n‑i)^{-p},   i = 0,…,n‑1 .
    Normalised to sum 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p) + np.power(n - i + 1, -p)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def cross_corr_grad(a: np.ndarray, w: np.ndarray,
                    conv_fft_size: int, corr_fft_size: int) -> np.ndarray:
    """
    Compute the gradient g_i = 2 * Σ_k w_k a_{k-i}
    (cross‑correlation of a with w) using FFTs.
    Returns a vector of length n.
    """
    a_rev = a[::-1]
    fft_rev = np.fft.rfft(a_rev, corr_fft_size)
    fft_w = np.fft.rfft(w, corr_fft_size)
    corr = np.fft.irfft(fft_rev * fft_w, corr_fft_size)
    n = a.shape[0]
    # slice that corresponds to indices i = 0,…,n‑1:
    #     g_i = 2·corr[n‑1+i]
    return 2.0 * corr[n - 1: n - 1 + n]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights with a low evaluation
    value.  Returns the best sequence (as a plain Python list) found
    before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5       # safety margin

    # -------------------------- 1. problem size --------------------------
    n = 4000                                      # a size that balances speed & quality

    # -------------------------- 2. cheap parametric scan -----------------
    best_seq = None
    best_val = np.inf

    # Uniform distribution (benchmark)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_val = fast_score(uniform)
    print(f"[init] uniform   value = {best_val:.6f}")

    # Symmetric power‑law families
    for p in [0.3, 0.4, 0.5, 0.6, 0.8]:
        cand = symmetric_power_law(n, p)
        val = fast_score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
        print(f"[init] power p={p:.2f}  value = {val:.6f}")

    # Symmetric Beta families (U‑shaped)
    alphas = np.linspace(0.30, 1.00, 15)          # α=0.5 is the arcsine case
    for α in alphas:
        cand = beta_u_shape(n, α)
        val = fast_score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
        print(f"[init] beta α={α:.2f}  value = {val:.6f}")

    # Optionally warm‑start from a previous construction
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                cand = prev[:n].copy()
            else:
                cand = np.zeros(n, dtype=np.float64)
                cand[:prev.size] = prev
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0.0:
                cand /= s
                val = fast_score(cand)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                print(f"[init] height_sequence_1 value = {val:.6f}")
        except Exception:
            pass

    # -------------------------- 3. smooth‑max gradient descent ----------
    a = best_seq.copy()
    m = np.zeros_like(a)          # Adam first moment
    v = np.zeros_like(a)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step_size = 0.05

    # FFT sizes (pre‑computed for speed)
    conv_fft_size = 1 << ((2 * n - 1).bit_length())
    corr_fft_size = 1 << ((3 * n - 2).bit_length())

    # schedule for the soft‑max temperature λ
    lam = 5.0
    lam_target = 200.0
    lam_inc = (lam_target - lam) / max(1, (deadline - time.time()))   # linear increase

    it = 1
    while time.time() < deadline:
        # ----- self‑convolution a∗a -----
        fft_a = np.fft.rfft(a, conv_fft_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_fft_size)[: 2 * n - 1]

        # current objective (sum(a)=1, so value = 2·n·max(conv))
        cur_val = 2.0 * n * conv.max()

        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            print(f"[iter {it:6d}] new best {best_val:.6f}")

        # ----- soft‑max approximation of the max -----
        maxc = conv.max()
        # (λ may become large; shifting by maxc avoids overflow)
        exp_weights = np.exp(lam * (conv - maxc))
        w = exp_weights / exp_weights.sum()

        # ----- gradient (cross‑correlation a with w) -----
        grad = cross_corr_grad(a, w, conv_fft_size, corr_fft_size)

        # ----- Adam update + simplex projection -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - step_size * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)                 # keep a on the ℓ¹‑simplex

        # ----- temperature schedule -----
        lam = min(lam_target, lam + lam_inc)

        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    return best_seq.tolist()
