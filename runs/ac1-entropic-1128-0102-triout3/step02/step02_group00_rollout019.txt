"""
Multi‑scale optimisation for the step‑function problem.

The evaluation function is homogeneous, so we optimise on the
probability simplex (∑a_i = 1).  For each chosen length we

* initialise with a discretised arcsine distribution (the ½‑Beta law);
* run a projected ADAM optimiser that follows a sub‑gradient of the
  max‑convolution (averaged over all almost‑active entries);
* finish with a cheap simulated‑annealing “mass‑transfer’’ local search;
* up‑sample the best sequence when moving to a longer vector.

Both the optimiser and the annealer respect the simplex constraints
by projecting onto the simplex after every update.
The whole routine respects the global time‑budget (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.sparse import csr_matrix

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σx_i = 1} (O(n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution a * a (linear, not circular) via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself by FFT (real‑valued)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient from a set of active convolution indices
# ------------------------------------------------------------
def _subgradient_from_active(a: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    """
    Average sub‑gradient of the max‑convolution over the active set.
    The gradient of (a*a)[k] w.r.t. a_j is 2*a_{k-j} (if the index exists).
    """
    n = a.shape[0]
    # limit the size of the active set to keep the loop cheap
    if act_idx.size > 200:
        act_idx = np.random.choice(act_idx, size=200, replace=False)

    g = np.zeros_like(a)
    for k in act_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]

    if act_idx.size > 0:
        g /= act_idx.size
    return g


# ------------------------------------------------------------
# Projected ADAM optimiser on the simplex
# ------------------------------------------------------------
def _run_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    max_iter: int = 200_000,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
):
    """
    Minimise 2*n*max(conv) on the simplex using ADAM.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"ADAM improve: n={n}, iter={it}, score={cur_score:.6f}"
            )

        # indices whose value is within a tiny relative tolerance of the max
        rel_tol = 1e-9
        act = np.where(conv >= max_val - rel_tol * max_val)[0]

        # sub‑gradient (average over the active set)
        grad = _subgradient_from_active(a, act)

        # ADAM update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = _project_simplex(a - step)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing local refinement (mass transfer moves)
# ------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_seq: np.ndarray,
    cur_best_score: float,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    max_steps: int = 30_000,
):
    """
    Tiny “move ε from i to j’’ annealing, returns possibly improved pair.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move a random fraction (up to 10 %) of the mass from i to j
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"annealing improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up /= up.sum()                      # re‑normalise to sum = 1
    return up


# ------------------------------------------------------------
# Initial sequence: discretised arcsine (Beta(½,½)) law
# ------------------------------------------------------------
def _arcsine_init(n: int) -> np.ndarray:
    """
    a_i ∝ 1 / sqrt(x_i (1‑x_i)),  x_i = (i+0.5)/n.
    The vector is normalised to sum = 1 (probability simplex).
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a /= a.sum()
    return a


# ------------------------------------------------------------
# Main entry point required by the evaluation harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s`
    seconds, combines a multi‑scale ADAM optimiser with a final
    simulated‑annealing refinement, and returns the best sequence
    (as a plain Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_score = float("inf")
    best_seq = None

    # length schedule – modest growth to stay inside the time budget
    lengths = [300, 600, 1200, 2400, 3600, 4800]
    np.random.shuffle(lengths)               # avoid bias if we stop early

    # keep a reference vector (in the simplex) for the next scale
    reference = None

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- initialise -------------------------------------------------
        if reference is not None:
            a0 = _upsample(reference, n)
        else:
            a0 = _arcsine_init(n)

        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        slice_deadline = time.time() + max(5.0, 0.12 * remaining)

        # ----- ADAM optimisation -----------------------------------------
        a_opt, score_opt = _run_adam(
            a0,
            n,
            slice_deadline,
            start_time,
            max_iter=120_000,
            lr=0.05,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"global improve after ADAM: n={n}, score={score_opt:.6f}"
            )

        # ----- simulated‑annealing refinement -------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(3.0, 0.04 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_seq,
                best_score,
                start_temp=5e-2,
                end_temp=1e-5,
                max_steps=30_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] "
                    f"global improve after annealing: n={n}, score={score_ref:.6f}"
                )

        # store the best vector for up‑sampling in the next round
        reference = best_seq.copy()

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # 100‑element uniform distribution – valid but not competitive
        best_seq = np.full(100, 1.0 / 100.0)

    return [float(x) for x in best_seq]
