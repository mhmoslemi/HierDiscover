Exponentiated‑gradient search on the probability simplex.

The algorithm builds several initial step‑functions (uniform, random Dirichlet,
Arcsine‑type, and an up‑sampled previous best).  For each length it runs a

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Score = 2 * n * max(conv(a, a)) for a simplex‑normalised vector."""
    n = a.shape[0]

    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        n = a.shape[0]
        L_cov = 1 << ((2 * n - 1).bit_length())
        step0 = 0.30
        eps_rel = 0.008

        for it in range(1, 15001):
            if time.time() >= deadline:
                break
            Af = np.fft.rfft(a, n=L_cov)
            conv = np.fft.irfft(Af * Af, n=L_cov)[: 2 * n - 1]
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] "
                    f"Final refine, iter={it}, score={cur_score:.6f}"
                )

            # active set of near‑max convolution indices
            eps = max_val * eps_rel
            K = np.where(conv >= max_val - eps)[0]
            g = np.zeros_like(a)
            for k in K:
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                i_idx = np.arange(low, high + 1)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]

            step = step0 / math.sqrt(it)
            a = a - step * g
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s

        # keep the refined candidate if it improved
        if a is not None:
            final_score = _score_of_seq(a)
            if final_score < best_score:
                best_seq = a.copy()
                best_score = final_score

    # ------------------------------------------------------------------
    # 5. Fallback (should never happen) and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200)

    # convert to plain Python floats as required by the caller
    return [float(x) for x in best_seq]
