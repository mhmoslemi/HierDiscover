    rows = 2*n -1

    """Return a normalized nonnegative sequence of length n approximating Beta(alpha,alpha) distribution."""
    # Use midpoints to avoid zero singularities
    xs = (np.arange(n) + 0.5) / n  # values in (0,1)
    # Compute raw density: x^{alpha-1} (1-x)^{alpha-1}
    # For alpha == 0, treat as uniform? but we won't use.
    raw = np.power(xs, alpha - 1) * np.power(1 - xs, alpha - 1)

    else:
        raw = raw / total
    return raw

"""
Propose a candidate step‑function that minimizes the given evaluation score.
The algorithm combines three ingredients:

1. **Parametric sweep** – a family of symmetric Beta(α,α) densities is
   discretised, normalised and evaluated.  This cheap scan often finds a
   shape much better than the uniform start.

2. **LP‑based refinement** – for a fixed length `n` the current sequence
   `f` defines a linear operator `T_f` (convolution with `f`).  Solving
   the linear programme

       max   Σ b_j
       s.t.  T_f b ≤ M          (M = max(f∗f))
             b ≥ 0

   yields a direction `b` that can increase the total mass while keeping
   the worst convolution entry ≤ M.  Mixing `b` back into `f` strictly
   reduces the objective.  A few such LP steps give a sizeable drop.

3. **Projected multi‑peak sub‑gradient descent** – the final polishing
   stage uses the same sub‑gradient scheme as the reference solution,
   but with a decreasing step‑size and a Euclidean projection onto the
   probability simplex.  It runs until the allotted 1000 s are exhausted.

The whole routine respects the 1000 s budget, prints intermediate
improvements and returns the best sequence found (as plain Python
floats).  It works with any previously discovered `height_sequence_1`
global variable but does not rely on it.

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_of_seq(a: np.ndarray) -> float:
    """Score 2·n·max(conv(a,a)) for a sequence that already sums to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _beta_sequence(alpha: float, n: int) -> np.ndarray:

    s = raw.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return raw / s

            max_val = conv.max()
            cur_sc = 2.0 * cur_n * max_val
            if cur_sc < best_score:
                best_score = cur_sc
                best_seq = a_vec.copy()
                print(f"[{time.time() - start:.2f}s] Upsample n={cur_n} it={it} → score={cur_sc:.6f}")
            eps = max_val * eps_rel
            K = np.where(conv >= max_val - eps)[0]
            g = np.zeros_like(a_vec)
            for k in K:
                lo = max(0, k - (cur_n - 1))
                hi = min(cur_n - 1, k)
                i_idx = np.arange(lo, hi + 1)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a_vec[j_idx]
            step = step0 / math.sqrt(it + 1)
            a_vec = _project_simplex(a_vec - step * g)
        best_seq = a_vec.copy()

    # ------------------------------------------------------------------
    # 3. LP‑based refinement (only for moderate lengths, requires cvxpy)
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp

        def _lp_refine(seq, time_limit):
            """Perform a few LP‑based improvement steps on `seq`."""
            f = seq.copy()
            n = f.shape[0]
            rows = 2 * n - 1
            best_local = f.copy()
            best_local_score = _score(f)

            while time.time() < time_limit:
                # current max convolution value
                fft_len = 1 << ((2 * n - 1).bit_length())
                Af = np.fft.rfft(f, n=fft_len)
                conv = np.fft.irfft(Af * Af, n=fft_len)[:rows]
                M = conv.max()

                # build the convolution matrix T (dense, but n is kept modest)
                T = np.zeros((rows, n))
                for j in range(n):
                    T[j:j + n, j] = f  # column j is f shifted

                # LP: maximise Σ b_j  subject to T b ≤ M and b ≥ 0
                b = cp.Variable(n, nonneg=True)
                constraints = [T @ b <= M * np.ones(rows)]
                prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)

                if b.value is None:
                    break
                sum_b = float(b.value.sum())
                # if we cannot increase total mass, stop
                if sum_b <= 1.0 + 1e-8:
                    break

                # mix the new direction into f
                mix = min(0.15, (sum_b - 1.0) / (2.0 * sum_b))
                f = (1.0 - mix) * f + mix * (b.value / sum_b)

                cur_sc = _score(f)
                if cur_sc < best_local_score:
                    best_local_score = cur_sc
                    best_local = f.copy()
                    print(f"[{time.time() - start:.2f}s] LP step improved → n={n}, score={cur_sc:.6f}")

                # early stopping if the said improvement is negligible
                if M * (sum_b - 1.0) < 1e-6:
                    break
            return best_local, best_local_score

        # only attempt LP refinement when the length is not too large
        if best_seq.shape[0] <= 2500:
            remaining = deadline - time.time()
            lp_time = max(5.0, 0.15 * remaining)   # at least a few seconds
            refined_seq, refined_sc = _lp_refine(best_seq, time.time() + lp_time)
            if refined_sc < best_score:
                best_score = refined_sc
                best_seq = refined_seq.copy()
    except Exception as e:
        # cvxpy not available or LP failed – silently continue
        print(f"LP refinement skipped ({e})")

    # ------------------------------------------------------------------
    # 4. Final projected multi‑peak sub‑gradient optimisation
    # ------------------------------------------------------------------
    n = best_seq.shape[0]
    a = best_seq.copy()
    fft_len = 1 << ((2 * n - 1).bit_length())
    step0 = 0.4
    eps_rel = 0.008
    max_iters = 150000
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_score:
            best_score = cur_sc
            best_seq = a.copy()
            print(f"[{time.time() - start:.2f}s] Sub‑grad it={it} improved → score={cur_sc:.6f}")

        eps = max_val * eps_rel
        K = np.where(conv >= max_val - eps)[0]
        g = np.zeros_like(a)
        for k in K:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_idx = np.arange(lo, hi + 1)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * g)

    # ------------------------------------------------------------------
    # 5. Return the best candidate (plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    return [float(v) for v in best_seq]
