def _lp_improve(f, M):
    n = f.shape[0]
    # build constraint matrix
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows i .. i + n - 1 contain coefficient f[i]
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # objective maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    # bounds for variables: (0, None)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        g = res.x
        sg = g.sum()
        if sg > 1e-12:
            g_norm = g / sg  # now sum=1
            return g_norm, sg
    # fallback - return original f
    return f, 0

"""
search for a non‑negative step function that minimizes evaluate_sequence.

Approach
--------
1. Work on the probability simplex (∑a_i = 1) – the score is scale invariant.
2. Use a multi‑scale schedule: start with a modest length, repeatedly up‑sample
   the best sequence to larger lengths.
3. For each length run a long projected‑Adam gradient descent that minimises the
   maximal entry of the self‑convolution.
4. When the length is moderate (≤2400) perform a few exact LP “capacity” steps:
   maximise ∑g_i under the current convolution bound M = max(f*f).  The LP
   solution g can be blended with the current f to strictly improve the bound.
5. Finish with a cheap simulated‑annealing refinement.
6. Keep the best sequence seen across all lengths and return it.

The algorithm respects the 1000 s time budget, prints progress and returns a
plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a using FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Indices of the K largest convolution entries
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, K: int) -> np.ndarray:
    m = conv.size
    if K >= m:
        return np.arange(m, dtype=int)
    idx = np.argpartition(-conv, K - 1)[:K]
    return idx.astype(int)


# ----------------------------------------------------------------------
# Sub‑gradient from the active convolution entries
# ----------------------------------------------------------------------
def _subgradient_from_active(a: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] += 2.0 * a[j]
    if act_idx.size > 0:
        g /= act_idx.size
    return g


# ----------------------------------------------------------------------
# Projected Adam gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_gradient_descent_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    K: int = 20,
    beta1: float = 0.9,
    beta2: float = 0.999,
    step0: float = 1.0,
    max_iter: int = 200000,
):
    """Adam with simplex projection; returns the best sequence seen."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    eps = 1e-8
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it}, score={cur_score:.6f}"
            )

        # sub‑gradient built from the K largest entries
        act = _active_indices(conv, K)
        grad = _subgradient_from_active(a, act)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(v_hat) + eps) / math.sqrt(it)

        a = _project_simplex(a - step * m_hat)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (capacity step)
# ----------------------------------------------------------------------
def _lp_solve(f: np.ndarray, M: float):
    """Solve max ∑g_i  s.t. f * g ≤ M, g ≥ 0.  Returns (g_norm, sum_g) or (f,0)."""
    n = f.shape[0]
    # build sparse matrix A where row k corresponds to (f * g)[k]
    rows = []
    cols = []
    data = []
    for i, fi in enumerate(f):
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
        data.extend([fi] * n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                     # maximise sum(g)  ⇒ minimise -sum(g)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if res.success:
        g = res.x
        s = g.sum()
        if s > 0:
            return g / s, s
    return f, 0.0


def _run_lp_refinement(a0: np.ndarray, n: int, deadline: float, start_time: float,
                       max_steps: int = 5, blend: float = 0.5):
    """Apply a few LP capacity steps to improve a0."""
    a = a0.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    best_a = a.copy()

    for step in range(max_steps):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        M = conv.max()
        g_norm, sum_g = _lp_solve(a, M)
        if sum_g <= 0.0:
            break
        new_a = (1.0 - blend) * a + blend * g_norm
        new_score = 2.0 * n * _conv_self(new_a).max()
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            a = new_a
            print(
                f"[{time.time() - start_time:6.2f}s] LP improve: n={n}, step={step+1}, score={new_score:.6f}"
            )
        else:
            blend *= 0.5
            if blend < 1e-3:
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 20000,
):
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear interpolation up‑sampling
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
