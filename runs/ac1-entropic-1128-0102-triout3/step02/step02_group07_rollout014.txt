"""
Propose a candidate sequence for ``evaluate_sequence``.

Algorithm
---------
1. **Initial seed** – use the globally supplied ``height_sequence_1`` (the
   best sequence found so far) or fall back to a power‑law seed.
   The seed is normalised to sum = 1.

2. **Linear‑programming refinement** – for the current sequence ``f`` we
   solve a small LP that *directly* minimises the peak of the
   cross‑convolution ``f * g`` while keeping the total mass of ``g``
   equal to that of ``f``.  The LP variables are the new coefficients
   ``g`` (non‑negative) and a scalar ``M`` which bounds every entry of
   the convolution.  The LP is:

        minimise      M
        subject to    (f * g)[k] ≤ M    for k = 0 … 2n‑2
                      Σ g_i = 1
                      g_i ≥ 0 , M ≥ 0

   The solution gives a candidate with (usually) a smaller self‑peak
   after a few iterations.

3. **Projected gradient descent** – starting from the LP solution we
   run a short, accelerated projected gradient descent that targets the
   *active* convolution entry.  The Euclidean projection onto the
   probability simplex restores the non‑negative, unit‑sum constraint.

4. **Loop** – the LP+gradient cycle is repeated while time permits
   (the outer loop is limited to a handful of iterations).  The best
   sequence encountered (lowest value of ``evaluate_sequence``) is
   returned.

The whole routine respects a 1000 s budget (950 s are used internally
to leave a safety margin).  All helper functions are defined at the
module level, and only NumPy, SciPy and the supplied ``evaluate_sequence``
are used.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Self‑convolution using a real FFT (length 2·len(x)-1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w | w_i ≥ 0, Σ w_i = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Sub‑gradient of the active convolution entry (a*a)[peak_idx]."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense Toeplitz matrix T such that T @ g = f * g.
    Shape: (2*n-1, n)  where n = len(f).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)

    # each column i contains f shifted by i positions
    for i in range(n):
        length = rows - i
        if length >= n:
            T[i:i + n, i] = f
        else:                     # tail of the matrix (i large)
            T[i:, i] = f[:length]
    return T


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative sequence minimising the evaluation function.
    Returns the best sequence discovered before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0     # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Initial seed -------------------------------------------------
    try:
        # ``height_sequence_1`` is assumed to be a global list of floats.
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
        print("[init] using supplied height_sequence_1, n =", n)
    except Exception:
        n = 2500
        init = power_law_sequence(n, p=0.5, offset=0.5)
        print("[init] fallback to power‑law seed, n =", n)

    # Normalise to unit sum (the objective is homogeneous).
    if init.sum() == 0.0:
        init[:] = 1.0 / n
    else:
        init /= init.sum()

    best_a = init.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print("[init] initial score =", best_score)

    # ------------------------------------------------------------------
    # 2️⃣  Outer optimisation loop (LP + gradient) ---------------------
    outer_iter = 0
    max_outer = 6                      # keep the total run time reasonable

    while time.time() < deadline and outer_iter < max_outer:
        outer_iter += 1
        print(f"\n[outer {outer_iter}] start, best_score={best_score:.6f}")

        # ---- current peak ------------------------------------------------
        conv_f = conv_fft(best_a)
        M_cur = conv_f.max()
        print(f"[outer {outer_iter}] current M = {M_cur:.8f}")

        # ---- build LP (minimise peak of f * g) -------------------------
        T = build_convolution_matrix(best_a)                 # (2n-1, n)
        # Append the column for the scalar M with coefficient –1
        A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)

        # Equality: Σ g_i = 1, M free (no contribution)
        A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
        b_eq = np.array([1.0])

        # Objective: minimise M -> c = [0,…,0, 1]
        c = np.zeros(n + 1)
        c[-1] = 1.0

        bounds = [(0.0, None)] * n + [(0.0, None)]   # g_i ≥0, M ≥0

        # Solve the LP (HiGHS is fast for dense problems)
        lp_start = time.time()
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs')
        lp_time = time.time() - lp_start

        if not res.success:
            print("[lp] solver failed:", res.message)
            break

        g = res.x[:n]
        M_opt = res.x[-1]
        print(f"[lp] solved in {lp_time:.2f}s – M_opt = {M_opt:.8f}")

        # Numerical cleanup (tiny negatives may appear)
        g = np.maximum(g, 0.0)
        g /= g.sum()                     # enforce exact simplex

        # ---- evaluate the LP candidate ---------------------------------
        conv_g = conv_fft(g)
        max_g = conv_g.max()
        score_g = 2.0 * n * max_g        # sum(g)=1 → denominator = 1
        print(f"[lp] candidate self‑peak = {max_g:.8f}, score = {score_g:.6f}")

        if score_g < best_score:
            best_score = score_g
            best_a = g.copy()
            print(f"[lp] *** new global best (LP) = {best_score:.6f}")

        # ---- 3️⃣  Gradient refinement starting from the LP point -------
        a = best_a.copy()
        cur_score = best_score
        eta = 0.08                     # initial step size
        decay_factor = 0.93
        decay_every = 1500
        max_grad_iters = 20000

        it = 0
        while time.time() < deadline and it < max_grad_iters:
            conv = conv_fft(a)
            peak_idx = int(conv.argmax())
            grad = gradient_of_peak(a, peak_idx)

            # step opposite to gradient, never erase more than 99 % of a component
            step = eta * grad
            step = np.minimum(step, a * 0.99)
            a_trial = a - step

            # tiny random jitter to help escape flat plateaux
            if rng.random() < 0.003:
                a_trial += (rng.random(n) - 0.5) * 1e-5

            a_trial = project_onto_simplex(a_trial)

            trial_score = 2.0 * n * conv_fft(a_trial).max()

            if trial_score < best_score:
                best_score = trial_score
                best_a = a_trial.copy()
                print(f"[grad iter {it:06d}] new global best = {best_score:.6f}")

            # accept if it improves; otherwise accept with a tiny probability
            if trial_score <= cur_score:
                a = a_trial
                cur_score = trial_score
            else:
                # simple Metropolis acceptance (temperature decays with eta)
                if rng.random() < 0.01:
                    a = a_trial
                    cur_score = trial_score

            if (it + 1) % decay_every == 0:
                eta = max(eta * decay_factor, 1e-6)

            it += 1

        # keep the refined vector as the start for the next outer iteration
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[outer {outer_iter}] after gradient, best_score={best_score:.6f}")

        # quick sanity check (calls the official evaluator)
        sanity = evaluate_sequence(best_a.tolist())
        if sanity < best_score - 1e-9:
            # should never happen – keep the evaluator's value
            best_score = sanity
            print(f"[sanity] evaluator returned an even better score {sanity:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final sanity check & return ------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\n[done] iterations={outer_iter}, final reported score={final_score:.6f}")
    return best_a.tolist()
