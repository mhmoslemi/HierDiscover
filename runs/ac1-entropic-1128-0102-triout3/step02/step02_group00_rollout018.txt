def _subgradient_full(a, conv, eps=1e-12):
    max_val = conv.max()
    act = np.where(conv >= max_val - eps)[0]  # active indices
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] += a[j]  # from a_{k-j} term
        # The second term (a_{j-k}) adds a[j] where (j-k) is in range, but that's just a[i]? Because for fixed k and i (i = index), j = k-i. The term a_{j - k} = a_{-i}? Actually a_{j-k} = a_{-(i)} => not valid unless i=0? hold on.
        # Let's derive: gradient_j = a_{k-j} + a_{j-k} if indices in range.
        # For j indexing over all positions, we need to add contributions from both sides.
        # In the loop over i = lo..hi (these are indices where i ranges over a_i that appear in sum_i a_i a_{k-i}).
        # For each such i, j = k-i. The term a_{i} a_{j} contributes to derivative w.r.t a_i: a_j; w.r.t a_j: a_i.
        # In our loop we add to g[i] += a[j] (covers derivative w.r.t a_i). We also need to add to g[j] += a[i] (derivative w.r.t a_j). This will double count when i=j.
        # To avoid double counting, we can add both contributions. Since both i and j belong to overlapping index set, we can just add both.
        g[j] += a[i]   # derivative w.r.t a_j
    # After loop, for each k, we have accounted both terms; we must divide by len(act) to average if we want average subgradient.
    if act.size > 0:
        g /= act.size
    return g

def _run_adam(a, n, deadline, start_time, tolerance=1e-6):
    a = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.1  # base step
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve: n={n}, iter={it}, score={cur_score:.6f}")
        # gradient
        grad = _subgradient_full(a, conv, eps=1e-12)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = _project_simplex(a)
        # optional early stop if max_val is close to target 1/(2n)
        if cur_score <= 1.5:  # maybe break early if good enough?
            pass
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative list) that minimizes
`evaluate_sequence`.  The algorithm consists of three stages:

1. **Exploration at a modest size** – a gradient descent on the simplex
   with a true sub‑gradient (all active convolution entries) is run for a
   short time on a relatively small sequence (n≈800).  This quickly finds
   a good non‑uniform shape.

2. **Upsampling + long optimisation** – the best sequence from stage 1
   is linearly interpolated to a larger length (n≈2400).  A much longer
   projected‑gradient descent (hundreds of thousands of iterations) refines
   the distribution.  The projection onto the probability simplex keeps
   the vector non‑negative and normalised (sum = 1), which makes the
   objective scale‑invariant.

3. **Local stochastic refinement** – a cheap simulated‑annealing walk
   performs small mass‑transfer moves (`ε` from one index to another) and
   accepts improvements or occasional uphill steps.  The final result is
   normalised again and returned as a plain Python list.

The gradient uses the exact sub‑gradient of the max‑convolution:
all indices where the convolution attains (or is within a tiny tolerance
of) its maximum contribute to the gradient.  Momentum (Nesterov‑style)
and a diminishing step size (`η₀/√t`) are employed, which together give a
stable and fast convergence without the need for an external LP
oracle.

The whole procedure respects the 1000 s time budget and prints progress
messages useful for debugging.
"""

import time
import math
import numpy as np
from scipy import fftpack  # used indirectly via np.fft

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Exact sub‑gradient from all active convolution entries
# ------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Average sub‑gradient of the max‑convolution.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    # active set – a few indices around the peak (usually 1‑3)
    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # derivative w.r.t. a_i part
        g[i] += a[j]
        # derivative w.r.t. a_j part (adds the symmetric contribution)
        g[j] += a[i]

    if act.size > 0:
        g /= float(act.size)
    return g


# ------------------------------------------------------------
# Projected (Nesterov‑style) gradient descent on the simplex
# ------------------------------------------------------------
def _run_gradient_descent_full(a0: np.ndarray, n: int,
                              deadline: float, start_time: float,
                              beta: float = 0.9,
                              step0: float = 0.6,
                              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent.
    Returns the best point found and its score (2·n·max(conv)).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # exact sub‑gradient from the active set
        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated annealing local refinement
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 80000) -> tuple[np.ndarray, float]:
    """
    Tiny “move ε from i to j’’ annealing.
    Returns possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a small mass transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
