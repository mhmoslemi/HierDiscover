"""
A multi‑restart L‑BFGS‑B optimizer for the “max self‑convolution” problem.

The algorithm works on the log‑variables (mirror descent on the simplex)
and minimises a smooth approximation of the maximal entry of the
self‑convolution:
    f_τ(a) = τ·log ∑_k exp((a∗a)[k]/τ)   (τ→0 ⇒ max_k (a∗a)[k]).

For a sequence ``log_theta`` we obtain ``a = exp(log_theta)`` and
renormalise it to the required ℓ¹‑norm  √(2 n).  The gradient of
f_τ with respect to the log‑variables is computed analytically via FFTs.
A decreasing temperature schedule (τ = 0.5 → 0.01) is used, and several
random restarts (perturbations of the current best solution) are performed
until the time budget expires.  The best sequence found is returned.

The code respects the required interface – a single function ``propose_candidate``
that returns a list of non‑negative floats.
"""

import time
import math
import numpy as np

# optional use of SciPy – fall back to a simple gradient descent if unavailable
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the first (2·len(a)‑1) entries (real values).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float, nfft: int) -> np.ndarray:
    """
    Gradient of τ·log ∑ exp(conv/τ) with respect to a.
    Uses a numerically stable soft‑max.
    """
    # stable soft‑max weights
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:                      # guard against pathological case
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a (FFT based)
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) at the index k_max.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


def _smooth_max_and_grad(log_theta: np.ndarray,
                         tau: float,
                         n: int,
                         S: float,
                         nfft: int):
    """
    Returns (objective, gradient) where the objective is the smooth
    approximation τ·log ∑ exp((a∗a)[k]/τ) and the gradient is with
    respect to the log‑variables.
    """
    a = np.exp(log_theta)
    # renormalise to the required ℓ¹‑norm
    a = a * (S / a.sum())

    conv = _convolve_self_fft(a, nfft)
    max_conv = conv.max()

    # smooth max value
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    obj = max_conv + tau * np.log(w_sum)

    # gradient w.r.t. a (soft‑max)
    grad_a = _grad_softmax(a, conv, tau, nfft)

    # chain rule for log‑variables
    grad_log = grad_a * a
    return obj, grad_log


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Uses several L‑BFGS‑B optimisations on a smoothed max‑objective with
    decreasing temperature, combined with random restarts.
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # initialise sequence length and ℓ¹‑norm (the problem normalisation)
    # ------------------------------------------------------------------
    try:
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.size
    except Exception:                         # no previous construction available
        n = 2401
        init_seq = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64)

    S = math.sqrt(2.0 * n)               # Σ a_i = √(2 n) (any scaling works)
    init_seq = init_seq * (S / init_seq.sum())
    eps = 1e-12
    log_theta = np.log(np.maximum(init_seq, eps))

    # bookkeeping of the best solution
    best_a = init_seq.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial score: {best_score:.6f}")

    nfft = _next_pow2(2 * n - 1)

    # temperature schedule for the smooth max
    tau_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    max_restarts = 8
    restart_counter = 0

    while time.time() < deadline and no_improve < 5000:

        a = a * (S / a.sum())
        conv = _convolve_self_fft(a, nfft)
        k_max = int(conv.argmax())

        a_new = a_new * (S / a_new.sum())
        new_score = evaluate_sequence(a_new.tolist())

            best_a = a_new.copy()

            cur_log = new_log
            no_improve += 1
            eta *= 0.995

    # ------------------------------------------------------------------
    # final sanity projection and verification
    # ------------------------------------------------------------------
    final_seq = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {restart_counter} restarts. Best score = {final_score:.6f}")
    return final_seq.tolist()
