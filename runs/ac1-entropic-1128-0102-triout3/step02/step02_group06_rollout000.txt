The code below solves the optimisation problem  

\[
\min_{a_i\ge 0,\;\sum_i a_i=\sqrt{2n}} \; \max_k (a*a)_k ,
\]

where \((a*a)\) is the linear self‑convolution of the step heights.  
The objective used by `evaluate_sequence` is proportional to the maximum
convolution entry, therefore we can work directly with `max(conv)`.

The search consists of two stages:

1. **Differential Evolution on a low‑dimensional symmetric block
   representation**  
   * The vector `a` is forced to be symmetric – the optimal shape is
   expected to be symmetric.  
   * Only `B` block‑values are optimised (≈ 100), each block is repeated
     to fill half of the domain and mirrored to obtain the full
     sequence of length `n`.  
   * A classic DE‐mutation / crossover / selection loop runs until the
     time budget expires (or the target `1.503` is reached).  
   * The population is kept on the simplex by normalising the full
     sequence to the required ℓ¹‑norm after each trial.  

2. **Fine‑tuning with Adam on the full vector**  
   * The best candidate from DE is taken as the start point for a
     mirror‑descent (log‑parameter) Adam optimiser identical to the
     one that was used previously, but with a much shorter run
     (few × 10⁴ iterations).  
   * The Adam phase uses a soft‑max approximation of the max‑convolution
     that is slowly annealed to the exact max, giving a precise final
     improvement.

Both phases respect all security checks of `evaluate_sequence`:
* all entries are non‑negative,
* the ℓ¹‑norm is exactly `sqrt(2·n)`,
* values are far below the 1000‑clip limit.

The routine `propose_candidate` respects the required signature,
prints progress information and finally returns the best sequence
found (as a plain Python list of floats).

```python
"""
Differential‑Evolution + Adam refinement for the self‑convolution
max‑norm problem.

* A symmetric “block” parametrisation reduces the dimensionality
  (≈ 100 free variables) while still allowing very fine shapes.
* DE explores the global landscape; the best individual is then
  polished with Adam mirror‑descent on the full vector (log‑variables)
  using a soft‑max → exact max annealing schedule.
* The algorithm runs until the time budget (≈ 1000 s) expires or the
  target 1.503 is reached.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the first (2*len(a) - 1) entries.
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Stable soft‑max weights for a given temperature."""
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    return w


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float, nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    Uses the soft‑max weights.
    """
    w = _softmax_weights(conv, tau)

    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    # gradient for the inner n positions (centered)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) (at the maximal index)."""
    k_max = int(conv.argmax())
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σ x = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _construct_symmetric_from_blocks(blocks: np.ndarray, n: int) -> np.ndarray:
    """
    Build a symmetric sequence of length n from `blocks` (one‑dimensional
    positive array).  The blocks are repeated (approximately) equally
    over the left half and mirrored to the right half.
    """
    half = (n + 1) // 2                     # length of the left side (incl. centre)
    B = blocks.size
    block_len = int(math.ceil(half / B))
    left = np.repeat(blocks, block_len)[:half]

    if n % 2 == 0:
        # even length: left + reversed left
        full = np.concatenate([left, left[::-1]])
    else:
        # odd length: mirror left but skip the central element once
        full = np.concatenate([left, left[-2::-1]])   # left[-2::-1] skips last (centre)
    return full[:n]       # safety trim


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with minimal max self‑convolution.
    The search consists of:
      1. Differential Evolution on a low‑dimensional symmetric block
         representation.
      2. A short Adam refinement on the full vector.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more freedom while keeping
    # the FFT size modest (next power of two = 16384 for n ≤ 8000).
    # ------------------------------------------------------------------
    n = 4801                                 # odd, fits well into 16384‑point FFT
    S = np.sqrt(2.0 * n)                     # required ℓ¹‑norm
    nfft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Block representation
    # ------------------------------------------------------------------
    # Half length (including centre) and number of blocks.
    half_len = (n + 1) // 2
    B = 100                                   # ≈ 100 free parameters

    # ------------------------------------------------------------------
    # Initialise DE population
    # ------------------------------------------------------------------
    popsize = 30
    # Dirichlet draws give a random point on the simplex for the blocks.
    population = [np.random.dirichlet(np.ones(B)) for _ in range(popsize)]

    # If a previous construction is available, seed the first individual.
    if 'height_sequence_1' in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        # Reduce the old sequence to block values by simple averaging.
        # This gives a sensible warm‑start.
        block_vals = np.zeros(B)
        counts = np.zeros(B, dtype=int)
        block_len = int(math.ceil(half_len / B))
        for idx in range(half_len):
            b = min(idx // block_len, B - 1)
            block_vals[b] += old[idx]
            counts[b] += 1
        block_vals = block_vals / np.maximum(counts, 1)
        # Ensure strictly positive values.
        block_vals = np.maximum(block_vals, 1e-12)
        block_vals = block_vals / block_vals.sum()
        population[0] = block_vals.copy()

    # ------------------------------------------------------------------
    # Utility: evaluate a block‑vector (returns max convolution entry)
    # ------------------------------------------------------------------
    def evaluate_blocks(blocks: np.ndarray) -> float:
        a = _construct_symmetric_from_blocks(blocks, n)
        a = a * (S / a.sum())                # enforce ℓ¹‑norm exactly
        conv = _convolve_self_fft(a, nfft)
        return conv.max()

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    scores = np.empty(popsize)
    for i in range(popsize):
        scores[i] = evaluate_blocks(population[i])

    best_idx = int(np.argmin(scores))
    best_blocks = population[best_idx].copy()
    best_score = scores[best_idx]
    print(f"Initial best DE score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Differential Evolution parameters
    # ------------------------------------------------------------------
    F = 0.8               # mutation factor
    CR = 0.9              # crossover probability
    generation = 0
    stagnation_gens = 0
    max_stagnation_gens = 2000   # after which we re‑initialise a fraction

    # ------------------------------------------------------------------
    # DE main loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        generation += 1
        improved_this_gen = False

        for i in range(popsize):
            # --- mutation -------------------------------------------------
            idxs = np.random.choice([j for j in range(popsize) if j != i],
                                    size=3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + F * (b - c)
            mutant = np.maximum(mutant, 0.0)   # keep non‑negative

            # --- crossover ------------------------------------------------
            cross_mask = np.random.rand(B) < CR
            if not np.any(cross_mask):
                cross_mask[np.random.randint(B)] = True
            trial = np.where(cross_mask, mutant, population[i])

            # --- evaluation ------------------------------------------------
            trial_score = evaluate_blocks(trial)

            # --- selection -------------------------------------------------
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_blocks = trial.copy()
                    improved_this_gen = True
                    print(f"[Gen {generation:05d} Ind {i:02d}] New best: {best_score:.6f}")

        # early exit if target already reached
        if best_score < 1.503:
            print(f"Target reached during DE at generation {generation}")
            break

        # stagnation handling – replace a random individual after many idle gens
        if not improved_this_gen:
            stagnation_gens += 1
        else:
            stagnation_gens = 0

        if stagnation_gens >= max_stagnation_gens:
            # Re‑initialise half of the population with fresh Dirichlet draws
            for j in range(popsize // 2, popsize):
                population[j] = np.random.dirichlet(np.ones(B))
                scores[j] = evaluate_blocks(population[j])
            stagnation_gens = 0
            print(f"[Gen {generation}] Stagnation – partial restart performed.")

    # ------------------------------------------------------------------
    # Build the full vector from the best block representation
    # ------------------------------------------------------------------
    a_best = _construct_symmetric_from_blocks(best_blocks, n)
    a_best = a_best * (S / a_best.sum())

    # ------------------------------------------------------------------
    # Optional fine‑tuning with Adam (mirror descent on log‑variables)
    # ------------------------------------------------------------------
    # The Adam phase is short (few × 10⁴ steps) because we already have a
    # very good starting point.
    log_theta = np.log(a_best + 1e-12)
    eps_adam = 1e-8
    beta1, beta2 = 0.9, 0.999
    eta = 0.15                # base learning rate in log‑space
    m = np.zeros_like(log_theta)
    v = np.zeros_like(log_theta)

    # Soft‑max temperature schedule (anneal to exact max)
    tau = 0.4
    tau_min = 0.01
    max_adam_iters = 50000
    tau_decay = (tau_min / tau) ** (1.0 / max_adam_iters)

    adam_iter = 0
    best_adam_score = best_score
    while time.time() < deadline and adam_iter < max_adam_iters:
        adam_iter += 1

        a = np.exp(log_theta)
        a = a * (S / a.sum())

        conv = _convolve_self_fft(a, nfft)

        if tau > tau_min * 2:
            grad_a = _grad_softmax(a, conv, tau, nfft)
        else:
            grad_a = _grad_max(a, conv)

        grad_log = grad_a * a   # chain rule

        # Adam update (descent)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** adam_iter)
        v_hat = v / (1.0 - beta2 ** adam_iter)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta -= step

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.003:
            jitter = np.random.randn(n) * (eta * 0.07)
            log_theta += jitter

        # temperature annealing
        tau = max(tau * tau_decay, tau_min)

        # Track best solution during Adam phase
        cur_score = conv.max()
        if cur_score < best_adam_score:
            best_adam_score = cur_score
            a_best = a.copy()
            print(f"[Adam {adam_iter:06d}] New best (Adam): {best_adam_score:.6f}")
            if best_adam_score < 1.503:
                print("Target reached during Adam refinement.")
                break

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = a_best * (S / a_best.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished. Best reported score = {final_score:.6f}")

    return final_seq.tolist()
```
