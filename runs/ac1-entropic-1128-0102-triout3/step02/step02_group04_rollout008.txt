Propose a step‑function sequence that minimizes the provided
``evaluate_sequence`` function.

The search proceeds in three stages:

1️⃣  **Initialisation** – a handful of random “Beta‑shaped” candidates
    (a_i ∝ x^{α‑1}(1‑x)^{β‑1}) and the classical half‑binomial
    series are evaluated quickly using an FFT based self‑convolution.
    The best of these seeds becomes the starting point.

2️⃣  **Smoothed Adam optimisation** – the objective is the ℓ∞‑norm of
    the self‑convolution, i.e. ``max(conv(a,a))`` (the factor
    ``2·n / sum(a)^2`` is handled by normalising the vector to sum 1).
    Direct sub‑gradients are very sparse, so we replace the hard max by
    a differentiable log‑sum‑exp surrogate

        f_β(a) = (1/β)·log∑_k exp(β·conv_k(a))

    with an increasing “temperature’’ β.  The gradient of the surrogate
    is a weighted sum of the exact sub‑gradients and can be obtained
    efficiently via a second FFT‑convolution.  An Adam optimiser (with
    bias‑correction) updates the coefficients, after which they are
    projected onto the probability simplex (non‑negative, sum = 1).

3️⃣  **Escapes & restarts** – if no improvement is seen for a long
    period we restart from a fresh random Beta‑shape; occasional tiny
    Gaussian perturbations help to leave shallow plateaus.

The whole routine runs for the allotted time (``budget_s``) and finally
returns the best sequence found (as a plain Python list of ``float``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def half_binomial(n: int) -> np.ndarray:
    """
    First ``n`` coefficients of the half‑binomial series
    a_i = C(2i,i) / 4**i, normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random Beta‑type shape:
        a_i ∝ x_i^{α‑1} (1‑x_i)^{β‑1},
    with ``x_i = (i+0.5)/n``.  The parameters α,β are drawn uniformly.
    The result is normalised to sum = 1.
    """
    # reasonable range for α,β – avoids extreme spikes
    alpha = rng.uniform(0.2, 4.0)
    beta = rng.uniform(0.2, 4.0)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a += 1e-12          # avoid exact zeros
    a /= a.sum()
    return a


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Fast self‑convolution a*a using a real FFT.
    Result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


def conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Fast convolution of two real vectors using FFT.
    Length of the result is len(x)+len(y)‑1.
    """
    n = len(x) + len(y) - 1
    X = np.fft.rfft(x, n=n)
    Y = np.fft.rfft(y, n=n)
    return np.fft.irfft(X * Y, n=n)


def fast_objective(a: np.ndarray) -> float:
    """
    Fast proxy for ``evaluate_sequence`` when ``sum(a)=1``:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    return 2.0 * n * conv.max()

    beta_end = 200.0             # final temperature

    beta1 = 0.90                 # Adam momentum parameter
    beta2 = 0.999                # Adam second‑moment parameter
    eps = 1e-8                   # Adam stabiliser
    restart_thresh = 250_000     # iterations without improvement → restart
    noise_prob = 0.006           # occasional perturbation probability
    noise_scale = 5e-4           # magnitude of Gaussian noise

    # ------------------------------------------------------------------
    # 1. initialise from a few random candidates
    best_a = half_binomial(n)
    best_score = fast_objective(best_a)
