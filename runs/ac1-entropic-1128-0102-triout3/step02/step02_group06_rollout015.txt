```python
"""
Propose a step‑function sequence by multi‑scale mirror‑descent.
1. A short search on a small problem (n≈401) discovers a coarse shape.
2. The shape is linearly interpolated to the full length (n≈2401 or 5001).
3. A second Adam‑mirror‑descent phase refines the full‑scale sequence.
   – log‑variables ensure positivity;
   – a soft‑max smoothing of the max‑convolution provides a differentiable
     surrogate; the temperature is annealed to the exact max.
   – Adam adapts per‑coordinate step sizes; occasional jitter and
     random restarts (both from the best point and from scratch) avoid
     stagnation.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the first (2·len(a)‑1) entries (real values).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    Uses a numerically stable soft‑max.
    """
    # stable soft‑max weights
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the index k_max.
    The derivative of (a*a)[k] w.r.t. a_i equals 2·a_{k‑i} if the index is
    inside the support, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


# ----------------------------------------------------------------------
# core optimisation routine (Adam on log‑variables)
# ----------------------------------------------------------------------


def _optimize_on_simplex(init_seq: np.ndarray,
                         target_score: float,
                         time_budget: float,
                         n_fft: int = None) -> np.ndarray:
    """
    Refine ``init_seq`` (non‑negative, sum fixed) using Adam on log‑variables.
    Returns the best sequence found during the allotted time.
    """
    n = init_seq.size
    S = np.sqrt(2.0 * n)                     # prescribed ℓ₁‑norm
    a = init_seq.astype(np.float64)
    a = a * (S / a.sum())                    # enforce Σa = S

    eps = 1e-12
    log_theta = np.log(a + eps)

    # bookkeeping
    best_a = a.copy()
    best_score = 2.0 * n * _convolve_self_fft(a, n_fft).max() / (a.sum() ** 2)

    print(f"[init] n={n}, score={best_score:.6f}")

    # Adam state
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta = 0.15

    m = np.zeros_like(log_theta)
    v = np.zeros_like(log_theta)

    # temperature for soft‑max smoothing
    tau = 0.5
    tau_min = 0.001
    # modest exponential decay – we let the time budget dictate the exact number
    tau_decay = 0.9999

    deadline = time.time() + time_budget - 0.5

    iteration = 0
    no_improve = 0
    restart_limit = 3000     # after this many non‑improving steps we restart
    while time.time() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        # reconstruct a, enforce Σa = S
        # ------------------------------------------------------------------
        a = np.exp(log_theta)
        total = a.sum()
        if total == 0.0:                    # numerical safety
            a = np.random.dirichlet(np.full(n, 0.5)) * S
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            continue
        a = a * (S / total)

        # ------------------------------------------------------------------
        # evaluate the current objective (exact, cheap version)
        # ------------------------------------------------------------------
        conv = _convolve_self_fft(a, n_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{iteration:08d}] new best: {best_score:.6f}")
        else:
            no_improve += 1

        # early stop if we beat the target
        if best_score < target_score:
            print(f"[{iteration}] target reached: {best_score:.6f}")
            break

        # ------------------------------------------------------------------
        # gradient of the smoothed max (or exact max once τ is tiny)
        # ------------------------------------------------------------------
        if tau > tau_min * 2:
            grad_a = _grad_softmax(a, conv, tau, n_fft)
        else:
            k_max = int(conv.argmax())
            grad_a = _grad_max(a, conv, k_max)

        grad_log = grad_a * a                     # chain rule

        # ------------------------------------------------------------------
        # Adam update (descent)
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step               # descent step

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * (eta * 0.05)
            log_theta = log_theta + jitter

        # anneal temperature
        tau = max(tau * tau_decay, tau_min)

        # ------------------------------------------------------------------
        # restart if we have been stuck for a while
        # ------------------------------------------------------------------
        if no_improve >= restart_limit:
            if np.random.rand() < 0.5:
                # completely random restart
                a = np.random.dirichlet(np.full(n, 0.5)) * S

                perturb = np.random.randn(n) * (eta * 0.3)
                a = a * np.exp(perturb)
                a = a * (S / a.sum())
                print(f"[{iteration}] perturbed restart")
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # final sanity projection
    final_seq = best_a * (S / best_a.sum())
    return final_seq
```

The code below implements a two‑stage optimisation:

1. **Exploratory phase (small problem)** –  
   A modest‑size vector (`n≈401`) is optimised with Adam on the log‑variables.  
   The soft‑max temperature is annealed from 0.5 down to 0.001, after which the
   exact max‑gradient is used.  Random restarts (both from scratch and from a
   perturbed best point) keep the search from stagnating.

2. **Refinement phase (full size)** –  
   The best small‑scale shape is linearly interpolated to the target length
   (`n≈2401`).  The same Adam‑mirror‑descent routine is run again, this time
   using the remaining time budget.  The optimiser respects the simplex
   constraint (`a_i ≥ 0`, `∑a_i = √(2n)`) by the log‑parameterisation; this
   guarantees that the final sequence is admissible for `evaluate_sequence`.

The implementation is self‑contained, uses only `numpy` (and `time` for the
budget), and prints progress information.  The best sequence found before the
deadline is returned as a plain Python list.  

```python
"""
Propose a step‑function sequence by multi‑scale mirror‑descent.

Stage 1  – optimise a small problem (n≈401) to obtain a coarse shape.
Stage 2  – interpolate that shape to the full length (n≈2401) and
           refine it with a second Adam‑mirror‑descent loop.

The optimiser works on log‑variables (ensuring positivity) and projects
onto the simplex Σa = √(2·n).  A soft‑max surrogate of the max‑convolution
is used while the temperature τ is annealed; once τ is tiny the exact
sub‑gradient of the true maximum is employed.  Adam adapts per‑coordinate
step sizes, occasional jitter and random restarts avoid stagnation.
The best sequence encountered before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the first (2·len(a)‑1) entries (real values).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    Uses a numerically stable soft‑max.
    """
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the index k_max.
    The derivative of (a*a)[k] w.r.t. a_i equals 2·a_{k‑i} if the index
    lies inside the support, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


# ----------------------------------------------------------------------
# core optimiser (Adam on the simplex via log‑variables)
# ----------------------------------------------------------------------


def _optimize_on_simplex(init_seq: np.ndarray,
                         target_score: float,
                         time_budget: float,
                         n_fft: int) -> np.ndarray:
    """
    Refine `init_seq` (non‑negative, Σa = √(2·n)) using Adam on the log‑variables.
    Returns the best sequence found within `time_budget` seconds.
    """
    n = init_seq.size
    S = np.sqrt(2.0 * n)                 # prescribed ℓ₁‑norm
    a = init_seq.astype(np.float64)
    a = a * (S / a.sum())                # enforce Σa = S

    eps = 1e-12
    log_theta = np.log(a + eps)

    # best‑so‑far bookkeeping
    best_a = a.copy()
    best_score = 2.0 * n * _convolve_self_fft(a, n_fft).max() / (a.sum() ** 2)
    print(f"[init] n={n}, score={best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta = 0.15

    m = np.zeros_like(log_theta)
    v = np.zeros_like(log_theta)

    # soft‑max temperature schedule
    tau = 0.5
    tau_min = 0.001
    tau_decay = 0.9999   # gentle exponential decay

    deadline = time.time() + time_budget - 0.5

    iteration = 0
    no_improve = 0
    restart_limit = 3000   # number of iterations without improvement before restart

    while time.time() < deadline:
        iteration += 1

        # -------- reconstruct a and normalise to Σa=S --------
        a = np.exp(log_theta)
        total = a.sum()
        if total == 0.0:               # fallback against numerical collapse
            a = np.random.dirichlet(np.full(n, 0.5)) * S
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            continue
        a = a * (S / total)

        # -------- evaluate the exact objective (cheap) ----------
        conv = _convolve_self_fft(a, n_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{iteration:08d}] new best: {best_score:.6f}")
        else:
            no_improve += 1

        if best_score < target_score:
            print(f"[{iteration}] target reached: {best_score:.6f}")
            break

        # -------- gradient of the (smoothed) max ---------------
        if tau > tau_min * 2:
            grad_a = _grad_softmax(a, conv, tau, n_fft)
        else:
            k_max = int(conv.argmax())
            grad_a = _grad_max(a, conv, k_max)

        grad_log = grad_a * a                      # chain rule

        # -------- Adam update (descent) -----------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step               # descent step

        # occasional jitter to shake off plateaus
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * (eta * 0.05)
            log_theta = log_theta + jitter

        # temperature annealing
        tau = max(tau * tau_decay, tau_min)

        # -------- restart on prolonged stagnation -------------
        if no_improve >= restart_limit:
            if np.random.rand() < 0.5:
                # complete random restart
                a = np.random.dirichlet(np.full(n, 0.5)) * S
                print(f"[{iteration}] random restart")
            else:
                # perturb the current best
                a = best_a.copy()
                perturb = np.random.randn(n) * (eta * 0.3)
                a = a * np.exp(perturb)
                a = a * (S / a.sum())
                print(f"[{iteration}] perturbed restart")
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # final projection onto the simplex (numerical safety)
    final_seq = best_a * (S / best_a.sum())
    return final_seq
```
