def compute_gradient(a, conv, eps_rel):

    max_val = conv.max()

    # Use weight = np.exp((conv - max_val)/ (eps_rel*max_val+1e-12))

    tau = max(eps_rel * max_val, 1e-12) # ensure non-zero

    w = np.exp((conv - max_val) / tau)

    w_sum = w.sum()

    if w_sum == 0: w = np.ones_like(w) / w.shape[0]; else w = w / w_sum

    # Compute g via convolution with a_rev

    a_rev = a[::-1]

    # length conv for g: w length 2n-1, a_rev length n

    fft_len = 1 << ((len(w) + len(a) - 1).bit_length()) # forward size

    W = np.fft.rfft(w, n=fft_len)

    A_rev = np.fft.rfft(a_rev, n=fft_len)

    conv2 = np.fft.irfft(W * A_rev, n=fft_len)[: len(w) + len(a) - 1] # length 3n-2

    n = len(a)

    g = 2 * conv2[n-1 : n-1 + n]  # slice

    return g

"""
Multi‑start, accelerated soft‑max projected gradient descent.

The evaluation function is scale‑invariant, so we keep the coefficients on the
probability simplex (sum = 1, a_i ≥ 0).  For a given length `n` we minimise

    score = 2·n·max(conv(a,a))

where `conv` is the linear auto‑convolution.  The objective is the maximum of a
family of quadratic forms; we approximate its sub‑gradient by a soft‑max over
the convolution entries.  An Euclidean projection onto the simplex (Duchi et
al.) keeps feasibility.  Momentum (Nesterov style) and a decreasing
temperature for the soft‑max give a smoother descent than the simple
sub‑gradient used previously.

The optimiser works in three phases:
1. several diverse initial shapes (uniform, triangular, Hann window, a
   previously found construction and a random Dirichlet) are refined for a
   modest number of iterations;
2. the best shape is progressively up‑sampled (≈1.5× larger) up to a maximum
   length, each size receiving a time‑budget proportional to its cost;
3. a final long refinement finishes when the overall time budget (≈1000 s)
   expires.

The routine returns the best sequence it discovered as plain Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Score for a sequence that already sums to 1 (2·n·max(conv))."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _refine_sequence(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    step0: float = 0.6,
    eps_start: float = 0.05,
    eps_end: float = 0.001,
    beta: float = 0.9,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Perform accelerated soft‑max projected gradient descent on a fixed length.

    Returns (final_a, best_seen_a, best_seen_score).
    """
    n = a.shape[0]
    # pre‑compute FFT lengths
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())  # length of w⊛a_rev
    v = np.zeros_like(a)  # momentum term

    best_a = a.copy()
    best_score = _score_of_seq(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution and current score -----
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights over the convolution entries -----
        eps_rel = eps_start + (eps_end - eps_start) * (it / max_iters)
        tau = max(eps_rel * max_val, 1e-14)  # temperature
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.shape[0])

        # ----- gradient of the smooth approximation -----
        # g_i = 2 * Σ_k w_k * a_{k-i}
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        g = 2.0 * conv2[n - 1 : n - 1 + n]

        # ----- Nesterov‑style momentum and projection -----
        step = step0 / math.sqrt(it)
        v = beta * v - step * g
        a = _project_simplex(a + v)

    return a, best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (a step function) with a low
    `evaluate_sequence` score.  The routine runs for at most ``budget_s``
    seconds and prints intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Assemble a small pool of diverse starting points (all on the simplex)
    # ------------------------------------------------------------------
    n0 = 1200
    candidates = []

    # a) previous construction, if available
    try:
        prev = np.array(height_sequence_1, dtype=float)
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0:
            prev = prev / s
            candidates.append(prev)
    except Exception:
        pass

    # b) uniform
    candidates.append(np.full(n0, 1.0 / n0))

    # c) triangular (tent) shape
    idx = np.arange(n0)
    tri = np.minimum(idx + 1, n0 - idx)
    candidates.append(tri / tri.sum())

    # d) Hann window (raised cosine, zero at the ends)
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n0 - 1)))
    candidates.append(hann / hann.sum())

    # e) a random Dirichlet point (single random start)
    candidates.append(np.random.dirichlet(np.ones(n0)))

    # ------------------------------------------------------------------
    # 2. Refine each candidate for a modest number of iterations
    # ------------------------------------------------------------------
    for i, cand in enumerate(candidates, 1):

        a = np.interp(x_new, x_old, current)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:

        else:
            a = a / s

        # ----- iteration budget for this resolution -----
        # keep total runtime balanced: larger n → fewer iterations

            step0=0.55,
            eps_start=0.04,
            eps_end=0.001,
            beta=0.9,
        )

        # keep the best shape seen at this scale
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(
                f"   Updated global best after up‑sampling: n={next_n}, "
                f"score={best_local_score:.6f}"
            )

        # prepare for next up‑sampling step

        cur_n = next_n

        if time.time() >= deadline:

    # ------------------------------------------------------------------

            f"{iters_final} iterations"

            best_seq,
            max_iters=iters_final,
            deadline=deadline,
            step0=0.45,
            eps_start=0.02,
            eps_end=0.0005,
            beta=0.9,
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"   Final improvement: score={best_local_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        # extremely unlikely fallback – small uniform vector

    # Ensure non‑negativity and normalisation (not strictly required but tidy)
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()

    print(
        f"[{time.time() - start_time:6.2f}s] Finished. Best score (2·n·max(conv)) = {best_score:.6f}"
