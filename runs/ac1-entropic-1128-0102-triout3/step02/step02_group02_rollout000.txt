"""
Propose a candidate step‑function sequence using a smooth‑max
gradient descent in log‑softmax space.

The algorithm works as follows:
1. Represent the non‑negative sequence a (length n) as a soft‑max
   of logits z, guaranteeing a_i ≥ 0 and Σ a_i = 1.
2. At each iteration compute the linear convolution c = a * a.
3. Approximate the non‑differentiable max(c) by a
   temperature‑controlled smooth‑max:
        Sα(c) = (1/α)·log Σ_k exp(α·c_k)
   Its gradient w.r.t. c is a probability vector p_k
   (soft‑max of α·c).  Using p we can obtain ∇_a Sα efficiently
   by two FFT‑based convolutions.
4. Propagate the gradient to the logits z (chain rule for the
   soft‑max) and apply the Adam optimiser.
5. The temperature α is increased slowly so the optimisation
   first explores a wide region (small α) and later concentrates
   on the true maximum (large α).
6. The procedure is repeated for several sequence lengths;
   the best sequence found (according to the official
   evaluate_sequence) is returned.

The optimisation is fully vectorised, uses only FFTs (O(n log n))
per iteration and respects the 1000 s time budget.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors a (len m) and b (len n)
    by FFT, returning a vector of length m+n‑1.
    """
    m = a.shape[0]
    n = b.shape[0]
    L = m + n - 1
    N = _next_pow2(L)
    A = np.fft.rfft(np.pad(a, (0, N - m)))
    B = np.fft.rfft(np.pad(b, (0, N - n)))
    C = A * B
    conv = np.fft.irfft(C, n=N)[:L]
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself (full length)."""
    return _conv_full_fft(a, a)


# ----------------------------------------------------------------------
# Smooth‑max utilities
# ----------------------------------------------------------------------


def _smooth_max_and_weights(conv: np.ndarray, alpha: float) -> Tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of max(conv) and its gradient weights.

    Returns:
        smooth_max – (1/α)·log Σ exp(α·conv)
        p         – soft‑max probabilities (∂smooth_max/∂conv)
    """
    # shift for numerical stability
    shift = conv.max()
    exp_term = np.exp(alpha * (conv - shift))
    sum_exp = exp_term.sum()
    smooth_max = (shift + np.log(sum_exp) / alpha)
    p = exp_term / sum_exp
    return smooth_max, p


# ----------------------------------------------------------------------
# Gradient wrt a using the soft‑max weights
# ----------------------------------------------------------------------


def _grad_wrt_a(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute ∂/∂a Σ_k p_k·(a*a)_k efficiently using two FFT‑based convolutions.
    a has length n, p has length 2n‑1.
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of p

    # convolution a * p  (size n+L‑1 = 3n‑2)
    conv_a_p = _conv_full_fft(a, p)

    # convolution a * p_rev
    p_rev = p[::-1]
    conv_a_p_rev = _conv_full_fft(a, p_rev)

    # gradient assembly
    #   g_i = (a * p)[i] + (a * p_rev)[(2n‑2)‑i]
    rev_idx = (2 * n - 2) - np.arange(n)
    grad = conv_a_p[:n] + conv_a_p_rev[rev_idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser in logit space
# ----------------------------------------------------------------------


def _adam_update(
    z: np.ndarray,
    grad_z: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Classic Adam update for logits."""
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, m, v


# ----------------------------------------------------------------------
# Helper for turning logits into a probability vector (soft‑max)
# ----------------------------------------------------------------------


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()

            # recompute a from updated logits
            a = _logits_to_a(z)

            # ------------------------------------------------------------------
            # 6) occasional jitter to escape flat regions
            # ------------------------------------------------------------------
            if t_iter % 2500 == 0:
                jitter = np.random.randn(n) * 1e-3
                z += jitter

            # ------------------------------------------------------------------
            # 7) temperature schedule – slowly make the smooth‑max sharper
            # ------------------------------------------------------------------
            if t_iter % alpha_inc_interval == 0 and alpha < alpha_max:
                alpha = min(alpha * alpha_mul, alpha_max)

            # ------------------------------------------------------------------
            # 8) early stopping for this length if progress stalls
            # ------------------------------------------------------------------
            # (We simply rely on the global deadline; no extra checks.)

    # ----------------------------------------------------------------------
    # Fallback – uniform distribution if nothing was discovered.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check (official evaluation)
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"Final best score (official) {final_score:.6f} "
              f"(length {len(best_seq)})")
    except Exception:
        print("Warning: final evaluation failed.")

    return best_seq
