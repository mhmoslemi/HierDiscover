"""
Optimises a non‑negative step function for the evaluation metric
defined in the task.  The optimiser works in three stages:

1. **Adam mirror‑descent on the simplex** – the variables are kept in
   log‑space so the positivity constraint is automatic, and after each
   step the vector is renormalised to the required ℓ₁‑norm
   `√(2·n)`.  A soft‑max smoothing of the `max` of the self‑convolution
   provides a differentiable surrogate; the temperature is slowly
   annealed to the exact max.

2. **LP‑boost** – every few thousand Adam steps we solve a linear programme

        maximise    Σ g_j
        subject to   (a * g)[k] ≤ M   for all k,
                     g_j ≥ 0,

   where `a` is the current best sequence and `M = max(a*a)`.  The LP
   finds a feasible direction that carries as much ℓ₁‑mass as possible
   while respecting the current peak bound.  The direction is mixed
   with the current iterate (small step size) which empirically pushes
   the peak down.

3. **Short local refinement** – a few random “swap‑or‑shift’’ moves are
   tried and kept if they improve the peak.

The routine is bounded by the supplied time budget (default 1000 s)
and always returns the best sequence found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a via FFT.
    Returns the first 2·len(a)-1 entries (real numbers).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    Uses a numerically stable soft‑max.
    """
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the index k_max.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------

                    v.fill(0.0)
                    print(f"[{iteration:08d}] swap improved to {best_score:.6f}")
                    break   # accept first improvement

    # ------------------------------------------------------------------
    # Final sanity check & return
    # ------------------------------------------------------------------
    final_seq = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations – final score = {final_score:.6f}")
    return final_seq.tolist()
