def conv_fft(x, y=None):

    n = len(x)
    m = len(y)

    X = np.fft.rfft(x, n=n_fft)

    conv = np.fft.irfft(X*Y, n=n_fft)

"""
A smooth‑max gradient descent with Adam on the simplex.

The objective to minimise (up to an irrelevant positive scale) is
    max_k (a * a)[k]     with  a_i ≥ 0 , Σ a_i = 1 .
A soft‑max with an increasing temperature λ approximates the
maximum, giving a differentiable surrogate.  The gradient of the
soft‑max surrogate can be written as a convolution, so each iteration
requires only two FFT‑based convolutions (self‑convolution of a and a
convolution of the soft‑max weights with a reversed a).  After a
gradient step an Euclidean projection onto the probability simplex is
performed.  Adam’s adaptive steps and a linear λ‑schedule (from a
very smooth surrogate to a sharp max) drive the search.  Occasionally
tiny random noise is added and a restart is performed if the best
value has not improved for a long time.

The routine runs for the time budget supplied by the caller (default
≈ 950 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of the half‑binomial series
        a_i = C(2i,i) / 4**i   (i = 0,…,n‑1)
    Normalised to have sum = 1 (i.e. a lies on the simplex).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0 , Σ x_i = 1}
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0

    if not np.any(cond):
        # all entries become equal
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray, y: np.ndarray | None = None) -> np.ndarray:
    """
    Fast (full) convolution of two real vectors using the FFT.
    If y is None, computes the auto‑convolution x*x.
    Returns a vector of length len(x)+len(y)-1 (or 2*len(x)-1 for auto‑conv).
    """
    if y is None:
        y = x
    n = x.shape[0]
    m = y.shape[0]
    L = n + m - 1
    # next power of two for efficient FFT
    N = 1 << (L - 1).bit_length()
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    prod = X * Y
    conv_full = np.fft.irfft(prod, n=N)
    return conv_full[:L]

    minimises the supplied ``evaluate_sequence`` metric.

        * Initialise with the classic half‑binomial series.
        * Repeatedly perform a smooth‑max Adam step on the simplex.
        * λ (soft‑max temperature) is increased from a very smooth
          surrogate to a sharp approximation of the true maximum.
        * Occasionally inject tiny random noise.
        * If no improvement is seen for many iterations, restart from
          a random Dirichlet draw.
        * Keep the best sequence ever seen and return it.
    """
    rng = np.random.default_rng(seed)

    # problem size – a larger n gives a finer discretisation but costs
    # more per iteration.  3000 is a good compromise.
    n = 3000

    # Adam hyper‑parameters

    lr_start = 0.12          # initial (unscaled) learning rate
    lr_end = 0.004           # final learning rate

    lam_start = 0.3          # start almost flat (≈ average)
    lam_max = 5000.0         # end very sharp → true max
    lam_exp = 2.0            # exponent for the schedule (slow‑start)

    # miscellaneous

    noise_scale = 5e-5       # magnitude of the perturbation
    stagnation_limit = 150_000   # iterations without improvement before restart
    max_iter = 10_000_000    # safety bound (real stopping is time‑based)

    best_score = first_score
    print(f"[init] n={n}, initial score={first_score:.6f}")

        m = beta1 * m + (1.0 - beta1) * grad

        lr = lr_start + (lr_end - lr_start) * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step

        # occasional tiny random perturbation (helps escape plateaux)
        if rng.random() < noise_prob:
            a_candidate += (rng.random(n) - 0.5) * noise_scale

        # 5) project back onto the simplex
        a_candidate = project_onto_simplex(a_candidate)

        # ----------------------------------------------------------
        # 6) accept the candidate unconditionally (the surrogate is
        #    smooth enough that a descent step is usually beneficial)
        a = a_candidate

        # ----------------------------------------------------------

            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            last_improve_it = it
            print(f"[iter {it:08d}] restart – new random start")

        # ----------------------------------------------------------
        it += 1

    # final evaluation (uses the official routine, which also applies the
    # +0/1000 clipping and the sum‑check)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
