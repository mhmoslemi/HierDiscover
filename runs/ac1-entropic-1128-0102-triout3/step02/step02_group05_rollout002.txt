"""
Propose a low‑value step‑function sequence.

The core idea is a **projected stochastic gradient descent** on a smooth
approximation of the objective  f(a)=max_k (a∗a)_k .  Instead of using the
hard max we consider the top‑K convolution entries (or all entries within
a small relative gap) and form a sub‑gradient that accounts for all of them.
The update is performed on the probability simplex (∑a_i = 1, a_i ≥ 0) using
the Euclidean projection `proj_simplex`.  A few random “mass‑transfer”
moves are interleaved to escape plateaus.  The algorithm starts from a
hand‑crafted family of power‑law / triangular shapes (including the
previous global `height_sequence_1` if present) and keeps track of the best
sequence seen so far.  It runs until the supplied time budget (≈1000 s) is
exhausted and finally returns the best list of floats.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` without the safety checks.
    The input is assumed to be a non‑negative 1‑D array.
    The function normalises nothing (the objective is scale‑invariant)
    and returns the same value as `evaluate_sequence`.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, ∑ x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def build_linear_descending(n: int) -> np.ndarray:
    """Linear decreasing shape a_i = (n-i) (i=0..n-1) normalised."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def gradient_multi_active(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient that aggregates the contributions of all
    active convolution indices in `top_idxs`.
    For each k ∈ top_idxs the contribution is
        g_i ← g_i + 2·a_{k-i}   (when 0 ≤ k-i < n)
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in top_idxs:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i = np.arange(low, high + 1)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation value.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # 1. Choose a problem size.
    n = 3000                       # length of the sequence

    # --------------------------------------------------------------
    # 2. Gather a small pool of analytically motivated initialisations.
    candidates = []

    # power‑law family
    exponents = [0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60]
    offsets   = [0.0, 0.1, 0.5, 1.0, 2.0]
    for p in exponents:
        for off in offsets:
            candidates.append(build_power_law(n, p, off))

    # linear decreasing shape
    candidates.append(build_linear_descending(n))

    # uniform (as a baseline)
    candidates.append(np.full(n, 1.0 / n))

    # If a previous best sequence is available, use its first n entries.
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                seq = np.zeros(n, dtype=np.float64)
                seq[:prev.size] = prev
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0:
                seq /= seq.sum()
                candidates.append(seq)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (according to the fast score).
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best initial value: {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop.
    a = best_seq.copy()                # current iterate (always on simplex)
    best_seq = a.copy()
    best_val = fast_score(a)

    # hyper‑parameters
    eta0 = 0.8                         # initial step size
    K = 5                              # number of active indices to aggregate
    delta_start = 0.06                 # initial relative gap for active set
    delta_end   = 0.01                 # final relative gap
    random_move_prob = 0.025           # prob. of a stochastic mass transfer
    random_move_frac = 0.20            # fraction of source mass to move
    uphill_accept = 0.001              # prob. to accept a non‑improving move
    it = 0
    last_improve = time.time()

    # pre‑compute FFT size for self‑convolution
    conv_size = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # ----------------------------------------------------------
        # (a) compute self‑convolution and its maximum
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_conv = conv.max()

        # (b) determine the set of active indices
        #    we use a decreasing relative gap (annealing)
        delta = max(delta_end,
                    delta_start * (1.0 - it / 200000.0))   # linear anneal
        mask = conv >= max_conv * (1.0 - delta)
        active_idxs = np.flatnonzero(mask)
        if active_idxs.size > K:
            # keep the K largest ones
            top_K = np.argpartition(-conv[active_idxs], K)[:K]
            active_idxs = active_idxs[top_K]

        # (c) compute the aggregated sub‑gradient
        grad = gradient_multi_active(a, active_idxs)

        # (d) step size (square‑root decay)
        eta = eta0 / np.sqrt(it + 1)

        # (e) projected gradient update
        a_new = proj_simplex(a - eta * grad)

        # (f) evaluate candidate
        cur_val = fast_score(a_new)

        # (g) accept if improvement, otherwise occasional uphill
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
        else:
            if np.random.rand() < uphill_accept:
                a = a_new

        # ----------------------------------------------------------
        # (h) random "hill‑climbing" mass transfer
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta_transfer = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta_transfer
                cand[j] += delta_transfer
                cand = proj_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                    last_improve = time.time()
                elif np.random.rand() < uphill_accept:
                    a = cand

        # ----------------------------------------------------------
        # reporting
        if it % 5000 == 0:
            print(f"[iter {it:7d}] current {cur_val:.6f}  best {best_val:.6f}")

        # (i) stagnation handling – if no improvement for a while, re‑seed
        if time.time() - last_improve > 30.0:
            # re‑initialise from a slight random perturbation of the best
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(0.9 * best_seq + 0.1 * noise)
            last_improve = time.time()

        it += 1

    print(f"[final] best value after {it} iterations: {best_val:.6f}")
    # The output does not need to be normalised – the external checker does it.
    return best_seq.tolist()
