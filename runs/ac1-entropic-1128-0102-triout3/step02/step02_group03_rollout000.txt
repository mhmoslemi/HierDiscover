def _smooth_gradient(a, beta, nfft):
    conv = _convolve_self_fft(a, nfft) # full conv length 2n-1
    # compute weights safely:
    mv = conv.max()
    # w = exp(beta * (conv - mv))  # subtract max to avoid overflow
    w = np.exp(beta * (conv - mv))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # compute conv of w with reversed a:
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[:2*a.size - 1]
    # central part:
    n = a.size
    grad = 2.0 * grad_full[n-1:n-1+n]
    return grad, conv

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # set n: maybe from height_sequence_1 length else 4000
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 4000
        a0 = None

    S = np.sqrt(2.0 * n)
    nfft = _next_pow2(2 * n - 1)

    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        # start with Dirichlet over n
        alpha = np.full(n, 0.5)  # moderate concentration
        a = np.random.dirichlet(alpha) * S

    best_a = a.copy()
    conv = _convolve_self_fft(a, nfft)
    best_score = conv.max()
    best_iter = 0
    temperature = 1.0
    temp_decay = 0.9999
    min_temp = 1e-6

    beta_start = 0.5
    beta_end = 200.0
    beta_iters = 20000
    # step base
    step_base = 0.5 * S / n  # typical scale for additive; for multiplicative we use small.

    iteration = 0
    no_improve = 0
    while time.time() < deadline:
        # compute smoothed gradient and conv
        beta = beta_start + (beta_end - beta_start) * min(1.0, iteration / beta_iters)
        grad, conv = _smooth_gradient(a, beta, nfft)
        cur_max = conv.max()
        cur_score = cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            best_iter = iteration
            print(f"[{iteration:06d}] New best: {best_score:.6f} (beta={beta:.2f})")
            no_improve = 0
        else:
            no_improve += 1

        # early exit if target
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration}, best {best_score:.6f}")
            break

        # decide move type
        r = np.random.rand()
        if r < 0.8:
            # gradient step (multiplicative)
            eta = step_base / np.sqrt(iteration + 1)  # diminishing
            # compute candidate
            exp_factor = np.exp(-eta * grad)
            a_candidate = a * exp_factor
            a_candidate = a_candidate * (S / a_candidate.sum())
        elif r < 0.95:
            # random mass transfer
            # choose source weighted by a
            src_probs = a / a.sum()
            i = np.random.choice(n, p=src_probs)
            j = np.random.choice(n)
            while j == i:
                j = np.random.choice(n)
            delta = 0.05 * a[i]  # move 5% of source
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = a_candidate * (S / a_candidate.sum())
        else:
            # restart
            alpha = np.full(n, 0.4)
            a_candidate = np.random.dirichlet(alpha) * S
            print(f"[{iteration:06d}] Restart performed.")
        
        # evaluate candidate exact objective
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        cand_score = cand_max

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
        else:
            # maybe jitter
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * step_base * 0.2
                a = a + jitter

"""
Hybrid smoothed‑gradient / simulated‑annealing search.

The ℓ₁‑norm of the step function is fixed to √(2n) (the scale at which
evaluate_sequence equals the maximal entry of the self‑convolution).
We minimise this maximal entry.

* A smoothed version of the max,  fβ(a)= (1/β)·log Σk exp(β·(a*a)[k]),
  is differentiable. Its gradient is
        ∇fβ(a) = 2·conv( w , a_rev )
  where   w[k] ∝ exp(β·(a*a)[k])  (a probability vector),
  and a_rev is a reversed copy of a.
  Using this gradient we perform a multiplicative (exponential) mirror‑
  descent step and then re‑project onto the ℓ₁‑simplex.

* To escape local minima we mix three stochastic moves:
  – the gradient step (≈80 % of the time),
  – a random “mass‑transfer’’ between two coordinates,
  – a full Dirichlet restart.
  Acceptance follows a Metropolis rule with a slowly cooling temperature.

* After the time budget is exhausted a short pure sub‑gradient polishing
  phase is applied to the best candidate found.

The routine returns the best sequence (list of floats) discovered before
the deadline.  All helper functions are top‑level as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a via FFT.
    Returns the full (2·len(a)‑1) result (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_gradient(a: np.ndarray, beta: float, nfft: int):
    """
    Compute the gradient of the smoothed max functional
        fβ(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
    and also return the exact self‑convolution (a*a).

    Returns
        grad : ndarray, shape (n,)
            Gradient ∇fβ(a).
        conv : ndarray, shape (2n‑1,)
            Exact linear convolution a*a (used for evaluating the true
            objective).
    """
    n = a.size
    # exact convolution
    conv = _convolve_self_fft(a, nfft)          # length 2n‑1

    # softmax weights, numerically stable
    m = conv.max()
    w = np.exp(beta * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2·conv(w, a_rev)   (central part of length n)
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * n - 1]

    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # central slice
    return grad, conv


def _grad_max_subgrad(a: np.ndarray, mask: np.ndarray, nfft: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the current active set.
    mask is a binary array (length 2n‑1) with ones at indices where the
    convolution attains its current maximum.
    Returns a vector of length n.
    """
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros_like(a)

    w = w / w_sum                     # normalise to a probability vector
    a_rev = a[::-1]

    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    evaluate_sequence.  The algorithm combines a smoothed‑gradient
    (mirror‑descent) phase with occasional random mass‑transfer moves,
    temperature‑based Metropolis acceptance and full Dirichlet restarts.
    After the time budget expires a short sub‑gradient polishing phase is
    run on the best solution found.

    Returns
        best_seq : list of floats (the sequence with the smallest
                   evaluate_sequence value observed).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 4000                # a larger size gives a better theoretical bound
        a0 = None

    S = np.sqrt(2.0 * n)        # ℓ₁‑norm we keep fixed
    nfft = _next_pow2(2 * n - 1)

    # initialise a
    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        a = np.random.dirichlet(np.full(n, 0.5)) * S   # moderate concentration

    # bookkeeping
    conv = _convolve_self_fft(a, nfft)
    best_score = conv.max()
    best_a = a.copy()
    best_iter = 0

    # ------------------------------------------------------------------
    # hyper‑parameters
    # ------------------------------------------------------------------
    temperature = 1.0
    temp_decay = 0.9997
    min_temp = 1e-6

    # smoothing schedule (β from 0.5 up to 200)
    beta_start = 0.5
    beta_end = 200.0
    beta_ramp_iters = 30000   # after this β is fixed at beta_end

    # step‑size for mirror descent (multiplicative update)
    base_eta = 0.5 * S / n     # typical magnitude of a_i

    # stagnation handling
    no_improve = 0
    max_no_improve = 5000

    iteration = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # current smoothing parameter
        beta = beta_start + (beta_end - beta_start) * min(1.0,
                                                          iteration / beta_ramp_iters)

        # smoothed gradient and exact convolution
        grad, conv = _smooth_gradient(a, beta, nfft)
        cur_max = conv.max()
        cur_score = cur_max                     # because Σa = √(2n)

        # update global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            best_iter = iteration
            no_improve = 0
            print(f"[{iteration:07d}] New best: {best_score:.6f} (β={beta:.1f})")
        else:
            no_improve += 1

        # early exit if target reached
        if best_score < 1.503:
            print(f"Target reached at iter {iteration}: {best_score:.6f}")
            break

        # --------------------------------------------------------------
        # choose a stochastic move
        # --------------------------------------------------------------
        r = np.random.rand()
        if r < 0.80:
            # ---- mirror‑descent (exponential) step -----------------
            eta = base_eta / np.sqrt(iteration + 1)   # diminishing
            # multiplicative update, then re‑normalise to ℓ₁‑simplex
            factor = np.exp(-eta * grad)
            a_candidate = a * factor
            a_candidate = a_candidate * (S / a_candidate.sum())
        elif r < 0.95:
            # ---- random mass‑transfer --------------------------------
            if a.sum() > 0:
                src_probs = a / a.sum()
                i = np.random.choice(n, p=src_probs)
            else:
                i = np.random.randint(n)
            j = i
            while j == i:
                j = np.random.randint(n)
            delta = 0.07 * a[i]          # move 7 % of the source mass
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = a_candidate * (S / a_candidate.sum())
        else:
            # ---- full Dirichlet restart ------------------------------
            a_candidate = np.random.dirichlet(np.full(n, 0.4)) * S
            print(f"[{iteration:07d}] Full restart performed.")

        # --------------------------------------------------------------
        # evaluate candidate (exact max‑convolution)
        # --------------------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        cand_score = cand_max

        # Metropolis acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
        else:
            # occasional jitter to shake shallow plateaux
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * base_eta * 0.3
                a = a + jitter
                a = np.maximum(a, 0.0)
                a = a * (S / a.sum())

        # temperature decay
        temperature = max(temperature * temp_decay, min_temp)

        # stagnation restart
        if no_improve >= max_no_improve:
            a = np.random.dirichlet(np.full(n, 0.4)) * S
            no_improve = 0
            print(f"[{iteration:07d}] Stagnation restart (no improvement).")

        iteration += 1

    # ------------------------------------------------------------------
    # Final polishing with exact sub‑gradient descent (few hundred iters)
    # ------------------------------------------------------------------
    polishing_iters = 800
    for _ in range(polishing_iters):
        conv = _convolve_self_fft(best_a, nfft)
        cur_max = conv.max()
        mask = (conv >= cur_max - 1e-12).astype(float)
        subgrad = _grad_max_subgrad(best_a, mask, nfft)
        grad_norm = np.linalg.norm(subgrad)
        if grad_norm == 0:
            break
        eta = base_eta * 0.2 / np.sqrt(_ + 1)          # small diminishing step
        a_candidate = best_a - eta * subgrad
        a_candidate = _project_to_simplex(a_candidate, S)
        cand_max = _convolve_self_fft(a_candidate, nfft).max()
        if cand_max < cur_max:
            best_a = a_candidate
            best_score = cand_max

    # ------------------------------------------------------------------
    # final verification and return
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)   # ensure feasibility
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations (best iter {best_iter}). "
          f"Best final score = {final_score:.6f}")
    return best_a.tolist()
