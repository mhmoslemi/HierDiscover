def _project_simplex(v):
    # Project onto simplex {x >=0, sum x =1}
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = np.where(u - cssv/np.arange(1, n+1) > 0)[0]
    if len(rho_arr) == 0:
        # all zero?
        return np.full_like(v, 1.0/n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function sequence.

Algorithm
---------
1. Work with a probability vector a (a_i ≥ 0, Σ a_i = 1).  The evaluation
   function is scale‑invariant, therefore we can optimise on the simplex
   and later return the raw vector (the official score will be the same).

2. Phase A – variance flattening.
   The convolution (a*a) is the distribution of X+X where X∼a.
   Its average value is 1/(2n‑1).  We minimise the variance
   Σ_k ( (a*a)_k – 1/(2n‑1) )².
   The gradient of this smooth objective is
        g = 2·corr( (a*a) – μ , a )
   where corr denotes the linear correlation (valid part) and μ = 1/(2n‑1).
   After a gradient step we project back onto the probability simplex.
   This phase quickly makes the auto‑correlation almost flat.

3. Phase B – max‑subgradient polishing.
   Starting from the best vector of Phase A we optimise the true objective
   max_k (a*a)_k.  We use Adam on the soft‑max parameters (logits) and
   a sub‑gradient coming from the single maximal convolution entry.
   Occasional random jitter helps to escape shallow local minima.

4. Multiple lengths and random restarts.
   The search is performed for several lengths (800‑2400) and a few
   random seeds.  The best sequence found over the whole run is returned.

The whole routine respects the 1000 s budget and prints intermediate
progress for monitoring.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the full convolution of length 2·n‑1.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    conv_full = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv_full


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (sums to 1)."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()

    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]               # largest index satisfying the condition
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_variance(a: np.ndarray) -> np.ndarray:
    """
    Gradient of the variance of the convolution a*a.
    Returns a vector of length n (same shape as a).
    """
    n = a.shape[0]
    conv = _conv_fft(a)                     # length 2n‑1
    L = conv.shape[0]                       # = 2n‑1
    mu = 1.0 / L                            # average value of conv (sum a = 1)
    diff = conv - mu                         # shape (L,)

    # correlation of diff with a (valid part) -> length n
    N = _next_pow2(L + n)                     # zero‑pad length
    diff_pad = np.zeros(N, dtype=np.float64)
    a_pad = np.zeros(N, dtype=np.float64)
    diff_pad[:L] = diff
    a_pad[:n] = a
    D = np.fft.rfft(diff_pad)
    A = np.fft.rfft(a_pad)
    corr_full = np.fft.irfft(D * np.conj(A), n=N)
    corr = corr_full[:n]                     # first n entries = valid correlation
    grad = 2.0 * corr
    return grad


# ----------------------------------------------------------------------
# Phase A – variance flattening
# ----------------------------------------------------------------------


def _phase_variance(a0: np.ndarray,
                    n: int,
                    iters: int,
                    lr0: float,
                    deadline: float) -> Tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on the variance of a*a.
    Returns the best vector (still on the simplex) and its evaluate_sequence score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    for it in range(iters):
        if time.time() >= deadline:
            break

        # gradient of variance
        grad = _gradient_variance(a)

        # learning‑rate schedule (slow decay)
        lr = lr0 / (1.0 + it * 1e-5)

        # gradient step + projection onto the simplex
        a = a - lr * grad
        a = _project_simplex(a)

        # occasional evaluation (every 200 steps)
        if (it % 200) == 0:
            # evaluate the true objective (scale does not matter)
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[variance] n={n} it={it:05d} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase B – max‑subgradient polishing (Adam on logits)
# ----------------------------------------------------------------------


def _adam_update(z: np.ndarray,
                 grad_z: np.ndarray,
                 m: np.ndarray,
                 v: np.ndarray,
                 t: int,
                 lr: float,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> Tuple[np.ndarray, Tuple[np.ndarray, np.ndarray]]:
    """Classic Adam update (returns new z and updated (m, v))."""
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, (m, v)


def _phase_max_subgradient(a_start: np.ndarray,
                           n: int,
                           iters: int,
                           lr0: float,
                           deadline: float) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits of a, using the sub‑gradient
    of the maximum convolution entry.
    """
    eps = 1e-12
    z = np.log(a_start + eps)          # initialise logits
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_a = a_start.copy()
    best_score = float("inf")

    for _ in range(iters):
        if time.time() >= deadline:
            break

        a = _softmax(z)                 # a stays on the simplex
        conv = _conv_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max    # identical to evaluate_sequence for sum=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[max] n={n} t={t:05d} new best {best_score:.6f}")

        # sub‑gradient from the (single) maximal entry
        k = int(np.argmax(conv))
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad_a = np.zeros_like(a)
        grad_a[i_range] = 2.0 * a[k - i_range]

        # chain rule to obtain gradient w.r.t. logits
        dot = a.dot(grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update (learning‑rate decays slowly)
        t += 1
        lr_t = lr0 / (1.0 + t * 1e-5)
        z, (m, v) = _adam_update(z, grad_z, m, v, t, lr_t)

        # occasional random jitter to avoid stagnation
        if t % 1500 == 0:
            z += np.random.randn(n) * 1e-3

    return best_a, best_score
