import time
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _convolve_self_fft(a, nfft):
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:2*a.size - 1]
    return conv

def _project_to_simplex(v, target_sum):
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _grad_max_subgrad(a, mask, nfft):
    # mask: same length as full convolution (2*n-1)
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros_like(a)
    w = w / w_sum
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[:2*a.size - 1]
    n = a.size
    grad = 2.0 * grad_full[n-1 : n-1 + n]
    return grad

                     budget_s: int = 1000,
                     **kwargs) -> list[float]:

    The algorithm keeps the ℓ₁‑norm fixed to √(2 n) so that the objective
    reduces to the maximum entry of the self‑convolution.  Starting from a
    previously found good sequence (if the global ``height_sequence_1`` is
    present) we perform iterative updates:

    * a smooth approximation of ``max(conv)`` via a temperature/β parameter;
    * the gradient of that smooth objective is obtained analytically with
      FFT‑based convolution;
    * an accelerated (heavy‑ball) update with momentum is applied;
    * after each step the iterate is projected back onto the simplex
      {x ≥ 0, Σx = √(2 n)}.
    * β is increased slowly so the algorithm gradually focuses on the true
      max; occasional random restarts avoid being trapped in poor local minima.

    The loop runs until the time budget expires (or the target bound 1.503 is
    reached) and returns the best sequence encountered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------- problem size -------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=np.float64)
        n = a0.shape[0]
        # guarantee that a0 respects non‑negativity
        a0 = np.maximum(0.0, a0)
    else:
        n = 2000
        a0 = None

    S = np.sqrt(2.0 * n)                     # ℓ₁‑norm we keep fixed
    nfft = _next_pow2(2 * n - 1)

    # ------------------- initial point -----------------
    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        a = np.full(n, S / n, dtype=np.float64)   # uniform start

    # momentum buffer
    v = np.zeros_like(a)

    # best-so‑far bookkeeping
    conv = _convolve_self_fft(a, nfft)
    best_score = float('inf')
    best_a = a.copy()

    # hyper‑parameters
    step_base = 0.6 * S / n       # base step size
    beta = 0.3                    # smoothing inverse temperature
    beta_max = 80.0
    beta_growth = 1.0015
    mu = 0.9                      # momentum coefficient
    eta_decay = 0.9995
    eta = step_base
    target = 1.503
    iteration = 0
    no_improve = 0
    max_no_improve = 3000

    while time.time() < deadline:
        # -------------------------------------------------
        # compute convolution and objective for current a
        # -------------------------------------------------
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = cur_max          # because sum == S

        # record best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{iteration:06d}] New best: {best_score:.6f} (max={cur_max:.6f})")
        else:
            no_improve += 1

        if best_score < target:
            print(f"Target {target:.3f} reached at iteration {iteration}, best={best_score:.6f}")
            break

        # -------------------------------------------------
        # compute smoothed gradient of max(conv)
        # -------------------------------------------------
        # softmax weights
        shift = conv - cur_max                # max conv -> 0
        # exponentiate; avoid overflow
        w = np.exp(beta * shift)
        w_sum = w.sum()
        if w_sum == 0:
            # numerical issue, fall back to hard mask
            mask = (conv >= cur_max - 1e-12).astype(float)
        else:
            mask = w / w_sum

        grad = _grad_max_subgrad(a, mask, nfft)

        # -------------------------------------------------
        # momentum update and projection
        # -------------------------------------------------
        v = mu * v - eta * grad
        a_candidate = _project_to_simplex(a + v, S)

        # -------------------------------------------------
        # evaluate candidate
        # -------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_score = cand_conv.max()

        # accept if improvement, otherwise keep old (still try momentum)
        if cand_score < cur_score:
            a = a_candidate
            # keep momentum v (already updated)
        else:
            # reject step: discard momentum and optionally try a small jitter
            v = np.zeros_like(a)   # reset momentum
            # occasional jitter to escape plateaux
            if np.random.rand() < 0.02:
                jitter = np.random.randn(n) * (step_base * 0.3)
                a = _project_to_simplex(a + jitter, S)

        # -------------------------------------------------
        # schedule updates
        # -------------------------------------------------
        beta = min(beta_max, beta * beta_growth)
        eta = max(eta * eta_decay, 1e-8)

        # restart if stagnating
        if no_improve >= max_no_improve:
            # random Dirichlet restart
            alpha = np.full(n, 0.3)
            a = np.random.dirichlet(alpha) * S
            v = np.zeros_like(a)
            no_improve = 0
            print(f"[{iteration:06d}] Stagnation restart.")
        elif iteration % 25000 == 0 and iteration > 0:

            a = np.random.dirichlet(alpha) * S
            v = np.zeros_like(a)
            print(f"[{iteration:06d}] Periodic full restart.")

        iteration += 1

    # -------------------------------------------------
    # final verification
    # -------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()

# Helper functions (top‑level, no closures)

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full (2·len(a)‑1) result as real numbers.
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_max_subgrad(a: np.ndarray, mask: np.ndarray, nfft: int) -> np.ndarray:
    """
    Gradient of the (soft) max of the self‑convolution at a.
    ``mask`` must have length 2·len(a)‑1 and is interpreted as a
    probability distribution (its entries are normalised internally).
    Returns a vector of length len(a).
    """
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros_like(a)

    w = w / w_sum
    a_rev = a[::-1]

    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    # slice that corresponds to the gradient entries

    return grad
