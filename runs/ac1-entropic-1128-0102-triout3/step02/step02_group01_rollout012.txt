"""
propose_candidate
-----------------

Search for a low‑score step‑function using an exponentiated‑gradient (EG)
sub‑gradient method on the simplex.  The algorithm:

1. Build a few diverse start vectors (previous solution, uniform,
   triangular, sine‑shaped, a few random draws) of length ≈1200.
2. For each start run a short EG optimisation (≈3 k iterations) and keep
   the best.
3. Repeatedly up‑sample the current best shape (linear interpolation) by
   ≈1.5× up to a maximum length (≈6000).  After each up‑sampling run a
   longer EG optimisation (iteration count scales inversely with the
   length so the total run‑time stays within the budget).
4. Finish with a final refinement on the best length found.

The EG update keeps the vector strictly non‑negative and automatically
enforces the simplex constraint by normalising after each step.
The sub‑gradient is obtained from all convolution entries that are
within a small relative tolerance of the current maximum, which makes
the optimisation smoother than the pure “single‑max” sub‑gradient.
A simple Adam‑style momentum is not required – the exponential update
provides a natural mirror‑descent on the probability simplex.

The whole routine runs until the supplied time budget is exhausted
(≈1000 s) and returns the best sequence found.
"""

import time

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Score for a sequence that already sums to 1: 2·n·max(conv)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _eg_optimize(a_init: np.ndarray,
                 deadline: float,
                 max_iters: int,
                 eps_rel: float = 0.01,
                 step0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation of a single sequence.

    Parameters
    ----------
    a_init : np.ndarray
        Starting vector (already normalised to sum = 1, non‑negative).
    deadline : float
        Absolute time (seconds since epoch) after which the routine must stop.
    max_iters : int
        Upper bound on the number of iterations.
    eps_rel : float
        Relative tolerance for “near‑max” convolution indices.
    step0 : float
        Initial step‑size (decays as 1/√t).

    Returns
    -------
    best_a : np.ndarray
        The best sequence seen.
    best_score : float
        Corresponding score.
    """
    n = a_init.shape[0]
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_of_seq(best_a)

    # FFT length – power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())
    # Pre‑allocate masks to avoid re‑allocation inside the loop
    mask = np.empty(2 * n - 1, dtype=float)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- auto‑convolution -------------------------------------------------
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]

        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Improved (n={n}) "
                  f"iter={it} score={cur_score:.6f}")

        # --- set of “active” convolution indices ------------------------------
        eps = max_val * eps_rel
        K = np.where(conv >= max_val - eps)[0]

        if K.size == 0:
            # No active indices – should not happen, but skip this iteration.
            continue

        # build binary mask for the active set
        mask.fill(0.0)
        mask[K] = 1.0

        # --- sub‑gradient (via convolution of a with the mask) ----------------
        Rf = np.fft.rfft(mask, n=fft_len)
        grad_full = np.fft.irfft(Af * Rf, n=fft_len)[:n]   # first n entries
        g = 2.0 * grad_full

        # --- exponentiated‑gradient update ------------------------------------
        step = step0 / math.sqrt(it)          # classic 1/√t schedule
        # protect against overflow/underflow in the exponent
        expo = np.exp(-np.clip(step * g, -50.0, 50.0))
        a = a * expo
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score

            break

    # ------------------------------------------------------------------
    # 4. Final short refinement on the best length found
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        a_ref, score_ref = _eg_optimize(best_seq, deadline,
                                        max_iters=20000,
                                        eps_rel=0.01,
                                        step0=0.6)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref
            print(f"Final refinement improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence (plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        # Very unlikely fallback – uniform 100‑element sequence
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
