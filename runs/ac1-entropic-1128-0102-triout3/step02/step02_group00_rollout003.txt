"""
Multi‑scale exact‑active projected gradient descent + annealing.

The evaluation function is scale‑invariant, so we optimise on the simplex
∑a_i = 1, a_i ≥ 0.  For a fixed length n we minimise

        2·n·max_k (a * a)_k

where (a*a) is the linear convolution of a with itself.
The gradient of the ∞‑norm is the convex hull of the gradients of the
active (maximal) convolution entries; we therefore use the *exact*
active set in every iteration (all indices whose value equals the current
maximum up to a tiny tolerance).  The update is projected onto the simplex
using the O(n) algorithm of Duchi et al.  After a gradient phase a cheap
simulated‑annealing refinement is applied.

A schedule of increasing lengths is used; the best sequence found so far
is up‑sampled and serves as the initial point for the next, larger, problem.
The whole search respects the supplied time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient from a set of active convolution indices
# ----------------------------------------------------------------------
def _subgradient_from_active(a: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    """
    Average sub‑gradient of the max‑convolution over the active indices.
    For each active k we add the gradient of (a*a)_k, i.e.
        ∂/∂a_i (a*a)_k = 2·a_{k-i}  (when 0 ≤ i ≤ n‑1 and 0 ≤ k-i ≤ n‑1)
    The result is averaged over all active indices.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] += 2.0 * a[j]
    if act_idx.size > 0:
        g /= act_idx.size
    return g


# ----------------------------------------------------------------------
# Exact‑active projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_gradient_descent_exact(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    beta: float = 0.9,
    step0: float = 0.5,
    max_iter: int = 200000,
    tol: float = 1e-12,
):
    """
    Projected (momentum) gradient descent where the sub‑gradient is computed
    from *all* maximal convolution entries (the exact sub‑gradient of the
    infinity norm).  Returns the best sequence seen and its score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] GD improve: n={n}, iter={it}, score={cur_score:.6f}"
            )

        # indices where the convolution attains the current maximum
        # tolerance chosen relative to the magnitude of max_val
        eps = max(tol, tol * max_val)
        active = np.where(conv >= max_val - eps)[0]

        # average sub‑gradient over the active set
        grad = _subgradient_from_active(a, active)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # gradient step + projection onto the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 30000,
):
    """
    Simple annealing: repeatedly choose two indices i≠j and move a small
    fraction of a[i] to a[j].  Accept moves that improve the score or,
    with probability exp(-Δ/T), worsen it.  The best sequence encountered
    during the run is returned.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.1  # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from an old length to a new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up_sum = up.sum()
    if up_sum > 0.0:
        up /= up_sum
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm runs for at most `budget_s` seconds,
    combining exact‑active projected gradient descent, multi‑scale up‑sampling
    and a final annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional external starter (e.g. a previously found construction)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Length schedule – geometric growth, shuffled to avoid bias if time ends early
    # ------------------------------------------------------------------
    lengths = [400, 800, 1200, 2400, 4800, 7200, 10000, 15000]
    np.random.shuffle(lengths)

    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise a for this length
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            # Random mixture of uniform and a simple “U‑shaped’’ start
            if np.random.rand() < 0.5:
                a0 = np.random.rand(n)
            else:
                idx = np.arange(n)
                a0 = 1.0 / np.sqrt((idx + 1) * (n - idx))
                a0 = np.where(np.isfinite(a0), a0, 0.0)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # Allocate a slice of the remaining budget to this length
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # give the current length at most 20 % of the remaining time, but not less than 8 s
        slice_deadline = time.time() + max(8.0, 0.2 * remaining)

        # ------------------------------------------------------------------
        # Gradient descent phase (exact active set)
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_gradient_descent_exact(
            a0,
            n,
            slice_deadline,
            start_time,
            beta=0.9,
            step0=0.6,
            max_iter=200000,
            tol=1e-12,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after GD: n={n}, score={score_opt:.6f}"
            )

        # ------------------------------------------------------------------
        # Simulated‑annealing refinement (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, 0.04 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=30000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n}, score={score_ref:.6f}"
                )

        # ------------------------------------------------------------------
        # Store the best sequence for the next (larger) length
        # ------------------------------------------------------------------
        global_a = best_seq.copy()

    # ----------------------------------------------------------------------
    # Final refinement on the largest length if we still have time left
    # ----------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        final_n = max(lengths)
        if final_n > best_seq.shape[0]:
            a0 = _upsample(best_seq, final_n)
        else:
            a0 = best_seq.copy()
        a_opt, score_opt = _run_gradient_descent_exact(
            a0,
            final_n,
            deadline,
            start_time,
            beta=0.9,
            step0=0.6,
            max_iter=300000,
            tol=1e-12,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] final GD improve: n={final_n}, score={score_opt:.6f}"
            )

    # ----------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # Final normalisation and conversion to plain Python list
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
