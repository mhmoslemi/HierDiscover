def compute_conv_and_max(a):
    n = len(a)
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n -1]
    return conv, conv.max()

"""
Propose a candidate step‑function using an exponentiated‑gradient (mirror‑descent)
scheme on the probability simplex, together with progressive up‑sampling and a
short stochastic hill‑climbing phase.

The objective (with sum(a)=1) is 2·n·max(conv(a,a)).  The algorithm
* starts from the best known sequence (``height_sequence_1``) or a random simplex
  point,
* repeatedly up‑samples the current shape (≈1.45× larger, up to 5000 points),
* runs an exponentiated‑gradient loop where the sub‑gradient is taken from the
  few (≤8) largest convolution entries,
* keeps the best sequence seen,
* finally performs a cheap random mass‑transfer local search until the time
  budget is exhausted.

All updates keep the vector strictly positive, so the simplex projection is
unnecessary – we simply renormalise after each multiplicative update.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """
    Evaluate the objective for a sequence that already sums to 1.
    Returns 2 * n * max(conv(a,a)).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray):
    """Return full auto‑convolution and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 8):
    """
    Determine a small set of convolution indices to use for the sub‑gradient.
    First collect all entries within rel_eps * max_val of the maximum.
    If there are fewer than max_add of them, also add the globally top indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.shape[0])
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Compute the sub‑gradient of max(conv) using the active set K.
    For each active convolution index k, the contribution to component i is
    2 * a_{k-i} (if the index is in range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        # accumulate 2 * a[j_idx] into g[i_idx]
        g[i_idx] += 2.0 * a[j_idx]
    return g

    Return a list of non‑negative heights (a step function) with a low
    `evaluate_sequence` score.  The routine runs for at most ``budget_s``
    seconds and prints intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise from a previously found construction, if available.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # may be defined globally
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s == 0.0:
            raise ValueError
        init = init / s
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"Initialised from height_sequence_1: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        # fall back to a random simplex point
        n0 = 1500
        init = np.random.rand(n0)
        init = init / init.sum()
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"Random initialisation: n={n0}, score={best_score:.6f}")

    current = best_seq.copy()
    cur_n = current.shape[0]

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling with exponentiated‑gradient optimisation.
    # ------------------------------------------------------------------
    MAX_LEN = 5000               # upper bound on the length we will try
    eta0 = 0.5                    # initial step size for exponentiated gradient
    rel_eps = 0.001               # tolerance for “near‑max” convolution entries
    max_add = 8                  # guarantee at most this many active indices

    while time.time() < deadline and cur_n < MAX_LEN:

            log_a -= np.max(log_a)
            a = np.exp(log_a)
            # enforce a tiny floor to avoid exact zeros later on
            a[a < 1e-15] = 1e-15
            a = a / a.sum()

        # prepare for next size
        current = a.copy()
        cur_n = next_n

    # ------------------------------------------------------------------
    # 3. Final short stochastic local search (mass transfers).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        it_local = 0
        # limit the number of random moves to a modest amount
        max_local_iters = 12000
        while time.time() < deadline and it_local < max_local_iters:
            it_local += 1
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # propose moving a small fraction of mass from i to j
            eps = np.random.rand() * 0.04          # up to 4 % of source mass

            if delta <= 0:

            a_new[i] -= delta

                print(f"[{time.time() - start_time:6.2f}s] Random improve: n={n}, "
                      f"it={it_local}, score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # convert to plain Python floats
    return [float(x) for x in best_seq]
