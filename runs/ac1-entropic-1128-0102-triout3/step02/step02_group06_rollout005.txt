"""
Entropy‑augmented Adam for the self‑convolution max‑norm problem.

The algorithm works on log‑variables (mirror descent on the simplex) and
optimises a smooth surrogate of the max entry of the self‑convolution.
Two regularisers are added:

* an entropy term  λ_ent·∑ a_i·log a_i that keeps the mass spread,
* a sum‑of‑squares term λ_var·∑ (a*a)_k² whose gradient 4·conv((a*a),a_rev)
  pushes the whole convolution towards a flat shape.

Both regularisers are linearly annealed to zero while a soft‑max temperature
τ is exponentially cooled.  After the main Adam loop a short stochastic
local‑search (mass‑transfer) refines the best solution found.
The best sequence discovered before the 1000 s budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# linear‑algebra helpers (no closures, no lambda)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the first (2·len(a)‑1) entries (real values).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _linear_conv_fft(x: np.ndarray, y: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the first (len(x)+len(y)-1) entries.
    """
    X = np.fft.rfft(x, nfft)
    Y = np.fft.rfft(y, nfft)
    prod = np.fft.irfft(X * Y, n=nfft)
    return prod[: len(x) + len(y) - 1]


def _grad_softmax(a: np.ndarray,
                  conv: np.ndarray,
                  tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a (soft‑max of the convolution).
    """
    c = conv.max()
    w = np.exp((conv - c) / tau)               # stable soft‑max numerator
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of the weights w with the reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the index k_max.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


# ----------------------------------------------------------------------
# main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The method combines Adam on log‑variables,
    a soft‑max smoothing of the max‑convolution, entropy and
    sum‑of‑squares regularisation, temperature annealing and a final
    stochastic refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ----- initialise sequence ------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.size
    else:
        # a slightly larger default length than the previous run
        n = 2801
        init_seq = None

    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    if init_seq is not None:
        a = init_seq.copy()
        a = a * (S / a.sum())
    else:
        a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64) * S

    # ----- Adam on log‑variables ---------------------------------------------
    eps = 1e-12
    log_theta = np.log(a + eps)

    # bookkeeping
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    # FFT planning
    nfft = _next_pow2(2 * n - 1)         # for a*a and soft‑max gradient
    nfft2 = _next_pow2(3 * n - 2)       # for the sum‑of‑squares gradient

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta = 0.18                           # base step size in log‑space
    m = np.zeros_like(log_theta)
    v = np.zeros_like(log_theta)

    # temperature schedule for the soft‑max approximation
    tau = 1.5
    tau_min = 0.005
    # estimate of total iterations (rough, will be capped by the budget)
    est_iters = 400_000
    tau_decay = (tau_min / tau) ** (1.0 / est_iters)

    # regularisation schedules (linear decay)
    lam_ent_initial = 0.08
    lam_var_initial = 0.03

    iteration = 0
    no_improve = 0
    restart_limit = 10_000

    while time.time() < deadline:
        iteration += 1

        # ----- recover a from log‑variables and enforce Σa = S ---------------
        a = np.exp(log_theta)
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64) * S
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            continue
        a = a * (S / sum_a)

        # ----- current objective ------------------------------------------------
        conv = _convolve_self_fft(a, nfft)          # self‑convolution
        cur_max = conv.max()
        cur_score = cur_max                         # equal to evaluate_sequence after normalisation

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{iteration:08d}] New best: {best_score:.6f}")
        else:
            no_improve += 1

        if best_score < 1.503:                     # target reached
            print(f"Target reached at iteration {iteration}")
            break

        # ----- gradient of the max (soft‑max or exact) -------------------------
        if tau > tau_min * 2:
            grad_conv = _grad_softmax(a, conv, tau, nfft)
        else:
            k_max = int(conv.argmax())
            grad_conv = _grad_max(a, conv, k_max)

        # ----- entropy regulariser --------------------------------------------
        progress = (time.time() - start_time) / (budget_s - 0.5)
        lam_ent = lam_ent_initial * max(0.0, 1.0 - progress)
        grad_entropy = np.log(a + eps) + 1.0        # d/da  (a·log a)

        # ----- sum‑of‑squares regulariser --------------------------------------
        lam_var = lam_var_initial * max(0.0, 1.0 - progress)
        if lam_var > 0.0:
            # gradient of Σ_k (conv_k)² = 4 * conv * a (via convolution)
            conv2 = _linear_conv_fft(conv, a[::-1], nfft2)   # length 3n‑2
            grad_sq = 4.0 * conv2[n - 1 : n - 1 + n]         # slice n entries
        else:
            grad_sq = np.zeros_like(a)

        # ----- total gradient w.r.t. a -----------------------------------------
        grad_a = grad_conv + lam_ent * grad_entropy + lam_var * grad_sq

        # ----- convert to gradient w.r.t. log‑variables -------------------------
        grad_log = grad_a * a

        # ----- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.006:
            log_theta += np.random.randn(n) * (eta * 0.04)

        # ----- anneal temperature ---------------------------------------------
        tau = max(tau * tau_decay, tau_min)

        # ----- restart on stagnation -------------------------------------------
        if no_improve >= restart_limit:
            a = best_seq.copy()
            perturb = np.random.randn(n) * (eta * 0.25)
            a = a * np.exp(perturb)                 # multiplicative shake‑up
            a = a * (S / a.sum())
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{iteration:08d}] Restarted after stagnation.")

    # ----------------------------------------------------------------------
    # final stochastic refinement (small mass transfers)
    # ----------------------------------------------------------------------
    a = best_seq.copy()
    for _ in range(5000):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue
        delta = a[i] * np.random.rand() * 0.12   # move up to 12 % of a[i]
        a[i] -= delta
        a[j] += delta
        conv = _convolve_self_fft(a, nfft)
        score = conv.max()
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta

    # ----------------------------------------------------------------------
    # final sanity check and return
    # ----------------------------------------------------------------------
    final_seq = best_seq * (S / best_seq.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
