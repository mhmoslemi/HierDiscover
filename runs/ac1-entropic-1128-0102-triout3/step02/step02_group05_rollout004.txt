def gradient_softmax(w, a):
    n = a.size
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    fft_w = np.fft.rfft(w, size)
    fft_a_rev = np.fft.rfft(a_rev, size)
    conv_full = np.fft.irfft(fft_w * fft_a_rev, size)
    # conv_full length = size; but we only need range n-1 : n-1+n
    g = 2.0 * conv_full[n-1: n-1+n]
    return g

"""
Proposed search for a low‑value step‑function.

The algorithm combines several ideas that differ from the previous
sub‑gradient descent:

*   A *soft‑max* approximation of the maximum of the self‑convolution is
    used most of the time.  This yields a smooth gradient that simultaneously
    reduces all large convolution values instead of moving only away from the
    single active index.

*   The gradient is obtained by a second FFT‑based convolution (the weights
    from the soft‑max are convolved with the reversed sequence).  This keeps
    the per‑iteration cost O(n log n).

*   An Adam‑style adaptive step is applied, followed by an Euclidean projection
    onto the probability simplex (non‑negative entries that sum to 1).

*   Occasionally a pure sub‑gradient step (active‑index) and a random
    “mass‑transfer” hill‑climb move are performed to escape plateaus.

*   The routine starts from a collection of analytical (power‑law) sequences,
    random Dirichlet draws and, if present, the previously found global
    `height_sequence_1`.  The best among them is taken as the initial point.

The search runs for the allotted time budget (default 1000 s) and returns the
best sequence discovered as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (identical to `evaluate_sequence`
    but without the safety checks).  The vector is assumed to be non‑negative.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry (active index k)."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Power‑law family a_i = (i+offset)^(-p) normalised to ℓ¹‑norm 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def gradient_softmax(conv: np.ndarray, a: np.ndarray, T: float,
                    grad_fft_size: int) -> np.ndarray:
    """
    Gradient of the soft‑max smoothed max‑convolution.
    w_i = exp((conv_i - max(conv))/T)  (normalised to sum 1)
    ∇ = 2 * (w ⋆ a) where ⋆ denotes cross‑correlation.
    The cross‑correlation is computed via FFT using the reversed sequence.
    """
    # normalise the weights
    w = np.exp((conv - conv.max()) / T)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # compute cross‑correlation w ⋆ a   (length n result)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, grad_fft_size)
    fft_a_rev = np.fft.rfft(a_rev, grad_fft_size)
    full = np.fft.irfft(fft_w * fft_a_rev, grad_fft_size)
    n = a.size
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.
    Returns the best sequence (list of non‑negative floats) found within
    the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Problem size (adjustable)
    n = 3000

    # ------------------------------------------------------------------
    # 2. Build a pool of diverse initial points
    init_pool = []

    # power‑law families
    for p in [0.3, 0.4, 0.5, 0.6, 0.8]:
        for off in [0.0, 0.1, 0.5, 1.0, 2.0]:
            init_pool.append(build_initial_sequence(n, p, off))

    # random Dirichlet draws
    for _ in range(12):

    # previous global solution, if available
    if "height_sequence_1" in globals():
        try:

                cand[: prev.size] = prev
            cand = np.maximum(cand, 0.0)
            if cand.sum() > 0:
                cand /= cand.sum()
                init_pool.append(cand)
        except Exception:

    # ------------------------------------------------------------------
