def _grad_softmax(a, conv, alpha, nfft):
    # conv is a np.ndarray of length 2*n-1
    # compute weights w based on softmax
    c = alpha * conv
    maxc = np.max(c)
    w = np.exp(c - maxc)   # shift for numerical stability
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # compute gradient = 2 * (w * a_rev) convolution
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]
    n = a.size
    grad = 2.0 * grad_full[n-1 : n-1 + n]
    return grad
```

    # a: ndarray length n
    # conv: ndarray length 2n-1 = precomputed convolution a*a
    # alpha: scalar
    # nfft: length for FFT (power of two >= 2n-1)

"""
Hybrid soft‑max gradient descent for the step‑function problem.

The algorithm works on the ℓ₁‑simplex {x ≥ 0, Σx = √(2·n)} so that the
objective 2·n·max(conv)/(Σx)² reduces to the maximal entry of the
self‑convolution.  At each iteration we

1. compute the full self‑convolution a*a by FFT,
2. form a smooth approximation of the max using a soft‑max (log‑sum‑exp)
   with a temperature α that slowly increases,
3. obtain the corresponding gradient (which is a convolution of the
   soft‑max weights with the reversed vector),
4. take a projected gradient step (step size η decays like 1/√t),
5. occasionally perform a random “mass‑transfer’’ move to increase
   exploration,
6. accept the new point if it improves the true objective, otherwise
   with a Metropolis probability that cools down,
7. restart from a random Dirichlet draw after long stagnation.

The use of the soft‑max gives a dense sub‑gradient that points
away from all large convolution entries, improving over the binary
active‑set sub‑gradient of the reference implementation.

The routine runs until the supplied time budget (≈1000 s) expires and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full (2·len(a)‑1) result.
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      alpha: float,
                      nfft: int,
                      A_cached: np.ndarray = None) -> np.ndarray:
    """
    Gradient of the smoothed max
        f(a) = (log Σ_k exp(α·(a*a)[k])) / α
    with respect to a.
    The result is a vector of length len(a).

    Parameters
    ----------
    a : ndarray, shape (n,)
        Current sequence (non‑negative, sum = √(2·n)).
    conv : ndarray, shape (2·n‑1,)
        Full self‑convolution of a (already computed).
    alpha : float
        Soft‑max temperature (larger → closer to true max).
    nfft : int
        Length of the FFT (power of two ≥ 2·n‑1).
    A_cached : ndarray, optional
        rfft(a) already computed; if None it will be computed.

    Returns
    -------
    grad : ndarray, shape (n,)
        The gradient ∇f(a).
    """
    n = a.shape[0]

    # soft‑max weights (normalized)
    c = alpha * conv
    c_max = np.max(c)
    w = np.exp(c - c_max)          # shift to avoid overflow
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # FFT of a (if not supplied) and of its reversal via a phase factor
    if A_cached is None:
        A = np.fft.rfft(a, n=nfft)
    else:
        A = A_cached
    # a_rev[k] = a[n-1-k]; its DFT is conj(A) times a linear phase
    freqs = np.arange(A.shape[0])
    twiddle = np.exp(-2j * np.pi * freqs * (n - 1) / nfft)
    A_rev = np.conj(A) * twiddle

    # FFT of the soft‑max weights
    W = np.fft.rfft(w, n=nfft)

    # convolution W * A_rev gives the (unnormalised) gradient
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * n - 1]

    # extract the central n entries (indices n‑1 … 2n‑2)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with low max self‑convolution.
    Returns the best sequence (list of floats) found before ``budget_s``
    seconds elapse.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2401                     # a reasonable default
        a0 = None

    S = np.sqrt(2.0 * n)            # target ℓ₁‑norm
    nfft = _next_pow2(2 * n - 1)    # FFT length (power of two)

    # ------------------------------------------------------------------
    # Initialise a
    # ------------------------------------------------------------------
    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        a = np.full(n, S / n, dtype=np.float64)   # uniform start

    # bookkeeping
    conv = _conv_self_fft(a, nfft)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (a.sum() ** 2)   # equals cur_max because Σa = S
    best_a = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    step0 = 0.5 * S                     # initial step magnitude (will decay)
    temperature = 0.12                  # Metropolis temperature
    temp_decay = 0.9995
    min_temp = 1e-6

    p_random = 0.25                     # probability of a random mass‑transfer move
    eps_transfer = 0.07                 # fraction of source mass when doing a transfer

    alpha_start = 1.0
    alpha_end = 45.0
    alpha = alpha_start
    alpha_inc = (alpha_end - alpha_start) / 200000.0   # slow increase

    stagnation_limit = 3000
    stagnation_counter = 0
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---------- compute current convolution & objective ----------
        A = np.fft.rfft(a, n=nfft)                     # cache FFT of a
        conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # ---------- keep global best ----------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{it:07d}] New best: {best_score:.6f} (max={cur_max:.6f})")
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        # ---------- stop early if we reach the target ----------
        if best_score < 1.503:
            print(f"Target bound reached after {it} iterations (best={best_score:.6f})")
            break

        # -----------------------------------------------------------
        # Propose a new candidate
        # -----------------------------------------------------------
        if np.random.rand() < p_random:
            # ----- simple random mass‑transfer move -----
            if a.sum() > 0:
                src_probs = a / a.sum()
                i = np.random.choice(n, p=src_probs)
            else:
                i = np.random.randint(n)
            j = i
            while j == i:
                j = np.random.randint(n)

            delta = eps_transfer * a[i]
            if delta <= 0:
                delta = eps_transfer * S / n

            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = _project_to_simplex(a_candidate, S)

        else:
            # ----- soft‑max gradient step -----
            # increase soft‑max temperature slowly
            alpha = min(alpha_end, alpha + alpha_inc)

            grad = _softmax_gradient(a, conv, alpha, nfft, A_cached=A)
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:
                eta = step0 / np.sqrt(it + 1.0)          # diminishing step
                step = eta * grad / grad_norm
                a_candidate = _project_to_simplex(a - step, S)
            else:
                a_candidate = a.copy()

        # ----- evaluate candidate (full objective) -----
        A_cand = np.fft.rfft(a_candidate, n=nfft)
        conv_cand = np.fft.irfft(A_cand * A_cand, n=nfft)[: 2 * n - 1]
        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max / (a_candidate.sum() ** 2)

        # ----- acceptance rule (greedy + Metropolis) -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
            cur_max = cand_max
        else:
            # occasional tiny jitter to escape plateaus
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * (step0 * 0.3)
                a = _project_to_simplex(a + jitter, S)

        # ----- temperature decay -----
        temperature = max(temperature * temp_decay, min_temp)

        # ----- stagnation handling -----
        if stagnation_counter >= stagnation_limit:
            # full restart from a random Dirichlet draw
            alpha = alpha_start
            a = np.random.dirichlet(np.full(n, 0.4)) * S
            stagnation_counter = 0
            print(f"[{it:07d}] Stagnation restart performed.")

        it += 1

    # ------------------------------------------------------------------
    # Final projection & verification
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {it} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()
