assistantfinal**Algorithm Overview**

The optimiser works on a probability simplex (the coefficients sum to 1 and are non‑negative).  
It repeatedly

1. computes the self‑convolution `c = a*a`,
2. builds a *soft‑max* weight `w` from `c`.  For a large temperature `τ` the weight is almost uniform, for a small `τ` it concentrates on the current peak – this gives a smooth approximation of the `max` operator,
3. computes the exact gradient of the weighted sum  
   \[
   g_i = 2\!\sum_k w_k\;a_{k-i}\;,
   \]  
   which is obtained efficiently by a single FFT‑based convolution `w * a[::-1]`,
4. updates the coefficients with an Adam‑style adaptive step,  
5. projects the result back onto the simplex (Duchi et al. projection),
6. accepts the move if it improves the score, otherwise with a Metropolis probability that slowly cools,
7. gradually lowers the temperature `τ` (sharpening the soft‑max) and the learning rate,
8. restarts from the current best solution if no improvement is seen for a long time.

The whole loop runs until the time budget (≈ 950 s) is exhausted.  
The best sequence encountered is finally checked with the supplied `evaluate_sequence` function and returned.

This approach differs from the earlier sub‑gradient method by
* using a smooth approximation of the `max` (soft‑max) – giving a more informative gradient,
* employing the Adam adaptive optimiser,
* adding an annealing schedule for both the temperature and the step size,
* using occasional random restarts to escape plateaus.

"""Adaptive soft‑max Adam optimiser for the step‑function bound.

The routine searches for a non‑negative coefficient vector a (∑a = 1) that
minimises   2·n·max(a * a) .
It uses a soft‑max weighting of the convolution to obtain a smooth
sub‑gradient, an Adam‑style adaptive step, projection onto the simplex,
and a simple annealing / restart scheme.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of the half‑binomial series,
    normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}
    (Duchi et al., 2008)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Fast self‑convolution a*a (full length 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


def conv_full_pre(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """
    Full convolution of x and y using a pre‑computed FFT length `size`
    (size must be ≥ len(x)+len(y)−1 and a power of two).
    Returns the first len(x)+len(y)−1 entries.
    """
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    n = len(x) + len(y) - 1
    return conv[:n]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    Parameters
    ----------
    seed : int
        Random seed.
    budget_s : float
        Approximate time budget (seconds).  A few seconds are kept as a
        safety margin.
    **kwargs : dict
        Ignored – kept for compatibility.

    Returns
    -------
    list[float]
        The best coefficient list found (plain Python list).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1️⃣  Initial sequence ------------------------------------------------
    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0:
            n = 2500
            a = sqrt_series(n)
        else:
            n = init.size
            a = init.copy()
            a = np.maximum(a, 0.0)
            if a.sum() == 0.0:
                a = sqrt_series(n)
    else:
        n = 2500
        a = sqrt_series(n)

    # Normalise to the probability simplex
    a = a / a.sum()

    # Pre‑compute the FFT size that will be reused for gradient convolutions
    # – w has length 2n‑1, a_rev has length n → length = 3n‑2
    grad_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

    # Initial convolution and score
    conv = conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Hyper‑parameters ------------------------------------------------
    max_iter = 10_000_000                     # safety bound
    beta1, beta2 = 0.9, 0.999                 # Adam coefficients
    eps = 1e-8                                 # Adam epsilon
    lr = 0.25                                  # initial learning rate
    lr_decay = 0.9999                          # learning‑rate decay factor
    tau = 5e-4                                 # soft‑max temperature
    tau_decay = 0.99995                        # temperature decay factor
    restart_after = 200_000                    # iterations without improvement
    last_improve = 0
    it = 0

    # Adam moment vectors
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 3️⃣  Main optimisation loop -----------------------------------------
    deadline = time.time() + budget_s - 5.0  # keep a few seconds margin

    while time.time() < deadline and it < max_iter:
        it += 1

        # ------------------------------------------------------------------
        # Soft‑max weighting of the current convolution
        # w_i ∝ exp( (c_i – max(c)) / τ )
        max_c = conv.max()
        w_exp = np.exp((conv - max_c) / tau)
        w = w_exp / w_exp.sum()                     # shape (2n‑1,)

        # Gradient of weighted sum: g = 2 * (w * a_rev)_{centered}
        a_rev = a[::-1]
        full = conv_full_pre(w, a_rev, grad_fft_len)  # length = 3n‑2
        start = n - 1
        grad = 2.0 * full[start:start + n]            # shape (n,)

        # Adam update (minimisation ⇒ move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_trial = a - step

        # Occasional random jitter to help escape plateaus
        if rng.random() < 0.001:
            a_trial += rng.normal(scale=1e-5, size=n)

        # Projection back onto the simplex (ensures non‑negativity & sum=1)
        a_trial = project_onto_simplex(a_trial)

        # ------------------------------------------------------------------
        # Evaluate the trial point (only one FFT needed – we reuse `conv_self`)
        conv_trial = conv_self(a_trial)
        trial_score = 2.0 * n * conv_trial.max()

        # Acceptance rule (deterministic improvement + Metropolis for uphill)
        if trial_score < cur_score:
            accept = True
        else:
            # Metropolis temperature linked to the soft‑max temperature
            accept = rng.random() < np.exp(-(trial_score - cur_score) / max(tau, 1e-8))

        if accept:
            a = a_trial
            conv = conv_trial
            cur_score = trial_score

            # Record the best ever seen
            if trial_score < best_score - 1e-12:
                best_score = trial_score
                best_a = a.copy()
                last_improve = it
                print(f"[iter {it:08d}] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Parameter schedules
        if it % 5000 == 0:
            tau = max(tau * tau_decay, 1e-7)
            lr = max(lr * lr_decay, 1e-4)

        # Restart if stuck for a long time
        if it - last_improve > restart_after:
            # Perturb the current best and resume from there
            a = best_a + rng.normal(scale=0.02, size=n)
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a)
            conv = conv_self(a)
            cur_score = 2.0 * n * conv.max()
            last_improve = it
            print(f"[iter {it:08d}] restart triggered – score {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final safety evaluation
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
