"""
Propose a step‑function sequence (non‑negative heights) that minimizes the
provided evaluation function.

Algorithm
---------
1. Work with a probability vector a (a_i ≥ 0, Σ a_i = 1) via a soft‑max
   parametrisation a = softmax(z).  The evaluation is then
        score = 2·n·max_k (a * a)_k,
   which is exactly the value returned by ``evaluate_sequence`` because the
   function is scale‑invariant.

2. At each iteration we compute the full linear self‑convolution
   ``c = a * a`` (FFT, O(n log n)).  Instead of using only the single
   maximal entry we form a smooth approximation of the max:
        w_k = exp(α·c_k) / Σ_j exp(α·c_j) .
   When α → ∞ the vector w concentrates on the true maximum; for small α it
   spreads the weight over many convolution entries, providing a useful
   sub‑gradient.

3. The gradient of the smooth objective
        J(a) = Σ_k w_k·c_k
   w.r.t. a is
        ∂J/∂a_i = 2· Σ_j w_{i+j}·a_j .
   This can be evaluated efficiently by one additional FFT‑based linear
   convolution of the reversed a with the weight vector w.

4. Convert the gradient w.r.t. a to the gradient w.r.t. the logits z using
   the chain rule for the soft‑max, and perform an Adam update.

5. α is increased slowly (geometric schedule) so that the optimisation
   starts with a smooth landscape and gradually focuses on the true max.
   Random jitter is added occasionally to escape plateaus.

6. The whole procedure is repeated for several lengths (the larger the
   better, but we keep them modest to stay within the 1000 s budget) and for
   several random restarts.  The best sequence seen overall is returned.

The implementation uses only NumPy, runs on a single CPU and respects the
time limit by checking the deadline on every iteration.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _linear_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using an FFT (real‑to‑real).
    Returns a vector of length len(u)+len(v)-1.
    """
    nu = u.shape[0]
    nv = v.shape[0]
    N = _next_pow2(nu + nv - 1)
    uf = np.zeros(N, dtype=np.float64)
    vf = np.zeros(N, dtype=np.float64)
    uf[:nu] = u
    vf[:nv] = v
    Uf = np.fft.rfft(uf)
    Vf = np.fft.rfft(vf)
    Cf = Uf * Vf
    conv_full = np.fft.irfft(Cf, n=N)
    return conv_full[: nu + nv - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT.
    Returns a vector of length 2·n‑1.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A                     # because a is real, this is |A|²
    conv_full = np.fft.irfft(C, n=N)
    return conv_full[: 2 * n - 1]


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _adam_update(z: np.ndarray,
                 grad_z: np.ndarray,
                 m: np.ndarray,
                 v: np.ndarray,
                 t: int,
                 lr: float,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> Tuple[np.ndarray, Tuple[np.ndarray, np.ndarray]]:
    """
    Classic Adam step for the parameter vector ``z``.
    Returns the updated ``z`` and the new (m, v).
    """
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, (m, v)


def _smooth_max_gradient(a: np.ndarray, alpha: float) -> Tuple[np.ndarray, float]:
    """
    Compute the gradient of the smooth max objective
        J(a) = Σ_k w_k·(a*a)_k,   w_k ∝ exp(α·(a*a)_k)

    Returns:
        grad_a  – ∂J/∂a  (shape n)
        cur_max – max_k (a*a)_k   (needed for the true score)
    """
    n = a.shape[0]

    # 1) convolution a*a
    conv = _conv_self(a)                     # length 2n‑1
    cur_max = conv.max()

    # 2) soft‑max over convolution entries (smooth approximation of the max)
    scaled = alpha * conv
    # subtract the max for numerical stability
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum == 0.0:                         # should never happen, but guard
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient w.r.t. a :
    #    ∂J/∂a_i = 2· Σ_j w_{i+j}·a_j
    #    This is the middle n entries of the linear convolution
    #    between reversed a and w.
    a_rev = a[::-1]
    conv_rev = _linear_convolve(a_rev, w)    # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

    return grad_a, cur_max

            # small Gaussian noise around zero (soft‑max ≈ uniform)
            z = np.random.randn(n) * 0.02

        # Adam state
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t = 0

        # hyper‑parameters
        lr0 = 0.08 / np.sqrt(n)      # base learning rate
        alpha = 0.5                  # start very smooth
        alpha_mul = 1.001            # geometric increase each step
        alpha_max = 200.0

        # ----------------------------------------------------------------
        # Inner optimisation loop
        # ----------------------------------------------------------------
        last_improve_iter = 0
        while time.time() < deadline and t < max_iters:
            a = _softmax(z)                     # Σ a = 1, a_i ≥ 0

            # gradient of smooth‑max objective and true max (for scoring)
            grad_a, cur_max = _smooth_max_gradient(a, alpha)

            # compute score (exact objective for sum=1)
            cur_score = float(2.0 * n * cur_max)

            # keep the best overall sequence
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                last_improve_iter = t
                print(
                    f"[n={n}] new global best {best_score:.6f} at iter {t}"
                )

            # chain rule through the soft‑max
            dot = a.dot(grad_a)
            grad_z = a * (grad_a - dot)

            # Adam update (learning‑rate decays very slowly)
            t += 1
            lr_t = lr0 / (1.0 + t * 1e-5)
            z, (m, v) = _adam_update(z, grad_z, m, v, t, lr_t)

            # increase the smoothness parameter α
            if alpha < alpha_max:
                alpha *= alpha_mul

            # occasional random jitter to escape flat regions
            if t % 5000 == 0:
                z += np.random.randn(n) * 1e-4

            # if we have not improved for a long while, inject a bigger jitter
            if t - last_improve_iter > 30_000:
                z += np.random.randn(n) * 5e-4
                last_improve_iter = t

        # --------------------------------------------------------------
        # End of one length‑run – try another length if time permits
        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (official evaluation)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(
            f"Final best score (official) {final_score:.6f} "
            f"(length {len(best_seq)})"
        )
    except Exception as exc:
        print(f"Warning: final evaluation failed – {exc}")

    return best_seq
