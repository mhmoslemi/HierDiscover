def smooth_max(conv, tau):
    if tau <= 0: return np.max(conv). Use standard max.
    max_conv = np.max(conv)
    exps = np.exp((conv - max_conv)/tau)  # stable
    sum_exps = np.sum(exps)
    smooth = max_conv + tau*np.log(sum_exps)
    # weight vector
    w = exps / sum_exps
    return smooth, w

def conv_fft(x):
   n = len(x)
   m = 2*n - 1
   X = np.fft.rfft(x, n=m)
   conv = np.fft.irfft(X * X, n=m)
   return conv

def compute_smooth_score_and_gradient(a, tau):
    n = a.shape[0]
    conv = conv_fft(a)  # length 2n-1
    max_c = conv.max()
    # if tau is extremely small (<1e-12), treat as exact max
    if tau <= 0:
        # weight vector is one-hot at max index
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        smooth = max_c
    else:
        # subtract max to improve stability
        shifted = conv - max_c
        # scaled
        exp_shifted = np.exp(shifted / tau)
        sum_exp = np.sum(exp_shifted)
        w = exp_shifted / sum_exp
        smooth = max_c + tau * np.log(sum_exp)
    # compute gradient
    a_rev = a[::-1]
    conv_wa = conv_fft_general(w, a_rev)  # length (2n-1)+n-1 = 3n-2
    start = n - 1
    grad = 2.0 * conv_wa[start:start + n]
    return smooth, grad

def conv_fft_general(x, y):
    n = len(x) + len(y) - 1
    X = np.fft.rfft(x, n=n)
    Y = np.fft.rfft(y, n=n)
    conv = np.fft.irfft(X * Y, n=n)
    return conv

"""
Propose a candidate step‑function sequence that minimizes
evaluate_sequence.

Algorithm
---------

1. **Generate a small pool of promising seeds**  
   – power‑law families a_i ∝ (i+offset)^(-p)  
   – exponential families a_i ∝ exp(-α·i)  
   – the classic half‑binomial sequence  
   – a few random Dirichlet draws (optional use of the previously
     discovered ``height_sequence_1``).

   All seeds are normalised to sum = 1 and the best few (by the simple
   surrogate 2·n·max(conv)) are kept.

2. **Refinement with Adam on the probability simplex**  
   The objective is 2 · n · max(conv(a,a)).  The max is replaced by a
   smooth approximation
       smooth_maxτ(x) = τ·log(∑ exp(x/τ))
   (implemented as a stable log‑sum‑exp).  For a given τ the gradient
   of smooth_maxτ w.r.t. a can be written in closed form using a single
   convolution:
       ∇a smooth_maxτ = 2·(w * a_rev)[n‑1 : n‑1+n]   (w are the soft‑max
       weights of the convolution entries).

   An Adam optimiser (first‑ and second‑moment estimates) updates the
   unconstrained vector; after each step the vector is projected back
   onto the probability simplex (Euclidean projection).

   The temperature τ is annealed: τ = max(conv)·τ_factor,
   τ_factor starts at 0.5 and is multiplied by 0.99 every 1000 updates,
   gradually turning the smooth objective into the true max.

3. **Final fine‑tuning** – a short projected gradient descent on the
   exact max (τ = 0) with a decaying step size cleans up the final
   result.

The whole routine respects a 950 s internal budget (leaving a safety
margin) and prints progress information.  The returned sequence is a
plain list of floats; a final call to ``evaluate_sequence`` guarantees
the reported score matches the official evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i = (i + offset)^{-p}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def exponential_sequence(n: int, alpha: float) -> np.ndarray:
    """a_i = exp(-α·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def half_binomial_sequence(n: int) -> np.ndarray:
    """Classic half‑binomial (recursive) sequence, normalised."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Self‑convolution x * x using a real FFT (length = 2·len(x)‑1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using a real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    X = np.fft.rfft(x, n=n)
    Y = np.fft.rfft(y, n=n)
    conv = np.fft.irfft(X * Y, n=n)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w | w_i ≥ 0, Σ w_i = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score_from_a(a: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv(a,a)) (sum(a) is 1)."""
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * conv.max())


def compute_smooth_score_and_gradient(a: np.ndarray, tau: float):
    """
    Returns (smooth_max, grad) where smooth_max is the τ‑smooth
    approximation of max(conv(a,a)) and grad = ∇_a smooth_max.
    """
    n = a.shape[0]
    conv = conv_fft(a)                     # length 2n‑1
    max_c = conv.max()
    if tau < 1e-12:                        # treat as exact max
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
        smooth = max_c
    else:
        # stable log‑sum‑exp
        shifted = conv - max_c
        # exponentials may under‑flow, guard against zero sum
        exp_shifted = np.exp(shifted / tau)
        sum_exp = exp_shifted.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            smooth = max_c
        else:
            w = exp_shifted / sum_exp
            smooth = max_c + tau * np.log(sum_exp)

    # gradient: 2·(w * a_rev) taken at positions n‑1 … n‑1+n‑1
    a_rev = a[::-1]
    conv_wa = conv_fft_general(w, a_rev)   # length 3n‑2
    start = n - 1
    grad = 2.0 * conv_wa[start:start + n]
    return smooth, grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative sequence of heights minimising the
    provided evaluation function.  Returns the best sequence found as
    a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # configuration
    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # safety margin
    n = 2500                                 # length of the vectors

    # hyper‑parameters for Adam
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.08

    # annealing for the smooth‑max temperature
    tau_factor_start = 0.5
    tau_decay = 0.99
    tau_decay_every = 1000

    # ------------------------------------------------------------------
    # 1) build an initial pool of candidates
    pool_candidates = []

    # – use the previously discovered sequence if it exists
    prev = globals().get('height_sequence_1')
    if isinstance(prev, (list, tuple)):
        a0 = np.array(prev[:n], dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        total = a0.sum()
        if total > 0.0:
            a0 /= total
            pool_candidates.append(a0)

    # – power‑law grid
    offsets = [0.5, 1.0, 2.0]
    p_grid = np.arange(0.45, 0.85, 0.02)
    for off in offsets:
        for p in p_grid:
            pool_candidates.append(power_law_sequence(n, p, offset=off))

    # – exponential grid
    alphas = np.arange(0.0003, 0.0061, 0.0004)
    for a in alphas:
        pool_candidates.append(exponential_sequence(n, a))

    # – half‑binomial seed
    pool_candidates.append(half_binomial_sequence(n))

    # – a few random Dirichlet draws
    for _ in range(8):
        alpha = 0.5 if rng.random() < 0.6 else 1.0
        a = rng.dirichlet(np.full(n, alpha))
        pool_candidates.append(a)

    # evaluate all seeds with the cheap surrogate and keep the top K
    scored = [(compute_score_from_a(a), a) for a in pool_candidates]
    scored.sort(key=lambda x: x[0])
    top_k = 5
    states = []  # each entry = dict with keys: a, m, v, t
    best_score = np.inf
    best_a = None

    for score, a in scored[:top_k]:
        best_score = min(best_score, score)
        best_a = a.copy() if best_a is None else (a.copy() if score < best_score else best_a)
        states.append({
            'a': a.copy(),
            'm': np.zeros_like(a),
            'v': np.zeros_like(a),
            't': 0,
            'tau_factor': tau_factor_start,
        })

    print(f"[init] kept {len(states)} seeds, best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam optimisation with smooth‑max annealing
    it = 0
    while time.time() < deadline and it < 200_000:
        for st in states:
            if time.time() >= deadline or it >= 200_000:
                break

            a = st['a']
            # compute smooth max and its gradient
            tau_factor = st['tau_factor']
            conv = conv_fft(a)                # we need max for τ
            max_c = conv.max()
            tau = max_c * tau_factor
            smooth, grad = compute_smooth_score_and_gradient(a, tau)

            # Adam update (minimise smooth)
            st['t'] += 1
            t = st['t']
            m = st['m']
            v = st['v']
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = -base_lr * m_hat / (np.sqrt(v_hat) + eps)

            # apply step and project onto the simplex
            a_new = project_onto_simplex(a + step)

            # occasional tiny random jitter to escape flat plateaus
            if rng.random() < 0.003:
                jitter = (rng.random(n) - 0.5) * 1e-5
                a_new = project_onto_simplex(a_new + jitter)

            st['a'] = a_new

            # periodic exact evaluation (every 20 updates per state)
            if t % 20 == 0:
                cur_score = compute_score_from_a(a_new)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a_new.copy()
                    print(f"[iter {it:07d}] new best surrogate score = {best_score:.6f}")

            # decay the temperature τ once in a while
            if (it + 1) % tau_decay_every == 0:
                for s in states:
                    s['tau_factor'] *= tau_decay

            it += 1

    # ------------------------------------------------------------------
    # 3) Final refinement using the true max (τ = 0) and a simple
    #    projected gradient descent with a decaying step size.
    a = best_a.copy()
    cur_score = compute_score_from_a(a)
    lr = 0.02
    decay_step = 0.92
    decay_every = 800
    for i in range(3000):
        if time.time() >= deadline:
            break
        conv = conv_fft(a)
        peak_idx = int(conv.argmax())
        # exact sub‑gradient of the active entry
        n = a.shape[0]
        grad = np.zeros_like(a)
        lo = max(0, peak_idx - (n - 1))
        hi = min(peak_idx, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        jdx = peak_idx - idx
        grad[idx] = 2.0 * a[jdx]                     # diagonal case also gets a factor 2 (good enough)

        # step & projection
        step = lr * grad
        step = np.minimum(step, a * 0.99)            # prevent overshooting to negative
        a = project_onto_simplex(a - step)

        # evaluate
        new_score = compute_score_from_a(a)
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[fine {i:04d}] improved final score = {best_score:.6f}")

        cur_score = new_score
        if (i + 1) % decay_every == 0:
            lr *= decay_step
            lr = max(lr, 5e-4)

    # ------------------------------------------------------------------
    # final sanity check with the official evaluator
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] best reported score = {final_score:.6f}")
    return best_a.tolist()
