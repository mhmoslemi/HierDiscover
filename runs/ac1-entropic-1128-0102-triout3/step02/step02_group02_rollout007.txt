"""
Propose a candidate step‑function sequence.

The algorithm searches for a non‑negative vector ``a`` (sum normalized to 1)
that minimizes the evaluation function

    2·n·max_k (a*a)_k .

A smooth approximation of the ``max`` is used:
    w_k ∝ exp(α·(conv_k – max(conv))) ,   Σ w_k = 1 .
Then
    ∂/∂a (α‑smoothed max) = 2·(w ⋆ a)     (valid linear correlation)
and the soft‑max parametrisation ``a = softmax(z)`` guarantees non‑negativity
and the normalisation constraint.

Gradient descent is performed in the log‑space ``z`` with Adam.  The
smoothing parameter ``α`` is increased during the run, turning the
objective gradually from an average to the true maximum.  The routine
optimises several lengths (2000, 4000, …) within the allotted time and
keeps the best sequence found.  If a previous solution is available via the
global variable ``height_sequence_1`` it is used as an initial candidate.
"""
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two 1‑D real arrays using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = x.shape[0]
    m = y.shape[0]
    N = _next_pow2(n + m - 1)
    X = np.fft.rfft(np.pad(x, (0, N - n)))
    Y = np.fft.rfft(np.pad(y, (0, N - m)))
    conv = np.fft.irfft(X * Y, n=N)[: n + m - 1]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _gradient_wrt_a(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the gradient of the α‑smoothed max(conv) w.r.t. the
    probability vector ``a``.
    """
    # Soft‑max weights over the convolution entries
    conv_max = conv.max()
    shifted = conv - conv_max          # for numerical stability
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # valid linear correlation w ⋆ a  (length len(a))
    a_rev = a[::-1]
    corr_full = _convolve_fft(w, a_rev)  # length 3·n‑2
    n = a.shape[0]
    # slice the “valid” part: indices n‑1 … 2·n‑2 (inclusive)
    grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad_a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights that (hopefully) beats the current
    best bound.  The function may run up to ``budget_s`` seconds; it returns
    the best sequence discovered before the timeout.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq: List[float] | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction exists, start from it
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(height_sequence_1)
            cand_score = evaluate_sequence(cand)
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[init] previous best score {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Search over a few lengths – the larger the better, but we keep them
    # modest to guarantee we finish within the budget.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a sub‑budget for this length (≈ remaining/len(lengths))
        remaining = deadline - time.time()
        sub_budget = min(remaining * 0.9, 200.0)   # at most 200 s per length
        sub_deadline = time.time() + sub_budget
        print(f"\n=== Optimising length n={n} (budget {sub_budget:.1f}s) ===")

        # hyper‑parameters (scale with n)
        base_lr = 0.05 / np.sqrt(n)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8

        # smoothing schedule for the soft‑max over the convolution
        alpha_start = 1.0
        alpha_max = 30.0
        # after ~150 k iterations we reach alpha_max – adjust later if needed
        alpha_increment = (alpha_max - alpha_start) / 150_000.0

        # initialise logits – small Gaussian noise around zero (soft‑max → uniform)
        z = rng.standard_normal(n) * 0.02
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t = 0

        # stagnation detector – restart if no improvement for many steps
        steps_since_best = 0

        while time.time() < sub_deadline:
            a = _softmax(z)                       # maintains Σ a_i = 1, a_i ≥ 0

            # linear convolution a*a (full length 2·n‑1)
            conv = _convolve_fft(a, a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max           # identical to official score for Σa=1

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                steps_since_best = 0
                print(f"[{n}] New global best {best_score:.6f}")
            else:
                steps_since_best += 1

            # ----- gradient of the smoothed max -----
            alpha = min(alpha_max, alpha_start + t * alpha_increment)
            grad_a = _gradient_wrt_a(a, conv, alpha)

            # chain rule for the soft‑max parametrisation
            dot = a.dot(grad_a)
            grad_z = a * (grad_a - dot)

            # ----- Adam update in log‑space -----
            t += 1
            lr_t = base_lr / (1.0 + t * 1e-5)

            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr_t * m_hat / (np.sqrt(v_hat) + eps)
            z = z - step

            # occasional random jitter to escape flat plateaus
            if t % 2000 == 0:
                z += rng.normal(scale=1e-3, size=n)

            # restart if we have been stuck for a long time
            if steps_since_best > 80_000:
                print(f"[{n}] Restarting after {steps_since_best} steps without improvement")
                z = rng.standard_normal(n) * 0.02
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                steps_since_best = 0

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform sequence score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (official evaluation)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"Final best score (official): {final_score:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print("Warning: final evaluation failed:", exc)

    return best_seq
