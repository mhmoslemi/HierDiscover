"""
Propose a step‑function sequence that minimizes `evaluate_sequence`.

Algorithm
---------
* Parameterise the non‑negative sequence `a` by a probability vector `p`
  (∑p = 1) and a fixed ℓ₁‑norm `S = sqrt(2·n)`.  The objective equals
  `2·n·max(p∗p)`, so only the shape of `p` matters.
* Optimise `p` on the simplex by **mirror descent** using the
  exponential map (i.e. soft‑max) together with an Adam‑style adaptive
  step‑size.  The sub‑gradient of `max(p∗p)` is obtained from the active
  set of convolution indices (the same sub‑gradient used in the previous
  baseline).
* The iteration performs:
      – compute `p = softmax(y)`;
      – evaluate the self‑convolution (`FFT`);
      – build the mask of maximal entries;
      – compute the sub‑gradient `g`;
      – map it to the dual space (`y`) by `g_y = p·(g‑⟨p,g⟩)`;
      – Adam update of `y`.
* Stagnation handling: if no improvement is seen for a long time the
  optimizer is re‑initialised from a Dirichlet draw.  Small Gaussian
  noise is added periodically to encourage exploration.
* The best sequence found within the time budget is returned (scaled back
  to the required ℓ₁‑norm).  All numbers are non‑negative, well below the
  1000‑limit, and the function respects the security checks of
  `evaluate_sequence`.

The method is a genuine alternative to the sub‑gradient / simulated‑annealing
hybrid used previously and often reaches a lower maximal convolution
value in the same time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full (2·len(a)‑1) result (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _grad_max_subgrad(p: np.ndarray, mask: np.ndarray, nfft: int) -> np.ndarray:
    """
    Sub‑gradient of  max_k (p∗p)[k]  at the current active set.
    `mask` is a binary array (same length as the convolution) with 1s at
    the indices where the convolution attains its current maximum.
    The gradient returned has length len(p).
    """
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros_like(p)

    w = w / w_sum                     # normalise to a probability vector
    p_rev = p[::-1]

    A_rev = np.fft.rfft(p_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * p.size - 1]

    n = p.size
    # slice that corresponds to the gradient entries
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Adam‑based mirror‑descent search for a low‑maximum self‑convolution
    step function.  Returns the best sequence found before the time budget
    expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # problem size and ℓ₁‑norm
    # ------------------------------------------------------------------
    try:
        # use the previously discovered construction if available
        n = len(height_sequence_1)                 # type: ignore[name-defined]
        a0 = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        n = 2000
        a0 = None

    S = np.sqrt(2.0 * n)                         # fixed ℓ₁‑norm (will be reapplied at the end)
    nfft = _next_pow2(2 * n - 1)                 # length for FFT (power of two)

    # ------------------------------------------------------------------
    # initialise the mirror‑descent iterate (log‑weights y)
    # ------------------------------------------------------------------
    if a0 is not None:
        # normalise a0 to a probability vector and take logs
        p0 = a0 / a0.sum()
        y = np.log(p0 + 1e-20)                   # avoid log(0)
    else:
        y = np.zeros(n, dtype=np.float64)

    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4                                    # base learning rate (tuned empirically)

    iteration = 0
    no_improve = 0

    eps_mask = 1e-12

        # ----- compute current probability vector via soft‑max -----
        y_max = np.max(y)

        p = exp_y / exp_y.sum()

        # ----- self‑convolution and its maximum -----
        conv = _convolve_self_fft(p, nfft)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            y = np.log(p_restart + 1e-20)

            v.fill(0.0)
            no_improve = 0
            print(f"[{iteration:07d}] Restart due to stagnation (no improvement).")

    # ------------------------------------------------------------------
    # final sequence (scale back to required ℓ₁‑norm) and verification
    # ------------------------------------------------------------------
    if best_p is None:
        # fallback – uniform distribution
        best_p = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = best_p * S                           # now Σ a_i = S = sqrt(2·n)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. "
          f"Best final score = {final_score:.9f}")
