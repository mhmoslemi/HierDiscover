Hybrid smooth‑max gradient descent + simulated‑annealing search for a
step‑function that minimises `evaluate_sequence`.  The method starts from a
deterministic decreasing 1/√t‑shaped sequence (which is close to the known
continuous optimum) or from the previously found best sequence if it is
available.  At each iteration the full self‑convolution is computed with FFT;
a softmax approximation of the max is used to obtain a smooth gradient.
A projected gradient step on the ℓ₁‑simplex (sum fixed to √(2n)) is taken,
occasionally combined with random mass‑transfer moves and occasional restarts.
The temperature of a Metropolis acceptance rule and the softmax temperature are
adapted during the run.  The best sequence encountered before the time budget
expires is returned.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using real FFT (full length 2·len(a)-1)."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv

def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = target_sum}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _init_sqrt_shape(n: int, target_sum: float) -> np.ndarray:
    """Return a decreasing shape a_i ∝ 1/√(i+1), scaled to sum = target_sum."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / np.sqrt(idx)
    a = a * (target_sum / a.sum())
    return a

def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return softmax probabilities proportional to exp(beta*conv)."""
    # use max trick for numerical stability
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    w = exps / exps.sum()
    return w

def _grad_softmax_max(a: np.ndarray, conv: np.ndarray, beta: float, nfft: int) -> np.ndarray:
    """
    Gradient of the smooth max f(a)= (1/beta) log sum exp(beta*conv)
    w.r.t. a.  The formula reduces to 2 * (a_rev * w) where w are the
    softmax weights.
    """
    w = _softmax_weights(conv, beta)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=nfft)
    A_rev = np.fft.rfft(a_rev, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]
    grad = 2.0 * grad_full[a.size - 1 : a.size - 1 + a.size]
    return grad

def _grad_max_subgrad(a: np.ndarray, conv: np.ndarray, eps_mask: float, nfft: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) (conv = a*a) obtained from the
    active set of entries within `eps_mask` of the current maximum.
    """
    mask = (conv >= conv.max() - eps_mask).astype(float)
    w_sum = mask.sum()
    if w_sum == 0:
        return np.zeros_like(a)
    w = mask / w_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=nfft)
    A_rev = np.fft.rfft(a_rev, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]
    grad = 2.0 * grad_full[a.size - 1 : a.size - 1 + a.size]
    return grad

    """

    else:
        n = 2000                 # fallback size
        a0 = None

    # ------------------------------------------------------------------
    # Initialise a – a decreasing 1/√t shape (close to continuous optimum)
    # ------------------------------------------------------------------
    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        a = _init_sqrt_shape(n, S)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    conv = _convolve_self_fft(a, nfft)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (S ** 2)

    best_a = a.copy()
    best_score = cur_score

    temp_decay = 0.9997
    min_temp = 1e-7

    return best_a.tolist()

    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _init_sqrt_shape(n: int, target_sum: float) -> np.ndarray:
    """Return a decreasing shape a_i ∝ 1/√(i+1), scaled to a given ℓ₁‑norm."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / np.sqrt(idx)
    a = a * (target_sum / a.sum())
    return a


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max probabilities proportional to exp(beta·conv)."""
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    w = exps / exps.sum()
    return w


def _grad_softmax_max(a: np.ndarray,
                     conv: np.ndarray,
                     beta: float,
                     nfft: int) -> np.ndarray:
    """
    Gradient of the smooth max f(a)= (1/β)·log Σ exp(β·conv)
    w.r.t. a.  The result is 2·(a_rev * w) where w are the soft‑max weights.
    """
    w = _softmax_weights(conv, beta)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=nfft)
    A_rev = np.fft.rfft(a_rev, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]
    grad = 2.0 * grad_full[a.size - 1 : a.size - 1 + a.size]
    return grad


def _grad_exact_subgrad(a: np.ndarray,
                       conv: np.ndarray,
                       eps_mask: float,
                       nfft: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) obtained from the active set
    (indices whose value is within `eps_mask` of the current maximum).
    """
    mask = (conv >= conv.max() - eps_mask).astype(float)
    weight_sum = mask.sum()
    if weight_sum == 0.0:
        return np.zeros_like(a)
    w = mask / weight_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=nfft)
    A_rev = np.fft.rfft(a_rev, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]
    grad = 2.0 * grad_full[a.size - 1 : a.size - 1 + a.size]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Hybrid optimisation for the step‑function problem.
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin for the terminator

    # ------------------------------------------------------------------
    # Problem size and ℓ₁‑norm that we keep fixed
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        a0 = None

    S = np.sqrt(2.0 * n)                     # Σa = √(2n)
    nfft = _next_pow2(2 * n - 1)             # convenient FFT length (power of two)

    # ------------------------------------------------------------------
    # Initial iterate
    # ------------------------------------------------------------------
    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        a = _init_sqrt_shape(n, S)           # 1/√t decreasing shape

    # ------------------------------------------------------------------
    # Book‑keeping of the best solution found so far
    # ------------------------------------------------------------------
    conv = _convolve_self_fft(a, nfft)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (S ** 2)

    best_a = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    step_base = 0.6 * S / n            # base step size for gradient moves
    temperature = 0.08                 # Metropolis temperature
    temp_decay = 0.9997
    min_temp = 1e-7

    p_grad = 0.55                      # probability of a smooth‑max gradient step
    p_random = 0.30                    # probability of a random mass‑transfer move
    # remaining probability → full restart

    eps_mask = 1e-12
    beta = 5.0                         # starting soft‑max temperature
    beta_factor = 1.001
    beta_max = 500.0

    no_improve = 0
    max_no_improve = 2000              # trigger a restart if too long without progress
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # recompute convolution and the objective for the current point
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (S ** 2)

        # keep the globally best sequence
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:06d}] New best: {best_score:.6f} (max={cur_max:.6f})")
            no_improve = 0
        else:
            no_improve += 1

        # early exit if the target bound is reached
        if best_score < 1.503:
            print(f"Target reached after {iteration} iterations (best={best_score:.6f})")
            break

        # ------------------------------------------------------------------
        # Choose a move type
        # ------------------------------------------------------------------
        r = np.random.rand()
        if r < p_grad:
            # ----- smooth‑max gradient step (projected) -----
            grad = _grad_softmax_max(a, conv, beta, nfft)
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:
                eta = step_base / np.sqrt(iteration + 1.0)
                step = eta * grad / grad_norm
                a_candidate = _project_to_simplex(a - step, S)
            else:
                a_candidate = a.copy()

        elif r < p_grad + p_random:
            # ----- random mass‑transfer (annealing style) -----
            if a.sum() > 0:
                src_probs = a / a.sum()
                i = np.random.choice(n, p=src_probs)
            else:
                i = np.random.randint(n)

            j = i
            while j == i:
                j = np.random.randint(n)

            eps_transfer = 0.07
            delta = eps_transfer * a[i]
            if delta <= 0:
                delta = eps_transfer * S / n

            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = _project_to_simplex(a_candidate, S)

        else:
            # ----- full restart from a Dirichlet draw -----
            alpha = np.full(n, 0.4)
            a_candidate = np.random.dirichlet(alpha) * S
            print(f"[{iteration:06d}] Full restart performed.")

        # ------------------------------------------------------------------
        # Evaluate the candidate (full objective)
        # ------------------------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max / (S ** 2)

        # acceptance rule (greedy + Metropolis)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
            cur_max = cand_max
        else:
            # occasional jitter to escape shallow plateaux
            if np.random.rand() < 0.012:
                jitter = np.random.randn(n) * (step_base * 0.4)
                a = _project_to_simplex(a + jitter, S)

        # ------------------------------------------------------------------
        # Update schedules
        # ------------------------------------------------------------------
        temperature = max(temperature * temp_decay, min_temp)
        beta = min(beta * beta_factor, beta_max)

        # restart if we have stagnated for a long time
        if no_improve >= max_no_improve:
            # blend a fresh Dirichlet restart with a small amount of noise
            alpha = np.full(n, 0.3)
            a = np.random.dirichlet(alpha) * S
            no_improve = 0
            print(f"[{iteration:06d}] Stagnation restart (no improvement).")

        iteration += 1

    # ------------------------------------------------------------------
    # Final projection and verification
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")

    return best_a.tolist()
