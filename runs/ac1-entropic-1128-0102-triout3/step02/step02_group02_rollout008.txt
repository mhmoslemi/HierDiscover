import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def lp_max_sum_g(f, M):
    n = len(f)
    # Build constraint matrix A (2n-1) x n
    # For each k, columns i such that 0 <= k-i < n => i in [max(0, k-(n-1)), min(k, n-1)]
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # iterate i
        i_range = np.arange(i_start, i_end+1)
        rows.extend([k]*len(i_range))
        cols.extend(i_range.tolist())
        # data: f[i]
        data.extend(f[i_range].tolist())
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # upper bound constraints: A * g <= M
    # convert to linprog format: A_ub @ g <= b_ub
    b = np.full(2*n-1, M, dtype=np.float64)
    # we want to maximize sum(g) i.e. minimize -sum(g)
    c = -np.ones(n)
    # bounds for variables: (0, None)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: return zeros
        return np.zeros_like(f)
    return res.x

        while time.time() < deadline:

            conv = _conv_fft(a)
            cur_max = conv.max()
            cur_score = float(2.0 * n * cur_max)
            # update best local
            if cur_score < best_local_score:

                best_local_a = a.copy()
            # global best
            nonlocal best_seq, best_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[n={n}] New global best {best_score:.6f}")
            # pick indices
            if top_k == 1:
                ks = np.array([int(np.argmax(conv))], dtype=int)
            else:
                ks = np.argpartition(conv, -top_k)[-top_k:]
            # gradient w.r.t a
            grad_a = _gradient_wrt_a(a, ks)
            # gradient w.r.t logits
            dot = a.dot(grad_a)

            m = 0.9 * m + 0.1 * grad_z

            # recompute a
            a = _logits_to_a(z)

            # schedule top_k
            if t == top_k_decay_iter:

def _next_pow2(x):
    return 1 << (x-1).bit_length()
def _conv_fft(a):
    n = a.shape[0]
    N = _next_pow2(2*n-1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A
    conv = np.fft.irfft(C, n=N)[:2*n-1]
    return conv

"""
Propose a candidate step‑function sequence.

The algorithm is a two‑stage search:

1️⃣  Gradient‑based optimisation in the soft‑max (log‑space) using Adam.
    The objective 2·n·max(a*a) is approximated by a sub‑gradient that
    only involves the few largest convolution entries (parameter *top_k*).
    Several lengths (up to a few × 10³) and a few random restarts are
    tried; the best sequence ever seen is kept globally.

2️⃣  A cheap linear‑programming refinement (only for moderate lengths ≤2400):
    for a fixed f we maximise  Σ g  under the linear constraints (f·g)≤M.
    The resulting g is blended with f (weight *t*) and renormalised.
    Repeating a few blend steps monotonically reduces the maximal
    auto‑correlation and often yields a small extra improvement.

All numbers are forced to be non‑negative; the score is scale‑invariant,
so we keep the vectors normalised to sum = 1.
The routine stops after the allotted budget (default 1000 s) and returns
the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (all top‑level – required by the platform)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (real, full length)."""
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A                      # |A|²  (convolution theorem)
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv_full


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Soft‑max (numerically stable) mapping from logits to a ∈ Δₙ."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _gradient_wrt_a(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)_k  w.r.t. a,
    summed over the selected offsets top_idxs.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    for k in top_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]

    return grad


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1, n)) such that
    A @ g == f * g  (linear convolution with fixed f).
    Entry A[k, i] = f_i if 0 ≤ k‑i < n, else 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Linear programme:
        maximise   Σ g_j
        subject to (f * g)_k ≤ M  (∀k),  g ≥ 0.
    Returns the optimiser g (may be all‑zero if LP fails).
    """
    A = _build_conv_matrix(f)
    c = -np.ones(f.shape[0])               # maximise → minimise – Σg
    b = np.full(A.shape[0], M, dtype=np.float64)

    # use the high‑performance HiGHS solver (available in recent SciPy)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=(0, None),
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    # fallback – return a zero vector (guaranteed feasible)
    return np.zeros_like(f)


def _lp_refine(f: np.ndarray, steps: int = 4, blend: float = 0.2) -> np.ndarray:
    """
    Apply a few LP‑based improvement steps to the vector f.
    Each step solves the LP for g, blends `f ← (1‑t)·f + t·g`,
    and renormalises to unit sum.
    """
    n = f.shape[0]
    for _ in range(steps):
        M = _conv_fft(f).max()
        g = _solve_lp_max_sum(f, M)
        f = (1.0 - blend) * f + blend * g
        s = f.sum()
        if s > 0:
            f /= s
    return f


# ----------------------------------------------------------------------
# Core optimiser – soft‑max Adam with a configurable top‑k schedule
# ----------------------------------------------------------------------


def _softmax_optimize(
    n: int,
    deadline: float,
    init_a: np.ndarray | None = None,
    max_steps: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent in log‑space (Adam) for a fixed length n.
    Stops when the global deadline is reached or when max_steps are
    exhausted.  Returns the best vector found (normalised to sum = 1)
    together with its objective value `2·n·max(conv)`.
    """
    # initialise logits
    if init_a is not None:
        eps = 1e-12
        z = np.log(np.clip(init_a, eps, None))
    else:
        z = np.random.randn(n) * 0.02

    a = _logits_to_a(z)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0

    # hyper‑parameters (scaled with n)
    lr0 = 0.10 / np.sqrt(n)               # base learning rate
    top_k = max(1, n // 200)              # start with a few largest entries
    decay_step = 8000                     # after this many iters keep only the true max
    best_local_a = a.copy()
    best_local_score = float("inf")

    while time.time() < deadline and t < max_steps:
        # ----- 1) convolution & current score -----
        conv = _conv_fft(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        # keep the best local version
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_a = a.copy()

        # ----- 2) build sub‑gradient from the top entries -----
        if top_k == 1:
            ks = np.array([int(np.argmax(conv))], dtype=int)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        grad_a = _gradient_wrt_a(a, ks)

        # ----- 3) chain rule through the soft‑max -----
        dot = a.dot(grad_a)
        grad_z = a * (grad_a - dot)

        # ----- 4) Adam update (classic) -----
        t += 1
        lr_t = lr0 / (1.0 + t * 1e-5)          # slow decay
        m = 0.9 * m + 0.1 * grad_z
        v = 0.999 * v + 0.001 * (grad_z ** 2)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        step = lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = z - step

        # ----- 5) recompute a from logits -----
        a = _logits_to_a(z)

        # ----- 6) occasional jitter to avoid stagnation -----
        if t % 1500 == 0:
            z += np.random.randn(n) * 1e-3

        # ----- 7) schedule shrinking of the active set -----
        if t == decay_step:
            top_k = 1

    return best_local_a, best_local_score
