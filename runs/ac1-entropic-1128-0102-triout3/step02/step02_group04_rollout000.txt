"""
Propose a step‑function sequence that (hopefully) improves the known bound.
The method is a two‑stage mirror‑descent on the probability simplex.
* Stage 1 works on a modest size (≈800) with a smoothed “soft‑max”
  approximation of the ‑∞ convolution norm.  The smoothness parameter β
  is increased gradually while a small step size η is decayed.
* Stage 2 up‑samples the best coarse solution to the target size (2500)
  and performs the same descent with tighter parameters.
The gradient of the smoothed objective is computed efficiently with
FFT‑based convolutions (full convolution of the weight vector with the
current sequence).  After each update the iterate is projected onto the
simplex using the Euclidean projection of Duchi et al.  Random
perturbations and occasional restarts help to escape plateaus.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolve_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of two real vectors using a real FFT.
    Length of result = len(x) + len(y) - 1.
    """
    m = next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=m)
    Y = np.fft.rfft(y, n=m)
    conv = np.fft.irfft(X * Y, n=m)
    return conv[: len(x) + len(y) - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a (full length 2·len(a)‑1) via FFT."""
    return convolve_full(a, a)


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x ≥ 0, Σ x = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of the half‑binomial series
        a_i = C(2i,i) / 4**i,
    normalised to have sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def objective_value(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when the sum of a is 1:
        2 * n * max(a * a)
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search the simplex for a non‑negative sequence minimising
    ``evaluate_sequence``.  Returns the best sequence found as a plain
    Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Timing
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Stage 1 – coarse optimisation (n ≈ 800)
    n_coarse = 800
    a = sqrt_series(n_coarse)                      # sum = 1
    best_a = a.copy()
    best_score = objective_value(a)
    print(f"[stage‑1] n={n_coarse}, initial score={best_score:.6f}")

    # hyper‑parameters for the coarse phase
    eta = 0.20            # step size for additive update
    beta = 0.5            # smooth‑max temperature
    eta_decay = 0.9995
    beta_growth = 1.001
    it = 0
    last_improve = 0
    restart_interval = 20000

    while time.time() < deadline and it < 500_000:
        # ----- self‑convolution and smooth‑max weights -----
        conv = conv_self_fft(a)
        max_val = conv.max()
        # soft‑max weights (numerically stable)
        exp_arg = beta * (conv - max_val)
        w = np.exp(exp_arg, dtype=np.float64)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # degenerate case – fallback to uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        # grad_i = 2 * Σ_k w_k * a_{k-i}  (treated as cross‑correlation)
        rev_w = w[::-1]                               # length = 2n‑1
        full = convolve_full(rev_w, a)                # length = 3n‑2
        L = 2 * n_coarse - 1
        grad = 2.0 * full[L - 1 - np.arange(n_coarse)]

        # ----- additive mirror‑descent step with simplex projection -----
        a_trial = a - eta * grad
        a_trial = np.maximum(a_trial, 0.0)            # keep non‑negative
        a_trial = project_onto_simplex(a_trial)

        # occasional tiny random jitter (helps escape plateaus)
        if rng.random() < 0.001:
            a_trial += (rng.random(n_coarse) - 0.5) * 1e-6
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = project_onto_simplex(a_trial)

        # ----- evaluate -----
        cur_score = objective_value(a_trial)

        # ----- keep the best ever -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_trial.copy()
            last_improve = it
            print(f"[stage‑1 iter {it:07d}] new best score = {best_score:.6f}")

        # ----- acceptance (always accept downhill, otherwise keep old) -----
        if cur_score <= objective_value(a):
            a = a_trial
        else:
            # occasional uphill moves (simulated annealing)
            temperature = max(0.01 * (1.0 - it / 500_000), 0.0001)
            if rng.random() < np.exp(-(cur_score - objective_value(a)) / temperature):
                a = a_trial

        # ----- schedule updates -----
        eta = max(eta * eta_decay, 1e-6)
        beta = beta * beta_growth

        # ----- restart if stuck for a long while -----
        if it - last_improve > restart_interval:
            # restart from a mixture of the best and a fresh Dirichlet draw
            a = best_a * 0.9 + rng.dirichlet(np.full(n_coarse, 0.6)) * 0.1
            a = project_onto_simplex(a)
            last_improve = it
            print(f"[stage‑1 iter {it:07d}] restart triggered, score={objective_value(a):.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to the final length and refine (n = 2500)
    n_fine = 2500
    # Linear interpolation of the best coarse solution
    xs = np.linspace(0, 1, n_coarse)
    xf = np.linspace(0, 1, n_fine)
    a = np.interp(xf, xs, best_a)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = objective_value(a)
    print(f"[stage‑2] up‑sampled to n={n_fine}, start score={best_score:.6f}")

    # hyper‑parameters for the fine phase
    eta = 0.08
    beta = 8.0
    eta_decay = 0.9999
    beta_growth = 1.0003
    it_fine = 0
    last_improve_fine = 0
    restart_interval_fine = 30000

    while time.time() < deadline:
        # ----- self‑convolution and smooth‑max weights -----
        conv = conv_self_fft(a)
        max_val = conv.max()
        cur_score = 2.0 * n_fine * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve_fine = it_fine
            print(f"[stage‑2 iter {it_fine:07d}] new best score = {best_score:.6f}")

        exp_arg = beta * (conv - max_val)
        w = np.exp(exp_arg, dtype=np.float64)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        rev_w = w[::-1]
        full = convolve_full(rev_w, a)           # length = 3n‑2
        L = 2 * n_fine - 1
        grad = 2.0 * full[L - 1 - np.arange(n_fine)]

        # ----- update -----
        a_trial = a - eta * grad
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = project_onto_simplex(a_trial)

        # ----- random jitter -----
        if rng.random() < 0.001:
            a_trial += (rng.random(n_fine) - 0.5) * 1e-5
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = project_onto_simplex(a_trial)

        # ----- acceptance -----
        trial_score = objective_value(a_trial)
        if trial_score <= cur_score:
            a = a_trial
        else:
            # occasional uphill acceptance
            temperature = max(0.005 * (1.0 - it_fine / 500_000), 1e-6)
            if rng.random() < np.exp(-(trial_score - cur_score) / temperature):
                a = a_trial

        # ----- schedule -----
        eta = max(eta * eta_decay, 5e-7)

            a = project_onto_simplex(a)
            last_improve_fine = it_fine
            print(f"[stage‑2 iter {it_fine:07d}] restart – score={objective_value(a):.6f}")

        it_fine += 1

    # ------------------------------------------------------------------
    # Final evaluation with the provided routine
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best reported score = {final_score:.6f}")
