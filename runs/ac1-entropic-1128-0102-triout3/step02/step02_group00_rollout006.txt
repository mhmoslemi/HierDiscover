def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float('inf')
    global_a = None

    # retrieve starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # define lengths schedule
    lengths = [800, 1600, 3200, 6400]  # Up to moderate size
    np.random.shuffle(lengths)

    for n in lengths:
        if time.time() >= deadline:
            break
        # initialize
        if global_a is not None:
            a0 = _upsample(global_a, n)
        elif starter is not None and starter.size == n:
            a0 = starter.copy()
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()
        # allocate slice time
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # give about 20% to each n or at least 20 seconds
        slice_deadline = time.time() + max(20.0, 0.2 * remaining)

        # gradient descent with smoothed max
        a_opt, score_opt = _run_smooth_gradient(a0, n, slice_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] global improvement after smoothed gradient: n={n}, score={score_opt:.6f}")

        # annealing refinement
        if time.time() < deadline:
            anneal_deadline = time.time() + max(5.0, 0.05 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(a_opt, n, anneal_deadline, start_time, best_score, best_seq)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:.2f}s] global improvement after annealing: n={n}, score={score_ref:.6f}")

        global_a = best_seq.copy()

    # fallback: uniform distribution if nothing found
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n)

    # ensure sum normalized (optional)
    best_seq = best_seq / best_seq.sum()
    return [float(x) for x in best_seq]

"""
Optimising a non‑negative step function for the “max‑self‑convolution’’ score.

The objective to minimise is

    score(a) = 2·n·max_k (a * a)[k]          (a lies on the probability simplex)

where * is the linear convolution.
The algorithm works on the simplex, repeatedly:
  * minimise a smooth approximation of the max‑norm
    (log‑sum‑exp with a gradually increasing smoothing parameter α);
  * update a by projected (Euclidean) gradient descent;
  * after the gradient phase perform a cheap simulated‑annealing refinement.
A multi‑scale schedule up‑samples the best solution to larger lengths.
The whole search respects the given time budget (default 1000 s) and returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (not circular)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) approximation of max(conv)
# ----------------------------------------------------------------------
def _grad_log_sum_exp(a: np.ndarray, alpha: float):
    """
    Returns (grad, conv) where
        conv = a * a
        grad = ∂/∂a ( (1/α)·log Σ_k exp(α·conv_k) )
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    # soft‑max weights (stable)
    maxc = conv.max()
    exps = np.exp(alpha * (conv - maxc))
    w = exps / exps.sum()                    # Σ w_k = 1, w ≥ 0

    # gradient = 2 * (w * a_rev)  (restricted to n coordinates)
    rev = a[::-1]
    m = len(w) + len(rev) - 1                # = 3n‑2
    fft_len = 1 << ((m - 1).bit_length())
    W = np.fft.rfft(w, n=fft_len)
    R = np.fft.rfft(rev, n=fft_len)
    conved = np.fft.irfft(W * R, n=fft_len)[:m]
    start = len(rev) - 1                     # n‑1
    grad = 2.0 * conved[start:start + n]
    return grad, conv

# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max‑norm
# ----------------------------------------------------------------------
def _run_smooth_gradient(a0: np.ndarray, n: int,
                         deadline: float, start_time: float):
    """
    Perform a continuation of projected gradient steps.
    The smoothing parameter α is increased geometrically.
    Returns the best sequence seen and its (exact) score.
    """
    a = a0.copy()
    # initial concrete score
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    # schedule for the smoothing parameter (log‑sum‑exp)
    alphas = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]

    it = 0
    step0 = 0.4               # base step size
    # do a fixed number of inner iterations per α (adjusted later if time runs out)
    max_iter_per_alpha = 2000

    for alpha in alphas:
        for _ in range(max_iter_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            grad, conv = _grad_log_sum_exp(a, alpha)

            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"smooth‑grad improve: n={n}, α={alpha:.1f}, "
                      f"iter={it}, score={cur_score:.6f}")

            # diminishing step size
            step = step0 / math.sqrt(it)
            a = _project_simplex(a - step * grad)

    return best_a, best_score

# ----------------------------------------------------------------------
# Very cheap simulated‑annealing refinement (mass transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2, end_temp: float = 1e-5,
                   steps: int = 20000):
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose moving ε from i to j
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1      # ≤10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"annealing improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
# Linear up‑sampling (preserves the simplex)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
