def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start+budget_s-1.0

    best_seq = None
    best_score = float('inf')
    n = 4000  # choose final length, flexible

    # 1. Parameter search over simple families
    # families = [('power', p) for p in np.linspace(0,1,21)] + [...]
    for family in families:
        a = generate_shape(n, family)
        score = _score_of_seq(a)
        if score < best_score:
            best_score = score; best_seq = a.copy()
            print(...)

    # 2. Refinement with subgradient
    a = best_seq.copy()
    cur_n = len(a)
    fft_len = 1 << ((2*cur_n - 1).bit_length())
    step0 = 0.5
    it = 0
    while time.time() < deadline and it < 300000:
        it+=1
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af*Af, n=fft_len)[:2*cur_n-1]
        k = np.argmax(conv)
        # gradient for max index
        g = np.zeros_like(a)
        lo = max(0, k - (cur_n - 1))
        hi = min(cur_n - 1, k)
        i_idx = np.arange(lo, hi+1)
        j_idx = k - i_idx
        g[i_idx] += 2 * a[j_idx]
        step = step0 / math.sqrt(it)
        a_new = a - step * g
        a = _project_simplex(a_new)
        # evaluate
        score = _score_of_seq(a)
        if score < best_score:
            best_score = score; best_seq = a.copy()
            print(...)

    # Return best_seq as list of floats
    if best_seq is None:
        best_seq = np.full(100, 1/100)
    return [float(x) for x in best_seq]

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    # set length
    n = 3000

    best_seq = None
    best_score = float('inf')

    # param search families
    families = []
    # power family
    for p in np.linspace(0.0, 1.5, 31):
        families.append(('power', p))
    # exponential family
    for lam in np.linspace(0.2, 5.0, 30):
        families.append(('exp', lam))
    # sine family
    for freq in np.linspace(1, 4, 10):
        families.append(('sin', freq))

    for typ, param in families:
        if typ == 'power':
            a = np.arange(n, 0, -1, dtype=float) ** param
        elif typ == 'exp':
            a = np.exp(- param * np.arange(n) / (n - 1))
        elif typ == 'sin':
            a = np.sin(np.pi * (np.arange(n) + 0.5) * param / n)  # ensure nonnegative maybe shift
            a = np.maximum(a, 0.0)
        else:
            continue

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            continue
        a = a / s

        score = _score_of_seq(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start:6.2f}s] param {typ}={param:.3f} => score {score:.6f}")

        # optionally break if time low
        if time.time() >= deadline:
            break

    # after param search, refine via gradient descent (subgradient)
    if best_seq is None:
        # fallback uniform
        best_seq = np.full(n, 1./n)

    a = best_seq.copy()
    cur_n = a.shape[0]
    fft_len = 1 << ((2*cur_n - 1).bit_length())
    step0 = 0.8
    mu = 0.9
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline and it < 200000:
        it += 1
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[: 2*cur_n - 1]
        max_val = conv.max()
        cur_score = 2.0 * cur_n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start:6.2f}s] grad iter={it}, score={cur_score:.6f}")

        k = conv.argmax()  # index of maximum
        # Build gradient for this index
        low = max(0, k - (cur_n - 1))
        high = min(cur_n - 1, k)
        i_idx = np.arange(low, high + 1)
        j_idx = k - i_idx
        g = np.zeros_like(a)
        g[i_idx] = 2.0 * a[j_idx]

        step = step0 / math.sqrt(it)
        v = mu * v - step * g
        a_new = a + v
        a = _project_simplex(a_new)

        # occasional random perturb
        if it % 5000 == 0:
            # add small noise
            noise = np.random.randn(cur_n) * 0.01
            a_noisy = a + noise
            a_noisy = np.maximum(a_noisy, 0.0)
            a_noisy = a_noisy / a_noisy.sum()
            # evaluate and maybe replace a
            a_score = _score_of_seq(a_noisy)
            if a_score < cur_score:
                a = a_noisy
                print(f"[{time.time() - start:6.2f}s] noise improvement: score {a_score:.6f}")

    # final output
    return [float(x) for x in best_seq]

"""
Propose a candidate step‑function.

The approach consists of three stages:

1. **Parameter sweep** – generate a few simple families of shapes
   (power‑law, exponential and a sinusoidal window) on a moderate
   length (≈2000).  Each shape is normalised to a probability simplex and
   scored with a fast FFT‑based evaluator.  The best shape from this
   sweep is kept as a starting point.

2. **Projected (momentum) sub‑gradient optimisation** – starting from the
   best shape, minimise the objective
   ``2·n·max(conv(a,a))`` using a sub‑gradient of the *max* convolution
   entry (the index where the convolution attains its maximum).  After
   each step the vector is projected back onto the simplex with the
   Euclidean projection of Duchi et al.  A simple momentum term is added
   to accelerate convergence.  The loop runs until the time budget is
   exhausted or a large iteration limit is reached.

3. **Optional random perturbations** – every few thousand iterations a
   small Gaussian perturbation is tried; if it improves the score the
   current iterate is replaced.

All intermediate improvements are printed.  The function returns the
best sequence found as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """
    Fast evaluation of `evaluate_sequence` for a sequence that already
    sums to 1 (the denominator of `evaluate_sequence` is then 1).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p, normalised to sum 1."""
    vec = np.arange(n, 0, -1, dtype=float) ** p
    s = vec.sum()
    return vec / s if s > 0 else np.full(n, 1.0 / n)


def _make_exp_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i/(n-1)), normalised."""
    if n == 1:
        return np.array([1.0])
    vec = np.exp(-lam * np.arange(n) / (n - 1))
    s = vec.sum()
    return vec / s if s > 0 else np.full(n, 1.0 / n)


def _make_sin_shape(n: int, freq: float) -> np.ndarray:
    """
    a_i ∝ sin(π * freq * (i+0.5) / n), clipped to non‑negative,
    then normalised.
    """
    x = np.arange(n, dtype=float) + 0.5
    vec = np.sin(np.pi * freq * x / n)
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    return vec / s if s > 0 else np.full(n, 1.0 / n)

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:

    Search for a step‑function with a low `evaluate_sequence` score.
    The routine runs for at most ``budget_s`` seconds and returns the
    best sequence (list of non‑negative floats) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # choose a working length (large enough to be expressive, but
    # still cheap for many FFTs)
    n = 2400
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1. initialise from any previously discovered sequence
    try:
        init = np.array(height_sequence_1, dtype=float)

        if init.sum() > 0:

        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] Init from saved sequence, "
              f"n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        # no prior data – fall back to uniform
        best_seq = np.full(n, 1.0 / n)
        best_score = _score_of_seq(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] No saved data – uniform init, "
              f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. coarse parameter sweep over simple families
    families = []

    # power‑law shapes
    for p in np.linspace(0.0, 1.6, 33):         # 0 … 1.6
        families.append(("power", p))

    # exponential shapes
    for lam in np.linspace(0.2, 5.0, 28):       # 0.2 … 5
        families.append(("exp", lam))

    # sinusoidal windows (non‑negative half‑wave)
    for freq in np.linspace(1.0, 4.0, 13):      # 1 … 4
        families.append(("sin", freq))

    # a few random Dirichlet seeds
    for _ in range(12):
        families.append(("rand", None))

    for typ, param in families:
        if time.time() >= deadline:
            break

        if typ == "power":
            a = _make_power_shape(n, param)
        elif typ == "exp":
            a = _make_exp_shape(n, param)
        elif typ == "sin":
            a = _make_sin_shape(n, param)
        elif typ == "rand":
            a = np.random.rand(n)
            a = a / a.sum()
        else:
            continue

        score = _score_of_seq(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()

                  f"score={score:.6f}")

    # --------------------------------------------------------------
    # 3. projected sub‑gradient optimisation (single max index, momentum)
    a = best_seq.copy()
    cur_n = a.shape[0]
    fft_len = 1 << ((2 * cur_n - 1).bit_length())

    step0 = 0.9          # initial step size
    mu = 0.9             # momentum coefficient
    velocity = np.zeros_like(a)

    it = 0
    max_iters = 250_000

    while time.time() < deadline and it < max_iters:
        it += 1

        # compute convolution via FFT
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * cur_n - 1]

        cur_max = conv.max()
        cur_score = 2.0 * cur_n * cur_max

        # record improvement
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Improvement iter={it}, "
                  f"score={cur_score:.6f}")

        # index of the (first) maximal convolution entry
        k = int(conv.argmax())

        # gradient of the active quadratic form
        low = max(0, k - (cur_n - 1))
        high = min(cur_n - 1, k)
        i_idx = np.arange(low, high + 1)
        j_idx = k - i_idx
        g = np.zeros_like(a)
        g[i_idx] = 2.0 * a[j_idx]

        # diminishing step size
        step = step0 / math.sqrt(it)

        # momentum update and projection
        velocity = mu * velocity - step * g
        a_new = a + velocity
        a = _project_simplex(a_new)

        # occasional random perturbation to escape flat regions
        if it % 5000 == 0:
            noise = np.random.randn(cur_n) * 0.008
            cand = a + noise
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand = cand / s
            cand_score = _score_of_seq(cand)
            if cand_score < cur_score:
                a = cand
                velocity = np.zeros_like(a)   # reset momentum
                print(f"[{time.time() - start_time:6.2f}s] Noise acceptance iter={it}, "
                      f"score={cand_score:.6f}")

    # --------------------------------------------------------------
    # 4. final sanity check (optional re‑evaluation with the provided
    #    `evaluate_sequence` function)
    if time.time() < deadline:
