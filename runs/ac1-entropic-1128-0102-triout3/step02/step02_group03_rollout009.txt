"""
Hybrid projected Adam‑gradient / targeted mass‑transfer search.

The objective for a non‑negative step‑function `a` with fixed ℓ₁‑norm
`S = sqrt(2·n)` equals the maximal entry of its self‑convolution
(`max(conv)`).  The routine keeps the iterate on the simplex
{a ≥ 0, Σa = S} by Euclidean projection and performs three kinds of
updates:

* **Adam‑style gradient step** – exact sub‑gradient of the max‑convolution
  (obtained from the active set) plus a small diffusion term
  (`‑Δ·laplacian(a)`) that spreads mass and helps escape plateaux.
* **Targeted mass‑transfer** – identifies the convolution index that attains
  the current maximum, finds the pair of entries that contribute most to
  that peak and moves a tiny fraction of mass from the larger contributor
  to a random location.
* **Random mass‑transfer** – a simple simulated‑annealing style move.

Updates are accepted greedily or with a Metropolis probability that
decreases according to a temperature schedule.  Stagnation triggers a
small random perturbation of the whole vector.  The best sequence seen
throughout the run is returned at the end (after a final call to
`evaluate_sequence` for verification).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full (2·len(a)‑1) result (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_max_subgrad(a: np.ndarray, mask: np.ndarray, nfft: int) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)[k]  at the current active set.
    `mask` is a binary array (length 2·n‑1) with 1s where the convolution
    attains its current maximum.
    Returns a gradient of length n.
    """
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros_like(a)

    w = w / w_sum                     # normalise to a probability vector
    a_rev = a[::-1]

    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    # slice that corresponds to the gradient entries
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _laplacian(a: np.ndarray) -> np.ndarray:
    """
    Discrete Laplacian (second difference) of a.
    Boundary handling uses zero‑padding (the same as np.convolve(...,mode='same')).
    """
    return np.convolve(a, [1.0, -2.0, 1.0], mode='same')


def _gaussian_kernel(size: int = 11, sigma: float = 2.0) -> np.ndarray:
    """Return a normalised 1‑D Gaussian kernel."""
    radius = size // 2
    x = np.arange(-radius, radius + 1, dtype=float)
    k = np.exp(-0.5 * (x / sigma) ** 2)
    return k / k.sum()


def _smooth(a: np.ndarray, kernel: np.ndarray) -> np.ndarray:
    """One step of Gaussian smoothing (same‑mode convolution)."""
    return np.convolve(a, kernel, mode='same')


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Hybrid projected Adam‑gradient / targeted move search for a low‑max
    self‑convolution step function.  Returns the best sequence found
    before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # problem size, target ℓ₁‑norm and FFT length
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=np.float64)
        n = a0.size
    else:
        n = 2000
        a0 = None

    S = np.sqrt(2.0 * n)                         # ℓ₁‑norm we keep fixed
    nfft = _next_pow2(2 * n - 1)

    # initialise the iterate
    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        a = np.full(n, S / n, dtype=np.float64)  # uniform start

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyper‑parameters
    base_lr = 0.3 * S / n                       # similar magnitude to original code
    temperature = 0.12
    temp_decay = 0.9995
    min_temp = 1e-6

    p_grad = 0.60          # probability of a gradient (Adam) step
    p_targeted = 0.25      # probability of a targeted mass‑transfer
    # remainder (0.15) -> random mass‑transfer

    eps_mask = 1e-12
    diff_coeff = 0.30      # weight of Laplacian diffusion term
    kernel = _gaussian_kernel(size=11, sigma=2.0)   # for occasional smoothing
    smooth_interval = 5000
    smooth_alpha = 0.2

    # bookkeeping
    best_a = a.copy()
    conv = _convolve_self_fft(a, nfft)
    best_score = conv.max()
    no_improve = 0
    max_no_improve = 30000                     # restart if stuck too long
    iteration = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = cur_max                      # because Σa = S ⇒ objective = max(conv)

        # keep the global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:06d}] New best: {best_score:.6f} (max={cur_max:.6f})")
            no_improve = 0
        else:
            no_improve += 1

        # early exit if target bound is reached
        if best_score < 1.503:
            print(f"Target bound reached after {iteration} iterations (best={best_score:.6f})")
            break

        # ------------------------------------------------------------------
        # choose a move type
        # ------------------------------------------------------------------
        r = np.random.rand()
        a_candidate = a.copy()

        if r < p_grad:
            # ---- Adam‑style gradient step ----
            mask = (conv >= cur_max - eps_mask)
            grad = _grad_max_subgrad(a, mask, nfft)

            # add a small diffusion (laplacian) term
            lap = _laplacian(a)
            grad = grad - diff_coeff * lap

            # Adam moments update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            # bias‑corrected estimates
            m_hat = m / (1.0 - beta1 ** (iteration + 1))
            v_hat = v / (1.0 - beta2 ** (iteration + 1))

            lr = base_lr / np.sqrt(iteration + 1)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_candidate = _project_to_simplex(a - step, S)

        elif r < p_grad + p_targeted:
            # ---- Targeted mass‑transfer move ----
            k_max = int(np.argmax(conv))
            idx = np.arange(n)
            jdx = k_max - idx
            valid = (jdx >= 0) & (jdx < n)

            contrib = np.zeros(n, dtype=float)
            contrib[valid] = a[idx[valid]] * a[jdx[valid]]

            src = int(np.argmax(contrib))
            if a[src] > 0:
                # move a small fraction of the source mass
                dest = src
                while dest == src:
                    dest = np.random.randint(n)
                delta = 0.05 * a[src]               # 5 % of source
                a_candidate[src] -= delta
                a_candidate[dest] += delta
                a_candidate = _project_to_simplex(a_candidate, S)
            else:
                # fall back to a random move if source mass is zero
                src = np.random.choice(n, p=a / a.sum())
                dest = src
                while dest == src:
                    dest = np.random.randint(n)
                delta = 0.08 * a[src] if a[src] > 0 else 0.08 * S / n
                a_candidate[src] -= delta
                a_candidate[dest] += delta
                a_candidate = _project_to_simplex(a_candidate, S)

        else:
            # ---- Simple random mass‑transfer move ----
            if a.sum() > 0:
                src = np.random.choice(n, p=a / a.sum())
            else:
                src = np.random.randint(n)
            dest = src
            while dest == src:
                dest = np.random.randint(n)

            delta = 0.08 * a[src] if a[src] > 0 else 0.08 * S / n
            a_candidate[src] -= delta
            a_candidate[dest] += delta
            a_candidate = _project_to_simplex(a_candidate, S)

        # occasional smoothing to encourage global flatness
        if iteration % smooth_interval == 0 and iteration > 0:
            a_candidate = (1.0 - smooth_alpha) * a_candidate + smooth_alpha * _smooth(a_candidate, kernel)
            a_candidate = _project_to_simplex(a_candidate, S)

        # ------------------------------------------------------------------
        # evaluate candidate (fast convolution, exact objective)
        # ------------------------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        cand_score = cand_max

        # acceptance rule (greedy + Metropolis)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
            cur_max = cand_max
        else:
            # tiny jitter to shake out of shallow plateaux (rare)
            if np.random.rand() < 0.005:
                jitter = np.random.randn(n) * (base_lr * 0.2)
                a = _project_to_simplex(a + jitter, S)

        # temperature decay
        temperature = max(temperature * temp_decay, min_temp)

        # restart / perturb if no improvement for a long stretch
        if no_improve >= max_no_improve:
            # small random perturbation of the whole vector (not a full restart)
            a = a + np.random.randn(n) * (S * 0.001)
            a = _project_to_simplex(a, S)
            m[:] = 0.0
            v[:] = 0.0
            no_improve = 0
            print(f"[{iteration:06d}] Stagnation perturbation applied.")

        iteration += 1

    # ------------------------------------------------------------------
    # final verification and return
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()
