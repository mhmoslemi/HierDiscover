"""
propose_candidate
-----------------

Search for a non‑negative step function (sequence of heights) that minimises
the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the linear convolution.  Because the objective is
homogeneous, we enforce the L1‑normalisation Σ a_i = sqrt(2·n) (the same
normalisation used in the reference code).  The search proceeds in two
phases:

1. **Cheap parametric sweep** – a handful of simple families (triangular,
   power‑law, exponential, Gaussian) are sampled on a coarse grid.
   Each candidate is normalised and its maximal self‑convolution is
   measured by an FFT based routine.  The best candidate from this sweep
   becomes the starting point for the next phase.

2. **Refinement with mirror‑descent on the simplex** – we optimise the log‑
   variables (a_i = exp(theta_i)) using Adam while projecting back onto
   the L1‑simplex after every step.  A sub‑gradient of the exact
   max‑convolution is used (no soft‑max smoothing).  Occasional jitter
   and a restart‑on‑stagnation mechanism help escape local minima.

The whole routine respects the 1000 s time‑budget and prints progress
information.  The returned list is the best sequence found (as plain
Python floats).  All helper functions are defined at the top level so
that there are no closures.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# basic linear‑algebra helpers (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _max_self_convolution(a: np.ndarray, nfft: int) -> float:
    """
    Return max entry of the linear self‑convolution a * a.
    Uses an FFT of length ``nfft`` (which should be ≥ 2·len(a)‑1).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return float(conv.max())


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the (unique) index ``k_max``.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto  {x ≥ 0, Σ x = target_sum}.
    Implements the classic O(n) algorithm of Duchi et al.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

        conv = np.fft.irfft(np.fft.rfft(a, nfft) * np.fft.rfft(a, nfft),
                            n=nfft)[: 2 * n - 1]
        cur_score = float(conv.max())

        if cur_score < best_score - 1e-12:
            best_score, best_a = cur_score, a.copy()
            no_improve = 0
            print(f"[{iteration:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # early stop if we are already under the target
        if best_score < 1.503:
            print(f"[{iteration}] target reached (score={best_score:.6f})")
            break

        # sub‑gradient of the max entry
        k_max = int(conv.argmax())
        grad_a = _grad_max(a, conv, k_max)

        # chain rule for log‑variables
        grad_log = grad_a * a

        # Adam update (descent)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.003:
            jitter = np.random.randn(n) * (eta * 0.07)
            log_theta += jitter

        # restart if stagnation is observed
        if no_improve >= restart_limit:
            print(f"[{iteration}] restart after {no_improve} non‑improving iters")
            a = best_a.copy()
            perturb = np.random.randn(n) * (eta * 0.2)
            a = a * np.exp(perturb)
            a = a * (S / a.sum())
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # ------------------------------------------------------------------
    # final sanity check (evaluate via the provided function)
    # ------------------------------------------------------------------
    final_seq = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iters. Final score = {final_score:.6f}")

    # Return plain Python list (required by the evaluator)
    return final_seq.tolist()
