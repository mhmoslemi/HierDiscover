"""
Propose a candidate step‑function by (i) progressive up‑sampling and (ii)
smooth‑max gradient descent on the probability simplex.
The objective 2·n·max(conv(a,a)) is invariant under scaling, so we enforce
∑a_i = 1 and minimise the smooth approximation
    F_α(a) = 2·n·log‑sum‑exp(α·conv(a,a))/α .
A linear schedule drives α from a modest value (≈1) to a large one
(≈80–120), smoothly turning the gradient of the smooth objective into a
sub‑gradient of the original max‑objective.  Adam with Euclidean simplex
projection yields fast convergence.  The procedure repeatedly up‑samples
the current best shape (≈1.5× larger) up to a configurable maximum length,
always keeping the best sequence seen so far.
"""

import time
import math
import numpy as np

# -------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_of_seq(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective used in ``evaluate_sequence``.
    Works for any scaling (the score is invariant under scaling).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    L = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    return 2.0 * n * max_val / (s * s)

        init_len = 1200

        print(f"Random initialisation: n={init_len}")

    # 2. Progressive up‑sampling schedule.
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_n = cur_seq.shape[0]
    MAX_LEN = 6000                # maximal length we will try
    factor = 1.5                  # growth factor per stage

    target_lengths = []

    for target_n in target_lengths:
