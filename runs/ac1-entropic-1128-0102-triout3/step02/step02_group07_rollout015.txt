def softmax_gradient(a, conv, alpha):
    max_val = conv.max()
    # compute weight exp; subtract max to avoid overflow
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum
    # reverse w and a
    w_rev = w[::-1]
    a_rev = a[::-1]
    # convolution
    grad_full = conv_fft2(w_rev, a_rev)
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * grad_full[start : start + n]
    return grad
```

"""
Propose a candidate step‑function sequence for ``evaluate_sequence``.

Algorithm
---------

1. **Problem size** – Use the length of the previously discovered
   ``height_sequence_1`` if it exists, otherwise fall back to ``n = 2500``.
2. **Family sweep** – Explore three simple parametric families
   (power‑law, centred Gaussian, symmetric triangular).  For each
   candidate sequence we evaluate the fast objective
   ``2·n·max(conv(a,a))`` (the full ``evaluate_sequence`` is called only
   for the final best vector).  The best shape found in this stage is
   used as a seed for the next phase.
3. **Smoothed‑max gradient descent** – Starting from the best seed we
   minimise a smooth approximation of the ``max`` using the
   log‑sum‑exp trick:
       fα(a)= (1/α)·log Σk exp(α·(a∗a)k)
   Its gradient can be computed by a single FFT‑based convolution.
   After each step the iterate is projected onto the probability
   simplex (non‑negative, Σa=1).  The step‑size, the smoothing
   parameter ``α`` and a Metropolis temperature are annealed during the
   run.
4. **Random restarts** – If time permits, a few additional runs are
   started from random Dirichlet seeds and the globally best vector is
   kept.
5. **Sanity check** – The final candidate is verified with the official
   ``evaluate_sequence`` before being returned.

The routine respects a 1000 s wall‑clock budget (default 950 s are used
internally) and prints progress information.  All helper functions are
defined at module level, have no closures and use only ``numpy``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i = (i+offset)^(-p), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def triangular_sequence(n: int) -> np.ndarray:
    """Symmetric triangle: a_i = min(i+1, n‑i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    total = a.sum()
    a /= total
    return a


def conv_fft2(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x)
    m = len(y)
    size = n + m - 1
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    return np.fft.irfft(X * Y, n=size)


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a∗a (full length)."""
    return conv_fft2(a, a)


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    Duchi‑et‑al. O(n log n) algorithm.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth max fα(a)= (1/α)·log Σ exp(α·(a∗a)k)
    w.r.t. a.  Uses the identity
         ∂(a∗a)k/∂a_i = 2·a_{k‑i}
    and FFT‑based convolution for efficiency.
    """
    max_val = conv.max()
    # stabilized weights
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * Σ_k w_k * a_{k‑i}
    # Equivalent to 2 * (rev(w) ∗ rev(a)), taking the middle n entries.
    w_rev = w[::-1]
    a_rev = a[::-1]
    full = conv_fft2(w_rev, a_rev)          # length = 3·n‑2
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def fast_score(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a,a)) – the numerator of the official score (sum = 1)."""
    conv = conv_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns the best list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
        base = base / base.sum()
        print(f"[info] using provided height_sequence_1 of length {n}")
    else:
        n = 2500
        base = None
        print(f"[info] no external sequence – using n = {n}")

    # ----- time management ---------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ----- bookkeeping --------------------------------------------------
    best_a = None
    best_score = np.inf
    best_descr = ""

    # ----- helper to possibly update the global best --------------------
    def try_update(candidate: np.ndarray, descr: str) -> None:
        nonlocal best_a, best_score, best_descr
        sc = fast_score(candidate, n)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            best_descr = descr
            print(f"[update] new global best: {descr}, score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 1. Family sweep
    # ------------------------------------------------------------------
    print("[search] starting parametric family sweep")

    # --- power‑law -----------------------------------------------------
    offsets = [0.5, 1.0, 2.0]
    p_grid = np.arange(0.40, 0.91, 0.01)
    for off in offsets:
        for p in p_grid:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=off)
            try_update(a, f"power p={p:.3f} off={off}")

    # --- Gaussian -------------------------------------------------------
    sigma_factors = np.linspace(0.10, 0.55, 46)   # inclusive
    for sf in sigma_factors:
        if time.time() > deadline:
            break
        sigma = sf * n
        a = gaussian_sequence(n, sigma)
        try_update(a, f"gauss σ={sigma:.1f}")

    # --- Triangular (single candidate) ---------------------------------
    a = triangular_sequence(n)
    try_update(a, "triangular")

    # If somehow no candidate was produced, fall back to uniform.
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = fast_score(best_a, n)
        best_descr = "uniform fallback"
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 2. Gradient descent (soft‑max smoothing)
    # ------------------------------------------------------------------
    def run_gradient(start_vec: np.ndarray,
                     max_iters: int,
                     name: str) -> None:
        """Perform projected gradient descent from ``start_vec``."""
        nonlocal best_a, best_score, best_descr

        a = start_vec.copy()
        cur_score = fast_score(a, n)

        # hyper‑parameters
        eta = 0.5                # step‑size multiplier
        decay_factor = 0.90
        decay_every = 2000
        temp = 0.05              # Metropolis temperature
        alpha = 2000.0           # soft‑max smoothing
        alpha_inc = 0.5          # increase per iteration
        alpha_max = 20000.0

        it = 0
        while time.time() < deadline and it < max_iters:
            conv = conv_fft(a)
            grad = softmax_gradient(a, conv, alpha)

            step = eta * grad
            # keep a safe distance from the boundaries
            step = np.minimum(step, a * 0.9)
            a_trial = a - step

            # occasional tiny jitter to avoid plateaus
            if rng.random() < 0.002:
                a_trial += (rng.random(n) - 0.5) * 1e-6

            a_trial = project_onto_simplex(a_trial)
            trial_score = fast_score(a_trial, n)

            # record any improvement
            if trial_score < best_score:
                best_score = trial_score
                best_a = a_trial.copy()
                best_descr = f"{name} + grad"
                print(f"[gd] iter {it:07d} new best (global) score = {best_score:.6f}")

            # acceptance rule
            if trial_score <= cur_score:
                a = a_trial
                cur_score = trial_score
            else:
                delta = trial_score - cur_score
                if temp > 0.0 and rng.random() < np.exp(-delta / temp):
                    a = a_trial
                    cur_score = trial_score
                # otherwise keep the old a

            # annealing of hyper‑parameters
            if (it + 1) % decay_every == 0:
                eta = max(eta * decay_factor, 1e-6)
                temp = max(temp * decay_factor, 1e-8)
                alpha = min(alpha + alpha_inc * decay_every, alpha_max)

            it += 1

    # Run a longer descent from the best family candidate
    print("[gd] starting main gradient descent")
    run_gradient(best_a, max_iters=250_000, name="main")

    # ------------------------------------------------------------------
    # 3. Random restarts (if time permits)
    # ------------------------------------------------------------------
    restarts = 3
    iters_per_restart = 80_000
    for r in range(restarts):
        if time.time() >= deadline:
            break
        a0 = rng.dirichlet(np.ones(n))          # random point on simplex
        print(f"[restart] random start {r+1}/{restarts}")
        run_gradient(a0, max_iters=iters_per_restart, name=f"restart{r+1}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print("[done] best reported score = {:.6f}".format(final_score))
    print("[info] description of best vector :", best_descr)
    print("[info] length =", n)

    return final_seq
