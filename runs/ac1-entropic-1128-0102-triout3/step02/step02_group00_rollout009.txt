"""Multi‑scale soft‑max projected gradient + annealing.

The objective 2·n·max(conv(a,a)) is scale‑invariant, so we work on the
probability simplex (∑a_i = 1).  For a given length *n* we perform a
gradient descent where the non‑smooth max is replaced by a soft‑max
approximation (temperature β).  β is increased from a small start value
to a large one, giving a smooth landscape early on and concentrating on
the true maximum later.  Each step uses a Euclidean projection onto the
simplex and a simple momentum term.  After the gradient phase a cheap
simulated‑annealing refinement moves small amounts of mass between
coordinates.  The procedure is repeated on an increasing sequence of
lengths, up‑sampling the best candidate each time.

The algorithm stays within the 1000 s budget and returns a list of
non‑negative heights (a step‑function) that aims to lower the evaluation
score below the previous best (≈1.514)."""

# ------------------------------------------------------------
# Projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear self‑convolution a*a (non‑circular) via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
    return conv

# ------------------------------------------------------------
def _conv_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    m = len(x) + len(y) - 1
    L = 1 << (m - 1).bit_length()
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    conv = np.fft.irfft(X * Y, L)[:m]
    return conv

# ------------------------------------------------------------
def _softmax_subgradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Returns the gradient of the smooth max approximation
    fβ(a)=log(∑_k exp(β·(a*a)[k])) up to a constant factor.
    The true sub‑gradient of the max is recovered when β→∞.
    """
    n = a.shape[0]

    # weights are numerically safe because conv-maxc ≤ 0

    idx = (2 * n - 2) - np.arange(n)  # pick the slice that gives g_i
    g = 2.0 * full[idx]
    return g
