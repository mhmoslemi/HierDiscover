"""
Multi‑scale Adam‑based projected gradient descent + diverse seed generation
and a final simulated‑annealing polishing step.

The optimisation is performed on the probability simplex (the scale of the
vector does not matter for the objective).  For each length we:

1. Build a small pool of promising seeds (uniform, power‑law, symmetric Beta,
   sin², a few random Dirichlet draws and an up‑sampled version of the best
   solution found so far).
2. Keep the best few seeds (by the target score 2·n·max(conv)).
3. Refine each seed with an Adam optimiser that uses a sub‑gradient built from
   all convolution entries whose value is within a relative tolerance of the
   current maximum (this is a smooth approximation of the ‑∞‑norm).
4. Occasionally inject a tiny Gaussian perturbation to escape plateaus.
5. After the gradient phase run a cheap simulated‑annealing local search.
6. The best sequence of a given length is up‑sampled and used as a seed for
   the next (larger) length.

The code respects the 1000 s time budget and always returns a non‑negative
list (the sum is normalised to 1, which is irrelevant for the evaluation
function because the score is scale‑invariant).
"""
import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – O(n) algorithm
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear convolution a * a (non‑circular) via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Active set: indices whose convolution value is close to the maximum
# ------------------------------------------------------------
def _active_set(conv: np.ndarray, rel_tol: float = 0.02, min_K: int = 5) -> np.ndarray:
    """Return indices where conv >= (1‑rel_tol)·max(conv).  Guarantees at least
    min_K entries by falling back to the K‑largest indices."""
    max_val = conv.max()
    if max_val <= 0:
        # degenerate case – just take the first few indices
        return np.arange(min_K, dtype=int)
    thresh = max_val * (1.0 - rel_tol)
    idx = np.where(conv >= thresh)[0]
    if idx.size >= min_K:
        return idx.astype(int)
    # fallback: simply take the K largest entries
    K = max(min_K, int(rel_tol * conv.size) + 1)
    if K >= conv.size:
        return np.arange(conv.size, dtype=int)
    idx = np.argpartition(-conv, K - 1)[:K]
    return idx.astype(int)


# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution (average over the active set)
# ------------------------------------------------------------
def _subgradient_from_active(a: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of the entries of a*a indexed by act_idx."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1)
        j = k - i
        # each pair (i,j) contributes 2·a[j] to ∂(a*a)[k]/∂a[i]
        g[i] += 2.0 * a[j]
    if act_idx.size > 0:
        g /= act_idx.size
    return g


# ------------------------------------------------------------
# Adam optimiser on the simplex (projected after each step)
# ------------------------------------------------------------
def _run_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    max_iter: int = 20000,
    rel_tol_start: float = 0.10,
    rel_tol_end: float = 0.01,
    min_K: int = 5,
    alpha0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the simplex using a tolerance‑based sub‑gradient."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, "
                f"iter={it}, score={cur_score:.6f}"
            )

        # adaptive relative tolerance (shrinks linearly)
        rel_tol = max(rel_tol_end, rel_tol_start * (1.0 - it / max_iter))
        active = _active_set(conv, rel_tol=rel_tol, min_K=min_K)
        grad = _subgradient_from_active(a, active)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha0 / (np.sqrt(v_hat) + eps)   # per‑coordinate step size
        a = _project_simplex(a - step * m_hat)

        # occasional stochastic perturbation to escape plateaus
        if it % 5000 == 0:
            noise = np.random.randn(n) * 0.02
            a = _project_simplex(a + noise)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated annealing (tiny “move ε from i to j’’) for polishing
# ------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 20000,
) -> tuple[np.ndarray, float]:
    """Local search by moving a random fraction of mass between two indices."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a small mass transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # up to 30 % of the source weight (larger jumps than the previous code)
        eps = a[i] * np.random.rand() * 0.30
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, "
                f"iter={it}, score={score:.6f}"
            )
    return best_seq, best_score


# ------------------------------------------------------------
# Upsample a probability vector by linear interpolation (preserves non‑negativity)
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ------------------------------------------------------------
# Candidate seeding strategies (parameterised shapes)
# ------------------------------------------------------------
def _candidate_shapes(n: int) -> list[np.ndarray]:
    """Return a modest list of diverse, normalised candidate vectors."""
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # symmetric power‑law shapes  a_i ∝ ((i+½)(n‑½‑i))^γ
    for gamma in (-1.5, -1.0, -0.5, 0.0, 0.5, 1.0, 1.5, 2.0):
        i = np.arange(n)
        pos = (i + 0.5) * (n - 0.5 - i)
        if gamma == 0.0:
            shape = np.ones_like(pos)
        else:
            # protect against overflow/underflow
            shape = np.power(pos, gamma, where=pos > 0, out=np.zeros_like(pos, dtype=float))
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        candidates.append(shape)

    # symmetric Beta(α,α) PDFs (including U‑shaped cases)
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        t = np.linspace(0.0, 1.0, n)
        # avoid infinities at the exact ends when α<1
        with np.errstate(divide='ignore', invalid='ignore'):
            shape = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
        shape[0] = shape[-1] = 0.0
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        candidates.append(shape)

    # sinusoidal (positive) shape
    sin_shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    sin_shape /= sin_shape.sum()
    candidates.append(sin_shape)

    # a few random Dirichlet draws
    for _ in range(5):
        rnd = np.random.rand(n)
        rnd /= rnd.sum()
        candidates.append(rnd)

    return candidates


# ------------------------------------------------------------
# Main entry point required by the evaluation harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative step heights that (hopefully) gives a low
    score for `evaluate_sequence`.  The algorithm combines multi‑scale up‑sampling,
    diverse seeding, an Adam optimiser on the simplex, and a final simulated
    annealing polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin – we want to finish before the harness kills us
    deadline = start_time + budget_s - 1.0

    best_score = float("inf")
    best_seq = None

    # --------------------------------------------------------------------
    # Optional starter supplied by the harness (if present)
    # --------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(0.0, starter)
            if starter.sum() > 0:
                starter = starter / starter.sum()
    except Exception:
        starter = None

    # Length schedule – geometric growth; we keep it modest to stay within the budget.
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    np.random.shuffle(lengths)          # avoid bias if time runs out early

    # Global best (always kept on the simplex)
    global_a = starter if starter is not None else None

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----------------------------------------------------------------
        # Build a pool of seeds for this length
        # ----------------------------------------------------------------
        seeds = []

        # up‑sample the current global best if we have one
        if global_a is not None:
            up = _upsample(global_a, n)
            seeds.append(up)

        # add a handful of parametric candidates
        seeds.extend(_candidate_shapes(n))

        # ----------------------------------------------------------------
        # Quick pre‑evaluation – keep the few most promising seeds
        # ----------------------------------------------------------------
        scores = []
        for s in seeds:
            conv = _conv_self(s)
            sc = 2.0 * n * conv.max()
            scores.append(sc)
        # keep the three best seeds (or all if fewer)
        order = np.argsort(scores)[: min(3, len(seeds))]
        top_seeds = [seeds[i] for i in order]

        # ----------------------------------------------------------------
        # Gradient‑based refinement (Adam) on each top seed
        # ----------------------------------------------------------------
        # Allocate a slice of remaining time to this length.
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # at most 20 % of the remaining budget for the whole refinement of this n
        slice_deadline = time.time() + max(5.0, 0.20 * remaining)

        for seed_vec in top_seeds:
            # a modest number of iterations; we let the time budget dictate the exact limit
            max_iter = 20000
            # if n is large we keep fewer iterations to stay within the slice
            if n > 4000:
                max_iter = 12000
            a_opt, sc_opt = _run_adam(
                a0=seed_vec,
                n=n,
                deadline=slice_deadline,
                start_time=start_time,
                max_iter=max_iter,
                rel_tol_start=0.12,
                rel_tol_end=0.008,
                min_K=5,
                alpha0=0.6,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after Adam: n={n}, "
                    f"score={sc_opt:.6f}"
                )
            # early exit if we are out of time for this length
            if time.time() >= slice_deadline:
                break

        # ----------------------------------------------------------------
        # Simulated‑annealing polishing (if time permits)
        # ----------------------------------------------------------------
        if best_seq is not None and time.time() < deadline:
            anneal_deadline = time.time() + max(3.0, 0.05 * (deadline - time.time()))
            a_refined, sc_refined = _run_annealing(
                a0=best_seq,
                n=n,
                deadline=anneal_deadline,
                start_time=start_time,
                cur_best_score=best_score,
                cur_best_seq=best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=20000,
            )
            if sc_refined < best_score:
                best_score = sc_refined
                best_seq = a_refined.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n}, "
                    f"score={sc_refined:.6f}"
                )

        # remember the best vector for the next scale
        if best_seq is not None:
            global_a = best_seq.copy()

    # --------------------------------------------------------------------
    # Fallback – should never be required
    # --------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # Convert to plain Python list for the harness
    return [float(x) for x in best_seq]
