"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(a * a) / (sum(a) ** 2)

using a projected Adam optimiser on a smooth approximation of the
max‑self‑convolution.  The soft‑max temperature (α) is increased
gradually so that the optimisation first smooths the objective and
later behaves like a sub‑gradient descent on the true maximum.
Occasional random “mass‑transfer” moves are tried to escape plateaus.
The best sequence encountered during the run is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i >= 0, sum_i x_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_power_law(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to ℓ¹‑norm 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def build_sine(n: int) -> np.ndarray:
    """a_i = sin(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = -np.log(np.random.rand(n))
    x /= x.sum()
    return x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a low‑value sequence within the given time budget.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    finish = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # 1. Choose a problem size.
    # Larger n gives more freedom; 5000 is a good trade‑off.
    n = 5000

    # FFT lengths (next powers of two for speed)
    conv_self_size = 1 << ((2 * n - 1).bit_length())       # for a*a
    corr_size = 1 << ((3 * n - 2).bit_length())           # for correlation a_rev * w

    # --------------------------------------------------------------
    # 2. Prepare a few promising starts.
    starts = []

    # a) previous construction, if available
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                a0 = prev[:n].copy()
            else:
                a0 = np.zeros(n, dtype=np.float64)
                a0[:prev.size] = prev
            a0 = np.maximum(a0, 0.0)
            if a0.sum() > 0:
                a0 /= a0.sum()
                starts.append(a0)
                print("[init] seeded from height_sequence_1")
        except Exception:
            pass

    # b) a few hand‑crafted families
    starts.append(build_power_law(n, p=0.5, offset=0.0))
    starts.append(build_power_law(n, p=0.6, offset=0.5))
    starts.append(build_power_law(n, p=0.7, offset=1.0))
    starts.append(build_sine(n))
    starts.append(random_simplex(n))

    # keep the best of the naive starts as a baseline
    best_seq = None
    best_val = np.inf
    for idx, a in enumerate(starts):
        val = fast_score(a)
        print(f"[init] start {idx:02d} value {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a.copy()

    # --------------------------------------------------------------
    # 3. Optimisation parameters
    lr = 0.08                     # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    max_alpha = 200.0             # final temperature (sharp soft‑max)
    tau_alpha = 6000.0            # controls how fast α grows
    random_move_prob = 0.04       # try a random mass‑transfer this often
    random_move_frac = 0.25       # max fraction of source mass to move

    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    a = best_seq.copy()
    it = 0
    report_interval = 4000

    while time.time() < finish:
        # ---- self‑convolution -------------------------------------------------
        fft_a = np.fft.rfft(a, conv_self_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_self_size)[: 2 * n - 1]
        max_b = conv.max()
        s = a.sum()
        cur_val = float(2.0 * n * max_b / (s * s))

        # keep global best
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            print(f"[iter {it:7d}] new best {best_val:.6f}")

        # ---- soft‑max weights -------------------------------------------------
        # α grows from 0 to max_alpha (exponential schedule)
        alpha = max_alpha * (1.0 - math.exp(-it / tau_alpha))
        # stable soft‑max
        shift = max_b
        w_raw = np.exp(alpha * (conv - shift))
        w = w_raw / w_raw.sum()          # w sums to 1

        # ---- gradient (cross‑correlation w ⋆ a) -------------------------------
        a_rev = a[::-1]                  # reversed view
        fft_w = np.fft.rfft(w, corr_size)
        fft_rev = np.fft.rfft(a_rev, corr_size)
        corr = np.fft.irfft(fft_w * fft_rev, corr_size)
        # slice that gives g_i = 2 * Σ_k w_k a_{k-i}
        grad = 2.0 * corr[n - 1: n - 1 + n]

        # ---- Adam update -----------------------------------------------------
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = proj_simplex(a - step)

        # ---- occasional random hill‑climb move ---------------------------------
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 1e-14:
                delta = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = proj_simplex(cand)
                # quick evaluation using the already computed convolution
                # (re‑compute because cand changed)
                cand_conv = np.fft.irfft(
                    np.fft.rfft(cand, conv_self_size) *
                    np.fft.rfft(cand, conv_self_size),
                    conv_self_size)[: 2 * n - 1]
                cand_val = float(
                    2.0 * n * cand_conv.max() / (cand.sum() ** 2))
                if cand_val < best_val:
                    best_val, best_seq = cand_val, cand.copy()
                    a_candidate = cand      # keep the improvement
                elif np.random.rand() < 0.001:
                    # tiny uphill acceptance to keep diversity
                    a_candidate = cand

        # ---- accept the candidate ------------------------------------------------
        a = a_candidate

        # ---- periodic reporting -------------------------------------------------
        if it % report_interval == 0:
            print(f"[iter {it:7d}] cur {cur_val:.6f} best {best_val:.6f}")

        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    # Ensure the returned sequence obeys the required format (list of floats)
    return best_seq.tolist()
