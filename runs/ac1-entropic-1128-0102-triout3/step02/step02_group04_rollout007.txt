```
def propose_candidate(seed=42, budget_s=950.0):
    rng = np.random.default_rng(seed)
    n = 2500
    # initialize a as sqrt_series (half-binomial)
    a = sqrt_series(n)
    # ensure sum=1
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    # hyperparameters
    eta = 0.3   # base step for exponentiated gradient
    eta_decay = 0.999   # per iteration
    beta = 1.0
    beta_inc = 1.003   # increase slowly
    beta_max = 200.0   # high smoothing up to approx max
    # main loop
    start_time = time.time()
    deadline = start_time + budget_s - 5.0
    it = 0
    while time.time() < deadline:
        # compute convolution
        conv = conv_fft(a)
        max_val = conv.max()
        # compute actual score
        cur_score = 2.0 * n * max_val
        # track best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[iter {it}] new best score={cur_score:.6f}, beta={beta:.2f}")
        # compute smoothed gradient using softmax
        # compute softmax probabilities
        # avoid overflow: subtract max
        cmax = conv.max()
        exp_vals = np.exp(beta * (conv - cmax))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        a_rev = a[::-1]
        # Compute convolution of w with a_rev
        # using FFT for efficiency
        m = len(w) + len(a_rev) - 1
        W = np.fft.rfft(w, n=m)
        A_rev = np.fft.rfft(a_rev, n=m)
        grad_full = np.fft.irfft(W * A_rev, n=m)
        # slice grad_a
        grad_a = 2.0 * grad_full[n-1 : n-1 + n]
        # exponentiated gradient update
        a = a * np.exp(-eta * grad_a)
        # renormalize to sum 1 (project onto simplex)
        a = a / a.sum()
        # clip very small values to avoid underflow
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        # adjust hyperparameters
        eta = max(eta * eta_decay, 1e-5)
        beta = min(beta * beta_inc, beta_max)
        it += 1
        # optional random perturbation with some probability
        if rng.random() < 0.005:
            perturb = (rng.random(n) - 0.5) * 1e-3
            a += perturb
            a = np.maximum(a, 0)
            a = a / a.sum()
        # restart condition: if no improvement for many iterations, random restart
        # but we may skip restarts unless needed
    # final evaluation
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best score={final_score:.6f}")
```

It uses a **soft‑max smoothed objective**, an **Adam optimizer on log‑its** (which automatically keeps the vector on the probability simplex) and a slowly increasing smoothing parameter β to approximate the true max more and more closely.  
The algorithm starts from the previously known half‑binomial sequence (or from the global variable `height_sequence_1` if it is present), performs many cheap FFT‑based updates, injects occasional random perturbations and restarts when progress stalls.

```python
"""
Propose a step‑function sequence that minimizes the evaluation function.

The method works on the probability simplex via a log‑it (soft‑max)
parameterisation and Adam optimisation.  At each iteration we
  * compute the self‑convolution a*a,
  * form a smooth approximation of the max using a soft‑max with
    temperature β (β is increased slowly so the objective becomes
    progressively closer to the true max),
  * obtain the gradient of the smoothed objective (a weighted sum of
    the exact max‑sub‑gradients) via a second FFT,
  * update the log‑its with Adam and map back to the simplex,
  * occasionally inject noise or restart from a random Dirichlet draw.

The approach provides smoother gradients than a pure sub‑gradient on the
hard max, uses adaptive step‑sizes, and converges reliably to a lower
bound (≈ 1.52 for n≈2500) within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities -------------------------------------------------------
# ----------------------------------------------------------------------


def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of the half‑binomial series a_i = C(2i,i)/4**i,
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Fast convolution of x with itself using a real FFT.
    Returns a vector of length 2*len(x)-1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def softmax(v: np.ndarray, beta: float) -> np.ndarray:
    """
    Numerically stable soft‑max with temperature β.
    """
    if beta == 0.0:
        # plain uniform weights – should not happen in practice
        return np.full_like(v, 1.0 / v.size)
    vmax = v.max()
    exp = np.exp(beta * (v - vmax))
    return exp / exp.sum()


def grad_smoothed_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the β‑soft‑max of the self‑convolution a*a
    w.r.t. the vector a.  The result has shape (n,).

    Uses a second FFT to evaluate   g_i = 2·∑_k p_k·a_{k-i},
    where p_k are the soft‑max probabilities of the convolution.
    """
    n = a.shape[0]

    # ----- convolution a*a ------------------------------------------------
    conv = conv_fft(a)                     # length 2n‑1

    # ----- soft‑max probabilities over the convolution entries -------------
    p = softmax(conv, beta)                # shape (2n‑1,)

    # ----- gradient = 2 * (p ⋆ a_rev)  -------------------------------
    a_rev = a[::-1]
    # Length needed for the full convolution of p (len 2n‑1) and a_rev (len n)
    full_len = len(p) + len(a_rev) - 1    # = 3n‑2
    # Use a power‑of‑two length for speed
    fft_len = 1 << ((full_len - 1).bit_length())

    P = np.fft.rfft(p, n=fft_len)
    AR = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(P * AR, n=fft_len)

    # Slice the central n entries (indices n‑1 … n‑1+n‑1)
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine ----------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that (approximately) minimises

        2 * n * max( conv(a,a) )  with  sum(a)=1.

    Returns the best sequence (as a plain Python list of floats) found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Parameters (tuned for the 1000 s budget)
    # --------------------------------------------------------------
    n = 2500                      # length of the sequence
    max_iter = 10_000_000         # safety upper bound
    eta0 = 0.6                    # initial Adam step size (log‑it scale)
    beta_start = 1.0              # start of the soft‑max smoothing
    beta_max = 400.0              # final smoothing (close to true max)
    beta_inc = 1.001              # multiplicative increase per iteration
    restart_thresh = 180_000      # iterations without improvement → restart
    noise_prob = 0.004            # probability per iteration of adding a tiny log‑it perturbation
    noise_scale = 5e-4            # magnitude of the log‑it noise

    # --------------------------------------------------------------
    # Initialise a (probability vector) --------------------------------
    # Use the global `height_sequence_1` if it exists, otherwise half‑binomial.
    # --------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)            # protect against negative entries
        if a0.sum() < 0.01 or a0.size != n:
            a0 = sqrt_series(n)
    else:
        a0 = sqrt_series(n)

    a0 = a0 / a0.sum()                     # enforce sum = 1

    # Log‑its (unconstrained parameters).  Using a small epsilon avoids log(0).
    eps = 1e-30
    v = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Tracking the best solution seen so far
    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin
    it = 0
    last_improve = 0
    eta = eta0
    beta = beta_start

    while time.time() < deadline and it < max_iter:
        # ----- map logits -> probability vector (simple soft‑max) -----
        max_v = v.max()
        a = np.exp(v - max_v)
        a_sum = a.sum()
        if a_sum == 0.0:                # numerical safeguard
            a = np.full_like(v, 1.0 / n)
        else:
            a = a / a_sum               # now sum(a)=1

        # ----- current objective (hard max) ---------------------------
        conv = conv_fft(a)               # length 2n‑1
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- record best -------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:08d}] new best score = {best_score:.6f}  β={beta:.2f}")

        # ----- gradient of the β‑smoothed max ------------------------
        grad_a = grad_smoothed_max(a, beta)   # shape (n,)

        # ----- convert gradient to log‑it space (chain rule) ---------
        #   ∂f/∂v = a * (∂f/∂a - 〈a,∂f/∂a〉)
        avg = np.dot(a, grad_a)
        grad_v = a * (grad_a - avg)

        # ----- Adam update on logits ---------------------------------
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** it1)
        v2_hat = v2 / (1.0 - beta2 ** it1)

        v -= eta * m_hat / (np.sqrt(v2_hat) + eps_adam)

        # ----- occasional stochastic perturbation --------------------
        if rng.random() < noise_prob:
            v += (rng.random(n) - 0.5) * noise_scale

        # ----- schedule updates ---------------------------------------
        #    – slowly decay the Adam step size
        if (it + 1) % 4000 == 0:
            eta = max(eta * 0.94, 1e-5)

        #    – increase the soft‑max temperature β → tighter approximation
        beta = min(beta * beta_inc, beta_max)

        # ----- restart if stuck --------------------------------------
        if it - last_improve > restart_thresh:
            # fresh random Dirichlet start (more exploration)
            a_new = rng.dirichlet(np.full(n, 0.6))
            v = np.log(a_new + eps)
            m.fill(0.0)
            v2.fill(0.0)
            last_improve = it
            print(f"[iter {it:08d}] restart triggered – random start")

        it += 1

    # ------------------------------------------------------------------
    # Final evaluation with the provided routine (extra safety)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    elapsed = time.time() - start_time
    print(f"[done] iterations={it}, elapsed={elapsed:.1f}s, best reported score={final_score:.6f}")

    return best_a.tolist()
```
