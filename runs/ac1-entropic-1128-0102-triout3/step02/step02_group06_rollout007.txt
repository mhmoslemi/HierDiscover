def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # initialize sequence length
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = init_seq.size
        # make odd length if even: if n0 % 2 == 0: extend with zero
        if n0 % 2 == 0:
            init_seq = np.append(init_seq, 0.0)
            n0 += 1
    else:
        n0 = 2401
        init_seq = np.random.dirichlet(np.full(n0, 0.5)).astype(np.float64)

    # ensure sum normalization
    def normalize(a):
        S = np.sqrt(2.0 * a.size)
        sum_a = a.sum()
        if sum_a == 0.0:
            return a * (S / a.size) # uniform? Use fallback
        return a * (S / sum_a)

    init_seq = normalize(init_seq)

    # Stage 1: optimize at initial size
    best_a, best_score = _optimize_simple(init_seq, deadline)

    # Stage 2: if time left, upsample and refine
    if time.time() < deadline - 5.0:
        # upsample to about 2x length
        n1 = best_a.size
        n2 = 2 * n1 - 1
        # ensure n2 odd
        if n2 % 2 == 0:
            n2 += 1
        # linear interpolation
        x_old = np.linspace(0, 1, n1)
        x_new = np.linspace(0, 1, n2)
        up_seq = np.interp(x_new, x_old, best_a)
        up_seq = normalize(up_seq)
        # run optimizer on up_seq with remaining time
        best_a2, best_score2 = _optimize_simple(up_seq, deadline)
        # keep better
        if best_score2 < best_score:
            best_a = best_a2
            best_score = best_score2

    # final sanity: maybe scale to sqrt(2*len) again (already)
    final_seq = best_a * (np.sqrt(2.0 * best_a.size) / best_a.sum())
    final_seq = np.clip(final_seq, 0.0, 1000.0)  # ensure within bounds

    # report
    try:
        final_score = evaluate_sequence(final_seq.tolist())
    except Exception as e:
        final_score = float('inf')
    print(f"Final score: {final_score:.6f}")
    return final_seq.tolist()

def _optimize_simple(init_seq, deadline):

"""
Propose a step‐function (non‑negative vector) minimising
    evaluate_sequence(sequence) = 2·n·max(a*a)/(∑a)² .

The algorithm works in two stages:
1️⃣  Optimise a moderately sized sequence (default ≈2401 entries) with
    Adam on log‑variables.  A soft‑max smoothing of the max‑convolution
    is annealed, after which the exact sub‑gradient of the maximal entry
    is used.  The iterate is always projected onto the simplex
    Σa = √(2·n) and symmetrised (a[i]=a[n‑1‑i]) – the optimum is known to
    be symmetric.
2️⃣  If time remains, the best vector is linearly up‑sampled (≈2× longer)
    and refined again under the same optimiser.  The longer vector has
    more degrees of freedom, allowing a lower maximal self‑convolution
    and therefore a lower final score.

The routine respects the 1000 s budget, prints progress and finally
returns the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT based convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the first (2·len(a)‑1) entries (real values).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    Implements a numerically stable soft‑max.
    """
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    # keep only the part that corresponds to the original vector
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the index k_max.
    The derivative of (a*a)[k] w.r.t. a_i equals 2·a_{k‑i}
    when the index is inside the support, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


# ----------------------------------------------------------------------
# Core optimiser (runs until a supplied deadline)
# ----------------------------------------------------------------------


def _optimise_fixed_length(init_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑based mirror‑descent on the simplex for a *fixed* length.
    Returns the best vector (already normalised to Σa = √(2n))
    and its maximal self‑convolution value.
    """
    eps = 1e-12
    a = init_a.copy()
    n = a.size
    S = np.sqrt(2.0 * n)                # target ℓ₁‑norm

    # enforce the simplex at the very start
    a = a * (S / a.sum())
    # enforce symmetry – the optimum is known to be symmetric
    a = (a + a[::-1]) / 2.0
    a = a * (S / a.sum())
    log_theta = np.log(a + eps)

    # Adam state
    m = np.zeros_like(log_theta)
    v = np.zeros_like(log_theta)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.30                     # learning rate in log‑space

    # FFT planning
    nfft = _next_pow2(2 * n - 1)

    # soft‑max temperature schedule
    tau = 2.0
    tau_min = 0.005
    # we anticipate a few hundred thousand updates; decay is exponential
    total_est = 300_000.0
    tau_decay = (tau_min / tau) ** (1.0 / total_est)

    # bookkeeping
    conv = _convolve_self_fft(a, nfft)
    best_score = conv.max()
    best_a = a.copy()
    no_improve = 0
    restart_limit = 6000
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ---- reconstruct a from log variables and re‑impose simplex & symmetry ----
        a = np.exp(log_theta)
        s = a.sum()
        if s == 0.0:
            # degenerate numerics – re‑initialise from the current best
            a = best_a.copy()
            a = a * (S / a.sum())
        else:
            a = a * (S / s)

        # symmetrise
        a = (a + a[::-1]) / 2.0
        a = a * (S / a.sum())          # re‑normalise after symmetrisation

        # ---- evaluate current objective ----
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()

        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
            print(f"[Iter {iteration:07d}] New best max = {best_score:.6f}")
        else:
            no_improve += 1

        # early stop if we have already reached the target
        if best_score < 1.503:
            print(f"Target score reached at iteration {iteration}")
            break

        # ---- gradient of the (smoothed) max convolution ----
        if tau > 5 * tau_min:
            grad_a = _grad_softmax(a, conv, tau, nfft)
        else:
            k_max = int(conv.argmax())
            grad_a = _grad_max(a, conv, k_max)

        # chain rule for log‑variables
        grad_log = grad_a * a

        # ---- Adam update (descent) ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.008:
            jitter = np.random.randn(n) * (base_lr * 0.07)
            log_theta = log_theta + jitter

        # anneal temperature
        tau = max(tau * tau_decay, tau_min)

        # restart if stagnation persists
        if no_improve >= restart_limit:
            # perturb the best solution multiplicatively
            perturb = np.random.randn(n) * (base_lr * 0.25)
            a = best_a * np.exp(perturb)
            a = a * (S / a.sum())
            a = (a + a[::-1]) / 2.0
            a = a * (S / a.sum())
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Iter {iteration:07d}] Restarted after stagnation.")

    # final projection onto the simplex (numerical safety)
    final_a = best_a * (S / best_a.sum())
    final_a = np.clip(final_a, 0.0, 1000.0)   # stay in the evaluator's safe range
    return final_a, best_score

    """
    Search for a step function that minimises `evaluate_sequence`.

    The best sequence found before the time budget expires is returned.
    """
