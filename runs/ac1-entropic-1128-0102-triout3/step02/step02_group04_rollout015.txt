"""
Hybrid optimizer for the step‑function bound.

Phase 1 (small size) – Alternating linear programmes:
    • Start from a short (n≈800) half‑binomial seed.
    • At each iteration compute M = max(a*a).
    • Solve   max Σ g_i   s.t.   a≥0 ,   (a ∗ g)_k ≤ M   (k=0…2n‑2) .
      The constraints are linear in g, so we use a sparse LP (SciPy
      “highs” solver).  The solution g₀ has a huge total mass; after
      normalising we obtain a feasible distribution g = g₀/Σg₀.
    • Fuse the old and the new distribution by a convex combination
      (several trial mixing ratios are examined and the one with the
      smallest new maximum convolution is kept).

Phase 2 (large size) – Projected sub‑gradient descent:
    • Upsample the best small‑n distribution to the target length
      (n≈2500) by linear interpolation and renormalise.
    • Perform many iterations of projected sub‑gradient descent on the
      simplex (the exact sub‑gradient of the peak of a*a is used).  The
      step size decays and occasional random perturbations (plus a
      simple restart policy) keep the search from stalling.

Both phases keep the globally best vector seen so far; after the time
budget is exhausted the best vector is returned as a plain Python list.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# -------------------------------------------------------------
# Helper utilities ------------------------------------------------
def sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑½, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Fast self‑convolution a∗a via a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


# -------------------------------------------------------------
# Linear‑programming sub‑routine (Phase 1)
def solve_lp_step(a: np.ndarray, M: float, row_idx, col_idx) -> np.ndarray:
    """
    Solve   max Σ g_i   s.t.   g_i ≥ 0  and  (a∗g)_k ≤ M.
    Returns the normalised solution g (sum(g)=1).  If the LP fails,
    returns the current a unchanged.
    """
    n = a.shape[0]
    # Data for the sparse matrix representation of the convolution a∗g
    data = np.repeat(a, n)                     # shape (n*n,)
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    # Linear programme: minimise –1ᵀg   (i.e. maximise 1ᵀg)
    c = -np.ones(n, dtype=np.float64)
    ub = M * np.ones(2 * n - 1, dtype=np.float64) * (1.0 + 1e-12)  # tiny slack
    bounds = [(0.0, None)] * n

    try:
        res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                          method="highs", options={"presolve": True})
    except Exception:      # pragma: no cover
        return a.copy()    # fall back to the current vector

    if not res.success:
        return a.copy()

    g0 = res.x
    mass = g0.sum()
    if mass <= 0.0:
        return a.copy()
    g = g0 / mass
    return g

def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine follows the hybrid strategy
    described in the module docstring.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 1️⃣  Phase 1 – small problem, LP based improvement
    n_small = 800                         # modest size for fast LPs
    a = sqrt_series(n_small)               # start from half‑binomial
    # pre‑compute the index pattern of the convolution matrix (independent of a)
    row_idx = (np.arange(n_small)[:, None] + np.arange(n_small)[None, :]).ravel()
    col_idx = np.repeat(np.arange(n_small, dtype=np.int64), n_small)

    best_a_small = a.copy()
    best_score_small = 2.0 * n_small * conv_fft(a).max()

    max_iter_big = 500_000
    eta_start = 0.15
    decay_every = 5000
    eta = eta_start
    no_improve = 0
    no_improve_limit = 80_000
    noise_prob = 0.006
    noise_scale = 8e-4
    start_big = time.time()

    while time.time() < deadline_big and max_iter_big > 0:
        conv_a = conv_fft(a_big)
        M_cur = conv_a.max()

        M_trial = conv_fft(a_trial).max()
        if M_trial < best_M - 1e-12:
            best_M = M_trial
            best_a = a_trial.copy()
            best_score = 2.0 * n_big * best_M
            no_improve = 0
            print(f"[phase‑2 iter {max_iter_big:08d}] new best score = "

            # (simulated‑annealing style) to escape shallow plateaus.
            if rng.random() < 0.001:

            no_improve = 0

        max_iter_big -= 1

    # -----------------------------------------------------------------
    # Final evaluation with the provided safety wrapper
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] total iterations (small + big) = {it_small + (500_000 - max_iter_big)}")
    print(f"[done] best reported score = {final_score:.6f}")
    return best_a.tolist()
