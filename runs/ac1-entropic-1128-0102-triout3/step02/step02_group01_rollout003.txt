"""
Propose a candidate step‑function using a continuation method.
The core idea:
* Work on the probability simplex (the coefficients sum to 1).
* Minimise a smooth approximation of the true objective
      f(a) = 2·n·max(conv(a,a))
  by replacing the max with a log‑sum‑exp (temperature β).
  The gradient of the smoothed objective is a simple correlation
  that can be obtained with two FFTs per iteration.
* Start with a low temperature (large smoothing) and increase β
  gradually – this “continuation” helps escape poor local minima.
* After the smooth phase perform a short projected sub‑gradient
  refinement on the original max‑objective.
* The whole procedure is applied repeatedly while progressively
  up‑sampling the current shape (≈1.5× growth) up to a configurable
  maximum length.  The best sequence found at any time is kept.
* Adam is used for adaptive step sizes; each update is projected
  back onto the simplex with the Euclidean projection of Duchi et al.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_of_seq(a: np.ndarray) -> float:
    """Exact evaluation used by the competition (max‑convolution version)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    # numerical noise may produce tiny negative values – clamp
    conv = np.maximum(conv, 0.0)
    return float(2.0 * n * conv.max())


def _smooth_score_and_grad(a: np.ndarray,
                           beta: float,
                           fft_len: int) -> tuple[float, np.ndarray]:
    """
    Smoothed objective (log‑sum‑exp) and its gradient.
    Returns (smooth_score, grad) where grad is the Euclidean gradient
    of the smoothed score (before projection).
    """
    n = a.shape[0]
    # ----- auto‑convolution of a (full length 2n‑1) -----
    Af = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    conv = np.maximum(conv_full, 0.0)          # guard against tiny negatives

    # ----- log‑sum‑exp (smooth max) -----
    max_c = conv.max()
    # use the log‑sum‑exp trick for stability
    exp_vals = np.exp(beta * (conv - max_c))
    sum_exp = exp_vals.sum()
    smooth_score = 2.0 * n * (max_c + np.log(sum_exp) / beta)

    # ----- weights for the gradient (softmax over convolution entries) -----
    w = exp_vals / sum_exp                     # size 2n‑1, sums to 1

    # ----- correlation a * w (full length n + (2n‑1) - 1 = 3n‑2) -----
    # correlation a ⋆ w = conv(a, reverse(w))
    w_rev = w[::-1]
    # reuse the same FFT length (covers 3n‑2)
    Aw_rev = np.fft.rfft(w_rev, n=fft_len)
    corr_full = np.fft.irfft(Af * np.conj(Aw_rev), n=fft_len)
    # linear correlation at non‑negative lags: 0 … n‑1
    m = 2 * n - 1                               # length of w (and w_rev)
    g_corr = corr_full[m - 1 - np.arange(n)]   # slice of length n

    # ----- gradient of the smoothed objective -----
    grad = 4.0 * n * g_corr
    return smooth_score, grad

                print(f"[{time.time() - start_time:6.2f}s] IMPROVED (subgrad) "
                      f"n={n}, sub_it={sub_it}, score={cur_score:.6f}")

            # --- multi‑peak sub‑gradient (near‑max indices) ---
            eps = max_val * eps_rel
            K = np.where(conv >= max_val - eps)[0]
            g = np.zeros_like(a)
            # for each near‑max index k, add its contribution
            for k in K:
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                i_idx = np.arange(low, high + 1, dtype=int)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]

            step = step0 / math.sqrt(sub_it)
            a = _project_simplex(a - step * g)

        # prepare for next up‑sampling stage
        current = a
        cur_n = n

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3. Final refinement on the best length (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        a = best_seq.copy()
        fft_len = 1 << ((3 * n - 2).bit_length())
        step0 = 0.30
        eps_rel = 0.009
        for it in range(1, 15001):
            if time.time() >= deadline:
                break
            Af = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
            conv = np.maximum(conv, 0.0)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] FINAL REFINE "
                      f"n={n}, iter={it}, score={cur_score:.6f}")

            eps = max_val * eps_rel
            K = np.where(conv >= max_val - eps)[0]
            g = np.zeros_like(a)
            for k in K:
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                i_idx = np.arange(low, high + 1, dtype=int)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]
            step = step0 / math.sqrt(it)
            a = _project_simplex(a - step * g)

        # keep the refined version if it improved
        if _score_of_seq(a) < best_score:
            best_seq = a.copy()
            best_score = _score_of_seq(best_seq)

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        # Should never happen – use a tiny uniform fallback
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
