"""
Propose a candidate step‑function using an exponentiated‑gradient
sub‑gradient method with progressive up‑sampling.

The evaluation score (scale‑invariant) is 2·n·max(conv(a,a)) for a
distribution a on the probability simplex.
We minimise the maximal auto‑convolution by iteratively:

* up‑sampling the current shape (linear interpolation) to a larger length,
* repeatedly computing the auto‑convolution via FFT,
* selecting the near‑maximal convolution indices (within a relative
  tolerance eps_rel) as the active set,
* forming the sub‑gradient g_i = 2·∑_{k∈K} a_{k‑i},
* updating the simplex point with a multiplicative (exponentiated‑gradient)
  step a_i ← a_i·exp(‑η·g_i) and re‑normalising,
* slowly reducing the step size η and the tolerance eps_rel.

Momentum‑free exponentiated updates keep the variables non‑negative
automatically and tend to converge faster than plain Euclidean
projection.  The method runs for at most ``budget_s`` seconds,
keeps the best sequence found, and prints intermediate improvements.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _auto_conv_and_grad(a: np.ndarray, eps_rel: float):
    """
    Compute auto‑convolution, its maximum, and a sub‑gradient based on the
    active set {k : conv[k] ≥ max_val – eps_rel·max_val}.
    Returns (max_val, grad) where grad has shape a.shape.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * eps_rel
    # active indices (near‑max)
    K = np.where(conv >= max_val - eps)[0]

    # build the sub‑gradient: g_i = 2·∑_{k∈K} a_{k‑i}
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return max_val, g

    Return a list of non‑negative heights (a step function) with a low
    `evaluate_sequence` score.  The routine runs for at most ``budget_s``
    seconds, progressively upsamples the sequence, and applies an
    exponentiated‑gradient optimiser.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    try:

        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 0:

        else:
            raise ValueError
        print(f"Initialised from height_sequence_1: n={len(init)}")

        init_len = 1500
        init = np.random.rand(init_len)
        init = init / init.sum()

    best_seq = init.copy()
    best_score = _score_of_seq(best_seq)

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling & optimisation -----------------------
    # ------------------------------------------------------------------
    current = best_seq.copy()
    cur_n = current.shape[0]

    # Upper bound for the length – chosen to give enough resolution but
    # still finish comfortably within the time budget.

        max_iter = int(base_iters * 1200 / n) + 15000
        max_iter = max(max_iter, 12000)

        # step‑size schedule (η decays like 1/√t)
        step0 = 0.7
        eps_rel = 0.025          # start with a relatively wide active set
        eps_decay = 0.90         # multiplicative decay of eps_rel every 2000 it.

        # ----- main optimisation loop ---------------------------------
        for it in range(1, max_iter + 1):
            if time.time() >= deadline:
                break

            max_val, grad = _auto_conv_and_grad(a, eps_rel)

            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.2f}s] Improved: n={n} iter={it} score={cur_score:.6f}")

            # exponentiated‑gradient update
            step = step0 / math.sqrt(it)
            a = a * np.exp(-step * grad)
            # re‑normalise onto the simplex (still non‑negative)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / a_sum

            # slowly tighten the active‑set tolerance
            if it % 2000 == 0:
                eps_rel = max(0.004, eps_rel * eps_decay)

        # --------------------------------------------------------------
        # 3. Prepare for the next up‑sampling stage
        # --------------------------------------------------------------
        current = a
        cur_n = n

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final refinement on the best length found
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        step0 = 0.45
        eps_rel = 0.015
        max_refine = 20000
        for it in range(1, max_refine + 1):
            if time.time() >= deadline:
                break
            max_val, grad = _auto_conv_and_grad(a, eps_rel)
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.2f}s] Final refine: n={n} iter={it} score={cur_score:.6f}")

            step = step0 / math.sqrt(it)
            a = a * np.exp(-step * grad)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / a_sum

            if it % 2000 == 0:
                eps_rel = max(0.001, eps_rel * 0.9)

        # keep refinement if it improved
        if _score_of_seq(a) < best_score:
            best_seq = a.copy()
            best_score = _score_of_seq(best_seq)

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform 100‑element sequence
        best_seq = np.full(100, 1.0 / 100)

    # Ensure compliance with the security checks (non‑negative, sum>0)
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() < 1e-12:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
