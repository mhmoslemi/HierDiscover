    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)

        cols[pos:pos+n] = i

    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver='ECOS', warm_start=False, verbose=False, max_iters=50000)
    if g.value is None:
        return None
    else:
        return np.maximum(g.value, 0)

"""
Propose a low‑value step‑function using a smooth‑max projected Adam optimizer.

The objective to minimise is
        2·n·max(a∗a) / (∑a)² .
Because the expression is homogeneous, we optimise on the probability simplex
(∑a = 1, a_i ≥ 0).  The optimiser minimises a differentiable “soft‑max”
approximation of the maximal self‑convolution entry:

    φ_α(a) = log‑sum‑exp( α·(a∗a) )

∂φ_α/∂a_i = 2· Σ_j a_j·p_{i+j},
where p = soft‑max(α·(a∗a)) is a probability vector of length 2n‑1.
The gradient is obtained by a single FFT‑based convolution with a reversed copy
of a.  An Adam‑style update is performed and the result is projected back onto
the simplex.  The smoothing parameter α is increased gradually (continuation
strategy) so that the algorithm first follows a smooth landscape and later
focuses on the true maximum.

The search runs several random restarts (different lengths, power‑law,
exponential or Dirichlet initialisations) and keeps the best sequence seen.
The final sequence is scaled to ‖a‖₁ = √(2 n) – the same normalisation used in
the reference implementation – before being returned.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def build_initial_sequence(n: int,
                           kind: str = "power",
                           p: float = 0.5,
                           offset: float = 0.0,
                           q: float = 0.9,
                           conc: float = 0.7) -> np.ndarray:
    """
    Return a random feasible start vector of length n.

    kind – one of "power", "exp", "dirichlet".
    """
    if kind == "power":
        idx = np.arange(1, n + 1, dtype=np.float64) + offset
        a = 1.0 / np.power(idx, p)
    elif kind == "exp":
        a = q ** np.arange(n, dtype=np.float64)
    else:  # dirichlet‑like random
        a = np.random.gamma(conc, 1.0, size=n)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def smooth_max_gradient(a: np.ndarray, alpha: float,
                        fft_size: int) -> np.ndarray:
    """
    Gradient of the smooth‑max φ_α at point a.

    The computation uses FFTs of size `fft_size` (≥ 3·n‑2) to obtain the
    self‑convolution of a and the convolution a_rev * p.
    """
    n = a.size
    # ----- self‑convolution a∗a (length 2n‑1) -----
    fft_a = np.fft.rfft(a, fft_size)
    conv_full = np.fft.irfft(fft_a * fft_a, fft_size)[: 2 * n - 1]

    # ----- soft‑max weights p (probability vector) -----
    mx = conv_full.max()
    # subtract mx for numerical stability
    w = np.exp(alpha * (conv_full - mx))
    w /= w.sum()

    # ----- gradient = 2 * (a_rev ∗ w) (extract middle n entries) -----
    a_rev = a[::-1]
    fft_rev = np.fft.rfft(a_rev, fft_size)
    fft_w = np.fft.rfft(w, fft_size)
    conv_rev = np.fft.irfft(fft_rev * fft_w, fft_size)
    # conv_rev length = fft_size, we only need indices n‑1 … 2n‑2
    grad = 2.0 * conv_rev[n - 1: 2 * n - 1]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation value.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1. Possibly seed from a previously found construction.
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            # keep the first part; if shorter, pad with zeros
            n_prev = prev.shape[0]
            n0 = min(3000, n_prev)
            cand = np.zeros(n0, dtype=np.float64)
            cand[:n0] = prev[:n0]
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand /= s
                val = fast_score(cand)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    print(f"[init] height_sequence_1 gave value {val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Main multi‑restart loop.
    while time.time() < deadline:
        # pick a length (larger n gives more freedom but more work)
        n = int(np.random.choice([2500, 3000, 3500, 4000]))
        # pick one of a few families for the start point
        kind = np.random.choice(["power", "exp", "dirichlet"], p=[0.4, 0.3, 0.3])
        if kind == "power":
            p = np.random.uniform(0.1, 1.0)
            offset = np.random.uniform(0.0, 2.0)
            a = build_initial_sequence(n, kind="power", p=p, offset=offset)
        elif kind == "exp":
            q = np.random.uniform(0.85, 0.98)   # decay factor
            a = build_initial_sequence(n, kind="exp", q=q)
        else:
            conc = np.random.uniform(0.2, 1.0)
            a = build_initial_sequence(n, kind="dirichlet", conc=conc)

        # keep the best of the randomly generated starts
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            print(f"[restart] initial value {cur_val:.6f} (n={n})")

        # --------------------------------------------------------------
        # optimizer hyper‑parameters
        fft_size = 1 << ((3 * n - 2).bit_length())   # size for gradient convolutions
        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr0 = 0.12                     # base learning rate
        it = 0

        # continuation schedule for the smoothing parameter α
        alpha_schedule = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
        steps_per_alpha = 4000        # how many Adam steps for each α

        for alpha in alpha_schedule:
            # run a batch of Adam steps at the current smoothing level
            for _ in range(steps_per_alpha):
                if time.time() >= deadline:
                    break

                # ----- gradient of the smooth‑max -----
                g = smooth_max_gradient(a, alpha, fft_size)

                # ----- Adam update -----
                it += 1
                m = beta1 * m + (1 - beta1) * g
                v = beta2 * v + (1 - beta2) * (g * g)
                m_hat = m / (1 - beta1 ** it)
                v_hat = v / (1 - beta2 ** it)
                step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

                # ----- projection onto the simplex -----
                a = proj_simplex(a - step)

                # ----- occasional stochastic move to escape plateaus -----
                if np.random.rand() < 0.015:
                    i, j = np.random.randint(0, n, size=2)
                    if i != j and a[i] > 0.0:
                        delta = 0.15 * a[i] * np.random.rand()
                        cand = a.copy()
                        cand[i] -= delta
                        cand[j] += delta
                        cand = proj_simplex(cand)
                        cand_val = fast_score(cand)
                        if cand_val < best_val:
                            best_val, best_seq = cand_val, cand.copy()
                            print(f"[rand] new best {best_val:.6f} (α={alpha:.2f})")
                        elif np.random.rand() < 0.001:      # occasional uphill accept
                            a = cand

                # ----- periodic true‑objective evaluation -----
                if it % 2500 == 0:
                    cur_val = fast_score(a)
                    if cur_val < best_val:
                        best_val, best_seq = cur_val, a.copy()
                        print(f"[iter {it:6d}] new best {best_val:.6f} (α={alpha:.2f})")

            if time.time() >= deadline:
                break

        # after finishing the current restart, keep the best found so far
        # (already stored in best_seq / best_val)

    # ------------------------------------------------------------------
    # 3. Final scaling to the conventional normalisation √(2 n).
    if best_seq is None:
        raise RuntimeError("No feasible sequence was produced.")
    n_best = best_seq.shape[0]
    scale = sqrt(2.0 * n_best)          # makes ∑a = √(2 n)
    best_scaled = best_seq * scale
    print(f"[final] best value {best_val:.6f} on length {n_best}")

    return best_scaled.tolist()
