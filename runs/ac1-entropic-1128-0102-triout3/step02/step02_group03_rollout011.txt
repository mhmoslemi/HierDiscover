   np.random.seed(seed)
   deadline = time.time() + budget_s - 0.5

   # Determine n: use existing height_sequence_1 if present else choose n=5000 maybe less.

   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8

   # learning rate
   eta0 = 0.5 * S / n  # maybe 0.5...

   # alpha schedule
   alpha = 5.0
   alpha_inc = 1.0005  # each iteration
   alpha_max = 200.0

   # keep best

   cur_max = cur_conv.max()

   best_a = a.copy()
   best_score = cur_score

   iteration = 0
   no_improve = 0
   max_no_improve = 5000

   while time.time() < deadline:
        # compute conv and score
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (S**2)
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration}] New best: {best_score:.6f} (max={cur_max:.6f})")
            no_improve = 0
        else:
            no_improve += 1

        if best_score < 1.503:
            print(f"Target reached at iteration {iteration} with best {best_score:.6f}")
            break

        # Choose move type
        r = np.random.rand()
        if r < 0.6:
            # gradient descent using smooth max
            # increase alpha
            alpha = min(alpha_max, alpha * alpha_inc)
            # compute softmax weights for conv

The goal is to minimise

\text{score}(a)=\frac{2\,n\,\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},

\sum_i a_i = S = \sqrt{2\,n}.
\]

The optimisation problem is non‑convex because of the `max` on the convolution,
so a pure gradient method can get stuck.  
The algorithm below combines three ideas that proved very effective in the
previous submission:

1. **Smoothed maximum** – instead of the binary mask `conv == max(conv)` we
   use a softmax weight  

   \[
   w_k = \frac{\exp(\alpha\,\text{conv}_k)}{\sum_j \exp(\alpha\,\text{conv}_j)},
   \]

   with a slowly increasing temperature `α`.  The gradient of the smoothed
   maximum can be obtained analytically and computed with a single FFT.
2. **Adam‑style adaptive steps** – the (smoothed) sub‑gradient is fed to an
   Adam optimiser; the adaptive learning‑rate handles the different scales
   of the variables automatically.
3. **Exploration moves** – occasional random mass‑transfer steps and full
   Dirichlet restarts keep the search from being trapped in local minima.

All iterates are projected onto the simplex \(\{x\ge 0,\;\sum x=S\}\) after
each move, guaranteeing validity for `evaluate_sequence`.  
The routine runs until the time budget expires (or a target score of 1.503 is
reached) and returns the best sequence discovered.

```python
"""
Hybrid Adam / smoothed‑max optimisation for the step‑function attack.
* Keeps the ℓ₁‑norm fixed ( Σa = √(2n) ).
* Uses a soft‑max approximation of the max‑convolution to obtain a smooth
  sub‑gradient (computed by a single FFT).
* Adam updates provide adaptive step sizes.
* Random mass‑transfer moves and occasional Dirichlet restarts add exploration.
* The best solution found before the deadline is returned.
"""

import time
import numpy as np
from math import log, sqrt

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full (2·len(a)‑1) result (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_max_subgrad(a: np.ndarray, mask: np.ndarray, nfft: int) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)[k]  (or of the smoothed maximum) at the current point.
    *mask* is a non‑negative vector of length 2·len(a)‑1 whose sum is 1.
    The function returns a gradient of length len(a).
    """
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros_like(a)
    w = w / w_sum                     # normalise to a probability vector
    a_rev = a[::-1]

    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    # slice that corresponds to the gradient entries
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Adam + smoothed‑max optimisation for the step‑function problem.
    Returns the best sequence found within the allocated time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # problem size, target ℓ₁‑norm and FFT length
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        a0 = None

    S = np.sqrt(2.0 * n)                         # ℓ₁‑norm we keep fixed
    nfft = _next_pow2(3 * n - 2)                 # enough for mask * a_rev

    # initialise the iterate
    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        a = np.full(n, S / n, dtype=np.float64)  # uniform start

    # --------------------------------------------------------------
    # Adam state
    # --------------------------------------------------------------
    m = np.zeros_like(a)       # first moment
    v = np.zeros_like(a)       # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # step‑size base (roughly 0.3·S/n – the same order as the previous code)
    step_base = 0.3 * S / n

    # smoothed‑max temperature schedule
    alpha = 5.0                 # start with a smooth approximation
    alpha_inc = 1.0003          # geometric increase per iteration
    alpha_max = 200.0

    # bookkeeping
    conv = _convolve_self_fft(a, nfft)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (S ** 2)

    best_a = a.copy()
    best_score = cur_score

    iteration = 0
    no_improve = 0
    max_no_improve = 5000  # after this many stagnant iterations we restart

    # main optimisation loop
    while time.time() < deadline:
        # ---- evaluate current point ----
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (S ** 2)

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:06d}] New best: {best_score:.6f} (max={cur_max:.6f})")
            no_improve = 0
        else:
            no_improve += 1

        # early exit if we have already reached the target bound
        if best_score < 1.503:
            print(f"Target reached after {iteration} iterations (best={best_score:.6f})")
            break

        # ----------------------------------------------------------
        # decide which move to perform
        # ----------------------------------------------------------
        r = np.random.rand()
        if r < 0.6:
            # ---- Adam step on the smoothed maximum ----
            # increase the temperature slowly
            alpha = min(alpha_max, alpha * alpha_inc)

            # soft‑max weights for the current convolution
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w = w / w.sum()                      # normalised mask

            # gradient of the smoothed max
            grad = _grad_max_subgrad(a, w, nfft)

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** (iteration + 1))
            v_hat = v / (1.0 - beta2 ** (iteration + 1))

            step = -step_base * m_hat / (np.sqrt(v_hat) + eps)

            a_candidate = _project_to_simplex(a + step, S)

        elif r < 0.9:
            # ---- Random mass‑transfer move (biased towards large contributors) ----
            # approximate contribution of each index to the current maximum using a binary mask
            mask_bin = (conv >= cur_max - 1e-12).astype(float)
            if mask_bin.sum() > 0:
                mask_bin = mask_bin / mask_bin.sum()
            else:
                mask_bin = np.full_like(mask_bin, 1.0 / mask_bin.size)

            contrib = _grad_max_subgrad(a, mask_bin, nfft) / 2.0   # ½ of the sub‑gradient gives the raw contribution
            contrib = np.maximum(contrib, 0.0)
            if contrib.sum() > 0:
                src_probs = contrib / contrib.sum()
                i = np.random.choice(n, p=src_probs)
            else:
                i = np.random.randint(n)

            # pick a distinct destination uniformly
            j = i
            while j == i:
                j = np.random.randint(n)

            # transfer a modest fraction of the source mass
            frac = 0.1
            delta = frac * a[i]
            if delta <= 0:
                delta = frac * S / n

            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = _project_to_simplex(a_candidate, S)

        else:
            # ---- Full Dirichlet restart ----
            alpha_dir = np.full(n, 0.4)
            a_candidate = np.random.dirichlet(alpha_dir) * S
            # reset Adam moments after a restart
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{iteration:06d}] Full restart performed.")

        # ---- evaluate the candidate (exact objective) ----
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max / (S ** 2)

        # ---- acceptance rule (greedy + occasional Metropolis) ----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # simple temperature schedule for uphill moves
            temp = max(0.05 * (1.0 - iteration / 200000.0), 0.001)
            prob = np.exp(-(cand_score - cur_score) / temp)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            # the moments (m, v) are already updated for Adam steps;
            # for non‑Adam moves we keep the old moments (they will decay automatically)
        else:
            # tiny jitter occasionally helps to leave shallow plateaux
            if np.random.rand() < 0.005:
                jitter = np.random.randn(n) * (step_base * 0.3)
                a = _project_to_simplex(a + jitter, S)

        # ---- stagnation handling ----
        if no_improve >= max_no_improve:
            a = np.random.dirichlet(np.full(n, 0.3)) * S
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{iteration:06d}] Stagnation restart (no improvement).")

        iteration += 1

    # ----------------------------------------------------------------------
    # final verification and return
    # ----------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()
```
