"""
Adam‑based exponentiated‑gradient descent on the probability simplex.

The objective is
    F(a) = 2 * n * max_k (a * a)[k]          (sum(a)=1, a_i ≥ 0)

For a fixed index k the gradient of the peak (a*a)[k] w.r.t. a is
    ∇_a (a*a)[k]_i = 2 * a_{k-i}    (if 0≤k-i<n, else 0).

The true sub‑gradient of the max‑function is any convex combination of the
gradients belonging to the *active* set of indices where the convolution
attains its maximum.  We therefore collect all indices whose value lies
within a tiny tolerance of the global maximum and average their gradients.

To stay on the simplex we re‑parameterise a_i = softmax(p)_i and apply the
Adam optimiser to the transformed variables p.  The (softmax) Jacobian is
handled explicitly:
    ∇_p F = a * (∇_a F – ⟨a,∇_a F⟩).

The algorithm performs many cheap FFT‑based convolutions, occasional
Gaussian perturbations of the logits and a restart when progress stalls.
The best sequence ever seen is kept and finally returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------
def sqrt_series(n: int) -> np.ndarray:
    """Half‑binomial coefficients C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Linear convolution x*x using a real FFT (length 2·len(x)‑1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    return np.fft.irfft(X * X, n=m)


def softmax(p: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    pmax = p.max()
    e = np.exp(p - pmax)
    return e / e.sum()


def active_subgradient(a: np.ndarray, conv: np.ndarray, tol_rel: float = 1e-12) -> np.ndarray:
    """
    Compute the (averaged) sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] within tol of the maximum are treated as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance – we guarantee at least the exact maximiser is included
    tol = max(tol_rel * max_val, 1e-15)
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    grad /= max(1, len(active))
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative step‑function (probability distribution)
    that minimises the evaluation function using Adam on the simplex.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – keep the same size as the previous best solution
    n = 2500

    # initialise from the provided best construction if available,
    # otherwise fall back to the classic half‑binomial start.
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.size != n:
            init = sqrt_series(n)
    except Exception:
        init = sqrt_series(n)

    # normalise to a proper probability vector (sum = 1, non‑negative)
    init = np.maximum(init, 0.0)
    init_sum = init.sum()
    if init_sum < 1e-12:               # guard against degenerate start
        init = sqrt_series(n)
    else:
        init /= init_sum

    # log‑it­erates for exponentiated gradient (softmax) parametrisation
    eps = 1e-30
    p = np.log(init + eps)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr_start = 0.20               # initial learning‑rate for p
    lr_decay = 1e-5               # linear decay
    noise_prob = 0.006
    noise_scale = 1e-3

    # bookkeeping
    best_a = init.copy()
    best_score = float("inf")
    last_improve = 0
    restart_threshold = 180_000   # iterations without improvement → restart
    max_iter = 5_000_000          # safety cap

    deadline = time.time() + budget_s - 5.0   # leave a safety margin
    it = 0

    while time.time() < deadline and it < max_iter:
        # current probability vector
        a = softmax(p)                 # Σa = 1, a ≥ 0 by construction

        # convolution & current score
        conv = conv_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max   # sum(a) is 1 ⇒ denominator = 1

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:08d}] new best score = {best_score:.6f}")

        # -----  sub‑gradient (averaged over all active maxima) -----
        grad_a = active_subgradient(a, conv, tol_rel=1e-12)

        # transform gradient to log‑it­erate space (softmax Jacobian)
        # ∇_p F = a * (∇_a F – <a,∇_a F>)
        dot_ag = np.dot(a, grad_a)
        grad_p = a * (grad_a - dot_ag)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)

        # bias‑corrected estimates
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        # learning‑rate schedule
        lr = lr_start / (1.0 + lr_decay * it)

        # parameter step
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional random perturbation of the logits to escape plateaus
        if rng.random() < noise_prob:
            p += rng.normal(scale=noise_scale, size=n)

        # restart if stuck for a long time
        if it - last_improve > restart_threshold:

            a_restart = rng.dirichlet(np.full(n, 0.6))
            p = np.log(a_restart + eps)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"[iter {it:08d}] restart triggered – new random start, score≈{2.0*n*conv_fft(a_restart).max():.6f}")

        it += 1

    # ------------------------------------------------------------------
    # final safety evaluation using the provided routine
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
