def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of the half‑binomial series
    a_i = C(2i,i) / 4**i, normalised to sum = 1.
    """

    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries become equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Fast convolution x * x using a real FFT.
    Length of the result is 2*len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that minimises the evaluation function.
    The routine runs for roughly ``budget_s`` seconds and returns the best
    sequence (as a list of floats) that was encountered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – a compromise between solution quality and speed
    n = 2500

    # ------------------------------------------------------------------
    # 1.  PARAMETRIC SWEEP – cheap families that often give a good start
    def evaluate_candidate(a_vec: np.ndarray) -> float:
        """Return 2·n·max(conv(a,a)) for a vector already normalised to sum=1."""
        conv = conv_fft(a_vec)
        return 2.0 * n * conv.max()

    best_a = None
    best_score = np.inf

    # half‑binomial (classic starting point)
    a0 = sqrt_series(n)
    score0 = evaluate_candidate(a0)
    best_a, best_score = a0, score0
    print(f"[sweep] half‑binomial score = {score0:.6f}")

    # power‑law family a_i ∝ (i+1)^{-p}
    powers = np.linspace(0.0, 2.0, 21)      # 0 → uniform, 2 → very steep
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in powers:
        a = (idx ** (-p))
        a /= a.sum()
        s = evaluate_candidate(a)
        if s < best_score:
            best_a, best_score = a, s
            print(f"[sweep] power p={p:.2f} score = {s:.6f}")

    # exponential family a_i ∝ exp(-λ·i)
    lambdas = np.linspace(0.0, 0.03, 16)    # λ=0 gives uniform
    for lam in lambdas:
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        s = evaluate_candidate(a)
        if s < best_score:
            best_a, best_score = a, s
            print(f"[sweep] expo λ={lam:.4f} score = {s:.6f}")

    # ------------------------------------------------------------------
    # 2.  ADAM‑BASED PROJECTED SUB‑GRADIENT DESCENT
    a = best_a.copy()                     # current iterate (sum=1)
    cur_score = best_score
    best_iter_a = a.copy()                # best ever seen
    best_iter_score = cur_score

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # hyper‑parameters
    lr_initial = 0.12                     # base learning‑rate
    lr = lr_initial
    entropy_weight = 0.05                 # λ for -entropy regulariser
    decay_rate = 0.93                     # multiplicative decay of lr every block
    decay_every = 2500
    noise_prob = 0.006                    # occasional tiny perturbations
    noise_scale = 1e-4
    max_iter = 10_000_000                # absolute safety bound
    last_improve = 0

    deadline = time.time() + budget_s - 5.0   # leave a small safety margin
    it = 0

    while time.time() < deadline and it < max_iter:
        # ---- convolution & peak location ----
        conv = conv_fft(a)
        max_val = conv.max()
        k = int(conv.argmax())

        # ---- exact sub‑gradient of the peak value ----
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        jdx_range = k - idx_range
        grad[idx_range] = 2.0 * a[jdx_range]          # g_i = 2·a_{k‑i}

        # ---- entropy regulariser (gradient of -H) ----
        ent_grad = np.log(a + eps) + 1.0               # d(-∑a log a)/da = log a + 1
        combined = grad + entropy_weight * ent_grad   # descent direction for the score

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * combined
        v = beta2 * v + (1.0 - beta2) * (combined ** 2)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # prevent stepping more than 90 % of any coordinate (keeps positivity)
        step = np.minimum(step, a * 0.90)

        a_trial = a - step

        # occasional random jitter to aid escape from plateaus
        if rng.random() < noise_prob:
            a_trial += (rng.random(n) - 0.5) * noise_scale

        # project back onto the simplex (sum = 1, non‑negative)
        a_trial = project_onto_simplex(a_trial)

        # ----- evaluate trial -----
        conv_trial = conv_fft(a_trial)
        trial_score = 2.0 * n * conv_trial.max()

        # update best‑ever record
        if trial_score < best_iter_score:
            best_iter_score = trial_score
            best_iter_a = a_trial.copy()
            last_improve = it
            print(f"[iter {it:08d}] new best = {best_iter_score:.6f}")

        # simple acceptance – always move (Adam already adapts); we keep the
        # current iterate for the next gradient step
        a = a_trial
        cur_score = trial_score

        # ----- learning‑rate decay -----
        if (it + 1) % decay_every == 0:
            lr = max(lr * decay_rate, 5e-5)

        # ----- symmetrisation every few thousand steps (helps flatten convolution) -----
        if (it + 1) % 5000 == 0:
            a = (a + a[::-1]) * 0.5
            a = project_onto_simplex(a)

        # ----- restart if stuck for a long time -----
        if it - last_improve > 200_000:
            # re‑initialise around the best found so far with a bit of noise
            a = best_iter_a + (rng.random(n) - 0.5) * 0.02
            a = np.clip(a, 0.0, None)
            a = project_onto_simplex(a)
            cur_score = 2.0 * n * conv_fft(a).max()
            last_improve = it
            print(f"[iter {it:08d}] restart – score reset to {cur_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 3.  Final evaluation with the provided routine (safety check)
    final_score = evaluate_sequence(best_iter_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_iter_a.tolist()
