"""
propose_candidate
-----------------
A multi‑scale, projected‐Adam optimiser for the step‑function problem.

The algorithm works in three stages:

1. **Initialisation** – start from the best known construction (if present)
   or from a deterministic “arcsine‑shaped’’ distribution that is known to
   flatten the self‑convolution.  The vector is normalised to unit sum.

2. **Progressive up‑sampling** – the current shape is repeatedly linearly
   interpolated to a larger length (≈1.5× each stage, up to a configurable
   maximum).  At each resolution an Adam optimiser is run on the objective

        score(a) = 2·n·max_k (a * a)[k]    (the sum of a is fixed to 1)

   The sub‑gradient is obtained from all convolution entries that lie
   within a small relative tolerance of the current maximum.
   The gradient is computed with a single FFT‑based correlation, making the
   per‑iteration cost O(n log n).  The Adam step is projected back onto the
   probability simplex (Euclidean projection of Duchi et al.).

3. **Final refinement** – after the time budget is almost exhausted a last
   pass on the best length found is performed with a tighter tolerance and
   a few thousand extra steps.

During the run the best sequence (lowest score) is remembered and finally
returned as a plain Python list.  All numbers are non‑negative and the
function respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal => uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """
    Return the evaluation score for a sequence that already sums to 1.
    The score is   2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _conv_max_and_vector(a: np.ndarray):
    """
    Compute (max_conv, full convolution vector) for a (sum(a)=1).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_mask(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of the (smoothed) max‑convolution with respect
    to a, assuming the active set is encoded in `mask` (0/1 per convolution
    index).  The gradient is
        g_i = 2 * Σ_{k} mask_k * a_{k-i}
    implemented via a single FFT‑based correlation.
    """
    n = a.shape[0]
    # correlation = a ⋆ mask  (cross‑correlation, not convolution)
    rev_mask = mask[::-1]
    L = 1 << ((n + len(mask) - 1).bit_length())
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(rev_mask, n=L)
    corr = np.fft.irfft(Af * Bf, n=L)[: n + len(mask) - 1]
    offset = len(mask) - 1               # index of zero shift in the full correlation
    g = 2.0 * corr[offset - np.arange(n)]  # pick the needed slice
    return g


def _adam_step(a, g, m, v, t,
               lr0=0.4, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Perform one Adam update on `a` with gradient `g`.  The step is then
    projected onto the simplex (ensuring non‑negativity and unit sum).
    """
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr0 / (np.sqrt(v_hat) + eps)
    a_new = a - step * m_hat
    # project back onto the simplex
    a_new = _project_simplex(a_new)
    return a_new, m, v


def _arcsine_initial(n: int) -> np.ndarray:
    """
    Deterministic arcsine‑shaped start: a_i ∝ 1/√((i+½)*(n‑½‑i)).
    This shape tends to flatten the self‑convolution.
    """
    i = np.arange(n, dtype=float)
    # shift by 0.5 to avoid division by zero at the boundaries
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0 / n)
    return w / w_sum
