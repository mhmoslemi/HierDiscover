def conv_matrix(f):
    n = len(f)
    # produce matrix A of shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    # So:
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # for j = i_start .. i_end: f_i where i = k - j => j = k - i
        # So we can set A[k, j] = f_i where i = k - j.
        # Equivalent loop:
        for j in range(i_start, i_end+1):
            i = k - j
            A[k, j] = f[i]
    return A

def solve_lp_max_sum(f, M):
    n = len(f)
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f   # shift f
    b_ub = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n)   # maximize sum
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a candidate step‑function sequence.

Algorithm
---------
1.  Initialise a random non‑negative vector `f` with total mass = 1.
2.  Compute the current bound M = max_k (f*f)_k.
3.  Solve the linear programme
        maximise   Σ_j b_j
        subject to (f*b)_k ≤ M  for all k = 0 … 2n‑2
                    b_j ≥ 0
   The matrix (f*b)_k is a simple Toeplitz convolution matrix, built
   by shifting the vector `f`.  `scipy.optimize.linprog` with the
   ``highs`` backend solves the LP quickly for n of a few hundred up
   to ~1500.
4.  Normalise the LP solution `b` to a probability vector `g`.
5.  Update the current step‑function by a convex combination
        f ← (1‑t)·f + t·g          (t starts at 0.2 and is adaptively
                                   shrunk if the bound does not improve).
    Renormalise `f` to keep Σf = 1.
6.  Re‑evaluate M = max_k (f*f)_k.  If the bound decreased, keep the
    new `f` and repeat the LP step.  The bound is monotone decreasing
    and typically drops from ≈ 2 (the uniform bound) towards the
    target ≈ 1.5.
7.  After the LP phase, a short Adam‑based soft‑max optimisation
    (sub‑gradient of the max‑convolution) is applied to polish the
    best sequence found.
8.  The routine runs for at most ``budget_s`` seconds, keeps the best
    sequence discovered (across several lengths and random restarts),
    and finally returns it.  All numbers are non‑negative; the
    ``evaluate_sequence`` function will return the official score.
"""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv_full


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())

def _gradient_wrt_a(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)_k w.r.t. a, using the indices in top_idxs.
    For each selected offset k we add 2·a_{k‑i} for the admissible i.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in top_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]
    return grad


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _adam_step(z: np.ndarray,
               grad_z: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """One Adam update (returns new z, new m, new v)."""
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, m, v


def _refine_with_adam(a: np.ndarray,
                      deadline: float,
                      top_k: int = 1) -> np.ndarray:
    """
    Short Adam‑based optimisation starting from array `a`.
    Only the (few) largest convolution entries are used for the sub‑gradient.
    """
    n = a.shape[0]
    # initialise logits
    eps = 1e-12
    z = np.log(a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_conv_fft(best_a))

    while time.time() < deadline:
        conv = _conv_fft(a)
        cur_max = conv.max()
        # indices of the (few) current maxima
        if top_k == 1:
            ks = np.array([int(np.argmax(conv))], dtype=int)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        grad_a = _gradient_wrt_a(a, ks)

        dot = a.dot(grad_a)
        grad_z = a * (grad_a - dot)

        t += 1
        lr = 0.05 / np.sqrt(n) / (1.0 + t * 1e-5)
        z, m, v = _adam_step(z, grad_z, m, v, t, lr)

        a = _logits_to_a(z)

        cur_score = 2.0 * n * np.max(_conv_fft(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress print
            print(f"[refine] new local best {best_score:.6f}")

        # jitter to avoid stagnation
        if t % 1500 == 0:
            jitter = np.random.randn(n) * 1e-3
            z += jitter

    return best_a

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A @ b = f * b (linear convolution).
    Column j of A is the vector f shifted by j positions, zero‑padded.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> Optional[np.ndarray]:
    """
    Solve the LP:
        maximise   Σ b_j
        subject to (f*b)_k ≤ M   for all k
                    b_j ≥ 0.
    Returns the optimal b (may be None if the solver fails).
    """
    if not _HAS_SCIPY:      # pragma: no cover
        return None
    n = f.shape[0]
    A = _build_conv_matrix(f)
    # a tiny tolerance avoids infeasibilities caused by floating‑point noise
    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)
    c = -np.ones(n)                     # linprog minimises → maximise Σb
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.success:
        return res.x
    return None

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function (the lower the better).  The routine tries several
    lengths, runs the LP‑based iterative improvement, and finally polishes the
    best candidate with a short Adam optimisation.  It respects the given
    time budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq: List[float] | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction is present, start from it.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(height_sequence_1)
            cand_score = evaluate_sequence(cand)
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[init] imported previous best {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to explore – a modest range keeps the LP cheap but still
    # provides enough flexibility for a good bound.
    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 1600]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Number of random restarts per length (adjusted to remaining time)
        restarts = 2
        for start_id in range(restarts):
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------------
            # 1) Initialise a random non‑negative vector with sum = 1.
            # ------------------------------------------------------------------
            a = np.random.rand(n).astype(np.float64)
            a /= a.sum()
            cur_score = 2.0 * n * np.max(_conv_fft(a))
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[init n={n} r={start_id}] start score {best_score:.6f}")

            # ------------------------------------------------------------------
            # 2) Core LP‑based improvement loop.
            # ------------------------------------------------------------------
            f = a.copy()
            M = np.max(_conv_fft(f))
            t_step = 0.2                # initial convex‑combination weight
            it = 0
            while time.time() < deadline and it < 500:
                b = _solve_lp_max_sum(f, M)
                if b is None:
                    # LP failed – abort this restart
                    print("[lp] solver failed, breaking")
                    break
                sum_b = b.sum()
                if sum_b == 0.0 or np.isnan(sum_b):
                    break
                g = b / sum_b               # normalise to unit sum

                # convex update
                f_new = (1.0 - t_step) * f + t_step * g
                f_new /= f_new.sum()        # renormalise (tiny numerical drift)

                M_new = np.max(_conv_fft(f_new))
                cur_score = 2.0 * n * M_new

                # accept if we made progress, otherwise shrink the step size
                if M_new < M - 1e-12:
                    f = f_new
                    M = M_new
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.tolist()
                        print(f"[n={n}] new global best {best_score:.6f}")
                    # enlarge step a little when we are improving
                    t_step = min(0.5, t_step * 1.05)
                else:
                    t_step *= 0.5          # become more conservative
                    if t_step < 1e-4:
                        # step became too small – stop this restart
                        break
                it += 1

            # ------------------------------------------------------------------
            # 3) Final polishing with Adam (only if we still have time).
            # ------------------------------------------------------------------
            remaining = deadline - time.time()
            if remaining > 5.0:
                f_polished = _refine_with_adam(f, deadline, top_k=1)
                final_score = 2.0 * n * np.max(_conv_fft(f_polished))
                if final_score < best_score:
                    best_score = final_score
                    best_seq = f_polished.tolist()
                    print(f"[n={n}] after polishing {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the official evaluator.
    # ------------------------------------------------------------------
    try:
        official_score = evaluate_sequence(best_seq)
        print(f"Final best official score {official_score:.6f} "
              f"(length {len(best_seq)})")
    except Exception as exc:  # pragma: no cover
        print(f"Warning: evaluate_sequence failed ({exc})")

    return best_seq
