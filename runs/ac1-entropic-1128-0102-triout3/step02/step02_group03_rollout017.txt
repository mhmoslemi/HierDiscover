    conv = _convolve_self_fft(a, nfft_self)

"""
Hybrid projected gradient descent with a log‑sum‑exp smoothing of the
max‑convolution (continuation) and enforced symmetry.

The algorithm works on the ℓ₁‑simplex {x ≥ 0, Σx = √(2n)}.
For a fixed ℓ₁‑norm the objective 2·n·max(conv)/(Σx)² reduces to the
maximal entry of the self‑convolution, so we minimise
    f(a) = max_k (a*a)[k].

A smooth approximation f_τ(a) = τ·log Σ_k exp((a*a)[k]/τ) is used.
Its gradient can be written in closed form via a single FFT‑based
convolution, giving a cheap direction that points towards a reduction
of the peak.  τ is decreased slowly (continuation) so that the method
first moves on a smooth landscape and then refines the true max.

After each gradient step the iterate is projected back onto the simplex
and symmetrised (a ← (a+a[::-1])/2).  If no improvement is observed for
a while a small symmetric jitter is added to escape plateaus.  The best
sequence seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full (2·len(a)‑1) result (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_max_logsumexp(a: np.ndarray,
                        tau: float,
                        nfft_self: int,
                        nfft_grad: int,
                        conv_self: np.ndarray = None) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution
          f_τ(a) = τ·log Σ_k exp( (a*a)[k] / τ )
    with respect to a.

    Parameters
    ----------
    a : ndarray, shape (n,)
        Current sequence (non‑negative, sum ≃ √(2n)).
    tau : float
        Smoothing parameter (τ > 0).  Smaller τ → closer to true max.
    nfft_self : int
        FFT length for the self‑convolution (≥ 2·n‑1).
    nfft_grad : int
        FFT length for the convolution w * a_rev (≥ 3·n‑2).
    conv_self : ndarray, optional
        If given, should be a*a already computed with `_convolve_self_fft`.
        Saves one FFT per iteration.

    Returns
    -------
    grad : ndarray, shape (n,)
        Gradient of f_τ at a.
    """
    n = a.size
    if conv_self is None:
        conv_self = _convolve_self_fft(a, nfft_self)

    # weights w_k ∝ exp( (conv_k) / τ ), numerically stabilised
    max_c = conv_self.max()
    w = np.exp((conv_self - max_c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.zeros_like(a)

    # convolution w * a_rev (full length = 3·n‑2)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=nfft_grad)
    A_rev = np.fft.rfft(a_rev, n=nfft_grad)
    conv_full = np.fft.irfft(W * A_rev, n=nfft_grad)

    needed = 3 * n - 2            # length of the exact linear convolution
    conv_full = conv_full[:needed]

    # cross‑correlation gives the desired weighted sum:
    #   g_i = Σ_k w_k * a_{k-i}
    start = n - 1
    grad = 2.0 * conv_full[start:start + n] / w_sum
    return grad

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,

    """
    Projected gradient descent with log‑sum‑exp smoothing and symmetry.
    Returns the best non‑negative step‑function sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # problem size, ℓ₁‑norm and FFT lengths
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():

        init_seq = None

    S = np.sqrt(2.0 * n)                # target ℓ₁‑norm
    nfft_self = _next_pow2(2 * n - 1)   # for a*a
    nfft_grad = _next_pow2(3 * n - 2)   # for w * a_rev

    # --------------------------------------------------------------
    # initialise iterate (uniform or from previous construction)
    # --------------------------------------------------------------
    if init_seq is not None:
        a = _project_to_simplex(init_seq, S)
    else:
        a = np.full(n, S / n, dtype=np.float64)   # uniform start

    # enforce symmetry from the very beginning
    a = (a + a[::-1]) / 2.0
    a = _project_to_simplex(a, S)

    # --------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------
    conv = _convolve_self_fft(a, nfft_self)
    cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_a = a.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # hyper‑parameters
    # --------------------------------------------------------------
    step_initial = S                     # start with a very large step
    tau = 2.0                            # initial smoothing parameter
    tau_min = 5e-4
    tau_decay = 0.997                    # geometric decay per iteration

    stagnation_counter = 0
    stagnation_limit = 2500

    iteration = 0
    max_line_search = 12                 # maximal back‑tracking steps

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # gradient of the smoothed objective
        grad = _grad_max_logsumexp(a, tau, nfft_self, nfft_grad, conv_self=conv)
        grad_norm = np.linalg.norm(grad)

        # direction = normalized gradient (ascending); we move opposite.
        if grad_norm > 0:
            direction = grad / grad_norm
        else:
            direction = np.zeros_like(a)

        # ----------------------------------------------------------
        # try a projected gradient step with back‑tracking line search
        # ----------------------------------------------------------
        step = step_initial / np.sqrt(iteration + 1.0)
        improved = False
        for _ in range(max_line_search):
            a_candidate = _project_to_simplex(a - step * direction, S)

            # enforce symmetry (helps a lot for the autocorrelation problem)
            a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
            a_candidate = _project_to_simplex(a_candidate, S)

            conv_cand = _convolve_self_fft(a_candidate, nfft_self)
            cand_score = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

            if cand_score < cur_score - 1e-12:
                improved = True
                break
            step *= 0.5                      # shrink step and try again

        # ----------------------------------------------------------
        # if back‑tracking failed, resort to a tiny symmetric jitter
        # ----------------------------------------------------------
        if not improved:
            jitter = np.random.randn(n) * (0.01 * S / n)
            jitter = (jitter + jitter[::-1]) / 2.0
            a_candidate = _project_to_simplex(a + jitter, S)
            conv_cand = _convolve_self_fft(a_candidate, nfft_self)
            cand_score = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        # ----------------------------------------------------------
        # accept the candidate if it improves the true objective
        # ----------------------------------------------------------
        if cand_score < cur_score:
            a = a_candidate
            conv = conv_cand
            cur_score = cand_score
            stagnation_counter = 0
            print(f"[{iteration:07d}] improved → {cur_score:.6f} (τ={tau:.6f})")
        else:
            stagnation_counter += 1

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----------------------------------------------------------
        # termination check
        # ----------------------------------------------------------
        if best_score < 1.503:
            print(f"Target reached after {iteration} iterations "
                  f"(best={best_score:.6f})")
            break

        # ----------------------------------------------------------
        # schedule the smoothing parameter τ → 0 (continuation)
        # ----------------------------------------------------------
        tau = max(tau * tau_decay, tau_min)

        # ----------------------------------------------------------
        # occasional stochastic escape when stuck too long
        # ----------------------------------------------------------
        if stagnation_counter >= stagnation_limit:
            jitter = np.random.randn(n) * (0.03 * S / n)
            jitter = (jitter + jitter[::-1]) / 2.0
            a = _project_to_simplex(a + jitter, S)
            conv = _convolve_self_fft(a, nfft_self)
            stagnation_counter = 0
            print(f"[{iteration:07d}] stagnation jitter applied")

        iteration += 1

    # ------------------------------------------------------------------
    # final verification and return (use the official evaluator)
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. "
          f"Best final score = {final_score:.6f}")
    return best_a.tolist()
