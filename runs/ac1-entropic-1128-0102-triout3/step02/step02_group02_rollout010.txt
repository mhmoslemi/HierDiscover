```
def _conv_fft(a):
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A
    conv_full = np.fft.irfft(C, n=N)[:2*n-1]
    return conv_full

def _gradient_wrt_a(a, ks):
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        grad[i_range] += 2.0 * a[k - i_range]
    return grad

def _weighted_gradient(a, ks, conv, alpha):
    # compute softmax weights among selected ks
    vals = conv[ks]
    # numeric stable
    maxv = np.max(vals)
    exp_vals = np.exp(alpha * (vals - maxv))
    weights = exp_vals / np.sum(exp_vals)
    # accumulate gradients weighted
    grad_a = np.zeros_like(a)
    for w, k in zip(weights, ks):
        # compute gradient contribution for k
        i_start = max(0, k - (a.shape[0] - 1))
        i_end = min(k, a.shape[0] - 1)
        i_range = np.arange(i_start, i_end + 1)
        grad_a[i_range] += w * 2.0 * a[k - i_range]
    return grad_a

def _adam_update(z, grad_z, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad_z
    v = beta2 * v + (1 - beta2) * (grad_z ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return z - step, m, v

"""
Propose a candidate non‑negative step‑function sequence.

Algorithm overview
-------------------
1. **Exploratory window sweep** – For several lengths (1200, 2400, 3600,
   4800) a handful of classical window shapes (Hann, Hamming, Blackman,
   Gaussian, Kaiser) are generated, normalised to unit sum and evaluated.
   The best few candidates are kept.

2. **Gradient‑based refinement** – Each stored candidate is turned into a
   soft‑max parametrisation (logits `z`).  Adam optimisation is applied in
   log‑space while the sequence is periodically symmetrised (a_i = a_{n‑1‑i})
   to keep the autocorrelation flat.  The gradient is a weighted sub‑gradient
   of the largest convolution entries (soft‑max of the top‐k).  Learning‑rate
   decays and the active set shrinks to the true maximum after a few
   thousand steps.

3. **Random restarts** – When time permits the whole procedure is repeated
   from random initial logits (different lengths) to explore other basins.

All operations use FFT‑based convolution (O(n log n)).  The search runs
until the supplied time budget (≈1000 s) expires and returns the best
sequence found.  The returned list contains non‑negative values that sum
to 1; this is compatible with `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.special import i0  # Kaiser window helper

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns a 1‑D array of length 2·n‑1.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A                     # |A|²  (real because A is conjugate‑symmetric)
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv_full


def _score_from_a(a: np.ndarray) -> float:
    """
    Compute the evaluation score used by `evaluate_sequence`
    for a sequence with Σ a = 1:
        score = 2·n·max( a * a )
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _adam_update(
    z: np.ndarray,
    grad_z: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Classic Adam update (returns new z, m, v)."""
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return z - step, m, v


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i} (simple averaging)."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Window generators (all return a non‑negative vector, not yet normalised)
# ----------------------------------------------------------------------


def hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))


def hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))


def blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )


def gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """σ = sigma_frac·n (fraction of the length)."""
    i = np.arange(n, dtype=np.float64)
    sigma = sigma_frac * n
    centre = (n - 1) / 2.0
    return np.exp(-0.5 * ((i - centre) / sigma) ** 2)


def kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    x = (i - centre) / centre  # in [-1, 1]
    w = i0(beta * np.sqrt(1.0 - x ** 2)) / i0(beta)
    return w

# Main optimisation routine
# ----------------------------------------------------------------------

    try:

        if cand_score < best_score:
            best_seq = cand

    except Exception:

    # ------------------------------------------------------------------
    # 1) exhaustive scan of classical windows (fast, cheap)
    # ------------------------------------------------------------------
    lengths = [1200, 2400, 3600, 4800]
