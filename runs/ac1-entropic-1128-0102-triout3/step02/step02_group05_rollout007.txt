"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(a∗a) / (sum(a)²)

using a smoothed‑max gradient descent on the probability simplex.
The algorithm mixes a softmax approximation of the max‑convolution,
projected gradient steps, occasional random mass‑transfer moves and
restarts when progress stalls.  It starts from several analytically
motivated families (power‑law, exponential, uniform, Dirichlet) and
keeps the best sequence seen so far.  The whole search stops after the
given time budget (default ≈ 1000 s) and returns the best list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` without safety checks.
    The vector ``a`` is assumed to be non‑negative.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # next power of two for an efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    Implements the O(n log n) algorithm from Wang & Carreira‑Perpiñán.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """
    Power‑law family a_i = (i + offset)^{-p},  i = 1,…,n,
    normalised to ℓ¹‑norm = 1 (i.e. a lies on the simplex).
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

        a = np.exp(-lam * idx)
        a /= a.sum()
        init_pool.append(a)

    # uniform
    init_pool.append(np.full(n, 1.0 / n))

    # a few random Dirichlet draws
    for _ in range(8):
        init_pool.append(np.random.dirichlet(np.ones(n)))

    # if a previous construction is available, try it as a seed
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seed_vec = prev[:n].copy()
            else:
                seed_vec = np.zeros(n, dtype=np.float64)
                seed_vec[: prev.size] = prev
            seed_vec = np.maximum(seed_vec, 0.0)
            if seed_vec.sum() > 0:
                seed_vec /= seed_vec.sum()
                init_pool.append(seed_vec)
                print("[init] incorporated height_sequence_1")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Pick the best of the initial candidates.
    best_seq = None
    best_val = np.inf
    for seq in init_pool:
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop – smoothed‑max gradient descent.
    a = best_seq.copy()                 # current point on the simplex
    best_seq = a.copy()
    best_val = fast_score(a)

    # hyper‑parameters (tuned empirically)
    eta0 = 0.6               # initial step size
    eta_decay = 1e-6         # linear decay per iteration
    rand_prob = 0.015        # probability of a random mass‑transfer move
    rand_step = 0.12         # fraction of source mass moved when random move occurs
    alpha_start = 0.3        # softmax temperature at iteration 0
    alpha_end = 30.0         # temperature after many iterations
    alpha_ramp = 300_000     # iterations over which alpha is linearly increased

    # pre‑allocate FFT size for the self‑convolution (fixed during the run)
    conv_fft_size = 1 << ((2 * n - 1).bit_length())

    it = 0
    last_improve = 0                     # iteration index of the last improvement
    max_no_improve = 200_000             # after this many stagnating steps we restart locally

    while time.time() < deadline:
        # ----- 4.1 self‑convolution and current objective value
        fft_a = np.fft.rfft(a, conv_fft_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_fft_size)[: 2 * n - 1]
        cur_val = fast_score(a)                 # same as 2·n·max(conv)/(sum(a)²)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = it
            print(f"[iter {it:7d}] new best {best_val:.6f}")

        # ----- 4.2 temperature for the smooth‑max
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / alpha_ramp)

        # ----- 4.3 compute softmax weights over the convolution entries
        max_conv = conv.max()
        # subtract max to keep the exponentials well‑behaved
        exp_term = np.exp(alpha * (conv - max_conv))
        w = exp_term / exp_term.sum()          # shape (2n‑1,)

        # ----- 4.4 gradient of the smooth‑max (via another convolution)
        rev_a = a[::-1]                        # a reversed
        # convolution of w (len 2n‑1) with rev_a (len n)
        size = 1 << ((len(w) + n - 1).bit_length())
        fft_w = np.fft.rfft(w, size)
        fft_rev = np.fft.rfft(rev_a, size)
        grad_full = np.fft.irfft(fft_w * fft_rev, size)[: len(w) + n - 1]
        # extract the n entries corresponding to indices 0…n‑1
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- 4.5 projected gradient step on the simplex
        eta = eta0 / (1.0 + it * eta_decay)
        a_next = proj_simplex(a - eta * grad)

        # ----- 4.6 occasional random mass‑transfer (exploration)
        if np.random.rand() < rand_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a_next[i] > 0.0:
                delta = rand_step * a_next[i] * np.random.rand()
                cand = a_next.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = proj_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a_next = cand
                elif np.random.rand() < 0.001:       # tiny uphill acceptance
                    a_next = cand

        a = a_next
        it += 1

        # ----- 4.7 periodic reporting
        if it % 5000 == 0:
            print(f"[iter {it:7d}] cur {cur_val:.6f} best {best_val:.6f} α {alpha:.2f} η {eta:.5f}")

        # ----- 4.8 stagnation handling – a light restart around the best point
        if it - last_improve > max_no_improve:
            # perturb the best point a little and continue
            perturb = np.random.dirichlet(0.5 * np.ones(n))
            a = 0.9 * best_seq + 0.1 * perturb
            a = proj_simplex(a)                # stay on the simplex
            last_improve = it
            print(f"[restart] perturbing after {it} iterations (best still {best_val:.6f})")

    # --------------------------------------------------------------
    print(f"[final] best value found = {best_val:.6f} after {it} iterations")
    # The sequence must be a plain Python list for the external checker.
    return best_seq.tolist()
