def propose_candidate(seed=0, budget_s=950.0):

    p_vals = np.arange(0.2, 1.2, 0.02)

            a = np.maximum(a, 0.0)

            score = evaluate_seq(a)

def refine_adam(start_a, n, deadline, max_iters=50000, lr=0.3, gamma=2000.0):
    a = start_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_local_a = a.copy()
    best_local_score = 2.0 * n * conv_fft(a).max()
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute conv and weights
        conv_full = conv_fft(a)   # shape 2n-1
        maxc = conv_full.max()
        # compute softmax weights for max
        # exponent: gamma*(conv - maxc)
        exp_arg = gamma * (conv_full - maxc)
        # Use np.exp which safe for negative moderate values.
        weights = np.exp(exp_arg)
        weight_sum = weights.sum()
        w = weights / weight_sum
        # gradient
        a_rev = a[::-1]
        conv_rev = conv_full_fft(w, a_rev)  # full conv length
        grad = 2.0 * conv_rev[n-1 : n-1 + n]   # shape n
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_candidate = a - step
        # project onto simplex
        a_candidate = project_onto_simplex(a_candidate)
        # evaluate candidate score
        conv_cand = conv_fft(a_candidate)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < best_local_score:
            best_local_score = cand_score
            best_local_a = a_candidate.copy()
        # acceptance

        if cand_score <= cur_score:
            a = a_candidate

"""
Search for a non‑negative step‑function (a list of heights) that minimises
the provided evaluate_sequence() function.

Algorithm
---------
1. **Parameter sweep** –  For a moderately large length (default n=8000)
   we generate three simple analytic families:
      • Symmetric power‑law   a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)
      • Beta‑distribution     a_i ∝ (i+1)^(α‑1)·(n‑i)^(β‑1)
      • Double‑sided power‑law a_i ∝ (i+off)^(‑p)+(n‑1‑i+off)^(‑p)
   The parameters (p,α,β,off) are swept on a coarse grid; the best
   candidate (smallest 2·n·max(conv)) is kept.

2. **Refinement** –  From the best seed we run a few independent runs of
   Adam optimiser on a smooth surrogate of the max‑convolution:
        L(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k)   (γ large)
   Its gradient can be written using a single convolution, which we
   evaluate via a real FFT.  After each step we project the vector back
   onto the probability simplex (non‑negative, sum = 1).  The global best
   across all runs is retained.

3. **Result** –  The final sequence is sanity‑checked with
   evaluate_sequence() and returned as a plain ``list``.

The routine respects the 1000 s time budget by checking a deadline in
every loop and prints intermediate scores for diagnostics.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Helper functions – top‑level, no closures
# ----------------------------------------------------------------------


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Self‑convolution (a*a) using a real FFT.
    Returns a vector of length 2·len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution x*y (both real) using a real FFT.
    Length = len(x)+len(y)-1.
    """
    n = x.shape[0] + y.shape[0] - 1
    X = np.fft.rfft(x, n=n)
    Y = np.fft.rfft(y, n=n)
    conv = np.fft.irfft(X * Y, n=n)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ_i w_i = 1 }.
    O(n log n) algorithm (Duchi et al. 2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def symmetric_power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """
    a_i ∝ ((i+offset)*(n‑1‑i+offset))^(‑p), i = 0…n‑1, normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑distribution shape on {0,…,n‑1}:
        a_i ∝ (i+1)^(α‑1)·(n‑i)^(β‑1),  normalised to sum = 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    a = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0))
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def double_power_law_sequence(n: int, p: float, offset: float = 0.5) -> np.ndarray:
    """
    a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p), symmetric double‑sided power‑law.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def score_of(a: np.ndarray, n: int) -> float:
    """
    Fast evaluation of the objective when the vector is already normalised
    (sum = 1).  Returns 2·n·max(a*a).
    """
    conv = conv_fft(a)
    return float(2.0 * n * conv.max())


def smooth_max_gradient(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        L(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k)
    Using the identity ∂L/∂a = 2·conv(w, a_rev)[n-1:n-1+n],
    where w are the soft‑max weights.
    """
    n = a.shape[0]
    conv_full = conv_fft(a)                     # length 2n‑1
    maxc = conv_full.max()
    # compute soft‑max weights (unnormalised for numerical stability)
    exp_arg = gamma * (conv_full - maxc)        # ≤ 0
    w = np.exp(exp_arg)
    w /= w.sum()                                # now Σ w = 1
    # gradient = 2 * conv(w, a_rev)[n-1 : n-1 + n]
    a_rev = a[::-1]
    conv_rev = conv_full_fft(w, a_rev)          # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


def adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 30000,
    lr: float = 0.25,
    gamma: float = 4000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate of max‑convolution.
    The vector is kept on the probability simplex.
    Returns (best_vector, best_score).
    """
    rng = np.random.default_rng()
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = score_of(a, n)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # ----- smooth gradient -----
        grad = smooth_max_gradient(a, gamma)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = project_onto_simplex(a_candidate)

        # ----- evaluate -----
        cur_score = score_of(a, n)
        cand_score = score_of(a_candidate, n)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[adam] iter {t:06d} new global best = {best_score:.6f}")

        # simple accept‑if‑better policy (no temperature)
        if cand_score <= cur_score:
            a = a_candidate
        else:
            # occasional uphill move with vanishing temperature
            temperature = max(1e-6, 0.02 * (1.0 - t / max_iters))
            if rng.random() < np.exp(-(cand_score - cur_score) / temperature):
                a = a_candidate

        # learning‑rate decay
        if t % 8000 == 0:
            lr *= 0.5

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Main entry point.  Searches for a sequence minimising evaluate_sequence().
    Returns the best sequence as a ``list[float]``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Choose problem size – larger n gives a lower theoretical bound.
    # ------------------------------------------------------------------
    n = 8000                       # feel free to adjust; still fast enough.

    # ------------------------------------------------------------------
    # 2️⃣  Initialise best holder.
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 3️⃣  If a previous solution exists, evaluate it as a candidate.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.maximum(cand, 0.0)
        if cand.sum() > 0.0:
            cand /= cand.sum()
        else:
            cand = np.full_like(cand, 1.0 / cand.size)
        if cand.size != n:
            # interpolate to current length
            x_old = np.linspace(0.0, 1.0, cand.size)
            x_new = np.linspace(0.0, 1.0, n)
            cand = np.interp(x_new, x_old, cand)
            cand /= cand.sum()
        sc = score_of(cand, n)
        best_a, best_score = cand, sc
        print(f"[init] existing height_sequence_1 score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Coarse grid search over analytic families.
    # ------------------------------------------------------------------
    print("[grid] starting coarse sweep")
    # Symmetric power‑law ------------------------------------------------
    for p in np.arange(0.30, 0.95, 0.025):
        if time.time() > deadline:
            break
        for off in (0.5, 1.0):
            a = symmetric_power_law_sequence(n, p, offset=off)
            sc = score_of(a, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[grid] sym‑pow p={p:.3f} off={off} score={sc:.6f}")

    # Double‑sided power‑law -------------------------------------------
    for p in np.arange(0.30, 0.95, 0.025):
        if time.time() > deadline:
            break
        a = double_power_law_sequence(n, p, offset=0.5)
        sc = score_of(a, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[grid] double‑pow p={p:.3f} score={sc:.6f}")

    # Beta‑distribution -------------------------------------------------
    alphas = np.arange(0.20, 3.01, 0.20)
    betas = alphas.copy()
    for alpha in alphas:
        for beta in betas:
            if time.time() > deadline:
                break
            a = beta_sequence(n, alpha, beta)
            sc = score_of(a, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[grid] beta α={alpha:.2f} β={beta:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Local refinement with Adam (multiple restarts).
    # ------------------------------------------------------------------
    print("[refine] starting Adam optimisation")
    runs = 4
    for run in range(runs):
        if time.time() > deadline:
            break
        # Start from the best found so far, but add a tiny jitter
        a_start = best_a.copy()
        a_start += (rng.random(n) - 0.5) * 1e-5
        a_start = project_onto_simplex(a_start)

        refined_a, refined_score = adam_optimize(
            a_start,
            n,
            deadline,
            max_iters=25000,
            lr=0.28,
            gamma=5000.0,
        )
        if refined_score < best_score:
            best_score, best_a = refined_score, refined_a.copy()
            print(f"[refine] run {run} improved global best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")

    return best_a.tolist()
