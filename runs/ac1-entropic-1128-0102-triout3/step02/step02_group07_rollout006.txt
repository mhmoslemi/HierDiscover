"""
Propose a candidate step‑function for the “max‑convolution” challenge.

Algorithm
---------
1. **Generate a diverse pool of seed vectors**  
   • Power‑law families  a_i ∝ (i+offset)^(-p)  
   • Beta‑distribution families  a_i ∝ x^{α‑1}(1‑x)^{β‑1} (x = i/(n‑1))  
   • Half‑Gaussian   a_i ∝ exp(−(i/σ)²)  
   • Full (centered) Gaussian   a_i ∝ exp(−((i‑c)/σ)²)  
   • A few random Dirichlet draws.  
   All vectors are projected onto the probability simplex (non‑negative,
   sum = 1).

2. **Quick screening** – compute the raw score
   S = 2·n·max(a*a) (the same value returned by ``evaluate_sequence``
   when the sum is one).  Keep the *k* best seeds (k≈5).

3. **Soft‑max gradient refinement**  
   For each of the selected seeds run a projected‑gradient descent that
   minimizes a smooth approximation of the max‑convolution:
   * compute the self‑convolution c = a⋆a;
   * form soft weights w_k = exp(α·(c_k−max c))   (α grows slowly);
   * the gradient of the smoothed objective is
        g = 2·(w ⋆ a)                (cross‑correlation)
     which we obtain by a single FFT‑based convolution.
   * Take a step opposite to g, clip the step so that no component becomes
     negative, and re‑project onto the simplex.
   * Learning‑rate η and smoothing α are decayed / increased each iteration.
   * The best vector seen throughout the refinement is kept.

   The refinement runs as long as time permits (max ≈ 10 000 000 iterations,
   but usually stops much earlier).

4. **Final sanity check** – evaluate the best vector with the official
   ``evaluate_sequence`` and return it.

The whole routine respects a user‑provided time budget (default 950 s,
leaving a 5 s safety margin).  Multiple starts, diverse families and a
soft‑max gradient typically improve the previous best score (≈ 1.546) and
have a good chance of reaching the target 1.503.

All helper functions are defined at module level; there are no closures
or file‑system access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1. BASIC TOOLS
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a⋆a using a real FFT (full length = 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


def cross_corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation w⋆a (full length = len(w)+len(a)‑1) via FFT.
    Returns the full array; the lag‑0 entry is at index len(a)-1.
    """
    L = len(w) + len(a) - 1
    nfft = 1 << ((L - 1).bit_length())
    W = np.fft.rfft(w, n=nfft)
    A = np.fft.rfft(a, n=nfft)
    corr = np.fft.irfft(W * np.conj(A), n=nfft)
    return corr[:L]


# ----------------------------------------------------------------------
# 2. PARAMETRIC SEEDS
# ----------------------------------------------------------------------
def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


def beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    # discretised Beta(α,β) on [0,1]
    x = np.linspace(0, 1, n, dtype=np.float64)
    # add a tiny epsilon to avoid exact zeros when α<1 or β<1
    eps = 1e-12
    a = np.maximum(x, eps) ** (alpha - 1.0) * np.maximum(1.0 - x, eps) ** (beta - 1.0)
    a_sum = a.sum()
    a /= a_sum
    return a


def half_gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i / sigma) ** 2)
    a_sum = a.sum()
    a /= a_sum
    return a


def full_gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(- ((i - c) / sigma) ** 2)
    a_sum = a.sum()
    a /= a_sum
    return a


# ----------------------------------------------------------------------
# 3. SMOOTH‑MAX GRADIENT REFINEMENT
# ----------------------------------------------------------------------
def refine_sequence(a_start: np.ndarray,
                    max_iters: int,
                    deadline: float,
                    rng: np.random.Generator,
                    eta_start: float = 0.08,
                    alpha_start: float = 10.0,
                    eta_decay: float = 0.9995,
                    alpha_factor: float = 1.001,
                    jitter_prob: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent that minimises a soft‑max approximation of
    max(a*a).  Returns the best vector seen and its raw score
    (2·n·max_convolution) – the same quantity that ``evaluate_sequence``
    returns when the sum equals one.
    """
    n = a_start.shape[0]
    a_cur = a_start.copy()
    # raw score for a vector with sum = 1
    cur_score = 2.0 * n * conv_fft(a_cur).max()
    best_a = a_cur.copy()
    best_score = cur_score

    eta = eta_start
    alpha = alpha_start

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # self‑convolution and its maximum
        conv = conv_fft(a_cur)
        max_val = conv.max()

        # soft weights (exponential smoothing of the peak)
        w = np.exp(alpha * (conv - max_val))   # all entries ≤ 1

        # gradient of the smoothed objective: 2·(w⋆a)[lag=0…n‑1]
        corr = cross_corr_fft(w, a_cur)
        grad = 2.0 * corr[n - 1: n - 1 + n]   # length n

        # take a step against the gradient
        step = eta * grad
        # never subtract more than 99 % of the current mass in any coordinate
        step = np.minimum(step, a_cur * 0.99)
        a_trial = a_cur - step

        # occasional tiny jitter to escape plateaus
        if rng.random() < jitter_prob:
            jitter = (rng.random(n) - 0.5) * 1e-6
            a_trial += jitter

        # project back onto the probability simplex
        a_trial = project_onto_simplex(a_trial)

        # evaluate trial
        trial_score = 2.0 * n * conv_fft(a_trial).max()

        # keep the globally best ever seen
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            print(f"[refine iter {it:06d}] new global best = {best_score:.6f}")

        # accept improvements unconditionally, otherwise keep the old vector
        if trial_score <= cur_score:
            a_cur = a_trial
            cur_score = trial_score

        # decay learning‑rate and increase smoothing
        eta = max(eta * eta_decay, 1e-7)
        alpha *= alpha_factor

    return best_a, best_score

    seeds = []

    # power‑law family
    p_vals = np.linspace(0.30, 0.80, 11)          # 0.30 … 0.80
    offsets = [0.3, 0.5, 1.0, 2.0]
    for p in p_vals:
        for off in offsets:
            seeds.append(power_law_sequence(n, p, offset=off))

    # beta family
    alphas = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    betas = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    for a in alphas:
        for b in betas:
            seeds.append(beta_sequence(n, a, b))

    # half‑Gaussian family
    sigma_factors = [0.05, 0.07, 0.10, 0.15, 0.20]   # fraction of n
    for sf in sigma_factors:
        sigma = sf * n
        seeds.append(half_gaussian_sequence(n, sigma))

    # full‑Gaussian (centered) family
    sigma_factors_center = [0.08, 0.12, 0.18]
    for sf in sigma_factors_center:
        sigma = sf * n
        seeds.append(full_gaussian_sequence(n, sigma))

    # a handful of pure random Dirichlet vectors
    for _ in range(5):
        a = rng.random(n)
        a /= a.sum()
        seeds.append(a)

    # ------------------------------------------------------------------
    # 4.2 quick screening – keep the best few raw seeds
    # ------------------------------------------------------------------
    best_raw_score = np.inf
    best_raw_vec = None
    screened = []          # (score, vector)

    print("[search] screening seed pool")
    for idx, a in enumerate(seeds):
        if time.time() > deadline:
            break
        # raw score (sum=1 ⇒ objective = 2·n·max)
        conv = conv_fft(a)
        raw_score = 2.0 * n * conv.max()
        screened.append((raw_score, a))
        if raw_score < best_raw_score:
            best_raw_score = raw_score
            best_raw_vec = a.copy()
        if (idx + 1) % 50 == 0:
            print(f"   screened {idx + 1}/{len(seeds)} – best raw = {best_raw_score:.6f}")

    # keep the top‑k seeds for refinement
    screened.sort(key=lambda t: t[0])
    top_k = 5
    top_candidates = [vec.copy() for (_, vec) in screened[:top_k]]
    print(f"[search] top‑{top_k} raw scores: {[f'{s:.6f}' for s,_ in screened[:top_k]]}")

    # ------------------------------------------------------------------
    # 4.3 refinement of the top candidates
    # ------------------------------------------------------------------
    best_score = best_raw_score
    best_vec = best_raw_vec.copy()
    refine_iters = 200_000          # generous upper bound; will stop early by time

    print("[refine] starting gradient refinement of top candidates")
    for cand_idx, a_start in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(f"   candidate {cand_idx + 1}/{len(top_candidates)} (raw score {2.0*n*conv_fft(a_start).max():.6f})")
        refined_a, refined_score = refine_sequence(
            a_start,
            max_iters=refine_iters,
            deadline=deadline,
            rng=rng,
            eta_start=0.08,
            alpha_start=12.0,
            eta_decay=0.9995,
            alpha_factor=1.001,
            jitter_prob=0.001,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_vec = refined_a.copy()
            print(f"[refine] *** new overall best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.4 opportunistic random restarts (if time permits)
    # ------------------------------------------------------------------
    print("[random] entering random‑restart phase")
    while time.time() < deadline:
        # generate a random simple‑x vector (Dirichlet with concentration 0.5)
        a_rand = rng.gamma(0.5, 1.0, size=n)
        a_rand /= a_rand.sum()
        # do a short refinement (few thousand steps)
        refined_a, refined_score = refine_sequence(
            a_rand,
            max_iters=15_000,
            deadline=deadline,
            rng=rng,
            eta_start=0.06,
            alpha_start=10.0,
            eta_decay=0.998,
            alpha_factor=1.0015,
            jitter_prob=0.002,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_vec = refined_a.copy()
            print(f"[random] new best from restart = {best_score:.6f}")

        # safety break if very little time left
        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 4.5 final sanity check & return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"[done] best reported score = {final_score:.6f}")
    return best_vec.tolist()
