"""
Simulated‑annealing / gradient‑descent hybrid for minimizing
the maximum entry of a self‑convolution of a non‑negative step function.

The algorithm works directly on the convolution using an O(n) incremental
update: moving a small amount of mass from index i to j changes the
full linear convolution only by two sliced additions and a few point
updates.  This makes it possible to perform millions of cheap moves
inside the 1000 s budget.  The search mixes:
  * a steepest‑descent step that moves mass from the source with the
    largest gradient (the index that contributes most to the current
    maximal convolution entry) to a low‑gradient target,
  * random small moves for exploration,
  * occasional global jitter when no improvement has been seen for a long
    time.
The best sequence encountered is returned (scaled to the usual
ℓ₁‑norm √(2 n) for compatibility with the reference implementation).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_and_scale(a: np.ndarray) -> np.ndarray:
    """
    Returns a copy of a with non‑negative entries, normalised to sum = 1.
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        # fall back to uniform distribution
        a = np.full_like(a, 1.0 / a.size)
        s = 1.0
    return a / s


def _recompute_conv(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a*a (real output).
    """
    return np.convolve(a, a)  # length = 2·n‑1


def _apply_move(a: np.ndarray, conv: np.ndarray,
                i: int, j: int, delta: float) -> None:
    """
    Perform an in‑place move of `delta` mass from index i to index j.
    Both the vector `a` and its convolution `conv` are updated.
    The caller must guarantee `delta <= a[i]`.
    """
    n = a.size
    # convolution update using the *old* a (still stored in the array)
    conv[j:j + n] += 2.0 * delta * a
    conv[i:i + n] -= 2.0 * delta * a
    conv[2 * j] += delta * delta
    conv[2 * i] += delta * delta
    conv[i + j] -= 2.0 * delta * delta

    # vector update
    a[i] -= delta
    a[j] += delta


def _revert_move(a: np.ndarray, conv: np.ndarray,
                 i: int, j: int, delta: float) -> None:
    """
    Undo the move applied by `_apply_move`.  `a` must already have been
    restored to its pre‑move values.
    """
    n = a.size
    conv[j:j + n] -= 2.0 * delta * a
    conv[i:i + n] += 2.0 * delta * a
    conv[2 * j] -= delta * delta
    conv[2 * i] -= delta * delta
    conv[i + j] += 2.0 * delta * delta


def _gradient_indices(a: np.ndarray, k_max: int):
    """
    Return two indices (src, tgt) suitable for a descent step based on the
    exact sub‑gradient of the maximal convolution entry at `k_max`.
    The source index has the largest gradient component, the target the
    smallest (within the active range of the gradient).
    """
    n = a.shape[0]
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    grads = 2.0 * a[k_max - idx]   # gradient values for the active indices
    src = idx[grads.argmax()]
    tgt = idx[grads.argmin()]
    return int(src), int(tgt)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # initialise the sequence ------------------------------------------------
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # a reasonably large default length
        a = np.random.dirichlet(np.ones(2401)).astype(np.float64)

    a = _project_and_scale(a)       # sum = 1, non‑negative
    n = a.size
    conv = _recompute_conv(a)

    cur_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    best_obj = cur_obj
    print(f"Initial objective (2·n·max conv): {cur_obj:.6f}")

    # ------------------------------------------------------------------
    # annealing / descent parameters
    # ------------------------------------------------------------------
    # move size as a fraction of the source mass
    move_frac = 0.25
    # probability of using a directed gradient move
    prob_grad = 0.60

    # Metropolis temperature (not used for purely improving moves, but kept
    # for occasional uphill acceptance)
    temperature = cur_obj * 0.02
    temp_decay = 0.99999

    # bookkeeping
    iter_counter = 0
    no_improve = 0
    RESTART_LIMIT = 150_000        # after this many non‑improving steps we jitter
    RECOMPUTE_INTERVAL = 50_000    # full recompute to wipe out rounding errors

    while time.time() < deadline:
        iter_counter += 1

        # --------------------------------------------------------------
        # choose a move
        # --------------------------------------------------------------
        if np.random.rand() < prob_grad:
            # directed move based on the exact sub‑gradient
            k_max = int(conv.argmax())
            src, tgt = _gradient_indices(a, k_max)
            if src == tgt or a[src] <= 1e-16:
                # fallback to a random move if gradient is degenerate
                src = np.random.randint(0, n)
                tgt = np.random.randint(0, n - 1)
                if tgt >= src:
                    tgt += 1
        else:
            # completely random move
            src = np.random.randint(0, n)
            tgt = np.random.randint(0, n - 1)
            if tgt >= src:
                tgt += 1

        # amount to transfer (cannot exceed what is available at src)
        max_delta = a[src]
        if max_delta <= 1e-16:
            continue
        delta = max_delta * np.random.rand() * move_frac
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # apply the candidate move (updates a and conv in‑place)
        # --------------------------------------------------------------
        _apply_move(a, conv, src, tgt, delta)

        new_obj = 2.0 * n * conv.max()

        # --------------------------------------------------------------

        accepted = False

        else:

            if np.random.rand() < np.exp(-(new_obj - cur_obj) / (temperature + 1e-12)):
                accepted = True

        if accepted:
            cur_obj = new_obj
            no_improve = 0
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                print(f"[{iter_counter:08d}] New best: {best_obj:.6f}")
        else:
            # reject: revert both a and conv
            a[src] += delta
            a[tgt] -= delta
            _revert_move(a, conv, src, tgt, delta)
            no_improve += 1

        # --------------------------------------------------------------
        # temperature annealing
        # --------------------------------------------------------------
        temperature = max(temperature * temp_decay, 1e-12)

        # --------------------------------------------------------------
        # periodic maintenance
        # --------------------------------------------------------------
        if iter_counter % RECOMPUTE_INTERVAL == 0:
            # re‑normalise to eliminate accumulated round‑off errors
            a = _project_and_scale(a)
            conv = _recompute_conv(a)
            cur_obj = 2.0 * n * conv.max()
            print(f"[{iter_counter:08d}] Re‑normalised. Obj={cur_obj:.6f}")

        if no_improve >= RESTART_LIMIT:
            # global jitter to escape a plateau
            jitter = np.random.randn(n) * 0.001
            a = np.maximum(a + jitter, 0.0)
            a = _project_and_scale(a)
            conv = _recompute_conv(a)
            cur_obj = 2.0 * n * conv.max()
            temperature = cur_obj * 0.02
            no_improve = 0
            print(f"[{iter_counter:08d}] Jitter applied. Obj={cur_obj:.6f}")

        # early exit if we beat the target bound
        if best_obj < 1.503:
            print(f"Target reached at iteration {iter_counter}")
            break

    # ------------------------------------------------------------------
    # final scaling to the conventional ℓ₁‑norm √(2·n) and verification
    # ------------------------------------------------------------------
    S = np.sqrt(2.0 * n)
    final_seq = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iter_counter} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
