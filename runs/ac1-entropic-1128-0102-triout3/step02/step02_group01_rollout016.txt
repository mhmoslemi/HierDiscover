def decode_params(p, K):
    w = np.exp(p[:K])
    mu = 1/(1+np.exp(-p[K:2*K]))
    sigma = np.exp(p[2*K:])
    return w, mu, sigma

def generate_seq_from_params(p, n):
    w, mu, sigma = decode_params(p, K)
    x = np.linspace(0,1,n)
    # compute gaussian contributions: using broadcasting
    # shape: K x n
    diff = x[None,:] - mu[:,None]  # K,n
    a = np.sum(w[:,None] * np.exp(-0.5 * (diff / sigma[:,None])**2), axis=0)
    # Ensure nonnegative (gaussian always >0). Normalize
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / s
    return a

class ScoreEvaluator:
    def __init__(self, n, K):
        self.n = n
        self.K = K
        self.x_grid = np.linspace(0.0, 1.0, n)
        self.fft_len = 1 << ((2*n - 1).bit_length())
        # define bounds
        self.bounds = []
        # log_w bounds:
        for _ in range(K):
            self.bounds.append((-5, 5))
        # mu_raw bounds:
        for _ in range(K):
            self.bounds.append((-5, 5))
        # log_sigma bounds:
        for _ in range(K):
            self.bounds.append((-4.5, 0.5))   # sigma approx [0.011, 1.65]
    def decode(self, p):
        K = self.K
        w = np.exp(p[:K])
        mu_raw = p[K:2*K]
        mu = 1.0/(1.0+np.exp(-mu_raw))
        sigma = np.exp(p[2*K:])
        return w, mu, sigma
    def sequence_from_params(self, p):
        w, mu, sigma = self.decode(p)
        # compute contributions
        diff = self.x_grid[None,:] - mu[:,None]
        a = np.sum(w[:,None] * np.exp(-0.5 * (diff / sigma[:,None])**2), axis=0)
        s = a.sum()
        if s == 0:
            a = np.full(self.n, 1.0/self.n)
        else:
            a = a / s
        return a
    def __call__(self, p):
        a = self.sequence_from_params(p)
        Af = np.fft.rfft(a, n=self.fft_len)
        conv = np.fft.irfft(Af * Af, n=self.fft_len)[:2*self.n - 1]
        max_b = conv.max()
        return 2.0 * self.n * max_b

def subgradient_step(a, eps_rel, step):
    n = a.shape[0]
    fft_len = 1 << ((3*n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    # conv for auto-convolution:
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    max_val = conv.max()
    # compute indicator e:
    eps = max_val * eps_rel
    e = np.zeros(2*n - 1, dtype=np.float64)
    e[conv >= max_val - eps] = 1.0
    # compute g via convolution with reversed a:
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    E = np.fft.rfft(e, n=fft_len)
    # convolution a_rev * e:
    conv_rev_e = np.fft.irfft(A_rev * E, n=fft_len)[:3*n-2]   # Actually result length = len(a_rev)+len(e)-1 = n + (2n-1) -1 = 3n-2
    # take slice:
    g = 2.0 * conv_rev_e[n-1 : n-1 + n]  # length n
    # step
    a_new = a - step * g
    # projection onto simplex
    a_proj = _project_simplex(a_new)
    return a_proj, max_val

    n = a.shape[0]
    # precompute FFT lengths
    fft_len_auto = 1 << ((2*n - 1).bit_length())
    fft_len_grad = 1 << ((3*n - 2).bit_length())
    best_seq = a.copy()
    best_score = _score_of_seq(a)
    it = 0
    # predetermined max iterations large but limited by time
    while time.time() < deadline:
        it += 1
        # auto convolution
        Af = np.fft.rfft(a, n=fft_len_auto)
        conv = np.fft.irfft(Af * Af, n=fft_len_auto)[:2*n-1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] iter {it}, n={n}, score={cur_score:.6f}")

Propose a candidate step‑function (list of non‑negative heights) with a low
evaluation score.  The algorithm proceeds as follows:

1.  Initialise from the previously discovered construction `height_sequence_1`
    if it exists; otherwise start from a random point on the probability simplex.

2.  Upsample the initial shape to a larger length (default 4000).  The length
    is chosen to give the optimiser more degrees of freedom while keeping the
    per‑iteration cost moderate.

3.  Run a long projected sub‑gradient descent on the convex objective
    `max(conv(a, a))`.  The sub‑gradient is computed efficiently with FFTs:
    for a set of “near‑max’’ convolution indices (those within a relative
    tolerance `eps_rel` of the current maximum) we build a binary mask `e`,
    then the gradient `g` is obtained as the convolution of the reversed
    sequence with this mask.  The step size decays as `step0 / sqrt(it)` and
    the tolerance `eps_rel` is linearly decreased from ~0.05 down to 0.001 as
    the allotted time elapses.  After each step the iterate is projected back
    onto the probability simplex (Euclidean projection).

4.  If any time remains a second short refinement pass with a tighter
    tolerance is performed.

The whole routine respects the supplied `budget_s` (default 1000 s) and
returns the best sequence found as a plain Python list of floats.

import time
import math

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Evaluation score for a sequence whose sum is 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        x_old = np.linspace(0.0, 1.0, len(init))
        x_new = np.linspace(0.0, 1.0, target_n)
        a = np.interp(x_new, x_old, init)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        print(f"Upsampled to target length {target_n}")
    else:
        a = init.copy()

    # ------------------------------------------------------------------
    # 3) intensive projected sub‑gradient optimisation
    # ------------------------------------------------------------------
    best_seq, best_score = _subgradient_optimize(a, deadline, start_time)

    # ------------------------------------------------------------------
    # 4) optional short final refinement with tighter tolerance
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _subgradient_optimize(best_seq, deadline, start_time)

    # ------------------------------------------------------------------
    # 5) fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(100, 1.0 / 100)

    # Return as plain Python floats
    return [float(x) for x in best_seq]
