"""
propose_candidate

Multi‑peak sub‑gradient descent on the simplex with simulated‑annealing,
exponential‑moving‑average momentum and occasional random restarts.
Instead of using only the single maximal convolution entry,
the gradient is assembled from the K largest convolution values
(so the algorithm flattens all near‑maxima simultaneously).
The step is projected onto the probability simplex after each update.
A simple annealing schedule and a small amount of random noise help escape
local plateaus.  The routine runs for the given time budget and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """Half‑binomial coefficients (1‑z)^‑½ truncated to length n,
    normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x*x using a real FFT (length 2*len‑1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def eval_score(a: np.ndarray) -> float:
    """Score = 2 * n * max(a*a)  (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = conv_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that minimizes
        2 * n * max(conv(a, a)) / (sum(a)**2)

    The implementation uses a multi‑peak sub‑gradient descent with
    momentum, simulated annealing and occasional random restarts.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Starting point – reuse a previously found sequence if available
    init_seq = globals().get('height_sequence_1')
    if isinstance(init_seq, list) and len(init_seq) > 0:
        a = np.array(init_seq, dtype=np.float64)
        a = np.clip(a, 0.0, None)                # safety
        s = a.sum()
        if s < 1e-12:
            n = 2500
            a = sqrt_series(n)
        else:
            n = a.shape[0]
            a = a / s                              # normalise to sum = 1
    else:
        n = 2500
        a = sqrt_series(n)

    # ------------------------------------------------------------------
    # Hyper‑parameters (feel free to adjust)
    max_iter = 20_000_000                  # hard upper bound
    eta_start = 0.12                       # base step‑size
    temperature_start = 0.04               # annealing temperature
    decay_interval = 2500                  # how often we decay eta / temp
    restart_patience = 120_000             # iterations without improvement
    noise_prob = 0.008                     # chance to add a tiny jitter
    noise_scale = 1e-4                     # magnitude of jitter
    top_k = 12                             # number of largest convolution entries used in the gradient
    momentum_coeff = 0.9                   # exponential moving average momentum

    # ------------------------------------------------------------------
    cur_score = eval_score(a)
    best_a = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # momentum accumulator (same shape as a)
    grad_mom = np.zeros_like(a)

    deadline = time.time() + budget_s - 5.0   # safety margin
    eta = eta_start
    temperature = temperature_start
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iter:
        # ----- convolution and peak identification -----
        conv = conv_fft(a)
        max_val = conv.max()
        # pick the top‑k indices (they may be symmetric – duplicates are harmless)
        if top_k >= conv.size:
            peak_idxs = np.arange(conv.size)
        else:
            # argpartition gives unsorted top‑k; we don't need ordering
            peak_idxs = np.argpartition(-conv, top_k - 1)[:top_k]

        # ----- multi‑peak sub‑gradient -----
        grad = np.zeros_like(a)
        # weight each peak proportionally to its relative height
        weights = conv[peak_idxs] / max_val
        for wk, k in zip(weights, peak_idxs):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = k - idx
            grad[idx] += 2.0 * wk * a[jdx]

        # ----- momentum update -----
        grad_mom = momentum_coeff * grad_mom + (1.0 - momentum_coeff) * grad

        # ----- take a step (additive) and project -----
        step = eta * grad_mom
        # protect against too‑large steps (no entry can become negative)
        step = np.minimum(step, a * 0.95)
        a_trial = a - step

        # occasional tiny random perturbation
        if rng.random() < noise_prob:
            a_trial += (rng.random(n) - 0.5) * noise_scale

        # project back onto the simplex (ensures non‑negativity and sum=1)
        a_trial = project_onto_simplex(a_trial)

        # ----- evaluate trial -----
        trial_score = eval_score(a_trial)

        # keep the global best
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            last_improve = it
            print(f"[iter {it:08d}] new best score = {best_score:.6f}")

        # ----- acceptance rule (simulated annealing) -----
        if trial_score <= cur_score:
            a = a_trial
            cur_score = trial_score
        else:
            delta = trial_score - cur_score
            if temperature > 0 and rng.random() < np.exp(-delta / temperature):
                a = a_trial
                cur_score = trial_score
            # else reject – keep current a

        # ----- schedule decay -----
        if (it + 1) % decay_interval == 0:
            eta = max(eta * 0.93, 1e-5)
            temperature = max(temperature * 0.92, 1e-7)

        # ----- restart if stuck -----
        if it - last_improve > restart_patience:
            a = rng.dirichlet(np.full(n, 0.6))
            a = project_onto_simplex(a)          # safety
            cur_score = eval_score(a)
            last_improve = it
            print(f"[iter {it:08d}] restart – new random start, score={cur_score:.6f}")

        it += 1

        # occasional lightweight progress report
        if it % 50000 == 0:
            print(f"[progress] it={it}, cur_score={cur_score:.6f}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final verification with the provided security‑checked evaluator
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
