"""
Hybrid top‑k sub‑gradient descent for the max‑self‑convolution bound.

The algorithm searches over non‑negative sequences (treated as a probability
distribution, i.e. sum = 1) and tries to make the self‑convolution as flat as
possible.  At each iteration we:

* Compute the full self‑convolution by FFT.
* Pick the `top_k` indices where the convolution is largest (these drive the
  current max).
* Build a sub‑gradient that reduces the contribution of those peaks
  (the exact sub‑gradient of the max‑convolution is the sum of the gradients
  of the selected indices).
* Add a tiny entropy‑regularisation term, which encourages a spread‑out
  solution.
* Perform a projected (Euclidean) gradient step onto the probability simplex.
* Occasionally (with probability `rand_prob`) move a random fraction of mass
  between two coordinates (exploration) and every `smooth_every`
  iterations apply a 3‑point smoothing filter.
* Keep the best sequence seen so far and restart from it (with a small
  perturbation) if no improvement occurs for many iterations.

The routine respects the 1000 s time budget, prints progress, and finally
returns the best sequence found as a plain Python list.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective (identical to `evaluate_sequence`
    but without the security checks).  The vector `a` is assumed to be
    non‑negative; its sum is used explicitly.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def compute_gradient(a: np.ndarray, top_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (approximate) max‑convolution objective.
    For each selected index k we add the gradient 2*a_{k-i}.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in top_idx:
        # Valid i satisfy 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad

    """
