"""
Propose a candidate sequence that (hopefully) beats the previous 1.546‑score.

The search proceeds in three stages:

1. **Broad parametric sweep** –  We generate sequences from three simple
   families (power‑law, exponential, power‑law × exponential) for a moderate
   length (default n = 4000).  For each family we sweep a few parameters,
   normalise the vector to sum‑to‑one and keep the best by the cheap
   proxy  score = 2·n·max(conv).

2. **Local adaptive gradient descent** –  Starting from the best proxy
   candidate we minimise the true objective using a projected sub‑gradient
   method.  The sub‑gradient is the sum of the gradients of *all* convolution
   entries that are (almost) maximal (active set).  An adaptive back‑tracking
   line‑search picks a step size; occasional random jitter and a simple
   simulated‑annealing acceptance rule help to escape plateaus.

3. **Random restarts & polishing** –  While time permits we repeatedly
   perturb the current best vector, re‑run a short gradient descent, and
   keep any improvement.

All operations are fully vectorised, use an FFT for the self‑convolution,
and the vector is always projected onto the probability simplex (non‑negative,
unit‑sum).  The routine respects the 1000 s budget: the default `budget_s`
parameter leaves a small safety margin.

The function returns a plain ``list[float]`` which can be handed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# Helper utilities (top‑level, no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a.fill(1.0 / n)
    else:
        a /= total
    return a


def exponential_sequence(n: int, q: float) -> np.ndarray:
    """a_i ∝ exp(-q·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i)
    total = a.sum()
    if total == 0.0:
        a.fill(1.0 / n)
    else:
        a /= total
    return a


def combined_sequence(n: int, p: float, q: float, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}·exp(-q·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) * np.exp(-q * i)
    total = a.sum()
    if total == 0.0:
        a.fill(1.0 / n)
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Discrete self‑convolution x*x using real FFT. Result length = 2·len(x)-1."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 } (Duchi et al. 2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal (or negative); return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_active_peaks(a: np.ndarray,
                             conv: np.ndarray,
                             tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of the *max* convolution entry (including ties).
    ``tol`` is an absolute tolerance; any index with
    conv[idx] ≥ max(conv) – tol is treated as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat nearly‑maximal entries as active
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # i runs from lo … hi, j = k-i
        i_vals = np.arange(lo, hi + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

            else:
                # simple back‑off
                eta_local *= 0.9
                if eta_local < 1e-6:
                    break
        # record improvement if any
        if cur_score < global_best_score:
            global_best_score = cur_score
            global_best_a = a.copy()
            print(f"[restart] improved proxy score = {global_best_score:.6f}")

    # ---------- final sanity check ----------
    # The evaluator rescales internally, but we keep sum = 1 for transparency.
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[done] best proxy score = {global_best_score:.6f}")
    print(f"[done] final reported score = {final_score:.6f}")
    print(f"[info] best family & params = {best_params}")

    return global_best_a.tolist()
