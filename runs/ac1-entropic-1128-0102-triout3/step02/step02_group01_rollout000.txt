def _score(a):
    n = a.shape[0]
    # compute conv using FFT power-of-2 length
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af*Af, n=fft_len)[:2*n-1]
    return 2.0*n*conv.max()

def _grad(a, eps_rel=0.001):
    n = a.shape[0]
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af*Af, n=fft_len)[:2*n-1]
    max_val = conv.max()
    # near-max indices
    eps = max_val * eps_rel
    K = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k-(n-1))
        high = min(n-1, k)
        i_idx = np.arange(low, high+1)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g, max_val

def _symmetrize(a):
    return (a + a[::-1]) / 2.0

def propose_candidate(seed=0, budget_s=1000):
    """Search for a step‑function (non‑negative heights) that minimises
    evaluate_sequence.  The algorithm:
    1. Starts from a family of symmetric beta‑shaped sequences of increasing

    2. Refines the best shape with an Adam‑style projected gradient descent

    3. Gradually upsamples the current best shape to larger lengths (≈1.5×)
       and repeats the refinement.
    The whole search respects the time budget and prints intermediate

            best_seq = a.copy()
            print(f"[{time.time()-start:.2f}s] New best: n={len(a)}, score={s:.6f}")
        return s

        s = init.sum()

1. **Initial shape** – start from the known construction (if available) or a

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a[i]=a[n-1-i] while keeping the sum unchanged."""
    return (a + a[::-1]) / 2.0


def _score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) for a already normalised to sum 1."""
    n = a.shape[0]

    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _grad(a: np.ndarray, eps_rel: float = 0.001):
    """
    Compute a sub‑gradient of the objective 2·n·max(conv) at `a`.
    Returns (gradient, current max convolution value).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * eps_rel
    K = np.where(conv >= max_val - eps)[0]  # near‑max indices
    g = np.zeros_like(a)

    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1)
        j_idx = k - i_idx
        # each pair (i,j) contributes 2*a_j to the gradient wrt a_i
        g[i_idx] += 2.0 * a[j_idx]

    return g, max_val
