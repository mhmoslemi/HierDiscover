"""Propose a step‑function sequence that (empirically) minimizes
`evaluate_sequence`.  

The search consists of two stages:

1. **Mirror‑descent on the simplex** (Adam on log‑variables).  
   A soft‑max smoothing of the max–convolution is annealed to the exact
   sub‑gradient.  Restarts and occasional jitter avoid stagnation.
2. **Simulated‑annealing fine‑tuning** around the best solution found
   with the mirror‑descent.  Small random “mass‑transfer” moves and
   occasional global multiplicative perturbations are accepted according
   to a temperature schedule.

Both stages keep the ℓ₁‑norm fixed to `sqrt(2·n)` (the problem
normalisation) and enforce non‑negativity.  The best sequence seen before
the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# low‑level linear‑algebra helpers (FFT‑based convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the first (2·len(a)‑1) entries (real values).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray,
                  conv: np.ndarray,
                  tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    Uses a numerically stable soft‑max.
    """
    # stable soft‑max weights
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:                     # safeguard
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a (the derivative of a*a)
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    # only the n entries that influence the “central” part of a*a
    grad = 2.0 * grad_full[n - 1: n - 1 + n]
    return grad


def _grad_max(a: np.ndarray,
              conv: np.ndarray,
              k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the index k_max.
    The derivative of (a*a)[k] w.r.t. a_i equals 2·a_{k‑i}
    if the index lies inside the support, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


# ----------------------------------------------------------------------
# main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for up to ``budget_s`` seconds,
    using a two‑phase optimisation (Adam‑mirror descent + simulated
    annealing) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # initialise sequence length and ℓ₁‑norm (problem normalisation)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        n = a0.size
    else:
        n = 2401                     # default length used in earlier runs
        a0 = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64)

    # required ℓ₁‑norm (the problem statement normalises to sqrt(2·n))
    S = np.sqrt(2.0 * n)
    a0 = a0 * (S / a0.sum())
    a0 = np.maximum(a0, 0.0)

    # bookkeeping of the best solution
    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[INIT] length={n}, initial score={best_score:.6f}")

    # pre‑compute FFT length (a power of two for speed)
    nfft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # PHASE 1 : Adam mirror‑descent on log‑variables
    # ------------------------------------------------------------------
    eps = 1e-12
    log_theta = np.log(best_a + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta = 0.18                     # base learning rate in log‑space

    m = np.zeros_like(log_theta)
    v = np.zeros_like(log_theta)

    # soft‑max temperature schedule
    tau = 0.4
    tau_min = 0.008
    total_iters_est = 300_000
    tau_decay = (tau_min / tau) ** (1.0 / total_iters_est)

    # stagnation handling
    no_improve = 0
    restart_limit = 7000

    start_time = time.time()
    iteration = 0
    while time.time() < deadline and (time.time() - start_time) < budget_s * 0.6:
        iteration += 1

        # ----------------------------------------------------------
        # reconstruct a from log‑variables and renormalise to sum S
        # ----------------------------------------------------------
        a = np.exp(log_theta)
        sum_a = a.sum()
        if sum_a == 0.0:                     # numerical safety
            a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64) * S
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            continue
        a = a * (S / sum_a)

        # ----------------------------------------------------------
        # evaluate current objective (max entry of self‑convolution)
        # ----------------------------------------------------------
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = cur_max                       # equals evaluate_sequence after normalisation

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{iteration:08d}] New best (Adam) = {best_score:.6f}")
        else:
            no_improve += 1

        # early exit if we are already below a reasonable target
        if best_score < 1.5:
            print("[STOP] Target‐like region reached by Adam.")
            break

        # ----------------------------------------------------------
        # gradient of the (smoothed) max‑convolution
        # ----------------------------------------------------------
        if tau > tau_min * 2:
            grad_a = _grad_softmax(a, conv, tau, nfft)
        else:
            k_max = int(conv.argmax())
            grad_a = _grad_max(a, conv, k_max)

        # chain rule to log‑space
        grad_log = grad_a * a

        # ----------------------------------------------------------
        # Adam update on log‑variables (descent)
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step                     # descent step

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.008:
            jitter = np.random.randn(n) * (eta * 0.07)
            log_theta = log_theta + jitter

        # temperature annealing
        tau = max(tau * tau_decay, tau_min)

        # restart if stagnated too long
        if no_improve >= restart_limit:
            a = best_a.copy()
            perturb = np.random.randn(n) * (eta * 0.25)
            a = a * np.exp(perturb)          # multiplicative perturbation
            a = a * (S / a.sum())
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{iteration:08d}] Restarted after stagnation.")
    # ------------------------------------------------------------------
    # PHASE 2 : Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_score = best_score

    # SA temperature schedule (geometric decay)
    T = cur_score * 0.25
    Tmin = cur_score * 0.0005
    decay = 0.99995

    # parameters for the two types of proposals
    global_sigma = 0.025        # std‑dev for global multiplicative noise
    prob_global = 0.18          # probability to use a global move

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # propose a new candidate
        # ------------------------------------------------------------------
        if np.random.rand() < prob_global:
            # global multiplicative perturbation
            eps_vec = np.random.randn(n) * global_sigma
            cand = cur_a * np.exp(eps_vec)
            cand = cand * (S / cand.sum())
        else:
            # local mass‑transfer move
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            # move a random fraction (up to 40 %) of a[i] to a[j]
            delta = cur_a[i] * (np.random.rand() * 0.4)
            if delta <= 0.0:
                continue
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
            # non‑negativity guaranteed by construction

        # evaluate candidate
        conv = _convolve_self_fft(cand, nfft)
        cand_score = conv.max()

        # acceptance test
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cur_score + 1e-12 < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[SA] New best = {best_score:.6f}")

        # decay temperature
        T = max(T * decay, Tmin)

    # ------------------------------------------------------------------
    # final sanity projection (ensures exact ℓ₁‑norm) and verification
    # ------------------------------------------------------------------
    final_a = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(final_a.tolist())
    print(f"[DONE] Finished after {iteration} Adam iters. Best score = {final_score:.6f}")
    return final_a.tolist()
