"""
Gradient descent on a smooth approximation of the max‐auto‑correlation.
Uses soft‑max parametrisation of the step heights and an Adam optimiser.
Beta (temperature) is increased during optimisation so the loss
smoothly turns into the true max.  Several sequence lengths and independent
restarts are tried within the given time budget.  The best sequence found
is returned.
"""
import time, math
import numpy as np
from typing import List, Tuple

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A
    conv = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv

def _conv_linear_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    X = np.fft.rfft(np.pad(x, (0, N - len(x))))
    Y = np.fft.rfft(np.pad(y, (0, N - len(y))))
    Z = X * Y
    conv = np.fft.irfft(Z)[:n]
    return conv

def _softmax(z: np.ndarray) -> np.ndarray:
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()

def _score_from_a(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())

def _adam_update(z, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, m, v

def _gradient_smooth_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Returns ∂ L / ∂ a where
        L(a) = (1/β) log Σ_i exp(β * (a*a)_i)
    """
    n = a.shape[0]
    conv = _conv_fft(a)                      # length 2n-1
    max_conv = conv.max()
    # stabilised exponentials
    w = np.exp(beta * (conv - max_conv))
    Z = w.sum()
    # convolution needed for gradient
    w_rev = w[::-1]

    # a * w   and a * w_rev  (linear convolutions)
    aw = _conv_linear_fft(a, w)              # length 3n-2
    aw_rev = _conv_linear_fft(a, w_rev)      # length 3n-2

    # extract terms for each index k = 0..n-1
    idx = np.arange(n)

"""
Gradient‑descent on a smooth approximation of the max‑auto‑correlation.

The step‑function is parametrised by logits `z` and the actual heights are
`a = softmax(z)` (hence non‑negative and ∑a = 1).  For a given vector
`a` the convolution `c = a * a` (linear, length 2·n‑1) is computed by
FFT.  The objective we would like to minimise is

    L(a) = max_k c_k

which is not differentiable.  We replace the max by a smooth log‑sum‑exp,
`L_β(a) = (1/β)·log Σ_k exp(β·c_k)`.  For a fixed β the gradient
∂L_β/∂a can be written analytically:

    ∂L_β/∂a = (1/Z)·Σ_k w_k·∂c_k/∂a,
    where w_k = exp(β·c_k) and Z = Σ_k w_k.

Since   ∂c_k/∂a_i = a_{k‑i} + a_{i‑k},
the weighted sum can be obtained with two linear convolutions
(a * w) and (a * w_rev).

An Adam optimiser updates the logits `z`.  The temperature β starts
small (smooth loss) and is multiplied by a factor each step until a
large upper bound, so the loss slowly turns into the true max.
Several sequence lengths are explored; each run keeps the best sequence
found.  After the gradient phase a cheap random‑swap local refinement
is applied.  The function respects the 1000 s budget and returns the
best‑scoring non‑negative height list.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A
    conv = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv


def _conv_linear_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    X = np.fft.rfft(np.pad(x, (0, N - len(x))))
    Y = np.fft.rfft(np.pad(y, (0, N - len(y))))
    Z = X * Y
    conv = np.fft.irfft(Z)[:n]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _score_from_a(a: np.ndarray) -> float:
    """Exact evaluation used during the search: 2·n·max(a*a)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def _adam_update(z: np.ndarray,
                 grad: np.ndarray,
                 m: np.ndarray,
                 v: np.ndarray,
                 t: int,
                 lr: float,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Classic Adam update for logits."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return z - step, m, v


def _gradient_smooth_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ exp(β·c) w.r.t. a,
    where c = a*a (linear convolution).
    """
    n = a.shape[0]

    # 1) convolution a*a
    conv = _conv_fft(a)                     # length 2n‑1
    max_conv = conv.max()

    # 2) soft‑max weights (stabilised)
    w = np.exp(beta * (conv - max_conv))    # all ≤ 1, max = 1
    Z = w.sum()

    # 3) weighted sum of ∂c/∂a
    w_rev = w[::-1]

    # a * w  and a * w_rev  (linear convolutions, length 3n‑2)
    aw = _conv_linear_fft(a, w)
    aw_rev = _conv_linear_fft(a, w_rev)

    idx = np.arange(n)
    term1 = aw_rev[2 * n - 2 - idx]         # contribution from a * w_rev
    term2 = aw[idx]                         # first n entries of a * w
    grad_raw = term1 + term2                # shape (n,)

    return grad_raw / Z
