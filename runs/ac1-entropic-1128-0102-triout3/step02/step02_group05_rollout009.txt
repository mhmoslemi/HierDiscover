def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """...
    Finds a non‑negative sequence on the simplex that 
    approximately minimises 2 n max(a*a) / (sum a)^2,
    i.e. the maximum of the self‑convolution.
    Uses a smoothed max (soft‑max) gradient with momentum,
    projected onto the probability simplex, plus occasional
    random mass‑transfer moves and a final hill‑climbing
    refinement.  The routine stops before the given time budget
    and returns the best sequence found as a plain Python list.
    """
    import numpy as np, time, math

    # parameters
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # choose length - try a moderate size.
    n = 2500  # maybe 2500
    # note: length can be changed for speed/quality trade‑off

    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                a0 = prev[:n].copy()
            else:
                a0 = np.zeros(n, dtype=np.float64)
                a0[:prev.size] = prev
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None or a0.sum() == 0:

        idx = np.arange(1, n+1, dtype=np.float64)

        a0 /= a0.sum()

"""
Propose a low‑value step‑function sequence.

The routine searches for a non‑negative vector `a` (interpreted as a step
function) that minimises

        2·n·max(a*a) / (sum(a)**2)

where `a*a` is the self‑convolution.  The objective is invariant to a
global scaling, therefore the algorithm works on the probability simplex
(sum = 1).

The method combines three ingredients:

1. **Smoothed‑max gradient** – the maximum of the convolution is
   approximated by a soft‑max with a temperature that gradually cools.
   This yields a smooth gradient that is a weighted average of the
   sub‑gradients `g_i = 2·a_{k‑i}` of each convolution entry.

2. **Adam‑style adaptive step** – the gradient is fed to an Adam optimiser
   and the resulting update is projected back onto the simplex.

3. **Exploration moves** – occasional random mass‑transfer steps and a
   stagnation‑triggered perturbation help escape local minima.  After the
   time budget is almost exhausted a cheap hill‑climbing refinement is
   performed.

The whole procedure stops before the given time budget and returns the
best sequence found as a plain Python list.  All numbers are non‑negative,
so the external `evaluate_sequence` routine can be called on the result
without further processing.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum_i x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)          # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def fast_score(arr: np.ndarray) -> float:
    """
    Fast evaluation of the objective (scale‑invariant version of
    `evaluate_sequence`).  Assumes `arr` is non‑negative.
    """
    s = arr.sum()
    if s <= 0:
        return np.inf
    n = arr.size
    size = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fft * fft, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that gives a low value
    for `evaluate_sequence`.  The function runs for at most ``budget_s``
    seconds, prints progress, and returns the best sequence found as a
    list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.20   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a sequence length.  3000 is a good trade‑off (used in the
    # reference implementation) and fits comfortably into memory.
    n = 3000

    # ------------------------------------------------------------------
    # Initialise the vector.
    if 'height_sequence_1' in globals():
        try:
            tmp = np.array(height_sequence_1, dtype=np.float64)
            if tmp.size >= n:
                a0 = tmp[:n].copy()
            else:
                a0 = np.zeros(n, dtype=np.float64)
                a0[:tmp.size] = tmp
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None or a0.sum() == 0:
        # power‑law start (a_i ∝ (i+1)^‑p) normalised to sum = 1
        p = 0.55
        idx = np.arange(1, n + 1, dtype=np.float64)
        a0 = 1.0 / np.power(idx, p)
        a0 /= a0.sum()
    else:
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0.sum()

    a = a0.copy()
    best_seq = a.copy()
    best_val = fast_score(a)

    # ------------------------------------------------------------------
    # Pre‑compute the FFT length needed for cross‑correlation
    # (convolution of length n with length 2n‑1 → 3n‑2 points).
    size_fft = 1 << ((3 * n - 2).bit_length())

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # Soft‑max temperature schedule
    lambda_start, lambda_end = 2.0, 200.0
    lambda_ramp_iters = 200_000       # after this λ stays at lambda_end

    # Random‑exploration settings
    rand_move_prob = 0.025
    rand_move_frac = 0.15

    # Stagnation detection (restart if no improvement for many iters)
    last_improve = 0
    patience_iters = 100_000

    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # ----- self‑convolution (needed for both objective and soft‑max) -----
        fft_a = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fft_a * fft_a, size_fft)[: 2 * n - 1]

        # ----- current objective (sum(a) should be ≈ 1) -----
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = t
            print(f"[iter {t:6d}] new best {best_val:.6f}")

        # ----- soft‑max weights -----
        lam = lambda_start + (lambda_end - lambda_start) * min(1.0,
                                                              t / lambda_ramp_iters)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w /= w.sum()                     # shape (2n‑1,)

        # ----- gradient = 2·(a ⋆ w) -----
        w_rev = w[::-1]                  # reversed weight vector
        fft_w = np.fft.rfft(w_rev, size_fft)
        grad_full = np.fft.irfft(fft_a * fft_w, size_fft)   # length = size_fft
        # The required slice is grad_full[n‑1 : 2n‑1] (length n) and then reversed.
        slice_rev = grad_full[n - 1: 2 * n - 1]
        grad = 2.0 * slice_rev[::-1]     # shape (n,)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)   # learning rate ≈ 0.05

        a_candidate = proj_simplex(a - step)

        # ----- occasional random mass‑transfer move (exploration) -----
        if np.random.rand() < rand_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a_candidate[i] > 1e-12:
                delta = rand_move_frac * a_candidate[i] * np.random.rand()
                cand = a_candidate.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = proj_simplex(cand)
                a_candidate = cand

        a = a_candidate

        # ----- stagnation handling -----
        if t - last_improve > patience_iters:
            # Perturb around the current best to escape flat regions
            noise = np.random.rand(n)
            noise /= noise.sum()
            a = proj_simplex(best_seq * 0.7 + 0.3 * noise)
            last_improve = t
            print(f"[iter {t:6d}] stagnation reset")

        # ----- light reporting every 30 000 iterations -----
        if t % 30000 == 0:
            print(f"[iter {t:6d}] current {cur_val:.6f} best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final cheap hill‑climbing refinement (few thousand trials)
    print("[final] refinement phase")
    a = best_seq.copy()
    for _ in range(8_000):
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        delta = 0.10 * a[i] * np.random.rand()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = proj_simplex(cand)
        cand_val = fast_score(cand)
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand.copy()
            a = cand
        elif np.random.rand() < 0.001:   # occasional uphill acceptance
            a = cand

    print(f"[final] best value found: {best_val:.6f}")
    # Convert to plain Python list (float) for the external evaluator.
    return best_seq.tolist()
