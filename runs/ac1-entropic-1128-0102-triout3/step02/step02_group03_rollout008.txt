"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Work on the ℓ₁‑simplex {x ≥ 0, Σx = √(2n)} – on this simplex the
  evaluation function reduces to the maximal entry of the self‑convolution.
* Initialise from the best known sequence (``height_sequence_1``) if it
  exists, otherwise from a uniform distribution.
* Repeatedly perform **projected Adam** updates on a *smooth* surrogate of
  the max‑convolution:
      f_λ(a) = (1/λ)·log ∑_k exp(λ·(a∗a)_k)      (log‑sum‑exp)
  When λ is small the surrogate is very smooth (easy to descend);
  λ is increased gradually so that the surrogate approaches the true max.
* Gradient of the surrogate can be written in closed form using a
  convolution with the soft‑max weights; both the convolution and the
  gradient are computed with a single FFT pair for efficiency.
* After each Adam step the iterate is projected back onto the ℓ₁‑simplex.
* To escape possible plateaux a small fraction of *random mass‑transfer*
  moves is performed, and very rarely a full restart from a Dirichlet
  draw.
* The best vector (according to the true max‑convolution) is remembered
  and returned when the time budget (≈1000 s) expires.

The method improves on the previous sub‑gradient/annealing scheme by
using a smooth objective, adaptive step‑sizes (Adam) and a systematic
λ‑schedule, which together give a more reliable descent towards the
theoretical bound ≈1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the full (2·len(a)‑1) real‑valued result.
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_softmax(a: np.ndarray, w: np.ndarray, nfft: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate with respect to ``a``.
    ``w`` must be a probability vector (soft‑max weights) of length
    2·len(a)‑1, i.e. the same shape as the full self‑convolution.
    Returns a vector of length ``len(a)``.
    """
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(A_rev * W, n=nfft)[: 2 * a.size - 1]
    # the gradient entries correspond to the middle block of length n
    grad = 2.0 * grad_full[a.size - 1 : a.size - 1 + a.size]
    return grad

            # soft‑max weights (stable implementation)
            w = np.exp(lam * (conv - cur_max))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum

            # gradient of the surrogate
            grad = _grad_softmax(a, w, nfft)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            # bias‑corrected estimates
            m_hat = m / (1.0 - beta1 ** (iteration + 1))
            v_hat = v / (1.0 - beta2 ** (iteration + 1))

            # diminishing step size
            lr = base_lr / np.sqrt(iteration + 1)

            step = -lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # candidate after projection onto the ℓ₁‑simplex
            a_candidate = _project_to_simplex(a + step, S)

        elif r < p_grad + p_rand:
            # ---- random mass‑transfer move (simple simulated annealing) ----
            # choose source according to its current mass (more likely big entries)
            if a.sum() > 0:
                src_probs = a / a.sum()
                i = np.random.choice(n, p=src_probs)
            else:
                i = np.random.randint(n)

            # choose different destination uniformly
            j = i
            while j == i:
                j = np.random.randint(n)

            # move a modest fraction of the source mass
            eps_transfer = 0.07
            delta = eps_transfer * a[i]
            if delta <= 0:
                delta = eps_transfer * S / n

            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = _project_to_simplex(a_candidate, S)

        else:
            # ---- very occasional full restart from a Dirichlet draw ----
            alpha = np.full(n, 0.4)
            a_candidate = np.random.dirichlet(alpha) * S
            print(f"[{iteration:07d}] Full restart (Dirichlet) performed.")
            # No projection needed – Dirichlet already satisfies the simplex

        # ----------------------------------------------------------
        # evaluate candidate using the true objective (max‑convolution)
        # ----------------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        cand_score = cand_max

        # acceptance rule: greedy + Metropolis
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
            cur_max = cand_max
        else:
            # tiny chance to add a random jitter to escape shallow plateaus
            if np.random.rand() < 0.006:
                jitter = np.random.randn(n) * (base_lr * 0.4)
                a = _project_to_simplex(a + jitter, S)

        # temperature decay
        temperature = max(temperature * temp_decay, min_temp)

        # increase smoothing parameter λ (makes the surrogate closer to the true max)
        lam = min(lam_max, lam * lam_factor)

        # restart if we have not improved for a long time
        if no_improve >= max_no_improve:
            alpha = np.full(n, 0.3)
            a = np.random.dirichlet(alpha) * S
            no_improve = 0
            print(f"[{iteration:07d}] Stagnation restart triggered.")

        # periodic progress report (lightweight)
        if iteration % print_interval == 0 and iteration > 0:
            elapsed = time.time() - (deadline - budget_s + 0.5)
            print(f"Iter {iteration:07d} – best so far {best_score:.6f}, λ={lam:.2f}, temp={temperature:.2e}")

        iteration += 1

    # ------------------------------------------------------------------
    # final verification and return
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)          # final safety projection
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Search finished after {iteration} iterations. Final best score = {final_score:.6f}")
    return best_a.tolist()
