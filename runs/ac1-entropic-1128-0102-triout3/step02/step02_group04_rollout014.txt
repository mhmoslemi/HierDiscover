"""
Propose a step‑function sequence that minimizes the given evaluation metric.

The algorithm consists of two phases:

1. **LP‑based redistribution** – an implementation of the iterative scheme
   described in the statement.  Starting from the classic half‑binomial
   (coefficients of (1‑z)^‑½) we repeatedly
      * compute M = max(f*f)
      * solve a linear program that maximises Σg subject to the linear
        constraints (f∗g) ≤ M and g ≥ 0,
      * rescale g to have the same ℓ¹‑norm as f and move a fraction τ towards g.
   This phase quickly drives down the peak of the self‑convolution.

2. **Projected sub‑gradient descent with simulated annealing** – after the
   LP phase we polish the solution using a fast FFT‑based sub‑gradient step
   (the exact gradient of the active convolution entry) together with a
   Euclidean projection onto the probability simplex, a decreasing step size,
   occasional random perturbations and restarts.

Both phases respect the time budget (default 950 s, leaving a safety margin)
and keep the best sequence encountered.  The returned list contains non‑negative
heights whose sum is 1, which is the normalisation naturally used by the
evaluation routine.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of a_i = C(2i,i) / 4**i (half‑binomial series)
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex {x ≥ 0,
    Σx = 1}.  Implements the Duchi‑Et‑Al algorithm (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Fast full convolution x * x using a real FFT.  Length of the result is
    2*len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def lp_optimize_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the linear program
        maximise   Σ g_i
        subject to (f ∗ g)_k ≤ M   for all k = 0..2n‑2,
                    g_i ≥ 0.
    Returns the optimal g (not normalised).
    """
    n = f.shape[0]
    # Build dense constraint matrix A (shape (2n‑1, n)).
    # Row k, column i contains f[i] iff 0 ≤ k‑i < n.
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σg  → minimise -Σg
    bounds = [(0, None)] * n
    b_ub = M * np.ones(2 * n - 1, dtype=np.float64)

    # Highs (interior‑point) is robust for moderate sizes.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        raise RuntimeError(f'LP failed: {res.message}')
    return res.x


def score_from_sequence(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(a,a)) assuming Σa = 1.
    """
    n = a.shape[0]
    return 2.0 * n * conv_fft(a).max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 950.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises
        2·n·max(conv(a,a)) / (sum(a)²).

    The routine first applies a few LP‑based redistribution steps and then
    refines the result with a projected sub‑gradient descent (with simulated
    annealing).  The best sequence found before the time budget expires is
    returned as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size
    n = 2500                     # keep the same length as the reference run
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # initialise – try to use a provided construction if available
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if seq0.shape == (n,):
            a = seq0.copy()
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            print("[init] starting from supplied height_sequence_1")
        else:
            a = sqrt_series(n)
            print("[init] supplied sequence has wrong size – using sqrt_series")
    else:
        a = sqrt_series(n)
        print("[init] using sqrt_series")

    best_a = a.copy()
    best_score = score_from_sequence(best_a)
    print(f"[init] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP driven redistribution
    # Parameters
    lp_iters = 6                     # number of outer LP updates
    tau = 0.4                        # convex‑combination weight for g
    lp_start = time.time()

    for it in range(lp_iters):
        if time.time() >= deadline:
            break

        # current peak of self‑convolution
        M = conv_fft(a).max()

        # solve LP
        try:
            g = lp_optimize_g(a, M)
        except Exception as e:
            print(f"[lp] iteration {it} failed: {e}")
            break

        sum_g = g.sum()
        if sum_g == 0.0:
            print("[lp] zero‑sum g – aborting LP phase")
            break

        # normalise g to have the same ℓ¹‑norm as a (here Σa = 1)
        g_norm = g / sum_g

        # convex update and projection back onto the simplex
        a = (1.0 - tau) * a + tau * g_norm
        a = project_onto_simplex(a)

        cur_score = score_from_sequence(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[lp iter {it}] improved score = {best_score:.6f}")

        # small safety check – if improvement is negligible, stop early
        if it > 0 and abs(best_score - cur_score) < 1e-7:
            print("[lp] negligible improvement – terminating LP phase")
            break

    print(f"[lp] finished after {it + 1} iterations, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent with simulated annealing
    cur_a = a.copy()
    cur_score = best_score
    eta = 0.12                     # initial step size
    temperature = 0.04             # initial annealing temperature
    decay_every = 2500
    last_improve = 0
    it = 0
    max_iter = 10_000_000          # safety cap

    while time.time() < deadline and it < max_iter:
        # compute convolution and locate the maximal index
        conv = conv_fft(cur_a)
        max_val = conv.max()
        k = int(conv.argmax())

        # exact sub‑gradient of the active convolution entry
        grad = np.zeros_like(cur_a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        jdx = k - idx
        grad[idx] = 2.0 * cur_a[jdx]

        # step (limit to 95 % of current values to stay non‑negative)
        step = np.minimum(eta * grad, cur_a * 0.95)
        trial = cur_a - step

        # tiny random jitter (helps escape shallow plateaus)
        if rng.random() < 0.008:
            trial += (rng.random(n) - 0.5) * 1e-4

        trial = project_onto_simplex(trial)

        # evaluate trial (fast, because sum(trial)=1)
        trial_score = 2.0 * n * conv_fft(trial).max()

        # accept if better, otherwise accept with Metropolis probability
        if trial_score <= cur_score:
            cur_a = trial
            cur_score = trial_score
        else:
            delta = trial_score - cur_score
            if temperature > 0 and rng.random() < np.exp(-delta / temperature):
                cur_a = trial
                cur_score = trial_score

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
            last_improve = it
            print(f"[iter {it:08d}] new best score = {best_score:.6f}")

        # schedule decay
        if (it + 1) % decay_every == 0:
            eta = max(eta * 0.93, 1e-5)
            temperature = max(temperature * 0.92, 1e-7)

            cur_a = project_onto_simplex(cur_a)
            cur_score = 2.0 * n * conv_fft(cur_a).max()
            last_improve = it
            print(f"[restart {it}] fresh Dirichlet start, score = {cur_score:.6f}")

        it += 1

    # final evaluation with the provided routine (adds clipping etc.)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")
