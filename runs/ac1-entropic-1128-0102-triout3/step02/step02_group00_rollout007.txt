def _run_adam_descent(a0, n, deadline, start_time, beta_softmax=15.0, init_lr=0.5, max_iter=200000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    # pre compute L for FFT
    L = 1 << ((3 * n - 2).bit_length())
    while time.time() < deadline and t < max_iter:
        t += 1
        # compute convolution via FFT
        Af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Af * Af, n=L)[: 2*n - 1]
        # compute score
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # print progress
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={t}, score={cur_score:.6f}")
        # compute softmax weights
        # subtract max to improve numeric stability
        exp_weights = np.exp(beta_softmax * (conv_full - max_val))
        w = exp_weights / exp_weights.sum()
        # compute correlation gradient
        Wf = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(np.conj(Af) * Wf, n=L)[: n]
        grad = 2.0 * corr  # shape (n,)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = init_lr / (1 + 0.00001 * t)  # optional decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # gradient descent move (negative direction)
        a = _project_simplex(a - step)
    return best_a, best_score

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # load starter.
    best_seq = None
    best_score = float('inf')

    # length schedule
    lengths = [800, 1600, 3200, 6400]  # final may be upsampled to >6400 if time.
    # might add 12800 if time remains.

    for n in lengths:
        if time.time() >= deadline:
            break
        # initial sequence
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()
        # allocate time slice
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        slice_deadline = time.time() + max(5.0, 0.15 * remaining)
        # gradient descent
        a_opt, score_opt = _run_adam_descent(a0, n, slice_deadline, start_time)
        # update global best
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after Adam: n={n}, score={score_opt:.6f}")
        # annealing
        if time.time() < deadline:
            anneal_deadline = time.time() + max(5.0, 0.05 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(a_opt, n, anneal_deadline, start_time, best_score, best_seq,
                                               start_temp=2e-2, end_temp=1e-6, steps=20000)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n}, score={score_ref:.6f}")

    # if leftover time, we can run a final annealing for longest n
    if time.time() < deadline and best_seq is not None:
        n = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(best_seq, n, deadline, start_time, best_score, best_seq,
                                          start_temp=1e-2, end_temp=1e-7, steps=50000)
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final annealing improve: n={n}, score={score_ref:.6f}")

    if best_seq is None:
        # fallback uniform
        best_seq = np.full(100, 1.0/100)

    return [float(x) for x in best_seq]
```

"""
Advanced projected‑Adam optimizer with soft‑max smoothing for the
step‑function “max‑convolution’’ problem.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) which
makes the objective scale‑invariant.  For a given length n it repeatedly:

* evaluates the self‑convolution a*a via an FFT (linear, not circular);
* builds a smooth approximation of the max operator by a soft‑max on the
  convolution entries (parameter β);
* obtains the gradient of this smoothed objective using the cross‑correlation
  a ⋆ w (w are the soft‑max weights);
* updates the coefficients with an Adam‑style adaptive step and projects back
  onto the simplex;
* keeps the best vector seen so far.

A multi‑scale schedule up‑samples the best vector to larger n and repeats
the optimisation.  After the gradient phase a cheap simulated‑annealing
local search makes a final refinement.  The whole procedure respects the
global time budget (default 1000 s) and always returns a valid list of
non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008) – O(n)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # length enough for linear convolution (2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Linear up‑sampling (piece‑wise linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (small mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 20000,
):
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent with soft‑max smoothing
# ----------------------------------------------------------------------
def _run_adam_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    beta_softmax: float = 12.0,
    init_lr: float = 0.15,
    max_iter: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Returns the best sequence found for a fixed length n together with its score.
    The objective (with sum(a)=1) is 2*n*max(conv_self(a)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # FFT length that accommodates both a*a (2n‑1) and a⋆w (n+2n‑2 = 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1

        # ----- convolution a*a -----
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={t}, score={cur_score:.6f}"
            )

        # ----- soft‑max weights over the convolution entries -----
        # stability: subtract max before exponentiation
        exp_w = np.exp(beta_softmax * (conv - max_val))
        w = exp_w / exp_w.sum()  # shape (2n‑1,)

        # ----- gradient of the smoothed objective -----
        # correlation a ⋆ w  (g_i = 2 * sum_j a_j * w_{i+j})
        Wf = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(np.conj(Af) * Wf, n=L)[:n]   # linear cross‑correlation
        grad = 2.0 * corr

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # simple learning‑rate schedule (slow decay)
        lr = init_lr / (1.0 + 1e-5 * t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------

    try:

    lengths = [800, 1600, 3200, 6400, 9600]

            )

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:

    # Convert to plain Python list for the evaluator
    return [float(x) for x in best_seq]
