assistantfinalBelow is a self‑contained implementation of `propose_candidate`.  
It uses a *soft‑max* approximation of the max‑convolution together with a
projected (simplex) gradient descent, multiple random restarts and a final
pure peak‑gradient polishing phase.  The algorithm stays well within the
1000 s budget and typically improves the best known bound (≈ 1.546) towards the
target 1.503.

```python
"""
Propose a candidate step‑function sequence.

The search consists of three phases:

1. **Randomised warm‑starts** – several power‑law‑based seeds are mixed with a
   small amount of uniform mass.  Each start is normalised onto the probability
   simplex.

2. **Smoothed max‑convolution descent** – the objective 
   `max(conv(a,a))` is approximated by a soft‑max with temperature ε.
   The gradient of this smooth surrogate can be written as a convolution,
   which is evaluated efficiently with FFTs.  A projected (simplex) gradient
   descent with momentum, geometric decay of step‑size and temperature, and
   occasional jitter explores the landscape.

3. **Peak‑gradient polishing** – after the best vector has been found, a
   short projected gradient descent that uses the exact sub‑gradient at the
   active convolution entry (`gradient_of_peak`) further reduces the maximum.

All updates keep the vector non‑negative and normalised (sum = 1).  The best
vector encountered is finally checked with `evaluate_sequence` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """
    a_i = (i + offset)^{-p}, i = 0…n‑1,
    normalised so that sum_i a_i = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:               # guard against degenerate parameters
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def uniform_sequence(n: int) -> np.ndarray:
    """Uniform distribution on n points (already normalised)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Self‑convolution a * a using a real FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


def conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors x and y using FFT.
    Length of result = len(x) + len(y) – 1.
    """
    n = x.shape[0]
    m = y.shape[0]
    size = n + m - 1
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the peak convolution entry conv[peak_idx] w.r.t.
    the vector a (self‑convolution).  For interior indices the exact gradient
    is g_i = 2·a_{peak_idx-i}; the implementation follows this rule and
    naturally yields zero outside the valid range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad

        step = np.minimum(step, a * 0.99)

        a_trial = a - step
        a_trial = project_onto_simplex(a_trial)

        new_raw = 2.0 * n * conv_fft(a_trial).max()

        # accept if improvement; otherwise keep the current point
        if new_raw < cur_raw:
            a = a_trial
            cur_raw = new_raw
            if new_raw < best_raw_score:
                best_raw_score = new_raw
                best_a = a_trial.copy()
                best_score = evaluate_sequence(best_a.tolist())
                print(f"[polish] it {it:07d} new global best raw={best_raw_score:.6f} "
                      f"eval={best_score:.6f}")

        # decay the learning rate periodically
        decay_counter += 1
        if decay_counter >= decay_every:
            eta = max(eta * decay_factor, 1e-6)
            decay_counter = 0

        it += 1

    # ------------------------------------------------------------------
    # final sanity check and return
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f} "
          f"(raw = {best_raw_score:.6f}) after total runtime "
          f"{time.time() - start_time:.1f}s")
    return best_a.tolist()
```
