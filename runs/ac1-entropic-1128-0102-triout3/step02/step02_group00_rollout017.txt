def _run_adam(a0, n, deadline, start_time, K=10, max_iter=30000, lr0=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    for t in range(1, max_iter+1):
        if time.time() >= deadline: break
        conv = _conv_self(a)
        maxc = conv.max()
        cur_score = 2 * n * maxc
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # active set via tolerance
        tol = 0.02
        act = np.where(conv >= maxc * (1 - tol))[0]
        g = _subgradient_from_active(a, act)
        # Adam updates (before projection)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)
    return best_a, best_score

def _run_greedy_local_search(a0, n, deadline, start_time, max_steps=100000, init_eps=0.05, eps_decay=0.995, min_eps=1e-7):
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    maxc = conv.max()
    best_score = 2.0 * n * maxc
    eps = init_eps
    steps = 0
    while time.time() < deadline and steps < max_steps:
        steps += 1
        # compute gradient
        act = np.where(conv >= maxc * 0.999)[0]
        grad = _subgradient_from_active(a, act)
        # indices with max and min grad
        i = np.argmax(grad)
        j = np.argmin(grad)
        if i == j or a[i] == 0:
            # random move fallback
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j or a[i] == 0:
                continue
        delta = min(eps * a[i], a[i])  # can't transfer more than a[i]
        if delta <= 0:
            eps *= eps_decay
            continue
        a[i] -= delta
        a[j] += delta
        # ensure non-neg (just in case)
        if a[i] < 0:
            a[i] = 0
        # evaluate new conv max
        conv = _conv_self(a)
        new_maxc = conv.max()
        new_score = 2.0 * n * new_maxc
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            maxc = new_maxc
            # maybe increase eps
        else:
            # revert move
            a[i] += delta
            a[j] -= delta
            eps *= eps_decay
        # maintain eps min
        if eps < min_eps:
            eps = min_eps
    return best_a, best_score

"""
Multi‑stage optimization for the step‑function minimisation problem.

1. Parametric families (U‑shaped, power‑law, cosine) generate a modest set
   of candidate sequences for a given length n.
2. The best candidate is refined by projected gradient descent on the
   probability simplex (using a sub‑gradient built from all convolution
   entries that are close to the maximum).
3. A short greedy local search moves mass from indices with large gradient
   to those with small gradient.
4. Finally a cheap simulated‑annealing phase attempts to escape the last
   local basin.
5. The process is repeated for increasing lengths, up‑sampling the
   current best solution before the next stage.
"""

import time
import math
import numpy as np
# evaluate_sequence is assumed to be imported in the execution environment

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries non‑positive – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution a * a using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_from_active(a: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of the max‑convolution over the active set."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    if act_idx.size > 0:
        g /= act_idx.size
    return g


def _run_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    max_iter: int = 50000,
    step0: float = 0.6,
    beta: float = 0.9,
    tol: float = 0.02,
):
    """Projected gradient descent on the simplex, using a sub‑gradient
    built from all convolution entries that are within (1‑tol) of the
    current maximum.
    Returns the best point seen and its (2·n·max_conv) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        maxc = conv.max()
        cur_score = 2.0 * n * maxc

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] gradient improve n={n} iter={it} score={cur_score:.6f}"
            )

        act = np.where(conv >= maxc * (1 - tol))[0]
        if act.size == 0:
            act = np.array([conv.argmax()], dtype=int)

        grad = _subgradient_from_active(a, act)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 20000,
):
    """Simple mass‑transfer simulated annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] annealing improve n={n} iter={it} score={score:.6f}"
            )
    return best_seq, best_score


def _run_greedy_local(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    init_eps: float = 0.05,
    eps_decay: float = 0.995,
    min_eps: float = 1e-8,
    max_iter: int = 20000,
):
    """Deterministic greedy improvement that repeatedly moves a small
    amount of mass from the index with the largest sub‑gradient to the
    one with the smallest."""
    a = a0.copy()
    conv = _conv_self(a)
    maxc = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * maxc
    eps = init_eps

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # active set for the current maximum (tight tolerance)
        act = np.where(conv >= maxc * 0.999)[0]
        grad = _subgradient_from_active(a, act)

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j or a[i] == 0.0:
            # fallback to random pair
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue

        delta = min(eps * a[i], a[i])
        if delta <= 0:
            eps *= eps_decay
            continue

        a[i] -= delta
        a[j] += delta

        conv = _conv_self(a)
        new_maxc = conv.max()
        new_score = 2.0 * n * new_maxc

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            maxc = new_maxc
        else:
            # revert & reduce step size
            a[i] += delta
            a[j] -= delta
            eps *= eps_decay
            if eps < min_eps:
                break

    return best_a, best_score


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, then re‑normalised."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _generate_parametric_candidates(n: int, starter: np.ndarray | None):
    """Return a list of candidate sequences built from a few low‑dimensional
    families (U‑shaped power laws, cosine, etc.). All candidates lie on the
    probability simplex."""
    candidates = []
    uniform = np.full(n, 1.0 / n)
    candidates.append(uniform)

    idx = np.arange(n, dtype=float)

    # 1. Power of (i+1)*(n-i) – both positive and negative exponents
    base = (idx + 1) * (n - idx)
    betas = np.linspace(-2.0, 2.0, 21)  # step 0.2
    for beta in betas:
        shape = np.power(base, beta)
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            candidates.append(shape)

    # 2. Absolute cosine – high at the ends, low in the centre
    cos_shape = np.abs(np.cos(np.pi * idx / (n - 1)))
    for p in [0.5, 1.0, 2.0, 4.0]:
        shape = np.power(cos_shape, p)
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            candidates.append(shape)

    # 3. Mixtures of the above with the uniform vector
    mixes = [0.25, 0.5, 0.75]
    # copy the non‑uniform candidates generated so far
    non_uniform = candidates[1:].copy()
    for shape in non_uniform:
        for mix in mixes:
            seq = mix * uniform + (1.0 - mix) * shape
            seq /= seq.sum()
            candidates.append(seq)

    # 4. Include a (possibly) provided starter up‑sampled to length n
    if starter is not None:
        up = _upsample(starter, n)
        candidates.append(up)

    return candidates


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The algorithm combines cheap parametric exploration, projected gradient
    descent, a greedy local mass‑transfer phase and a final simulated‑annealing
    refinement.  It works on several lengths, each time up‑sampling the best
    solution found so far.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # Globals for the best solution seen across all lengths
    best_seq = None
    best_score = float("inf")
    best_n = None

    # Try to use an externally supplied starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # Length schedule – we gradually increase the dimension
    lengths = [800, 1600, 2400, 3600, 4800]
    # Shuffle a bit to avoid deterministic early‑stop bias
    np.random.shuffle(lengths)

    # The current best (normalized) sequence that will be up‑sampled
    current_best = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------
        # 1) generate a modest set of parametric candidates
        # ------------------------------------------------------------
        cand = _generate_parametric_candidates(n, current_best)
        local_best = None
        local_best_score = float("inf")

        for seq in cand:
            if time.time() >= deadline:
                break
            conv = _conv_self(seq)
            maxc = conv.max()
            score = 2.0 * n * maxc
            if score < local_best_score:
                local_best_score = score
                local_best = seq.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] n={n} param candidate score={score:.6f}"
                )

        # ------------------------------------------------------------
        # 2) Gradient‑descent refinement
        # ------------------------------------------------------------
        if local_best is None:
            continue
        # allocate a slice of remaining budget (≈30 % of the remaining time)
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(5.0, 0.30 * remaining)

        a_grad, score_grad = _run_gradient_descent(
            local_best,
            n,
            slice_deadline,
            start_time,
            max_iter=40000,
            step0=0.6,
            beta=0.9,
            tol=0.02,
        )
        if score_grad < best_score:
            best_score = score_grad
            best_seq = a_grad.copy()
            best_n = n
            print(
                f"[{time.time() - start_time:6.2f}s] new global best after gradient n={n} score={score_grad:.6f}"
            )

        # ------------------------------------------------------------
        # 3) Greedy local search (mass transfer guided by sub‑gradient)
        # ------------------------------------------------------------
        if time.time() < deadline:
            # give a small budget for greedy search
            greedy_deadline = time.time() + max(3.0, 0.10 * (deadline - time.time()))
            a_greedy, score_greedy = _run_greedy_local(
                a_grad,
                n,
                greedy_deadline,
                start_time,
                init_eps=0.05,
                eps_decay=0.995,
                max_iter=20000,
            )
            if score_greedy < best_score:
                best_score = score_greedy
                best_seq = a_greedy.copy()
                best_n = n
                print(
                    f"[{time.time() - start_time:6.2f}s] new global best after greedy n={n} score={score_greedy:.6f}"
                )
        else:
            a_greedy = a_grad

        # ------------------------------------------------------------
        # 4) Final annealing stage (if time permits)
        # ------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, 0.10 * (deadline - time.time()))
            a_anneal, score_anneal = _run_annealing(
                a_greedy,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=15000,
            )
            if score_anneal < best_score:
                best_score = score_anneal
                best_seq = a_anneal.copy()
                best_n = n
                print(
                    f"[{time.time() - start_time:6.2f}s] new global best after anneal n={n} score={score_anneal:.6f}"
                )

        # ------------------------------------------------------------
        # 5) Prepare for the next scale – up‑sample the current global best
        # ------------------------------------------------------------
        if best_seq is not None:
            current_best = best_seq / best_seq.sum()  # maintain simplex
        else:
            current_best = None

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a modest uniform distribution
        n = 1200
        best_seq = np.full(n, 1.0 / n)

    # Convert to plain Python list as required by the harness
    return [float(x) for x in best_seq]

"""
Multi‑stage optimisation for the step‑function minimisation problem.

The programme works in increasing dimensions.  For every length n it

1. builds a modest catalogue of low‑dimensional (“U‑shaped”, cosine,
   arcsine, power‑law) candidate sequences on the probability simplex,
2. refines the best candidate by projected gradient descent where the
   sub‑gradient is formed from all convolution entries that are close to
   the current maximum,
3. runs a short greedy mass‑transfer phase guided by the sub‑gradient,
4. optionally performs a cheap simulated‑annealing sweep,
5. up‑samples the current global best and proceeds to a larger n.

All steps respect the overall time budget (default 1000 s).  The final
list of non‑negative heights is returned in plain Python list form, ready
for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic geometry utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find the largest rho satisfying the KKT condition
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution a * a via FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_from_active(a: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max (a*a) over the active index set."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    if act_idx.size > 0:
        g /= act_idx.size
    return g


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _run_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    max_iter: int = 50000,
    step0: float = 0.6,
    beta: float = 0.9,
    tol: float = 0.02,
):
    """
    Projected gradient descent on the simplex.
    The sub‑gradient is the average of ∇(a*a)_k over all k with
    (a*a)_k ≥ (1‑tol)·max(a*a).
    Returns the best point observed and its 2·n·max(conv) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        maxc = conv.max()
        cur_score = 2.0 * n * maxc

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] gradient improve n={n} iter={it} score={cur_score:.6f}"
            )

        act = np.where(conv >= maxc * (1.0 - tol))[0]
        if act.size == 0:
            act = np.array([conv.argmax()], dtype=int)

        grad = _subgradient_from_active(a, act)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 20000,
):
    """Tiny mass‑transfer simulated annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] annealing improve n={n} iter={it} score={score:.6f}"
            )
    return best_seq, best_score


def _run_greedy_local(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    init_eps: float = 0.05,
    eps_decay: float = 0.995,
    min_eps: float = 1e-8,
    max_iter: int = 20000,
):
    """
    Greedy improvement that repeatedly slides a small amount of mass from the
    index with the largest sub‑gradient to the one with the smallest.
    """
    a = a0.copy()
    conv = _conv_self(a)
    maxc = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * maxc
    eps = init_eps

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        act = np.where(conv >= maxc * 0.999)[0]
        grad = _subgradient_from_active(a, act)

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j or a[i] == 0.0:
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue

        delta = min(eps * a[i], a[i])
        if delta <= 0:
            eps *= eps_decay
            continue

        a[i] -= delta
        a[j] += delta

        conv = _conv_self(a)
        new_maxc = conv.max()
        new_score = 2.0 * n * new_maxc

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            maxc = new_maxc
        else:
            # revert move and shrink step
            a[i] += delta
            a[j] -= delta
            eps *= eps_decay
            if eps < min_eps:
                break

    return best_a, best_score


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric candidate generation
# ----------------------------------------------------------------------


def _generate_parametric_candidates(n: int, starter):
    """Low‑dimensional families that are known to flatten a*a."""
    candidates = []

    # 0) uniform baseline
    uniform = np.full(n, 1.0 / n)
    candidates.append(uniform)

    idx = np.arange(n, dtype=float)

    # 1) Power of (i+1)*(n-i)   – both positive and negative exponents
    base = (idx + 1) * (n - idx)
    betas = np.linspace(-2.0, 2.0, 21)  # step 0.2
    for beta in betas:
        shape = np.power(base, beta)
        s = shape.sum()
        if s > 0:
            shape /= s
            candidates.append(shape)

    # 2) Cosine‑based U‑shape (high at the ends, low in the centre)
    cos_shape = np.abs(np.cos(np.pi * idx / (n - 1)))
    for p in (0.5, 1.0, 2.0, 4.0):
        shape = np.power(cos_shape, p)
        s = shape.sum()
        if s > 0:
            shape /= s
            candidates.append(shape)

    # 3) Arcsine‑like U‑shape
    arcsine = 1.0 / np.sqrt((idx + 0.5) * (n - 0.5 - idx))
    s = arcsine.sum()
    if s > 0:
        arcsine /= s
        candidates.append(arcsine)
        for p in (0.5, 1.0, 2.0):
            shape = np.power(arcsine, p)
            s2 = shape.sum()
            if s2 > 0:
                shape /= s2
                candidates.append(shape)

    # 4) Mixtures of the non‑uniform candidates with uniform
    mixes = (0.25, 0.5, 0.75)
    for shape in candidates[1:]:
        for mix in mixes:
            seq = mix * uniform + (1.0 - mix) * shape
            seq /= seq.sum()
            candidates.append(seq)

    # 5) Optional external starter (up‑sampled)
    if starter is not None:
        candidates.append(_upsample(starter, n))

    return candidates
