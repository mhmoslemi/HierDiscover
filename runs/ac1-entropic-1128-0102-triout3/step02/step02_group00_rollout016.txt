"""
Proposed solution:
- Minimises 2*n*max(conv) on the probability simplex.
- Uses a smoothed max (soft‑max) to obtain a dense sub‑gradient.
- Performs accelerated projected gradient descent with momentum.
- Refines the result with a simple simulated‑annealing local search.
- Progressively upsamples the best solution to larger lengths.
"""

import time, math, numpy as np

# Helper: Euclidean projection onto simplex
def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Convolution a*a (linear, not circular) using FFT
def _conv_self(a, L):
    """Return linear convolution a*a of length 2*len(a)-1 using FFT length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*len(a)-1]
    return conv, Af

# Linear correlation of a (length n) with w (length 2n-1):
# g_i = sum_j a_j * w[i+j]   (i=0..n-1)
def _correlation(a, w, L, Af=None):
    """Compute vector c where c[i]=sum_j a[j]*w[i+j] for i=0..n-1."""
    n = a.shape[0]
    if Af is None:
        Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(Af * np.conj(Wf), n=L)
    start = len(w) - 1          # = 2*n-2
    return corr_full[start:start + n]

# Upsample a sequence by linear interpolation, then renormalise.
def _upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Gradient descent with a smooth approximation of the max.
# ------------------------------------------------------------
def _run_smooth_gradient(a0, n, deadline, start_time,
                         beta_start=0.5, beta_end=200.0,
                         momentum_beta=0.9, step0=0.6,
                         max_iter=50000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    # FFT length able to hold linear correlation (3n-2)
    L = 1 << ((3 * n - 2).bit_length())
    momentum = np.zeros_like(a)

    while time.time() < deadline and it < max_iter:
        it += 1
        # convolution and its FFT (re‑use Af for the correlation)
        conv, Af = _conv_self(a, L)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"smooth‑grad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # --------- soft‑max weights for the active convolutions ----------
        # schedule β from β_start to β_end (log‑linear)
        progress = it / max_iter
        beta = beta_start * (beta_end / beta_start) ** progress

        # w_i ∝ exp(β (conv_i - max_val))  – numerically stable
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()   # size 2n‑1, sums to 1

        # --------- gradient = 2 · correlation(a, w)  --------------------
        corr = _correlation(a, w, L, Af=Af)   # length n
        grad = 2.0 * corr

        # momentum and diminishing step size
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
        step = step0 / math.sqrt(it)

        # Euclidean projected gradient step onto the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Very cheap simulated‑annealing fine tuning.

def _run_annealing(a0, n, deadline, start_time,
                   cur_best_score, cur_best_seq,
                   start_temp=5e-2, end_temp=1e-5, steps=20000):
    a = a0.copy()
    score = 2.0 * n * _conv_self(a, 1 << ((2*n-1).bit_length()))[0].max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # move up to 15 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop, 1 << ((2*n-1).bit_length()))[0].max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

        # keep the best solution for the next scale
        global_a = best_seq.copy()

    # -----------------------------------------------------------------
    # safety fallback
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    return [float(x) for x in best_seq]

"""
search for a step‑function minimising `evaluate_sequence`.

* Work on the probability simplex (∑aᵢ = 1); the objective becomes
  `score = 2·n·max(conv(a,a))`.
* Replace the non‑smooth max by a soft‑max with a temperature β that
  grows during the run.  The gradient of the smoothed objective is a
  weighted sum of the gradients of the individual convolution entries,
  which we compute efficiently by a single cross‑correlation via FFT.
* Perform a momentum projected‑gradient descent on the simplex.
* Finish each length with a cheap simulated‑annealing local search.
* Progressively up‑sample the best solution to larger lengths.
"""
import time
import math
import numpy as np

# ------------------------------------------------------------------
# Linear‑algebra helpers (FFT based)
# ------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, ∑x=1} (O(n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_self(a: np.ndarray, L: int):
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT length L.
    Returns (conv, Af) where Af is the rFFT of a (used later for correlation).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * len(a) - 1]
    return conv, Af


def _correlation(a: np.ndarray, w: np.ndarray, L: int, Af=None):
    """
    Linear correlation c[i] = Σⱼ aⱼ·w[i+j]   (i=0 … n‑1).

    w must have length 2·n‑1.  Returns a vector of length n.
    """
    n = a.shape[0]
    if Af is None:
        Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(Af * np.conj(Wf), n=L)
    start = len(w) - 1                     # = 2·n‑2
    return corr_full[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ------------------------------------------------------------------
# Smoothed‑max projected gradient descent on the simplex
# ------------------------------------------------------------------
def _run_smooth_gradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    beta_start: float = 0.5,
    beta_end: float = 150.0,
    momentum_beta: float = 0.9,
    step0: float = 0.6,
    max_iter: int = 40000,
):
    """
    Returns (best_seq, best_score) for a fixed length n.
    The objective is `2·n·max(conv(a,a))`.  A soft‑max with temperature β
    provides a dense sub‑gradient; the update is a projected gradient with
    momentum on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    # FFT length sufficient for a (len = n) and w (len = 2n‑1) correlation:
    L = 1 << ((3 * n - 2).bit_length())
    momentum = np.zeros_like(a)

    while time.time() < deadline and it < max_iter:
        it += 1
        conv, Af = _conv_self(a, L)               # self‑convolution
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"smooth‑grad improve: n={n}, iter={it}, score={cur_score:.6f}"
            )

        # -------- soft‑max weights for the active convolution entries ----------
        progress = it / max_iter
        beta = beta_start * (beta_end / beta_start) ** progress
        exp_vals = np.exp(beta * (conv - max_val))   # all ≤ 1, max = 1
        w = exp_vals / exp_vals.sum()                # length 2n‑1, ∑w = 1

        # -------- gradient = 2·correlation(a, w) -----------------------------
        corr = _correlation(a, w, L, Af=Af)          # length n
        grad = 2.0 * corr

        # momentum + diminishing step
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Tiny simulated‑annealing refinement (same idea as the reference code)
# ------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 15000,
):
    a = a0.copy()
    # exact score (no soft‑max) for the current point
    conv, _ = _conv_self(a, 1 << ((2 * n - 1).bit_length()))
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # move up to 15 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop, _ = _conv_self(a_prop, 1 << ((2 * n - 1).bit_length()))
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"anneal improve: n={n}, iter={it}, score={score:.6f}"
            )
    return best_seq, best_score
