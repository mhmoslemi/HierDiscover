"""
Hybrid projected‑gradient / stochastic search for a low‑max‑auto‑correlation
step‑function.

The algorithm works on the simplex {x ≥ 0, Σx = √(2 n)}.
* Sub‑gradient direction is obtained from the active (maximal) entries of the
  self‑convolution and is accumulated with a momentum term (β‑averaging).
* The update is performed with an exponential‑gradient (EG) step
  a_i ← a_i·exp(‑η · g_i) and then projected back onto the ℓ₁‑simplex.
* Random mass‑transfer moves and occasional full Dirichlet restarts keep the
  search diversified.
* A simple Metropolis acceptance rule together with a geometrically
  decaying temperature enables occasional uphill moves.
* The best sequence seen so far is kept and reported after the time budget
  (≈1000 s) expires.

The returned sequence is guaranteed to be non‑negative; the final objective
value equals max(self‑convolution) because the ℓ₁‑norm is fixed to √(2 n).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full (2·len(a)‑1) result (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_max_subgrad(a: np.ndarray, mask: np.ndarray, nfft: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the current active set.
    ``mask`` is a binary (0/1) array of length 2·len(a)‑1 where 1 marks the
    indices achieving the current maximal convolution value.
    The returned gradient has length len(a) and is the (averaged) gradient of
    those active entries.
    """
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.zeros_like(a)

    w = w / w_sum                     # turn into a probability vector
    a_rev = a[::-1]

    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative list) that minimises
    evaluate_sequence.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a small safety margin

    # --------------------------------------------------------------
    # Problem size – if a previous construction exists we continue from it
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2401                     # default length (same as previous run)
        a0 = None

    S = np.sqrt(2.0 * n)            # ℓ₁‑norm we keep fixed throughout
    nfft = _next_pow2(2 * n - 1)    # length for FFT‑based convolution

    # --------------------------------------------------------------
    # Initial point (projected onto the simplex)
    # --------------------------------------------------------------
    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        # a moderately concentrated Dirichlet draw – avoids the trivial uniform start
        alpha = np.full(n, 0.4)
        a = np.random.dirichlet(alpha) * S

    # bookkeeping
    conv = _convolve_self_fft(a, nfft)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    # momentum vector (used only for sub‑gradient steps)
    momentum = np.zeros_like(a)

    # --------------------------------------------------------------
    # Hyper‑parameters (hand‑tuned)
    # --------------------------------------------------------------
    beta = 0.9                     # momentum averaging coefficient
    step0 = 0.8 * S / n            # base step size for the EG update
    temperature = 0.2              # initial Metropolis temperature
    temp_decay = 0.9997            # geometric decay per iteration
    min_temp = 1e-7

    p_subgrad = 0.70
    p_random = 0.25                # the rest (≈0.05) are full restarts
    eps_mask = 1e-12

    no_improve = 0
    max_no_improve = 3000          # restart if stalled too long
    iteration = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---- current state -------------------------------------------------
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = cur_max                      # because Σa = √(2n) ⇒ the objective equals max(conv)

        # keep the global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:07d}] New best: {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # early termination if the target bound is met
        if best_score < 1.503:
            print(f"Target bound reached after {iteration} iterations (best={best_score:.6f})")
            break

        # --------------------------------------------------------------
        # Choose a move type
        # --------------------------------------------------------------
        r = np.random.rand()
        if r < p_subgrad:
            # -------- sub‑gradient EG step ---------------------------------
            mask = (conv >= cur_max - eps_mask).astype(float)
            grad = _grad_max_subgrad(a, mask, nfft)          # non‑negative sub‑gradient
            # momentum update (exponential moving average)
            momentum = beta * momentum + (1.0 - beta) * grad
            # EG update: a_i ← a_i · exp(‑η·momentum_i)
            eta = step0 / np.sqrt(iteration + 1.0)
            a_candidate = a * np.exp(-eta * momentum)
            a_candidate = _project_to_simplex(a_candidate, S)

        elif r < p_subgrad + p_random:
            # -------- random mass‑transfer move ----------------------------
            if a.sum() > 0.0:
                src_probs = a / a.sum()
                i = np.random.choice(n, p=src_probs)
            else:
                i = np.random.randint(n)
            j = i
            while j == i:
                j = np.random.randint(n)

            # transfer a fraction of the source mass
            eps_transfer = 0.12               # a bit larger than before
            delta = eps_transfer * a[i]
            if delta <= 0.0:
                delta = eps_transfer * S / n

            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = _project_to_simplex(a_candidate, S)

        else:
            # -------- full restart (Dirichlet) ----------------------------
            alpha = np.full(n, 0.3)          # slightly more peaked than the initial draw
            a_candidate = np.random.dirichlet(alpha) * S
            print(f"[{iteration:07d}] Full restart performed.")

        # --------------------------------------------------------------
        # Evaluate the candidate (fast convolution)
        # --------------------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        cand_score = cand_max                      # same normalisation as above

        # --------------------------------------------------------------
        # Acceptance rule (greedy + Metropolis)
        # --------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis probability with current temperature
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
            cur_max = cand_max
        else:
            # occasional tiny jitter to avoid getting stuck on flat regions
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * (step0 * 0.4)
                a = _project_to_simplex(a + jitter, S)

        # temperature decay
        temperature = max(temperature * temp_decay, min_temp)

        # restart if stagnation detected
        if no_improve >= max_no_improve:
            alpha = np.full(n, 0.3)
            a = np.random.dirichlet(alpha) * S
            momentum = np.zeros_like(a)
            no_improve = 0
            print(f"[{iteration:07d}] Stagnation restart (no improvement for {max_no_improve} iters).")

        iteration += 1

    # ------------------------------------------------------------------
    # Final verification (using the provided evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)   # ensure perfect feasibility
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()
