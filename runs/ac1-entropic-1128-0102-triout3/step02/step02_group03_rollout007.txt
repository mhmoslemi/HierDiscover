    # w length = len(conv) = 2*n-1
    # Normalize w to sum 1.
    w_norm = w / w.sum()

assistantfinal**Solution Overview**

The score returned by `evaluate_sequence` for a non‑negative vector `a` of length `n` is  

\[
\text{score}= \frac{2n\;\max (a*a)}{(\sum_i a_i)^2}.
\]

If we keep the ℓ₁‑norm fixed to  

\[
\sum_i a_i = S = \sqrt{2n},
\]

the denominator equals `2n` and the score reduces to the **maximum entry of the self‑convolution** `a*a`.  
Consequently the optimisation problem becomes:

```
minimise   max_k   (a * a)[k]
subject to a_i ≥ 0 ,   Σ a_i = √(2n) .
```

The algorithm below works on the ℓ₁‑simplex and repeatedly
* a **soft‑max (exponentiated‑gradient) step** – a smooth approximation of the
  non‑differentiable `max` that yields a dense sub‑gradient,
* a **random mass‑transfer move** that helps escape shallow plateaus,
* an occasional **full restart** when progress stalls.

All iterates are projected back onto the simplex using the Euclidean
projection of Duchi et al. (2008).  
The method is deliberately simple, fast (FFT‑based convolutions) and
does not require any external solvers.

The best vector seen before the time limit is returned.

```python
"""
Hybrid exponentiated‑gradient / random‑move search for the step‑function
that minimises ``evaluate_sequence``.  The ℓ₁‑norm is kept fixed to
sqrt(2·n), therefore the objective equals the maximum entry of the
self‑convolution.  Each iteration:

* computes the convolution with an FFT,
* builds a soft‑max weight vector over the convolution entries,
* obtains a (soft) sub‑gradient of the smoothed max,
* performs an exponentiated‑gradient update and projects onto the simplex,
* with probability p_random executes a random mass‑transfer,
* accepts uphill moves with a tiny Metropolis probability (annealed).

The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length nfft.
    Returns the full (2·len(a)‑1) array of real values.
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = target_sum}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_weighted(a: np.ndarray, w: np.ndarray, nfft: int) -> np.ndarray:
    """
    Gradient of  max_k (a*a)[k]  when the max is approximated by a
    weighted average with non‑negative weights w (same length as the
    linear convolution a*a).  The weight vector is first normalised to
    sum to one.  The returned gradient has length len(a).
    """
    w = np.asarray(w, dtype=np.float64)
    s = w.sum()
    if s == 0.0:
        return np.zeros_like(a)
    w = w / s                     # normalise to a probability vector
    a_rev = a[::-1]

    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad

# Main search routine
# ----------------------------------------------------------------------

    """

        a = _project_to_simplex(a0, S)

        # ----------------------------------------------------------

        cand_score = cand_conv.max()

        # ----------------------------------------------------------

    best_a = _project_to_simplex(best_a, S).tolist()
    final_score = evaluate_sequence(best_a)
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a
```
