def propose_candidate(seed=0, budget_s=950.0):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0
    n = 2500

    # generate initial candidates in list
    candidates = []

    # half-binomial (preset)
    candidates.append(half_binomial_sequence(n))

    # power law grid
    for offset in [0.5, 0.8, 1.0, 2.0]:
        for p in np.arange(0.45, 0.66, 0.03):
            candidates.append(power_law_sequence(n, p, offset))

        candidates.append(exp_sequence(n, lam))

        a = a / a.sum()
        candidates.append(a)

    best_score = np.inf
    best_a = None

    for a0 in candidates:
        if time.time() > deadline:
            break
        a = a0.copy()
        # perform smoothed gradient descent
        # parameters schedule
        beta = 1.0
        beta_max = 80.0
        beta_factor = 1.05
        lr = 0.2
        decay = 0.99
        # Adam vars
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0

        while time.time() < deadline:
            # evaluate smoothed loss
            conv = conv_fft(a)
            max_val = conv.max()
            # compute softmax
            shifted = conv - max_val
            exp_scaled = np.exp(beta * shifted)
            w = exp_scaled / exp_scaled.sum()
            # compute weighted correlation
            corr = weighted_correlation(a, w)  # returns size n
            grad = 2.0 * corr  # gradient of L (log-sum-exp) w.r.t a

            # Adam step

            m = 0.9 * m + (1 - 0.9) * grad_obj

            # update

            # add jitter occasionally
            if rng.random() < 0.01:
                a_new += (rng.random(n) - 0.5) * 1e-6
            # project on simplex

"""
Advanced search for a non‑negative step‑function minimizing
evaluate_sequence.

The algorithm works with three ideas:

1. **Diverse seeds** – half‑binomial, power‑law families, exponential
   tails and a few random Dirichlet vectors (and the optional global
   variable ``height_sequence_1`` if present).

2. **Smooth max optimisation** – the true objective
   `2·n·max(conv)` is approximated by a log‑sum‑exp
   `Lβ(a)= (1/β)·log Σ exp(β·conv_k)`.  Its gradient is a weighted
   correlation `2·corr(a,w)` with soft‑max weights `w`.  The weights
   become increasingly peaked as β grows, allowing the optimizer to
   first move in a smooth landscape and later focus on the true peak.

3. **Projected Adam** – after each gradient step the vector is projected
   onto the probability simplex (non‑negative, sum = 1).  Adam’s
   adaptive moments provide stable steps without hand‑tuned decay.

All computation uses FFT‑based convolutions so that each iteration is
~O(n log n).  The routine respects a 950 s internal budget (leaving a
few seconds safety margin) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all defined at module level (no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:                     # fallback (should never happen)
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def half_binomial_sequence(n: int) -> np.ndarray:
    """Classic half‑binomial series, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def exp_sequence(n: int, lam: float = 0.001, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ exp(-lam·(i+offset)), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * (i + offset))
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Self‑convolution x*x via real FFT (length 2·len(x)‑1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def weighted_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the correlation Σ_i a_i·w_{i+j} for all j=0…n‑1.
    Implemented with a single FFT pair (O(n log n)).
    """
    n = a.shape[0]
    len_w = w.shape[0]            # should be 2·n‑1
    size = 1 << ((n + len_w - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)          # reversed w
    prod = A * W_rev
    conv = np.fft.irfft(prod, n=size)             # length = size
    start = len_w - 1
    idx = start - np.arange(n)                    # descending slice
    return conv[idx]                              # length n


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for the best non‑negative sequence under the given time budget.
    Returns the sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # configuration
    n = 2500                         # length of the candidate
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # create a pool of starting vectors
    candidates = []

    # half‑binomial seed (classic baseline)
    candidates.append(half_binomial_sequence(n))

    # power‑law grid
    offsets = [0.5, 0.8, 1.0, 2.0]
    for offset in offsets:
        for p in np.arange(0.45, 0.66, 0.03):
            candidates.append(power_law_sequence(n, float(p), offset))

    # exponential tails
    for lam in [0.0005, 0.001, 0.002, 0.005]:
        candidates.append(exp_sequence(n, lam))

    # a few random Dirichlet points
    for _ in range(8):
        a = rng.gamma(1.0, size=n)
        candidates.append(a / a.sum())

    # optional global variable from the environment
    if 'height_sequence_1' in globals():
        hs = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if hs.shape[0] == n:
            candidates.append(hs / hs.sum())
        else:
            # resample/re‑normalise if length differs
            if hs.shape[0] > n:
                candidates.append(hs[:n] / hs[:n].sum())
            else:
                # pad with tiny mass
                pad = np.full(n - hs.shape[0], 1e-12, dtype=np.float64)
                cand = np.concatenate([hs, pad])
                candidates.append(cand / cand.sum())

    # ------------------------------------------------------------------
    # global best trackers
    best_score = np.inf
    best_a = None

    def real_score(vec: np.ndarray) -> float:
        """Exact score using the known closed form (sum(vec)=1)."""
        conv = conv_fft(vec)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # optimisation routine for a single seed
    def optimise_from(start_vec: np.ndarray):
        nonlocal best_score, best_a

        a = start_vec.copy()
        # ensure simplex feasibility
        a = project_onto_simplex(a)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8

        cur_score = real_score(a)

        # smoothing / annealing parameters
        beta = 2.0           # start of soft‑max temperature
        beta_max = 80.0
        beta_inc = 1.025

        lr = 0.25            # base learning rate (will be decayed)
        lr_decay = 0.9995

        iteration = 0
        max_iter = 40000    # hard cap per seed

        while time.time() < deadline and iteration < max_iter:
            # ---------------------------------------------------------
            # 1) convolution and soft‑max weights
            conv = conv_fft(a)
            max_conv = conv.max()

            # soft‑max (log‑sum‑exp) weights, numerically stable
            shifted = conv - max_conv
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()          # shape (2n‑1,)

            # ---------------------------------------------------------
            # 2) gradient of the smoothed objective
            #    grad_L = 2 * correlation(a, w)
            corr = weighted_correlation(a, w)      # length n
            grad_L = 2.0 * corr                     # d/d a of log‑sum‑exp

            # full objective = 2·n·L   →   grad = 2·n·grad_L
            grad = 2.0 * n * grad_L

            # ---------------------------------------------------------
            # 3) Adam update
            iteration += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** iteration)
            v_hat = v / (1 - beta2 ** iteration)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---------------------------------------------------------
            # 4) take a step, add occasional jitter, project back
            a_new = a - step
            if rng.random() < 0.01:          # tiny random perturbation
                a_new += (rng.random(n) - 0.5) * 1e-6
            a_new = project_onto_simplex(a_new)

            # ---------------------------------------------------------
            # 5) evaluate new point (exact objective)
            new_score = real_score(a_new)

            # accept if better (always) otherwise with a tiny metropolis chance
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
            else:
                if rng.random() < 0.001:
                    a = a_new
                    cur_score = new_score

            # update global best
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[global] new best score = {best_score:.6f}")

            # ---------------------------------------------------------
            # 6) schedule updates
            if beta < beta_max:
                beta = min(beta_max, beta * beta_inc)
            lr *= lr_decay

            # optional early stop when the step becomes negligible
            if lr < 1e-8:
                break

        # end of optimisation for this seed
        return

    # ------------------------------------------------------------------
    # run the optimisation from every seed (subject to time)
    for idx, start in enumerate(candidates):
        if time.time() >= deadline:
            break
        print(f"[seed {idx:02d}] starting optimisation from seed")
        optimise_from(start)

    # ------------------------------------------------------------------
    # final sanity check via the provided evaluate_sequence()
    if best_a is None:
        # extremely unlikely – fall back to half‑binomial
        best_a = half_binomial_sequence(n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] best reported score = {final_score:.6f}")
    return best_a.tolist()
