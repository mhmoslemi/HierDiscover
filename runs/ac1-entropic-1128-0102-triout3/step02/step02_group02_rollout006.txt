"""
Propose a candidate step‑function sequence.

Algorithm
---------
* Uses a soft‑max parametrisation a_i = exp(z_i)/Σexp(z) to guarantee
  non‑negativity and Σa_i = 1 (the scale of the objective is invariant).
* Optimises a smooth approximation of the objective  
      f_τ(a) = (1/τ)·log Σ_k exp(τ·(a*a)_k)            (τ → ∞ gives max)
  The gradient of f_τ is a weighted sum of the gradients of each
  convolution entry.  The weighted sum can be computed with a single FFT
  convolution, avoiding an O(n·|K|) loop.
* Starts with a low smoothing parameter τ (≈5) and linearly increases it
  to a large value (≈200) – this “anneals’’ the loss from a very smooth
  landscape to the true max‑convolution.
* Adam optimiser (log‑space) with a learning‑rate that decays with the
  iteration count.
* Multi‑start: several random initialisations for each problem size.
* After the gradient phase a cheap simulated‑annealing step performs
  tiny mass transfers between coordinates to escape possible local minima.
* The routine respects the 1000 s time budget, falls back to a uniform
  sequence if nothing is found, and finally returns the best list of
  heights.
"""

import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·n‑1) using FFT.
    a is a real 1‑D array.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A                     # |A|²  → convolution of a with itself
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv_full


def _conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary real vectors x and y
    (full length len(x)+len(y)-1) using FFT.
    """
    n, m = x.shape[0], y.shape[0]
    N = _next_pow2(n + m - 1)
    x_pad = np.zeros(N, dtype=np.float64)
    y_pad = np.zeros(N, dtype=np.float64)
    x_pad[:n] = x
    y_pad[:m] = y
    X = np.fft.rfft(x_pad)
    Y = np.fft.rfft(y_pad)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[: n + m - 1]
    return conv


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _adam_update(z: np.ndarray,
                 grad_z: np.ndarray,
                 m: np.ndarray,
                 v: np.ndarray,
                 t: int,
                 lr: float,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, tuple[np.ndarray, np.ndarray]]:
    """Classic Adam update; returns new (z, (m, v))."""
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, (m, v)


def _smooth_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     tau: float) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate
        fτ(a) = (1/τ)·log Σ_k exp(τ·(a*a)_k)
    w.r.t. the probability vector a.
    """
    # stable soft‑max weights over the convolution entries
    max_c = conv.max()
    w = np.exp(tau * (conv - max_c))          # shape (2n‑1,)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * cross‑correlation of w with a
    a_rev = a[::-1]
    conv_wr = _conv_fft_general(w, a_rev)    # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]   # slice of length n
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current best bound.  The function may run up to ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a little safety margin

    best_seq: List[float] | None = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Initialise from a previously stored construction, if any.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(height_sequence_1)     # type: ignore[name-defined]
            cand_score = evaluate_sequence(cand)
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[init] Imported previous best score {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # Parameter sweep over a few problem sizes.
    # --------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000]   # larger sizes give more freedom
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # Number of independent restarts for this size.
        restarts = 3 if n <= 4000 else 2

        # Hyper‑parameters that depend on n
        lr_base = 0.06 / np.sqrt(n)               # base learning rate
        t_ramp = 12000                            # iterations over which τ grows
        tau_start = 5.0
        tau_end = 200.0

        for restart in range(restarts):
            if time.time() >= deadline:
                break

            # ---- initialise logits -------------------------------------------------
            # Start from a Dirichlet‑type random vector (exponential draws)
            a0 = np.random.exponential(scale=1.0, size=n)
            a0 /= a0.sum()
            # Convert to logits (log‑probabilities) – add a tiny jitter
            z = np.log(a0 + 1e-12) + np.random.randn(n) * 0.02

            # ---- Adam state --------------------------------------------------------
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            t = 0

            # ---- optimisation loop -------------------------------------------------
            while time.time() < deadline:
                # 1) current distribution and its auto‑correlation
                a = _logits_to_a(z)                 # Σa = 1
                conv = _conv_fft(a)                  # length 2n‑1
                cur_max = conv.max()
                cur_score = 2.0 * n * cur_max        # identical to evaluate_sequence

                # 2) keep the global best
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.tolist()
                    print(f"[{n} r{restart}] New global best {best_score:.6f}")

                # 3) smoothing parameter τ (annealed)
                if t < t_ramp:
                    tau = tau_start + (tau_end - tau_start) * (t / t_ramp)
                else:
                    tau = tau_end

                # 4) gradient of the smooth‑max surrogate
                grad_a = _smooth_gradient(a, conv, tau)

                # 5) chain‑rule to obtain gradient w.r.t. logits
                dot = a.dot(grad_a)
                grad_z = a * (grad_a - dot)

                # 6) Adam update (learning rate decays very slowly)
                t += 1
                lr = lr_base / (1.0 + t * 1e-5)
                # optional gradient clipping to keep updates stable
                gnorm = np.linalg.norm(grad_z)
                if gnorm > 10.0:
                    grad_z = grad_z * (10.0 / gnorm)

                z, (m, v) = _adam_update(z, grad_z, m, v, t, lr)

                # 7) occasional jitter to avoid being trapped in plateaus
                if t % 3000 == 0:
                    z += np.random.randn(n) * 1e-3

                # 8) early‑stop the inner loop if progress stalls
                if t % 8000 == 0:
                    # very cheap check – if the gradient norm becomes tiny we stop
                    if np.linalg.norm(grad_z) < 5e-7:
                        break

    # ------------------------------------------------------------------
    # Fine‑tuning with a tiny simulated‑annealing on the best candidate.
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, dtype=np.float64)
        a /= a.sum()                     # ensure Σa = 1
        n = a.shape[0]

        # a few thousand cheap annealing steps
        anneal_steps = 50000
        for step in range(anneal_steps):
            if time.time() >= deadline:
                break

            # propose moving a small amount of mass from i → j
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # draw a candidate transfer amount (exponential tail)
            eps = np.random.exponential(scale=0.02) * a[i]
            eps = min(eps, a[i])          # keep non‑negative
            a_new = a.copy()
            a_new[i] -= eps
            a_new[j] += eps

            # evaluate the proposal
            conv_new = _conv_fft(a_new)
            score_new = 2.0 * n * conv_new.max()

            # temperature linearly decreasing from 1.0 to 0.0
            T = max(1e-8, 1.0 - step / anneal_steps)

            # Metropolis acceptance
            if score_new < best_score or np.random.rand() < np.exp(-(score_new - best_score) / T):
                a = a_new
                if score_new < best_score:
                    best_score = score_new
                    best_seq = a.tolist()
                    print(f"[anneal] New best {best_score:.6f} at step {step}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be taken).
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"Final best score (official): {final_score:.6f} "
              f"(length {len(best_seq)})")
    except Exception:
        print("Warning: official evaluation of the final sequence failed.")

    return best_seq
