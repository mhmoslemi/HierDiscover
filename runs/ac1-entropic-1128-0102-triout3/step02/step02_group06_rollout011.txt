"""
Mirror‑descent with a continuously‑smoothed max‑convolution objective.
Improvements over the reference implementation:

*   Uses a *soft‑max* gradient for the whole run (no abrupt switch to a
    sub‑gradient of a single active index).  This yields a smoother
    descent direction and avoids the plateaus caused by the exact‑max
    phase.
*   When the temperature is very low we still keep a few top indices
    (within a relative tolerance) and average their exact gradients –
    this is a cheap approximation of a bundle‑method update.
*   More aggressive restart (5 000 non‑improving steps) together with a
    multiplicative perturbation of the current best solution.
*   Optional “coarse‑to‑fine’’ schedule: if a global variable
    `height_sequence_1` is present we initialise from it, otherwise a
    Dirichlet draw is used.
*   All updates are performed on the logarithmic variables (mirror
    descent on the simplex).  After each step the full sequence is
    renormalised to the required ℓ₁‑norm `√(2·n)` – the same
    normalisation used by `evaluate_sequence`, so the objective value
    equals the maximum entry of the self‑convolution.
*   Extensive timing checks keep the search within the allotted budget.

The function `propose_candidate` returns the best sequence found before
the time limit expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries (real part).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray,
                  tau: float, nfft: int) -> np.ndarray:
    """
    Gradient of   τ·log Σ exp(conv/τ)   w.r.t. a.
    Uses a numerically stable soft‑max.
    """
    # stable soft‑max weights
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1: n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at a single index k_max.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


def _grad_max_multi(a: np.ndarray, conv: np.ndarray,
                    idxs: np.ndarray) -> np.ndarray:
    """
    Gradient averaged over a set of active indices.
    """
    grad = np.zeros_like(a)
    for k in idxs:
        grad += _grad_max(a, conv, int(k))
    if idxs.size > 0:
        grad /= float(idxs.size)
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm performs mirror descent on the
    simplex with a soft‑max smoothing of the max‑convolution and a
    multi‑index bundle gradient near the end of the schedule.
    The best sequence encountered before the time limit is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose a problem size and an initial point
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_full.size
        # make sure the start is non‑negative and has a reasonable L1 norm
        init_full = np.maximum(init_full, 0.0)
        total = init_full.sum()
        if total == 0.0:
            init_full = np.random.dirichlet(np.full(n, 0.5))
    else:
        n = 2401                      # default length (same as reference)
        init_full = np.random.dirichlet(np.full(n, 0.5))

    # target ℓ₁‑norm (the normalisation used by evaluate_sequence)
    S = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Initialise logarithmic variables (mirror descent on the simplex)
    # --------------------------------------------------------------
    eps = 1e-12
    log_theta = np.log(init_full + eps)   # size n

    a = a * (S / a.sum())
    conv = _convolve_self_fft(a, _next_pow2(2 * n - 1))
    best_score = conv.max()
    best_a = a.copy()
    print(f"Initial score (max conv) = {best_score:.6f}")

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta = 0.20                     # base learning rate

    v_adam = np.zeros_like(log_theta)

    tau_min = 0.01
    # estimate a generous number of iterations; decay is exponential
    total_iters_est = 300_000
    tau_decay = (tau_min / tau) ** (1.0 / total_iters_est)

    # stagnation handling
    no_improve = 0
    restart_limit = 5_000
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # Re‑build the full sequence and enforce the ℓ₁‑norm
        # ----------------------------------------------------------
        a = np.exp(log_theta)
        sum_a = a.sum()
        if sum_a == 0.0:
            # Numerical degeneration – re‑initialise from scratch
            a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64) * S
            log_theta = np.log(a + eps)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            continue
        a = a * (S / sum_a)

        # ----------------------------------------------------------
        # Evaluate current objective (max entry of self‑convolution)
        # ----------------------------------------------------------
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = cur_max                         # equals evaluate_sequence(...)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{iteration:08d}] New best: {best_score:.6f}")
        else:
            no_improve += 1

        # early exit if target bound is reached
        if best_score < 1.503:
            print(f"Target achieved at iteration {iteration}")
            break

        # ----------------------------------------------------------
        # Gradient of the (smoothed) max‑convolution
        # ----------------------------------------------------------
        if tau > tau_min:
            # soft‑max gradient (smooth approximation)
            grad_a = _grad_softmax(a, conv, tau, nfft)
        else:
            # near zero temperature: use a small bundle of top indices
            cur_max = conv.max()
            # indices whose value is within 0.01 % of the maximum (or absolute 1e‑6)
            tol = max(1e-6, cur_max * 1e-4)
            active = np.where(conv >= cur_max - tol)[0]
            if active.size > 1:
                grad_a = _grad_max_multi(a, conv, active)
            else:
                grad_a = _grad_max(a, conv, int(active[0]))

        # ----------------------------------------------------------
        # Gradient w.r.t. log‑variables (chain rule for mirror descent)
        # ----------------------------------------------------------
        grad_log = grad_a * a

        # ----------------------------------------------------------
        # Adam update on log‑variables (descent)
        # ----------------------------------------------------------
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_log
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m_adam / (1.0 - beta1 ** iteration)
        v_hat = v_adam / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step

        # occasional stochastic jitter to help escape plateaus
        if np.random.rand() < 0.005:
            jitter = np.random.randn(n) * (eta * 0.05)
            log_theta = log_theta + jitter

        # anneal temperature
        tau = max(tau * tau_decay, tau_min)

        # restart if stagnation persists
        if no_improve >= restart_limit:
            # perturb the current best sequence multiplicatively
            a_best = best_a.copy()
            # work in log‑space to keep positivity
            log_best = np.log(a_best + eps)
            perturb = np.random.randn(n) * (eta * 0.2)
            log_theta = log_best + perturb
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            no_improve = 0
            print(f"[{iteration:08d}] Restarted after {restart_limit} non‑improving steps.")

    # --------------------------------------------------------------
    # Final sanity check and return the best sequence found.
    # --------------------------------------------------------------
    final_seq = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
