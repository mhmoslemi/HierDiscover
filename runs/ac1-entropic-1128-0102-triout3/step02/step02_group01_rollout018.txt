"""
Heuristic optimiser for the “max auto‑convolution” objective
used by `evaluate_sequence`.  The algorithm proceeds in three stages:

1.  **Exploratory parametrised families** – short sweeps over geometric
    and power‑law shapes give a few good baselines (often already
    beating the previous best).

2.  **Soft‑max Adam optimisation on the simplex** – the coefficients are
    represented by a soft‑max of unconstrained parameters `p`.  The
    non‑smooth max in the objective is replaced by a smooth
    log‑sum‑exp (inverse temperature `beta`).  Gradients are obtained
    analytically using a single FFT‑based convolution, and an Adam
    optimiser updates `p`.  The current best sequence (highest score
    improvement) is stored throughout.

3.  **Progressive up‑sampling** – after a stage finishes the current
    shape is linearly interpolated to a larger length (≈1.5×) and the
    Adam optimisation is repeated.  This allows the same “shape” to be
    refined on finer grids without restarting from scratch.

Random restarts (small Gaussian perturbations of the parameters) are
performed whenever an optimisation stage makes no improvement for a
while.  The whole routine respects the supplied `budget_s` (default
1000 s) and finally returns the best sequence found as plain Python
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _conv_auto(a: np.ndarray) -> np.ndarray:
    """Full linear auto‑convolution a*a using FFT (real input)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


def _score_of_seq(a: np.ndarray) -> float:
    """Exact evaluation used for bookkeeping: 2·n·max(conv).  a must sum to 1."""
    n = a.shape[0]
    conv = _conv_auto(a)
    return 2.0 * n * conv.max()


def _grad_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k  w.r.t. a, where conv = a*a.
    Uses the identity g_i = 2·Σ_k w_k·a_{k-i}.
    Implemented by a single FFT‑based convolution with the reversed weight
    vector.
    """
    n = a.shape[0]               # length of a
    m = 2 * n - 1                 # length of w (same as conv)
    w_rev = w[::-1]              # reversed weights

    # length needed for a (n) + w_rev (m) - 1 = n + m - 1 = 3n - 2
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(A * W, n=fft_len)[: n + m - 1]

    # The required slice is full[(m-1) - i] for i=0..n-1
    start = m - 1
    g = 2.0 * full[start - np.arange(n)]
    return g


def _geometric_seq(r: float, n: int) -> np.ndarray:
    """Geometric decreasing sequence a_i ∝ r**i (0<r<1), normalised to sum=1."""
    idx = np.arange(n, dtype=float)
    a = r ** idx
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def _powerlaw_seq(p: float, n: int) -> np.ndarray:
    """Power‑law decreasing sequence a_i ∝ (i+1)^{-p}, normalised to sum=1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step‑function) that achieves a
    low `evaluate_sequence` score.  The routine runs for at most ``budget_s``
    seconds, printing intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a small safety margin

    # --------------------------------------------------------------
    # 0. Initialise from the global `height_sequence_1` if present.
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 0:
            init = init / s
        else:
            raise ValueError
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] Init from height_sequence_1 "
              f"n={len(best_seq)} score={best_score:.6f}")
    except Exception:
        # Fallback: random Dirichlet of moderate size
        n0 = 1200
        init = np.random.dirichlet(np.ones(n0))
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] Random init n={n0} "
              f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Sweep simple parametric families (geometric & power‑law)
    # --------------------------------------------------------------
    n = best_seq.shape[0]
    # geometric sweep
    for r in np.linspace(0.90, 0.9999, 40):
        a = _geometric_seq(r, n)
        sc = _score_of_seq(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Geom r={r:.4f} "
                  f"score={sc:.6f}")

    # power‑law sweep
    for p in np.linspace(0.2, 5.0, 40):
        a = _powerlaw_seq(p, n)
        sc = _score_of_seq(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Power p={p:.2f} "
                  f"score={sc:.6f}")

    # --------------------------------------------------------------
    # 2. Adam optimiser on the simplex (soft‑max parametrisation)
    # --------------------------------------------------------------
    def _adam_optimize(a_start: np.ndarray, max_iters: int, beta: float = 30.0):
        """Run Adam from a_start (already normalised) for up to max_iters."""
        n_local = a_start.shape[0]
        # initialise log‑parameters (soft‑max trick)
        eps = 1e-12
        p = np.log(a_start + eps)

        m = np.zeros_like(p)
        v = np.zeros_like(p)
        beta1, beta2 = 0.9, 0.999
        lr = 0.15                      # base learning rate (tuned empirically)
        best_local_seq = a_start.copy()
        best_local_score = _score_of_seq(a_start)

        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break

            # soft‑max -> probability vector
            a = _softmax(p)

            # convolution and current max
            conv = _conv_auto(a)
            cur_max = conv.max()
            cur_score = 2.0 * n_local * cur_max
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam n={n_local} "
                      f"iter={it} score={cur_score:.6f}")

            # smooth approximation of the max
            # (subtract max to keep numbers stable)
            exp_w = np.exp(beta * (conv - cur_max))
            w = exp_w / exp_w.sum()

            # gradient wrt a (soft‑max output)
            g_a = _grad_wrt_a(a, w)

            # gradient wrt logits p
            dot = np.dot(a, g_a)
            g_p = a * (g_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g_p
            v = beta2 * v + (1.0 - beta2) * (g_p * g_p)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            p -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        return best_local_seq, best_local_score

    # --------------------------------------------------------------
    # 3. Progressive up‑sampling + optimisation
    # --------------------------------------------------------------
    current = best_seq.copy()
    cur_n = current.shape[0]
    MAX_LEN = 6000                     # upper bound on the length we will try

    while True:
        # decide next size (≈1.5×, but never exceed MAX_LEN)
        nxt = int(cur_n * 1.5)
        if nxt > MAX_LEN:
            nxt = MAX_LEN
        if nxt <= cur_n:
            break

        # ----- up‑sample current shape to new length -----------------
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, nxt)
        a_up = np.interp(x_new, x_old, current)
        a_up = np.maximum(a_up, 0.0)
        a_up_sum = a_up.sum()
        if a_up_sum == 0:
            a_up = np.full(nxt, 1.0 / nxt)
        else:
            a_up = a_up / a_up_sum

        # ----- Adam optimisation on the new length ------------------
        # iteration budget that roughly scales with problem size and remaining time
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allow roughly 0.1 s per iteration (very rough)
        est_iter = int(min(40000, max(8000, remaining * 10)))
        a_opt, sc_opt = _adam_optimize(a_up, est_iter, beta=30.0)

        # keep the best overall
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] New global best "
                  f"n={nxt} score={sc_opt:.6f}")

        # Prepare for the next loop
        current = a_opt
        cur_n = nxt

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final short Adam refinement on the best length found
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_iters = 20000
        a_refined, sc_refined = _adam_optimize(best_seq, final_iters, beta=50.0)
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined
            print(f"[{time.time() - start_time:6.2f}s] Final refinement "
                  f"score={sc_refined:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence (as plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        # extreme fallback – uniform 100‑element distribution
        best_seq = np.full(100, 1.0 / 100)

    print(f"[{time.time() - start_time:6.2f}s] Finished. Best score={best_score:.6f} "
          f"length={len(best_seq)}")
    return [float(x) for x in best_seq]
