"""
Propose a step‑function that (hopefully) beats the previous best score
(~1.514).  

The search consists of three stages  

1. **Random shape sampling** – generate many candidate probability
   vectors using beta‑shaped densities, Dirichlet draws and a small
   Gaussian blur.  Keep the best one.

2. **Mirror‑descent refinement** – starting from the best vector,
   repeatedly move mass away from the convolution index that attains the
   current maximum.  The update is
        a_i ← a_i · exp(‑η·g_i)  (g is the sub‑gradient of the max entry)
   and a renormalisation onto the simplex.  An adaptive step size
   η_t = η₀ / t**decay is used.  The best observable vector is stored.

3. **Light simulated‑annealing** – a final local‑search phase that
   swaps a small amount of mass between two positions while accepting
   uphill moves with a decreasing temperature.

All helper functions are defined at the top level, no closures are used.
The algorithm respects the 1000 s budget by splitting the remaining time
among the stages and between several lengths (the “multi‑scale” idea
from the reference code).  The returned list contains non‑negative
numbers that sum to one (the evaluator rescales anyway, but keeping the
simplex helps the optimisation).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution a * a (linear, not circular) via FFT – fast O(n log n)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple up‑sampling (linear interpolation) from old to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Score (the objective) for a vector that already lives on the simplex
# ----------------------------------------------------------------------
def _score_on_simplex(a: np.ndarray) -> float:
    n = a.shape[0]
    max_conv = _conv_self(a).max()
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Sub‑gradient of the *maximum* convolution entry (K = 1)
# ----------------------------------------------------------------------
def _max_conv_subgradient(a: np.ndarray) -> np.ndarray:
    """Gradient of the entry attaining the maximum of a*a."""
    n = a.shape[0]
    conv = _conv_self(a)
    k = int(conv.argmax())               # index of the maximal entry
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k - i
    g = np.zeros_like(a)
    g[i] = 2.0 * a[j]                     # each pair contributes twice
    return g


# ----------------------------------------------------------------------
# Mirror‑descent on the simplex (exponentiated‑gradient style)
# ----------------------------------------------------------------------
def _run_mirror_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    eta0: float = 0.6,
    decay: float = 0.55,
    max_iter: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Perform mirror‑descent (KL‑prox) on the simplex to minimise the max
    self‑convolution.  Returns the best vector seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_on_simplex(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ---------- evaluate ----------
        cur_score = _score_on_simplex(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] mirror improve: n={n}, "
                f"iter={it}, score={cur_score:.6f}"
            )

        # ---------- sub‑gradient ----------
        g = _max_conv_subgradient(a)

        # ---------- step size ----------
        eta = eta0 / (it ** decay)

        # ---------- exponentiated update ----------
        # scale gradient to avoid overflow in exp
        if g.max() > 0:
            g_scaled = g / (g.max() + 1e-12)
        else:
            g_scaled = g
        a = a * np.exp(-eta * g_scaled)

        # ---------- project back onto simplex ----------
        s = a.sum()
        if s > 0:
            a /= s
        else:                              # numerical safety – re‑initialise
            a = np.random.rand(n)
            a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (same spirit as the reference)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 20000,
) -> tuple[np.ndarray, float]:
    """Neighbourhood moves that transfer a little mass between two bins."""
    a = a0.copy()
    score = _score_on_simplex(a)
    best_seq = cur_best_seq
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12          # up to 12 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _score_on_simplex(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, "
                f"iter={it}, score={score:.6f}"
            )

    return best_seq, best_score

        if remaining <= 0:
            break
        slice_deadline = time.time() + max(10.0, 0.1 * remaining)

        # ----- mirror descent -----
        a_opt, sc_opt = _run_mirror_descent(
            a0,
            n,
            slice_deadline,
            start_time,
            eta0=0.65,
            decay=0.58,
            max_iter=60000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] mirror best improve: "
                f"n={n}, score={sc_opt:.6f}"
            )

        # ----- brief annealing pass -----
        if time.time() < deadline:
            anneal_deadline = time.time() + max(5.0, 0.05 * (deadline - time.time()))
            a_ref, sc_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=3e-2,
                end_temp=1e-5,
                steps=18000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] annealing best improve: "
                    f"n={n}, score={sc_ref:.6f}"
                )

        # store the current global best for the next upscale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 800
        best_seq = np.full(n, 1.0 / n)

    # Convert to a plain Python list (the harness expects a list of floats)
    return [float(v) for v in best_seq]
