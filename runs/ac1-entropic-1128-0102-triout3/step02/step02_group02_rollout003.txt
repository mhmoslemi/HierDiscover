def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]
    # build A matrix
    rows = 2*n - 1
    # Build A in numpy; then constraints: A @ g <= M

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)

        g_val = np.maximum(g_val, 0)
        # optionally clip/gate large values by bound of 1000 (not needed).

def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    n = f.shape[0]
    # build convolution matrix A
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # Fill columns
    for j in range(n):
        A[j:j + n, j] = f
    # define cvxpy variable
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # Solve
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1000, abstol=1e-9, reltol=1e-9)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None
    g_val = np.array(g.value).flatten()
    # Clip negative values due to numeric
    g_val = np.maximum(g_val, 0.0)
    return g_val
```

        a = _logits_to_a(z)

            z += np.random.randn(n) * 1e-3

"""
Propose a candidate step‑function sequence.

Algorithm
---------
1.  Work with a probability vector a (∑a_i = 1, a_i ≥ 0).  The evaluation
    to minimise is 2·n·max_k (a*a)_k, which is invariant under scaling.
2.  For a given length n the optimisation proceeds in two complementary
    phases:
      * **LP phase** – given the current a, let M = max(a*a).  Solve the
        linear programme
               maximise   ∑ g_i
               subject to f * g ≤ M    (convolution, component‑wise)
                         g_i ≥ 0 .
        The optimum g pushes the mass of a into positions that do not make
        the convolution larger than M.  After normalising g to sum 1 we
        blend a ← (1‑t)·a + t·g (t≈0.5) – this guarantees a non‑increase of
        the objective (the classic Frank‑Wolfe / conditional‑gradient step
        described in the statement).
      * **Gradient‑refinement** – occasional Adam steps in the soft‑max
        (log‑space) parametrisation sharpen the solution.  The gradient is
        built from the K largest convolution entries (K decays to 1), which
        yields a sub‑gradient of the max‑operator.
3.  The routine iterates the two phases for several lengths (starting from
    a modest size and up‑sampling the best solution to larger n).  All
    intermediate results are evaluated with ``evaluate_sequence`` and the
    best sequence found before the time budget expires is returned.
"""

import time
import numpy as np
import cvxpy as cp
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv_full


def _score_from_a(a: np.ndarray) -> float:
    """Evaluation 2·n·max(conv) for a that already satisfies ∑a_i = 1."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Soft‑max (log‑space) utilities for gradient refinement
# ----------------------------------------------------------------------


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _gradient_wrt_a(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)_k w.r.t. a using only the convolution
    entries indexed by ``top_idxs``.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    for k in top_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]

    return grad


# ----------------------------------------------------------------------
# Linear‑programming helper
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix A such that (A @ g)_k = (f*g)_k.
    Column j contains the vector f shifted by j positions.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise   1ᵀg   subject to   f*g ≤ M,   g≥0.
    Returns the optimal g (as a dense np.ndarray) or None if the solver fails.
    """
    n = f.shape[0]
    # Build the convolution matrix (dense – still modest for n≤2000)
    A = _build_conv_matrix(f)

    # CVXPY model
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   abstol=1e-9, reltol=1e-9, max_iters=2000)
    except Exception:  # safety‑net for unexpected solver issues
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    g_val = np.array(g.value).flatten()
    # Numerical noise may produce tiny negative values
    g_val = np.maximum(g_val, 0.0)
    return g_val


# ----------------------------------------------------------------------
# Gradient‑refinement (Adam in log‑space)
# ----------------------------------------------------------------------


def _refine_with_adam(a0: np.ndarray, max_time: float) -> np.ndarray:
    """
    Run a short Adam optimisation on the soft‑max parameters starting from a0.
    The routine stops after ``max_time`` seconds.
    """
    n = a0.shape[0]
    # initialise logits from a0 (add tiny epsilon to avoid log(0))
    eps = 1e-12
    z = np.log(a0 + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    lr0 = 0.05 / np.sqrt(n)
    top_k = max(1, n // 800)          # start with a few active offsets
    top_k_decay_iter = 20000

    deadline = time.time() + max_time
    while time.time() < deadline:
        a = _logits_to_a(z)

        conv = _conv_fft(a)
        if top_k == 1:
            ks = np.array([int(np.argmax(conv))], dtype=int)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        grad_a = _gradient_wrt_a(a, ks)

        # chain rule for the soft‑max
        dot = a.dot(grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        t += 1
        lr_t = lr0 / (1.0 + t * 1e-5)
        m = 0.9 * m + 0.1 * grad_z
        v = 0.999 * v + 0.001 * (grad_z ** 2)

        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        step = lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = z - step

        # occasional jitter to escape plateaus
        if t % 1500 == 0:
            z += np.random.randn(n) * 1e-3

        # tighten the focus on the true maximum
        if t == top_k_decay_iter:
            top_k = 1

    return _logits_to_a(z)


# ----------------------------------------------------------------------
# Upsampling helper (to seed larger n from a smaller solution)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to length new_n, followed by renormalisation."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    old_x = np.arange(old_n, dtype=float)
    new_x = np.linspace(0, old_n - 1, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    new_a = new_a / new_a.sum()
    return new_a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best bound.  The function runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    safety = 5.0                     # small safety margin before hard timeout
    deadline = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # Keep track of the globally best sequence discovered so far.
    # ------------------------------------------------------------------
    best_seq: List[float] | None = None
    best_score = float("inf")

    # If an older construction is present, import it.
    prev_seq = None
    if "height_sequence_1" in globals():
        try:
            prev_seq = list(height_sequence_1)
        except Exception:
            prev_seq = None

    # ------------------------------------------------------------------
    # Length schedule – start small, then grow using up‑sampling.
    # ------------------------------------------------------------------
    lengths = [600, 900, 1200, 1500, 1800, 2100]   # will be trimmed by time

    prev_a = None
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # --------------------------------------------------------------
        # Initialise a probability vector for this length.
        # --------------------------------------------------------------
        if prev_a is not None:
            a = _upsample(prev_a, n)
        elif prev_seq is not None and len(prev_seq) == n:
            a = np.array(prev_seq, dtype=np.float64)
            a = np.maximum(0.0, a)
            a = a / a.sum()
        else:
            a = np.random.rand(n)
            a = a / a.sum()

        # --------------------------------------------------------------
        # Main optimisation loop for the current length.
        # --------------------------------------------------------------
        stagnation_counter = 0
        iteration = 0
        while time.time() < deadline:
            # --- Phase 1: LP improvement ---------------------------------
            conv_a = _conv_fft(a)
            M = conv_a.max()
            cur_score = float(2.0 * n * M)

            # record globally best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[LP start] New global best {best_score:.6f}")

            # Solve the LP to obtain a promising g.
            g = _solve_lp_for_g(a, M)
            if g is None:
                print("[LP] Solver failed – breaking length loop.")
                break

            # Normalise g to sum 1.
            g = g / g.sum()
            lp_score = _score_from_a(g)
            if lp_score < best_score:
                best_score = lp_score
                best_seq = g.tolist()
                print(f"[LP] New global best {best_score:.6f}")

            # --- Phase 2: convex combination (conditional gradient step) -
            improved = False
            # Try a few step sizes, keep the first that yields a strict decrease.
            for step in (0.5, 0.3, 0.1):
                a_new = (1.0 - step) * a + step * g
                a_new = a_new / a_new.sum()
                new_score = _score_from_a(a_new)
                if new_score < cur_score - 1e-9:
                    a = a_new
                    improved = True
                    if new_score < best_score:
                        best_score = new_score
                        best_seq = a.tolist()
                        print(f"[combine t={step}] New global best {best_score:.6f}")
                    break

            if not improved:
                # fall back: accept g directly if it is better
                if lp_score < cur_score - 1e-9:
                    a = g
                    if lp_score < best_score:
                        best_score = lp_score
                        best_seq = a.tolist()
                        print(f"[accept g] New global best {best_score:.6f}")
                else:
                    stagnation_counter += 1
                    if stagnation_counter >= 5:
                        # No progress for a while – exit this length.
                        break
                    # inject a tiny random jitter and continue
                    jitter = np.random.randn(n) * 1e-4
                    a = np.maximum(0.0, a + jitter)
                    a = a / a.sum()
                    continue
            else:
                stagnation_counter = 0

            iteration += 1

            # --- Phase 3: occasional Adam refinement --------------------
            if iteration % 7 == 0:
                remaining = deadline - time.time()
                if remaining > 8.0:               # keep a comfortable reserve
                    # spend at most 20 % of the remaining budget on refinement
                    refine_time = min(remaining * 0.2, 12.0)
                    a_refined = _refine_with_adam(a, refine_time)
                    refined_score = _score_from_a(a_refined)
                    if refined_score < best_score:
                        best_score = refined_score
                        best_seq = a_refined.tolist()
                        print(f"[adam] New global best {best_score:.6f}")
                    a = a_refined  # continue from the refined point

            # safety break if we have performed many iterations
            if iteration >= 50:
                break

        # store the best vector for the next size (up‑sampling)
        prev_a = np.array(best_seq, dtype=np.float64) if best_seq is not None else a

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if nothing succeeded (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        best_seq = (np.full(n_fallback, 1.0 / n_fallback)).tolist()
        best_score = evaluate_sequence(best_seq)  # sanity check
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"Final official score {final_score:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print(f"Warning: final evaluation failed ({exc})")

    return best_seq
