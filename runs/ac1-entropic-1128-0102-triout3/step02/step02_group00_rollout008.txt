"""
soft‑max gradient descent + symmetric projection + annealing

* Work on the probability simplex (∑a_i = 1).  The evaluation metric is
  scale invariant, so fixing the sum to 1 makes the objective 2·n·max(conv).
* For a given length n we minimise a smooth approximation of the max:
      f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k) .
  Its gradient is a weighted sum of the self‑convolution gradients.
* The gradient is obtained efficiently by one extra FFT (a_rev * w).
* We run several stages with an increasing β (0.5 → … → 200) and a
  diminishing step size with Adam‑style momentum.
* After each update we enforce symmetry a_i = a_{n‑1-i} and project onto
  the simplex.
* A cheap simulated‑annealing refinement (small mass transfers) is run
  at the end of each length.
* Lengths are increased geometrically; the best sequence is up‑sampled
  for the next stage.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution a * a (non‑circular) using FFT
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv_full


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) approximation of max(conv)
# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a  ( (1/β) log Σ_k exp(β·conv_k) )
    where conv = a * a (already supplied).
    """
    n = a.shape[0]
    max_val = conv.max()
    # weights = softmax(β·conv)
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (a_rev * w)_{slice}
    a_rev = a[::-1]
    conv_len = len(a_rev) + len(w) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len.bit_length())
    Af = np.fft.rfft(a_rev, n=fft_len)
    Wf = np.fft.rfft(w, n=fft_len)
    rev = np.fft.irfft(Af * Wf, n=fft_len)[:conv_len]
    start = n - 1
    grad = 2.0 * rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves ordering, then renormalises)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_softmax_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    betas,
    max_iter_per_beta: int = 2000,
    step0: float = 0.8,
):
    """
    Perform Adam‑style gradient descent on the smooth max objective.
    Returns the best sequence seen and its (non‑scaled) score 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = float("inf")
    t = 0

    for beta in betas:
        for _ in range(max_iter_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            conv = _conv_self_fft(a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] improve: n={n}, "
                    f"β={beta:.1f}, it={t}, score={cur_score:.6f}"
                )

            # gradient of the smooth max
            grad = _grad_softmax(a, conv, beta)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = step0 / math.sqrt(t)
            a = a - step * m_hat / (np.sqrt(v_hat) + eps)

            # enforce symmetry and simplex
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)

            if time.time() >= deadline:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 2e-2,
    end_temp: float = 1e-6,
    steps: int = 30000,
):
    a = a0.copy()
    conv = _conv_self_fft(a)
    cur_score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] anneal init improve: n={n}, score={cur_score:.6f}"
        )

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # move ≤10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop = _conv_self_fft(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - cur_score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, it={it}, score={cur_score:.6f}"
            )
    return best_seq, best_score
