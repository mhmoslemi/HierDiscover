    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 3000  # might increase later if time left

    # generate initial vectors via many seeds
    best_a = None
    best_val = np.inf
    for p in [0.2,0.3,0.35,0.4,0.45,0.5,0.55,0.6,0.65]:
        for offset in [0.0,0.2,0.5,1.0]:
            a = build_initial_sequence(n, p, offset)
            v = fast_score(a)
            if v < best_val:
                best_val, best_a = v, a.copy()
    # also try uniform distribution
    a = np.full(n, 1.0 / n)
    v = fast_score(a)
    if v < best_val:
        best_val, best_a = v, a.copy()

    # also include previous known best if present
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                seq = np.zeros(n, dtype=np.float64)
                seq[:prev.size] = prev
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0:
                seq /= seq.sum()
                v = fast_score(seq)
                if v < best_val:
                    best_val, best_a = v, seq.copy()
        except Exception:
            pass

    a = best_a.copy()
    best_seq = a.copy()
    best_val = fast_score(a)

    # parameters
    eta0 = 0.5
    eta_decay = 5e-6
    lambda0 = 5.0
    lambda_inc = 1e-5   # increase per iteration
    lambda_max = 200.0
    random_move_prob = 0.02
    random_move_frac = 0.1

    size = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        # compute self-convolution
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]

        # compute smoothed weights
        lam = min(lambda_max, lambda0 + lambda_inc * it)
        max_conv = conv.max()
        # subtract max to avoid overflow
        w = np.exp(lam * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:  # numerical, fallback to uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via cross-correlation
        fw = np.fft.rfft(w, size)
        # correlation = ifft(conj(FA) * FW)
        corr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
        grad = 2.0 * corr

        # exponentiated gradient step
        eta = eta0 / (1.0 + it * eta_decay)
        # ensure gradient not too large for numeric stability
        # we can clip grad values
        grad_clipped = np.clip(grad, -1e3, 1e3)
        a = a * np.exp(-eta * grad_clipped)
        # renorm to sum 1
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / sum_a

        # occasional random mass transfer for exploration
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = cand / cand.sum()
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand.copy()
                    a = cand
                else:

                    if np.random.rand() < 0.001:

        # update best periodically

            if cur_val < best_val:

                best_seq = a.copy()
                print(f"[iter {it}] new best {best_val:.6f}")
            else:
                print(f"[iter {it}] cur {cur_val:.6f} best {best_val:.6f}")

        it += 1

    print(f"[final] best value {best_val:.6f}")
    return best_seq.tolist()

"""
propose_candidate
-----------------

Search routine that tries to minimise the value returned by
``evaluate_sequence``.  The optimisation works on the simplex
(∑ a_i = 1, a_i ≥ 0) because the evaluation function is
scale‑invariant.  The algorithm is a *mirror‑descent* (exponentiated
gradient) method that uses a *soft‑max* approximation of the max‑norm
of the self‑convolution:

    w_k ∝ exp( λ·(a*a)_k )          ( λ grows during the run)

The gradient of the smoothed objective is a weighted cross‑correlation
between the current vector a and the weight vector w.
A multiplicative update a_i ← a_i·exp(−η·grad_i) automatically keeps
the vector non‑negative; after every step we renormalise to keep the
ℓ¹‑norm equal to one (simplex projection).  Random “mass‑transfer”
moves are injected from time to time to explore new basins.

The routine keeps the best sequence seen (according to the cheap
``fast_score``) and returns it as a plain Python list when the time
budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the provided `evaluate_sequence` but
# without the security checks).  Used internally for speed.
def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a probability vector (∑a=1)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Power‑law family a_i ∝ (i+1+offset)^{-p}, normalised to ℓ¹‑norm 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0.0:
        a.fill(1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Return a probability vector w_k ∝ exp(lam·conv_k)."""
    # subtract the max to avoid overflow
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:                      # numerical fallback
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def grad_cross_correlation(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Compute  ∇_a  ( Σ_k w_k·(a*a)_k ) = 2·corr(a,w)
    where corr(a,w)[i] = Σ_j a_j·w_{i+j}.  Implemented with FFT.
    """
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    # cross‑correlation via ifft(conj(Fa)·Fw)
    corr = np.fft.irfft(np.conj(fa) * fw, size)[: a.size]
    return 2.0 * corr


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    evaluation value.  The method runs until the time budget expires
    and returns the best sequence encountered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # 1. Length of the sequence – 3000 proved a good compromise.
    n = 3000

    # --------------------------------------------------------------
    # 2. Scan a few analytically motivated families for a good start.
    best_seq = None
    best_val = np.inf
    exponents = [0.20, 0.30, 0.40, 0.45, 0.50, 0.55, 0.60, 0.65]
    offsets   = [0.0, 0.2, 0.5, 1.0, 2.0]

    for p in exponents:
        for off in offsets:
            a = build_initial_sequence(n, p, off)
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
    # uniform also serves as a baseline
    a_uni = np.full(n, 1.0 / n)
    val_uni = fast_score(a_uni)
    if val_uni < best_val:
        best_val, best_seq = val_uni, a_uni.copy()
    print(f"[init] best value after scanning families: {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. If a previous construction exists, try to initialise from it.
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                cand = prev[:n].copy()
            else:
                cand = np.zeros(n, dtype=np.float64)
                cand[:prev.size] = prev
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0.0:
                cand /= s
                val = fast_score(cand)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                print(f"[init] height_sequence_1 gave value {val:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 4. Main optimisation loop (exponentiated‑gradient + soft‑max).
    a = best_seq.copy()
    best_seq = a.copy()
    best_val = fast_score(a)

    # hyper‑parameters
    eta0        = 0.70          # initial step size for exponentiated update
    eta_decay   = 4e-6          # linear decay of step size
    lam0        = 5.0           # initial soft‑max temperature
    lam_inc     = 2e-5          # increase per iteration
    lam_max     = 200.0         # cap for λ
    rnd_prob    = 0.02          # probability of a random perturbation
    rnd_frac    = 0.12          # max fraction of mass moved in a perturbation
    log_interval= 2000          # how often we print progress / test best

    size_fft = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFTs

    it = 0
    while time.time() < deadline:
        # ----- a * a (self‑convolution) ---------------------------------
        fa = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]

        # ----- soft‑max weights over the convolution entries ------------
        lam = min(lam_max, lam0 + lam_inc * it)
        w = softmax_weights(conv, lam)

        # ----- gradient of the smoothed objective -----------------------
        grad = grad_cross_correlation(a, w, size_fft)   # length n

        # ----- exponentiated‑gradient step (mirror descent) ------------
        eta = eta0 / (1.0 + it * eta_decay)
        # avoid overflow/under‑flow in the exponent
        exp_arg = -eta * np.clip(grad, -50.0, 50.0)
        a = a * np.exp(exp_arg)
        s = a.sum()
        if s == 0.0:                       # numerical safety
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ----- occasional random mass‑transfer move (exploration) ------
        if np.random.rand() < rnd_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = rnd_frac * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = cand / cand.sum()
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_seq = cand_val, cand.copy()
                    a = cand                                   # keep improvement
                else:
                    if np.random.rand() < 0.001:               # tiny uphill acceptance
                        a = cand

        # ----- bookkeeping ------------------------------------------------
        if it % log_interval == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
                print(f"[iter {it:6d}] NEW BEST {best_val:.6f}  (λ={lam:.1f})")
            else:
                print(f"[iter {it:6d}] cur {cur_val:.6f}  best {best_val:.6f}")

        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    # Return a plain Python list – the external `evaluate_sequence` will
    # perform its own security checks.
    return best_seq.tolist()
