    # compute full convolution length len(x)+len(y)-1
    n = x.shape[0] + y.shape[0] - 1

"""
A fast Adam‑based optimizer for the step‑function problem.

The algorithm works with a probability vector `a` (non‑negative, sum = 1).
It repeatedly

1. Computes the self‑convolution `c = a*a`.
2. Forms a smooth approximation of the maximum using a softmax with a
   temperature `λ` (λ grows from a small value to a large one, turning the
   smooth max into the exact max in later stages).
3. The gradient of the smooth max is the weighted autocorrelation
   `∇f(a) = 2·(w ⋆ a)` where the weights `w` are the softmax probabilities.
   This correlation is obtained efficiently with a real FFT.
4. An Adam update (first‑ and second‑moment estimates) is applied,
   the step is capped so that no component is reduced by more than 95 %,
   and the result is projected back onto the simplex (Euclidean projection).
5. The candidate is accepted if the true objective
   `2·n·max(a*a) / (sum(a)²)` improves; otherwise it may be accepted
   with a Metropolis probability that slowly cools.
6. Occasionally a tiny random perturbation is added to escape plateaus,
   and a restart is triggered if no improvement is seen for many
   iterations.

The routine stops after the allotted time budget and returns the best
sequence found (as a plain Python list).  The returned sequence can be
evaluated with the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of a_i = C(2i,i) / 4**i (half‑binomial series),
    normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries become equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using a real FFT (full length)."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:m]                            # length 2n‑1


def corr_fft(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """Correlation w ⋆ a_rev (full length) via FFT.
    `a_rev` must be the reversed version of a."""
    m = w.shape[0] + a_rev.shape[0] - 1
    size = 1 << (m - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A, n=size)
    return conv[:m]                            # length 3n‑2 for our use


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 950.0,
                     **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimizes
    evaluate_sequence.  The method combines a smooth‑max approximation,
    the Adam optimiser, simplex projection and a simple simulated‑annealing
    acceptance rule.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # configurable parameters
    n = 4000                     # length of the sequence (larger → better bound)
    max_iter = 10_000_000        # safety bound (real stopping is time‑driven)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    step_size = 0.07             # base step size for Adam (scaled later)

    # soft‑max (smooth‑max) temperature schedule
    lam = 5.0                    # start very smooth
    lam_inc = 0.0006             # linear increase per iteration
    lam_max = 200.0

    # annealing temperature for Metropolis acceptance
    temperature = 0.02
    temp_decay = 0.9995
    min_temperature = 1e-7

    # restart / random‑perturbation controls
    restart_thresh = 200_000     # iterations without improvement → restart
    noise_prob = 0.003           # chance of injecting tiny noise each iteration
    noise_scale = 5e-5

    # --------------------------------------------------------------
    # initial vector (half‑binomial, already sums to 1)
    a = sqrt_series(n)

    # Adam moment estimates
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # current objective (sum(a) = 1 ⇒ eval = 2·n·max(conv))
    conv = conv_self_fft(a)
    max_val = conv.max()
    cur_score = 2.0 * n * max_val
    best_a = a.copy()
    best_score = cur_score
    last_improve = 0

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # safety margin
    it = 0

    while time.time() < deadline and it < max_iter:
        # ---------- evaluate current convolution ----------
        conv = conv_self_fft(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[iter {it:08d}] new best score = {best_score:.6f}")

        # ---------- smooth‑max weight ----------
        # w_i ∝ exp(lam * (conv_i - max_val))  (numerically stable)
        exp_arg = lam * (conv - max_val)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:          # extreme underflow – fall back to hard max
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w /= w_sum

        # ---------- gradient of smooth‑max ----------
        # gradient = 2 * (w ⋆ a_rev)[n-1 : 2n-1]
        a_rev = a[::-1]
        corr_full = corr_fft(w, a_rev)               # length 3n‑2
        grad = 2.0 * corr_full[n - 1: 2 * n - 1]      # shape (n,)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = step_size * m_hat / (np.sqrt(v_hat) + epsilon)

        # move opposite to the gradient (since grad points towards increase)
        step = np.minimum(step, a * 0.95)   # avoid making any component negative
        a_trial = a - step

        # occasional tiny random perturbation
        if rng.random() < noise_prob:
            a_trial += (rng.random(n) - 0.5) * noise_scale

        # project back onto the probability simplex
        a_trial = project_onto_simplex(a_trial)

        # ---------- evaluate trial ----------
        conv_trial = conv_self_fft(a_trial)
        max_val_trial = conv_trial.max()
        trial_score = 2.0 * n * max_val_trial

        # accept if better, otherwise with Metropolis probability
        if trial_score <= cur_score:
            a = a_trial
            cur_score = trial_score
        else:
            delta = trial_score - cur_score
            if temperature > 0 and rng.random() < np.exp(-delta / temperature):
                a = a_trial
                cur_score = trial_score
            # else reject: keep current a

        # ---------- schedule updates ----------
        lam = min(lam + lam_inc, lam_max)
        temperature = max(temperature * temp_decay, min_temperature)

        # restart if stagnated
        if it - last_improve > restart_thresh:
            # perturb the best solution lightly and restart from there
            a = best_a + rng.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a)
            # reset Adam moments to avoid stale direction
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            last_improve = it
            print(f"[iter {it:08d}] restart – perturbed best, score={cur_score:.6f}")

        # occasional progress report
        if (it + 1) % 20000 == 0:
            print(f"[iter {it+1:08d}] best={best_score:.6f}, cur={cur_score:.6f}, λ={lam:.2f}")

        it += 1

    # ------------------------------------------------------------------
    # final evaluation with the supplied routine (extra safety checks)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
