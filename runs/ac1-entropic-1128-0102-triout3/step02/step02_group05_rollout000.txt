"""
Simulated‑annealing + Adam‑projected gradient descent for the max‑self‑convolution
problem.

The algorithm works as follows:

1. Choose a length `n` (here 5000, a good trade‑off between quality and speed).
2. Initialise several candidate probability vectors:
   * uniform,
   * a centred Gaussian,
   * a power‑law family,
   * the optional previous global `height_sequence_1`.
   The best of those seeds is taken as the start.
3. Main loop (until the time budget expires):
   * Compute the self‑convolution `c = a*a` by FFT.
   * Locate all indices whose convolution value is within 1e-14 of the
     maximum (the active set of constraints).
   * Form the sub‑gradient of the maximum‐convolution value as the average of
     the sub‑gradients of the active indices.
   * Update the probability vector using the Adam optimiser
     (first‐order method with adaptive step‑size).  After the Adam step the
     vector is projected onto the probability simplex by simple
     non‑negative clipping and renormalisation.
   * With a small probability a random mass‑transfer (hill‑climbing) move
     is tried; with another small probability a short smoothing kernel
     (`[0.25,0.5,0.25]`) is applied.  Improvements are kept, and occasional
     uphill moves are accepted to escape plateaus.
   * The current objective `2*n*max(c)` (identical to `evaluate_sequence`
     for a normalised vector) is monitored and the best sequence seen so far
     is stored.
4. Return the best sequence as a plain Python list.

The combination of Adam’s adaptive learning rates, handling of all active
constraints, and occasional stochastic moves tends to improve over the
plain sub‑gradient method used previously, and the larger dimension gives
the algorithm a richer search space.
"""

import numpy as np
import time
from typing import List

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective for a normalised probability vector."""
    # a is assumed to sum to 1 (or close to it)
    n = a.size
    # convolution via FFT (next power of two)
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
def build_uniform(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


# ----------------------------------------------------------------------
def build_gaussian(n: int, sigma_factor: float = 0.15) -> np.ndarray:
    """Centered Gaussian, sigma = sigma_factor * n."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    total = a.sum()
    if total == 0:
        return build_uniform(n)
    return a / total


# ----------------------------------------------------------------------
def build_powerlaw(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Power‑law a_i ∝ (i+1+offset)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        return build_uniform(n)
    return a / total


# ----------------------------------------------------------------------
def subgradient_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Compute the average sub‑gradient of the max‑convolution value
    over all active indices.
    """
    n = a.size
    grad = np.zeros_like(a)
    i = np.arange(n)
    for k in active_idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
    if active_idxs.size > 0:
        grad /= float(active_idxs.size)
    return grad


# ----------------------------------------------------------------------
def adam_projected_step(
    a: np.ndarray,
    grad: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    alpha: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform one Adam update and project onto the probability simplex
    by simple clipping and renormalisation.
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    a_new = a - alpha * m_hat / (np.sqrt(v_hat) + eps)

    # clip to non‑negative and renormalise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new = build_uniform(a.size)
    else:
        a_new /= s
    return a_new, m, v


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (default 1000 s).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose sequence length.
    n = 5000

    # ------------------------------------------------------------------
    # 2. Initialise several candidates and keep the best.
    best_seq = None
    best_val = np.inf

    candidates = [
        build_uniform(n),
        build_gaussian(n, sigma_factor=0.12),
        build_gaussian(n, sigma_factor=0.18),
        build_powerlaw(n, p=0.4, offset=0.0),
        build_powerlaw(n, p=0.55, offset=0.5),
    ]

    # Include previous global solution if available.
    if "height_sequence_1" in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                cand = prev[:n].copy()
            else:
                cand = np.zeros(n, dtype=np.float64)
                cand[: prev.size] = prev
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand /= s
                candidates.append(cand)
        except Exception:
            pass

    for idx, seq in enumerate(candidates):
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
        # debug output
        print(f"[init] candidate {idx:02d} value {val:.6f}")

    print(f"[init] best after seeding: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (Adam + stochastic moves).
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 1

    # hyper‑parameters
    alpha = 0.02          # base Adam step size
    random_move_prob = 0.04
    random_move_scale = 0.25
    smoothing_prob = 0.015
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    # pre‑compute FFT size
    conv_size = 1 << ((2 * n - 1).bit_length())

    # timing statistics
    last_report = time.time()
    report_interval = 5.0   # seconds

    while time.time() < deadline:
        # ----- 3.1 self‑convolution and active set -----
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        max_val = conv.max()
        # active indices – include those within a tiny tolerance
        tol = 1e-14
        active = np.where(conv >= max_val - tol)[0]

        # ----- 3.2 compute sub‑gradient (average over active set) -----
        grad = subgradient_active(a, active)

        # ----- 3.3 Adam step with projection -----
        a, m, v = adam_projected_step(a, grad, m, v, t_iter,
                                      alpha=alpha, beta1=0.9, beta2=0.999, eps=1e-8)

        # ----- 3.4 occasional random hill‑climbing move -----
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_scale * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = np.maximum(cand, 0.0)
                cand_sum = cand.sum()
                if cand_sum > 0:
                    cand /= cand_sum
                    cand_val = fast_score(cand)
                    if cand_val < best_val:
                        best_val, best_seq = cand_val, cand.copy()
                        a = cand.copy()          # keep improvement
                        # reset Adam moments to avoid stale direction
                        m.fill(0.0)
                        v.fill(0.0)

        # ----- 3.5 occasional smoothing move -----
        if np.random.rand() < smoothing_prob:
            cand = np.convolve(a, kernel, mode='same')
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand /= s
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_seq = cand_val, cand.copy()
                    a = cand.copy()
                    m.fill(0.0)
                    v.fill(0.0)

        # ----- 3.6 bookkeeping and reporting -----
        cur_val = 2.0 * n * max_val   # identical to fast_score(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
        # periodic status print
        if time.time() - last_report >= report_interval:
            print(f"[iter {t_iter:07d}] cur {cur_val:.6f} best {best_val:.6f}")
            last_report = time.time()

        t_iter += 1

    print(f"[final] best value found: {best_val:.6f}")
    # Return plain Python list as required by the evaluator.
    return best_seq.tolist()
