def gradient_of_top_k(a, conv, k):
    # find indices of top k conv values
    if k >= len(conv):
        idxs = np.arange(len(conv))
    else:
        idxs = np.argpartition(conv, -k)[-k:]
    # weights proportional to conv value
    vals = conv[idxs]
    # avoid zero
    if vals.sum() == 0:
        weights = np.full_like(vals, 1.0/len(vals))
    else:
        weights = vals / vals.sum()
    grad = np.zeros_like(a)
    for idx, w in zip(idxs, weights):
        grad += w * gradient_of_peak(a, idx)
    return grad

def propose_candidate(seed=0, budget_s=950.0):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    # Phase1: coarse search
    ...
    # After coarse, we have best_a

    # Phase2: generate multiple starting points and gradient refine
    best_global = best_a.copy()
    best_score = 2 * n * conv_fft(best_a).max()
    # number of restarts determined by time
    restart = 0
    while time.time() < deadline and restart < max_restarts:
        # pick start: if restart==0 use best_a; else random Dirichlet
        if restart == 0:
            a = best_a.copy()
        else:
            # random Dirichlet: sample exp(rng.random) and normalize
            a = rng.exponential(scale=1.0, size=n)
            a /= a.sum()
        cur_score = 2 * n * conv_fft(a).max()
        # gradient descent loop with limited iterations
        eta = 0.1
        for it in range(max_iter_per_restart):
            if time.time() > deadline: break
            conv = conv_fft(a)
            max_val = conv.max()
            # find top K indices
            K = 5
            idx_top = np.argpartition(conv, -K)[-K:]
            vals = conv[idx_top]
            # compute weights: softmax based on val - max_val
            beta = 5.0  # smoothing parameter, maybe increase later
            w = np.exp(beta * (vals - max_val))
            w_sum = w.sum()
            if w_sum > 0:
                w = w / w_sum
            else:
                w = np.ones_like(w) / len(w)

            grad = np.zeros_like(a)
            for idx, weight in zip(idx_top, w):
                grad += weight * gradient_of_peak(a, idx)

            # step
            step = eta * grad
            step = np.minimum(step, a * 0.99)  # no negative
            a_trial = a - step

            # jitter occasional
            if rng.random() < 0.003:
                jitter = (rng.random(n) - 0.5) * 1e-6
                a_trial += jitter

            # project onto simplex
            a_trial = project_onto_simplex(a_trial)

            # evaluate
            conv_trial = conv_fft(a_trial)
            trial_score = 2 * n * conv_trial.max()

            # update global best
            if trial_score < best_score:
                best_score = trial_score
                best_global = a_trial.copy()
                print(f"[restart {restart} iter {it}] new best score = {best_score:.6f}")

            # accept move if improves or small probability
            if trial_score <= cur_score:
                a = a_trial
                cur_score = trial_score
            else:
                # Metropolis acceptance with small temperature
                if rng.random() < np.exp(-(trial_score - cur_score) / (0.01 + restart*0.001)):
                    a = a_trial
                    cur_score = trial_score

            # decay eta periodically
            if (it+1) % 2000 == 0:
                eta = max(eta * 0.7, 1e-5)

        restart += 1
        # optionally update max_restarts based on time

    # final evaluate
    final_score = evaluate_sequence(best_global.tolist())
    print("[final] best score = ", final_score)

    return best_global.tolist()

"""
Propose a candidate step‑function sequence that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

using a three‑stage search:

1.  **Fast power‑law sweep** – generate a_i ∝ (i+offset)^(–p) for a
    moderate length (default n=2500).  A coarse grid over (p,offset) is
    evaluated and the best point is kept.

2.  **Multi‑restart projected gradient descent** – improve the best
    candidate (and a few random Dirichlet starts) with many iterations.
    Rather than following only the single maximal convolution entry,
    the gradient is a weighted combination of the top‑K entries
    (soft‑max weighting).  After each step the vector is projected onto
    the probability simplex (non‑negative, unit sum).  Step size,
    temperature and decay are scheduled; occasional jitter avoids
    plateaus.

3.  **Return the best found sequence** – the final list is checked with
    ``evaluate_sequence`` before returning.

The routine respects a ~950 s internal budget (leaving a safety margin)
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Return a non‑negative vector a_i = (i+offset)^(-p) (i=0…n‑1) normalised
    to unit L1‑norm."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:          # defensive fallback
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Full discrete self‑convolution x*x using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w ≥ 0, Σw = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Exact sub‑gradient of the convolution entry conv[peak_idx] w.r.t. a.
    The gradient is g_i = 2·a_{peak_idx-i} for all valid i, else 0."""
    n = a.shape[0]
    grad = np.zeros_like(a)

    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


def gradient_of_top_k(a: np.ndarray, conv: np.ndarray, k: int,
                      beta: float = 5.0) -> np.ndarray:
    """Weighted blend of gradients of the top‑k convolution entries.
    Weights are soft‑maxed with temperature 1/β."""
    m = conv.shape[0]
    if k >= m:
        idxs = np.arange(m)
    else:
        idxs = np.argpartition(conv, -k)[-k:]
    vals = conv[idxs]
    maxv = conv.max()
    # soft‑max weights (shifted by max to avoid overflow)
    w = np.exp(beta * (vals - maxv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w_sum

    grad = np.zeros_like(a)
    for idx, weight in zip(idxs, w):
        grad += weight * gradient_of_peak(a, int(idx))
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """Search for a non‑negative sequence with a low evaluation score.
    The function respects the time budget and returns the best list found.
    """
    rng = np.random.default_rng(seed)

    # -------------------------------------------------- configuration
    n = 2500                      # length of the candidate vector
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ---------- phase 1 : coarse power‑law sweep ----------
    best_a = None
    best_score = np.inf
    best_p = None
    best_offset = None

    p_grid = np.arange(0.40, 0.71, 0.015)   # coarse grid
    offsets = [0.3, 0.5, 1.0, 2.0]

    print("[search] phase 1 – coarse power‑law sweep")
    for offset in offsets:
        for p in p_grid:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=offset)
            conv = conv_fft(a)
            cur = 2.0 * n * conv.max()          # sum(a)=1 → denominator=1
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                best_p = p
                best_offset = offset
                print(f"[grid] new best: p={p:.4f}, off={offset}, score={cur:.6f}")

    # ---------- phase 2 : fine local search around the best ----------
    if best_a is not None:
        fine_range = 0.02
        p_fine = np.arange(max(0.01, best_p - fine_range),
                           min(0.99, best_p + fine_range) + 0.001,
                           0.001)
        print("[search] phase 2 – fine sweep")
        for offset in offsets:
            for p in p_fine:
                if time.time() > deadline:
                    break
                a = power_law_sequence(n, p, offset=offset)
                conv = conv_fft(a)
                cur = 2.0 * n * conv.max()
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    best_p = p
                    best_offset = offset
                    print(f"[fine] new best: p={p:.4f}, off={offset}, score={cur:.6f}")

    # If for any reason we have no candidate (should not happen), fall back
    if best_a is None:
        # half‑binomial seed as a safe fallback
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
        a /= a.sum()
        best_a = a
        best_score = 2.0 * n * conv_fft(a).max()
        print("[fallback] using half‑binomial seed")

    # ---------- phase 3 : multi‑restart projected gradient descent ----------
    print("[grad] phase 3 – multi‑restart projected gradient descent")
    max_restarts = 8                # number of independent starts
    max_iter_per_restart = 300_000  # upper bound per start (will be cut by time)
    top_k = 5                       # number of dominant convolution entries
    beta_start = 3.0                # softness of the weighting
    beta_inc = 0.003                # increase per 10 000 iterations
    eta_start = 0.09                # base step size
    decay_factor = 0.85
    decay_every = 5000

    global_best_a = best_a.copy()
    global_best_score = best_score

    for restart in range(max_restarts):
        if time.time() > deadline:
            break

        # initialise the vector for this restart
        if restart == 0:
            a = best_a.copy()
        else:
            # random Dirichlet (Gamma(0.7) gives a bit sparse start)
            a = rng.gamma(0.7, 1.0, size=n)
            a /= a.sum()

        cur_score = 2.0 * n * conv_fft(a).max()
        eta = eta_start
        beta = beta_start

        it = 0
        while time.time() < deadline and it < max_iter_per_restart:
            conv = conv_fft(a)
            max_val = conv.max()

            # weighted gradient from the top‑k peaks
            grad = gradient_of_top_k(a, conv, top_k, beta=beta)

            # step opposite to the gradient (reduces the active peaks)
            step = eta * grad
            # never overshoot: keep each component ≥ 1 % of its current value
            step = np.minimum(step, a * 0.99)
            a_trial = a - step

            # occasional small random jitter to escape flat regions
            if rng.random() < 0.004:
                jitter = (rng.random(n) - 0.5) * 1e-6
                a_trial += jitter

            # project back onto the simplex (maintains non‑negativity & unit sum)
            a_trial = project_onto_simplex(a_trial)

            # cheap estimate of the score (sum is 1 by construction)
            trial_score = 2.0 * n * conv_fft(a_trial).max()

            # record global improvement
            if trial_score < global_best_score:
                global_best_score = trial_score
                global_best_a = a_trial.copy()
                print(f"[restart {restart} iter {it:07d}] global best = {global_best_score:.6f}")

            # accept if improved; otherwise Metropolis‑type acceptance
            if trial_score <= cur_score:
                a = a_trial
                cur_score = trial_score
            else:
                # tiny temperature that slowly cools
                temperature = 0.02 * np.exp(-it / 200_000.0)
                if rng.random() < np.exp(-(trial_score - cur_score) / max(temperature, 1e-12)):
                    a = a_trial
                    cur_score = trial_score

            # periodic decay of step size and increase of β (sharper focus)
            if (it + 1) % decay_every == 0:
                eta = max(eta * decay_factor, 1e-6)
                beta = beta + beta_inc

            it += 1

        # after each restart report the best found so far
        print(f"[restart {restart}] finished – local best score = {cur_score:.6f}")

    # ----------- final sanity check with the official evaluator -------------
    final_score = evaluate_sequence(global_best_a.tolist())
    print(f"[done] final reported score = {final_score:.6f}")
    print(f"[info] best parameters from sweep: p={best_p:.5f}, offset={best_offset}")

    return global_best_a.tolist()
