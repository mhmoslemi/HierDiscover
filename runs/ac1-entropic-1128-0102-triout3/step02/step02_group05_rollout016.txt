Propose candidate step function via momentum sub‑gradient descent
augmented with occasional linear‑programming “direction” steps.
The algorithm:
1. Initialise a probability vector a on the simplex using a power‑law scan
   (and possibly the previously found `height_sequence_1`).
2. Main loop (time‑limited):
   * Compute self‑convolution a*a with FFT; locate the active index
     where the maximum occurs.
   * Sub‑gradient for that index is g_i = 2·a_{k−i}.  A momentum term
     averages the sub‑gradient and a projected‑gradient step on the
     simplex is taken.
   * With a small probability a random mass‑transfer move is tried.
   * Every `lp_interval` seconds a linear program
     max Σ b_j  s.t. b≥0 , (a * b) ≤ current_max is solved.
     The optimal b is normalised to a simplex vector g and a
     line‑search in the segment a → g is performed – the best point
     is kept.
   * The best sequence seen so far (according to the fast evaluation
     routine) is stored.
3. Return the best sequence as a plain list.
"""

import time

# Optional SciPy imports – if unavailable the LP phase is simply skipped.

    SCIPY_AVAILABLE = False


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """Fast version of the provided evaluate_sequence (no safety checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def build_initial_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Power‑law |i+offset|^{-p} initialisation, normalised to sum‑1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def lp_direction(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t. b ≥ 0 , (a * b)_k ≤ M  for all k.
    Return the optimiser b (still on the simplex) or ``None`` if the
    LP fails or SciPy is not available.
    """
    if not SCIPY_AVAILABLE:
        return None
    n = a.shape[0]
    # Build the (2n‑1)×n convolution matrix with diags.
    # For offset d (0…n‑1) the diagonal entries are a[d].
    data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int64)
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    # Upper‑bound vector
    ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σ b_j  → maximise Σ b_j
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return None
    g = b / sum_b            # normalise to the simplex
    return g

    the evaluation function.  The routine runs for ``budget_s`` seconds and

    n = 3000

    best_seq = None

        k_max = int(np.argmax(conv_full))
        cur_max = conv_full[k_max]
        cur_val = 2.0 * n * cur_max   # sum(a)=1, so denominator =1

        # ----- 4.2 sub‑gradient with momentum -----

        grad = np.zeros_like(a)

        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad

        eta = eta0 / (1.0 + it * eta_decay)

                cand[j] += delta

                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_seq = cand_val, cand.copy()
                    a = cand
                else:
                    # tiny uphill acceptance to maintain exploration
                    if np.random.rand() < 0.001:
                        a = cand

                best_local_val = cur_val

                    coeff_aa = (1.0 - t) ** 2

                    conv_new = coeff_aa * conv_full + coeff_ag * conv_ag + coeff_gg * conv_gg

                print(f"[lp] improved to {best_val:.6f} at iter {it}")

                print(f"[iter {it:6d}] cur {cur_val:.6f} best {best_val:.6f}")

        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    return best_seq.tolist()

"""
Propose candidate step function via momentum sub‑gradient descent
augmented with occasional linear‑programming (LP) direction steps.

Algorithm
---------
1. Initialise a probability vector `a` (sum = 1) using a scan of power‑law
   families and, if available, the previously discovered `height_sequence_1`.
2. Main optimisation loop (runs until the time budget expires):
   * Compute the self‑convolution `a*a` with an FFT and locate the index
     `k_max` where the maximum occurs.
   * The sub‑gradient of the active peak is `g_i = 2·a_{k_max‑i}`.
     A momentum term smooths the updates and a projected gradient step
     (projection onto the probability simplex) is performed.
   * With a small probability a random “mass‑transfer” move between two
     coordinates is tried; improvements are kept, with a tiny chance of
     accepting a worsening move to retain diversity.
   * Every ~30 s an LP is solved:
         maximise Σ b_j  subject to  b ≥ 0  and  (a * b)_k ≤ M  ∀k,
       where `M` is the current maximal convolution entry.
     The optimiser `b` is normalised to a simplex vector `g`.  A cheap
     line‑search on the segment `a → g` (a few mixing coefficients) is
     performed using pre‑computed convolutions, and the best point replaces
     `a`.  This LP step usually yields a noticeable drop in the
     objective.

                cand[j] += delta
                cand = proj_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_seq = cand_val, cand.copy()
                    a = cand
                else:
                    if np.random.rand() < 0.001:
                        a = cand

        # ----- 4.4 periodic LP‑based direction step -----
        if SCIPY_AVAILABLE and time.time() >= next_lp_time:
            g = lp_direction(a, cur_max)
            if g is not None:
                # Pre‑compute the three necessary convolutions.
                fft_g = np.fft.rfft(g, conv_sz)
                conv_ag = np.fft.irfft(fft_a * fft_g, conv_sz)[: 2 * n - 1]
                conv_gg = np.fft.irfft(fft_g * fft_g, conv_sz)[: 2 * n - 1]

                # Try a few mixing coefficients.
                t_candidates = [0.0, 0.05, 0.10, 0.20, 0.40, 0.60, 0.80, 1.0]
                best_local_a = a
                best_local_val = cur_val
                for t in t_candidates[1:]:      # skip t=0 (already current a)
                    a_new = (1.0 - t) * a + t * g        # stays on the simplex
                    coeff_aa = (1.0 - t) ** 2
                    coeff_ag = 2.0 * t * (1.0 - t)
                    coeff_gg = t ** 2
                    conv_new = (coeff_aa * conv_full +
                                coeff_ag * conv_ag +
                                coeff_gg * conv_gg)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_local_val:
                        best_local_val = val_new
                        best_local_a = a_new.copy()

                a = proj_simplex(best_local_a)   # numerical safety
                cur_val = best_local_val
                if cur_val < best_val:
                    best_val, best_seq = cur_val, a.copy()
                print(f"[lp] improved to {best_val:.6f} at iter {it}")

            next_lp_time = time.time() + 30.0   # schedule the next LP

        # ----- 4.5 reporting & bookkeeping -----
        if it % report_every == 0:
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
                print(f"[iter {it:6d}] NEW BEST {best_val:.6f}")
            else:
                print(f"[iter {it:6d}] cur {cur_val:.6f} best {best_val:.6f}")

        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    return best_seq.tolist()
