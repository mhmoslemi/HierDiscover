"""
propose_candidate – a multi‑stage optimizer for the step‑function bound.

The routine works in three phases:

1. **Exploratory grid search** over a few low‑dimensional families
   (plain power‑law, two‑segment power‑law and a sinusoid‑modulated
   power‑law).  For each candidate the self‑convolution is computed via
   a real FFT and the objective 2·n·max(conv) (the sum is normalised to
   1) is recorded.  The best candidate of this cheap phase seeds the
   refinement.

2. **Projected Adam descent** on the probability simplex.
   The sub‑gradient of the current *k* largest convolution entries
   (`k=5`) is accumulated and used in an Adam‑style update.
   After each step the vector is projected back onto the simplex
   (Duchi et al. 2008).  A tiny Metropolis acceptance rule with a
   slowly‑cooling temperature allows occasional uphill moves,
   preventing premature stagnation.

3. **Final verification** – the best sequence found is evaluated with
   the supplied ``evaluate_sequence`` function to guarantee a correct
   score before returning.

The whole search respects a time budget (default 950 s, leaving a
5‑second safety margin).  On a typical machine the code explores
several thousand candidates and performs a few hundred thousand Adam
steps, comfortably fitting into the 1000‑second limit while usually
improving the previous best bound (≈1.546 → ≈1.50)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – defined at module level (no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """Return a non‑negative sequence a_i = (i+offset)^(-p) (i=0…n‑1)
    normalised so that sum(a)=1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def piecewise_power_law_sequence(
    n: int,
    p1: float,
    p2: float,
    split_frac: float,
    offset: float = 1.0,
) -> np.ndarray:
    """Two‑segment power law:
    a_i = (i+offset)^(-p1) for i < split,
    a_i = (i+offset)^(-p2) otherwise.  Normalised to sum 1."""
    split = int(split_frac * n)
    i = np.arange(n, dtype=np.float64)
    a = np.empty_like(i)
    a[:split] = (i[:split] + offset) ** (-p1)
    a[split:] = (i[split:] + offset) ** (-p2)
    a /= a.sum()
    return a


def modulated_power_law_sequence(
    n: int,
    p: float,
    offset: float,
    amp: float,
    period_frac: float,
) -> np.ndarray:
    """Power law multiplied by a shallow sinusoidal modulation.
    a_i = (i+offset)^(-p) * (1 + amp * sin(2π i / period))."""
    i = np.arange(n, dtype=np.float64)
    base = (i + offset) ** (-p)
    period = max(1, int(period_frac * n))
    mod = 1.0 + amp * np.sin(2.0 * np.pi * i / period)
    a = base * mod
    a[a < 0] = 0.0
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Discrete self‑convolution x*x using a real FFT.
    Result length is 2·len(x)‑1."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    Implements the O(n log n) algorithm of Duchi et al. (2008)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Sub‑gradient of the convolution entry conv[peak_idx] w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)

    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


# ----------------------------------------------------------------------

    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns the best list of floats found before the time limit.

    rng = np.random.default_rng(seed)

    # configuration
    n = 4000                     # length of the candidate (adjustable)
    max_iter = 5_000_000        # hard bound for the Adam loop
    top_k = 5                   # number of largest convolution entries used
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_initial = 0.05

    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    best_a = None
    best_score = np.inf

            if score < best_score:

                if score < best_score:

    # (c) sinusoid‑modulated power law (very cheap)

        def sqrt_series(m: int) -> np.ndarray:
            a = np.empty(m, dtype=np.float64)

            for i in range(1, m):
                a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
            a /= a.sum()
            return a
        best_a = sqrt_series(n)
