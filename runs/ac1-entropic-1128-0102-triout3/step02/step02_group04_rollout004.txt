The loop runs until the allotted time expires (≈950 s, leaving a safety
margin).  The best sequence encountered is returned; the caller will
evaluate it with the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of a_i = C(2i,i) / 4**i (half‑binomial series),
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Fast convolution x * x using a real FFT.
    Length of result is 2*len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the valid cross‑correlation corr(w, a) (length len(a))
    using FFTs.  Result[i] = Σ_j w[j+i]·a[j]   (0 ≤ i < len(a)).
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    L = w.shape[0]                      # = 2n‑1
    m = L + n - 1                       # = 3n‑2
    # next power‑of‑two is slightly faster for FFTs
    N = 1 << ((m - 1).bit_length())
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a_rev, n=N)
    conv = np.fft.irfft(W * A, n=N)
    # The valid correlation is the reversed first n elements of conv
    return conv[n - 1 :: -1][:n]         # length n


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that minimizes the provided
    evaluation function.  Returns the best sequence found (list of floats).
    """
    rng = np.random.default_rng(seed)

    # -------------------- configuration --------------------
    n = 2500                    # sequence length
    max_iter = 10_000_000       # hard cap (time will stop us before)
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # schedules (linear in iteration count, capped after ramp_iters)
    lambda_start = 8.0          # soft‑max start
    lambda_end = 250.0          # final sharpness
    eta_start = 0.15            # initial learning‑rate
    eta_end = 0.001             # final learning‑rate
    ramp_iters = 250_000        # iterations over which schedules transition
    restart_gap = 150_000       # iterations without improvement before a restart
    uphill_prob = 0.001         # probability to accept a worsening move
    noise_prob = 0.002          # occasional random nudges
    noise_scale = 5e-5

    # -------------------- initial point --------------------
    a = sqrt_series(n)          # already sums to 1
    conv = conv_fft(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # Adam accumulators
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # bookkeeping
    start_time = time.time()
    deadline = start_time + budget_s - 5.0     # a few seconds safety margin
    it = 0
    last_improve = 0

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # -------------------- optimisation loop --------------------
    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- schedule for λ and η -----
        progress = min(it / ramp_iters, 1.0)
        lam = lambda_start + (lambda_end - lambda_start) * progress
        eta = eta_start * (eta_end / eta_start) ** progress

        # ----- convolution and soft‑max weights -----
        conv = conv_fft(a)                     # full self‑convolution
        max_conv = conv.max()
        # stable soft‑max
        shift = conv - max_conv                 # max goes to zero
        w = np.exp(lam * shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (exact) of smoothed max -----
        grad = correlation_fft(w, a)           # corr(w, a)
        grad *= 2.0

        # ----- Adam update -----
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
        m_hat = m_vec / (1.0 - beta1_pow)
        v_hat = v_vec / (1.0 - beta2_pow)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- trial point -----
        a_trial = a - step

        # occasional random perturbation to escape flat regions
        if rng.random() < noise_prob:
            a_trial += (rng.random(n) - 0.5) * noise_scale

        # project back onto the simplex (ensures non‑negativity and sum=1)
        a_trial = project_onto_simplex(a_trial)

        # ----- evaluate trial -----
        conv_trial = conv_fft(a_trial)
        trial_score = 2.0 * n * conv_trial.max()

        # keep global best
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            last_improve = it
            if it % 5000 == 0 or it == 1:
                print(f"[iter {it:08d}] new best score = {best_score:.6f}")

        # ----- acceptance rule -----
        accept = False
        if trial_score <= cur_score:
            accept = True
        elif rng.random() < uphill_prob:
            accept = True

        if accept:
            a = a_trial
            cur_score = trial_score

        # ----- restart if stuck -----
        if it - last_improve > restart_gap:
            # restart from a fresh Dirichlet draw (more exploratory)
            a = rng.dirichlet(np.full(n, 0.7))
            a = project_onto_simplex(a)                # safety
            cur_score = 2.0 * n * conv_fft(a).max()
            last_improve = it
            print(f"[iter {it:08d}] restart triggered – new random start, "
                  f"score={cur_score:.6f}")

        # optional early break if we have already reached the target bound
        if best_score <= 1.505:
            # we are very close; continue a little longer for safety
            if it - last_improve > 10_000:
                break

    # -------------------- final evaluation --------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
