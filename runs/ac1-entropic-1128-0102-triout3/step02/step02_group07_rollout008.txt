"""
Propose a candidate sequence that minimizes `evaluate_sequence`.

Algorithm
---------

1. **Warm‑start** – if a global `height_sequence_1` exists we use it
   (renormalised to sum = 1).  Otherwise we start from scratch.

2. **Parametric sweep** – generate sequences of the form  

        a_i ∝ (i + offset)^{‑p} · r^{i}          (i = 0 … n‑1)

   where `p` (power‑law exponent), `offset` (shifts the singularity) and
   `r` (geometric damping) are scanned on a coarse grid.
   The best grid point (lowest `2·n·max(conv)`) is kept.

3. **Fine grid** – zoom around the best `(p, r, offset)` with a fine
   step (Δp ≈ 0.001, Δr ≈ 0.001).  This usually beats the half‑binomial
   baseline already.

4. **Projected gradient descent** – using the exact sub‑gradient of the
   *active* convolution entries (all indices whose value is within 95 %
   of the current maximum) we take a step opposite to the gradient,
   project back onto the probability simplex and slowly decay the step
   size and a Metropolis temperature.

5. **Stochastic hill‑climbing** – the refined vector is thereafter
   perturbed by random mass transfers `a_i → a_j`.  The perturbation size
   shrinks with the temperature.  Moves are accepted if they improve
   the score or with probability `exp(-(Δ)/T)`.  This phase is able to
   escape tiny local minima that the gradient step may miss.

6. **Return the best vector** – the globally best sequence discovered
   during the whole run is validated with `evaluate_sequence` and
   returned as a plain ``list``.

The routine respects the 1000 s overall budget (default 950 s are used
internally to leave a safety margin) and prints progress information.
"""

def power_law_exp_sequence(
    n: int,
    p: float,
    r: float,
    offset: float = 1.0,
) -> np.ndarray:
    """
    a_i = (i+offset)^{-p} * r^{i}  (i = 0 … n‑1)
    Returned vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) * (r ** i)
    total = a.sum()
    if total == 0.0:                         # fall‑back – uniform
        a[:] = 1.0 / n
    else:
        a /= total
    return a

    n: int,

    offset: float = 1.0,
) -> np.ndarray:
    """Legacy variant without geometric damping (r = 1)."""
    return power_law_exp_sequence(n, p, r=1.0, offset=offset)


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Compute the discrete self‑convolution x * x using a real FFT.
    Length of the result is 2·len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry `conv[peak_idx]`
    w.r.t. the vector `a` (self‑convolution).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


def score_from_a(a: np.ndarray) -> float:
    """
    Fast surrogate score: 2·n·max(conv(a,a))  (assuming sum(a)=1).
    The real evaluator also divides by (sum(a))², but we keep sum=1.
    """
    n = a.shape[0]
    conv = conv_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative sequence that minimises the provided
    evaluation function within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Seconds allocated for the search (default 950 s, leaving a 5 s
        safety margin for the final sanity check).

    Returns
    -------
    list[float]
        The best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)

    # -------------------------- configuration ---------------------------
    n = 2500                     # length of the candidate
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_a = None
    best_score = np.inf
    best_params = None   # (p, r, offset)

    # ------------------- optional warm‑start ---------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            if cand.ndim == 1 and cand.size > 1:
                # adjust length if necessary
                if cand.size != n:
                    if cand.size > n:
                        cand = cand[:n]
                    else:
                        cand = np.concatenate([cand,
                                               np.zeros(n - cand.size)])
                # normalise to sum = 1
                s = cand.sum()
                if s == 0.0:
                    cand[:] = 1.0 / n
                else:
                    cand /= s
                sc = score_from_a(cand)
                best_a = cand.copy()
                best_score = sc
                best_params = ('warm', 'warm', 'warm')
                print(f"[warm‑start] score={sc:.6f}")
        except Exception as e:
            # any problem – simply ignore the warm start
            print(f"[warm‑start] failed ({e})")

    # ------------------- parameter sweep (coarse) --------------------
    print("[search] starting coarse parametric sweep")
    p_grid = np.arange(0.40, 1.05, 0.02)       # power‑law exponent
    r_grid = np.linspace(0.85, 0.999, 12)      # geometric damping
    offsets = [0.5, 1.0, 2.0, 3.0, 5.0]        # shift of the singularity

    for offset in offsets:
        for r in r_grid:
            for p in p_grid:
                if time.time() > deadline:
                    break
                a = power_law_exp_sequence(n, p, r, offset)
                sc = score_from_a(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    best_params = (p, r, offset)
                    print(f"[grid] new best  p={p:.3f}, r={r:.3f}, "
                          f"off={offset}, score={sc:.6f}")

    # ------------------- fine grid around the best point ---------------
    if best_params is not None:
        p0, r0, off0 = best_params
        p_fine = np.arange(max(0.01, p0 - 0.02),
                           min(0.99, p0 + 0.02) + 1e-9,
                           0.001)
        r_fine = np.arange(max(0.5, r0 - 0.005),
                           min(0.9999, r0 + 0.005) + 1e-9,
                           0.0005)
        offs_fine = [off0]    # keep the offset fixed in the fine phase

        print("[search] starting fine sweep")
        for offset in offs_fine:
            for r in r_fine:
                for p in p_fine:
                    if time.time() > deadline:
                        break
                    a = power_law_exp_sequence(n, p, r, offset)
                    sc = score_from_a(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        best_params = (p, r, offset)
                        print(f"[fine] new best  p={p:.4f}, r={r:.4f}, "
                              f"off={offset}, score={sc:.6f}")

    # ------------------------------------------------------------------
    # At this point we have a decent seed (best_a).  If still None we
    # fall back to a simple half‑binomial seed.
    if best_a is None:
        # classic half‑binomial (Square‑root) construction
        def half_binomial(m: int) -> np.ndarray:
            a = np.empty(m, dtype=np.float64)
            a[0] = 1.0
            for i in range(1, m):
                a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
            a /= a.sum()
            return a
        best_a = half_binomial(n)
        best_score = score_from_a(best_a)
        print("[fallback] half‑binomial seed score=", best_score)

    # -------------------------- gradient descent ------------------------
    print("[grad] projected gradient descent start")
    a_cur = best_a.copy()
    cur_score = best_score

    eta = 0.07                     # initial step size
    temp = 0.03                    # Metropolis temperature
    decay_every = 2500
    decay_factor = 0.93
    max_grad_iters = 4_000_000

    it = 0
    while time.time() < deadline and it < max_grad_iters:
        conv = conv_fft(a_cur)
        max_val = conv.max()
        # active set: all indices whose value is at least 95 % of max
        active = np.where(conv >= 0.95 * max_val)[0]

        # combined gradient of the active entries
        grad = np.zeros_like(a_cur)
        for k in active:
            grad += gradient_of_peak(a_cur, int(k))

        # step opposite to the gradient
        step = eta * grad

        # keep moves small enough to stay non‑negative
        step = np.minimum(step, a_cur * 0.99)

        a_next = a_cur - step
        # projection onto the simplex (keeps sum = 1)
        a_next = project_onto_simplex(a_next)

        # evaluate trial
        trial_score = score_from_a(a_next)

        # keep globally best
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_next.copy()
            print(f"[grad iter {it:08d}] new global best score={trial_score:.6f}")

        # accept / reject
        if trial_score <= cur_score:
            a_cur = a_next
            cur_score = trial_score
        else:
            delta = trial_score - cur_score
            if temp > 0.0 and rng.random() < math.exp(-delta / temp):
                a_cur = a_next
                cur_score = trial_score
            # otherwise reject

        # schedule decay
        if (it + 1) % decay_every == 0:
            eta = max(eta * decay_factor, 1e-6)
            temp = max(temp * decay_factor, 1e-9)

        it += 1

    # ------------------------ stochastic hill‑climbing -----------------
    print("[hill] stochastic refinement start")
    a_cur = best_a.copy()
    cur_score = best_score

    temp = 0.02                     # start temperature for Metropolis
    step_factor = 0.15              # relative amount of mass moved each trial
    max_hill_iters = 3_000_000

    it = 0
    while time.time() < deadline and it < max_hill_iters:
        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # amount of mass to transfer – up to `step_factor` of a[i]
        delta = a_cur[i] * rng.random() * step_factor
        if delta == 0.0:
            # nothing to move – skip
            it += 1
            continue

        a_trial = a_cur.copy()
        a_trial[i] -= delta
        a_trial[j] += delta   # sum stays exactly 1, non‑negative guaranteed

        trial_score = score_from_a(a_trial)

        # keep global best
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            print(f"[hill iter {it:08d}] new global best score={trial_score:.6f}")

        # accept according to Metropolis rule
        if trial_score <= cur_score:
            a_cur = a_trial
            cur_score = trial_score
        else:
            delta_score = trial_score - cur_score
            if temp > 0.0 and rng.random() < math.exp(-delta_score / temp):
                a_cur = a_trial
                cur_score = trial_score

        # anneal temperature slowly
        if (it + 1) % 5000 == 0:
            temp = max(temp * 0.97, 1e-9)
            step_factor = max(step_factor * 0.995, 0.001)

        it += 1

    # ----------------------- final sanity check -----------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] best reported score = {final_score:.6f}")
    if best_params is not None:
        print("[info] best parameters (p, r, offset) =", best_params)

    return best_a.tolist()
