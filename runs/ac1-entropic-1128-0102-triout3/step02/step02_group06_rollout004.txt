"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm combines two phases:

1. **Adam mirror‑descent on the simplex** (log‑parameterisation).  
   A soft‑max approximation of the maximum entry of the self‑convolution
   is annealed to the exact max, and Adam adapts per‑coordinate step sizes.
   This quickly reaches a good local minimum.

2. **Simulated‑annealing local search** starting from the best Adam
   solution.  Small amounts of mass are moved between two positions
   while preserving the ℓ₁‑norm; moves that improve the maximal
   convolution are always accepted, worsening moves are accepted with
   a probability that decays with a temperature schedule.  The move
   size and temperature shrink over time, allowing fine‑grained
   refinement.

Both phases respect the simplex constraint Σa = √(2·n) and keep all
entries non‑negative.  The best sequence found before the time budget
expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the first (2·len(a)‑1) entries (real values).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a (soft‑max approximation of max).
    """
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1: n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the index k_max.
    The derivative of (a*a)[k] w.r.t. a_i equals 2·a_{k‑i}
    if the index lies inside the support, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σ x = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1729,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step function that minimises `evaluate_sequence`.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # initialise sequence length, simplex target and starting point
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.size
    else:
        n = 2401                                     # default size
        init_seq = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64)

    # required ℓ₁‑norm (problem normalisation)
    S = np.sqrt(2.0 * n)
    init_seq = init_seq * (S / init_seq.sum())

    # log‑variables for Adam (mirror descent on the simplex)
    eps = 1e-12
    log_theta = np.log(init_seq + eps)

    # bookkeeping of the best solution
    best_a = init_seq.copy()
    best_score = float('inf')
    # evaluate the initial point once
    cur_conv = _convolve_self_fft(best_a, _next_pow2(2 * n - 1))
    best_score = cur_conv.max()
    print(f"[init] n={n}, initial max‑conv = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam on log‑variables (soft‑max annealing)
    # --------------------------------------------------------------
    nfft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta = 0.25                         # base learning rate in log‑space
    m = np.zeros_like(log_theta)
    v = np.zeros_like(log_theta)

    # temperature schedule for softmax
    tau_start = 0.8
    tau_min = 1e-4
    # approximate number of Adam iterations we would like to afford
    adam_iterations_est = 250_000
    tau_decay = (tau_min / tau_start) ** (1.0 / adam_iterations_est)
    tau = tau_start

    # restart handling
    no_improve = 0
    restart_limit = 6_000

    iteration = 0
    adam_deadline = start_time + int(budget_s * 0.55)

    while time.time() < adam_deadline:
        iteration += 1

        # reconstruct a from log‑variables and enforce Σa = S
        a = np.exp(log_theta)
        sum_a = a.sum()
        if sum_a == 0.0:
            # degenerate situation – re‑initialise from a random Dirichlet
            a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64) * S
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            continue
        a = a * (S / sum_a)

        # current convolution and objective
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = cur_max                     # scaling factor is 1

        # bookkeeping
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:08d}] new best max‑conv = {best_score:.6f}")
        else:
            no_improve += 1

        # gradient of the smoothed max (or exact max)
        if tau > tau_min * 2:
            grad_a = _grad_softmax(a, conv, tau, nfft)
        else:
            k_max = int(conv.argmax())
            grad_a = _grad_max(a, conv, k_max)

        # chain rule for log‑variables
        grad_log = grad_a * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step   # descent

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.004:
            jitter = np.random.randn(n) * (eta * 0.07)
            log_theta = log_theta + jitter

        # temperature annealing
        tau = max(tau * tau_decay, tau_min)

        # restart after prolonged stagnation
        if no_improve >= restart_limit:
            # perturb the best solution multiplicatively and continue
            a_pert = best_a * np.exp(np.random.randn(n) * (eta * 0.25))
            a_pert = a_pert * (S / a_pert.sum())
            log_theta = np.log(a_pert + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Adam {iteration:08d}] restart after stagnation")

    # --------------------------------------------------------------
    # Phase 2 – Simulated‑annealing local refinement
    # --------------------------------------------------------------
    sa_deadline = deadline
    a = best_a.copy()
    conv = _convolve_self_fft(a, nfft)
    cur_max = conv.max()
    cur_score = cur_max
    best_score = cur_max
    best_a = a.copy()

    # SA hyper‑parameters
    temp = 0.12                     # initial temperature
    temp_min = 1e-6
    temp_decay = 0.9995

    step_factor = 0.025 * S         # initial move magnitude (≈2.5 % of total mass)
    step_min = 1e-9 * S
    step_decay = 0.9994

    it_sa = 0
    while time.time() < sa_deadline:
        it_sa += 1

        # recompute max index and gradient only when needed
        k_max = int(conv.argmax())
        grad_vec = _grad_max(a, conv, k_max)
        # support indices where gradient > 0 (i contributes to the current max)
        support = np.where(grad_vec > 0)[0]
        if support.size == 0:
            # fall back to uniform random pick if something unexpected happens
            i = np.random.randint(n)
        else:
            i = np.random.choice(support)

        # choose a distinct destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose moving a small amount of mass from i to j
        max_move = min(step_factor, a[i])
        if max_move <= 0.0:
            continue
        delta = np.random.rand() * max_move

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # convolution of the candidate
        conv_new = _convolve_self_fft(a_new, nfft)
        new_max = conv_new.max()
        new_score = new_max

        # acceptance rule
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            if temp > temp_min:
                prob = np.exp(-(new_score - cur_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_score = new_score
            cur_max = new_max
            # record global best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[SA {it_sa:08d}] new best max‑conv = {best_score:.6f}")

        # anneal temperature and step size
        temp = max(temp * temp_decay, temp_min)
        step_factor = max(step_factor * step_decay, step_min)

        # periodic re‑normalisation to compensate floating‑point drift
        if it_sa % 1024 == 0:
            a = a * (S / a.sum())
            conv = _convolve_self_fft(a, nfft)
            cur_score = conv.max()
            cur_max = cur_score

    # --------------------------------------------------------------
    # final sanity projection and verification
    # --------------------------------------------------------------
    final_seq = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    elapsed = time.time() - start_time
    print(f"Finished (elapsed {elapsed:.1f}s). Best max‑conv = {final_score:.6f}")
    return final_seq.tolist()
