```
def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = float('inf')

    # If height_sequence_1 defined, use it as starting shape scaled to new lengths.
    base_seq = None
    if 'height_sequence_1' in globals():
        base_seq = np.array(height_sequence_1, dtype=np.float64)

    # define lengths to try in increasing order
    lengths = [800, 1600, 3200, 4800]  # adjust based on time

    for n in lengths:
        if time.time() >= deadline:
            break

        # Initialize a: if we have base_seq of different length, upsample/resample to n.
        if base_seq is not None and base_seq.size != n:
            # simple linear interpolation to size n
            x_old = np.linspace(0, 1, base_seq.size)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, base_seq)
            a = np.maximum(a, 0)
            a /= a.sum()
        else:
            # start uniform or random gaussian.
            a = np.full(n, 1.0/n, dtype=np.float64)
            # could add small random noise
            a += 0.01 * np.random.randn(n)
            a = np.clip(a, 0, None)
            a /= a.sum()

        # ADAM parameters
        z = np.log(a + 1e-12)   # log-prob parameters
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t = 0

        # schedule
        lr0 = 0.1 / np.sqrt(n)
        top_k = max(3, n // 1000)   # start with few top indices
        decay_iter = 50000

        while time.time() < deadline:
            # compute conv
            conv = _conv_fft(_logits_to_a(z))
            # compute score
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = _logits_to_a(z).tolist()
                print(f"[n={n}] New best {best_score:.6f}")

            # pick indices for gradient
            if top_k == 1:
                ks = np.array([int(np.argmax(conv))])
            else:
                ks = np.argpartition(conv, -top_k)[-top_k:]

            # compute gradient w.r.t a
            a_curr = _logits_to_a(z)
            grad_a = _gradient_wrt_a(a_curr, ks)

            # gradient for logits (softmax)
            dot = a_curr.dot(grad_a)
            grad_z = a_curr * (grad_a - dot)

            # Adam update
            t += 1
            lr_t = lr0 / (1 + t * 1e-5)
            z, (m, v) = _adam_update(z, grad_z, m, v, t, lr_t)

            # occasionally jitter
            if t % 2000 == 0:
                z += np.random.randn(*z.shape) * 1e-4

            if t == decay_iter:
                top_k = 1

        # after finish for this length, set base_seq to best for next length
        if best_seq is not None:
            base_seq = np.array(best_seq[:], dtype=np.float64)

    # fallback if still none
    if best_seq is None:
        n = 500
        a = np.full(n, 1.0/n)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)

    # final evaluation
    final_score = evaluate_sequence(best_seq)
    print(f"Final best {final_score:.6f} length {len(best_seq)}")
    return best_seq
```

The function `propose_candidate` builds a non‑negative step‑function that minimizes  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` is the linear convolution of `a` with itself.  
The score is *scale‑invariant*, therefore we enforce `∑a_i = 1` and work with a *soft‑max* (logit) parameterisation

```
a_i = exp(z_i) / Σ_j exp(z_j)   ,   a_i ≥ 0 ,   Σ a_i = 1 .
```

The optimisation proceeds in three stages:

1. **Multi‑scale warm‑start** – we start with a short sequence (≈ 800 elements), optimise it,
   then linearly interpolate the best obtained shape to a longer length and continue
   optimisation.  This gives a good initial shape for large `n` without spending too much time
   on a random start in a huge space.

2. **Adam‑based sub‑gradient descent** – the non‑differentiable `max` is replaced by a
   *partial‑max* sub‑gradient built from the `k` largest convolution entries (`k` is
   decreasing during the run).  Adam updates are performed in the log‑space `z`.  
   The learning‑rate, the number of active indices and a small jitter are scheduled so
   that early iterations explore widely and later iterations focus on the true maximum.

3. **Fast simulated‑annealing refinement** – after the gradient phase we spend the
   remaining time performing cheap mass‑transfer moves (move a tiny amount of weight from
   one bin to another) accepted with a decreasing temperature.  This helps escape the
   final “flat” region that the sub‑gradient descent may get stuck in.

All heavy work (convolution, gradient) is carried out with FFT‑based `np.fft` which
runs in `O(n log n)`.  The whole routine respects the 1000 s budget and prints
progress whenever a new global best is found.

```python
"""
Proposes a step‑function (list of non‑negative heights) that minimises
the evaluation score defined by `evaluate_sequence`.

The algorithm consists of three phases:
  1. Multi‑scale warm‑start (short → long sequences).
  2. Adam optimisation on the log‑soft‑max parametrisation using a
     sub‑gradient built from the several largest convolution entries.
  3. Final simulated‑annealing refinement based on tiny mass transfers.

All heavy operations use FFT‑based convolution, and the routine stops
gracefully before the given time budget expires.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two not smaller than ``x``."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·n‑1) using FFT.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Non‑negative vector, assumed to sum to 1.

    Returns
    -------
    np.ndarray, shape (2·n‑1,)
        Full convolution.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A                     # |A|²  → convolution of a with itself
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv_full


def _score_from_a(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(conv) for a vector with Σ a_i = 1.
    This equals `evaluate_sequence` up to the scaling‑invariant normalisation.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max conversion."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _adam_update(z: np.ndarray,
                 grad_z: np.ndarray,
                 m: np.ndarray,
                 v: np.ndarray,
                 t: int,
                 lr: float,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> Tuple[np.ndarray, Tuple[np.ndarray, np.ndarray]]:
    """
    Classic Adam update (returns new ``z`` and updated moments).
    """
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, (m, v)


def _gradient_wrt_a(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (partial) maximum of the convolution w.r.t. ``a``.
    For every active offset k we add the contribution
        ∂ (a*a)_k / ∂ a_i = a_{k-i} + a_{i-k}
    (the second term is the symmetric counterpart).

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector.
    top_idxs : np.ndarray
        Indices of the convolution entries that are currently active.

    Returns
    -------
    np.ndarray, shape (n,)
        Gradient w.r.t. ``a`` (not yet transformed to the log‑space).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    for k in top_idxs:
        # admissible i satisfy 0 ≤ i < n and 0 ≤ k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]

    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of
    Wang & Carreira‑Perpiñán (2013).  Used only for a possible fall‑back
    (the main optimisation works in the log‑space).
    """
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Local (simulated‑annealing) refinement
# ----------------------------------------------------------------------


def _local_refine(a: np.ndarray,
                  time_budget: float,
                  current_best_score: float) -> Tuple[np.ndarray, float]:
    """
    Very cheap stochastic refinement: repeatedly move a tiny amount of mass
    from a random index i to a random index j and accept the move either
    if it improves the objective or with a Metropolis probability that
    shrinks according to a linear temperature schedule.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Starting distribution (already normalised).
    time_budget : float
        Seconds available for the refinement.
    current_best_score : float
        Score of the input ``a`` (used to seed the temperature schedule).

    Returns
    -------
    (a_best, best_score) : Tuple[np.ndarray, float]
        The best distribution found during the refinement and its score.
    """
    start = time.time()
    n = a.shape[0]
    a_best = a.copy()
    best_score = current_best_score

    # Temperature schedule: start relatively hot, end close to zero.
    T_start = 0.15
    T_end = 0.001

    while time.time() - start < time_budget:
        # linear interpolation of temperature
        frac = (time.time() - start) / time_budget
        T = (1.0 - frac) * T_start + frac * T_end

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # propose a mass transfer; δ is at most a few percent of total mass
        delta = np.random.rand() * 0.010  # up to 1 % of total weight
        if a_best[i] < delta:
            delta = a_best[i]          # keep non‑negative

        cand = a_best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = _score_from_a(cand)

        # Metropolis acceptance
        if cand_score < best_score or np.random.rand() < np.exp((best_score - cand_score) / max(T, 1e-12)):
            a_best = cand
            best_score = cand_score

    return a_best, best_score

    """

    best_score = float("inf")

    # ------------------------------------------------------------------
    base_seq = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.array(height_sequence_1, dtype=np.float64)
            tmp = np.maximum(tmp, 0.0)
            if tmp.sum() > 0:
                base_seq = tmp / tmp.sum()
                print("[init] Loaded height_sequence_1")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Length schedule – we grow the problem size progressively.
    # ------------------------------------------------------------------
    lengths = [800, 1600, 3200, 4800]        # feel free to adjust
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length n = {n} ===")

        # --------------------------------------------------------------
        # Initialise a (logits) – either upscale a previous best or start
        # from a mildly perturbed uniform distribution.
        # --------------------------------------------------------------
        if base_seq is not None and base_seq.size != n:
            # linear interpolation of the previous best shape
            x_old = np.linspace(0.0, 1.0, base_seq.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, base_seq)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0.sum()
        else:
            # uniform + tiny Gaussian noise
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
            a0 += 0.005 * np.random.randn(n)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()

        # log‑space representation (soft‑max)
        z = np.log(a0 + 1e-15)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t = 0

        # --------------------------------------------------------------
        # Hyper‑parameters that scale with n
        # --------------------------------------------------------------
        lr0 = 0.12 / np.sqrt(n)                 # baseline learning rate
        top_k = max(5, n // 400)                # start with several active offsets
        decay_iter = 80000                      # after this we focus on the true max

        # --------------------------------------------------------------
        # Main optimisation loop (Adam + sub‑gradient)
        # --------------------------------------------------------------
        while time.time() < deadline and t < 200000:
            a = _logits_to_a(z)                 # always satisfies Σa = 1
            conv = _conv_fft(a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep the best ever seen
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[n={n}] New global best {best_score:.6f}")

            # ----------------------------------------------------------
            # Choose the active convolution indices.
            #   * early on we take a few of the largest entries (`top_k`);
            #   * later we shrink to the single true maximum.
            # ----------------------------------------------------------
            if top_k == 1:
                ks = np.array([int(np.argmax(conv))], dtype=int)
            else:
                # use a tolerance‑based selection to keep a consistent batch size
                thresh = cur_max - max(1e-7, 0.001 * cur_max)
                cand = np.where(conv >= thresh)[0]
                if cand.size > top_k:
                    ks = np.random.choice(cand, size=top_k, replace=False)
                else:
                    ks = cand
                if ks.size == 0:                 # fallback – should never happen
                    ks = np.array([int(np.argmax(conv))], dtype=int)

            # ----------------------------------------------------------
            # Sub‑gradient in the original space and chain‑rule to logits.
            # ----------------------------------------------------------
            grad_a = _gradient_wrt_a(a, ks)
            dot = a.dot(grad_a)
            grad_z = a * (grad_a - dot)          # dL/dz = Jᵀ·∇_a L  (soft‑max Jacobian)

            # ----------------------------------------------------------
            # Adam update (step size decays very slowly)
            # ----------------------------------------------------------
            t += 1
            lr_t = lr0 / (1.0 + t * 1e-5)
            z, (m, v) = _adam_update(z, grad_z, m, v, t, lr_t)

            # occasional jitter to avoid pathological plateaus
            if t % 2500 == 0:
                z += np.random.randn(*z.shape) * 4e-4

            # after a predefined number of steps we tighten the focus to the exact max
            if t == decay_iter:
                top_k = 1

        # Store the best shape to upscale for the next length
        if best_seq is not None:
            base_seq = np.array(best_seq, dtype=np.float64)

    # ------------------------------------------------------------------
    # Final refinement with a cheap simulated‑annealing style local search.
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 30.0:                     # ensure we have enough time left
            a_start = np.array(best_seq, dtype=np.float64)
            a_refined, refined_score = _local_refine(a_start, remaining * 0.9, best_score)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = a_refined.tolist()
                print(f"[refine] Improved final best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a_fallback = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a_fallback.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the *official* evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"Final best official score {final_score:.6f} (length {len(best_seq)})")
    except Exception as exc:  # pragma: no cover
        print(f"Warning: official evaluation failed ({exc})")
        final_score = None

    return best_seq
```
