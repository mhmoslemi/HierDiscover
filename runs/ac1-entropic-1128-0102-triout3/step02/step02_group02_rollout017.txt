def proj_simplex(v):
    # v: np.array
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The search combines three cheap ideas:

1. **Adam on a soft‑max parametrisation** – a smooth way to stay on the
   probability simplex while following a sub‑gradient of the maximum
   convolution entry.  The gradient is built from the *top‑k* convolution
   entries, starting with a modest k (≈ 3–5) and shrinking to 1 after a
   while so that the optimiser first sees a smoother landscape and then
   concentrates on the true maximum.

2. **Random restarts** – each length is tried several times with independent
   random initial logits.  Because the problem is highly non‑convex, restarts
   give a good chance to escape poor local minima.

3. **Cheap random sampling** – after the gradient runs a batch of Dirichlet
   (uniform‑on‑the‑simplex) candidates is generated and scored with a fast
   FFT based convolution.  Occasionally a random candidate beats the best so
   far.

The routine adapts the learning rate, adds small jitter from time to time
and respects a global time budget (default 1000 s).  The best sequence found
is finally checked with the official ``evaluate_sequence`` function and
returned.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A                     # |A|²  (since a is real)
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv_full


def _gradient_wrt_a(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the sum of the convolution entries indexed by *top_idxs*.
    For each selected offset k we add the contribution
        ∂ (a*a)_k / ∂ a_i = a_{k-i} + a_{i-k}
    which reduces to 2·a_{k-i} for admissible i.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in top_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]
    return grad


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _adam_update(
    z: np.ndarray,
    grad_z: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, tuple[np.ndarray, np.ndarray]]:
    """Classic Adam step."""
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, (m, v)

# Main optimisation routine
# ----------------------------------------------------------------------

    best sequence discovered.
    """

    deadline = time.time() + budget_s - 5.0  # 5 s safety margin

        # hyper‑parameters that scale with n

                    ks = np.argpartition(conv, -top_k)[-top_k:]

                # sub‑gradient w.r.t. a and then chain‑rule to logits
                grad_a = _gradient_wrt_a(a, ks)
                dot = a.dot(grad_a)
                grad_z = a * (grad_a - dot)

                # Adam update
                t += 1
                lr_t = lr0 / (1.0 + t * 1e-5)   # very slow decay
                z, (m, v) = _adam_update(z, grad_z, m, v, t, lr_t)

                # occasional jitter to shake out of plateaus
                if t % 2000 == 0:
                    z += np.random.randn(n) * 2e-3

                # bring top_k down to 1 once we have “cooled”
                if t == decay_iter:
                    top_k = 1

        # ------------------------------------------------------------------
        # 3) Pure random sampling on the simplex (very cheap)
        # ------------------------------------------------------------------
        # Number of random candidates is chosen adaptively – we try a few
        # thousand but stop early if the time budget is exhausted.
        random_trials = 3000
        for _ in range(random_trials):
            if time.time() >= deadline:
                break
            # Dirichlet(1) = exponential normalised
            a = np.random.exponential(scale=1.0, size=n)
            a /= a.sum()
            cur_score = 2.0 * n * _conv_fft(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[{n} rnd] New global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final sanity check (official evaluation)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"Final best official score {final_score:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print("Warning: official evaluation failed:", exc)

    return best_seq
