"""
Propose a step‑function using a soft‑max smoothed sub‑gradient descent with
Adam updates on the probability simplex.  The algorithm works in stages:
1. initialise from the previously discovered construction (or a random
   uniform vector) and normalise it to sum‑1;
2. for a sequence length n perform many Adam steps where the gradient of the
   smooth approximation  f(a)=∑_k w_k·(a*a)[k]  (weights w_k are a
   soft‑max over the auto‑convolution values) is computed efficiently by
   FFT‑based cross‑correlation;
3. periodically jitter the iterate to escape shallow local minima;
4. after a fixed budget up‑sample the current best shape (linear interpolation)
   to a larger length and repeat the optimisation; the final best sequence is
   returned.  The whole routine respects the 1000 s time budget and prints
   intermediate improvements.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Return the evaluation score for a probability vector a (sum(a)=1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _gradient_softmax(a: np.ndarray, alpha: float):
    """
    Compute a smoothed sub‑gradient of the objective.
    Returns (g, max_conv) where g_i = 2 * Σ_k w_k a_{k-i}
    and w_k = softmax(alpha * (conv_k - max_conv)).
    """
    n = a.shape[0]

    # ----- auto‑convolution -----
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()

    # ----- soft‑max weights over the convolution entries -----
    # shift by max_val for numerical stability (max weight = 1)
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # ----- gradient = 2 * cross‑correlation(a, w) -----
    # cross‑correlation a ⋆ w can be obtained as convolution with w reversed
    w_rev = w[::-1]
    Lg = 1 << ((n + w.shape[0] - 1).bit_length())
    Ag = np.fft.rfft(a, n=Lg)
    Wg = np.fft.rfft(w_rev, n=Lg)
    corr = np.fft.irfft(Ag * Wg, n=Lg)[: n + w.shape[0] - 1]

    # indices that give the cross‑correlation at shifts 0 … n‑1
    start = w.shape[0] - 1                # = 2*n-2
    g = 2.0 * corr[start - np.arange(n)]

    return g, max_val


def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   max_iters: int | None = None):
    """
    Adam optimisation on the simplex using the soft‑max gradient.
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _score_of_seq(best_a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.20                     # base learning rate (tuned empirically)

    # soft‑max schedule
    alpha = 5.0
    alpha_max = 200.0

    last_improve = 0
    it = 0

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1

        # gradient of the smoothed objective
        g, cur_max = _gradient_softmax(a, alpha)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected Adam step
        a = _project_simplex(a - step)

        # occasional random jitter to escape plateaus
        if it % 2500 == 0:
            jitter = np.random.normal(scale=1e-4, size=n)
            a = _project_simplex(a + jitter)

        # evaluation of the true (non‑smoothed) objective
        cur_score = 2.0 * n * cur_max
        if cur_score + 1e-12 < best_score:          # strict improvement
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"n={n:4d} iter={it:6d} score={cur_score:.6f}")

        # increase the soft‑max sharpness slowly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.001)

        # early stop if stagnated for many iterations
        if it - last_improve > 15000:
            break

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step function.
    The routine runs for at most ``budget_s`` seconds and prints intermediate
    improvements.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5     # safety margin

    # ------------------------------------------------------------------
    # 1. initialise
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # may be provided
    except Exception:
        init = np.random.rand(1200)

    init = np.maximum(init, 0.0)
    s = init.sum()
    if s == 0.0:
        init = np.full_like(init, 1.0 / init.shape[0])
    else:
        init = init / s

    best_seq = init.copy()
    best_score = _score_of_seq(best_seq)
    print(f"[{time.time() - start:6.2f}s] Initial score: n={len(best_seq)} "
          f"score={best_score:.6f}")

    cur_seq = best_seq.copy()
    cur_n = len(cur_seq)

    # ------------------------------------------------------------------
    # 2. progressive up‑sampling stages
    # ------------------------------------------------------------------
    # target lengths (≈1.5× each stage, capped at 6000)
    target_lengths = []
    while True:
        nxt = int(cur_n * 1.5)
        if nxt <= cur_n:
            break
        if nxt > 6000:
            nxt = 6000
        if nxt not in target_lengths:
            target_lengths.append(nxt)
        if nxt == 6000:
            break
        cur_n = nxt
    # ensure at least one stage (the original length)
    cur_n = len(cur_seq)
    if not target_lengths:
        target_lengths = [cur_n]

    for nxt in target_lengths:
        if time.time() >= deadline:
            break

        # up‑sample current shape to the new length
        x_old = np.linspace(0.0, 1.0, cur_seq.shape[0])
        x_new = np.linspace(0.0, 1.0, nxt)
        a = np.interp(x_new, x_old, cur_seq)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(nxt, 1.0 / nxt)
        else:
            a = a / a_sum

        # allocate a per‑stage deadline (roughly equal share of remaining time)
        remaining = deadline - time.time()
        # give at most 70 % of the remaining time to the current stage
        stage_deadline = time.time() + min(remaining * 0.7, remaining - 0.5)

        # run Adam optimisation for this length
        a_opt, score_opt = _adam_optimize(a, start, stage_deadline)

        # keep global best
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start:6.2f}s] ** Global best ** "
                  f"n={nxt} score={best_score:.6f}")

        cur_seq = a_opt
        cur_n = nxt

    # ------------------------------------------------------------------
    # 3. final short refinement on the best length found
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        a_refined, score_refined = _adam_optimize(best_seq, start,
                                                  final_deadline,
                                                  max_iters=10000)
        if score_refined < best_score:
            best_seq = a_refined.copy()
            best_score = score_refined
            print(f"[{time.time() - start:6.2f}s] Final refinement "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. return the best sequence (as plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
