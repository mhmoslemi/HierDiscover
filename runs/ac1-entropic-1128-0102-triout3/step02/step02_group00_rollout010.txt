def grad_logsumexp(a, beta):
    conv = _conv_self(a)
    # compute softmax weights
    max_c = conv.max()
    # exponentiate stable
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / exp_term.sum()
    # compute gradient via cross-correlation
    rev_w = w[::-1]
    n = a.shape[0]
    L = 1 << ((3*n - 1).bit_length())
    fft_a = np.fft.rfft(a, L)
    fft_rev_w = np.fft.rfft(rev_w, L)
    conv_rev = np.fft.irfft(fft_a * fft_rev_w, L)
    # extract segment
    seg = conv_rev[n-1:n-1 + n]
    grad = 2 * seg[::-1]  # shape n
    return conv, grad, w

"""
Smooth‐max Adam optimiser on the simplex + final annealing.

The objective to minimise (scale‑invariant) is
    2·n·max_k (a*a)[k]   with   a_i ≥ 0,  Σ a_i = 1 .
Instead of the non‑smooth max we minimise a smoothed version
    f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
which converges to the true max when β→∞.
The gradient of f_β can be written in closed form using a
cross‑correlation and is computed by FFT (O(n log n)).
An Adam update is applied and the iterate is projected onto the
probability simplex after every step.
A geometric schedule raises β from a modest start to a large final
value, allowing the optimiser to first shape the distribution
coarsely and then to focus on the actual peak.
After the Adam phase a cheap simulated‑annealing move‑ε‑transfer
refines the result.
The procedure is run on a schedule of growing lengths, each time
upsampling the best sequence found so far.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution a * a (full, not circular) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _grad_logsumexp(a: np.ndarray, beta: float):
    """
    Returns the convolution (a*a), the gradient of
    f_β(a) = (1/β)·log Σ exp(β·conv), and the softmax weights.
    """
    n = a.shape[0]
    conv = _conv_self(a)                         # length 2n‑1
    max_c = conv.max()
    # stable softmax on β·conv
    shifted = conv - max_c
    exp_term = np.exp(beta * shifted)
    w = exp_term / exp_term.sum()                # length 2n‑1, Σw = 1

    # gradient = 2·(a ⋆ w) where ⋆ is cross‑correlation.
    # Compute via convolution with the reversed weight vector.
    rev_w = w[::-1]
    L = 1 << ((3 * n - 1).bit_length())          # power of two ≥ 3n‑2
    fft_a = np.fft.rfft(a, L)
    fft_rev_w = np.fft.rfft(rev_w, L)
    conv_rev = np.fft.irfft(fft_a * fft_rev_w, L)  # length L
    # segment that corresponds to lags 0 … n‑1
    seg = conv_rev[n - 1 : n - 1 + n]
    grad = 2.0 * seg[::-1]                       # reverse to obtain g_j
    return conv, grad, w


# ----------------------------------------------------------------------
# Adam optimiser on the simplex using the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    beta_start: float = 1.0,
    beta_end: float = 2000.0,
    max_iters: int = 12000,
    lr0: float = 0.02,
):
    """Run projected Adam on the smoothed objective, return best (a, score)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = a.copy()
    best_score = float("inf")

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # geometric schedule for β
        beta = beta_start * (beta_end / beta_start) ** ((t - 1) / (max_iters - 1))

        conv, grad, _ = _grad_logsumexp(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 / math.sqrt(t)          # diminishing learning rate
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

        # every 400 steps evaluate the true (non‑smoothed) score
        if t % 400 == 0:
            true_score = 2.0 * n * conv.max()
            if true_score < best_score:
                best_score = true_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] "
                    f"Adam improve: n={n}, iter={t}, β={beta:.1f}, score={true_score:.6f}"
                )

    # final check
    conv = _conv_self(a)
    final_score = 2.0 * n * conv.max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (move ε from i to j)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 25000,
):
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"Annealing improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up_sum = up.sum()
    if up_sum > 0:
        up /= up_sum
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    np.random.seed(seed)
    start_time = time.time()

    best_score = float("inf")
