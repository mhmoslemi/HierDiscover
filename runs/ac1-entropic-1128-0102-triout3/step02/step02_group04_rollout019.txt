"""
Propose a step‑function sequence that (almost) minimizes

        2 * n * max(conv(a,a)) / (sum(a)**2)

The routine consists of two stages:

* **Stage 1 – smooth‑max gradient descent.**  
  The L∞‑norm of the self‑convolution is replaced by a
  soft‑max (log‑sum‑exp) with a temperature β that slowly
  increases.  The exact gradient of this smooth surrogate is
  obtained by an FFT‑based correlation, after which the iterate
  is updated with an Adam‑style step and projected onto the
  probability simplex.

* **Stage 2 – linear‑programming refinement.**  
  Given the current sequence `a`, we compute `M = max(conv(a,a))`
  and solve the linear programme

        maximise   Σ b_i
        subject to b_i ≥ 0,  (a * b)_k ≤ M   for all k

  (the convolution is linear in `b`).  The optimal `b` has a
  larger L¹‑norm than `a` while respecting the same peak bound.
  After rescaling `b` to unit sum we take a small convex
  combination `a ← (1‑t)·a + t·b`.  Repeating this step (a few
  times) strictly reduces the peak and often gives a noticeable
  improvement over pure gradient descent.

Both stages respect the security checks in ``evaluate_sequence``:
all entries are non‑negative, the sum is fixed to 1 (scaling does
not affect the objective), and individual values stay well below
the 1000‑limit.

The implementation runs for the allocated time (≈ 950 s) and
returns the best sequence discovered.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def sqrt_series(n: int) -> np.ndarray:
    """Half‑binomial coefficients (C(2i,i)/4**i) normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1} (Duchi et al.)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    return np.maximum(v - theta, 0.0)


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Self‑convolution a*a using a real FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=nfft)
    full = np.fft.irfft(fa * fa, n=nfft)
    return full[: 2 * a.shape[0] - 1]


def grad_smooth_max(a: np.ndarray, beta: float, nfft: int):
    """
    Gradient of the smooth maximum of a*a.
    Returns (full convolution array, gradient w.r.t. a, current peak).
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, n=nfft)
    conv_full = np.fft.irfft(fa * fa, n=nfft)
    conv = conv_full[: 2 * n - 1]
    peak = conv.max()

    # soft‑max weights (numerically stable)
    w = np.exp(beta * (conv - peak))
    w /= w.sum()

    fw = np.fft.rfft(w, n=nfft)
    corr_full = np.fft.irfft(fw * np.conj(fa), n=nfft)

    # valid part of the correlation gives the gradient
    start = n - 1
    grad = 2.0 * corr_full[start : start + n]      # length n
    return conv, grad, peak

    # problem size – 2500 gives a good trade‑off between quality and LP size
    n = 2500

    # initialise from the classic half‑binomial or from a provided hint
    if 'height_sequence_1' in globals():
        hs = globals()['height_sequence_1']
        if isinstance(hs, list) and len(hs) == n:
            a = np.maximum(np.array(hs, dtype=np.float64), 0.0)
            a /= a.sum()
        else:
            a = sqrt_series(n)
    else:
        a = sqrt_series(n)

    best_a = a.copy()
    conv0 = conv_fft(a, next_pow2(3 * n))
    best_score = 2.0 * n * conv0.max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # ------- Stage 1 : smooth‑max Adam gradient descent -------------
    max_iter = 2_000_000          # safety cap – iteration count limited by time
    lr = 0.06                     # initial learning rate for Adam
    beta_start = 0.5
    beta_end = 150.0
    beta = beta_start
    beta_growth = 1.001           # exponential increase of β
    eps = 1e-8
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    pow1, pow2 = 1.0, 1.0

    nfft = next_pow2(3 * n)       # large enough for both convolution and correlation
    deadline = time.time() + budget_s - 10.0   # keep a few seconds margin
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iter:
        # compute (smooth) gradient
        conv, grad, peak = grad_smooth_max(a, beta, nfft)
        cur_score = 2.0 * n * peak

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[grad {it:07d}] new best score = {best_score:.6f}")

        # Adam update (step = -lr * m_hat / (sqrt(v_hat)+eps))
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        pow1 *= beta1
        pow2 *= beta2
        m_hat = m / (1 - pow1)
        v_hat = v / (1 - pow2)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        a_trial = a + step

        # occasional tiny stochastic kick
        if rng.random() < 0.005:
            a_trial += (rng.random(n) - 0.5) * 1e-4

        # project back onto the simplex (sum=1, non‑negative)
        a_trial = project_onto_simplex(a_trial)

        # evaluate the trial point (fast convolution)
        trial_peak = conv_fft(a_trial, nfft).max()
        trial_score = 2.0 * n * trial_peak

        # Metropolis‑style acceptance
        if trial_score <= cur_score:
            a = a_trial
        else:
            # logarithmic cooling
            temperature = max(1e-4, 0.02 * (1.0 - it / max_iter))
            if rng.random() < math.exp(-(trial_score - cur_score) / temperature):
                a = a_trial

        # schedule updates
        beta = min(beta * beta_growth, beta_end)
        if (it + 1) % 5000 == 0:
            lr = max(lr * 0.95, 1e-4)

        # occasional restart if stuck
        if it - last_improve > 200_000:
            a = rng.dirichlet(np.full(n, 0.8))
            a = project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            pow1, pow2 = 1.0, 1.0
            last_improve = it
            print(f"[grad {it:07d}] restart (random Dirichlet)")

        it += 1

    # --------------------------------------------------------------
    # ------- Stage 2 : linear‑programming refinement ----------------
    # Use the best sequence found in Stage 1 as the starting point.
    a = best_a.copy()
    A_ub = np.empty((2 * n - 1, n), dtype=np.float64)  # will be refilled each LP
    t_mix = 0.25               # convex‑combination weight
    lp_iter = 0

    while time.time() < deadline and lp_iter < 30:
        # current peak M
        M = conv_fft(a, nfft).max()

        # fill the Toeplitz‑type constraint matrix (a * b ≤ M)
        A_ub.fill(0.0)
        for i in range(n):
            A_ub[i : i + n, i] = a

        # solve   maximise Σ b_i   s.t.   A_ub @ b ≤ M, b ≥ 0
        # (linprog minimises –c·x, so we supply -1 as objective)
        res = linprog(
            -np.ones(n),
            A_ub=A_ub,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0, None)] * n,
            method="highs",
        )
        if not res.success:
            print(f"[lp {lp_iter}] linprog failed: {res.message}")
            break

        b0 = res.x
        s = b0.sum()
        if s <= 0:
            print(f"[lp {lp_iter}] zero sum returned by LP – abort")
            break
        # rescale so that the candidate also sums to 1
        g = b0 / s

        # convex combination and projection
        a_new = (1 - t_mix) * a + t_mix * g
        a_new = project_onto_simplex(a_new)

        # evaluate improvement
        new_score = 2.0 * n * conv_fft(a_new, nfft).max()
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[lp {lp_iter}] new best score = {best_score:.6f}")

        # accept the move if it decreased the peak, otherwise shrink t_mix
        if new_score < 2.0 * n * M:
            a = a_new
        else:
            t_mix = max(t_mix * 0.8, 0.01)

        lp_iter += 1

    # --------------------------------------------------------------
    # final safety evaluation (uses the provided routine)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] best reported score = {final_score:.6f}")
    return best_a.tolist()
