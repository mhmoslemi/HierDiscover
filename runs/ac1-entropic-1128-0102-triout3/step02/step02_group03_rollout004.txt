The algorithm below implements exactly this idea:

1. **Initialisation** – start from the best known sequence (`height_sequence_1`)

"""
Hybrid LP‑based optimiser for the max‑self‑convolution problem.

The algorithm follows the method from the statement:
    – keep the ℓ₁‑norm fixed to √(2 n) (so the objective equals max‑convolution)
    – repeatedly solve a linear programme that maximises the ℓ₁‑mass of a
      candidate sequence under the current convolution bound,
    – scale the LP solution back to the required norm and mix it with the
      current iterate, accepting only steps that strictly lower the maximal
      convolution entry.
The LP is solved with SciPy’s HiGHS solver; Toeplitz constraint matrix
indices are pre‑computed once, only the data vector changes each iteration.
"""
import time
from typing import List

import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = target_sum}.
    Returns a new array with the same shape as `v`.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray) -> float:
    """
    Returns the maximal entry of the linear (non‑circular) convolution a*a.
    """
    conv = np.convolve(a, a)
    return float(conv.max())


def _build_toeplitz_indices(n: int):
    """
    Pre‑computes the row and column index arrays for the Toeplitz matrix
    that maps b ↦ f∗b (linear convolution).
    The matrix has shape (2 n‑1, n) and contains exactly n² non‑zero entries.
    """
    # rows[i, j] = i + j, cols[i, j] = j
    idx_i = np.arange(n, dtype=np.int32)[:, None]          # shape (n,1)
    idx_j = np.arange(n, dtype=np.int32)[None, :]          # shape (1,n)
    rows = (idx_i + idx_j).ravel()                         # length n²
    cols = np.tile(np.arange(n, dtype=np.int32), n)        # length n²
    return rows, cols


def _solve_lp(f: np.ndarray, M: float, rows, cols) -> np.ndarray:
    """
    Solves the linear programme:
        maximise   Σ b_i
        subject to T(f)·b ≤ M·1 ,  b_i ≥ 0 ,
    where T(f) is the Toeplitz matrix of f (pre‑computed index arrays are
    supplied to avoid rebuilding them).
    Returns the optimal vector `b` (may be zero if the solver fails).
    """
    n = f.shape[0]
    # data of the sparse matrix = repeat each entry of f exactly n times
    data = np.repeat(f, n)
    A_ub = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # objective: minimise - Σ b_i  <=> maximise Σ b_i
    c = -np.ones(n, dtype=np.float64)

    # use HiGHS (fast interior‑point/ simplex implementation)
    res = opt.linprog(
        c=c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # Return an empty (zero) vector – the caller will handle the failure.
        return np.zeros_like(f)
    return res.x
