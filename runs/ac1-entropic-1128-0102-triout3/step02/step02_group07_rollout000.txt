   best = a.copy(); best_score = ...

   for it in range(max_iters):

        if time.time() > time_deadline: break

        conv = conv_fft(a); max_val = conv.max(); peak_idx = conv.argmax()

        grad = gradient_of_peak(a, peak_idx)

        # step size
        step_factor = eta * (1.0 / np.sqrt(it+1)) # or decay

        step = step_factor * grad

        step = np.minimum(step, a * 0.999) # ensure non-negative

        a_trial = a - step

        a_trial = project_onto_simplex(a_trial)

        # compute score
        conv_trial = conv_fft(a_trial); trial_score = 2 * n * conv_trial.max()

        if trial_score < best_score: best_score=trial_score; best = a_trial.copy(); print

"""
Propose a candidate sequence for ``evaluate_sequence``.

Algorithm
---------
1. **Parameterized family search** –  Generate truncated
   power‑law / exponential sequences
       a_i ∝ (i+offset)^(-p)·exp(-λ·i)
   for several values of (p,offset,λ) on a modest length (n0≈3000).
   The best vector (lowest 2·n·max(conv)) is kept as a seed.

2. **Scale up** –  Re‑generate the same parameters on a larger
   length (n≈5000) and keep the best of those as the starting point
   for the local optimiser.

3. **Projected gradient descent** –  Repeatedly:
      * compute the self‑convolution by FFT,
      * locate the K (default 5) largest convolution entries,
      * form a sub‑gradient as the average of the gradients of those
        entries (exact expression via ``gradient_of_peak``),
      * take a step opposite to the gradient with a decaying step‑size,
      * project back onto the probability simplex (non‑negative,
        sum = 1) using the O(n log n) Duchi algorithm,
      * accept the step if it improves the objective, otherwise
        accept with a small Metropolis probability (simulated‑annealing).

   The loop runs until the time budget expires (≈950 s).  The current
   global best vector is updated whenever a better objective is found.

4. **Final sanity check** –  The best vector is evaluated with the
   provided ``evaluate_sequence`` and returned as a plain ``list``.

The method combines a wide exploration of a low‑dimensional parametric
family with a fine‑grained optimisation on the full‑dimensional space,
and it works comfortably within the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def tapered_sequence(n: int, p: float, offset: float = 1.0, lam: float = 0.0) -> np.ndarray:
    """Return a non‑negative sequence

           a_i = (i + offset)^(-p) * exp(-lam * i)

       normalised so that Σ a_i = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    if lam != 0.0:
        a *= np.exp(-lam * i)
    total = a.sum()
    if total == 0.0:
        a.fill(1.0 / n)
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Self‑convolution x*x using a real FFT (length 2·len(x)-1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Exact sub‑gradient of the convolution entry ``conv[peak_idx]``."""
    n = a.shape[0]
    grad = np.zeros_like(a)

    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    if lo > hi:
        return grad
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    # each valid pair contributes 2*a[jdx] to grad[idx]
    grad[idx] = 2.0 * a[jdx]
    return grad


def combined_gradient(a: np.ndarray, conv: np.ndarray, top_k: int = 5) -> np.ndarray:
    """Average gradient of the top_k convolution entries."""
    # indices of the largest entries (ties are fine)
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        # ``argpartition`` gives unsorted top_k indices efficiently
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    for idx in idxs:
        grad += gradient_of_peak(a, int(idx))
    return grad / float(len(idxs))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative sequence minimising ``evaluate_sequence``.
    The routine performs a parametric sweep followed by a projected
    gradient descent with occasional simulated‑annealing steps.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    # two lengths: a quick sweep on a smaller array, then a full run.
    n_small = 3000
    n_full = 5000

    # parametric families
    p_vals = np.arange(0.20, 0.81, 0.04)          # exponent
    offsets = [0.2, 0.5, 1.0, 2.0, 5.0]          # additive shift
    lam_vals = [0.0, 0.0002, 0.0005, 0.001, 0.002]  # exponential decay

    # gradient descent hyper‑parameters
    eta0 = 0.12                # initial step‑size multiplier
    decay_factor = 0.97        # factor applied every ``decay_every`` steps
    decay_every = 2000
    temperature0 = 0.02        # Metropolis temperature
    temp_decay = 0.985
    top_k = 5                  # number of peaks to blend into the gradient
    max_iter = 10_000_000      # safety bound – loop also stops by time

    # ------------------------------------------------------------------
    # Time handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    best_vec = None
    best_score = np.inf
    best_params = None

    # ------------------------------------------------------------------
    # Phase 1 – coarse sweep on the smaller length
    # ------------------------------------------------------------------
    print("[search] phase 1 – coarse sweep (n_small =", n_small, ")")
    for lam in lam_vals:
        for offset in offsets:
            for p in p_vals:
                if time.time() > deadline:
                    break
                a = tapered_sequence(n_small, p, offset, lam)
                conv = conv_fft(a)
                score = 2.0 * n_small * conv.max()
                if score < best_score:
                    best_score = score
                    best_vec = a.copy()
                    best_params = (p, offset, lam)
                    print(f"[grid] new best: p={p:.3f}, off={offset}, λ={lam:.5f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refined sweep on the full length using the same params
    # ------------------------------------------------------------------
    if best_params is None:
        # fallback: uniform distribution (should never happen)
        best_vec = np.full(n_full, 1.0 / n_full)
        best_score = 2.0 * n_full * conv_fft(best_vec).max()
        best_params = (0.5, 1.0, 0.0)
        print("[fallback] using uniform seed")
    else:
        p0, off0, lam0 = best_params
        print("[search] phase 2 – refinement on full length (n_full =", n_full, ")")
        # explore a tiny neighbourhood around the best parameters
        p_fine = np.arange(max(0.01, p0 - 0.02), min(0.99, p0 + 0.02) + 0.001, 0.002)
        off_fine = np.array([max(0.0, off0 - 0.5), off0, off0 + 0.5])
        lam_fine = np.array([max(0.0, lam0 - 0.0002), lam0, lam0 + 0.0002])

        for lam in lam_fine:
            for offset in off_fine:
                for p in p_fine:
                    if time.time() > deadline:
                        break
                    a = tapered_sequence(n_full, p, offset, lam)
                    conv = conv_fft(a)
                    score = 2.0 * n_full * conv.max()
                    if score < best_score:
                        best_score = score
                        best_vec = a.copy()
                        best_params = (p, offset, lam)
                        print(f"[fine] new best: p={p:.4f}, off={offset:.2f}, λ={lam:.5f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected gradient descent on the full length
    # ------------------------------------------------------------------
    n = n_full
    a = best_vec.copy()
    cur_score = best_score
    eta = eta0
    temperature = temperature0
    it = 0
    last_improve_iter = 0

    print("[grad] starting projected gradient descent (n =", n, ")")
    while time.time() < deadline and it < max_iter:
        conv = conv_fft(a)
        max_val = conv.max()
        peak_idx = int(conv.argmax())

        # sub‑gradient averaged over the largest K entries
        grad = combined_gradient(a, conv, top_k=top_k)

        # step with decaying learning rate
        step = eta * grad
        # enforce non‑negativity by not removing more than 99 % of any entry
        step = np.minimum(step, a * 0.99)
        a_trial = a - step

        # occasional random jitter (helps to escape flat plateaus)
        if rng.random() < 0.007:
            jitter = (rng.random(n) - 0.5) * 2e-5
            a_trial += jitter

        # projection onto the probability simplex
        a_trial = project_onto_simplex(a_trial)

        # evaluate trial
        conv_trial = conv_fft(a_trial)
        trial_score = 2.0 * n * conv_trial.max()

        # global best update
        if trial_score < best_score:
            best_score = trial_score
            best_vec = a_trial.copy()
            last_improve_iter = it
            print(f"[iter {it:08d}] new global best score = {best_score:.6f}")

        # decide acceptance
        if trial_score <= cur_score:
            a = a_trial
            cur_score = trial_score
        else:
            # Metropolis acceptance
            delta = trial_score - cur_score
            if temperature > 0.0 and rng.random() < np.exp(-delta / temperature):
                a = a_trial
                cur_score = trial_score
            # otherwise reject

        # schedule updates
        if (it + 1) % decay_every == 0:
            eta = max(eta * decay_factor, 1e-6)
            temperature = max(temperature * temp_decay, 1e-9)

        # occasional restart if stagnation for many iterations
        if it - last_improve_iter > 250_000 and it > 0:
            # perturb the current best slightly and continue
            noise = rng.normal(scale=0.01, size=n)
            a = project_onto_simplex(best_vec + noise)
            cur_score = 2.0 * n * conv_fft(a).max()
            last_improve_iter = it
            print(f"[restart @ {it}] perturbed best for escape")

        it += 1

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"[done] total iterations={it}, final reported score={final_score:.6f}")
    if best_params is not None:
        p, off, lam = best_params
        print(f"[info] best parameters after search: p={p:.5f}, offset={off}, lambda={lam}")

    return best_vec.tolist()
