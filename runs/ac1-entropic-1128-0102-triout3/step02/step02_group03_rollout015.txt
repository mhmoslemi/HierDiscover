"""Hybrid LP‑based optimiser for the step‑function max‑convolution problem.

The routine keeps a non‑negative vector `a` on the ℓ₁‑simplex
{a ≥ 0, Σ a = √(2 n)}.  In each outer iteration it

1. computes the current maximal self‑convolution entry `M`,
2. solves a linear programme that **maximises the total mass** that can be
   placed in a new vector `b` while keeping the convolution `a*b` bounded by `M`,
3. rescales `b` to the correct ℓ₁‑norm, mixes it with the current iterate,
   and projects back onto the simplex,
4. occasionally performs cheap random mass‑transfer moves and full Dirichlet
   restarts to avoid stagnation.

The LP step moves mass from entries that contribute heavily to the peak
towards positions that have little influence, leading to a rapid reduction of
the maximal convolution value.  The algorithm stops when the 1000 s budget is
exhausted or a target score (< 1.503) is reached.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length `nfft`."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2 n‑1) × n matrix `A` that maps `b` to the linear convolution
    `a * b`.  The k‑th row of `A` contains the coefficients a_i that appear in
    (a*b)[k].
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=a.dtype)
    # column j corresponds to a shifted copy of `a`
    for j in range(n):
        A[j : j + n, j] = a
    return A


def _lp_maxsum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise    Σ b_j
        subject to  a * b ≤ M   (componentwise)
                    b ≥ 0
    and return the optimal `b`.  If the LP fails, ``None`` is returned.
    """
    n = a.size
    A = _build_conv_matrix(a)           # shape (2n‑1, n)
    rows = A.shape[0]

    # upper‑bound for each convolution entry
    b_ub = np.full(rows, M, dtype=np.float64)

    # minimise -Σ b_j  <=> maximise Σ b_j
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n

    # SciPy’s HiGHS backend is fast and robust for large dense LPs
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={"presolve": True})

    if res.success:
        return np.maximum(res.x, 0.0)     # guard against tiny negatives
    return None


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function sequence that minimises
    `evaluate_sequence`.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # spare a little margin

    # ------------------------------------------------------------------
    # problem size, target ℓ₁‑norm and FFT length
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2401                # a reasonably large size (≈ 2 500)
        a0 = None

    S = np.sqrt(2.0 * n)        # ℓ₁‑norm we keep fixed
    nfft = _next_pow2(2 * n - 1)

    # initialise the iterate
    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        a = np.full(n, S / n, dtype=np.float64)   # uniform start

    # bookkeeping
    best_a = a.copy()
    best_score = float('inf')
    iteration = 0
    no_improve = 0
    max_no_improve = 2500          # restart after this many idle steps

    # mixing coefficient (how much of the LP solution we take)
    mix = 0.9
    mix_decay = 0.9995

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --- current objective -------------------------------------------------
        conv = _convolve_self_fft(a, nfft)

        cur_score = cur_max                     # because Σa = √(2 n)

        # keep global best

            print(f"[{iteration:06d}] New best: {best_score:.6f} (max={cur_max:.6f})")

        # ------------------------------------------------------------------
        # LP‑based improvement step

        b = _lp_maxsum(a, cur_max)

        if b is not None:

            if sum_b > 0:
