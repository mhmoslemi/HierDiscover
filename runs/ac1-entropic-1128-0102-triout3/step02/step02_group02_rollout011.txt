"""
Propose a step‑function sequence using a smooth approximation to the
max‑convolution objective and Adam optimisation in the soft‑max (logit)
parameterisation.

Key ideas
---------
* Represent a non‑negative sequence a of length n as a soft‑max of logits
  z: a_i = exp(z_i) / Σ_j exp(z_j).  This guarantees a_i ≥ 0 and Σ a_i = 1,
  which makes the evaluation score equal to 2·n·max(conv(a,a)).
* Approximate the non‑differentiable max by the smooth log‑sum‑exp:
      M_α(a) = (1/α)·log Σ_k exp(α·c_k) ,   c_k = (a*a)_k .
  The gradient of M_α is a weighted sum of the gradients of each convolution
  entry, where the weights w_k = exp(α·c_k)/Σ exp(α·c_j) form a probability
  distribution.
* The gradient of a single convolution entry (a*a)_k with respect to a is
  guaran­ted to be a simple Toeplitz‑convolution:
      ∂(a*a)_k/∂a = 2·a shifted by k.
  Consequently the gradient of the smoothed max can be obtained with **one**
  FFT‑based convolution:
      grad_a = 2·conv_rev[(2·n‑2)‑np.arange(n)]
  where conv_rev = a * w_rev (linear convolution) and w_rev is w reversed.
* An Adam optimiser updates the logits z.  The temperature α is increased
  slowly (annealed) so the optimiser first sees a smooth landscape and later
  focuses on the true maximum.
* Several lengths (up to a few thousand) and a few random restarts are
  tried within the time budget.  The best sequence encountered is returned.

The routine runs for at most ``budget_s`` seconds (default 1000) and never
exceeds the allowed API – it only calls the provided ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """
    Return the smallest power of two that is >= x.
    """
    return 1 << (x - 1).bit_length()


def _conv_fft_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    n, m = a.shape[0], b.shape[0]
    L = n + m - 1
    N = _next_pow2(L)
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)[:L]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """
    Numerically stable soft‑max (returns a probability vector that sums to 1).
    """
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _grad_wrt_a(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed‑max objective w.r.t. the sequence a.

    Parameters
    ----------
    a      : current sequence (Σ a = 1, a ≥ 0)
    conv   : linear convolution a*a (length 2·n‑1)
    alpha  : temperature of the log‑sum‑exp approximation

    Returns
    -------
    grad_a : ndarray of shape (n,) – ∂M_α/∂a
    """
    # ----- compute the soft‑max weights for each convolution entry -----
    # stability: shift by the maximum before exponentiation
    max_c = np.max(conv)
    shifted = conv - max_c                # prevents overflow
    w = np.exp(alpha * shifted)
    w /= w.sum()                          # probability distribution over offsets

    # ----- gradient through the weighted sum of convolutions ----------
    # The gradient of Σ_k w_k (a*a)_k  w.r.t. a is:
    #   2 * Σ_k w_k * (a shifted by +k)
    # This can be computed as a single convolution with the reversed weight vector.
    w_rev = w[::-1]
    conv_rev = _conv_fft_full(w_rev, a)   # length 3·n‑2
    L = conv.shape[0]                     # = 2·n‑1
    idx = (L - 1) - np.arange(a.shape[0])  # indices (2·n‑2‑i) for i=0…n‑1
    grad_a = 2.0 * conv_rev[idx]
    return grad_a


def _adam_step(
    z: np.ndarray,
    grad_z: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform a single Adam update on the logits.
    Returns the updated (z, m, v).
    """
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, m, v

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation score.

    Parameters
    ----------
    seed       : random seed for reproducibility.
    budget_s   : time budget in seconds (default 1000).
    **kwargs   : ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    List[float] – the best sequence discovered (list of Python floats).
    """
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    deadline = time.time() + budget_s - 2.0   # small safety margin
    best_seq: List[float] | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction exists, evaluate it once.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(height_sequence_1)
            cand_score = evaluate_sequence(cand)
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[init] Imported previous best score {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore.  Larger n gives a better potential
    # bound but also higher per‑iteration cost.
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2400, 3200]   # adaptively skipped if time runs out

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # time allocation per length (roughly equal share of remaining budget)
        remaining = deadline - time.time()
        # give at most 30 % of the remaining budget to the current length
        length_budget = min(remaining * 0.3, 200.0)

        # hyper‑parameters that scale with n
        lr0 = 0.12 / np.sqrt(n)          # base learning rate for Adam
        alpha0 = 0.5                      # start temperature for log‑sum‑exp
        alpha_max = 200.0                 # final temperature (approaches true max)

        # ------------------------------------------------------------------
        # Prepare a few different initialisations to avoid being trapped.
        # ------------------------------------------------------------------
        init_strategies = []

        # 1) Uniform (soft‑max of zeros) – a useful baseline
        init_strategies.append(("uniform", np.zeros(n)))

        # 2) Small random Gaussian noise
        init_strategies.append(("random", rng.randn(n) * 0.05))

        # 3) “U‑shape”: large values at the two ends, small in the centre
        u_shape = np.linspace(1.0, 0.0, n // 2 + 1)
        u_vals = np.concatenate([u_shape, u_shape[-2::-1]])[:n]
        init_strategies.append(("ushape", np.log(u_vals + 1e-8)))   # logits from pattern

        # 4) If we already have a good sequence, embed it (resample) as a start
        if best_seq is not None and len(best_seq) > 0:
            arr = np.array(best_seq, dtype=float)
            # simple linear interpolation to the target length
            xs_old = np.linspace(0, 1, arr.size)
            xs_new = np.linspace(0, 1, n)
            interpolated = np.interp(xs_new, xs_old, arr)
            init_strategies.append(("interpolated", np.log(interpolated + 1e-8)))

        # ------------------------------------------------------------------
        # Run a optimisation pass for each initialisation.
        # ------------------------------------------------------------------
        for tag, z0 in init_strategies:
            if time.time() >= deadline:
                break
            print(f"[{n}] start {tag} initialisation")
            z = z0.astype(np.float64).copy()
            a = _softmax(z)

            # Adam state
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            t = 0

            # temperature schedule
            alpha = alpha0

            # time budgeting for this run
            run_deadline = time.time() + length_budget

            while time.time() < run_deadline:
                # ------------------------------------------------------------
                # 1) current convolution and score
                # ------------------------------------------------------------
                conv = _conv_fft_full(a, a)          # length 2·n‑1
                cur_max = conv.max()
                cur_score = float(2.0 * n * cur_max)  # sum(a)=1 ⇒ denominator =1

                # record the best overall sequence
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.tolist()
                    print(f"[{n}/{tag}] new global best {best_score:.6f}")

                # ------------------------------------------------------------
                # 2) gradient of the smoothed‑max objective w.r.t. a
                # ------------------------------------------------------------
                grad_a = _grad_wrt_a(a, conv, alpha)

                # ------------------------------------------------------------
                # 3) back‑propagation through the soft‑max (logits)
                # ------------------------------------------------------------
                dot = a.dot(grad_a)
                grad_z = a * (grad_a - dot)   # chain rule for soft‑max

                # ------------------------------------------------------------
                # 4) Adam update in log‑space
                # ------------------------------------------------------------
                t += 1
                # a very mild decay of the learning rate helps stability
                lr = lr0 / (1.0 + t * 1e-5)
                z, m, v = _adam_step(z, grad_z, m, v, t, lr)

                # re‑compute a from updated logits
                a = _softmax(z)

                # ------------------------------------------------------------
                # 5) temperature (α) annealing – slowly focus on the true max
                # ------------------------------------------------------------
                if alpha < alpha_max:
                    # increase α by ~2 % every 500 steps (exponential schedule)
                    if t % 500 == 0:
                        alpha = min(alpha_max, alpha * 1.02)

                # ------------------------------------------------------------
                # 6) occasional random jitter to escape tiny plateaus
                # ------------------------------------------------------------
                if t % 3000 == 0:
                    z += rng.randn(n) * 1e-3

                # ------------------------------------------------------------
                # 7) optional early break if progress stalls
                # ------------------------------------------------------------
                # (here we simply keep looping until the allotted time expires)

        # end of all initialisations for length n
        print(f"[{n}] finished – best score so far {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should never happen) use uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform sequence score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional – evaluate_sequence is cheap)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"Final best official score {final_score:.6f} (length {len(best_seq)})")
    except Exception:
        print("Warning: final evaluation failed.")

    return best_seq
