"""
Multi‑scale smoothed‑gradient descent + final annealing.

* Work on the probability simplex (∑aᵢ = 1).  The objective
  2·n·max(conv) is scale‑invariant, so we keep the sum exactly 1.
* Replace the non‑smooth ℓ∞‑norm by a log‑sum‑exp surrogate.
  For λ>0  Lλ(a)=λ·log(∑exp(conv/λ)) → max(conv) as λ→0.
* The gradient of Lλ is a weighted sum of the true sub‑gradients.
  The weights are the soft‑max of the convolution entries and are
  obtained from an FFT‑based cross‑correlation (O(n log n)).
* λ is annealed during the run (initially a constant fraction of the
  current maximum convolution value, then divided by two every few
  thousand iterations).
* A small momentum term (β≈0.9) and a diminishing step size
  step₀/√t are used; after each step the vector is projected onto the
  simplex (Duchi et al. 2008).
* The optimisation proceeds on a schedule of increasing lengths.
  After each length the best distribution is up‑sampled (linear
  interpolation) and used as a warm‑start for the next length.
* When the time budget is almost exhausted a cheap simulated‑annealing
  sweep (mass transfers between two coordinates) is performed to
  polish the result.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, ∑x_i = 1}.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (real‑to‑real)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors x and y via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed gradient descent on the simplex
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute the gradient of the log‑sum‑exp surrogate.
    Parameters
    ----------
    a   : np.ndarray, shape (n,)
        Current probability vector (sum = 1).
    conv: np.ndarray, shape (2n‑1,)
        Self‑convolution a*a.
    lam  : float
        Smoothing parameter λ>0.
    Returns
    -------
    grad: np.ndarray, shape (n,)
        The (dense) gradient ∇Lλ(a).
    """
    n = a.shape[0]
    # soft‑max over convolution entries, numerically stable
    max_val = conv.max()
    # subtract max to avoid overflow
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation (weights * a_rev)
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)                # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]              # 2·∂conv/∂a weighted sum
    return grad


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent with momentum on the probability simplex.
    Returns the best vector seen and its (exact) objective value.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    # compute objective via the public evaluator for safety
    best_score = float('inf')
    beta = 0.9
    step0 = 0.8
    it = 0

    # λ‑schedule: start as a fixed fraction of the current max and halve
    # every λ_decay_interval iterations.
    lam_factor = 0.25            # initial λ = lam_factor * current max
    lam_decay = 0.5
    lam_decay_interval = 2_000   # halve every 2000 steps

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        # exact score (sum(a)=1, so 2·n·max_b)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, it={it}, score={cur_score:.6f}")

        # update smoothing parameter
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        # gradient of the surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # momentum + diminishing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing: move a small ε from i to j.
    Returns possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # ε up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Finds a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm runs for at most `budget_s` seconds
    and combines multi‑scale smoothed gradient descent with a final
    simulated‑annealing polishing step.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter (if the runner provided a previously found solution)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # length schedule – geometric growth, modest sizes to stay within the budget
    lengths = [800, 1600, 3200, 4800, 6400, 9600]
    np.random.shuffle(lengths)          # avoid systematic bias if we stop early

    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # initialise a0
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # allocate a slice of the remaining budget for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_deadline = time.time() + max(8.0, 0.13 * remaining)

        # --------------------------------------------------------------
        # smoothed gradient descent
        # --------------------------------------------------------------
        a_opt, score_opt = _run_smooth_descent(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after gradient: n={n}, score={score_opt:.6f}")

        # --------------------------------------------------------------
        # final annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, 0.04 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=30_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n}, score={score_ref:.6f}")

        # remember the best distribution for the next upscale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # return plain Python list
    return [float(x) for x in best_seq]
