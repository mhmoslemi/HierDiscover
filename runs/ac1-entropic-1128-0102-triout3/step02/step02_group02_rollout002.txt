"""
Propose a candidate step‑function sequence.

Algorithm
---------
* Parameterise the step heights with a soft‑max (logits → a) so that
  a_i ≥ 0 and Σ a_i = 1 (the evaluation is scale‑invariant for sum > 0).
* For a given length n we minimise the smooth approximation of the
  max‑convolution:
        f(a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
  When α → ∞ this coincides with the true objective.
* The gradient of the weighted sum Σ_k w_k·(a*a)_k (with
  w_k ∝ exp(α·conv_k)) can be written as
        ∇_a f = 2·corr(w, a)               (valid mode correlation)
  which we compute efficiently with FFTs.
* Adam updates are performed on the logits (z).  The temperature α is
  increased linearly from a small value (α≈0.5) to a larger one
  (α≈30–40) so the optimiser first smooths the landscape and later
  focuses on the true maximum.
* A multi‑scale schedule is employed: we start with a modest length,
  optimise it, then up‑sample the best shape to a longer length and
  continue optimisation.  This lets us obtain a good initial guess for
  large n without spending many iterations on the expensive FFTs.
* Random jitter is added occasionally to escape shallow local minima.
* The best sequence seen across all lengths (and the optional previous
  construction `height_sequence_1`) is returned.

The whole routine respects the 1000 s time budget and prints progress
information.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities – pure functions, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (full length = 2·n‑1) using FFT.
    a is assumed to be real and 1‑D.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A                     # |A|²
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv_full


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _adam_update(z: np.ndarray,
                 grad_z: np.ndarray,
                 m: np.ndarray,
                 v: np.ndarray,
                 t: int,
                 lr: float,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Classic Adam update.  Returns (z_new, m_new, v_new).
    """
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, m, v


def _grad_a_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a ( Σ_k w_k (a*a)_k ) = 2 * corr_valid(w, a).

    This uses FFT based correlation:
        corr_valid(w, a) = (w ⋆ a)[n‑1 : 2n‑1]
    where ⋆ denotes full correlation.
    """
    n = a.shape[0]
    # reverse a for correlation
    a_rev = a[::-1]

    # length of the full correlation = len(w) + n - 1 = (2n‑1)+n‑1 = 3n‑2
    L = len(w) + n - 1
    N = _next_pow2(L)

    w_pad = np.zeros(N, dtype=np.float64)
    a_rev_pad = np.zeros(N, dtype=np.float64)
    w_pad[: len(w)] = w
    a_rev_pad[: n] = a_rev

    W = np.fft.rfft(w_pad)
    A_rev = np.fft.rfft(a_rev_pad)

    # correlation = irfft(W * conj(A_rev))
    corr_full = np.fft.irfft(W * np.conj(A_rev), n=N)[:L]

    # valid part: indices n‑1 … 2n‑2 (inclusive) → slice [n‑1 : 2n‑1)
    start = n - 1
    end = start + n
    grad = 2.0 * corr_full[start:end]
    return grad


def _evaluate_score(a: np.ndarray) -> float:
    """
    Compute the official evaluation score for a vector a that already
    sums to 1 (i.e. the same as 2·n·max(conv)).
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation for a single length
# ----------------------------------------------------------------------


def _optimise_length(n: int,
                     deadline: float,
                     init_a: np.ndarray | None = None,
                     seed: int = 0) -> tuple[List[float], float]:
    """
    Optimise a sequence of length n.
    Returns (best_sequence_as_list, best_official_score).
    """
    np.random.seed(seed)

    # initialise logits
    if init_a is not None:
        # use the supplied distribution as a warm start
        a = np.maximum(0.0, init_a.astype(np.float64))
        a = a / a.sum()
        # logits = log(a) (add tiny epsilon to avoid log(0))
        z = np.log(a + 1e-16)
    else:
        # start from a small random perturbation of the uniform distribution
        z = np.random.randn(n) * 0.02

    a = _softmax(z)                     # ensures Σ a = 1
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # learning‑rate schedule (scale with sqrt(n) as a rule of thumb)
    base_lr = 0.15 / np.sqrt(n)

    # temperature (α) schedule – start smooth, end sharp
    alpha_start = 0.5
    alpha_target = 35.0
    warmup_iters = 3000                # number of iterations over which α grows
    max_iters = 200_000                # safety cap

    best_seq = a.copy()
    best_score = _evaluate_score(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        # --------------------------------------------------------------
        # 1) convolution and temperature
        # --------------------------------------------------------------
        conv = _conv_fft(a)                 # length 2n‑1, all entries ≥ 0
        # linear increase of α, then keep at α_target
        if t < warmup_iters:
            alpha = alpha_start + (alpha_target - alpha_start) * (t / warmup_iters)
        else:
            alpha = alpha_target

        # --------------------------------------------------------------
        # 2) soft‑max over convolution entries → weights w_k
        # --------------------------------------------------------------
        # To avoid overflow we subtract the max α·conv before exponentiation
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_term = np.exp(scaled - max_scaled)
        w = exp_term / exp_term.sum()        # shape (2n‑1), sums to 1

        # --------------------------------------------------------------
        # 3) gradient w.r.t. a (sub‑gradient of weighted sum)
        # --------------------------------------------------------------
        grad_a = _grad_a_from_weights(w, a)   # shape (n,)

        # --------------------------------------------------------------
        # 4) chain rule to obtain gradient w.r.t. logits (soft‑max)
        # --------------------------------------------------------------
        dot = a.dot(grad_a)
        grad_z = a * (grad_a - dot)

        # --------------------------------------------------------------
        # 5) Adam update in logit space
        # --------------------------------------------------------------
        t += 1
        lr_t = base_lr / (1.0 + t * 1e-5)    # very gentle decay
        z, m, v = _adam_update(z, grad_z, m, v, t, lr_t)

        # re‑compute probabilities
        a = _softmax(z)

        # --------------------------------------------------------------
        # 6) occasional jitter to escape plateaus
        # --------------------------------------------------------------
        if t % 1500 == 0:
            jitter = np.random.randn(n) * 5e-4
            z += jitter
            a = _softmax(z)   # keep a consistent

        # --------------------------------------------------------------
        # 7) track the best official score (evaluate_sequence)
        # --------------------------------------------------------------
        if t % 200 == 0 or t == 1:
            cur_score = _evaluate_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{n}] iter {t:6d}  score {best_score:.6f}")

    # final sanity check with the real evaluate_sequence (the wrapper may clip)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        best_score = float(final_score)
    except Exception:
        # fallback to the internally computed score
        pass

    return best_seq.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    # --------------------------------------------------------------
    # 0) start from a previously discovered construction if present
    # --------------------------------------------------------------
    best_seq: List[float] | None = None
    best_score = float("inf")
    prev_a: np.ndarray | None = None

    if "height_sequence_1" in globals():
        try:
            old = np.array(height_sequence_1, dtype=np.float64)
            old = np.maximum(0.0, old)
            if old.sum() > 0:
                old = old / old.sum()
                prev_a = old
                cand_score = evaluate_sequence(old.tolist())
                best_seq = old.tolist()
                best_score = float(cand_score)
                print(f"[init] loaded previous sequence, score {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1) multi‑scale optimisation: 800 → 1600 → 3200 → 4800 (adaptable)
    # --------------------------------------------------------------
    lengths = [800, 1600, 3200, 4800]

    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # up‑sample previous best shape if we have one
        if prev_a is not None:
            # linear interpolation on the interval [0, 1]
            x_old = np.linspace(0.0, 1.0, len(prev_a))
            x_new = np.linspace(0.0, 1.0, n)
            upsampled = np.interp(x_new, x_old, prev_a)
            # add a tiny random perturbation to avoid exact symmetry
            upsampled += np.random.randn(n) * 1e-4 * upsampled.mean()
            upsampled = np.clip(upsampled, 0.0, None)
            if upsampled.sum() == 0:
                upsampled = np.full(n, 1.0 / n)
            else:
                upsampled = upsampled / upsampled.sum()
        else:
            upsampled = None

        seq, score = _optimise_length(
            n=n,
            deadline=deadline,
            init_a=upsampled,
            seed=seed + n  # different seed per length
        )

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[global] New overall best {best_score:.6f} (len={n})")

        # store shape for the next (larger) length
        prev_a = np.array(seq, dtype=np.float64)

    # --------------------------------------------------------------
    # 2) fallback: uniform sequence (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        uniform = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = uniform.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) final sanity check (optional – cheap)
    # --------------------------------------------------------------
    try:
        final = evaluate_sequence(best_seq)
        print(f"Final best official score {final:.6f} (length {len(best_seq)})")
    except Exception:
        print("Warning: final evaluation failed.")

    return best_seq
