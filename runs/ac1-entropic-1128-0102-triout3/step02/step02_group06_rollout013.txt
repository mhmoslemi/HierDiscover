Propose a step‑function sequence via a hybrid mirror‑descent / variance‐reduction
algorithm.

The objective of `evaluate_sequence` (after normalising the ℓ¹‑norm) equals the
maximum entry of the self‑convolution of the coefficient vector.
We minimise this quantity by:

* Optimising on the simplex with a log‑mirrored parametrisation
  a_i = exp(θ_i)  and Σ a_i = √(2·n)  (projection by simple scaling).
* Using Adam on the log‑variables.
* Gradient = ( sub‑gradient of the maximum entry )
            + β·(gradient of the variance of the convolution).
  The variance term flattens the whole convolution early in the run and
  therefore helps to reduce the eventual peak.
* A soft‑max smoothing of the max is employed while a temperature τ is
  annealed; once τ is small we switch to the exact sub‑gradient.
* Occasional jitter and a restart mechanism avoid stagnation.
* The best sequence encountered before the time budget expires is returned.

The implementation relies only on NumPy and the provided
`evaluate_sequence`. No file or network access is performed.

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the first (2·len(a)‑1) real entries.
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int, a_rev_fft: np.ndarray) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a (soft‑max approximation of max).
    a_rev_fft is the FFT of a[::-1] (pre‑computed to avoid recomputation).
    """
    # stable soft‑max weights
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # convolution of w with reversed a in the frequency domain
    W = np.fft.rfft(w, nfft)
    grad_full = np.fft.irfft(W * a_rev_fft, n=nfft)[: 2 * a.size - 1]
    # extract the part that corresponds to the original vector length
    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the index k_max.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


def _grad_variance(a: np.ndarray, conv: np.ndarray, avg: float,
                   nfft: int, a_rev_fft: np.ndarray, A_sq: np.ndarray) -> np.ndarray:
    """
    Gradient of the variance (mean squared deviation) of the convolution.
    Uses the fact that Σ a_i = const, so the average of conv is constant.
    """
    n = a.size
    m = conv.size                               # = 2·n‑1
    # FFT of (conv - avg) : A_sq holds FFT(a)^2 = FFT(conv)
    residual_fft = A_sq.copy()
    residual_fft[0] -= avg * m                  # DC adjustment for the constant shift
    # convolution of the residual with reversed a (frequency domain)
    conv_full = np.fft.irfft(residual_fft * a_rev_fft, n=nfft)
    start = n - 1
    grad = conv_full[start : start + n]
    grad = (4.0 / m) * grad
    return grad


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σ x = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # initialise sequence length and ℓ₁‑norm (the problem normalisation)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.size
    else:
        n = 2401                      # reasonable default length
        init_seq = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64)

    S = np.sqrt(2.0 * n)               # required ℓ₁‑norm
    init_seq = init_seq * (S / init_seq.sum())

    # mirror‑descent in log‑space (θ such that a = expθ)
    eps = 1e-12
    log_theta = np.log(init_seq + eps)

    # bookkeeping of the best solution
    best_a = init_seq.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial score (max conv): {best_score:.6f}")

    # pre‑compute FFT length (must cover 3·n‑2 for the variance gradient)
    nfft = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters (operating on log‑variables)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta = 0.20                     # base learning rate in log‑space

    m_adam = np.zeros_like(log_theta)
    v_adam = np.zeros_like(log_theta)

    # temperature schedule for the soft‑max smoothing of the max
    tau = 1.0
    tau_min = 0.001
    total_iters_est = 500_000
    tau_decay = (tau_min / tau) ** (1.0 / total_iters_est)

    # variance‑gradient weight schedule
    beta0 = 5.0                       # initial weight of the variance term
    # linear decay to zero over the estimated number of iterations
    # (beta_cur = max(0, beta0*(1‑t/T)))
    # stagnation handling
    no_improve = 0
    restart_limit = 8_000

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # reconstruct a from log‑variables and enforce Σa = S
        # ----------------------------------------------------------
        a = np.exp(log_theta)
        sum_a = a.sum()
        if sum_a == 0.0:                     # numerical degenerate case → re‑initialise
            a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64) * S
            log_theta = np.log(a + eps)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            continue
        a = a * (S / sum_a)                  # Euclidean scaling onto the ℓ¹‑simplex

        # ----------------------------------------------------------
        # FFT of a and its reverse (used many times)
        # ----------------------------------------------------------
        A = np.fft.rfft(a, nfft)                     # FFT of a
        a_rev_fft = np.fft.rfft(a[::-1], nfft)       # FFT of reversed a
        A_sq = A * A                                 # FFT of self‑convolution

        # ----------------------------------------------------------
        # self‑convolution and current objective value
        # ----------------------------------------------------------
        conv = np.fft.irfft(A_sq, n=nfft)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = cur_max                           # identical to evaluate_sequence after normalisation

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{iteration:08d}] New best: {best_score:.6f}")
        else:
            no_improve += 1

        # early exit if target already reached
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration}")
            break

        # ----------------------------------------------------------
        # gradients
        # ----------------------------------------------------------
        # (1) gradient of the (smoothed) max‑convolution
        if tau > tau_min * 2:
            grad_max = _grad_softmax(a, conv, tau, nfft, a_rev_fft)
        else:
            k_max = int(conv.argmax())
            grad_max = _grad_max(a, conv, k_max)

        # (2) gradient of the variance of the convolution
        avg_conv = S * S / (2 * n - 1)                 # constant because Σa = S
        grad_var = _grad_variance(a, conv, avg_conv,
                                  nfft, a_rev_fft, A_sq)

        # combine the two gradient contributions
        beta_cur = max(0.0, beta0 * (1.0 - iteration / total_iters_est))
        grad = grad_max + beta_cur * grad_var

        # chain rule for log‑variables
        grad_log = grad * a

        # ----------------------------------------------------------
        # Adam update in log‑space (descent)
        # ----------------------------------------------------------
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_log
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m_adam / (1.0 - beta1 ** iteration)
        v_hat = v_adam / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step                     # descent step

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.005:
            jitter = np.random.randn(n) * (eta * 0.05)
            log_theta = log_theta + jitter

        # temperature annealing
        tau = max(tau * tau_decay, tau_min)

        # restart if stagnation persists
        if no_improve >= restart_limit:
            a = best_a.copy()
            perturb = np.random.randn(n) * (eta * 0.2)
            a = a * np.exp(perturb)                     # multiplicative perturbation
            a = a * (S / a.sum())
            log_theta = np.log(a + eps)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            no_improve = 0
            print(f"[{iteration:08d}] Restarted after stagnation.")

    # --------------------------------------------------------------
    # final sanity projection and verification
    # --------------------------------------------------------------
    final_seq = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
