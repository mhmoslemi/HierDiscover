"""
Hybrid Adam‑based optimisation of the max self‑convolution.

The objective 2·n·max(conv)/sum(a)² reduces (for sum(a)=√(2n))
to minimise max(conv) over non‑negative vectors a with fixed ℓ₁‑norm.
We replace the non‑smooth max by a log‑sum‑exp (soft‑max) with a
temperature β that is increased during the run, obtaining a smooth
gradient.  Adam (with projection onto the ℓ₁‑simplex) provides
efficient descent; occasional random mass‑transfer moves and a
restart‑on‑stagnation scheme help escape local minima.
The routine stops when the time budget is exhausted or a target
value (< 1.503) is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must stay top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full (2·len(a)‑1) result (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Adam optimisation on a soft‑max approximation of the maximal entry
    of the self‑convolution, with projection onto the ℓ₁‑simplex.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size and initial point
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=np.float64)
        n = a0.shape[0]
    else:
        n = 2401                     # a modest size that worked previously
        a0 = None

    S = np.sqrt(2.0 * n)            # ℓ₁‑norm we keep fixed
    nfft_conv = _next_pow2(2 * n - 1)          # for self‑convolution
    nfft_grad = _next_pow2(3 * n - 2)          # for correlation (grad)

    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        a = np.full(n, S / n, dtype=np.float64)   # uniform start

    # --------------------------------------------------------------
    # Adam state and hyper‑parameters
    # --------------------------------------------------------------
    m = np.zeros_like(a)       # first moment
    v = np.zeros_like(a)       # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    lr = 0.5 * S / n            # base learning rate (≈ step size)

    temperature = 0.2
    temp_decay = 0.9997
    min_temp = 1e-6

    beta_soft = 5.0             # start temperature for log‑sum‑exp
    beta_factor = 1.001
    beta_max = 200.0

    # bookkeeping
    best_a = a.copy()
    best_score = float('inf')
    no_improve = 0
    stagnation_limit = 2500
    iteration = 0

    # --------------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # current convolution and objective (max entry)
        conv = _convolve_self_fft(a, nfft_conv)
        cur_max = conv.max()
        cur_score = cur_max                     # evaluate_sequence == max for our normalisation

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:07d}] New best max={best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # early exit if target reached
        if best_score < 1.503:
            print(f"Target reached after {iteration} iterations (best={best_score:.6f})")
            break

        # ----------------------------------------------------------
        # smooth approximation of the max (log‑sum‑exp)
        # ----------------------------------------------------------
        # shift by cur_max to avoid overflow
        exp_arg = beta_soft * (conv - cur_max)
        w_raw = np.exp(exp_arg)
        w = w_raw / w_raw.sum()                # softmax weights, sum to 1

        # ----------------------------------------------------------
        # gradient of the soft‑max (via correlation)
        # ----------------------------------------------------------
        # g_i = 2 * sum_k w_k * a_{k-i}
        corr = np.correlate(w, a, mode='full')
        grad = 2.0 * corr[n - 1: n - 1 + n]     # slice = indices 0 … n‑1

        # ----------------------------------------------------------
        # Adam update with projection onto the simplex
        # ----------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = _project_to_simplex(a - step, S)

        # ----------------------------------------------------------
        # evaluate candidate (exact max) and accept/reject
        # ----------------------------------------------------------
        conv_c = _convolve_self_fft(a_candidate, nfft_conv)
        max_c = conv_c.max()

        accept = False
        if max_c < cur_max:
            accept = True
        else:
            prob = np.exp(-(max_c - cur_max) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_max = max_c

        # ----------------------------------------------------------
        # occasional random mass‑transfer move (simulated annealing)
        # ----------------------------------------------------------
        if np.random.rand() < 0.1:
            # pick a source proportional to its mass
            if a.sum() > 0:
                src_probs = a / a.sum()
                i = np.random.choice(n, p=src_probs)
            else:
                i = np.random.randint(n)
            # distinct destination
            j = i
            while j == i:
                j = np.random.randint(n)

            delta = 0.07 * a[i]            # move ~7 % of the source mass
            a_rand = a.copy()
            a_rand[i] -= delta
            a_rand[j] += delta
            a_rand = _project_to_simplex(a_rand, S)

            conv_r = _convolve_self_fft(a_rand, nfft_conv)
            max_r = conv_r.max()
            if max_r < cur_max or np.random.rand() < 0.05:
                a = a_rand
                cur_max = max_r

        # ----------------------------------------------------------
        # temperature & soft‑max schedule
        # ----------------------------------------------------------
        temperature = max(temperature * temp_decay, min_temp)
        beta_soft = min(beta_soft * beta_factor, beta_max)

        # ----------------------------------------------------------
        # restart if we have been stuck too long
        # ----------------------------------------------------------
        if no_improve >= stagnation_limit:
            print(f"[{iteration:07d}] Stagnation – performing a full restart.")
            alpha = np.full(n, 0.4)
            a = np.random.dirichlet(alpha) * S
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # -----------------------------------------------------------------
    # final verification and return
    # -----------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()
