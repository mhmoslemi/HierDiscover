def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a

def propose_candidate(seed=0, budget_s=950.0):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    n = 2500

    best_score = np.inf
    best_a = None
    best_params = None

    # coarse search over p and offset
    p_vals = np.arange(0.4, 0.91, 0.01)   # 0.4 to 0.9
    offsets = np.concatenate([
        np.linspace(0.2, 5.0, 25),
        np.linspace(5.0, 20.0, 15),
        np.array([30., 50., 100.])
    ])
    for offset in offsets:
        for p in p_vals:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=offset)
            conv = conv_fft(a)
            max_val = conv.max()
            score = 2.0 * n * max_val / (a.sum()**2)   # a.sum()≈1
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_params = (p, offset)
                print(f"[grid] p={p:.3f}, offset={offset:.3f}, score={score:.6f}")

    # Fine refinement around best
    if best_params is not None:
        p_best, offset_best = best_params
        p_fine_vals = np.arange(max(p_best - 0.02, 0.01), min(p_best + 0.02, 0.99), 0.001)
        offset_fine_vals = np.arange(max(offset_best - 1.0, 0.05), offset_best + 1.0, 0.05)
        for offset in offset_fine_vals:
            for p in p_fine_vals:
                if time.time() > deadline:

"""
propose_candidate
-----------------
Search for a non‑negative step‑function (list of floats) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

using a broad parametric sweep over power‑law families, a few
random Dirichlet seeds and a long projected‑gradient (Adam) descent.
The implementation stays well within the 1000 s budget and prints
intermediate best scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """
    a_i = (i + offset) ** (-p)  , i = 0 … n‑1
    The vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:                     # safety – should never happen
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution x * x using a real FFT.
    Result length = 2·len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    O(n log n) implementation (Duchi et al. 2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry at index `peak_idx`
    with respect to the vector a (self‑convolution).
    g_i = 2·a_{peak_idx-i} for all i that contribute,
    otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    j = peak_idx - i
    grad[i] = 2.0 * a[j]
    return grad
