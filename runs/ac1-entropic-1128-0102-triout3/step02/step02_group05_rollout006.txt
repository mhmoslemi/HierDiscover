"""
Propose a step‑function (non‑negative vector) that minimises the
max‑self‑convolution bound used by ``evaluate_sequence``.

The search consists of three phases:

1. **Diverse initialisation** – a short scan of analytic families
   (power‑law, exponential) together with a few random Dirichlet draws.
   The best of these vectors (normalised to ℓ¹‑norm = 1) seeds the next
   phase.

2. **Mirror‑descent refinement** – exponentiated‑gradient (entropy
   mirror descent) on the probability simplex.  At each iteration we
   compute the self‑convolution by FFT, pick the few largest
   convolution entries (the “active set”) and use the sum of their
   sub‑gradients as a descent direction.  The iterate is updated by
   multiplicative weights and renormalised.  This deterministic phase
   occupies roughly 60 % of the time budget.

3. **Simulated annealing fine‑tuning** – simple mass‑transfer moves
   (move a fraction of the mass from one coordinate to another) are
   accepted if they improve the objective, otherwise with a
   temperature‑dependent probability.  Temperature decays geometrically.
   The best vector seen during the whole run is returned.

All operations respect the 1000 s time limit, print periodic progress
and use only NumPy (no external LP/SDP solvers).  The returned list
contains non‑negative numbers and will be accepted by ``evaluate_sequence``.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Fast objective (no security checks, used internally)
def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2)  (fast version without security)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # Use a power‑of‑2 size for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of (a*a)[k] w.r.t. a (single active index)."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad


def batch_subgradient(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """Sum of sub‑gradients for a list/array of active convolution indices."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i = np.arange(n)
    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
    return grad


# ----------------------------------------------------------------------
def power_law_sequence(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}, normalised to ℓ¹‑norm = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·(i‑1)), normalised to ℓ¹‑norm = 1."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def random_dirichlet_sequence(n: int, alpha: float = 0.5) -> np.ndarray:
    """Draw a random point from Dirichlet(alpha,…,alpha)."""
    a = np.random.gamma(alpha, 1.0, size=n)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a

    return best_seq.tolist()
