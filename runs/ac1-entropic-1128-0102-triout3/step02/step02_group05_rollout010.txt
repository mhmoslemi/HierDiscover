assistantfinal**Solution Overview**

The search routine builds a non‑negative step‑function `a` (a probability vector) and tries to minimise  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2}\;,
\]

which is exactly the value returned by `evaluate_sequence`.  
Because the objective is *scale invariant* we keep the ℓ¹‑norm of `a` equal to 1
by projecting onto the probability simplex after every update.

The optimiser works as follows

1. **Initialisation** – try several analytic families (power‑law, exponential,
   uniform) and the previously discovered sequence `height_sequence_1`.  
   The best of these is taken as the start point.

2. **Smooth max‑approximation** – instead of using only the active index of the
   self‑convolution we smooth the `max` with a soft‑max:
   \[
   w_k=\frac{\exp(\alpha\,c_k)}{\sum_j\exp(\alpha\,c_j)},\qquad
   c=a*a .
   \]
   The parameter α is increased linearly during the run so that the optimisation
   first sees a global landscape and later focuses on the true peak.

3. **Gradient computation** – the gradient of the soft‑max objective is the
   weighted correlation of `a` and `w`:
   \[
   \nabla_a F \;=\;2\;(a\star w)[\cdot] ,\qquad
   (a\star w)_i=\sum_{k} a_k\,w_{i+k}.
   \]
   Both the self‑convolution and the correlation are evaluated with FFTs
   ( O(n log n) ).

4. **Adam update & simplex projection** – the gradient is fed to the Adam
   optimiser (adaptive step‑sizes).  After the Adam step the vector is projected
   back onto the simplex (non‑negative, sum = 1).

5. **Exploration moves** – with a small probability a random mass transfer
   between two coordinates is attempted; improvements are kept, occasional
   uphill moves are accepted to escape plateaus.

6. **Book‑keeping** – the true objective (via the fast evaluator) is computed
   every few thousand iterations; the best sequence seen so far is stored.

The whole loop runs until the time budget (≈ 1000 s) expires and finally returns
the best sequence as a plain list.

"""
Propose a low‑value step‑function using a soft‑max
gradient descent with Adam on the probability simplex.
The optimiser combines a smooth approximation of the max
self‑convolution with occasional random mass‑transfer moves.
"""

import numpy as np
import time


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum_i x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)          # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence (no security checks).
    Returns 2 * n * max(a*a) / (sum(a)**2).
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def softmax(x: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max with temperature 1/alpha."""
    # The larger alpha, the closer to the true max.
    mx = np.max(x)
    y = np.exp(alpha * (x - mx))
    return y / np.sum(y)


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, kind: str = "power", **kw) -> np.ndarray:
    """
    Produce a normalised (ℓ¹‑norm = 1) non‑negative vector.
    ``kind`` may be ``"power"``, ``"exp"`` or ``"uniform"``.
    """
    if kind == "uniform":
        a = np.full(n, 1.0 / n)
    elif kind == "exp":
        # a_i ∝ exp(-λ·i)
        lam = kw.get("lam", 0.001)
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * idx)
        a /= a.sum()
    else:   # power‑law
        p = kw.get("p", 0.5)
        off = kw.get("offset", 0.0)
        idx = np.arange(1, n + 1, dtype=np.float64) + off
        a = 1.0 / np.power(idx, p)
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation value.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence (as a plain Python list) found so far.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5     # safety margin

    # --------------------------------------------------------------
    # 1. Choose a problem size.
    #    3000–5000 is a good compromise between expressivity and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Scan a few analytically motivated families for a good start.
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # power‑law family
    for p in (0.3, 0.4, 0.5, 0.6, 0.8):
        for off in (0.0, 0.1, 0.5, 1.0, 2.0):
            a = build_initial_sequence(n, kind="power", p=p, offset=off)
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()

    # exponential family
    for lam in (5e-4, 1e-3, 2e-3):
        a = build_initial_sequence(n, kind="exp", lam=lam)
        val = fast_score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()

    # uniform (just for completeness)
    a = build_initial_sequence(n, kind="uniform")
    val = fast_score(a)
    if val < best_val:
        best_val, best_seq = val, a.copy()

    # --------------------------------------------------------------
    # 3. If a previous construction exists, try to reuse it.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                a = prev[:n].copy()
            else:
                a = np.zeros(n, dtype=np.float64)
                a[:prev.size] = prev
            a = np.maximum(a, 0.0)
            if a.sum() > 0:
                a /= a.sum()
                val = fast_score(a)
                if val < best_val:
                    best_val, best_seq = val, a.copy()
                print(f"[init] height_sequence_1 gave value {val:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 4. Main optimisation loop – Adam + soft‑max gradient + random moves.
    # --------------------------------------------------------------
    a = best_seq.copy()
    best_seq = a.copy()
    best_val = fast_score(a)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # FFT sizes
    size_self = 1 << ((2 * n - 1).bit_length())          # for a*a
    size_corr = 1 << ((3 * n - 2).bit_length())          # for correlation a ⋆ w

    it = 0
    while time.time() < deadline:
        # ---------- self‑convolution ----------
        fft_a_self = np.fft.rfft(a, size_self)
        conv_self = np.fft.irfft(fft_a_self * fft_a_self, size_self)[: 2 * n - 1]

        # ---------- soft‑max weights ----------
        # α grows linearly from 0.5 to ~30 throughout the run
        elapsed = time.time() - start
        frac = min(1.0, elapsed / budget_s)
        alpha = 0.5 + 30.0 * frac
        w = softmax(conv_self, alpha)

        # ---------- gradient via correlation ----------
        # reuse FFT of a (but with larger size)
        fft_a = np.fft.rfft(a, size_corr)
        w_rev = w[::-1]                                 # reversed for correlation
        fft_w_rev = np.fft.rfft(w_rev, size_corr)

        corr_full = np.fft.irfft(fft_a * fft_w_rev, size_corr)
        # gradient_i = 2 * corr_full[n-1 : 2*n-1][::-1]
        grad = 2.0 * corr_full[n - 1 : 2 * n - 1][::-1]

        # ---------- Adam update ----------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.1 / (np.sqrt(v_hat) + eps)   # base learning‑rate 0.1
        a_new = a - step * m_hat

        # ---------- projection onto simplex ----------
        a = proj_simplex(a_new)

        # ---------- occasional random hill‑climb ----------
        if np.random.rand() < 0.02:    # 2 % probability
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = proj_simplex(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_seq = cand_val, cand.copy()
                    a = cand                     # keep the improvement
                elif np.random.rand() < 0.001:
                    a = cand                     # occasional uphill move

        # ---------- periodic bookkeeping ----------
        if it % 2500 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
                print(f"[iter {it:6d}] new best {best_val:.6f}")
            else:
                print(f"[iter {it:6d}] current {cur_val:.6f} best {best_val:.6f}")

    print(f"[final] best value found: {best_val:.6f}")
    return best_seq.tolist()
