def _resample_to_length(arr, new_n):
    old_n = len(arr)
    if old_n == new_n:
        return arr / arr.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_arr = np.interp(new_x, old_x, arr)
    # enforce nonneg
    new_arr = np.clip(new_arr, 0, None)
    if new_arr.sum() == 0:
        new_arr = np.full_like(new_arr, 1.0 / len(new_arr))
    else:
        new_arr = new_arr / new_arr.sum()
    return new_arr
```

    N = 1 << ((2*n - 2).bit_length()) # next pow2 >= 2*n-1
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a

    C = A * A
    conv_full = np.fft.irfft(C, n=N)[:2*n-1]
    return conv_full

def _project_simplex(v):
    # v: np.ndarray
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all zero case, return uniform
        w = np.full_like(v, 1.0 / n)
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a refined step‑function sequence.

The algorithm consists of three stages:

1. **Heuristic initialisation** – several classic window functions
   (Hann, Hamming, Blackman, triangular, Gaussian, exponential,
   uniform) are generated for a range of lengths.  They are normalised
   to sum to one and the best among them (according to the official
   evaluation function) is kept as a starting point.

2. **Local optimisation** – from this starting point we run a projected
   Adam optimiser directly on the coefficients (instead of on soft‑max
   logits).  After each Adam step the coefficients are projected onto
   the probability simplex (non‑negative, sum = 1) using the O(n log n)
   algorithm of Duchi et al.  The gradient is the sub‑gradient of the
   current maximal convolution entry (or all entries attaining the
   maximum).  A small random jitter is injected every few thousand
   steps to escape shallow minima.

3. **Interpolation & multi‑length search** – the best sequence found so
   far is interpolated to the next candidate length and the optimisation
   is repeated.  All the while the total wall‑clock time is limited by
   the ``budget_s`` argument (≈ 1000 s).

The code relies only on NumPy (no external solvers) and respects the
evaluation routine’s security checks.  It returns the best list of
non‑negative heights discovered before the timeout.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the full convolution of length 2*len(a)-1.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A                       # |A|^2
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values – clip for safety
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0 , sum x_i = 1}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries are zero or negative
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                      # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # due to floating errors the sum may be slightly off; renormalise
    s = w.sum()
    if s == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= s
    return w


def _resample_to_length(arr: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a 1‑D array to a new length.
    The result is normalised to sum to one.
    """
    old_n = arr.shape[0]
    if old_n == new_n:
        out = arr.copy()
    else:
        old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
        new_x = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
        out = np.interp(new_x, old_x, arr)
    out = np.clip(out, 0.0, None)
    tot = out.sum()
    if tot == 0.0:
        out = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        out /= tot
    return out


# ----------------------------------------------------------------------
# Window factories (deterministic starting points)
# ----------------------------------------------------------------------


def _uniform_window(n: int) -> np.ndarray:
    w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    w /= w.sum()
    return w


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2.0 * np.pi * i / (n - 1))
         + a2 * np.cos(4.0 * np.pi * i / (n - 1)))
    w /= w.sum()
    return w


def _triangular_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs((2.0 * i) / (n - 1) - 1.0)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _gaussian_window(n: int, sigma_factor: float = 0.35) -> np.ndarray:
    """Gaussian centred window; sigma = sigma_factor * n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _exponential_window(n: int, tau_factor: float = 0.45) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    tau = tau_factor * n
    w = np.exp(-np.abs(i - centre) / tau)
    w /= w.sum()
    return w


def _generate_candidate_windows(n: int) -> List[np.ndarray]:
    """Return a collection of basic window shapes for length n."""
    windows = [
        _uniform_window(n),
        _hann_window(n),
        _hamming_window(n),
        _blackman_window(n),
        _triangular_window(n),
    ]
    # couple of Gaussians / exponentials with different spread
    for sf in (0.25, 0.35, 0.45):
        windows.append(_gaussian_window(n, sigma_factor=sf))
    for tf in (0.30, 0.45, 0.60):
        windows.append(_exponential_window(n, tau_factor=tf))
    return windows


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective under a fixed wall‑clock budget.
    The returned list is already normalised (its sum is 1) so the
    official score equals ``2 * n * max(conv)``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Global best bookkeeping
    # ------------------------------------------------------------------
    best_seq: List[float] | None = None
    best_score = float("inf")

    # Load previously found construction if it exists
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev /= prev.sum()
            cur = float(2.0 * len(prev) *
                        np.max(_conv_fft(prev)))     # sum == 1 ⇒ exact score
            best_seq = prev.tolist()
            best_score = cur
            print(f"[init] loaded previous sequence (len={len(prev)}) "
                  f"score={best_score:.6f}")
        except Exception as exc:                     # pragma: no cover
            print(f"[init] failed to import previous best: {exc}")

    # ------------------------------------------------------------------
    # Lengths to explore – start modest, finish with larger ones
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1800, 2400, 3000, 3600, 4800]

    # Main loop over lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # --------------------------------------------------------------
        # 1) Initialise from the best known sequence (interpolated) or
        #    from a set of deterministic windows.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # a) deterministic windows
        candidates.extend(_generate_candidate_windows(n))

        # b) interpolated previous best, if any
        if best_seq is not None:
            interp = _resample_to_length(np.asarray(best_seq), n)
            candidates.append(interp)

        # ----------------------------------------------------------------
        # Evaluate all seeds and keep the one with the smallest score.
        # ----------------------------------------------------------------
        local_best_a = None
        local_best_score = float("inf")
        for a0 in candidates:
            # safety – enforce simplex
            a0 = np.clip(a0, 0.0, None)
            total = a0.sum()
            if total == 0.0:
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a0 = a0 / total
            score = float(2.0 * n * np.max(_conv_fft(a0)))
            if score < local_best_score:
                local_best_score = score
                local_best_a = a0
        # Record if this simple step already beats the global optimum
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.tolist()
            print(f"[{n}] window‑based initial best {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) Projected Adam optimisation starting from local_best_a
        # --------------------------------------------------------------
        a = local_best_a.copy()
        m = np.zeros_like(a)                     # first moment
        v = np.zeros_like(a)                     # second moment
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        # base learning rate – scaled with sqrt(n) for stability
        base_lr = 0.15 / np.sqrt(n)

        # parameters for occasional jitter
        jitter_interval = 2500
        jitter_scale = 1e-4

        # iteration counter
        t = 0
        # a small cap on iterations per length – the loop will also stop on deadline
        max_iters = 80000

        while time.time() < deadline and t < max_iters:
            # ---------------------------------------------------------
            # Compute convolution and current score
            # ---------------------------------------------------------
            conv = _conv_fft(a)
            cur_max = conv.max()
            cur_score = float(2.0 * n * cur_max)

            # Record globally best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[{n}] new global best {best_score:.6f}  (iter {t})")

            # ---------------------------------------------------------
            # Sub‑gradient of the maximal convolution entry
            # ---------------------------------------------------------
            ks = np.where(conv == cur_max)[0]   # all active max positions
            grad = np.zeros_like(a)
            for k in ks:
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                idx = np.arange(i_start, i_end + 1, dtype=int)
                grad[idx] += 2.0 * a[k - idx]

            # ---------------------------------------------------------
            # Adam update (in the original space) + simplex projection
            # ---------------------------------------------------------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # learning‑rate decay (very mild)
            lr = base_lr / (1.0 + t * 1e-5)
            step = lr * (m_hat / (np.sqrt(v_hat) + eps))

            a_new = a - step
            a = _project_simplex(a_new)

            # ---------------------------------------------------------
            # Jitter to avoid stagnation
            # ---------------------------------------------------------
            if t % jitter_interval == 0:
                noise = np.random.randn(n) * jitter_scale
                a = _project_simplex(a + noise)

            # ---------------------------------------------------------
            # Optional early break – if we have not improved for a while
            # ---------------------------------------------------------
            if t % 5000 == 0:
                remaining = deadline - time.time()
                if remaining < 5.0:               # leave a tiny margin
                    break

        # ------------------------------------------------------------------
        # Report progress for this length
        # ------------------------------------------------------------------
        print(f"[{n}] finished (best score for this n = {local_best_score:.6f})")
        print(f"[{n}] current global best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Fallback in the extremely unlikely event that nothing was found
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check (using the official evaluate_sequence)

        final_score = evaluate_sequence(best_seq)
        print(f"Final best official score {final_score:.6f} "
              f"(length {len(best_seq)})")
    except Exception as exc:                     # pragma: no cover
        print(f"Warning: final evaluation failed – {exc}")

    return best_seq
