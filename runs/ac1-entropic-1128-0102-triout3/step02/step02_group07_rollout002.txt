"""
Propose a candidate step‑function sequence that minimizes
``evaluate_sequence``.

Algorithm
---------
1. Build a diverse pool of seed vectors:
   * a power‑law a_i ∝ (i+offset)^(-p) (the previously best guess);
   * a half‑sine shape a_i ∝ sin((i+1)·π/(2·n+2));
   * the historic ``height_sequence_1`` if it exists;
   * a few random Dirichlet draws.

2. For each seed keep the one with the lowest reported score.

3. Run a *soft‑max* (smooth‑max) projected gradient descent.
   * The true objective is  max(conv(a,a)).  We replace the max by
     f_α(a)= (1/α)·log Σ exp(α·conv_k), whose gradient is a smooth
     weighted sum of the individual convolution gradients.
   * The gradient can be obtained by a single cross‑correlation,
     efficiently computed with FFTs.
   * An Adam‑style step is taken and the iterate is projected onto the
     probability simplex (non‑negative, sum = 1).
   * The temperature‑parameter α is increased geometrically; for large α
     the smoothed problem approaches the original max‑problem.

4. After the smooth phase perform a short *hard‑max* refinement that
   uses the exact sub‑gradient of the active convolution entry (the
   technique from the reference solution).  This cleans up the final
   vector.

5. Return the best vector found (converted to a plain list).  A final
   call to ``evaluate_sequence`` guarantees the reported score matches
   the evaluator.

The whole routine respects a ~950 s internal budget; it prints progress
messages and continuously updates the best solution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i = (i+offset)^(-p), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def sine_sequence(n: int) -> np.ndarray:
    """a_i ∝ sin((i+1)·π/(2·n+2)), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin((i + 1) * np.pi / (2 * n + 2))
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv


def convolve_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution (linear) of two real vectors using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    N = 1 << ((n + m - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[: n + m - 1]
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Exact sub‑gradient of the active convolution entry (peak_idx)."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad

    """
    Search for a non‑negative sequence that minimises the evaluation
    function.  The method combines several seeds, a smooth‑max Adam
    optimiser and a final hard‑max refinement.
