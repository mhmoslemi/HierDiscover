def optimize_at_length(a0, n_iter, deadline, start_time, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    conv_len = 2*n -1
    Lc = 1 << ((conv_len).bit_length())
    # schedule
    step0 = 0.6
    alpha = 0.5
    alpha_max = 120.0
    alpha_factor = 1.12
    it_total = 0
    while time.time() < deadline and alpha < alpha_max:
        # inner loop
        # compute iteration budget for current alpha: maybe 2000 steps or dependent on n.
        inner_budget = max(2000, int(200000 / n)) # ensure some min
        for _ in range(inner_budget):
            if time.time() >= deadline:
                break
            # compute conv of a
            Af = np.fft.rfft(a, n=Lc)
            conv = np.fft.irfft(Af * Af, n=Lc)[:conv_len]
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] New best: n={n}, score={cur_score:.6f}")
            # compute softmax weights
            w = softmax_weights(conv, alpha)
            # gradient
            grad = gradient_softmax(a, w)
            # step
            it_total += 1
            step = step0 / math.sqrt(it_total)
            # exponentiated gradient update
            a = a * np.exp(-step * grad)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum
        alpha *= alpha_factor
    return a, best_seq, best_score

"""
Propose a candidate step‑function using a soft‑max (log‑sum‑exp) smoothing
of the convolution max and exponentiated‑gradient descent on the probability
simplex.  The algorithm works in three stages:

1.  Initialise from the best known construction (global ``height_sequence_1``)
    or from a uniform distribution.
2.  Progressive up‑sampling: repeatedly enlarge the vector (≈1.5×) while
    keeping its shape by linear interpolation.
3.  For each length run a schedule of “temperature” values α.
    For a fixed α we minimise  log‑sum‑exp(α·conv(a,a))  by exponentiated‑gradient
    descent (a_i ← a_i·exp(−η·∇) , renormalised).  The gradient of the
    smoothed objective is 2·(w⋆a_rev) where w are the soft‑max weights of
    the current convolution and a_rev is a reversed copy of a.
    The step size η decays like η₀/√t.  Improvements are logged and the
    best sequence (in terms of the original evaluation function) is kept.
    A final refinement with a larger α is performed on the best length found.

The whole routine respects the 1000 s time limit and returns a list of
non‑negative floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Evaluation score for a sequence that already sums to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over the convolution entries (stable implementation)."""
    # shift by the max to avoid overflow
    c = conv.max()
    exps = np.exp(alpha * (conv - c))
    s = exps.sum()
    if s == 0.0:
        # fallback – uniform weights
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exps / s


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp objective.
    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector.
    w : np.ndarray, shape (2n‑1,)
        Soft‑max weights (sum to 1).
    Returns
    -------
    g : np.ndarray, shape (n,)
        Gradient ∂/∂a (  log Σ_i exp(α·conv_i) ).
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed copy
    # length needed for the convolution w * a_rev is (2n‑1)+n‑1 = 3n‑2
    L = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, n=L)
    far = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(fw * far, n=L)[: len(w) + n - 1]
    # the slice conv[n‑1 : n‑1+n] gives the needed gradient components
    g = 2.0 * conv[n - 1 : n - 1 + n]
    return g


def _optimize_at_length(
    a0: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    start_time: float,
    deadline: float,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Run the soft‑max exponentiated‑gradient optimiser for a fixed length.
    Returns the (possibly) improved a, the global best sequence and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len.bit_length())          # power of two ≥ 2n‑1

    # annealing schedule for the soft‑max temperature
    alpha = 0.5
    alpha_max = 128.0
    alpha_factor = 1.25

    # step‑size parameters
    eta0 = 0.6
    it_total = 0

    while time.time() < deadline and alpha <= alpha_max:
        # number of inner iterations for the current temperature
        # (scale inversely with n so that total work stays roughly constant)
        inner_iters = max(5000, int(3_000_000 / n))

        for _ in range(inner_iters):
            if time.time() >= deadline:
                break

            # ----- convolution a*a -------------------------------------------------
            Af = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(Af * Af, n=fft_len)[:conv_len]
            max_val = conv.max()
            cur_score = 2.0 * n * max_val

            # ----- record improvement ----------------------------------------------
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] "
                    f"Improved: n={n}, α={alpha:.2f}, iter={it_total}, score={cur_score:.6f}"
                )

            # ----- soft‑max weights --------------------------------------------------
            w = _softmax_weights(conv, alpha)

            # ----- gradient ---------------------------------------------------------
            g = _gradient_from_weights(a, w)

            # ----- exponentiated‑gradient step ---------------------------------------
            it_total += 1
            eta = eta0 / math.sqrt(it_total)
            # multiplicative update (keeps positivity automatically)
            a = a * np.exp(-eta * g)
            s = a.sum()
            if s == 0.0:                     # numeric safety
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

        # increase the temperature (focus more on the true max)
        alpha *= alpha_factor

    return a, best_seq, best_score
