def weighted_gradient_fft(a, conv, beta):
    n = len(a)
    w = np.exp(beta * (conv - conv.max()))
    w /= w.sum()
    # compute (rev_a) * w convolution of length 2n-1 + n -1 = 3n-2
    # Want full convolution via FFT length L= next pow2 >= n + (2n-1) maybe.
    L = 1
    while L < n + (2*n -1):
        L <<= 1
    rev_a = a[::-1]
    # compute rfft with length L
    A = np.fft.rfft(rev_a, n=L)
    W = np.fft.rfft(w, n=L)
    grad_full = np.fft.irfft(A * W, n=L)
    # get slice
    start = n-1
    grad = 2.0 * grad_full[start:start+n]
    return grad
```

def adam_optimize(a0, n, rng, deadline, best_a, best_score):
    a = a0.copy()
    best = best_a.copy()
    best_s = best_score
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.2
    beta = 5.0
    iteration = 0
    cur_score = best_s
    while time.time() < deadline:
        iteration += 1
        conv = conv_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        # weighted gradient
        grad = weighted_gradient_fft(a, conv, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        # Keep step size bounded to avoid negative
        step = np.minimum(step, a * 0.99)
        a_new = a - step
        # occasional random jitter
        if rng.random() < 0.001:
            a_new += rng.normal(scale=1e-5, size=n)
        # project onto simplex
        a_new = project_onto_simplex(a_new)
        conv_new = conv_fft(a_new)
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        # update best
        if new_score < best_s:
            best_s = new_score
            best = a_new.copy()
            print(f"[adam] iter {iteration:06d}, new best {best_s:.6f}")
        # accept if improves; else accept with small probability
        if new_score <= cur_score:
            a = a_new
        else:
            # use simulated annealing acceptance
            temp = max(0.01, 0.5 / iteration)
            if rng.random() < np.exp(-(new_score - cur_score) / temp):
                a = a_new
        # update hyperparameters schedule
        if iteration % 2000 == 0:
            beta = min(beta * 1.1, 30.0)
            eta = max(eta * 0.9, 1e-5)
        # early stop if score < 1.503 (target)
        if best_s <= 1.503:
            print(f"[adam] target achieved at iter {iteration}")
            break
    return best, best_s

    a = a0.copy()
    best = best_a.copy()
    best_s = best_score
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.15
    beta = 5.0
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = conv_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        # find top indices
        top_k = min(max_peaks, conv.size)
        # Use partition for top_k largest values
        idxs = np.argpartition(-conv, top_k-1)[:top_k]
        # compute weights
        vals = conv[idxs]
        # subtract max for numerical stability
        w = np.exp(beta * (vals - cur_max))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / top_k)
        else:
            w = w / w_sum
        # compute gradient as weighted sum of peak gradients
        grad = np.zeros_like(a)
        for idx, weight in zip(idxs, w):
            grad += weight * gradient_of_peak(a, idx)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        # bound step size
        step = np.minimum(step, a * 0.99)
        a_new = a - step
        # occasional jitter
        if rng.random() < 0.001:
            a_new += rng.normal(scale=1e-5, size=n)
        a_new = project_onto_simplex(a_new)
        conv_new = conv_fft(a_new)
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        # update best
        if new_score < best_s:
            best_s = new_score
            best = a_new.copy()
            print(f"[adam] iter {iteration}, new best {best_s:.6f}")
        # acceptance
        if new_score <= cur_score:
            a = a_new
        else:
            # Metropolis-like acceptance
            temp = max(0.001, 0.5 / iteration)
            if rng.random() < np.exp(-(new_score - cur_score) / temp):
                a = a_new
        # schedule adjustments
        if iteration % 5000 == 0:
            beta = min(beta * 1.1, 30.0)
            eta = max(eta * 0.9, 1e-5)
        if best_s <= 1.503:
            print("[adam] Achieved target bound.")
            break
    return best, best_s

"""
Improved search for a low‑value step‑function.

The algorithm consists of three phases:

1. **Cheap shape search (small n)** – Randomly sample a few
   parameterised families (power‑law and exponential).  The best
   parameters are kept as a seed for the final optimisation.

2. **Upscaling** – The chosen shape is instantiated at a larger
   length (default 5000).  The vector is normalised to sum = 1 so that
   the evaluation function reduces to `2·n·max(conv)`.

3. **Projected Adam optimisation** – A fast Adam optimiser works on
   the probability simplex.  At each iteration the gradient is a
   weighted sum of the gradients of the *K* largest convolution entries
   (K≈4).  This multi‑peak sub‑gradient helps to reduce the global
   maximum more efficiently than using a single peak.  Adaptive
   learning‑rate decay, occasional jitter and a Metropolis‑style
   acceptance rule avoid premature stagnation.

The routine respects a 1000 s overall budget (default 950 s usable) and
prints a short progress log.  It returns the best sequence found as a
plain ``list[float]`` – the final call to ``evaluate_sequence`` guarantees
that the reported score matches the competition evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all at module level, no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+offset)^(-p) , normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:          # safety fallback (should never happen)
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def exponential_sequence(n: int, lam: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ lam^{i+offset} , normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = lam ** (i + offset)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution x*x using a real FFT.
    Result length = 2·len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry conv[peak_idx] w.r.t. a.
    The exact gradient is
        g_i = 2 * a_{peak_idx-i}   if the index is valid,
        otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    grad[idx] = 2.0 * a[jdx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser that uses a multi‑peak sub‑gradient

def adam_optimize_multi(
    a0: np.ndarray,
    n: int,
    rng: np.random.Generator,
    deadline: float,
    best_a: np.ndarray,
    best_score: float,
    max_peaks: int = 4,
    start_eta: float = 0.15,
    start_beta: float = 5.0,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex.
    Gradient = weighted sum of the gradients of the ``max_peaks`` largest
    convolution entries (soft‑max weighting with temperature ``beta``).
    """
    a = a0.copy()
    best = best_a.copy()
    best_s = best_score

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    eta = start_eta
    beta = start_beta
    it = 0

    while time.time() < deadline:
        it += 1

        conv = conv_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # --------------------------------------------------------------
        # build a multi‑peak sub‑gradient
        k = min(max_peaks, conv.size)
        # indices of the k largest values
        idxs = np.argpartition(-conv, k - 1)[:k]
        vals = conv[idxs]
        # soft‑max weights (more focus on the absolute maximum)
        w = np.exp(beta * (vals - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / k)
        else:
            w = w / w_sum

        grad = np.zeros_like(a)
        for idx, weight in zip(idxs, w):
            grad += weight * gradient_of_peak(a, int(idx))

        # --------------------------------------------------------------
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # keep the step from destroying non‑negativity
        step = np.minimum(step, a * 0.99)
        a_new = a - step

        # occasional random jitter to escape plateaus
        if rng.random() < 0.001:
            a_new += rng.normal(scale=1e-5, size=n)

        # projection onto the simplex
        a_new = project_onto_simplex(a_new)

        # evaluate the trial point
        conv_new = conv_fft(a_new)
        new_score = 2.0 * n * conv_new.max()

        # keep the globally best vector
        if new_score < best_s:
            best_s = new_score
            best = a_new.copy()
            print(f"[adam] iter {it:07d} – new best {best_s:.6f}")

        # acceptance rule (always accept improvements)
        if new_score <= cur_score:
            a = a_new
        else:
            # Metropolis‑style acceptance with a slowly cooling temperature
            temp = max(0.001, 0.5 / it)
            if rng.random() < np.exp(-(new_score - cur_score) / temp):
                a = a_new

        # schedule – slowly tighten focus and shrink step size
        if it % 5000 == 0:
            beta = min(beta * 1.1, 30.0)
            eta = max(eta * 0.9, 1e-6)

        # early stop if we have already beaten the target bound
        if best_s <= 1.503:
            print("[adam] target bound reached – terminating early")
            break

    return best, best_s

def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine first samples a few parametric families on a modest size,
    then builds a longer candidate and refines it with a projected Adam
    optimiser that uses a multi‑peak gradient.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – optional fall‑back to a known good seed
    try:
        height_sequence_1
        init = np.asarray(height_sequence_1, dtype=np.float64)
        init = np.maximum(init, 0.0)
        init /= init.sum()
        best_a = init.copy()
        best_score = 2.0 * len(init) * conv_fft(best_a).max()
        print("[init] using provided height_sequence_1, score =", best_score)
    except NameError:
        best_a = None
        best_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – cheap exploration on a small problem size
    n_small = 2000
    samples = 300                      # total number of random families
    print("[search] phase 1 – random family sampling (n =", n_small, ")")
    for _ in range(samples):
        if time.time() > deadline:
            break
        # randomly choose a family
        if rng.random() < 0.55:      # power‑law majority
            p = rng.uniform(0.05, 1.2)
            offset = rng.uniform(0.1, 5.0)
            a = power_law_sequence(n_small, p, offset)
        else:                        # exponential family
            lam = rng.uniform(0.90, 0.9999)
            a = exponential_sequence(n_small, lam)
        conv = conv_fft(a)
        score = 2.0 * n_small * conv.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[search] new small‑scale best {score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to a larger length (more optimisation room)
    n_large = 5000
    if best_a is None:                 # should never happen, but safety
        best_a = power_law_sequence(n_small, 0.5, 1.0)

    # Re‑instantiate the best‑found shape at the larger n
    # Detect which family gave the best result (very roughly)
    # For simplicity we reuse the *type* of the best candidate:
    # - if it looks like a power‑law (fast decay) we use power‑law,
    # - otherwise exponential.
    if np.allclose(best_a[:5], np.sort(best_a[:5])[::-1]):
        # assume power‑law – re‑fit p and offset by a quick linear regression
        # on log‑log values (robust enough for a quick guess)
        i = np.arange(n_small, dtype=np.float64) + 1.0
        log_i = np.log(i)
        log_a = np.log(best_a + 1e-30)
        p_est, log_const = np.polyfit(log_i, log_a, 1)
        p_est = -p_est
        offset_est = max(0.1, np.exp(log_const) ** (-1.0 / p_est) - 1.0)
        a_large = power_law_sequence(n_large, p_est, offset_est)
    else:
        # assume exponential – infer lambda from the ratio of successive entries
        ratio = best_a[1] / (best_a[0] + 1e-30)
        lam_est = max(0.90, min(0.9999, ratio))
        a_large = exponential_sequence(n_large, lam_est)

    a_large /= a_large.sum()
    cur_score_large = 2.0 * n_large * conv_fft(a_large).max()
    if cur_score_large < best_score:
        best_score = cur_score_large
        best_a = a_large.copy()
    print("[upscale] length =", n_large, "initial score =", cur_score_large)

    # ------------------------------------------------------------------
    # Phase 3 – projected Adam optimisation with multi‑peak gradient
    print("[optim] starting Adam optimisation")
    best_a, best_score = adam_optimize_multi(
        a0=best_a,
        n=n_large,
        rng=rng,
        deadline=deadline,
        best_a=best_a,
        best_score=best_score,
        max_peaks=4,
        start_eta=0.15,
        start_beta=5.0,
    )

    # ------------------------------------------------------------------
    # Final sanity check via the official evaluator
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print("[done] final reported score =", final_score)
    print("[info] length =", len(final_seq), "best score =", best_score)

    return final_seq
