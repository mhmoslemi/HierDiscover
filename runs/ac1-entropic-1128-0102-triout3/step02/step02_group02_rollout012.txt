"""
Improved optimisation of a step‑function sequence.

The objective of ``evaluate_sequence`` (with sum normalised to 1) is

    score = 2·n·max_k (a*a)_k ,

where ``a`` is a non‑negative vector of length *n*.
The maximum is approximated by a smooth “log‑sum‑exp’’ (soft‑max) with a
temperature α that is increased during optimisation.
The gradient of the smooth objective can be written in closed form as a
convolution of the current weight vector with the current distribution,
which we evaluate efficiently with FFTs.  Adam in the log‑space (soft‑max
parametrisation) together with a few restarts and a simple time budget
gives a better bound than the previous implementation.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT helpers (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x)
    m = len(y)
    N = _next_pow2(n + m - 1)
    X = np.fft.rfft(np.pad(x, (0, N - n)))
    Y = np.fft.rfft(np.pad(y, (0, N - m)))
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[: n + m - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Convolution a*a (full) – convenience wrapper."""
    return _full_convolution_fft(a, a)


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Gradient for the smooth (log‑sum‑exp) approximation of the max
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Given the convolution vector ``conv`` and temperature ``alpha``,
    return the normalised soft‑max weights w_k = exp(α·conv_k) / Σ.
    """
    scaled = alpha * conv
    # shift for numerical stability
    scaled_max = np.max(scaled)
    w = np.exp(scaled - scaled_max)
    w_sum = np.sum(w)
    if w_sum == 0.0:          # should never happen, but be safe
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    return w


def _grad_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t. the probability
    vector ``a`` (simplex).  The formula is

        ∇_a L = 2 * (w_rev * a)_{[(L‑n) : L][::-1]}

    where ``w_rev`` is ``w`` reversed and ``*`` denotes linear convolution.
    """
    n = len(a)
    L = len(w)                     # = 2·n‑1
    w_rev = w[::-1]
    conv = _full_convolution_fft(a, w_rev)   # length n+L‑1 = 3·n‑2
    # slice part that maps to the gradient, then reverse order
    grad = 2.0 * conv[L - n : L][::-1]        # shape (n,)
    return grad


def _grad_wrt_logits(a: np.ndarray, grad_a: np.ndarray) -> np.ndarray:
    """
    Convert the gradient w.r.t. ``a`` (which lies on the simplex)
    to the gradient w.r.t. the logits ``z`` used in the soft‑max.
    Standard chain‑rule for the soft‑max:
        ∂L/∂z = a ⊙ (∂L/∂a – (a·∂L/∂a)·1)
    """
    dot = a.dot(grad_a)
    return a * (grad_a - dot)


# ----------------------------------------------------------------------
# Adam optimiser (plain, no closures)
# ----------------------------------------------------------------------


def _adam_step(z: np.ndarray, g: np.ndarray,
               m: np.ndarray, v: np.ndarray,
               t: int, lr: float,
               beta1: float = 0.9, beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Perform one Adam update (returns new z, new m, new v)."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, m, v

# Main candidate generator
# ----------------------------------------------------------------------

    best_seq: List[float] | None = None

        except Exception:
            prev_seq = None

    # ------------------------------------------------------------------

        print(f"Final best score (official) {final_score:.6f} "

    except Exception:
