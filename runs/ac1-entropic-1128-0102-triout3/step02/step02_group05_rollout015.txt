"""
Gradient‑softmax optimisation of the self‑convolution peak.

We search for a non‑negative probability vector ``a`` (∑a_i = 1) that
minimises  2·n·max_k (a∗a)_k.
The objective is approximated by the smooth loss

    L(a) = (1/β)·log∑_k exp(β·c_k)               with  c = a∗a

which converges to the true maximum as β→∞.
Its gradient can be written as a weighted sum of the usual sub‑gradients

    ∇L_i = 2·∑_k p_k·a_{k-i}      with  p_k = softmax_β(c_k).

The weighted sum is evaluated by a single FFT‑based convolution.
We perform an exponentiated‑gradient (mirror‑descent w.r.t. KL) update

    a_i ← a_i·exp(−η·∇L_i)   and renormalise to the simplex.

A small entropy regulariser (λ·∑a_i log a_i) is added to the gradient to
encourage a flatter distribution.  The algorithm also employs occasional
random mass‑transfer moves to escape plateaus.

The routine runs for at most ``budget_s`` seconds, keeps the best sequence
encountered (according to a fast evaluation function) and finally returns
it as a plain Python list.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Fast evaluation used internally (identical to the provided evaluate_sequence
# but without the safety checks).  It works on vectors that already sum to 1.
def fast_score(a: np.ndarray) -> float:
    """Return the objective value for a probability vector a (∑a=1)."""
    n = a.size
    s = a.sum()
    if s <= 0.0:
        return np.inf
    # size for the self‑convolution
    conv_len = 2 * n - 1
    size_fft = 1 << ((conv_len).bit_length())
    fft_a = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fft_a * fft_a, size_fft)[:conv_len]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
def build_initial_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """
    Power‑law initialisation:
        a_i ∝ (i + offset)^{‑p},
    normalised to have ℓ¹‑norm equal to 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a
