"""
Propose a candidate step‑function by optimizing a smoothed
ℓ∞‑objective via Adam on the probability simplex.

The objective to minimise is
    score = 2·n·max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant, so we keep the coefficients on the simplex
(sum = 1, a_i ≥ 0).  Instead of the non‑smooth max we minimise the
soft‑max
    F(a) = (1/β)·log Σ_k exp(β·conv_k(a)),
whose gradient is a weighted sum of the convolution gradients.
The weights are the soft‑max probabilities p_k ∝ exp(β·conv_k).
A standard Adam update is applied to the gradient and the resulting
vector is projected back onto the simplex using the Euclidean projection
of Duchi et al.  The algorithm proceeds with progressive up‑sampling,
starting from the previously known construction (if any) or a random
simplex point, and finishes with a short high‑β refinement.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal after projection
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Fast evaluation of the original score for a simplex‑normalised sequence."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _gradient_softmax(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the gradient of   (1/β)·log Σ_k exp(β·conv_k(a))
    with respect to a, assuming a lies on the simplex.
    """
    n = a.shape[0]

    # ---- auto‑convolution of a (conv = a*a) ----
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]

    # ---- soft‑max weights ----
    max_val = conv.max()
    # shift by max to avoid overflow
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        # numerical safeguard – fall back to uniform weights
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum

    # ---- gradient = 2 * (p ⋆ a) ----
    # Correlation of p with a can be performed as a convolution of p
    # with the reversed a.
    a_rev = a[::-1]
    nfft = 1 << ((3 * n - 2).bit_length())
    Pf = np.fft.rfft(p, n=nfft)
    Ar = np.fft.rfft(a_rev, n=nfft)
    corr = np.fft.irfft(Pf * Ar, n=nfft)[: 3 * n - 2]

    # slice the part corresponding to shifts 0 … n−1
    grad = 2.0 * corr[n - 1 : 2 * n - 1]
    return grad

                break

            grad = _gradient_softmax(a, beta)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.02 / (math.sqrt(it) + 1.0)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            if it % 100 == 0 or it == max_refine_iters:
                cur_score = _score_of_seq(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] "
                        f"Final refine: n={n}, iter={it}, score={cur_score:.6f}"
                    )

        # one final check with the exact evaluation routine
        final_score = _score_of_seq(best_seq)
        if final_score < best_score:
            best_score = final_score

    # ------------------------------------------------------------------
    # 3. Safeguard fallback.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # return a plain Python list (required by the harness)
    return [float(x) for x in best_seq]
