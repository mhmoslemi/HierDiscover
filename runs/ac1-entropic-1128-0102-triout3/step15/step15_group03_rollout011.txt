def _lp_update(f, M, t, n):
    # Build constraint matrix
    # Row for each conv index (size 2n-1)
    # variable g of size n
    from scipy.optimize import linprog
    # Build sparse matrix A_ub shape (2n-1, n)
    # using lists
    rows = []
    cols = []
    data = []
    for i in range(n):
        # matrix entries for rows k from i to i+n-1
        rows.extend(range(i, i + n))
        cols.extend([i]*n)
        data.extend([f[i]]*n)
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f  # fallback
    g = res.x
    S = g.sum()
    if S == 0:
        return f
    # normalize g to have same sum as f (which currently sum=1)
    g_norm = g / S  # sum becomes 1
    # mix
    new_f = (1 - t) * f + t * g_norm
    # project onto simplex to keep sum=1 (should already be)
    new_f = _project_simplex(new_f)  # ensure positivity
    return new_f
```

"""
Propose a non‑negative step‑function (list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of several stages, each respecting the overall
wall‑clock budget (default 1000 s).

1.  A quick parametric sweep (geometric, power‑law, two‑exponential “U‑shaped”
    and a Gaussian family) to get a reasonable starting point.
2.  Many random Dirichlet initialisers.  Each is refined by a projected
    sub‑gradient descent that directly reduces the quantity
    ``max(conv(a,a))``; the descent uses a diminishing step‑size
    proportional to the gradient norm and projects back onto the probability
    simplex after every update.
3.  A final deterministic hill‑climb that moves tiny amounts of mass from the
    “over‑represented’’ index (according to the sub‑gradient) to the
    “under‑represented’’ one; it terminates when the allotted time expires.

All stages keep the sequence normalised to unit sum, so the reported score
is ``2·n·max(conv)`` – exactly the value returned by ``evaluate_sequence``.
The code uses only NumPy (FFT based convolution) and runs comfortably within
the 1 CPU limit.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (real vectors)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (vectors are assumed to be normalised to sum = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum = 1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgrad_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time()-start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 151):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time()-start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    r_grid = np.linspace(0.90, 0.995, 25)
    uniform = np.full(n, 1.0 / n)

    for r in r_grid:
        if time.time() >= deadline:
            break
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        for w_left in np.linspace(0.0, 0.6, 7):
            if time.time() >= deadline:
                break
            for w_right in np.linspace(0.0, 0.6, 7):
                w_unif = 1.0 - w_left - w_right
                if w_unif < 0.0:
                    continue
                seq = w_left * left + w_right * right + w_unif * uniform
                seq /= seq.sum()
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time()-start_time:.2f}s] TwoExp r={r:.3f} wL={w_left:.2f} wR={w_right:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_gaussian(n, best_seq, best_score, deadline):
    """Gaussian centre with optional uniform mixing."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sig_grid = np.logspace(np.log10(0.5), np.log10(n / 2.0), 30)

    for sigma in sig_grid:
        if time.time() >= deadline:
            break
        gauss = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        if gauss.sum() == 0:
            continue
        gauss /= gauss.sum()
        # pure Gaussian
        sc = _objective_score(gauss)
        if sc < best_score:
            best_score, best_seq = sc, gauss.copy()
            print(f"[{time.time()-start_time:.2f}s] Gauss σ={sigma:.3f} score={sc:.6f}")
        # mixture with uniform
        for lam in (0.2, 0.5, 0.8):
            mix = lam * gauss + (1.0 - lam) * (np.full(n, 1.0 / n))
            mix /= mix.sum()
            sc = _objective_score(mix)
            if sc < best_score:
                best_score, best_seq = sc, mix.copy()
                print(f"[{time.time()-start_time:.2f}s] Gauss σ={sigma:.3f} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------
def _random_dirichlet(n):
    rng = np.random.default_rng()
    logc = rng.uniform(-2.0, 1.0)          # concentration ≈ 0.01 … 10
    c = 10 ** logc
    a = rng.gamma(shape=c, scale=1.0, size=n)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray, n: int, deadline: float, cur_best: float):
    """
    Projected sub‑gradient descent that directly minimises
    max(conv(a,a)).  The iterate is always kept on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    it = 0
    step0 = 0.8

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time()-start_time:.2f}s] Descent it={it} score={cur_score:.6f}")

        # stop early if we have already beaten the global best and no further
        # improvement is likely within this short sub‑run
        if cur_best - best_score > 1e-4 and (time.time() > deadline - 0.5):
            break

        grad = _subgrad_maxconv(a, conv)
        gnorm = np.linalg.norm(grad) + 1e-12
        step = step0 / (np.sqrt(it) * gnorm)
        a = a - step * grad
        a = _project_simplex(a)

    return best_a, best_score

# Final hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray, deadline: float):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgrad_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        eps = a[src] * (rng.random() * 0.004 + 1e-6)   # up to 0.4 % of source
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time()-start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # occasional jitter to escape plateaus
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score
