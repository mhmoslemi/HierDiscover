"""
Improved search for a low‑score step‑function.

The algorithm builds a large initial pool of candidates (including a
deterministic Sidon‑set construction), then runs a smooth‑max
gradient descent using an exponentiated‑gradient (mirror‑descent)
update.  The smoothed objective
   F(a)= (1/α)·log∑exp(α·conv(a,a))
has gradient 2·corr(a,w) where w is the soft‑max over the
convolution entries; this yields a dense, well‑behaved descent
direction.  A final short simulated‑annealing polish is applied.
All computation is done with FFTs, so the method scales to vectors
of size ≈10⁵ within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Full self‑convolution (linear, length 2·n‑1) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max of a vector and the associated soft‑max weights
# ----------------------------------------------------------------------
def _smooth_max_weights(conv: np.ndarray, alpha: float):
    """Return (smooth_max, weights) where weights sum to 1."""
    # shift for numerical stability
    m = conv.max()
    exp_vals = np.exp(alpha * (conv - m))
    sum_exp = exp_vals.sum()
    # smoothed max = m + log(sum_exp)/α
    smooth = m + np.log(sum_exp) / alpha
    w = exp_vals / sum_exp
    return smooth, w


# ----------------------------------------------------------------------
# Gradient of the smoothed max w.r.t. a (via cross‑correlation)
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a F where F is the smoothed max defined above.
    The gradient is 2 * correlation(a, w) for shifts 0…n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]                     # 2·n‑1
    L = _next_pow2(n + m - 1)          # size for linear correlation
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w, n=L)
    # correlation = ifft( Af * conj(Wf) )
    corr_full = np.fft.irfft(Af * np.conj(Wf), n=L)[: n + m - 1]
    # linear correlation: shift s appears at index (m‑1)+s
    grad = 2.0 * corr_full[m - 1 : m - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (Σ a = 1, a ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators (same as previous version)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    s = shape.sum()
    shape = shape / s if s > 0 else np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    s = shape.sum()
    shape = shape / s if s > 0 else np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    s = shape.sum()
    shape = shape / s if s > 0 else np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Sidon set construction (B₂ sequence) – deterministic candidate
# ----------------------------------------------------------------------
def _sidon_candidate(n: int) -> np.ndarray:
    """Place equal mass on a Sidon set of size ≈ sqrt(n)."""
    # largest prime ≤ sqrt(n)
    limit = int(math.isqrt(n))
    p = limit
    while p > 2:
        is_prime = True
        for d in range(2, int(math.isqrt(p)) + 1):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p -= 1
    # generate Sidon set { i·p + (i² mod p) : i=0…p‑1 }
    positions = [i * p + (i * i) % p for i in range(p) if i * p + (i * i) % p < n]
    k = len(positions)
    vec = np.zeros(n, dtype=float)
    if k == 0:
        return vec
    weight = 1.0 / k
    vec[positions] = weight
    return vec


# ----------------------------------------------------------------------
# Keep the K best candidates (sorted by score)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (mirror‑descent)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    rng: np.random.Generator,
                    eta0: float = 1.0,
                    alpha_start: float = 1.0,
                    alpha_max: float = 200.0,
                    alpha_mul: float = 1.001,
                    max_exp: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed max‑convolution.
    Returns the best probability vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    it = 0
    alpha = alpha_start
    while time.time() < deadline:
        it += 1

        # exact objective for monitoring
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Descent iter={it:,} "
                  f"new best={cur_score:.6f}")

        # ---- smoothed max and soft‑max weights ----
        smooth, w = _smooth_max_weights(conv, alpha)

        # ---- gradient of smoothed max ----
        g = _grad_smooth_max(a, w)

        # ---- exponentiated‑gradient (KL‑mirror) update ----
        step = eta0 / math.sqrt(it)
        # avoid overflow/under‑flow in the exp
        expo = np.exp(-np.clip(step * g, -max_exp, max_exp))
        a = a * expo
        # simple projection onto simplex (renormalise)
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum) or np.isinf(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ---- slowly increase smoothing parameter ----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional random jitter to escape flat regions
        if it % 4000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * 0.015 * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.04) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = 2.0 * len(a) * _conv_self(a).max()
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.25
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        new_score = 2.0 * n * _conv_self(a_prop).max()

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} "
                      f"new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats (not required to
    sum to 1 – the evaluator normalises internally).
    """
    rng = np.random.default_rng(seed + 12345)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a discretisation size (larger N gives more freedom)
    # --------------------------------------------------------------
    # Use the full budget if possible; we try for N=131072.
    remaining = deadline - time.time()
    if remaining > 800:
        N = 131072
    elif remaining > 500:
        N = 65536
    elif remaining > 250:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build an initial pool of diverse candidates
    # --------------------------------------------------------------
    TOP_K = 12
    pool = []   # list of (score, vector)

    # Uniform
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, 2.0 * N * _conv_self(uniform).max(), uniform, TOP_K)

    # Power‑ramps
    for p in np.linspace(0.2, 2.2, 10):
        for plat in np.linspace(0.0, 0.5, 6):
            a = _make_shape(N, p, plat)
            _push_top(pool, 2.0 * N * _conv_self(a).max(), a, TOP_K)

    # Beta shapes
    for alpha in np.arange(0.05, 1.05, 0.05):
        a = _make_beta_shape(N, alpha)
        _push_top(pool, 2.0 * N * _conv_self(a).max(), a, TOP_K)

    # Gaussian‑like bumps
    for sigma in [N * 0.02, N * 0.05, N * 0.12]:
        a = _make_gaussian_shape(N, sigma)
        _push_top(pool, 2.0 * N * _conv_self(a).max(), a, TOP_K)

    # Random Dirichlet draws (very sparse)
    for conc in [0.001, 0.005, 0.01, 0.03, 0.07]:
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            _push_top(pool, 2.0 * N * _conv_self(a).max(), a, TOP_K)

    # Deterministic Sidon set candidate
    try:
        sid = _sidon_candidate(N)
        if sid.sum() > 0:
            sid /= sid.sum()
            _push_top(pool, 2.0 * N * _conv_self(sid).max(), sid, TOP_K)
    except Exception:
        pass

    # Use previously best sequence if present (up‑sample if needed)
    try:
        prev = globals().get('height_sequence_1')
        if isinstance(prev, (list, tuple, np.ndarray)):
            prev = np.asarray(prev, dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                if prev.size != N:
                    x_old = np.linspace(0, 1, prev.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    prev_up = np.interp(x_new, x_old, prev)
                else:
                    prev_up = prev.copy()
                prev_up = np.maximum(prev_up, 0.0)
                if prev_up.sum() > 0:
                    prev_up /= prev_up.sum()
                    _push_top(pool,
                              2.0 * N * _conv_self(prev_up).max(),
                              prev_up,
                              TOP_K)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Pick the current best candidate
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Smoothed‑max gradient descent on the best few candidates
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # allocate about 80 % of remaining time for descent, split among
        # the top 3 candidates
        descent_time = time.time() + remaining * 0.80
        per_candidate = (descent_time - time.time()) / 3.0
        for i in range(min(3, len(pool))):
            cand = pool[i][1]
            sub_deadline = time.time() + per_candidate
            refined, score = _smooth_descent(cand, sub_deadline,
                                             start_time, rng,
                                             eta0=1.2,
                                             alpha_start=0.8,
                                             alpha_max=250.0,
                                             alpha_mul=1.002,
                                             max_exp=70.0)
            if score < best_score:
                best_score, best_seq = score, refined.copy()
                print(f"[{time.time() - start_time:.2f}s] After descent best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Short SA polish (if any time left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, rng,
                                                    max_move_frac=0.04)

    # --------------------------------------------------------------
    # 4) Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = 2.0 * N * _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
