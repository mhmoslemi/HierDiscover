def _smooth_max_gradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    beta = 1.0
    step_scale = 0.1
    iteration = 0
    while time.time() < deadline:
        # gradually increase beta
        if iteration % 1000 == 0 and beta < 256:
            beta *= 2

        conv = _conv_self(a)
        max_val = conv.max()
        # compute softmax weights
        exp_term = np.exp(beta * (conv - max_val))
        w = exp_term / exp_term.sum()

        # gradient
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)  # length 3n-2
        grad = 2.0 * grad_full[n-1:n-1+n]

        # step size
        gnorm = np.linalg.norm(grad) + 1e-12
        step = step_scale / gnorm

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # update best
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[...] Smooth descent it={iteration} beta={beta:.1f} score={cur_score:.6f}")

        a = a_new
        iteration += 1
    return best_seq, best_score
```

"""
Propose a non‑negative step‑function (list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines:
  1. Fast parametric families (geometric, power, beta, U‑shaped arcsine).
  2. Many random Dirichlet draws.
  3. A smooth‑max gradient descent (log‑sum‑exp surrogate) with an
     accelerating β schedule and projection onto the probability simplex.
  4. A final small‑mass hill‑climb to fine‑tune the best candidate.

All stages respect the 1000 s wall‑clock budget.  The best sequence found
is returned as a plain Python list ready for ``evaluate_sequence``.
"""
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full result)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (expects sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalized to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) (average over all active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣ Parametric families
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for eps in np.logspace(-5, -1, 180):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 251):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        if seq.sum() == 0:
            continue
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_beta(n, best_seq, best_score, deadline):
    """Symmetric and asymmetric beta families (a,b ∈ (0,4])."""
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.05, 4.0, 81)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta a={a:.2f} b={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_exponential(n, best_seq, best_score, deadline):
    """Mixture of left‑geometric, right‑geometric and uniform components."""
    idx = np.arange(n, dtype=np.float64)
    r_grid = np.linspace(0.90, 0.995, 25)          # geometric ratio
    for r in r_grid:
        if time.time() >= deadline:
            break
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        uniform = np.full(n, 1.0 / n)

        for w_left in np.linspace(0.0, 0.6, 7):
            if time.time() >= deadline:
                break
            for w_right in np.linspace(0.0, 0.6, 7):
                w_unif = 1.0 - w_left - w_right
                if w_unif < 0.0:
                    continue
                seq = w_left * left + w_right * right + w_unif * uniform
                seq /= seq.sum()
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] TwoExp r={r:.3f} wL={w_left:.2f} wR={w_right:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_arcsine(n, best_seq, best_score, deadline):
    """U‑shaped “arcsine” density: 1/√(i+½)(n‑½‑i)."""
    idx = np.arange(n, dtype=np.float64)
    seq = 1.0 / np.sqrt((idx + 0.5) * (n - 0.5 - idx))
    seq /= seq.sum()
    sc = _objective_score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Arcsine init score={sc:.6f}")
    return best_seq, best_score

# 2️⃣ Random initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=800):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration in [0.01, 3] (log‑uniform)
        logc = rng.uniform(-2.0, 0.48)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(n, best_seq, best_score, deadline, trials=300):
    """Random mixtures of left‑geometric, right‑geometric and uniform."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.80, 0.999)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] RandMix r={r:.3f} w={w} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Smooth‑max gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(a0: np.ndarray, deadline, best_seq, best_score):
    """
    Minimise 2·n·max(conv) using a smooth log‑sum‑exp surrogate.
    β grows geometricly (1 → 256) to sharpen the approximation.
    """
    a = a0.copy()
    n = a.shape[0]

    beta = 1.0          # start smoothness
    max_beta = 256.0
    step_scale = 0.12   # initial step size (will be normalised)
    it = 0

    while time.time() < deadline:
        it += 1

        # increase β every 2000 iterations (or sooner if close to deadline)
        if it % 2000 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        conv = _conv_self(a)                    # length 2n‑1
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it} β={beta:.1f} score={cur_score:.6f}")

        # ----- smooth max (log‑sum‑exp) -----
        max_val = conv.max()
        exp_term = np.exp(beta * (conv - max_val))   # stable
        w = exp_term / exp_term.sum()                # soft‑max weights (length 2n‑1)

        # gradient of the surrogate (see analysis)
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)              # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # normalise the step
        gnorm = np.linalg.norm(grad) + 1e-12
        step = step_scale / gnorm

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # gentle decay of step scale
        step_scale *= 0.9997
        if step_scale < 5e-5:
            step_scale = 5e-5

        a = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣ Directed hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.005 + 1e-7)   # up to 0.5 % of source
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                # small random jitter to escape flat plateaus
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.005 + 1e-7)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score
