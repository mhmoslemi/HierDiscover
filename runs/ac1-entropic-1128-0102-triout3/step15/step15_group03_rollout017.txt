"""
Propose a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The search runs inside a 1000 s wall‑clock budget and consists of three
phases:

1️⃣  **Broad random exploration** – a few thousand Dirichlet draws give a
    good starting point that is already noticeably better than the uniform
    baseline.

2️⃣  **Alternating linear‑programming flattening** – this follows the
    state‑of‑the‑art iteration described in the problem statement.
    For a fixed sequence ``f`` we compute ``M = max(f∗f)`` and then solve

        maximise   Σ b_j
        subject to b ≥ 0 ,  f∗b ≤ M   (entrywise)

    The programme is a pure LP (the convolution is linear in ``b``); we
    solve it with CVXPY/ECOS.  The optimal ``b`` is renormalised to a
    probability vector and mixed with the current iterate
    ``f ← (1‑t)·f + t·b`` (t = 0.2).  Repeating this step pushes the
    self‑convolution towards the constant value 1/(2·n‑1) and typically
    drives the score well below the previous 1.5078.

3️⃣  **Sub‑gradient descent on the max‑convolution** – after the LP
    phase we run a projected sub‑gradient method that directly
    minimises ``max(conv(a,a))``.  The sub‑gradient is given by
    ``_subgradient_maxconv``.  A diminishing step size (∝1/√k) and a
    projection onto the probability simplex keep the iterate feasible.
    The final few seconds are spent in a tiny hill‑climb that shuffles
    minute amounts of mass from the “over‑represented’’ index to the
    “under‑represented’’ one.

All heavy linear‑algebra (full convolutions) uses an FFT based routine,
and the Euclidean projection onto the simplex is O(n log n).  The
implementation respects the 1000 s time limit and prints intermediate
scores for monitoring.  The returned list is ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using the FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    return irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    return irfft(Af * Af, n=L)[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a sequence with sum(a)=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣ Random Dirichlet initialise
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=500):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration in [0.01, 5]
        logc = rng.uniform(-2.0, 0.7)
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} "
                  f"score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Alternating LP flattening
# ----------------------------------------------------------------------
def _lp_flatten(f: np.ndarray, t: float, deadline, best_seq, best_score):
    """
    One LP‑based flattening step.
    Returns the updated sequence, its score and the (possibly) improved best.
    """
    n = f.shape[0]
    M = _conv_self(f).max()
    try:
        import cvxpy as cp
    except Exception:
        # LP phase unavailable – simply return the original vector
        return f, _objective_score(f), best_seq, best_score

    # Build and solve the LP: maximise sum(b) subject to f* b ≤ M, b ≥ 0
    b = cp.Variable(n, nonneg=True)
    conv_expr = cp.conv(cp.Constant(f), b)          # length 2n‑1
    constraints = [conv_expr <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # Solve – we use a modest iteration limit to stay within the budget
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1500,
               abstol=1e-7, reltol=1e-7, feastol=1e-7)

    if b.value is None:
        # solver failed – abort the LP phase
        return f, _objective_score(f), best_seq, best_score

    b_opt = np.maximum(np.asarray(b.value).flatten(), 0.0)
    sum_b = b_opt.sum()
    if sum_b < 1e-12:
        return f, _objective_score(f), best_seq, best_score

    # Renormalise to a probability vector
    b_norm = b_opt / sum_b

    # Convex combination with the current iterate
    f_new = (1.0 - t) * f + t * b_norm
    f_new = _project_simplex(f_new)     # guarantees sum=1, non‑negative

    cur_score = _objective_score(f_new)
    if cur_score < best_score:
        best_score, best_seq = cur_score, f_new.copy()
        print(f"[{time.time() - start_time:.2f}s] LP step score={cur_score:.6f}")

    return f_new, cur_score, best_seq, best_score

# 3️⃣ Sub‑gradient descent on max(conv)

def _subgrad_descent_max(a0: np.ndarray, deadline, best_seq, best_score):
    """Projected sub‑gradient descent directly on max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it} "
                  f"score={cur_score:.6f}")

        # Sub‑gradient of the max‑convolution
        grad = _subgradient_maxconv(a, conv)
        norm = np.linalg.norm(grad) + 1e-12
        step = 0.5 / np.sqrt(it) / norm     # classic 1/√t schedule

        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

    return best_seq, best_score

def _hill_climb(start_a, deadline):
    """Very small deterministic transfers that usually shave the last bits."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # source = index that contributes most to the current max
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)          # index most “under‑represented”

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        # move a modest fraction of src to dst
        eps = a[src] * (rng.random() * 0.02 + 1e-6)   # up to 2 % of src
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                # random jitter to escape a plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.01 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score

    Search for a step‑function that minimises the evaluation function.
    The algorithm follows three stages (random Dirichlet, alternating LP
    flattening, sub‑gradient descent) and respects a 1000 s wall‑clock
    budget.  The returned list is ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate n makes the LP cheap while still
    # allowing a very low theoretical bound 2·n/(2·n‑1)≈1.0.
    # --------------------------------------------------------------
    n = 2048
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Random Dirichlet exploration (≈120 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 500.0     # keep plenty of time for later phases
    if time.time() < phase_deadline:

            n, best_seq, best_score, phase_deadline, trials=800)

    # --------------------------------------------------------------
    # 2️⃣ Alternating LP flattening (≈260 s)
    # --------------------------------------------------------------
    lp_deadline = deadline - 250.0
    t_mix = 0.2                # mixing factor (1‑t)·f + t·g
    max_lp_iters = 40
    lp_iter = 0
    while time.time() < lp_deadline and lp_iter < max_lp_iters:
        best_seq, cur_score, best_seq, best_score = _lp_flatten(
            best_seq, t_mix, lp_deadline, best_seq, best_score)
        lp_iter += 1

    # --------------------------------------------------------------
    # 3️⃣ Sub‑gradient descent on the max‑convolution (≈210 s)
    # --------------------------------------------------------------
    subgrad_deadline = deadline - 30.0
    if time.time() < subgrad_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_seq, best_score = _subgrad_descent_max(
            best_seq, subgrad_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After sub‑grad score={best_score:.6f}")

    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise to a probability vector and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
