def _subgrad_descent(a0, deadline, start_time, rng, step0=500.0, min_step=1e-6):

    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it=0
    while time.time() < deadline:
        it +=1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2*len(a)*max_val # same as _objective
        if cur_score < best_score:
            best_score = cur_score

"""
Fast multi‑start mirror‑descent / simulated‑annealing optimizer.

* Build a diverse initial pool (uniform, power‑ramps, Beta, Gaussian,
  exponential‑decay, power‑law, Dirichlet draws and the previous best if
  available).  All candidates are normalised to sum 1.

* Keep the TOP_K best candidates (lowest score).  Re‑use the pool as a
  source of restarts.

* Repeatedly pick a candidate from the pool and run a short
  exponentiated‑gradient (mirror‑descent) optimisation on the exact
  objective  `2·n·max(conv(a,a))`.  The sub‑gradient of the max‑convolution
  is computed analytically; the update is `a_i ← a_i·exp(‑η·g_i)` followed
  by a simplex projection (the exponentiated‑gradient is a natural mirror
  descent on the probability simplex).

* Occasionally perform a cheap random “mass‑move’’ to escape flat regions
  and, when the budget is low, a short simulated‑annealing polish.

* The whole search respects the 1000 s budget; the best vector found is
  returned as a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (fast convolution of a vector with itself)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (size 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Objective for a probability vector (sum == 1).  value = 2·n·max(conv)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used only as a safety net.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape factories (all produce a probability vector of length n)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with optional flat plateau."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape = shape.astype(float)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential_shape(n: int, lam: float) -> np.ndarray:
    """Monotone exponential decay, normalised."""
    shape = np.exp(-lam * np.arange(n, dtype=float) / n)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Power‑law (i+1)^{-p}, normalised."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Pool handling – keep the best K candidates
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the sorted pool if it belongs to the top k."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Sub‑gradient / mirror‑descent on the exact objective
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     rng: np.random.Generator,
                     step0: float = 600.0,
                     max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent (mirror descent) on
    f(a)=2·n·max(conv(a,a)).
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SG iter={it:,} best={cur_score:.6f}")

        # --- sub‑gradient of the max‑convolution --------------------
        ks = np.where(conv >= max_val - 1e-14)[0]          # indices where max is attained
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # --- exponentiated‑gradient update -------------------------
        step = step0 / math.sqrt(it)
        expo = np.exp(-step * g)
        a = a * expo
        s = a.sum()
        if s > 0 and np.isfinite(s):
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # --- occasional random mass move (escape) ------------------
        if it % 2000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Simple mass‑swap SA on the true objective.
    """
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------

    """
    rng = np.random.default_rng(seed + 12345)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Choose a discretisation size (large for final quality)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()

        N = 65536
    elif remaining > 500:
        N = 32768
    else:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    TOP_K = 12

    for p in np.linspace(0.2, 3.0, 13):
        for plat in np.linspace(0.0, 0.6, 9):
            a = _make_shape(N, p, plat)
            _push_top(pool, _objective(a), a, TOP_K)

    # symmetric Beta shapes
    for alpha in np.arange(0.05, 0.96, 0.05):

    # Gaussian‑like bumps
    for sigma in [N * 0.03, N * 0.07, N * 0.15]:

    # power‑law tails (decreasing)
