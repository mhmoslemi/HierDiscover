    while time.time() < deadline:
        # compute grad for source/dest selection

        grad = _subgradient_maxconv(a, conv)
        src = np.argmax(grad)
        dst = np.argmin(grad)
        # compute max_frac based on remaining time

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
    else:
        w = np.full_like(v, 1.0/n)
    return w

"""
search for a non‑negative step function (probability vector) that
minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣  A quick set of random Dirichlet initialisations to obtain a decent
    starting point (the best of the trials is kept).

2️⃣  Simulated‑annealing on the simplex.  At each iteration a small amount
    of mass is transferred from the index that contributes most to the
    current maximal convolution (obtained from a sub‑gradient) to the index
    that contributes least.  Moves that increase the objective are accepted
    with probability `exp(-(Δ)/T)` and the temperature `T` is cooled
    geometrically.  The allowed transfer fraction is reduced slowly so the
    search becomes finer with time.

3️⃣  A final projected‑gradient descent (back‑tracking line‑search) that
    repeatedly moves opposite to the sub‑gradient of the max‑convolution
    and projects back onto the probability simplex.  This fine‑tunes the
    best candidate found by the annealer.

All heavy work (convolution, gradient) uses an FFT based routine and the
simplex projection uses the O(n) algorithm of Duchi et al.  The whole
search respects the 1000 s budget; if the time runs out the best vector
found so far is returned as a plain Python list.
"""

import time
import numpy as np

# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1). Uses an FFT of length power‑of‑2."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, so denominator equals 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Random Dirichlet trials
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Simulated annealing on the simplex
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, cur_score: float, deadline: float):
    """Anneal the vector by moving mass between the most/least contributing indices."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # temperature starts as a small fraction of the current objective
    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9999
    max_frac_start = 0.30   # early moves may transfer up to 30 % of source mass
    max_frac_end = 0.01     # late moves become very fine‑grained

    while time.time() < deadline:
        # current gradient – useful for picking source / destination
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        # linearly decay the allowed transfer fraction
        t_rem = deadline - time.time()
        total_rem = deadline - start_time
        frac = max_frac_end + (max_frac_start - max_frac_end) * max(t_rem / total_rem, 0.0)
        delta = a[src] * rng.random() * frac
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        if cand[src] < 0.0:
            continue
        cand_score = _objective_score(cand)
        diff = cand_score - cur_score
        # Metropolis acceptance
        if diff <= 0.0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best {best_score:.6f}")
        # cool down
        T = max(T * decay, 1e-12)
    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Projected gradient descent (back‑tracking line search)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray, cur_score: float, deadline: float):
    """Fine‑tune the vector by moving opposite to the sub‑gradient and re‑projecting."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    step = 0.5          # initial step length (in the gradient direction)
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        # try a few decreasing step factors
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            # keep feasibility (non‑negative) before projection
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)   # shrink step if no progress
        else:
            # modestly increase step to accelerate when things go well
            step = min(step * 1.05, 1.0)
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Locate a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain list) found within the time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) if it exists in the environment
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet initialisers (≈30 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 30.0)
    if time.time() < dirichlet_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=300, deadline=dirichlet_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] After Dirichlet best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Simulated annealing – the heavy lifting phase
    # ------------------------------------------------------------------
    anneal_deadline = min(deadline, start_time + 720.0)   # up to ~12 min
    if time.time() < anneal_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(best_seq, best_score, anneal_deadline)
        print(f"[{time.time() - start_time:.2f}s] Annealing finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final projected‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting projected‑gradient descent")
        best_seq, best_score = _projected_gradient_descent(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Gradient descent finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
