        # compute conv
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:

        expargs = np.clip(expargs, -700, 0)  # exp(-700) ~ 5e-305

            w = w / w_sum

"""
Step‑function optimizer – smoothed exponentiated‑gradient descent.

Goal
----
Minimise   2·n·max(conv(a,a))   over non‑negative vectors a with
∑a = 1 (the evaluator rescales any positive vector to this simplex).

Method
------
1.  Broad random Dirichlet draws to obtain a decent initial point.
2.  Smoothed exponential‑gradient descent on the probability simplex.
    The non‑smooth max‑operator is approximated by a soft‑max
    (log‑sum‑exp) with a temperature ε that is slowly increased.
    The gradient of the smooth surrogate is
        ∇_a f = 2·corr(w , a[::-1])
    where w are the soft‑max weights of the convolution entries.
    An entropic (multiplicative) update
        a_i ← a_i·exp(−η·∇_i)
    is performed and a is renormalised.
    η decays like 1/√t; ε grows geometrically to tighten the approximation.
3.  A final fine‑grained convex combination with the uniform vector.

The routine is time‑aware (≈1000 s), prints progress and returns the
best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used during optimisation: 2·n·max(conv(a,a))."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

    while time.time() < deadline:
        # concentration drawn log‑uniformly in [0.02, 20]
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        total = sample.sum()
        if total <= 0:
            continue
        a = sample / total
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smoothed exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_expgrad_descent(
    a0: np.ndarray,
    n: int,
    L_fft: int,
    L_corr: int,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Mirrors descent with a soft‑max approximation of the max‑convolution.
    ε (smoothness) starts small and grows; step size η decays as 1/√t.
    """
    a = a0.copy()
    it = 0
    # smoothing schedule
    eps = 20.0          # start temperature
    eps_max = 300.0     # final temperature
    eps_mult = 1.001    # multiplicative increase per iteration
    # step‑size schedule
    eta0 = 0.6

    while time.time() < deadline:
        it += 1
        # ---------- current objective ----------
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD it={it:,} score={cur_score:.6f}")

        # ---------- soft‑max weights ----------
        conv_max = conv.max()
        # for stability shift by the maximum before exponentiation
        expargs = eps * (conv - conv_max)
        # Clip to avoid overflow/underflow
        expargs = np.clip(expargs, -700, 0)   # exp(-700) ≈ 5e‑305
        w = np.exp(expargs)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the smooth surrogate ----------
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        # compute via convolution of w with reversed a
        a_rev = a[::-1]
        A_rev_fft = np.fft.rfft(a_rev, n=L_corr)
        W_fft = np.fft.rfft(w, n=L_corr)
        corr = np.fft.irfft(A_rev_fft * W_fft, n=L_corr)
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # length n

        # ---------- multiplicative update ----------
        eta = eta0 / np.sqrt(it)
        exp_factor = np.exp(-eta * grad)
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a_new = a * exp_factor
        s = a_new.sum()
        if s == 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / s

        # ---------- accept‑or‑reject ----------
        conv_new = _conv_self(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            a = a_new
        else:
            # simulated‑annealing style acceptance
            T0 = 1e-3
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = T0 * (remaining / total)
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a_new

        # ---------- schedule adjustments ----------
        eps = min(eps_max, eps * eps_mult)   # slowly tighten the max approximation

        # occasional tiny random jitter to help escape plateaus
        if it % 5000 == 0:
            jitter = np.random.normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final fine‑grained mixing with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(
    a: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Scan λ∈[0,1] for seq = λ·a + (1‑λ)·uniform and keep the best."""
    uniform = np.full(n, 1.0 / n)
    lam_grid = np.linspace(0.0, 1.0, 41)  # coarse sweep
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        sc = _objective_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`. The routine respects the time budget and prints
    intermediate scores.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.20

    # problem size – 4096 proved effective in earlier experiments
    n = 4096
    L_fft = _next_pow2(2 * n - 1)      # for self‑convolution
    L_corr = _next_pow2(3 * n - 2)     # for correlation in gradient

    # ---------- baseline (uniform) ----------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # ---------- optional global starter ----------
    starter_res = _maybe_load_global_starter(n)
    if starter_res is not None:
        st_seq, st_score = starter_res
        if st_score < best_score:
            best_seq, best_score = st_seq.copy(), st_score
            print(f"[0.00s] Global starter improved score={st_score:.6f}")

    # ---------- Phase 1: random Dirichlet ----------
    phase1_deadline = deadline - 300.0   # keep ~5 min for later phases
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_search(
            n, best_seq, best_score, phase1_deadline
        )

    # ---------- Phase 2: smoothed exponentiated‑gradient descent ----------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed exp‑grad descent")
        best_seq, best_score = _smooth_expgrad_descent(
            best_seq,
            n,
            L_fft,
            L_corr,
            start_time,
            deadline,
            best_seq,
            best_score,
        )
        print(f"[{time.time() - start_time:.2f}s] After exp‑grad best score≈{best_score:.6f}")

    # ---------- Phase 3: mix with uniform ----------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Mixing with uniform")
        best_seq, best_score = _mix_with_uniform(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # final normalisation and conversion to list
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
