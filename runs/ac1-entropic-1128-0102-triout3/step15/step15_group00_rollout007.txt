"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in four stages within a 1000 s budget:

1️⃣  A fast parametric sweep (geometric & exponential families) to obtain a
    decent starting point.
2️⃣  Random Dirichlet draws with a wide range of concentration parameters.
3️⃣  Smooth‑max gradient descent: we minimise a smooth approximation of the
    max‑convolution (log‑sum‑exp) by projected gradient steps on the simplex.
    The smoothing parameter β is increased gradually so that the optimisation
    moves from a convex landscape to the true objective.
4️⃣  A final hill‑climb that transfers mass from indices with large gradient
    to indices with small gradient, with occasional random shakes.

All updates are projected onto the probability simplex (∑a_i = 1, a_i ≥ 0)
using the O(n) algorithm of Duchi et al.  The routine is written without
closures so that it can be imported and called directly.  The best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is kept = 1 → denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Simple parametric families
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    """Geometric decay mixed with uniform."""
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 120)               # fewer points for speed
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential decay."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score

# 2️⃣  Random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 400
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration c ∈ [1e‑3, 10] on a log‑scale
        logc = rng.uniform(-3.0, 1.0)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smooth‑max gradient descent
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns a smooth approximation of max(conv(a,a)) (log‑sum‑exp) and its
    gradient w.r.t. a.  The gradient is computed via two FFT‑based convolutions.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2·n‑1
    max_val = conv.max()
    # stable softmax on the shift axis
    w_raw = np.exp(beta * (conv - max_val))
    sum_w = w_raw.sum()
    w = w_raw / sum_w                         # weights sum to 1

    # gradient = conv(a, w) + conv(a, w_rev) (central part only)
    start = n - 1
    conv_aw = _conv_fft(a, w)                # length 3·n‑2
    part1 = conv_aw[start:start + n]
    w_rev = w[::-1]
    conv_aw_rev = _conv_fft(a, w_rev)
    part2 = conv_aw_rev[start:start + n]
    grad = part1 + part2

    smooth_val = max_val + np.log(sum_w) / beta
    return smooth_val, grad


def _smooth_descent(a0: np.ndarray, deadline, best_seq, best_score):
    """
    Projected gradient descent on the smooth max‑convolution.
    The smoothing factor β is increased gradually so that the method
    first sees a convex landscape and then focuses on the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    beta = 1.0           # start with a mild smoothing
    beta_max = 200.0
    eta0 = 0.15          # base step size

    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        # smooth objective & gradient
        smooth_val, grad = _smooth_max_and_grad(a, beta)

        # true objective (cheap to compute once per 10 iterations)
        if it % 10 == 0:
            cur_score = _objective_score(a)
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SmoothDescent it={it} beta={beta:.1f} score={cur_score:.6f}")

        # centred gradient → preserves the simplex constraint
        grad_centered = grad - grad.mean()
        eta = eta0 / (1.0 + it * 1e-4)
        a = a - eta * grad_centered
        a = _project_simplex(a)

        # increase smoothing slowly
        if it % 200 == 0:
            beta = min(beta * 1.15, beta_max)

        # occasional random shake‑up to escape plateaus
        if it % 1500 == 0:
            eps = rng.normal(scale=0.02, size=n)
            a = a + eps
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (larger transfer steps)
# ----------------------------------------------------------------------
def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        max_frac = 0.12                     # allow larger moves than before
        frac = rng.random() * max_frac + 1e-5
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-14:            # safety check
            continue

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill++ score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                # random shake‑up
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps2 = a[i1] * (rng.random() * max_frac + 1e-5)
                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine consists of a parametric sweep, random Dirichlet draws,
    a smooth‑max gradient descent, and a final hill‑climb.  The final
    vector (normalised to sum = 1) is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size – using a larger n gives a tighter lower bound.
    # --------------------------------------------------------------
    n = 8192                     # feel free to adjust; larger n → tighter bound
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1, if defined)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.size != n:
                # stretch / truncate to the chosen size
                if starter.size < n:
                    # pad with zeros
                    pad = np.zeros(n - starter.size)
                    starter = np.concatenate([starter, pad])
                else:
                    starter = starter[:n]
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep (fast)
    # --------------------------------------------------------------
    phase_deadline = deadline - 400.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max gradient descent
    # --------------------------------------------------------------
    md_deadline = deadline - 150.0
    if time.time() < md_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_seq, best_score = _smooth_descent(best_seq, md_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final directed hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
