assistantfinal**Solution Explanation**

The evaluation function  

\[
\text{score}(a)=\frac{2\,n\;\max(a*a)}{(\sum_i a_i)^2}
\]

is homogeneous – scaling the vector does not change the score – therefore we may
restrict ourselves to *probability vectors* (`a_i ≥ 0`, `∑ a_i = 1`).  
With this normalisation  

\[
\text{score}(a)=2\,n\;\max (a*a)
\]

so the task is to make the maximal entry of the self‑convolution as small as
possible.

The search consists of three phases

1. **Coarse optimisation on a small grid**  
   A short but intensive search on a small problem (`n≈512`) is cheap and
   produces a good “shape’’ of the distribution.  
   • random draws from Dirichlet distributions (many concentration values)  
   • a few steps of exponentiated‑gradient descent on the exact `max‑conv`
   objective  

2. **Upscaling** – the small vector is interpolated to the target length
   (`n=4096`).  Because the objective is scale‑invariant, the shape obtained on
   the coarse grid remains a good starting point for the full problem.

3. **Refinement on the full grid** – three standard gradient‑based blocks are
   executed one after another, each using the remaining time budget.  
   – *Adam* on a smooth‑max surrogate (log‑sum‑exp approximation)  
   – *Exponentiated gradient* on the exact `max` (sub‑gradient)  
   – *Simulated‑annealing / hill‑climbing* with adaptive temperature.

All steps respect the global time limit (default 1000 s).  The best vector
encountered during the whole run is returned.

The code below implements exactly this strategy; it can be used as a drop‑in
replacement for the previous `propose_candidate` function.

```python
"""
Multi‑scale optimisation for the self‑convolution L∞‑objective.

The algorithm works in three stages:
  1. Small‑scale search (n≈512) – many random Dirichlet draws + a short
     exponentiated‑gradient descent on the exact max‑convolution.
  2. Upscaling the best small vector to the target size (n=4096) by linear
     interpolation.
  3. Full‑scale refinement on the target size using
        * Adam on a smooth‑max (log‑sum‑exp) surrogate,
        * exponentiated‑gradient descent on the exact max‑convolution,
        * a simulated‑annealing hill‑climb.

All operations keep the vector on the probability simplex (non‑negative,
sum = 1).  The best vector seen at any time is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a∗a (full length) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max(conv)  at  a .
    For each index attaining the maximum the corresponding Jacobians are summed.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]          # contribution from a_i * a_{k-i}
    if active.size:
        grad = 2.0 * grad / active.size   # factor 2 because of symmetry
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score for a probability vector – 2·n·max(conv(seq,seq))."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Small‑scale optimisation (n≈512)
# ----------------------------------------------------------------------
def _small_scale_opt(n_small: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Returns a good probability vector of length n_small.
    The routine uses random Dirichlet draws and a short EG descent.
    """
    rng = np.random.default_rng()
    # start with uniform
    best = np.full(n_small, 1.0 / n_small)
    best_score = _objective_score(best)

    # ---- random Dirichlet -------------------------------------------------
    while time.time() < deadline:
        # log‑uniform concentration in [0.01, 10]
        log_c = rng.uniform(np.log(0.01), np.log(10.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n_small)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best, best_score = a, sc
            print(f"[{time.time() - start_time:.2f}s] small Dirichlet c={c:.3f} score={sc:.6f}")

        # keep sampling until the deadline for the small phase is reached
    # ---- tiny EG descent -------------------------------------------------
    # one pass of exponentiated gradient (larger step than in the large phase)
    a = best.copy()
    L_fft = _next_pow2(2 * n_small - 1)
    eta = 0.2                     # relatively aggressive
    cur_score = best_score
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)
        eta_t = eta / (1.0 + it * 1e-4)
        exp_factor = np.exp(np.clip(-eta_t * grad, -700, 700))
        a_new = a * exp_factor
        a_new = np.maximum(a_new, 1e-12)
        a_new /= a_new.sum()
        new_score = _objective_score(a_new)
        # accept if improvement
        if new_score < cur_score - 1e-12:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best, best_score = a_new.copy(), new_score
                print(f"[{time.time() - start_time:.2f}s] small EG it={it:,} score={new_score:.6f}")
        else:
            # slight decay of step size at stagnation
            eta *= 0.99
    return best, best_score


# ----------------------------------------------------------------------
# 2️⃣  Adam optimisation of the smooth‑max surrogate (large n)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    tau = 20.0
    tau_factor = 1.2
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of soft‑max
        a_rev = a[::-1]
        conv_w = np.fft.irfft(np.fft.rfft(w, n=_next_pow2(3 * n - 2)) *
                              np.fft.rfft(a_rev, n=_next_pow2(3 * n - 2)),
                              n=_next_pow2(3 * n - 2))[:n]
        grad = 2.0 * conv_w

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Exponentiated‑gradient descent on exact max‑convolution (large n)
# ----------------------------------------------------------------------
def _eg_maxconv_descent(a0: np.ndarray,
                        start_time: float,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the sub‑gradient of the L∞‑norm.
    The update is  a ← normalize(a * exp(-η·grad)).
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.05                 # initial step size
    it = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        eta_t = eta / (1.0 + it * 1e-4)

        exp_factor = np.exp(np.clip(-eta_t * grad, -700, 700))
        a_new = a * exp_factor
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        new_score = _objective_score(a_new)

        if new_score < cur_score - 1e-12:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] EG it={it:,} score={new_score:.6f}")
        else:
            # gentle decay of the step size
            eta *= 0.99
            # occasional random shake to escape plateaus
            if np.random.rand() < 0.001:
                a, cur_score = a_new, new_score
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb / simulated‑annealing (large n)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb.
    At each step a tiny amount of mass is moved from a “source’’ index to a
    “destination’’ index; moves are accepted with the Metropolis rule.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # temperature schedule (linear decay to ~1e‑6)
    T0 = 0.02
    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # move a *relative* amount; larger range than the original script
        frac = rng.random() * 0.02 + 1e-6        # up to 2 % of source mass
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a, cur_score = a_prop, new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²).

    The algorithm proceeds in three stages:
        1) small‑scale random/EG search,
        2) up‑sampling to the target length,
        3) full‑scale Adam → EG → hill‑climb refinement.

    The best sequence found (as a list of floats) is returned.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Small‑scale optimisation (n≈512)
    # ------------------------------------------------------------------
    n_small = 512
    small_deadline = min(deadline, start_time + 150.0)   # at most 150 s
    best_small, best_small_score = _small_scale_opt(n_small, small_deadline)
    print(f"[{time.time() - start_time:.2f}s] Small‑scale best score={best_small_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Upscaling to the final size (n=4096)
    # ------------------------------------------------------------------
    n_large = 4096
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_large)
    a_large = np.interp(x_large, x_small, best_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    best_seq = a_large.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Upscaled score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Full‑scale refinement (remaining time)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left <= 0:
        # time exhausted – return what we already have
        best_seq = best_seq / best_seq.sum()
        return [float(x) for x in best_seq]

    # split the remaining budget into three roughly equal parts
    portion = time_left / 3.0
    adam_deadline = time.time() + portion
    best_seq, best_score = _adam_optimize(best_seq, start_time, adam_deadline,
                                          best_seq, best_score)

    eg_deadline = time.time() + portion
    best_seq, best_score = _eg_maxconv_descent(best_seq, start_time, eg_deadline,
                                               best_seq, best_score)

    final_deadline = deadline
    best_seq, best_score = _final_hill_climb(best_seq, start_time, final_deadline,
                                             best_seq, best_score)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n_large, 1.0 / n_large)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
