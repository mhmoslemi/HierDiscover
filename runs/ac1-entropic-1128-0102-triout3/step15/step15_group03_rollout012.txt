def hill_climb_optimal(a, deadline):
    n = len(a)
    conv = _conv_self(a)  # length 2n-1
    M = conv.max()
    best_a = a.copy()
    best_score = 2*n*M
    no_improve = 0
    while time.time() < deadline:
        # compute active indices
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        if active.size == 0: break
        # compute gradient (subgrad) using loops as before (could reuse function)
        grad = _subgradient_maxconv(a, conv)
        src = np.argmax(grad)
        dst = np.argmin(grad)
        if src == dst:
            # random fallback
            src, dst = np.random.choice(n, size=2, replace=False)
        # compute diff vector shape (2n-1)
        diff = np.zeros(2*n - 1, dtype=np.float64)
        # add a contributions
        diff[dst:dst+n] += a
        diff[src:src+n] -= a
        # compute possible epsilon based on non-active entries where diff>0
        # for indices where diff>0, we need epsilon <= (M - conv[i]) / diff[i]
        # if diff <= 0, they do not increase max, ignore.
        mask_pos = diff > 1e-15
        if np.any(mask_pos):
            eps_candidates = (M - conv[mask_pos]) / diff[mask_pos]
            # Ensure positive
            eps_candidates = eps_candidates[eps_candidates > 0]
            if eps_candidates.size > 0:
                eps_star = np.min(eps_candidates)
            else:
                # no positive diff entries? Then moving will only decrease max, can take max allowed epsilon = a[src]
                eps_star = a[src]
        else:
            # all diff <=0, we can move as much as possible limited by a[src]
            eps_star = a[src]
        # Also clip eps_star by a[src] to ensure nonnegativity
        eps_star = min(eps_star, a[src])
        if eps_star <= 0:
            # break if cannot move
            break
        # Update a
        a[src] -= eps_star
        a[dst] += eps_star
        # Update conv linearly
        conv = conv + eps_star * diff
        M = conv.max()
        cur_score = 2 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill-opt step score={best_score:.6f}")
        else:
            no_improve += 1
        # Add occasional random jitter if stuck
        if no_improve > 2000:
            # random small redistribution
            i1, i2 = np.random.choice(n, size=2, replace=False)
            eps_jitter = a[i1] * (np.random.random() * 0.01 + 1e-6)

            M = conv.max()
            no_improve = 0

"""
Improved search for a non‑negative step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is deliberately simple yet effective:

1.  Load the globally stored best construction (``height_sequence_1``) if
    available, otherwise start from a uniform distribution.
2.  Run a few quick random Dirichlet initialisations to see whether a
    better starting point exists.
3.  Perform a *deterministic* hill‑climb that moves mass between two
    indices in the *optimal* amount (the largest ε that does not make any
    convolution entry exceed the current peak).  The direction is chosen
    from the sub‑gradient of the peak‑convolution.
4.  Occasionally inject a tiny random jitter to escape plateaus.

All linear algebra uses FFT based convolution for speed, the vector is
kept normalised (sum = 1) throughout, and the whole search respects the
1000 s wall‑clock budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
#  Global timer (set inside ``propose_candidate``)
# ----------------------------------------------------------------------
start_time = None

# ----------------------------------------------------------------------
#  Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using the FFT.
    Length of the result is len(x)+len(y)-1.
    """
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
#  Objective helpers (assume sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return the evaluation score 2·n·max(conv(a,a)) for a normalised vector."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
#  Simple random initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_once(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector via a Dirichlet distribution."""
    logc = rng.uniform(-2.0, 0.7)          # concentration between ~0.01 and 5
    c = 10 ** logc
    a = rng.gamma(shape=c, scale=1.0, size=n)
    s = a.sum()
    if s == 0:
        return None
    return a / s


# ----------------------------------------------------------------------
#  Deterministic optimal‑step hill‑climb
# ----------------------------------------------------------------------
def hill_climb_optimal(a_start: np.ndarray, deadline: float):
    """
    Move mass from the index with largest sub‑gradient to the one with
    smallest sub‑gradient.  The transferred amount ε is the *largest*
    possible that does not let any entry of the self‑convolution exceed
    the current peak.  This yields a fast, monotone descent.
    """
    a = a_start.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    M = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # sub‑gradient direction
        grad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        # diff = shift(dst) - shift(src)
        diff = np.zeros(2 * n - 1, dtype=np.float64)
        diff[dst:dst + n] += a
        diff[src:src + n] -= a

        # compute admissible ε (largest that keeps all conv entries ≤ M)
        pos = diff > 1e-15                     # only entries that would increase
        if np.any(pos):
            cand = (M - conv[pos]) / diff[pos]   # candidate ε for each rising entry
            cand = cand[cand > 0]                # keep only positive candidates
            eps_star = np.min(cand) if cand.size else a[src]
        else:
            eps_star = a[src]                     # everything only goes down

        # never move more than we own
        eps_star = min(eps_star, a[src])
        if eps_star <= 0:
            # fallback – tiny random jitter
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 0.01 + 1e-8)
            eps = min(eps, a[i1])
            a[i1] -= eps
            a[i2] += eps
            conv = _conv_self(a)
            M = conv.max()
            continue

        # apply the optimal move
        a[src] -= eps_star
        a[dst] += eps_star
        conv += eps_star * diff
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill‑opt improved score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 5000:
                # random jitter to escape a plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps = a[i1] * (rng.random() * 0.01 + 1e-8)
                eps = min(eps, a[i1])
                a[i1] -= eps
                a[i2] += eps
                conv = _conv_self(a)
                M = conv.max()
                no_improve = 0

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the evaluation function.
    The routine respects the 1000 s wall‑clock budget and returns a plain
    Python list ready for ``evaluate_sequence``.
    """
    global start_time
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (keep moderate to stay within the time limit)
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Initialise from the global best construction if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter /= s
                best_seq = starter.copy()
            else:
                best_seq = np.full(n, 1.0 / n)
        else:
            best_seq = np.full(n, 1.0 / n)
    except Exception:
        best_seq = np.full(n, 1.0 / n)

    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Quick random Dirichlet tries (very cheap)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(120):
        if time.time() >= deadline:
            break
        cand = _random_dirichlet_once(n, rng)
        if cand is None:
            continue
        sc = _objective_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Random Dirichlet improved score={sc:.6f}")

    # ------------------------------------------------------------------
    # Deterministic optimal‑step hill‑climb for the rest of the budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting optimal‑step hill‑climb")
        best_seq, best_score = hill_climb_optimal(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Hill‑climb finished – best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return the result as a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score≈{_objective_score(best_seq):.6f}")
    return [float(x) for x in best_seq]
