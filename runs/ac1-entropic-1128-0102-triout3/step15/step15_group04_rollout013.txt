"""
Mirror‑descent on the soft‑max of the self‑convolution.

The search uses three phases:

1️⃣  **Initialization** – start from several points (uniform, the global
    starter if present and a few random perturbations).  A vector *x*
    represents a probability distribution via the soft‑max
    a_i = exp(x_i) / Σ_j exp(x_j); this guarantees non‑negativity and a
    unit sum without an explicit projection.

2️⃣  **Adam (mirror) optimisation** – minimise a smooth approximation of
    the max‑convolution.  For a temperature β we compute the soft‑max
    weights p_k ∝ exp(β·conv_k) and the exact gradient
    g_i = 2·Σ_j a_j·p_{i+j}.  The gradient w.r.t. the log‑variables x
    is g_x = a·(g – ⟨a,g⟩).  An Adam optimiser with a modest learning‑rate
    (decreased for large β) updates x.  β is increased in a continuation
    schedule (0.5 → … → 512) so that the optimisation first sees a smooth
    landscape and gradually focuses on the true L∞‑objective.

3️⃣  **Exact sub‑gradient polishing** – the best distribution found is
    further refined by a classic projected sub‑gradient method on the
    true objective (the same routine as the reference solution).

All phases respect a 1000 s time budget; the best sequence is normalised
to sum = 1 and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max objective:
        g_i = 2 * Σ_j a_j * p_{i+j}
    Computed via FFT as a cross‑correlation.
    """
    n = a.shape[0]
    # reverse the weight vector
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    conv_full = np.fft.irfft(Af * Pf, n=L_grad)
    # keep the central 2n‑1 entries (full linear convolution length)
    # then extract the n positions needed for the gradient
    start = n - 1
    g = conv_full[start:start + n][::-1]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Return the soft‑max weights p_k = exp(beta*conv_k) / Σ exp(.)
    with a shift for numerical stability.
    """
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

    Search for a non‑negative step‑function that minimises

            2 * n * max(conv(a, a)) / (sum(a)**2)

    followed by a short exact sub‑gradient polish.  It returns a list of
    floats (summing to 1) that is the best candidate found before the
    time budget expires.

    # ------------------------------------------------------------------

    L_grad = _next_pow2(3 * n - 2)   # length needed for the weighted gradient

            # aborts automatically when the global deadline is reached.
            iters_beta = 6000
            for it in range(iters_beta):
                if time.time() >= deadline:
                    break

                t_step += 1
                a = _a_from_x(x)                 # current probability vector

                # compute full self‑convolution and soft‑max weights

# Exact sub‑gradient descent used in the final polishing stage.
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float):
    """
    Projected sub‑gradient descent on   2·n·max(conv(a,a)) /
    (sum(a)**2).  The routine runs until *deadline* and returns the
    best vector and its score.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))

        # sub‑gradient for the index k where the max is attained
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        # diminishing step size
        step = step0 / (it ** 0.6)
        a_candidate = a - step * g
        # project onto the simplex (Euclidean projection)
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(
                f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                f"step={step:.5e} score={cur_score:.6f}"
            )
        else:
            # occasional uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # small jitter every few thousand iterations
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w
