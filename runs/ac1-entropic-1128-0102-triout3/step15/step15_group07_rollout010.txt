def _additive_grad_desc(a0, sub_deadline, start_time, rng, step0=0.5, max_move_frac=0.02):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < sub_deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # subgradient
        ks = np.where(conv >= max_val - 1e-12)[0]

        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            # a[k - idx] yields corresponding a values
            g[idx] += 2.0 * a[k - idx]

        # additive update
        step = step0 / math.sqrt(it)
        a = a - step * g
        # project to simplex
        a = _project_simplex(a)

        # occasional random move
        if it % 2000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()
    return best_a, best_score

"""
Hybrid multi‑restart optimizer for the self‑convolution L∞ problem.

1. Build a diverse initial pool (uniform, Beta, power‑ramps, Gaussians,
   Dirichlet draws and the previous best if available).  Keep the TOP_K
   lowest‑scoring candidates.

2. Repeatedly pick a pool member and improve it:
   * exponentiated‑gradient (mirror descent w.r.t. KL) for coarse
     exploration,
   * additive projected‑gradient (Euclidean mirror) for fine tuning.
   Both phases use the exact sub‑gradient of the max‑convolution
   (the set of indices attaining the current maximum).  The step size
   decays as 1/√t.

3. Occasionally mix two pool members (convex combination + tiny noise) to
   create fresh starting points.

4. When the time budget is low perform a short simulated‑annealing polish.

All updates respect the simplex (non‑negative, sum = 1).  The routine
returns a plain list of floats; the evaluation function will compute the
final bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two (FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Self‑convolution via FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
#   val = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical safety
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators for the initial pool
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Power‑ramp with optional flat plateau, symmetrised."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on {0,…,n‑1}, normalised."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Keep the TOP_K best candidates (sorted by score)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the true max‑convolution
# ----------------------------------------------------------------------
def _exp_grad_descent(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      rng: np.random.Generator,
                      step0: float = 300.0,
                      max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EXP it={it:,} best={cur_score:.6f}")

        # ----- sub‑gradient of the max convolution -----
        ks = np.where(conv >= max_val - 1e-14)[0]
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # ----- multiplicative (exponentiated) update -----
        step = step0 / math.sqrt(it)
        expo = np.exp(-step * g)
        a = a * expo
        # renormalise onto simplex
        s = a.sum()
        if s <= 0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random mass move to escape plateaux -----
        if it % 4000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Additive projected‑gradient descent (Euclidean mirror) using the same
# sub‑gradient.  Works well for fine‑grained polishing.
# ----------------------------------------------------------------------
def _additive_grad_descent(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           rng: np.random.Generator,
                           step0: float = 0.8,
                           max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ADD it={it:,} best={cur_score:.6f}")

        ks = np.where(conv >= max_val - 1e-14)[0]
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        step = step0 / math.sqrt(it)
        a = a - step * g
        a = _project_simplex(a)

        if it % 3000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Very light simulated‑annealing polish (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mix two candidates (convex combination + tiny random jitter)
# ----------------------------------------------------------------------
def _mix_vectors(v1: np.ndarray, v2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    w = rng.random()
    mixed = w * v1 + (1.0 - w) * v2
    # add a barely perceptible jitter to avoid exact ties
    jitter = 1e-12 * rng.random(mixed.shape)
    mixed += jitter
    mixed = np.maximum(mixed, 0.0)
    s = mixed.sum()
    if s > 0:
        mixed /= s
    else:
        mixed = np.full_like(mixed, 1.0 / mixed.shape[0])
    return mixed


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats summing to 1.
    """
    rng = np.random.default_rng(seed + 123456)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose a discretisation size adapted to the remaining budget
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build an initial pool of diverse candidates
    # --------------------------------------------------------------
    TOP_K = 8
    pool = []   # list of tuples (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # power‑ramps
    for p in np.linspace(0.2, 2.0, 9):
        for plat in np.linspace(0.0, 0.4, 5):
            a = _make_shape(N, p, plat)
            _push_top(pool, _objective(a), a, TOP_K)

    # symmetric Beta shapes (including U‑shaped)
    for alpha in np.arange(0.08, 1.00, 0.08):
        a = _make_beta_shape(N, alpha)
        _push_top(pool, _objective(a), a, TOP_K)

    # Gaussian‑like bumps
    for sigma in (N * 0.04, N * 0.08, N * 0.16):
        a = _make_gaussian_shape(N, sigma)
        _push_top(pool, _objective(a), a, TOP_K)

    # random Dirichlet draws (varied concentration)
    concs = [0.02, 0.07, 0.15, 0.35, 0.6]
    for conc in concs:
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            _push_top(pool, _objective(a), a, TOP_K)

    # include the previous best if it exists
    try:
        prev = globals().get('height_sequence_1')
        if prev is not None:
            prev = np.asarray(prev, dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                if prev.size != N:
                    # linear interpolation to required size
                    x_old = np.linspace(0, 1, prev.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    prev_up = np.interp(x_new, x_old, prev)
                else:
                    prev_up = prev.copy()
                prev_up = np.maximum(prev_up, 0.0)
                if prev_up.sum() > 0:
                    prev_up /= prev_up.sum()
                    _push_top(pool, _objective(prev_up), prev_up, TOP_K)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # keep refining until time runs out
    while True:
        remaining = deadline - time.time()
        if remaining < 6.0:
            break

        # pick a candidate from the top‑K (biased towards the current best)
        idx = rng.integers(min(len(pool), TOP_K))
        cur_score, cur_vec = pool[idx]

        # split the remaining budget for this refinement
        sub_deadline = time.time() + min(remaining * 0.3, 30.0)

        # 1) exponentiated‑gradient phase
        exp_dead = time.time() + (sub_deadline - time.time()) * 0.55
        vec, sc = _exp_grad_descent(cur_vec, exp_dead, start_time, rng,
                                    step0=350.0, max_move_frac=0.025)

        # 2) additive projected‑gradient phase on the remaining slice
        if time.time() < sub_deadline:
            vec, sc = _additive_grad_descent(vec, sub_deadline, start_time, rng,
                                            step0=0.9, max_move_frac=0.025)

        # push the refined result back into the pool
        _push_top(pool, sc, vec, TOP_K)

        # occasional mixing of two random pool members
        if rng.random() < 0.25 and len(pool) >= 2:
            v1 = pool[rng.integers(len(pool))][1]
            v2 = pool[rng.integers(len(pool))][1]
            mixed = _mix_vectors(v1, v2, rng)
            mixed_score = _objective(mixed)
            _push_top(pool, mixed_score, mixed, TOP_K)

        # update the global best for printing
        if pool[0][0] < best_score:
            best_score, best_seq = pool[0][0], pool[0][1].copy()
            print(f"[{time.time() - start_time:.2f}s] New global best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Short SA polish if any time remains
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, rng,
                                                    max_move_frac=0.03)

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
