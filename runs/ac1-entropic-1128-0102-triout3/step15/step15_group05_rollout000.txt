def _exp_gradient_descent(a0, deadline, start_time, best_score_init, step0=0.3, decay=0.9999, momentum=0.9):

   a = a0.copy()
   v = np.zeros_like(a)
   best_a = a.copy()
   best_score = best_score_init
   it = 0
   while time.time() < deadline:
       it += 1
       conv = _conv_self_fft(a)
       g = _compute_subgrad_multi(a, conv)
       # update momentum
       v = momentum * v + (1 - momentum) * g
       # step size decays
       eta = step0 * (decay ** it)
       # exponentiated update
       a_new = a * np.exp(-eta * v)
       # projection onto simplex
       a_new = _project_simplex(a_new)
       cur_score = _eval_fast(a_new, _next_pow2(2*len(a)-1))
       if cur_score < best_score:
           best_score = cur_score
           best_a = a_new.copy()
           # maybe reset momentum
           v = np.zeros_like(a)
           a = a_new
       else:
           # occasionally accept uphill with small probability
           if np.random.rand() < np.exp(-(cur_score-best_score) * 0.1):
               a = a_new
   return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a multi‑stage stochastic / first‑order optimizer.

Stages
-------
0️⃣  Initialise on the simplex (uniform or global starter).
1️⃣  Randomised parametric search in a flexible 4‑parameter family
    a_i ∝ r**i·(i+1)^p·(n−i)^q mixed with uniform background.
2️⃣  Stochastic hill‑climb on the four parameters.
3️⃣  First‑order optimisation on the true objective:
    * Adam‑style projected optimisation on the simplex,
      using the sub‑gradient of the max‑convolution (active set).
4️⃣  Tiny random mass‑transfer hill‑climb for final polishing.

All work respects the 1000 s budget.  The best vector found is
normalised to sum = 1 and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective value given a self‑convolution array and length n,
    assuming the vector is already normalised to sum = 1."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Shape generation
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components
    proportional to r**i * (i+1)**p * (n-i)**q   (i = 0…n‑1).
    Computed in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    # log‑space expression; protect against log(0)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(np.maximum(n - i, 1.0))
    log_shape -= np.max(log_shape)               # shift for stability
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return lam·uniform + (1‑lam)·shape  (both already sum to 1).
    The result is renormalised to guard against numerical drift.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def _compute_subgrad_multi(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) with respect to a,
    aggregating over all shifts whose value is within `tol` of the maximum.
    """
    max_val = conv.max()
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        # contribution a_j to ∂/∂a_i
        g[i_vals] += a[j_vals]

        # symmetric contribution for i≠j (adds a_i to ∂/∂a_j)
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
    return g

def _random_param_search(n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray,
                                                     float,
                                                     tuple[float, float, float, float]]:
    """
    Uniform‑in‑log random sampling of (r,p,q,λ) with wider ranges.
    Keeps the best vector.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        # r in (0.5, 0.999999) – log‑uniform
        r = np.exp(rng.uniform(np.log(0.5), np.log(0.999999)))
        p = rng.uniform(-6.0, 6.0)
        q = rng.uniform(-6.0, 6.0)
        lam = rng.uniform(0.0, 1.0)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        # compute objective (sum is already 1)
        conv = _conv_self_fft(cand, L_fft)
        sc = _score_from_conv(conv, n)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam)
            print(f"[{time.time() - start_time:.2f}s] random search r={r:.4f} "
                  f"p={p:.3f} q={q:.3f} λ={lam:.3f} score={sc:.6f}")

    return best_seq, best_score, best_params

# Hill‑climb on the four shape parameters

def _hillclimb_params(initial_params: tuple[float, float, float, float],
                      start_seq: np.ndarray,
                      start_score: float,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb on (r,p,q,λ) with exponentially shrinking
    perturbations and occasional jitter.
    """
    cur_r, cur_p, cur_q, cur_lam = initial_params
    cur_seq = start_seq.copy()
    cur_score = start_score

    sigma_r = 0.020
    sigma_p = 0.35
    sigma_q = 0.35
    sigma_lam = 0.15

    rng = np.random.default_rng()
    n = cur_seq.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_seq = cur_seq.copy()
    best_score = cur_score

    while time.time() < deadline:
        # propose new parameters
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        q_new = cur_q + sigma_q * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        # clamp
        r_new = float(np.clip(r_new, 0.5, 0.999999))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_shape_general(n, r_new, p_new, q_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)

        conv = _conv_self_fft(cand_new, L_fft)
        sc_new = _score_from_conv(conv, n)

        if sc_new < cur_score:
            cur_r, cur_p, cur_q, cur_lam = r_new, p_new, q_new, lam_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(f"[{time.time() - start_time:.2f}s] param hill r={r_new:.6f} "
                      f"p={p_new:.4f} q={q_new:.4f} λ={lam_new:.4f} score={sc_new:.6f}")

        # decay perturbation scales
        sigma_r *= 0.9994
        sigma_p *= 0.9994
        sigma_q *= 0.9994
        sigma_lam *= 0.9994

        # occasional jitter
        if rng.random() < 0.006:
            cur_seq = np.maximum(cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq, 0.0)
            cur_seq = _project_simplex(cur_seq)
            conv = _conv_self_fft(cur_seq, L_fft)
            cur_score = _score_from_conv(conv, n)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam‑style projected optimisation on the simplex
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style update on the simplex using the sub‑gradient of
    max(conv(a,a)).  Momentum‑free restarts on improvement.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    best_a = a.copy()
    best_score = best_score_initial
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        conv = _conv_self_fft(a, L_fft)
        cur_score = _score_from_conv(conv, n)

        # sub‑gradient of max‑convolution (active set)
        g = _compute_subgrad_multi(a, conv, tol=1e-12)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        conv_cand = _conv_self_fft(a_candidate, L_fft)
        cand_score = _score_from_conv(conv_cand, n)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{time.time() - start_time:.2f}s] adam it={t:,} score={cand_score:.6f}")
        else:
            # occasional uphill acceptance (tiny probability)
            if rng.random() < 0.001:
                a = a_candidate

        # optional decay of learning rate for very long runs
        if t % 2000 == 0:
            lr *= 0.97

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two
    coordinates, keeping the vector on the simplex.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3   # ≤0.2 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        conv = _conv_self_fft(a_prop, L_fft)
        new_sc = _score_from_conv(conv, n)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            if rng.random() < 0.001:
                a = a_prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising the target
    evaluation function.  The routine honours the time budget and
    returns the best sequence (list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a good compromise between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Baseline (uniform) – also used as a fallback.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    base_conv = _conv_self_fft(uniform, L_fft)
    best_seq = uniform.copy()
    best_score = _score_from_conv(base_conv, n)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Pull in a globally supplied starter if present.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / s
            cur_conv = _conv_self_fft(starter, L_fft)
            cur_score = _score_from_conv(cur_conv, n)
            if cur_score < best_score:
                best_seq = starter.copy()
                best_score = cur_score
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    remaining = deadline - time.time()
    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep – generous budget.
    # ------------------------------------------------------------------
    sweep_deadline = time.time() + min(200.0, remaining * 0.25)
    rp_seq, rp_score, rp_params = _random_param_search(n, sweep_deadline, start_time)
    if rp_seq is not None and rp_score < best_score:
        best_seq = rp_seq.copy()
        best_score = rp_score
        best_params = rp_params
        print(f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}")
    else:
        # reasonable fallback if random sweep gave nothing better
        best_params = (0.97, 0.0, 0.0, 0.3)

    remaining = deadline - time.time()
    # ------------------------------------------------------------------
    # 2️⃣  Fine hill‑climb on the four shape parameters.
    # ------------------------------------------------------------------
    hill_deadline = time.time() + min(150.0, remaining * 0.2)
    best_seq, best_score = _hillclimb_params(best_params,
                                             best_seq,
                                             best_score,
                                             hill_deadline,
                                             start_time)

    remaining = deadline - time.time()
    # ------------------------------------------------------------------
    # 3️⃣  Adam‑style projected optimisation on the true objective.
    # ------------------------------------------------------------------
    adam_deadline = time.time() + min(400.0, remaining * 0.5)
    if time.time() < adam_deadline:
        a_adam, sc_adam = _adam_optimize(best_seq,
                                         adam_deadline,
                                         start_time,
                                         best_score)
        if sc_adam < best_score:
            best_seq = a_adam.copy()
            best_score = sc_adam
            print(f"[{time.time() - start_time:.2f}s] adam final score={best_score:.6f}")

    remaining = deadline - time.time()
    # ------------------------------------------------------------------
    # 4️⃣  Final cheap random “mass‑transfer” hill‑climb.
    # ------------------------------------------------------------------
    if remaining > 2.0:
        a_final, sc_final = _final_local_search(best_seq,
                                                deadline,
                                                start_time,
                                                best_score)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Log the official evaluation with the provided security‑checked function.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
