"""
Hybrid search for a low‑value step‑function.

The routine follows a multi‑stage strategy:

1️⃣  Build a large, diverse pool of candidate shapes (analytic families,
    random Dirichlet draws, noisy perturbations, etc.) on a modest size
    N≈8192 and keep the TOP_K best.

2️⃣  Run a *multi‑point Frank–Wolfe* (linear‑oracle) descent on each
    top candidate.  The linear oracle is cheap: we take the indices with
    smallest sub‑gradient of the max‑convolution and move a small
    convex‑combination of the current point towards the uniform mass on
    those indices.  This quickly reduces the peak of the self‑convolution.

3️⃣  Refine the best survivor with exponentiated sub‑gradient descent
    (multiplicative updates) and a final smooth‑max Adam polishing.

4️⃣  If time permits, up‑sample the best vector to a larger dimension
    (≈32768) and repeat steps 2–3.  Upsampling spreads the mass, often
    lowering the bound further.

All operations stay on the probability simplex (∑a=1, a≥0), use
FFT‑based convolutions for speed, and respect the 1000 s time budget.
The returned list is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog  # optional, not used in the fast path

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negatives caused by rounding.
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with optional flat top."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    shape /= shape.sum()
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    shape /= shape.sum()
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta density ∝ x^{α‑1}(1‑x)^{β‑1} on discrete grid."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_cosine_decay(n: int, power: float = 1.0, reverse: bool = False) -> np.ndarray:
    """a_i ∝ cos(π·i/(2·(n‑1)))^power, optionally mirrored."""
    x = np.linspace(0, np.pi / 2, n)
    shape = np.cos(x) ** power
    if reverse:
        shape = shape[::-1]
    shape /= shape.sum()
    return shape


def _make_gaussian_mixture(
    n: int,
    mu1: float,
    sigma1: float,
    mu2: float,
    sigma2: float,
    w1: float = 0.5,
) -> np.ndarray:
    x = np.arange(n, dtype=float)
    g1 = np.exp(-0.5 * ((x - mu1) / sigma1) ** 2)
    g2 = np.exp(-0.5 * ((x - mu2) / sigma2) ** 2)
    shape = w1 * g1 + (1.0 - w1) * g2
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _add_noise(shape: np.ndarray, scale: float = 0.01) -> np.ndarray:
    noisy = shape + np.random.normal(scale=scale, size=shape.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s > 0:
        noisy /= s
    else:
        noisy = shape.copy()
    return noisy


# ----------------------------------------------------------------------
# Gradient of the max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_i (a*a)_i w.r.t. a."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_val = conv.max()
    tol = max(1e-12, max_val * 1e-12)
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv)
    w[active] = 1.0 / max(active.size, 1)
    a_rev = a[::-1]
    L = _next_pow2(3 * n - 2)
    tmp = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
    grad = 2.0 * tmp[n - 1 : n - 1 + n]   # factor 2 from symmetry
    return grad


# ----------------------------------------------------------------------
# Multi‑point Frank–Wolfe descent (linear oracle = several best indices)
# ----------------------------------------------------------------------
def _multi_fw_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    batch: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style updates that move the current point towards a uniform
    distribution on the ``batch`` coordinates with smallest sub‑gradient.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad = _grad_max_conv(a)
        # pick ``batch`` indices with smallest gradient components
        idx = np.argpartition(grad, batch)[:batch]
        s = np.zeros_like(a)
        s[idx] = 1.0 / batch
        gamma = 2.0 / (it + 2.0)          # classic diminishing step‑size
        a = (1.0 - gamma) * a + gamma * s
        cur = _objective(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] FW iter={it:,} best={cur:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (multiplicative updates)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    it = 0
    eta0 = 0.7
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(active.size, 1)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]
        eta = eta0 / (1.0 + it / 3000.0)
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)
        cur = _objective(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max Adam polishing
# ----------------------------------------------------------------------
def _adam_smooth_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.12
    T0 = 0.3
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur < best_score:
            best_score = cur
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} best={cur:.6f}")

        # temperature schedule (exponential decay)
        T = T_end + (T0 - T_end) * (0.5 ** (it / 3500.0))

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best, best_score


# ----------------------------------------------------------------------
# Pool handling utilities
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Maintain a sorted top‑k pool (ascending scores)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low upper bound.
    The algorithm works in two resolution stages (≈8k → ≈32k)
    and combines analytic initialisation, multi‑point Frank‑Wolfe,
    exponentiated sub‑gradient descent and a final Adam polishing.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5       # tiny safety margin

    # ------------------------------------------------------------------
    # Stage 0 – choose a modest size for the exploratory pool.
    # ------------------------------------------------------------------
    if deadline - start > 800:
        N0 = 8192
    elif deadline - start > 400:
        N0 = 4096
    else:
        N0 = 2048

    TOP_K = 12
    pool = []   # elements are (score, vector)

    # --------------------------------------------------------------
    # Helper to push a candidate into the pool.
    # --------------------------------------------------------------
    def _add_candidate(vec: np.ndarray) -> None:
        # ensure non‑negative and normalised
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s > 0:
            vec = vec / s
        else:
            vec = np.full_like(vec, 1.0 / vec.shape[0])
        _push_top(pool, _objective(vec), vec, TOP_K)

    # --------------------------------------------------------------
    # 1) Analytic families (including more extreme Beta parameters)
    # --------------------------------------------------------------
    _add_candidate(_make_uniform(N0))

    # Power ramps
    for p in np.linspace(0.2, 4.0, 9):
        for pll in np.linspace(0.0, 0.45, 3):
            _add_candidate(_make_power_ramp(N0, p, pll))

    # Exponential decays
    for lam in np.linspace(0.2, 6.0, 12):
        _add_candidate(_make_exponential(N0, lam))

    # Power‑law decays
    for p in np.linspace(0.2, 4.0, 12):
        _add_candidate(_make_power_law(N0, p))

    # Asymmetric Beta – extend to very sharp ends
    for a in np.array([0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]):
        for b in np.array([0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]):
            _add_candidate(_make_beta_shape_asym(N0, float(a), float(b)))

    # Cosine‑decay families (left‑leaning and right‑leaning)
    for pw in [0.5, 1.0, 2.0]:
        _add_candidate(_make_cosine_decay(N0, power=pw, reverse=False))
        _add_candidate(_make_cosine_decay(N0, power=pw, reverse=True))

    # Random Dirichlet draws – both sorted (monotone) and unsorted.
    rng = np.random.default_rng(seed + 12345)
    for conc in [0.02, 0.05, 0.1, 0.3, 0.7, 1.5]:
        for _ in range(20):
            if time.time() >= deadline:
                break
            vec = rng.gamma(conc, 1.0, size=N0)
            # unsorted version
            _add_candidate(vec)
            # sorted decreasing version
            vec_sorted = np.sort(vec)[::-1]
            _add_candidate(vec_sorted)

    # Gaussian mixtures – randomised parameters.
    for _ in range(30):
        if time.time() >= deadline:
            break
        mu1 = rng.uniform(0, N0 * 0.4)
        sigma1 = rng.uniform(N0 * 0.02, N0 * 0.15)
        mu2 = rng.uniform(N0 * 0.6, N0)
        sigma2 = rng.uniform(N0 * 0.02, N0 * 0.15)
        w1 = rng.uniform(0.3, 0.7)
        mix = _make_gaussian_mixture(N0, mu1, sigma1, mu2, sigma2, w1)
        _add_candidate(mix)

    # Add a noisy perturbation of the best known global (if present).
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # Resample to current size if needed
            if starter.size != N0:
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N0, endpoint=False)
                starter = np.interp(x_new, x_old, starter)
            _add_candidate(_add_noise(starter, scale=0.02))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Identify the current best.
    # ------------------------------------------------------------------
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial pool best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Multi‑point Frank‑Wolfe on the top candidates.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        fw_deadline = time.time() + 0.45 * remaining
        # iterate cyclically over the pool
        ptr = 0
        while time.time() < fw_deadline:
            vec = pool[ptr % len(pool)][1]
            candidate, cand_score = _multi_fw_descent(vec, fw_deadline, start, batch=7)
            if cand_score < best_score:
                best_score, best_vec = cand_score, candidate.copy()
                print(f"[{time.time() - start:.2f}s] FW improvement → {cand_score:.6f}")
            ptr += 1

    # ------------------------------------------------------------------
    # Stage 2 – Exponentiated sub‑gradient descent (aggressive multiplicative).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        exp_deadline = time.time() + 0.40 * remaining
        candidate, cand_score = _exp_subgradient_descent(best_vec, exp_deadline, start)
        if cand_score < best_score:
            best_score, best_vec = cand_score, candidate.copy()
            print(f"[{time.time() - start:.2f}s] ExpGD improvement → {cand_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Adam polishing on the current best.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        adam_deadline = time.time() + 0.15 * remaining
        candidate, cand_score = _adam_smooth_descent(best_vec, adam_deadline, start)
        if cand_score < best_score:
            best_score, best_vec = cand_score, candidate.copy()
            print(f"[{time.time() - start:.2f}s] Adam improvement → {cand_score:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a larger resolution if time permits.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 100.0:
        # Double the resolution (or go to 32768 if we started smaller)
        N_big = max(32768, N0 * 2)
        # Linear interpolation to the larger grid.
        x_old = np.linspace(0, 1, best_vec.size, endpoint=False)
        x_new = np.linspace(0, 1, N_big, endpoint=False)
        upsampled = np.interp(x_new, x_old, best_vec)
        upsampled = np.maximum(upsampled, 0.0)
        upsampled /= upsampled.sum()

        # Re‑run the three optimisation phases on the up‑sampled vector.
        extra_deadline = time.time() + remaining - 0.05
        # Phase A – FW
        if time.time() < extra_deadline:
            fw_deadline2 = time.time() + 0.40 * (extra_deadline - time.time())
            candidate, cand_score = _multi_fw_descent(
                upsampled, fw_deadline2, start, batch=9
            )
            if cand_score < best_score:
                best_score, best_vec = cand_score, candidate.copy()
                print(f"[{time.time() - start:.2f}s] Upsampled FW → {cand_score:.6f}")

        # Phase B – ExpGD
        if time.time() < extra_deadline:
            exp_deadline2 = time.time() + 0.35 * (extra_deadline - time.time())
            candidate, cand_score = _exp_subgradient_descent(best_vec, exp_deadline2, start)
            if cand_score < best_score:
                best_score, best_vec = cand_score, candidate.copy()
                print(f"[{time.time() - start:.2f}s] Upsampled ExpGD → {cand_score:.6f}")

        # Phase C – Adam
        if time.time() < extra_deadline:
            adam_deadline2 = time.time() + 0.25 * (extra_deadline - time.time())
            candidate, cand_score = _adam_smooth_descent(best_vec, adam_deadline2, start)
            if cand_score < best_score:
                best_score, best_vec = cand_score, candidate.copy()
                print(f"[{time.time() - start:.2f}s] Upsampled Adam → {cand_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a plain Python list.
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.shape[0])

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished – final bound ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
