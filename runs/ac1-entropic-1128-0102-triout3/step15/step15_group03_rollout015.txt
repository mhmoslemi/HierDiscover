def _smooth_desc(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    beta = 1.0
    # schedule: increase beta after some iterations or after certain time intervals.
    # We'll maintain loop while time < deadline.
    # adapt step_scale.
    step_scale = 0.5
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        # actual score
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={iteration}, beta={beta:.2f} score={cur_score:.6f}")
        # compute gradient for smoothed max
        # compute softmax weights w
        # For numerical stability compute max_conv = conv.max()
        max_conv = conv.max()
        # to avoid overflow, shift by max_conv
        exp_term = np.exp(beta * (conv - max_conv))
        w = exp_term / exp_term.sum()
        # compute gradient
        a_rev = a[::-1]
        # conv of w with a_rev using FFT
        # length for conv: w length 2n-1, a_rev n
        L = _next_pow2(len(w) + len(a_rev) - 1)

        grad_slice = conv_full[n-1: n-1 + n]  # central part
        grad = 2.0 * grad_slice
        # step
        norm = np.linalg.norm(grad) + 1e-12
        step = step_scale / norm

def _solve_lp_for_g(a, M, time_limit=10.0):
    n = len(a)
    # Build A_ub matrix
    from scipy.linalg import toeplitz
    c = np.concatenate([a, np.zeros(n-1)])
    r = np.concatenate([np.array([a[0]]), np.zeros(n-1)])
    A = toeplitz(c, r)  # shape (2n-1, n)
    b = np.full(2*n-1, M)
    # objective to maximize sum(g)
    obj = -np.ones(n)
    bounds = [(0, None)] * n
    # Solve
    # Use HiGHS interior point; we can set options for max time if needed.
    # SciPy's linprog doesn't have time limit option directly, but we can manage by checking elapsed time before solving.
    res = linprog(obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        print(f"LP failed: {res.message}")
        return None
    return res.x

    a = a_init.copy()
    n = len(a)
    beta = 1.0
    step_scale = 0.5
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Compute current conv and score
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-9:

        # Compute gradient for smoothed max
        max_conv = conv.max()
        exp_terms = np.exp(beta * (conv - max_conv))
        w = exp_terms / exp_terms.sum()
        # gradient via FFT
        a_rev = a[::-1]
        L_fft = _next_pow2(len(w) + len(a_rev) - 1)
        Wf = np.fft.rfft(w, n=L_fft)

        grad = 2.0 * grad_slice
        # step
        norm = np.linalg.norm(grad) + 1e-12
        step = step_scale / norm
        a_new = a - step * grad

"""
Improved search for a step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages:

1.  Random Dirichlet initialisation (quick exploration).
2.  Alternating linear‑programming refinement:
        – given a, compute M = max(a*a).
        – solve   max Σ g_i   s.t.   a * g ≤ M ,  g ≥ 0 .
        – mix the solution into a with a small step size.
   The LP is dense but inexpensive for n≈2000.
3.  Smoothed‑max gradient descent (continuation on β):
        – optimise a smooth approximation of the L∞ norm
          using FFT‑based gradients and projection onto the simplex.
        – β grows slowly, turning the smooth problem into the original one.
4.  Fine‑grained hill‑climb that nudges tiny amounts of mass between
    indices to squeeze out the last bits of improvement.

All stages respect a total wall‑clock budget of 1000 s.
The returned sequence is normalised to sum=1, non‑negative and ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (always used)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (linear, not circular)."""
    return np.convolve(a, a)               # exact and fast for n≈2000


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assume sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return the raw evaluation score 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _smooth_max_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        L(a)= (1/β) log Σ exp(β·conv(a,a))
    evaluated at a (already normalised).  Returns ∇L.
    """
    n = a.shape[0]
    conv = _conv_self(a)                 # length 2n‑1
    shift = conv.max()
    exp_terms = np.exp(beta * (conv - shift))
    w = exp_terms / exp_terms.sum()      # softmax weights, Σ w = 1

    # conv of w with the reversed a (FFT‑based, O(n log n))
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    Wf = np.fft.rfft(w, n=L)
    Af = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(Wf * Af, n=L)[: len(w) + len(a_rev) - 1]
    # central slice corresponds to the derivative w.r.t each a_i
    grad_slice = full[n - 1: n - 1 + n]
    grad = 2.0 * grad_slice               # factor 2 from ∂conv/∂a
    return grad


def _solve_lp_for_g(a: np.ndarray, M: float):
    """
    Solve the LP:
        maximise Σ g_i
        s.t.   a * g ≤ M      (component‑wise linear convolution)
               g_i ≥ 0
    Returns the optimal g (numpy array) or None on failure.
    """
    n = a.shape[0]
    # Build the Toeplitz matrix that implements linear convolution a*g.
    # First column: a followed by n‑1 zeros, length 2n‑1.
    # First row: a[0] followed by n‑1 zeros.
    col = np.concatenate([a, np.zeros(n - 1)])
    row = np.concatenate([np.array([a[0]]), np.zeros(n - 1)])
    A = toeplitz(col, row)               # shape (2n‑1, n)
    b = np.full(2 * n - 1, M)

    # maximise Σ g_i  → minimise -Σ g_i
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        print(f"[LP] failed: {res.message}")
        return None
    return res.x


# ----------------------------------------------------------------------
# Search stages
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=300):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration parameter sampled log‑uniformly in [0.01, 5]
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


def _lp_refinement(a_start, deadline, best_seq, best_score):
    """
    Alternating linear‑programming refinement.
    Mixes the LP solution into the current point with a shrinking step size.
    """
    a = a_start.copy()
    n = len(a)
    mix = 0.20          # initial convex‑combination weight
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-9:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP it={iteration} score={cur_score:.6f}")

        g = _solve_lp_for_g(a, M)
        if g is None:
            break

        # normalise g to a proper probability vector (the LP only guarantees feasibility)
        s = g.sum()
        if s <= 0:
            break
        g = g / s

        # convex combination and projection
        a = (1.0 - mix) * a + mix * g
        a = _project_simplex(a)

        # slowly shrink the step size – this stabilises the process
        mix = max(0.01, mix * 0.97)

        # stop after a reasonable number of iterations if we are not improving
        if iteration >= 30:
            # keep iterating only while we still see improvement
            if time.time() > deadline - 10:
                break
    return best_seq, best_score


def _smooth_descent(a_start, deadline, best_seq, best_score):
    """
    Gradient descent on a smooth approximation of the max‑norm.
    The smoothing parameter β grows slowly (continuation method).
    """
    a = a_start.copy()
    n = len(a)
    beta = 1.0
    step_scale = 0.5
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-9:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={iteration} β={beta:.2f} score={cur_score:.6f}")

        # gradient of the smoothed objective
        grad = _smooth_max_gradient(a, beta)

        # normalised step
        gnorm = np.linalg.norm(grad) + 1e-12
        step = step_scale / gnorm

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # accept if the true max‑norm improves, otherwise shrink step
        new_score = _objective_score(a_new)
        if new_score < cur_score - 1e-9:
            a = a_new
            step_scale = min(1.0, step_scale * 1.02)   # gently increase step if we are moving downhill
        else:
            step_scale *= 0.7                          # otherwise become more cautious

        # increase β every few hundred iterations – makes the problem harder
        if iteration % 200 == 0:
            beta = min(beta * 1.1, 80.0)

        # stop condition if step becomes tiny
        if step_scale < 1e-6:
            step_scale = 1e-6
    return best_seq, best_score


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) with respect to a."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _hill_climb(start_a, deadline, best_seq, best_score):
    """
    Tiny mass transfers that greedily decrease the max‑convolution.
    Occasionally injects random jitter to escape plateaus.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = best_score
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)   # up to 0.4 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function defined in the challenge.  The routine respects
    an overall wall‑clock limit (default 1000 s) and returns the best list
    found.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate n keeps the LP tractable while still
    # allowing a very low theoretical bound (≈1.0).
    # ------------------------------------------------------------------
    n = 2048
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) – truncate / pad to n.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if len(starter) != n:
                if len(starter) > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - len(starter)), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet initialisation (broad exploration)
    # ------------------------------------------------------------------
    phase_deadline = start_time + budget_s * 0.25
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline, trials=350)

    # ------------------------------------------------------------------
    # 2️⃣  Alternating LP refinement
    # ------------------------------------------------------------------
    phase_deadline = start_time + budget_s * 0.55
    if time.time() < phase_deadline:
        best_seq, best_score = _lp_refinement(best_seq, phase_deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # 3️⃣  Smoothed‑max gradient descent (continuation on β)
    # ------------------------------------------------------------------
    phase_deadline = start_time + budget_s * 0.90
    if time.time() < phase_deadline:
        best_seq, best_score = _smooth_descent(best_seq, phase_deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # 4️⃣  Final directed hill‑climb for fine‑grained polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
