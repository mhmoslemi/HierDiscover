def evaluate_sequence(sequence: list[float]) -> float:
    # Security checks...
    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers

"""
A fast multi‑stage optimiser for the “max‑self‑convolution / sum²’’ problem.

Stages
-------
1️⃣  Randomised Dirichlet draws with *non‑uniform* concentration
    vectors – this explores a far richer family of shapes than a plain
    symmetric Dirichlet.

2️⃣  Projected gradient descent on a smooth‑max surrogate of the
    convolution.  The surrogate uses a soft‑max (temperature τ) and the
    gradient is obtained by a single FFT‑based convolution.
    The step size is tiny‑adjusted by a back‑tracking line‑search and τ
    is increased gradually so that the optimisation finally targets the
    true L∞‑norm.

3️⃣  Mass‑transfer simulated‑annealing (SA).  An active shift (the one
    attaining the current maximum) is identified, a source index inside
    the overlapping band is chosen (probability ∝ current mass) and a
    destination outside the band is chosen at random.  A small amount of
    mass is moved; the move is accepted according to a temperature schedule.
    This stage removes the remaining “active‑set’’ bottlenecks.

All time limits are measured against the global deadline (default 1000 s);
the routine always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑v=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for sequences with Σseq = 1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Random Dirichlet with per‑coordinate concentrations
# ----------------------------------------------------------------------
def _random_dirichlet_var(n: int, rng: np.random.Generator) -> np.ndarray | None:
    """
    Generate a Dirichlet draw with concentration α_i = c·(i+1)^p,
    where c>0 and p are sampled at random.
    """
    # pick random base concentration (log‑uniform) and random power
    c = np.exp(rng.uniform(np.log(0.01), np.log(10.0)))
    p = rng.uniform(-2.0, 2.0)
    idx = np.arange(n, dtype=np.float64)
    alphas = c * (idx + 1) ** p
    alphas = np.maximum(alphas, 1e-6)          # keep them strictly positive
    a = rng.gamma(shape=alphas, scale=1.0)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# 2️⃣  Projected gradient descent on a smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray, start_time: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Smoothed max optimisation using a soft‑max over convolution values.
    The gradient of the surrogate is computed via FFT, then a back‑track
    line‑search is performed.  τ (temperature) is increased gradually.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    tau = 5.0                     # initial temperature
    tau_factor = 1.5
    best_a = a.copy()
    best_score = _objective_score(a)

    while time.time() < deadline:
        # convolution and soft‑max weights
        conv = _conv_self(a, L_fft)
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of the smooth‑max: 2·a_rev (*) w
        a_rev = a[::-1]
        L = _next_pow2(len(a_rev) + len(w) - 1)
        grad = 2.0 * np.fft.irfft(
            np.fft.rfft(a_rev, n=L) * np.fft.rfft(w, n=L), n=L
        )[:n]

        # back‑tracking line‑search
        step = 0.02
        improved = False
        for _ in range(5):
            a_candidate = a - step * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)
            sc = _objective_score(a_candidate)
            if sc < best_score - 1e-12:
                a = a_candidate
                best_score = sc
                best_a = a.copy()
                improved = True
                # larger steps may be possible – keep the current step size
                break
            step *= 0.5                       # shrink step and retry
        if not improved:
            # no progress – increase the temperature to sharpen the objective
            tau = min(tau * tau_factor, 1e5)

        # occasional tiny random perturbation to escape plateaus
        if np.random.rand() < 0.01:
            noise = np.random.normal(scale=1e-4, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Simulated annealing using mass transfers
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, start_time: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from a source index inside the active
    (max‑conv) band to a destination outside it.  Accept uphill moves
    according to a temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # temperature schedule (exponential decay)
    T0 = 0.05
    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        max_idx = int(conv.argmax())
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)

        # choose a source index inside the band (prefer larger mass)
        if lo <= hi and np.any(a[lo:hi + 1] > 0):
            src_pool = np.arange(lo, hi + 1)
            probs = a[lo:hi + 1]
            probs = probs / probs.sum()
            src = rng.choice(src_pool, p=probs)
        else:
            src = rng.integers(n)

        # choose a destination outside the band
        if lo > 0:
            dst_pool = np.arange(0, lo)
        elif hi < n - 1:
            dst_pool = np.arange(hi + 1, n)
        else:
            dst_pool = np.arange(n)               # fallback (very unlikely)
        dst = rng.choice(dst_pool)

        # move a fraction of the source mass (up to a few percent)
        eps_frac = rng.random() * 0.02 + 1e-5
        eps = a[src] * eps_frac
        if eps <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score

        # temperature based on remaining time
        remaining = deadline - time.time()
        total = deadline - start_time
        frac = remaining / total
        T = T0 * ((1e-6 / T0) ** (1 - frac))

        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improved score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine implements the three stages described
    above and respects the overall time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # problem size (the original challenge used 4096)
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # initialise with the uniform distribution
    best_seq = np.full(n, 1.0 / n)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (if the previous run stored a sequence)
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1:
                # if the length differs, truncate or pad to n
                if starter.size > n:
                    starter = starter[:n]
                elif starter.size < n:
                    pad = np.zeros(n - starter.size)
                    starter = np.concatenate([starter, pad])
                starter = np.maximum(starter, 0.0)
                s = starter.sum()
                if s > 0:
                    starter = starter / s
                    sc = _objective_score(starter)
                    if sc < best_score:
                        best_seq, best_score = starter.copy(), sc
                        print(f"[{time.time() - start_time:.2f}s] Global starter improved score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet draws with per‑coordinate alphas
    # ------------------------------------------------------------------
    phase_deadline = deadline - 300.0   # keep ~5 min for later stages
    while time.time() < phase_deadline:
        a = _random_dirichlet_var(n, rng)
        if a is None:
            continue
        sc = _objective_score(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet‑var score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed‑max projected gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:   # reserve ~2 min for SA
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max GD")
        best_seq, best_score = _smooth_max_descent(best_seq, start_time, deadline - 120.0)
        print(f"[{time.time() - start_time:.2f}s] GD finished – best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated annealing (mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting SA")
        best_seq, best_score = _simulated_annealing(best_seq, start_time, deadline)
        print(f"[{time.time() - start_time:.2f}s] SA finished – final best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best overall score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
