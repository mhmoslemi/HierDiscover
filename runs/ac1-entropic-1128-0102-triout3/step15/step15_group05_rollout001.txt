"""
Step‑function optimiser.

The algorithm builds on the previous multi‑stage stochastic optimiser but
adds a **multi‑restart projected sub‑gradient descent** phase.  After a
short parametric sweep + hill‑climb (as before) we launch several
independent sub‑gradient runs, each started from a randomly perturbed
copy of the current best vector (or from a fresh Dirichlet sample).  The
runs share the same fast FFT convolution code and use momentum +
adaptive step‑size, but they are limited in time so that many restarts
can be tried within the 1000 s budget.  The best result among all runs
is then polished by a tiny random “mass‑transfer’’ hill‑climb.

This diversification often escapes the local minima that limited a
single long descent and pushes the bound well below the previous
≈ 1.507 – target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the full (2·len(a)‑1) array.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (unimodal)
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components
    proportional to  r**i * (i+1)**p * (n-i)**q   (i = 0…n‑1).
    Computed in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)                     # avoid overflow
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return lam·uniform + (1‑lam)·shape  (both already sum to 1).
    The result is renormalised to guard against numerical drift.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _compute_subgrad_multi(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) with respect to a, aggregating over
    all shifts whose value is within `tol` of the maximum.
    """
    max_val = conv.max()
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)

    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        # contribution a_j to ∂/∂a_i
        g[i_vals] += a[j_vals]

        # symmetric contribution for i≠j (adds a_i to ∂/∂a_j)
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def _subgradient_descent_multi(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               best_score_initial: float,
                               active_tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the true objective,
    using the active set of near‑maximal convolution entries.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    beta = 0.9
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=active_tol)

        # momentum update
        v = beta * v + (1.0 - beta) * g

        step = step0 / (it ** 0.6)
        a_candidate = a - step * v
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            v = np.zeros_like(a)          # reset momentum after improvement
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional uphill move
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter every few thousand iterations
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            v = np.zeros_like(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small random “mass‑transfer’’ hill‑climb
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two coordinates,
    staying on the simplex.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3           # ≤ 0.2 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            if rng.random() < 0.001:                # rare uphill acceptance
                a = a_prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Random parametric sweep (unimodal family)
# ----------------------------------------------------------------------


def _random_param_search(n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray,
                                                     float,
                                                     tuple[float, float, float, float]]:
    """
    Uniform‑in‑log random sampling of (r,p,q,λ) and keep the best vector.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        r = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))   # geometric decay
        p = rng.uniform(-3.0, 3.0)                               # polynomial left
        q = rng.uniform(-3.0, 3.0)                               # polynomial right
        lam = rng.uniform(0.0, 1.0)                              # uniform mixing

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam)
            print(f"[{time.time() - start_time:.2f}s] random sweep r={r:.4f} "
                  f"p={p:.2f} q={q:.2f} λ={lam:.2f} score={sc:.6f}")

    return best_seq, best_score, best_params


# ----------------------------------------------------------------------
# Hill‑climb on the four shape parameters
# ----------------------------------------------------------------------


def _hillclimb_params(initial_params: tuple[float, float, float, float],
                      start_seq: np.ndarray,
                      start_score: float,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb on (r,p,q,λ) with exponentially shrinking perturbations.
    """
    cur_r, cur_p, cur_q, cur_lam = initial_params
    cur_seq = start_seq.copy()
    cur_score = start_score

    sigma_r = 0.015
    sigma_p = 0.30
    sigma_q = 0.30
    sigma_lam = 0.12

    rng = np.random.default_rng()
    n = cur_seq.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_seq = cur_seq.copy()
    best_score = cur_score

    while time.time() < deadline:
        # propose new parameters
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        q_new = cur_q + sigma_q * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        # clamp
        r_new = float(np.clip(r_new, 0.85, 0.999999))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_shape_general(n, r_new, p_new, q_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = _eval_fast(cand_new, L_fft)

        if sc_new < cur_score:
            cur_r, cur_p, cur_q, cur_lam = r_new, p_new, q_new, lam_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(f"[{time.time() - start_time:.2f}s] param hill r={r_new:.6f} "
                      f"p={p_new:.4f} q={q_new:.4f} λ={lam_new:.4f} score={sc_new:.6f}")

        # decay perturbations
        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_q *= 0.9995
        sigma_lam *= 0.9995

        # occasional jitter
        if rng.random() < 0.005:
            cur_seq = np.maximum(cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq, 0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Multi‑restart sub‑gradient driver
# ----------------------------------------------------------------------


def _run_multi_subgrad(initial_best: np.ndarray,
                      initial_score: float,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Run several independent sub‑gradient descents, each with its own
    random start (jittered copy of the current best or fresh Dirichlet).
    Return the best vector found.
    """
    n = initial_best.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_seq = initial_best.copy()
    best_score = initial_score
    rng = np.random.default_rng()

    # keep a small safety margin for the final local search
    final_local_reserve = 30.0

    run_id = 0
    while time.time() < deadline - final_local_reserve:
        run_id += 1
        # pick a start point
        if rng.random() < 0.3:
            # jittered copy of current best
            eps = 1e-3 * rng.random(n) * best_seq
            start_a = np.maximum(best_seq + eps, 0.0)
        else:
            # fresh Dirichlet‑like random vector
            start_a = rng.exponential(scale=1.0, size=n)
            start_a = np.maximum(start_a, 0.0)

        start_a_sum = start_a.sum()
        if start_a_sum > 0.0:
            start_a /= start_a_sum
        else:
            start_a = np.full(n, 1.0 / n, dtype=np.float64)

        # allocate a few tens of seconds to each run (adaptive)
        remaining = deadline - time.time() - final_local_reserve
        run_time = min(40.0, remaining * 0.35)    # at most 40 s, otherwise a share
        if run_time < 5.0:
            break

        run_deadline = time.time() + run_time
        a_opt, sc_opt = _subgradient_descent_multi(start_a,
                                                   run_deadline,
                                                   start_time,
                                                   best_score)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] multi‑run #{run_id} "
                  f"improved score to {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising
        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine respects the 1000 s time budget and returns the best
    found sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 points gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline (score = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try to use a globally supplied starter if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / total
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_seq = starter.copy()
                best_score = cur
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep (geometric‑power‑tail family)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 120.0
    rp_seq, rp_score, rp_params = _random_param_search(n, sweep_deadline, start_time)
    if rp_seq is not None and rp_score < best_score:
        best_seq = rp_seq.copy()
        best_score = rp_score
        best_params = rp_params
        print(f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}")
    else:
        best_params = (0.97, 0.0, 0.0, 0.3)   # fallback parameters

    # ------------------------------------------------------------------
    # 2️⃣  Fine hill‑climb on the four parameters
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    hill_deadline = time.time() + min(120.0, remaining * 0.20)
    best_seq, best_score = _hillclimb_params(best_params,
                                             best_seq,
                                             best_score,
                                             hill_deadline,
                                             start_time)

    # ------------------------------------------------------------------
    # 3️⃣  Multi‑restart projected sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    sub_deadline = time.time() + min(600.0, remaining * 0.65)
    if time.time() < sub_deadline:
        a_multi, sc_multi = _run_multi_subgrad(best_seq,
                                               best_score,
                                               sub_deadline,
                                               start_time)
        if sc_multi < best_score:
            best_seq = a_multi.copy()
            best_score = sc_multi
            print(f"[{time.time() - start_time:.2f}s] after multi‑restart subgrad best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny hill‑climb (random mass transfers)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_final, sc_final = _final_local_search(best_seq,
                                                deadline,
                                                start_time,
                                                best_score)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Log the official evaluation (using the supplied security‑checked function)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
