"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The routine combines:

1️⃣  A richer parametric sweep that also tests symmetric power‑law families
    (a_i ∝ (i+Δ)^p + (n‑1‑i+Δ)^p) and geometric‑power hybrids.
2️⃣  An Adam‑based smoothed‑max gradient descent (soft‑max of the
    self‑convolution) with a continuation on the temperature β.
3️⃣  A final projected sub‑gradient descent (same routine as the reference
    solution) for polishing.
4️⃣  A simulated‑annealing hill‑climb in the last seconds.

All phases respect the 1000 s budget.  The best sequence found is returned
as a plain Python list of floats (normalised to sum = 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generators – richer families than the original sweep
# ----------------------------------------------------------------------


def _make_pow_shape(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i+shift)^p."""
    i = np.arange(n, dtype=np.float64)
    shape = (i + shift) ** p
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape


def _make_sym_pow_shape(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i+shift)^p + (n-1-i+shift)^p   (symmetric power law)."""
    i = np.arange(n, dtype=np.float64)
    shape = (i + shift) ** p + (n - 1 - i + shift) ** p
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape


def _make_geom_pow_shape(n: int, r: float, p: float, shift: float) -> np.ndarray:
    """a_i ∝ r^i·(i+shift)^p."""
    i = np.arange(n, dtype=np.float64)
    shape = (r ** i) * ((i + shift) ** p)
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape


def _make_sym_geom_pow_shape(n: int, r: float, p: float,
                             shift: float) -> np.ndarray:
    """
    Symmetric geometric‑power shape:
        a_i ∝ r^i·(i+shift)^p + r^{n-1-i}·(n-1-i+shift)^p
    """
    i = np.arange(n, dtype=np.float64)
    shape = (r ** i) * ((i + shift) ** p) + (r ** (n - 1 - i)) * (
        (n - 1 - i + shift) ** p)
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Mix λ∈[0,1] of uniform and the supplied shape; result already sums to 1."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Soft‑max utilities (smooth approximation of max)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max probabilities p_k = exp(β·conv_k) / Σ exp(·) (stable)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max objective:
        g_i = 2·Σ_j a_j·p_{i+j}
    Computed via a cross‑correlation using FFT.
    """
    n = a.shape[0]
    # reverse p for correlation
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    corr = np.fft.irfft(Af * Pf, n=L_grad)
    # only the middle part corresponds to the needed indices
    start = n - 1
    g = corr[start:start + n][::-1]
    return 2.0 * g


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same as the reference implementation)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        step = step0 / (it ** 0.6)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(
                f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cur_score:.6f}"
            )
        else:
            # occasional uphill moves
            prob = np.exp(-(cur_score - best_score) * 12.0 /
                          (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)

    if total > 0.0:

    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
