def _adam_descent(a0, deadline, start_time, rng, step0=0.02, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0
    best_a=a.copy(); best_score=_objective(a)
    while time.time() < deadline:
        conv=_conv_self(a)
        max_val=conv.max()
        score=2*a.shape[0]*max_val
        if score<best_score:
            best_score=score; best_a=a.copy()
        ks=np.where(conv>=max_val-1e-14)[0]
        # compute gradient g
        g = np.zeros_like(a)
        for k in ks:
            i_start=max(0,k-a.shape[0]+1)
            i_end=min(a.shape[0]-1,k)
            idx=np.arange(i_start,i_end+1)
            g[idx] += 2*a[k-idx]
        # ADAM update
        t+=1
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # compute step size; step0 may be scaled adaptively
        step = step0 / (np.sqrt(t) + 1) # we can incorporate sqrt(t) if needed
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_simplex(a)
        # Occasionally random mass move
        if t % 2000 == 0:
            src = rng.choice(a.shape[0], p=a)
            dst = rng.integers(a.shape[0])
            if src != dst:
                delta = a[src] * 0.02 * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = _project_simplex(a)
    return best_a, best_score

"""
Hybrid multi‑restart optimizer for the max‑auto‑convolution problem.

The algorithm proceeds in four phases:

1. **Initial pool** – generate a diverse set of probability vectors
   (uniform, power ramps, Beta families, Gaussian bumps, edge‑pair bumps
   and a few random Dirichlet draws).  Keep the K best candidates.

2. **Multi‑restart descent** – repeatedly pick a candidate from the pool,
   perturb it slightly and run a short exponentiated‑gradient (mirror‑descent)
   sub‑gradient optimisation.  Any improvement updates the global best.

3. **Long Adam‑style refinement** – on the current best sequence run a
   projected Adam optimiser (first‑ and second‑moment adaptation) for the
   remaining bulk of the budget.

4. **Final simulated‑annealing polish** – a brief temperature‑controlled
   mass‑swap local search to squeeze out the last bit of improvement.

All updates stay on the probability simplex via Euclidean projection.
The implementation is fully vectorised, uses FFT for fast convolutions and
fits comfortably into the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Full linear convolution a*a using FFT (length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
#   value = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators (used for the initial pool)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Power‑ramp with optional flat plateau, symmetrised."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Keep the K best (lowest) candidates (sorted)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Sub‑gradient descent (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     rng: np.random.Generator,
                     step0: float = 300.0,
                     max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SG iter={it:,} best={cur_score:.6f}")

        # sub‑gradient (all shifts attaining the maximum)
        ks = np.where(conv >= max_val - 1e-14)[0]
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # exponentiated‑gradient update
        step = step0 / math.sqrt(it)
        expo = np.exp(-step * g)
        a = a * expo
        a = _project_simplex(a)

        # occasional random mass move to inject stochasticity
        if it % 4000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected optimisation (first‑ and second‑moment adaptation)
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  rng: np.random.Generator,
                  step0: float = 0.8,
                  max_move_frac: float = 0.02,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} best={cur_score:.6f}")

        # sub‑gradient at current max locations
        ks = np.where(conv >= max_val - 1e-14)[0]
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size slowly decays with sqrt(it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # occasional random move (helps escape plateaus)
        if it % 5000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start = 0.2
    T_end = 1e-9
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        a_prop = _project_simplex(a_prop)

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score

                continue
            a /= s
            _push_top(pool, _objective(a), a, TOP_K)

    # Incorporate the previous best if available
    if globals().get('height_sequence_1') is not None:
        old = np.asarray(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != N:
                x_old = np.linspace(0, 1, old.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                up = np.interp(x_new, x_old, old)
            else:
                up = old.copy()
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                up = up / up.sum()
                _push_top(pool, _objective(up), up, TOP_K)

    # --------------------------------------------------------------
    # Pick the current best candidate
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Multi‑restart short sub‑gradient descents
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate roughly 70 % of the remaining budget for restarts
    restart_budget = remaining * 0.7
    per_restart = max(5.0, restart_budget / 12.0)   # at most ~12 short runs
    restart_start = time.time()

    while time.time() < deadline - 5.0 and (time.time() - restart_start) < restart_budget:
        # pick a random elite candidate, add a tiny jitter and optimise
        idx = rng.integers(len(pool))
        a0 = pool[idx][1].copy()
        a0 = a0 + 0.001 * rng.random(N)   # small random perturbation
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

        sub_deadline = time.time() + per_restart
        cand_seq, cand_score = _subgrad_descent(a0, sub_deadline, start_time, rng,
                                                step0=300.0, max_move_frac=0.02)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart improved best = {best_score:.6f}")

        # keep the new point in the pool if it is promising
        _push_top(pool, cand_score, cand_seq, TOP_K)

    # --------------------------------------------------------------
    # 3) Long Adam‑style refinement on the best candidate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        refine_deadline = time.time() + remaining * 0.6
        refined_seq, refined_score = _adam_descent(best_seq, refine_deadline,
                                                   start_time, rng,
                                                   step0=0.8, max_move_frac=0.02)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam refinement best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final simulated‑annealing polish (if any time left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, rng,
                                                    max_move_frac=0.03)

    # --------------------------------------------------------------
    # 5) Normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
