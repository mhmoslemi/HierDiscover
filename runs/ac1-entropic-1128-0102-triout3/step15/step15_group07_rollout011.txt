def propose_candidate(seed=0, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed + 12345)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    # Choose N based on time left
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # function to compute objective for sum=1 case.
    def objective(a):
        conv = _conv_self(a)
        return float(2.0 * N * conv.max())
    # initial pool
    TOP_K = 5
    pool = []

    # base shapes
    # uniform
    uniform = np.full(N, 1.0/N)
    _push_top(pool, objective(uniform), uniform, TOP_K)
    # random Dirichlet draws
    for conc in [0.005, 0.01, 0.02, 0.05, 0.1]:
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            a /= a.sum()
            _push_top(pool, objective(a), a, TOP_K)

    # add previously known best if exists
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1:
            if prev.size != N:
                x_old = np.linspace(0, 1, prev.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                prev = np.interp(x_new, x_old, prev)
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0:
                prev /= prev.sum()
                _push_top(pool, objective(prev), prev, TOP_K)
    # sort
    pool.sort(key=lambda x: x[0])
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time()-start_time:.2f}s] Initial best={best_score:.6f}")

    # gradient descent with smooth max
    # schedule parameters
    # calculate total approximate iterations based on remaining time:
    est_iter = int( max(1, (deadline - time.time()) // 0.01) )  # assume 0.01 sec per iter
    # but we cap to maybe 5000
    max_iters = min(5000, est_iter)
    # epsilon schedule
    eps_start = 5e-5
    eps_end = 5e-7
    # geometric factor
    if max_iters > 1:
        eps_factor = (eps_end/eps_start) ** (1.0/(max_iters-1))
    else:
        eps_factor = 1.0

    eps = eps_start

    # Use variable step0 = 200.0 as previous.
    step0 = 200.0

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # evaluate current best on the best candidate
        a = best_seq.copy()
        # compute conv and objective
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * N * max_conv

        # track improvements
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] GD iter={it} new best={best_score:.6f}")

        # compute smooth weights
        # shift conv by subtracting max to avoid overflow
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # compute gradient g = 2 * conv(w_rev, a)[:N]
        w_rev = w[::-1]
        L = _next_pow2(len(w_rev) + N - 1)
        Wa = np.fft.rfft(w_rev, n=L)
        Aa = np.fft.rfft(a, n=L)
        conv_wa = np.fft.irfft(Wa * Aa, n=L)[:N]
        g = 2.0 * conv_wa

        # Step size
        step = step0 / math.sqrt(it)
        # gradient step with projection onto simplex
        a_new = a - step * g
        a_new = _project_simplex(a_new)

        # assign for next iteration's start (use best_seq as seed)
        best_seq = a_new

        # decaying epsilon
        eps *= eps_factor

    # after gradient descent, possibly refine via SA
    # short SA with mass transfers
    # use remaining time
    remaining = deadline - time.time()
    if remaining > 2.0:
        a = best_seq.copy()
        cur_score = 2.0 * N * _conv_self(a).max()
        best_a = a.copy()
        best_s = cur_score
        T_start = 0.2
        T_end = 1e-6
        it = 0
        while time.time() < deadline:
            it += 1
            elapsed = time.time() - start_time
            frac = min(1.0, elapsed / budget_s)
            T = T_start * ((T_end / T_start) ** frac)

            # propose move: pick two positions i,j and transfer mass
            i = rng.integers(N)
            j = rng.integers(N)
            if i == j:
                continue
            # amount to move
            delta = a[i] * 0.02 * rng.random()
            if delta <= 0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            # project onto simplex via simple renorm (since only small changes)
            # ensure non-negative
            a_prop = np.maximum(a_prop, 0.0)
            s = a_prop.sum()
            if s == 0:
                continue
            a_prop /= s

            new_score = 2.0 * N * _conv_self(a_prop).max()
            if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
                a = a_prop
                cur_score = new_score
                if new_score < best_s:
                    best_s = new_score
                    best_a = a.copy()
                    print(f"[{time.time()-start_time:.2f}s] SA iter={it} best={best_s:.6f}")

        best_seq = best_a
        best_score = best_s

    # final normalisation (should already sum to 1)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0/N)

    final_score = objective(best_seq)
    print(f"[{time.time()-start_time:.2f}s] Finished best≈{final_score:.6f}")
    return [float(v) for v in best_seq]

"""
Propose a low‑score step‑function.

Algorithm
---------
1.  Choose a problem size N (larger N gives more degrees of freedom;
    N≈65536 for the full budget, smaller N if time runs low).
2.  Build a tiny pool of diverse initial probability vectors:
    uniform, a few Dirichlet draws (very spiky) and, if available,
    the previous best `height_sequence_1`.
3.  Take the best vector and run a smooth‑max gradient descent.
    * The true objective for a probability vector a (∑a=1) is
          F(a)=2·N·max_k (a∗a)[k].
      We replace the hard max by a soft‑max:
          w_k = exp(( (a∗a)[k] − max ) / ε)  (normalized to sum 1).
    * The gradient of the smoothed objective is
          g_i = 2·∑_k w_k·a_{k−i},
      which is a cross‑correlation and is evaluated efficiently with FFT.
    * Update a by a Euclidean projected (sub)gradient step with a
      decreasing stepsize η_t=η₀/√t and a geometrically decaying
      smoothing parameter ε_t.
4.  A short simulated‑annealing polish (mass transfers) is applied
    while time permits.
5.  Return the best vector seen, normalised to sum 1 (list of floats).

The combination of a soft‑max surrogate, FFT‑based gradients and a
final SA phase improves upon the previous sub‑gradient / SA approach
and typically reaches a bound below 1.503 within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a via FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Keep only the k best (lowest‑score) items in *pool*."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Search for a step‑function with a low evaluation score."""
    rng = np.random.default_rng(seed + 9999)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Choose a discretisation size (larger N → more flexibility)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Initialise a tiny pool of diverse candidates
    # --------------------------------------------------------------
    TOP_K = 6
    pool = []   # list of (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, 2.0, uniform, TOP_K)   # uniform always gives score 2

    # a few Dirichlet draws with very small concentration (spiky)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05):
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            # true objective for a probability vector is 2·N·max(conv)
            cur = 2.0 * N * _conv_self(a).max()
            _push_top(pool, cur, a, TOP_K)

    # bring in the previous best if it exists
    try:
        prev = globals().get('height_sequence_1')
        if prev is not None:
            prev = np.asarray(prev, dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                if prev.size != N:
                    # linear interpolation to required length
                    old_x = np.linspace(0, 1, prev.size, endpoint=False)
                    new_x = np.linspace(0, 1, N, endpoint=False)
                    prev = np.interp(new_x, old_x, prev)
                prev = np.maximum(prev, 0.0)
                if prev.sum() > 0:
                    prev /= prev.sum()
                    cur = 2.0 * N * _conv_self(prev).max()
                    _push_top(pool, cur, prev, TOP_K)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Pick the current best candidate
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Smooth‑max gradient descent
    # --------------------------------------------------------------
    # decide how many iterations we can afford (roughly 0.01 s per iter)
    est_iters = int(max(1, (deadline - time.time()) // 0.012))
    max_iters = min(5000, est_iters)

    cur_a = best_vec.copy()
    # initialise smoothing parameter from the first objective value
    conv0 = _conv_self(cur_a)
    max0 = conv0.max()
    eps_start = max(1e-8, max0 * 5.0)      # a few multiples of the current max
    eps_end   = max(1e-10, max0 * 0.001)   # shrink towards zero
    if max_iters > 1:
        eps_factor = (eps_end / eps_start) ** (1.0 / (max_iters - 1))
    else:
        eps_factor = 1.0
    eps = eps_start

    step0 = 300.0                         # initial step size

    # keep the overall best seen during the descent
    overall_best_a = cur_a.copy()
    overall_best_score = 2.0 * N * max0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- evaluate current point -------------------------------------------------
        conv = _conv_self(cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * N * max_conv
        if cur_score < overall_best_score:
            overall_best_score = cur_score
            overall_best_a = cur_a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD iter={it:,}  new best={cur_score:.6f}")

        # ---- soft‑max weights -------------------------------------------------------
        # w_k = exp((conv_k - max)/eps)  and normalise
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed objective (cross‑correlation) ---------------
        # g_i = 2 * Σ_k w_k * a_{k-i}
        w_rev = w[::-1]                      # reverse for cross‑correlation
        Lg = _next_pow2(3 * N - 2)           # length for convolution a (N) + w_rev (2N‑1) - 1
        Af = np.fft.rfft(cur_a, n=Lg)
        Wf = np.fft.rfft(w_rev, n=Lg)
        conv_aw = np.fft.irfft(Af * Wf, n=Lg)[: 3 * N - 2]   # full cross‑correlation
        # extract the first N entries (indices 2N‑2 … N‑1, reversed)
        g = 2.0 * conv_aw[2 * N - 2 : N - 2 : -1]

        # ---- gradient step with Euclidean projection onto the simplex -------------
        step = step0 / math.sqrt(it)
        a_tmp = cur_a - step * g
        cur_a = _project_simplex(a_tmp)

        # ---- decay the smoothing parameter -----------------------------------------
        eps *= eps_factor

    # --------------------------------------------------------------
    # 3) Optional short simulated‑annealing polish
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a = overall_best_a.copy()
        cur_score = overall_best_score
        best_a = a.copy()
        best_s = cur_score

        T_start = 0.2
        T_end = 1e-6
        it = 0
        while time.time() < deadline:
            it += 1
            elapsed = time.time() - start_time
            frac = min(1.0, elapsed / budget_s)
            T = T_start * ((T_end / T_start) ** frac)

            # propose a small mass transfer between two random positions
            src = rng.integers(N)
            dst = rng.integers(N)
            if src == dst:
                continue
            delta = a[src] * 0.02 * rng.random()
            if delta <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta
            a_prop = np.maximum(a_prop, 0.0)
            s = a_prop.sum()
            if s == 0:
                continue
            a_prop /= s

            new_score = 2.0 * N * _conv_self(a_prop).max()
            if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
                a = a_prop
                cur_score = new_score
                if new_score < best_s:
                    best_s = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SA iter={it:,}  best={new_score:.6f}")

        overall_best_a = best_a
        overall_best_score = best_s

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    overall_best_a = np.maximum(overall_best_a, 0.0)
    total = overall_best_a.sum()
    if total > 0:
        overall_best_a /= total
    else:
        overall_best_a = np.full(N, 1.0 / N)

    final_score = 2.0 * N * _conv_self(overall_best_a).max()
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")

    return [float(v) for v in overall_best_a]
