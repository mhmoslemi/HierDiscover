    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} best={cur_score:.6f}")

        ks = np.where(conv >= max_val - 1e-14)[0]
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]
        # scale gradient of objective (objective = 2n * max)

        # Adam update

        v = beta2 * v + (1.0 - beta2) * (g * g)

        lr = lr0 / math.sqrt(it)

        # project onto simplex

        # occasional random move for escape

            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = _project_simplex(a)

    return best_a, best_score

"""
Hybrid multi‑start exponentiated‑gradient / simulated‑annealing optimizer.

* Choose a problem size N based on the remaining budget (up to 131 072).
* Build a small, diverse pool of candidate probability vectors:
  - uniform, power ramps, symmetric beta‑densities, Gaussian bumps,
    random Dirichlet draws, and the previous best if available.
* Keep the TOP_K vectors (lowest current objective).
* Run a short exponentiated‑gradient (sub‑gradient) refinement on each
  of the top candidates – this helps escape the uniform basin and
  discovers several promising basins.
* Pick the best of those refinements and spend the bulk of the remaining
  time on a long exponentiated‑gradient run on that vector (large step0,
  decaying as 1/√it, occasional random mass moves for escape).
* Finish with a short simulated‑annealing polish.
* The algorithm works directly on the probability simplex (sum = 1, a_i≥0)
  and prints progress information.  The returned list sums to 1, so the
  harness’s `evaluate_sequence` returns the same objective value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Objective for a probability vector (sum == 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes for the initial pool
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Power‑ramp with optional symmetric flat plateau."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Keep the k best (lowest‑score) items in *pool*."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Sub‑gradient descent with exponentiated‑gradient updates
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     rng: np.random.Generator,
                     step0: float = 250.0,
                     max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on  f(a)=2·n·max(conv(a,a)).
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SG iter={it:,} best={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        ks = np.where(conv >= max_val - 1e-14)[0]
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # exponentiated‑gradient update
        step = step0 / math.sqrt(it)
        expo = np.exp(-step * g)
        a = a * expo
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum) or np.isinf(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # occasional random mass move to escape flat regions
        if it % 4000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 0:
                    a /= a_sum
                else:
                    a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score

        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] After long SG best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final simulated‑annealing polish (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        best_seq, best_score = _simulated_annealing(
            best_seq, deadline, start_time, rng, max_move_frac=0.03)

    # --------------------------------------------------------------
    # 5) Normalise and convert to a plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
