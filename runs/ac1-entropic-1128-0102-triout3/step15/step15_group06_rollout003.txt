"""
Hybrid optimiser for the step‑function bound

    2·n·max(conv(a,a)) / (sum(a)²)

The routine works with a modest problem size (N≈4‑8 k) so that many
iterations of cheap updates fit into the 1000 s budget:

1. Build a diverse pool of analytic and random shapes (uniform,
   symmetric/asymmetric ramps, exponentials, power‑laws, Beta,
   Dirichlet draws).  Keep the TOP_K best candidates.

2. Run an **exponentiated sub‑gradient descent** with a simple back‑tracking
   line‑search on each top candidate.  The update is
   a_i ← a_i·exp(−η·g_i) and the vector is renormalised onto the
   probability simplex after every step.  The sub‑gradient g is obtained
   from the active set of convolution indices attaining the current
   maximum.  A small amount of symmetrisation is applied periodically.

3. Perform a few **random‑perturbation** local searches around the best
   solution found so far.  Each perturbation is followed by a short
   exponentiated descent.

4. Finish with a **smooth‑max Adam** polishing phase.  The hard max is
   approximated by a soft‑max with an exponentially decreasing temperature.

All steps are fully vectorised (FFT based convolutions).  The routine
returns a plain Python list of non‑negative heights, normalised to sum = 1.
"""

import time
import math

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Make the vector symmetric: (v + v[::-1]) / 2."""
    return (v + v[::-1]) * 0.5

def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power ramp; plateau_frac∈[0,0.5]."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_asym_ramp(n: int, p_left: float, p_right: float, plateau_frac: float) -> np.ndarray:
    """Asymmetric power ramp, independent left/right steepness."""

            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        # temperature schedule (exponential decay)
        T = T_end + (T0 - T_end) * (0.5 ** (it / 3000.0))

        # soft‑max weights (stable)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of soft‑max surrogate (correlation with reversed a)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a)

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation bound.
    The routine builds a pool of analytic/random shapes, runs exponentiated
    mirror descent with line‑search on the top candidates, performs random
    perturbations around the best solution and finishes with a smooth‑max
    Adam polishing.  The final vector is returned as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.05  # small safety margin

    # --------------------------------------------------------------
    # Choose a modest problem size – many cheap iterations are valuable
    # --------------------------------------------------------------
    remaining_initial = deadline - start
    if remaining_initial > 750:
        N = 8192
    elif remaining_initial > 400:
        N = 4096
    else:
        N = 2048

    # --------------------------------------------------------------
    # 1) Build a diverse initial pool (TOP_K kept)
    # --------------------------------------------------------------
    TOP_K = 12
    pool = []          # elements are (score, vector)

    def _add(vec: np.ndarray) -> None:
        _push_top(pool, _objective(vec), vec, TOP_K)

    # deterministic analytic families
    _add(_make_uniform(N))

    # symmetric power ramps
    for p in np.linspace(0.2, 3.0, 7):
        for plat in np.linspace(0.0, 0.35, 3):
            _add(_make_power_ramp(N, p, plat))

    # asymmetric ramps
    for pl in np.linspace(0.2, 3.0, 4):
        for pr in np.linspace(0.2, 3.0, 4):
            _add(_make_asym_ramp(N, pl, pr, plateau_frac=0.1))

    # exponential decays
    for lam in np.linspace(0.4, 3.5, 9):
        _add(_make_exponential(N, lam))

    # power‑law decays
    for p in np.linspace(0.5, 3.5, 9):
        _add(_make_power_law(N, p))

    # asymmetric Beta families
    for a in np.linspace(0.3, 2.5, 5):
        for b in np.linspace(0.3, 2.5, 5):
            _add(_make_beta_shape_asym(N, a, b))

    # U‑shapes (high at both ends)
    for ef in np.linspace(0.05, 0.25, 4):
        _add(_make_u_shape(N, edge_frac=ef))

    # Random Dirichlet draws (both sorted and unsorted)
    rng = np.random.default_rng(seed + 12345)
    conc_params = [0.02, 0.05, 0.1, 0.3, 0.7, 1.5, 3.0]
    for conc in conc_params:
        for _ in range(30):
            if time.time() >= deadline:
                break
            vec = rng.gamma(conc, 1.0, size=N)
            # make a monotone decreasing version as well
            vec_desc = np.sort(vec)[::-1]
            vec_desc /= vec_desc.sum()
            _add(vec_desc)

    # Include the previous best construction if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:
                starter_up /= s
                _add(starter_up)
    except Exception:
        pass

    # Ensure pool is not empty
    if not pool:
        _add(_make_uniform(N))

    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Exponentiated sub‑gradient descent on the current top pool
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + 0.70 * remaining
        idx = 0
        while time.time() < sub_deadline:
            vec = pool[idx % len(pool)][1]
            seq, sc = _exp_descent_line_search(vec, sub_deadline, start)
            if sc < best_score:
                best_score, best_vec = sc, seq.copy()
                print(f"[{time.time() - start:.2f}s] ExpGD improvement → {sc:.6f}")
            idx += 1

    # --------------------------------------------------------------
    # 3) Random perturbation local search around the current best
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 12.0:
        pert_deadline = time.time() + remaining - 5.0
        while time.time() < pert_deadline:
            # decreasing Gaussian noise magnitude
            sigma = 0.02 * (1.0 - (time.time() - start) / budget_s)
            cand = best_vec + np.random.normal(scale=sigma, size=N)
            cand = np.clip(cand, 0.0, None)
            cand /= cand.sum() if cand.sum() > 0 else 1.0
            cand = _symmetrize(cand)
            # short descent (time‑boxed)
            short_end = time.time() + 2.0
            seq, sc = _exp_descent_line_search(cand, short_end, start)
            if sc < best_score:
                best_score, best_vec = sc, seq.copy()
                print(f"[{time.time() - start:.2f}s] Perturb improvement → {sc:.6f}")

    # --------------------------------------------------------------
    # 4) Final smooth‑max Adam polishing
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        seq, sc = _smooth_gradient_descent(best_vec, deadline, start)
        if sc < best_score:
            best_score, best_vec = sc, seq.copy()
            print(f"[{time.time() - start:.2f}s] SmoothGD final improvement → {sc:.6f}")

    # --------------------------------------------------------------
    # 5) Normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    # Return a plain Python list as required by the evaluator.
    return [float(v) for v in best_vec]
