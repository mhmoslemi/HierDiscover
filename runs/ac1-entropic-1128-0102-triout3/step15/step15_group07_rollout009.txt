"""
Improved search for a low‑score step‑function.

The algorithm works in three phases under a 1000 s budget.

1. **Initial pool** – many analytic shapes (uniform, power ramps,
   symmetric Beta, Gaussian) and a few Dirichlet draws are generated,
   evaluated with the exact objective (2·n·max(conv)) and the TOP_K
   candidates are kept.

2. **Soft‑max gradient descent** – the hard max in the objective is
   replaced by a temperature‑controlled soft‑max.  Each iteration
   computes the full self‑convolution (FFT) and the soft‑max weights,
   then the gradient of the smoothed objective (a correlation of the
   weights with the current vector).  An additive step with a
   diminishing learning‑rate is taken followed by Euclidean projection
   onto the probability simplex.  Temperature is exponentially cooled
   from a fraction of the current maximal convolution down to a tiny
   value, thereby smoothly turning the problem from “flatten the whole
   autocorrelation” into the original max‑convolution problem.
   Several short restarts from random Dirichlet draws are performed,
   after which a long run is executed on the best found candidate.

3. **Simulated‑annealing polish** – a lightweight SA with random
   mass‑moves finishes the search when time is low.

All vectors are kept non‑negative and normalised to sum to 1, which
does not affect the evaluation function (the latter is homogeneous of
degree 0).  The final sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: smallest power of two ≥ x (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Full self‑convolution a*a (linear convolution, length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
#   value = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes for the initial pool
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Power‑ramp with optional flat plateau, symmetrised."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Keep the K best (lowest) candidates (sorted)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Soft‑max gradient descent (temperature annealing)
# ----------------------------------------------------------------------
def _soft_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  rng: np.random.Generator,
                  step0: float = 10.0,
                  max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed objective:
        F_T(a) = 2·n·T·log Σ_k exp(conv_k / T)
    with an exponential cooling schedule for T.
    Uses additive updates with projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- self‑convolution (FFT) -----
        Lc = _next_pow2(2 * n - 1)
        Af = np.fft.rfft(a, n=Lc)
        conv = np.fft.irfft(Af * Af, n=Lc)[: 2 * n - 1]

        # ----- temperature (exponential cooling) -----
        max_conv = conv.max()
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total) if total > 0 else 1.0
        T_start = max_conv * 0.5          # start a moderate fraction of the peak
        T_end = max_conv * 1e-5           # end at a very low temperature
        T = T_start * ((T_end / T_start) ** frac) if T_start > 0 else 1e-12

        # ----- soft‑max weights -----
        # shift for numerical stability
        shifted = conv - max_conv               # ≤ 0
        exps = np.exp(shifted / max(T, 1e-12))
        w = exps / exps.sum()                  # length 2n‑1, sums to 1

        # ----- gradient of the smoothed max -----
        # gradient_i = 2 * Σ_k w_k * a_{k-i}  (correlation)
        a_rev = a[::-1]
        Lg = _next_pow2(len(w) + len(a_rev) - 1)   # ≈ 3·n
        W_fft = np.fft.rfft(w, n=Lg)
        A_rev_fft = np.fft.rfft(a_rev, n=Lg)
        corr_fft = W_fft * A_rev_fft
        corr_full = np.fft.irfft(corr_fft, n=Lg)
        grad = 2.0 * corr_full[n - 1: n - 1 + n]    # length n, non‑negative

        # ----- step size (decreasing) -----
        step = step0 / math.sqrt(it)

        # normalise the direction to keep the absolute update roughly step
        if grad.max() > 0:
            direction = grad / grad.max()
        else:
            direction = grad

        a = a - step * direction
        # ----- projection onto simplex -----
        a = _project_simplex(a)

        # ----- occasional random mass move (helps escape) -----
        if it % 2000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()

        # ----- track best -----
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] soft‑desc iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total) if total > 0 else 1.0
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats whose sum is 1.
    """
    rng = np.random.default_rng(seed + 9999)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose a discretisation size (larger sizes give finer resolution)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build an initial pool of diverse candidates
    # --------------------------------------------------------------
    TOP_K = 8
    pool: list[tuple[float, np.ndarray]] = []   # (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # power‑ramps
    p_grid = np.linspace(0.1, 3.0, 13)
    plate_grid = np.linspace(0.0, 0.5, 9)
    for p in p_grid:
        for plat in plate_grid:
            a = _make_shape(N, p, plat)
            _push_top(pool, _objective(a), a, TOP_K)

    # symmetric Beta shapes (including very U‑shaped)
    for alpha in np.concatenate([np.arange(0.05, 0.26, 0.05),
                                 np.arange(0.3, 1.01, 0.1),
                                 np.arange(1.2, 3.1, 0.4)]):
        a = _make_beta_shape(N, float(alpha))
        _push_top(pool, _objective(a), a, TOP_K)

    # Gaussian‑like bumps (different widths)
    for sigma in [N * 0.02, N * 0.04, N * 0.08, N * 0.15]:
        a = _make_gaussian_shape(N, sigma)
        _push_top(pool, _objective(a), a, TOP_K)

    # random Dirichlet draws with varied concentration
    concs = [0.02, 0.05, 0.1, 0.2, 0.5]
    for conc in concs:
        for _ in range(12):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            if a.sum() == 0:
                continue
            a /= a.sum()
            _push_top(pool, _objective(a), a, TOP_K)

    # include previously found best if available
    try:
        prev = globals().get('height_sequence_1')
        if prev is not None:
            prev = np.asarray(prev, dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                if prev.size != N:
                    x_old = np.linspace(0, 1, prev.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    prev_up = np.interp(x_new, x_old, prev)
                else:
                    prev_up = prev.copy()
                prev_up = np.maximum(prev_up, 0.0)
                if prev_up.sum() > 0:
                    prev_up /= prev_up.sum()
                    _push_top(pool, _objective(prev_up), prev_up, TOP_K)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Keep the best candidate from the pool
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Short warm‑up runs (several random restarts) with soft‑desc
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate up to 35 % of the remaining budget to quick restarts
        restart_budget = remaining * 0.35
        n_restarts = 8
        per_restart = restart_budget / n_restarts
        for i in range(n_restarts):
            if time.time() >= deadline:
                break
            a0 = rng.gamma(0.02, 1.0, size=N)
            a0 /= a0.sum()
            sub_dead = time.time() + per_restart
            cand_seq, cand_score = _soft_descent(a0, sub_dead, start_time, rng,
                                                step0=12.0, max_move_frac=0.025)
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Restart {i+1}/{n_restarts} gave new best {cand_score:.6f}")

    # --------------------------------------------------------------
    # 3) Long soft‑max descent on the current best candidate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        long_deadline = time.time() + remaining * 0.75
        refined_seq, refined_score = _soft_descent(best_seq, long_deadline,
                                                  start_time, rng,
                                                  step0=15.0, max_move_frac=0.02)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] After long soft‑desc best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final simulated‑annealing polish (if any time left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, rng,
                                                    max_move_frac=0.03)

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
