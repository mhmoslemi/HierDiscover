"""
Hybrid stochastic search for a low‑peak self‑convolution step‑function.

The evaluation function is
    score = 2·n·max(conv(a,a)) / (sum(a)**2) ,

where `conv` is the full linear convolution.  The score is invariant under
global scaling, therefore we keep the vector normalised to sum = 1.

Algorithm (time‑aware, ≤ budget_s seconds)

1. **Initialisation** – start from the globally supplied `height_sequence_1`
   (if present) otherwise from a uniform distribution of length 4096.

2. **Broad Dirichlet sweep** – for a short initial period generate random
   Dirichlet samples with a wide range of concentration parameters and keep
   the best one.

3. **Gradient‑guided simulated annealing** – repeatedly
      * compute the sub‑gradient of the max‑convolution,
      * move a small amount of mass from a high‑gradient index to a low‑gradient
        index (preserving the total sum),
      * accept the move if the score improves, otherwise accept with a
        temperature‑dependent probability that cools linearly to zero.
   The step size adapts to the acceptance ratio.

4. **Optional LP refinement** – every few 10 000 iterations (if SciPy is
   available) solve a small linear program that maximises the total mass of a
   candidate `b` under the current peak constraint.  Mixing a fraction of the
   normalised `b` into the current solution often yields a strict improvement.

5. **Final uniform mixing** – sweep a convex combination with the uniform
   distribution to smooth residual peaks.

The routine prints progress and returns the best sequence found as a plain
list of non‑negative floats.
"""

import time
import numpy as np

# optional linear‑programming fallback
try:
    from scipy.optimize import linprog  # type: ignore
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full (linear) self‑convolution of `a` using FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv(a,a))``.
    The contribution of each active shift is averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat values within 1e‑12 of the maximum as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]               # a_{k-i}
    if active.size:
        grad = 2.0 * grad / active.size        # factor 2 because each term appears twice
    return grad


def score_from_conv(conv: np.ndarray, n: int) -> float:
    """Score used by the optimiser (sum of a is 1 ⇒ 2·n·max(conv))."""
    return float(2.0 * n * conv.max())


def solve_lp_max_sum(a: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve   max 1ᵀb   s.t.   b ≥ 0 ,   (a * b)[k] ≤ M   for k in `active_idx`.

    Returns the optimal `b` (may have any total mass) or ``None`` on failure.
    Only a few rows (`len(active_idx)`) are used, so the LP is cheap.
    """
    if not _SCIPY_AVAILABLE:
        return None
    n = a.shape[0]
    m = active_idx.size
    A = np.zeros((m, n), dtype=float)
    for row, k in enumerate(active_idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1, dtype=int)
        A[row, i_range] = a[k - i_range]
    c = -np.ones(n)                     # maximise sum(b)  ⇔ minimise -sum(b)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0️⃣  Load a starter (global variable) or fall back to uniform.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                raise ValueError
            a = starter / starter.sum()
        else:
            raise ValueError
    except Exception:
        n = 4096
        a = np.full(n, 1.0 / n)

    n = a.shape[0]
    L_fft = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Helper to compute score quickly.
    # ------------------------------------------------------------------
    cur_conv = conv_self(a, L_fft)
    cur_score = score_from_conv(cur_conv, n)

    best_seq = a.copy()
    best_score = cur_score

    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # small safety margin

    print(f"[0.00s] Initial score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Broad random Dirichlet sweep (≈ first 5 % of time)
    # ------------------------------------------------------------------
    dirichlet_deadline = start_time + 0.05 * budget_s
    while time.time() < dirichlet_deadline:
        # draw concentration log‑uniformly from [0.02, 20]
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        if sample.sum() == 0.0:
            continue
        candidate = sample / sample.sum()
        conv = conv_self(candidate, L_fft)
        sc = score_from_conv(conv, n)
        if sc < best_score:
            best_score, best_seq, a, cur_conv, cur_score = sc, candidate.copy(), candidate.copy(), conv, sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Gradient‑guided simulated annealing.
    # ------------------------------------------------------------------
    it = 0
    accept_cnt = 0
    step_factor = 0.15           # start moving about 15 % of the source mass
    min_step = 1e-8
    T0 = 0.02                    # initial temperature for SA

    # parameters for occasional LP refinement
    lp_interval = 20000          # every N iterations
    lp_t = 0.3                   # mixing weight for LP‑derived candidate

    while time.time() < deadline:
        it += 1

        # ----- compute sub‑gradient of the max‑convolution -----
        grad = subgradient_maxconv(a, cur_conv)

        # ----- choose source (high gradient, non‑zero mass) & target (low gradient) -----
        mask_positive = a > 0
        if not mask_positive.any():
            break
        # source index – probability proportional to grad where there is mass
        prob_src = grad * mask_positive
        if prob_src.sum() == 0:
            src = rng.integers(0, n)
        else:
            prob_src = prob_src / prob_src.sum()
            src = rng.choice(n, p=prob_src)

        # target index – prefer small gradient values
        inv = (grad.max() - grad) + 1e-12
        inv[src] = 0.0
        if inv.sum() == 0:
            tgt = rng.integers(0, n)
            while tgt == src:
                tgt = rng.integers(0, n)
        else:
            prob_tgt = inv / inv.sum()
            tgt = rng.choice(n, p=prob_tgt)

        # amount of mass to transfer
        delta = step_factor * a[src]
        delta = max(min_step, min(delta, a[src]))
        if delta <= 0:
            continue

        # ----- propose a new vector (mass transfer) -----
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # renormalise to avoid tiny drift
        total = cand.sum()
        if total == 0.0:
            continue
        cand /= total

        # ----- evaluate -----
        cand_conv = conv_self(cand, L_fft)
        cand_score = score_from_conv(cand_conv, n)

        # ----- SA acceptance -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # linear cooling of the temperature
            remaining = max(0.0, deadline - time.time())
            total_time = max(0.1, deadline - start_time)
            T = T0 * (remaining / total_time)
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a, cur_conv, cur_score = cand, cand_conv, cand_score
            accept_cnt += 1
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] it={it:,}  new best = {best_score:.6f}")

        # ----- adapt step size every 10 000 iterations -----
        if it % 10000 == 0:
            acc_ratio = accept_cnt / 10000
            if acc_ratio > 0.3:
                step_factor = min(step_factor * 1.2, 0.4)
            elif acc_ratio < 0.05:
                step_factor = max(step_factor * 0.7, 1e-4)
            accept_cnt = 0

        # ----- occasional LP refinement (if SciPy available) -----
        if _SCIPY_AVAILABLE and (it % lp_interval == 0):
            # active indices of the current peak
            M = cur_conv.max()
            active = np.where(cur_conv >= M - 1e-12)[0]
            b = solve_lp_max_sum(a, M, active)
            if b is not None and b.sum() > 0:
                # normalise candidate to sum=1
                g = b / b.sum()
                # convex combination with the current solution
                cand_lp = (1 - lp_t) * a + lp_t * g
                cand_lp /= cand_lp.sum()
                cand_lp_conv = conv_self(cand_lp, L_fft)
                cand_lp_score = score_from_conv(cand_lp_conv, n)
                if cand_lp_score < cur_score:
                    a, cur_conv, cur_score = cand_lp, cand_lp_conv, cand_lp_score
                    if cand_lp_score < best_score:
                        best_score, best_seq = cand_lp_score, cand_lp.copy()
                        print(f"[{time.time() - start_time:.2f}s] LP mix → new best = {best_score:.6f}")

        # ----- early exit if target reached -----
        if best_score <= 1.503:
            print(f"[{time.time() - start_time:.2f}s] target reached (score={best_score:.6f})")
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final uniform mixing sweep (coarse + refined).
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    # coarse sweep
    for lam in np.linspace(0.0, 0.2, 41):
        seq = lam * best_seq + (1.0 - lam) * uniform
        seq /= seq.sum()
        conv = conv_self(seq, L_fft)
        sc = score_from_conv(conv, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] mix λ={lam:.3f} → score={sc:.6f}")

    # refined sweep around the best λ (if any improvement was seen)
    # Find the λ that gave the best score in the coarse sweep
    # (here we simply reuse the same grid for brevity)
    for lam in np.linspace(0.0, 0.2, 21):
        seq = lam * best_seq + (1.0 - lam) * uniform
        seq /= seq.sum()
        conv = conv_self(seq, L_fft)
        sc = score_from_conv(conv, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] fine‑mix λ={lam:.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
