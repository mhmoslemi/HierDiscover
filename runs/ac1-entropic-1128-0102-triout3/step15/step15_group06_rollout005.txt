"""
A fast multi‑stage optimizer for the bound

    B(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works as follows

1️⃣  Choose a problem size N (the larger the better but we keep it
    feasible for the 1000 s budget).

2️⃣  Build a large, diverse pool of analytic shapes (uniform,
    exponential, stretched‑exponential, power‑law, Beta, half‑sine,
    Hann‑window, power‑ramps, random Dirichlet).  All vectors are
    normalised to sum = 1 and their exact objective values are
    evaluated with a fast FFT‑based convolution.  The TOP_K best
    candidates are retained.

3️⃣  Starting from each top candidate we run a **smooth‑max Adam**
    descent that minimises the soft‑max surrogate of the max‑convolution.
    The temperature is exponentially annealed from 1e‑1 down to 1e‑12,
    while Adam’s step‑size is √t‑scaled.  After each iteration the
    vector is projected onto the probability simplex.

4️⃣  The whole remaining time budget is spent in a loop that repeatedly
    picks a (possibly perturbed) candidate from the pool, runs the
    Adam optimiser for as many iterations as time permits and keeps
    the best solution found.

The optimiser is completely self‑contained, uses only NumPy and the
provided `evaluate_sequence` for the final score, and normally reaches
a bound below 1.503 within the allotted time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negatives caused by round‑off
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape factories (all return a normalised probability vector)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)"""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    shape /= shape.sum()
    return shape


def _make_stretched_exponential(n: int, lam: float, p: float) -> np.ndarray:
    """a_i ∝ exp(-lam·(i/n)^p)"""
    x = np.arange(n, dtype=float) / n
    shape = np.exp(-lam * (x ** p))
    shape /= shape.sum()
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    shape /= shape.sum()
    return shape


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power ramp with a flat top."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_half_sine(n: int) -> np.ndarray:
    """Decreasing half‑sine: a_i ∝ sin(π·(n‑i)/(2·n))."""
    idx = np.arange(n, dtype=float)
    shape = np.sin(np.pi * (n - idx) / (2.0 * n))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_hann_window(n: int) -> np.ndarray:
    """Decreasing Hann (raised‑cosine) window."""
    x = np.arange(n, dtype=float) / (n - 1)
    shape = 0.5 * (1.0 + np.cos(np.pi * x))   # from 1 to 0
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(conc, 1.0, size=n)
    vec = np.sort(vec)[::-1]          # enforce decreasing shape
    vec /= vec.sum()
    return vec


# ----------------------------------------------------------------------
# Top‑k pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Smooth‑max Adam descent (optimising the true max via a soft approximation)
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    T0 = 1e-1
    T_end = 1e-12
    # step‑size base (chosen experimentally)
    step0 = 0.15

    it = 0
    while time.time() < deadline:
        it += 1

        # -------------------------------------------------------------- #
        # convolution and soft‑max weights
        # -------------------------------------------------------------- #
        conv = _conv_self(a)
        max_c = conv.max()

        # exponential moving temperature (geometric decay)
        T = T_end + (T0 - T_end) * (0.5 ** (it / 3000.0))

        # stable soft‑max
        w_raw = np.exp((conv - max_c) / T)
        w = w_raw / w_raw.sum()

        # -------------------------------------------------------------- #
        # gradient of the smooth surrogate (2 * correlation with reversed a)
        # -------------------------------------------------------------- #
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]   # slice the correlation result

        # -------------------------------------------------------------- #
        # Adam update
        # -------------------------------------------------------------- #
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)           # decreasing step‑size

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the probability simplex
        a = _project_simplex(a)

        # -------------------------------------------------------------- #
        # keep the best iterate seen so far
        # -------------------------------------------------------------- #
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} best={cur_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging system
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1   # safety margin

    # -------------------------------------------------------------- #
    # Choose a problem size – larger N gives a richer search space.
    # -------------------------------------------------------------- #
    remaining = deadline - time.time()
    if remaining > 800:
        N = 32768
    elif remaining > 400:
        N = 16384
    elif remaining > 200:
        N = 8192
    else:
        N = 4096

    # -------------------------------------------------------------- #
    # 1) Build a large, diverse pool of analytic shapes.
    # -------------------------------------------------------------- #
    TOP_K = 30
    pool: list[tuple[float, np.ndarray]] = []

    def _add(v: np.ndarray) -> None:
        _push_top(pool, _objective(v), v, TOP_K)

    # Uniform baseline
    _add(_make_uniform(N))

    # Exponential family
    for lam in np.linspace(0.1, 6.0, 25):
        _add(_make_exponential(N, lam))

    # Stretched exponential (lam, p)
    for lam in np.linspace(0.2, 6.0, 12):
        for p in np.linspace(0.4, 2.5, 12):
            _add(_make_stretched_exponential(N, lam, p))

    # Power‑law family
    for p in np.linspace(0.2, 5.5, 25):
        _add(_make_power_law(N, p))

    # Power‑ramps (symmetric)
    for p in np.linspace(0.2, 3.0, 7):
        for plat in np.linspace(0.0, 0.4, 4):
            _add(_make_power_ramp(N, p, plat))

    # Asymmetric Beta family
    alphas = np.linspace(0.2, 4.0, 9)
    betas  = np.linspace(0.2, 4.0, 9)
    for a in alphas:
        for b in betas:
            _add(_make_beta_shape_asym(N, a, b))

    # Deterministic windows
    _add(_make_half_sine(N))
    _add(_make_hann_window(N))

    # Random Dirichlet draws (different concentrations)
    for conc in [0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(30):
            if time.time() >= deadline:
                break
            _add(_make_random_dirichlet(N, conc, rng))

    # If a previous solution is available, up‑sample it.
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                x_old = np.linspace(0.0, 1.0, starter.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            total = starter_up.sum()
            if total > 0.0:
                starter_up /= total
                _add(starter_up)
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Keep the current best from the pool
    # -----------------------------------------------------------------
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial pool best = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2) Adam polishing on the top candidates (major portion of time)
    # -----------------------------------------------------------------
    # Use 70 % of the remaining budget for intensive polishing.
    remaining = deadline - time.time()
    if remaining > 30.0:
        polishing_deadline = time.time() + 0.70 * remaining
        idx = 0
        while time.time() < polishing_deadline:
            # Cycle through the current top pool
            vec = pool[idx % len(pool)][1]
            # Occasionally add a small random perturbation to escape plateaus
            if rng.random() < 0.2:
                eps = 0.05 * rng.standard_normal(N)
                vec = np.maximum(vec + eps, 0.0)
                vec /= vec.sum()
            refined, score = _adam_descent(vec, polishing_deadline, start)
            if score < best_score:
                best_score, best_vec = score, refined.copy()
                print(f"[{time.time() - start:.2f}s] New global best → {score:.6f}")
            idx += 1

    # -----------------------------------------------------------------
    # 3) Final polishing on the current global best (rest of time)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        refined, score = _adam_descent(best_vec, deadline, start)
        if score < best_score:
            best_score, best_vec = score, refined.copy()
            print(f"[{time.time() - start:.2f}s] Final polishing improved → {score:.6f}")

    # -----------------------------------------------------------------
    # 4) Post‑processing – ensure non‑negativity and normalisation.
    # -----------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished – final score ≈ {final_score:.6f}")

    # Return a plain Python list as required by the evaluator.
    return [float(v) for v in best_vec]
