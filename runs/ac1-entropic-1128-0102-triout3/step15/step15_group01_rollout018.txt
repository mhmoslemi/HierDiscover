"""Optimization of non‑negative step‑function sequence via multi‑start
Adam‑style projected gradient descent on the simplex.

The evaluator used by the challenge is:
    score = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works as follows:
1. Initialise with a uniform vector (sum = 1).  If a global starter
   ``height_sequence_1`` is present, it is used.
2. Perform a short random‑Dirichlet sampling phase to obtain a decent
   starting point.  Each sample is evaluated using a FFT‑based convolution.
3. Repeatedly launch short Adam‑style descent runs from either the current
   best candidate or a fresh random start.  The update is additive, the
   iterate is projected onto the probability simplex after each step.
   Sub‑gradients of the max‑convolution are obtained from the active (tight)
   shifts.
4. After the time budget is almost exhausted a final coarse sweep mixing the
   best candidate with the uniform distribution is performed.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    If several shifts attain the maximum, gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat values within 1e-12 of the max as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]  # contribution a_{k-i}
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used during optimisation: 2·n·max(conv(a,a))."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i >= 0, sum x_i = 1}.
    Implements the algorithm from Duchi et al. (2008).
    """
    # shift by max for numerical stability
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # compute rho
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣  Random Dirichlet sampling
# ----------------------------------------------------------------------
def _random_dirichlet_phase(n: int,
                            rng: np.random.Generator,
                            best_seq: np.ndarray,
                            best_score: float,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    Sample Dirichlet vectors with a broad range of concentration parameters.
    Keep the best according to the internal objective.
    """
    while time.time() < deadline:
        # concentration log‑uniform in [log(0.02), log(20)]
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        s = sample.sum()
        if s <= 0:
            continue
        a = sample / s
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Adam‑style projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  n: int,
                  L_fft: int,
                  max_iters: int,
                  start_time: float,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    projected Adam descent for the max‑convolution objective.
    The iterate is kept on the probability simplex.
    """
    a = a0.copy()
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # base learning rate – a small constant works well because the
    # gradient magnitude is O(1) for our scale.
    alpha = 0.02

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_maxconv(a, conv)

        # Adam moments update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # additive update
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # project back onto simplex
        a_new = _project_simplex(a_new)

        a = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final mixing with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Scan λ∈[0,1] for seq = λ·a + (1‑λ)·uniform.
    Keep the mixture that yields the lowest objective.
    """
    uniform = np.full(n, 1.0 / n)
    lam_grid = np.linspace(0.0, 1.0, 41)   # coarse sweep
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        seq = lam * a + (1.0 - lam) * uniform
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq = seq / seq_sum
        sc = _objective_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")
    return best_seq, best_score

        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Adam‑style projected gradient descent
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  n: int,
                  L_fft: int,
                  max_iters: int,
                  start_time: float,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam descent on the probability simplex for the max‑convolution
    objective.
    """
    a = a0.copy()
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.02               # base learning rate

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = _project_simplex(a_new)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final mixing with uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Coarse sweep over λ∈[0,1] for seq = λ·a + (1‑λ)·uniform.
    The best mixture is retained.
    """
    uniform = np.full(n, 1.0 / n)
    for lam in np.linspace(0.0, 1.0, 41):
        if time.time() >= deadline:
            break
        seq = lam * a + (1.0 - lam) * uniform
        tot = seq.sum()
        if tot == 0:
            continue
        seq = seq / tot
        sc = _objective_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget, prints
    progress information and returns the best sequence found.
    """
    global start_time
    start_time = time.time()
    # small safety margin before the runner kills the process
    deadline = start_time + budget_s - 0.15

    # problem size – 4096 gives a good trade‑off between resolution and speed
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0️⃣  Initialise (uniform or optional global starter)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad random Dirichlet search (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_phase(
            n, rng, best_seq, best_score, phase1_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Multi‑start Adam descent (the bulk of the budget)
    # ------------------------------------------------------------------
    max_iters_per_restart = 5000
    while time.time() < deadline:
        # 35 % chance to start from a fresh random Dirichlet, otherwise continue from best
        if rng.random() < 0.35:
            log_c = rng.uniform(np.log(0.02), np.log(20.0))
            c = np.exp(log_c)
            sample = rng.gamma(shape=c, scale=1.0, size=n)
            s = sample.sum()
            if s == 0:
                continue
            a0 = sample / s
        else:
            a0 = best_seq.copy()

        best_seq, best_score = _adam_descent(
            a0,
            n,
            L_fft,
            max_iters=max_iters_per_restart,
            start_time=start_time,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # 3️⃣  Fine‑grained mixing with the uniform distribution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _mix_with_uniform(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Normalise one final time and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
