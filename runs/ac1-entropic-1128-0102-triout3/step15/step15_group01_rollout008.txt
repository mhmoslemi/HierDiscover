    n = len(f)
    m = len(active_shifts)
    A = np.zeros((m, n), dtype=float)

"""
Step‑function optimizer – LP‑guided alternating improvement.

The evaluator:
    score = 2·n·max(conv(a,a)) / (sum(a)²)   (a is any non‑negative vector)

Goal
----
The algorithm is a two‑phase procedure:

1️⃣  **Coarse optimisation** – a fast random‑Dirichlet / exponentiated‑gradient
    phase (a few seconds) that yields a decent non‑uniform distribution.
    If the global variable ``height_sequence_1`` is present we start from it
    directly.

2️⃣  **LP‑guided refinement** – an alternating scheme closely following the
    “linear‑programming + mixing” idea from the paper.  For the current
    sequence ``f`` we compute ``M = max(conv(f,f))`` and solve the linear
    program

        maximise   Σ g_i
        s.t.       (f * g)[k] ≤ M   for every *active* shift k
                   g_i ≥ 0

    (only the shifts where the convolution attains the maximum are kept,
    which makes the LP tiny).  The optimal ``g`` is then rescaled to the
    target sum ``√(2·n)`` and mixed back into ``f``.  Whenever the new
    candidate improves the bound it is kept; otherwise a convex combination
    with a modest mixing weight is performed.  The loop stops when the time
    budget expires or the target score 1.503 is reached.

The final candidate is optionally fine‑tuned with a short exponentiated‑gradient
sweep and a last mixture with the uniform distribution.

All numbers are non‑negative; the routine respects the 1000 s time budget
and returns a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (only needed for the LP step)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score = 2·n·max(conv(a,a)) / (sum(a)²).  ``a`` may be any non‑negative vector."""
    conv = _conv_self(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:      # mimics the evaluator’s safety check
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _build_active_constraints(f: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (|active| × n) such that
        (f * g)[k] = A[k, :] · g
    for every active shift k.
    """
    n = f.size
    m = active_shifts.size
    A = np.zeros((m, n), dtype=float)
    for row, k in enumerate(active_shifts):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        A[row, idx] = f[k - idx]
    return A


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """A cheap Dirichlet draw (concentration drawn log‑uniformly)."""
    log_c = rng.uniform(np.log(0.02), np.log(20.0))
    c = np.exp(log_c)
    sample = rng.gamma(shape=c, scale=1.0, size=n)
    s = sample.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return sample / s


def _expgrad_descent(
    a0: np.ndarray,
    n: int,
    L_fft: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Simple exponentiated‑gradient (mirror descent) on the simplex.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _objective_score(a, n)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score, best = cur_score, a.copy()
        # sub‑gradient of max(conv)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size

        # decaying step size
        eta = 0.7 / np.sqrt(it)
        factor = np.exp(-eta * grad)
        factor = np.clip(factor, 1e-12, 1e12)
        a_new = a * factor
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / a_new_sum

        # accept if improvement, else keep old a
        conv_new = _conv_self(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)
        if new_score < cur_score:
            a = a_new
        else:
            # occasional simulated‑annealing acceptance
            T0 = 1e-3
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = T0 * (remaining / total)
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_new
    return best, best_score


def _mix_with_uniform(
    a: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Scan λ∈[0,1] for seq = λ·a + (1‑λ)·uniform and keep the best.
    """
    uniform = np.full(n, 1.0 / n)
    lam_grid = np.linspace(0.0, 1.0, 41)
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        seq = lam * a + (1.0 - lam) * uniform
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq = seq / seq_sum
        sc = _objective_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.25

    # problem size – 4096 is a good trade‑off between quality and speed
    n = 4096
    sqrt2n = np.sqrt(2.0 * n)
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – try to use the globally supplied starter
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s * sqrt2n
                f = starter
                print(f"[0.00s] Using global starter, initial score={_objective_score(f, n):.6f}")
            else:
                raise ValueError
        else:
            raise ValueError
    except Exception:
        # fallback: a single random Dirichlet draw
        rng = np.random.default_rng()
        f = _random_dirichlet(n, rng) * sqrt2n
        print(f"[0.00s] No starter – random init, score={_objective_score(f, n):.6f}")

    best_seq = f.copy()
    best_score = _objective_score(f, n)
    print(f"[0.00s] Baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  (optional) quick Exponentiated‑gradient warm‑up (few seconds)
    # ------------------------------------------------------------------
    warmup_deadline = start_time + 30.0  # 30 s of quick exp‑grad
    if time.time() < warmup_deadline:
        print(f"[{time.time() - start_time:.2f}s] Warm‑up exp‑grad")
        best_seq, best_score = _expgrad_descent(
            best_seq, n, L_fft, warmup_deadline, start_time
        )
        f = best_seq.copy()
        print(f"[{time.time() - start_time:.2f}s] After warm‑up best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  LP‑guided alternating improvement
    # ------------------------------------------------------------------
    lp_deadline = deadline - 100.0  # keep ~100 s for final polishing
    it = 0
    while time.time() < lp_deadline:
        it += 1
        # current convolution data
        conv_f = _conv_self(f, L_fft)
        M = conv_f.max()
        cur_score = 2.0 * n * M / (f.sum() ** 2)

        # keep global best
        if cur_score < best_score:
            best_score, best_seq = cur_score, f.copy()
            print(f"[{time.time() - start_time:.2f}s] LP‑iter {it:,}  new best score={cur_score:.6f}")

        # stop early if we have already beaten the target
        if best_score < 1.503:
            break

        # --------------------------------------------------------------
        # build the active‑set linear programme
        # --------------------------------------------------------------
        # relative tolerance – we treat anything within 1e‑12·M as active
        tol = max(1e-12, M * 1e-10)
        active_shifts = np.where(conv_f >= M - tol)[0]
        if active_shifts.size == 0:
            break

        A = _build_active_constraints(f, active_shifts)   # shape (m, n)
        b_ub = np.full(active_shifts.size, M, dtype=float)
        c = -np.ones(n, dtype=float)   # maximise Σ g_i  → minimise −Σ g_i
        bounds = [(0, None)] * n

        if not _SCIPY_AVAILABLE:
            # Without SciPy we simply skip the LP phase.
            break

        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "tol": 1e-9},
            )
        except Exception as e:
            print(f"[{time.time() - start_time:.2f}s] LP failed ({e}) – abort LP phase")
            break

        if not res.success:
            # If the LP is infeasible (should not happen) we stop the LP loop.
            print(f"[{time.time() - start_time:.2f}s] LP not successful: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            # nothing to gain – abort the LP loop
            break

        # scale g to the target sum √(2·n)
        g = g0 * (sqrt2n / sum_g0)

        # --------------------------------------------------------------
        # decide whether to replace f by g or mix
        # --------------------------------------------------------------
        g_score = _objective_score(g, n)

        # a small mixing weight that shrinks as we approach the deadline
        remaining = max(0.0, deadline - time.time())
        ratio = remaining / (deadline - start_time)
        t = 0.3 * ratio + 0.05 * (1 - ratio)   # start with ~0.3, finish with ~0.05

        if g_score < cur_score:
            # g is already better – accept it completely
            f = g
            cur_score = g_score
        else:
            # otherwise perform a convex combination
            f = (1.0 - t) * f + t * g
            cur_score = _objective_score(f, n)

        # keep the best ever seen
        if cur_score < best_score:
            best_score, best_seq = cur_score, f.copy()
            print(f"[{time.time() - start_time:.2f}s] LP‑iter {it:,}  improved to {cur_score:.6f}")

    print(f"[{time.time() - start_time:.2f}s] LP phase finished – best score ≈ {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final short exponentiated‑gradient sweep (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final exp‑grad sweep")
        best_seq, best_score = _expgrad_descent(
            best_seq, n, L_fft, deadline, start_time
        )
        print(f"[{time.time() - start_time:.2f}s] After final exp‑grad best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Mixing with the uniform distribution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Mixing with uniform")
        best_seq, best_score = _mix_with_uniform(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Normalise one final time (any scaling works – we keep the current sum)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total * sqrt2n   # keep the convention used in the LP phase
    else:
        best_seq = np.full(n, sqrt2n / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final best score ≈ {best_score:.6f}")
    # convert to plain Python list of floats
    return [float(v) for v in best_seq]
