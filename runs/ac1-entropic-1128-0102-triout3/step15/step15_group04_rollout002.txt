"""
Optimisation of a non‑negative step function (vector) for the objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages, all respecting a 1000 s budget.

1️⃣  **Broad parametric sweep** – a dense grid over a geometric×power family,
    mixed with the uniform distribution, to obtain a set of promising seeds.

2️⃣  **Smoothed‑max Adam optimisation** – minimise a smooth soft‑max
    approximation of the max‑convolution using the Adam optimiser.
    The temperature β is increased in a continuation scheme (0.5 → 256).

3️⃣  **Projected sub‑gradient descent** – polish the best candidate with the
    exact L∞‑sub‑gradient (the same routine used in the reference solution).

4️⃣  **Simulated‑annealing hill‑climb** – random mass‑transfer moves with a
    decreasing temperature to escape any remaining local minima.

The routine works with a vector length n=8192 (a good trade‑off between
expressivity and speed) and returns a plain Python list of floats whose
sum is 1.  All numbers are non‑negative and safely within the limits of
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0 , Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation utilities
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """
    Geometric × power shape: a_i ∝ r**i * (i+1)**p, normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    # work in log‑space to avoid overflow / underflow
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)      # prevent overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Convex mixture of a shape with the uniform distribution.
    lam∈[0,1] gives lam·uniform + (1‑lam)·shape.
    The result is already normalised.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Smooth‑max utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights p_k = exp(beta·conv_k)/∑exp(.) (numerically stable)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max objective:
        g_i = 2 * Σ_j a_j * p_{i+j}
    computed via FFT as a cross‑correlation.
    """
    n = a.shape[0]
    # reverse p for correlation
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    conv_full = np.fft.irfft(Af * Pf, n=L_grad)
    # length of conv_full is ≥ 3n‑2 ; we need the middle n entries reversed
    start = n - 1
    g = conv_full[start:start + n][::-1]
    return 2.0 * g


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same as reference)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.20          # a slightly larger initial step
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # Build the sub‑gradient for the active shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        # add symmetric part where i ≠ j
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        # diminishing step size
        step = step0 / (it ** 0.58)          # a bit slower decay
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional uphill moves (tiny probability)
            prob = np.exp(-(cur_score - best_score) * 10.0 /
                          (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # small jitter every few thousand iterations to avoid stagnation
        if it % 2000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

    # 3️⃣  Exact sub‑gradient descent (polishing).
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:      # reserve ~80 s for final steps
        sub_deadline = deadline - 80.0
        a_sub, sc_sub = _subgradient_descent(best_seq, sub_deadline,
                                             start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient final score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing hill‑climb (fine‑grained polishing).
    # ------------------------------------------------------------------
    # Temperature linearly decays from T0 to near‑zero.
    T0 = 0.12
    while time.time() < deadline:
        # choose a source index proportionally to its mass (more effect)
        src = rng.choice(n, p=best_seq)
        if best_seq[src] < 1e-18:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # transfer up to 0.8 % of the source mass (occasionally larger)
        max_factor = 0.008
        if rng.random() < 0.02:          # rare larger jumps
            max_factor = 0.04
        eps = best_seq[src] * rng.random() * max_factor

        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = np.maximum(a_prop, 0.0)
        a_prop /= a_prop.sum()

        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improvement score={new_sc:.6f}")
        else:
            # accept with Boltzmann probability
            remaining = deadline - time.time()
            total = deadline - start_time
            T = T0 * (remaining / total)   # linear cooling
            if T > 0 and rng.random() < np.exp(-(new_sc - best_score) / T):
                best_seq = a_prop
                best_score = new_sc

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
