def _make_edge_power(n, p):
    i = np.arange(n, dtype=float) + 0.5
    # distances to edges
    left = i
    right = n + 0.5 - i
    shape = (left * right) ** (-p)
    shape[shape == np.inf] = 0.0
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0/n)
    return shape
```

"""Propose a low‑score step‑function for the bound
2·n·max(conv(a,a))/sum(a)².

Algorithm
---------
1. **Random analytic search (small size).**  
   Build a diverse set of candidate shapes (Beta, exponential,
   power‑law, edge‑peaked power, sine‑power) on a modest length
   `N_small≈8 k`.  Randomly sample parameters (log‑uniform where
   appropriate) and keep the best `TOP_K` candidates (simple
   top‑k pool).

2. **Upsample to a larger size.**  
   Interpolate the best candidates to `N_large` (≈16–32 k) and
   renormalise to sum = 1.

3. **Local refinement.**  
   – *Exponentiated sub‑gradient descent* (mirror‑descent on the
     probability simplex) – fast many‑iteration phase.  
   – *Adam on a smooth‑max surrogate* – fine‑grained polishing with
     an exponentially‑annealed temperature.

4. **Random convex combinations.**  
   Occasionally mix a few of the top‑k vectors with random simplex
   weights, evaluate the objective and keep any improvement.

All steps respect the 1000 s budget and use only NumPy (no I/O)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    if n == 0:
        return np.inf
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# Analytic shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_edge_power(n: int, p: float) -> np.ndarray:
    """U‑shaped (edge‑peaked) power law: a_i ∝ ((i+1)*(n-i))^{-p}."""
    i = np.arange(n, dtype=float) + 0.5
    left = i
    right = n + 0.5 - i
    shape = (left * right) ** (-p)
    shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sine_power(n: int, gamma: float) -> np.ndarray:
    """sin‑power shape; negative gamma produces edge‑peaked curves."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    shape = np.sin(np.pi * x) ** gamma
    shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror‑descent on the L∞‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 1.0                   # a slightly larger base step
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # sub‑gradient of max(conv) w.r.t. a
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + it / 4000.0)          # slower decay than before
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        s = a.sum()
        if s > 0.0:

    """Adam optimisation of a soft‑max approximation of the max‑convolution."""

    best_score = _objective(a)

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)

        v = beta2 * v + (1 - beta2) * (grad ** 2)
