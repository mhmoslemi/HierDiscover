"""
Propose a better step‑function sequence.

The algorithm builds on the previous stochastic optimiser but adds a
few extra ingredients that help to push the bound below the previous
record (~1.5079 → target 1.503).

Main ideas
----------
1️⃣  Wide random sampling of a 4‑parameter *geometric‑power‑tail* family,
    extended ranges for the exponents.
2️⃣  Several restarts of a stochastic hill‑climb on the same four parameters.
3️⃣  A *smooth‑max* gradient descent (log‑sum‑exp approximation of the
    ‑∞‑norm) with an Adam‑style step‑size schedule.
4️⃣  Classic active‑set sub‑gradient descent (as in the reference code) for
    fine‑tuning.
5️⃣  Random “mix‑with‑shape’’ moves: blend the current best vector with a new
    randomly generated shape (small mixing weight) and keep any improvement.
6️⃣  A cheap random mass‑transfer local search (the same final hill‑climb as
    before).

All stages respect the global 1000 s budget, and the best sequence found
is projected onto the probability simplex (∑a_i = 1) before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two vectors via FFT."""
    L = _next_pow2(len(a) + len(b) - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: len(a) + len(b) - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Shape generation utilities

def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Shape proportional to r**i * (i+1)**p * (n-i)**q   (i = 0…n‑1).
    Returned vector is already normalised (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)               # stabilise
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return lam·uniform + (1‑lam)·shape (both already sum to 1)
    and renormalise to guard against rounding.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

            cur_seq = np.maximum(cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq, 0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent (Adam‑style)
# ----------------------------------------------------------------------


def _smooth_max_descent(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of max(conv(a,a)).
    Uses an Adam‑like adaptive step size and steadily increases the
    smoothing parameter β.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta_smooth = 5.0          # start smoothing
    beta_target = 180.0
    beta_inc = 0.6

    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        max_c = conv.max()

        # weights for the log‑sum‑exp smoothing
        w = np.exp(beta_smooth * (conv - max_c))
        w_sum = w.sum()
        w /= w_sum                     # normalised

        # gradient = 2 * B a   where B_{ij} = w_{i+j}
        a_rev = a[::-1]
        B_a_full = _conv_fft(w, a_rev)               # length 3n‑2
        # extract the n entries needed (indices n‑1 … 2n‑2)
        grad = 2.0 * B_a_full[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step_size = 0.15 / (np.sqrt(v_hat) + eps)
        a_candidate = a - step_size * m_hat
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            # reset Adam buffers on improvement
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{time.time() - start_time:.2f}s] smooth‑descent it={it:,} "
                  f"β={beta_smooth:.1f} score={cur_score:.6f}")
        else:
            # occasional acceptance of uphill move
            if rng.random() < 0.005:
                a = a_candidate

        # increase smoothing parameter
        beta_smooth = min(beta_target, beta_smooth + beta_inc)

        # jitter every few thousand iterations
        if it % 3000 == 0:
            eps_vec = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps_vec, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (active set) – same as before, but with a
# slightly more aggressive schedule.
# ----------------------------------------------------------------------


def _compute_subgrad_multi(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) with respect to a, aggregating over all
    entries within `tol` of the current maximum.
    """
    max_val = conv.max()
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
    return g


def _subgradient_descent_multi(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               best_score_initial: float,
                               active_tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the true objective,
    using the full active set of near‑maximal convolution entries.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.14
    beta = 0.9
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=active_tol)

        v = beta * v + (1.0 - beta) * g
        step = step0 / (it ** 0.55)
        a_candidate = a - step * v
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            v[:] = 0.0
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            prob = np.exp(-(cur_score - best_score) * 10.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            v[:] = 0.0

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two
    coordinates, keeping the vector on the simplex.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3   # up to 0.2 % transfer
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            if rng.random() < 0.001:
                a = a_prop
    return best_a, best_sc


# ----------------------------------------------------------------------
# Mixing random shapes with the current best (helps escape plateaus)
# ----------------------------------------------------------------------


def _mix_random_shapes(best_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       best_score: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly generate a fresh parametric shape and blend it with the
    current best vector (mixing weight t∈[0.05,0.30]).
    Keep any improvement.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)
    cur_seq = best_seq.copy()
    cur_score = best_score

    while time.time() < deadline:
        # random shape (same ranges as the initial sweep)
        r = np.exp(rng.uniform(np.log(0.5), np.log(1.2)))
        p = rng.uniform(-6.0, 6.0)
        q = rng.uniform(-6.0, 6.0)
        lam = rng.uniform(0.0, 0.8)

        shape = _make_shape_general(n, r, p, q)
        shape = _mix_with_uniform(shape, lam)

        t = rng.uniform(0.05, 0.30)
        cand = (1.0 - t) * cur_seq + t * shape   # already sums to 1
        sc = _eval_fast(cand, L_fft)

        if sc < cur_score - 1e-9:
            cur_seq = cand.copy()
            cur_score = sc
            print(f"[{time.time() - start_time:.2f}s] mix‑shape t={t:.3f} "
                  f"score={sc:.6f}")
    return cur_seq, cur_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the target
    evaluation function within the given time budget.
    The best sequence (normalised to sum 1) is returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0️⃣  Uniform baseline (score = 2.0)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Use a globally supplied starter if it exists
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_seq = starter.copy()
                    best_score = cur
                    print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Random parametric sweep (broader ranges)
    # --------------------------------------------------------------
    sweep_deadline = start_time + 180.0
    rp_seq, rp_score, rp_params = _random_param_search(n, sweep_deadline, start_time)
    if rp_seq is not None and rp_score < best_score:
        best_seq = rp_seq.copy()
        best_score = rp_score
        best_params = rp_params
        print(f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}")
    else:
        # fallback params (reasonable centre)
        best_params = (0.95, 0.0, 0.0, 0.2)

    # --------------------------------------------------------------
    # 2️⃣  Hill‑climb on the four shape parameters (multiple restarts)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    hill_deadline = time.time() + min(200.0, remaining * 0.25)
    best_seq, best_score = _hillclimb_params(best_params,
                                             best_seq,
                                             best_score,
                                             hill_deadline,
                                             start_time)

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max gradient descent (log‑sum‑exp)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    smooth_deadline = time.time() + min(150.0, remaining * 0.2)
    if time.time() < smooth_deadline:
        a_smooth, sc_smooth = _smooth_max_descent(best_seq,
                                                  smooth_deadline,
                                                  start_time,
                                                  best_score)
        if sc_smooth < best_score:
            best_seq = a_smooth.copy()
            best_score = sc_smooth
            print(f"[{time.time() - start_time:.2f}s] smooth‑max score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Sub‑gradient descent with active set (fine‑tuning)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    sub_deadline = time.time() + min(350.0, remaining * 0.3)
    if time.time() < sub_deadline:
        a_sub, sc_sub = _subgradient_descent_multi(best_seq,
                                                   sub_deadline,
                                                   start_time,
                                                   best_score)
        if sc_sub < best_score:
            best_seq = a_sub.copy()
            best_score = sc_sub
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient final score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Random mixing of fresh shapes (helps escape remaining plateaus)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    mix_deadline = time.time() + min(120.0, remaining * 0.15)
    if time.time() < mix_deadline:
        a_mix, sc_mix = _mix_random_shapes(best_seq, n, mix_deadline,
                                           start_time, best_score)
        if sc_mix < best_score:
            best_seq = a_mix.copy()
            best_score = sc_mix
            print(f"[{time.time() - start_time:.2f}s] post‑mix best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Final tiny mass‑transfer hill‑climb
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_final, sc_final = _final_local_search(best_seq,
                                                deadline,
                                                start_time,
                                                best_score)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Log the official evaluation (uses the provided security‑checked function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
