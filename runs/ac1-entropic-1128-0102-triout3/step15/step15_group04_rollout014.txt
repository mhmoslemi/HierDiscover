```
def _lp_optimize(f, M):
    n = f.shape[0]
    # Build sparse matrix A of size (2n-1, n)
    # Constraint: A * g <= M
    # Precompute row_idx, col_idx once globally for given n (caching)
    # Use data = np.repeat(f, n)
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n - 1, n))
    # Solve linear program: max sum(g) -> min -sum(g)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        # Rescale to sum=1
        s = g.sum()
        if s > 0:
            g = g / s
        else:
            g = f.copy()
        return g
    else:
        # fallback to original f
        return f
```

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a` is a non‑negative vector (the step‑function).  
For a normalised vector (`\sum a = 1`) the score is simply  

\[
\text{score}(a)=2\,n\;\max (a*a).
\]

The optimisation is convex: each entry of the self‑convolution `a*a` is a
positive‑semidefinite quadratic form, therefore their point‑wise maximum is
convex.  We exploit this structure with three ideas that improve on the
reference implementation.

1. **Fast smoothed‑max gradient descent** – a continuation on the temperature
   β ∈ {0.5,…,256} with Adam‑style adaptive steps, always projected onto the
   probability simplex.

2. **Exact linear‑programming refinement** – for the current vector `f`
   compute the maximal‑mass vector `g` that satisfies `f*g ≤ M` (where
   `M = max(f*f)`).  This is a linear programme:
   ```
   maximise   Σ g_j
   subject to  (f * g)_k ≤ M   for all k
               g_j ≥ 0
   ```
   Mixing `f` with `g` (`f ← (1‑t)f + t g`) provably reduces the maximal
   convolution while keeping the sum fixed.  The LP is solved with the
   high‑performance HiGHS solver from SciPy.  The constraint matrix is a
   Toeplitz matrix that we build once (sparse, O(n²) entries).

3. **Projected sub‑gradient polishing** – the classic sub‑gradient method from
   the reference code, now used only as a final “polish”.

All phases respect the 1000 s wall‑clock budget.  The routine returns a plain
Python list of floats that can be fed directly to `evaluate_sequence`.

```python
"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four phases:

1️⃣  Fast smoothed‑max gradient descent with Adam‑style adaptation and a
    continuation on the temperature β.

2️⃣  Exact linear‑programming refinement: for the current vector f compute the
    maximal‑mass vector g that satisfies f*g ≤ M (M = max(f*f)).  Mixing
    f←(1‑t)f+t g provably lowers the objective.  The LP is solved with the
    HiGHS solver (SciPy) using a pre‑computed sparse Toeplitz constraint
    matrix.

3️⃣  Classic projected sub‑gradient descent (same idea as the reference
    solution) as a final polish.

4️⃣  Tiny random hill‑climb (mass transfers) for the last few seconds.

All phases observe the 1000 s budget and the routine finally returns the best
sequence as a list of floats (normalised to sum = 1).
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·len(a)‑1) using FFT.
    Returns a real array of length 2·n‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1.
    Returns ∞ if the sum is too tiny.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed‑max utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights p_k = exp(beta*conv_k) / Σ exp(.) (numerically stable)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective:
        g_i = 2 * Σ_j a_j * p_{i+j}
    Implemented by a cross‑correlation computed via FFT.
    """
    n = a.shape[0]
    p_rev = p[::-1]                     # reverse for correlation
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    corr = np.fft.irfft(Af * Pf, n=L_grad)   # full correlation length ≈ 3n‑2
    # The relevant part is the middle n entries (indices n‑1 … 2n‑2)
    start = n - 1
    g = corr[start:start + n][::-1]     # reverse back to obtain gradient
    return 2.0 * g


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _build_lp_indices(n: int):
    """
    Pre‑compute the row/column indices of the Toeplitz constraint matrix
    that encodes (f * g)_k = Σ_i f_i g_{k-i}.
    Returns two 1‑D integer arrays of length n².
    """
    # rows[i,j] = i + j  (0‑based)
    rows = np.arange(n).reshape(-1, 1) + np.arange(n).reshape(1, -1)
    row_idx = rows.ravel(order="F")
    col_idx = np.tile(np.arange(n), n)
    return row_idx.astype(np.int64), col_idx.astype(np.int64)


def _lp_refine(f: np.ndarray, M: float, row_idx, col_idx, t: float) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t. (f * g)_k ≤ M,  g ≥ 0,
    then mix f and the normalised solution:
        new_f = (1‑t) f + t g ,  projected onto the simplex.
    Returns the mixed vector.
    """
    n = f.shape[0]

    # Build the constraint matrix A (sparse, shape (2n‑1, n))
    data = np.repeat(f, n)               # f_i appears n times, once per column
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    c = -np.ones(n)                      # maximise Σ g_j  ⇒ minimise –Σ g_j
    b = np.full(2 * n - 1, M, dtype=np.float64)

    bounds = [(0.0, None)] * n
    # HiGHS is the default method in recent SciPy versions and is very fast.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # LP failed – fall back to the original vector
        return f.copy()

    g = res.x
    s = g.sum()
    if s > 0.0:
        g /= s
    else:
        g = f.copy()

    new_f = (1.0 - t) * f + t * g
    new_f = _project_simplex(new_f)
    return new_f

# Exact sub‑gradient descent (reference implementation)
# ----------------------------------------------------------------------

    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    it = 0

        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals

    # 5️⃣ Tiny random hill‑climb for the last few seconds
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to 0.5 % of the source mass
        eps = best_seq[src] * rng.random() * 5e-3
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```
