    L_fft = _next_pow2(2*n - 1)
    a = a0.copy()
    best_a = a.copy()

    rng = np.random.default_rng()

    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    alpha = 0.3  # base learning rate
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=1e-12)

"""
Propose a step function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in several stages inside the 1000 s budget:

1️⃣  Baselines (uniform and a deterministic arcsine‑type shape).

2️⃣  A brief random exploration of the 4‑parameter family
    a_i ∝ r**i·(i+1)ᵖ·(n‑i)ᵠ, mixed with a uniform background.
    The best parameters are kept.

3️⃣  A short stochastic hill‑climb on the four parameters.

4️⃣  Many short restarts of an **Adam**‑style projected sub‑gradient
    descent on the true objective.  Each run starts either from the
    current best vector (with jitter) or from a freshly sampled
    parametric shape.  The active set of convolution entries that are
    within a tiny tolerance of the current maximum is used for the
    sub‑gradient, giving a stable direction.

5️⃣  A cheap final “mass‑transfer’’ hill‑climb that moves a tiny
    amount of probability mass between two coordinates.

All vectors are kept on the probability simplex (non‑negative,
summing to 1) by a Euclidean projection.  The code uses FFT‑based
convolution for speed and prints progress information.  The final
vector is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components
    proportional to r**i * (i+1)**p * (n-i)**q   (i = 0…n‑1).
    Computed in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)                     # avoid overflow
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return lam·uniform + (1‑lam)·shape  (both already sum to 1).
    The result is re‑normalised to guard against numerical drift.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _arcsine_shape(n: int) -> np.ndarray:
    """
    Deterministic “arcsine‑type’’ shape:
        a_i ∝ 1/√((i+0.5)·(n‑i‑0.5))
    Normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    shape = 1.0 / np.sqrt(i * (n - i + 0.5))
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient of the active set
# ----------------------------------------------------------------------


def _compute_subgrad_multi(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) with respect to a, aggregating over
    all shifts whose value is within `tol` of the maximum.
    """
    max_val = conv.max()
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)

    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        # a_j contributes to ∂/∂a_i
        g[i_vals] += a[j_vals]

        # symmetric term (i ≠ j) adds a_i to ∂/∂a_j
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

    return g


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent
# ----------------------------------------------------------------------


def _adam_subgradient_descent(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              best_score_initial: float,
                              active_tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the true objective.
    Resets momentum when an improvement is found.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    alpha = 0.35               # base learning rate

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=active_tol)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            # reset Adam moments after improvement
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            print(f"[{time.time() - start_time:.2f}s] Adam it={it} score={cur_score:.6f}")
        else:
            # occasional uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter every few thousand iterations to escape plateaus
        if it % 2500 == 0:
            eps_j = 1e-5 * rng.random(n) * a
            a = np.maximum(a + eps_j, 0.0)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            it = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Random parametric search (geometric‑power‑tail family)
# ----------------------------------------------------------------------


def _random_param_search(n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray,
                                                     float,
                                                     tuple[float, float, float, float]]:
    """
    Uniform‑in‑log sampling of (r,p,q,λ) and keep the best vector.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        # r in (0.8, 0.999999) – log‑uniform (broader than before)
        r = np.exp(rng.uniform(np.log(0.8), np.log(0.999999)))
        p = rng.uniform(-5.0, 5.0)
        q = rng.uniform(-5.0, 5.0)
        lam = rng.uniform(0.0, 1.0)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam)
            print(f"[{time.time() - start_time:.2f}s] random search r={r:.4f} "
                  f"p={p:.3f} q={q:.3f} λ={lam:.3f} score={sc:.6f}")

    return best_seq, best_score, best_params


# ----------------------------------------------------------------------
# Local hill‑climb on the four shape parameters
# ----------------------------------------------------------------------


def _hillclimb_params(initial_params: tuple[float, float, float, float],
                      start_seq: np.ndarray,
                      start_score: float,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb on (r,p,q,λ) with exponentially shrinking
    perturbations.
    """
    cur_r, cur_p, cur_q, cur_lam = initial_params
    cur_seq = start_seq.copy()
    cur_score = start_score

    sigma_r = 0.012
    sigma_p = 0.25
    sigma_q = 0.25
    sigma_lam = 0.10

    rng = np.random.default_rng()
    n = cur_seq.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_seq = cur_seq.copy()
    best_score = cur_score

    while time.time() < deadline:
        # propose a new point in parameter space
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        q_new = cur_q + sigma_q * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        r_new = float(np.clip(r_new, 0.8, 0.999999))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_shape_general(n, r_new, p_new, q_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = _eval_fast(cand_new, L_fft)

        if sc_new < cur_score:
            cur_r, cur_p, cur_q, cur_lam = r_new, p_new, q_new, lam_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(f"[{time.time() - start_time:.2f}s] param hill r={r_new:.6f} "
                      f"p={p_new:.4f} q={q_new:.4f} λ={lam_new:.4f} score={sc_new:.6f}")

        # decay the perturbation scales
        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_q *= 0.9995
        sigma_lam *= 0.9995

        # occasional jitter to shake the solution
        if rng.random() < 0.004:
            cur_seq = np.maximum(cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq, 0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final tiny hill‑climb (random mass transfers)
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two
    coordinates, keeping the vector on the simplex.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            if rng.random() < 0.001:
                a = a_prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the target
    evaluation function within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Baselines
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # deterministic arcsine‑type shape
    try:
        arcs = _arcsine_shape(n)
        arcs_score = _eval_fast(arcs, L_fft)
        if arcs_score < best_score:
            best_seq = arcs.copy()
            best_score = arcs_score
            print(f"[{time.time() - start_time:.2f}s] arcsine baseline score={arcs_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Use the globally supplied starter if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_seq = starter.copy()
                best_score = cur
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep (geometric × power × tail)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + min(120.0, deadline - time.time())
    if time.time() < sweep_deadline:
        rp_seq, rp_score, rp_params = _random_param_search(n,
                                                          sweep_deadline,
                                                          start_time)
        if rp_seq is not None and rp_score < best_score:
            best_seq = rp_seq.copy()
            best_score = rp_score
            best_params = rp_params
            print(f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}")
        else:
            best_params = (0.95, 0.0, 0.0, 0.3)   # fallback
    else:
        best_params = (0.95, 0.0, 0.0, 0.3)

    # ------------------------------------------------------------------
    # 2️⃣  Fine hill‑climb on the four parameters
    # ------------------------------------------------------------------
    hill_deadline = min(deadline, time.time() + 150.0)
    best_seq, best_score = _hillclimb_params(best_params,
                                             best_seq,
                                             best_score,
                                             hill_deadline,
                                             start_time)

    # ------------------------------------------------------------------
    # 3️⃣  Multiple short Adam descents from diverse starts
    # ------------------------------------------------------------------
    max_restarts = 8
    restarts_done = 0
    while time.time() + 30.0 < deadline and restarts_done < max_restarts:
        restarts_done += 1
        # either jitter the current best or draw a fresh parametric shape
        if rng.random() < 0.5:
            a0 = np.maximum(best_seq + 1e-4 * rng.standard_normal(n) * best_seq,
                            0.0)
            a0 = _project_simplex(a0)
        else:
            # fresh random parametric shape
            r = np.exp(rng.uniform(np.log(0.8), np.log(0.999999)))
            p = rng.uniform(-5.0, 5.0)
            q = rng.uniform(-5.0, 5.0)
            lam = rng.uniform(0.0, 1.0)
            shape = _make_shape_general(n, r, p, q)
            a0 = _mix_with_uniform(shape, lam)

        sub_deadline = time.time() + 30.0
        a_opt, sc_opt = _adam_subgradient_descent(a0,
                                                  sub_deadline,
                                                  start_time,
                                                  best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] new best after Adam descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny hill‑climb (random mass transfers)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_final, sc_final = _final_local_search(best_seq,
                                                deadline,
                                                start_time,
                                                best_score)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Log the official evaluation (uses the provided security‑checked function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
