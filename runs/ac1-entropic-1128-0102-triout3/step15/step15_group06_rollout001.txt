"""
Hybrid optimiser for the step‑function bound

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works as follows (all steps respect the 1000 s wall‑clock limit).

1. **Problem size** – pick a length N (up to 32768) based on the available time.
2. **Diverse initial pool** – generate many analytic shapes
   (uniform, power‑ramps, exponentials, power‑laws, asymmetric Beta,
   Gaussian, random bimodal mixtures and Dirichlet draws) and keep the best
   TOP_K candidates (by the exact objective).
3. **Exponentiated sub‑gradient descent** – run the classic multiplicative
   mirror‑descent on the active‑set ℓ∞ formulation for each top candidate.
4. **Simulated annealing** – starting from the current best vector,
   perform random mass‑exchange moves (preserving the simplex constraint)
   and accept them with a temperature schedule, allowing escape from local
   minima.
5. **Adam polishing** – finish the run with a smooth‑max (log‑sum‑exp)
   surrogate of the max‑convolution using Adam and a decreasing soft‑max
   temperature.
6. Return the best vector (normalised to sum = 1) as a plain Python list.

All heavy operations (convolutions, FFTs, projections) are vectorised.
The routine checks the wall‑clock frequently and stops just before the
deadline, guaranteeing a valid answer.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip possible tiny negatives from numerical noise
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be non‑negative and already normalised to sum = 1.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    # sum(seq) == 1 by construction, but we keep the formula for safety.
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian(n: int, sigma: float, center_frac: float = 0.5) -> np.ndarray:
    """Discrete Gaussian on {0,…,n‑1}."""
    center = center_frac * (n - 1)
    idx = np.arange(n, dtype=float)
    shape = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_bimodal_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of two Gaussians (different centres, sigmas, weight)."""
    w = rng.random()
    sigma1 = rng.uniform(n * 0.01, n * 0.2)
    sigma2 = rng.uniform(n * 0.01, n * 0.2)
    c1 = rng.uniform(0.0, 1.0)
    c2 = rng.uniform(0.0, 1.0)
    g1 = _make_gaussian(n, sigma1, c1)
    g2 = _make_gaussian(n, sigma2, c2)
    shape = w * g1 + (1.0 - w) * g2
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Simple top‑k pool utilities
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (active‑set ℓ∞ optimisation)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent on max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad_fft = np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L)
        grad = np.fft.irfft(grad_fft, n=L)[n - 1 : n - 1 + n]
        grad = 2.0 * grad

        eta = eta0 / (1.0 + it / 5000.0)
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of a log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.1
    T0 = 0.1
    T_end = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        T = T_end + (T0 - T_end) * (0.5 ** (it / 4000.0))
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad_fft = np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L)
        grad = np.fft.irfft(grad_fft, n=L)[n - 1 : n - 1 + n]
        grad = 2.0 * grad

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best, best_score

# Simulated annealing on the simplex
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Simple SA that moves a small amount of mass between two coordinates."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best = a.copy()
    best_score = _objective(a)

    # Temperature starts as a fraction of the current score and cools linearly.
    init_temp = max(0.1, best_score * 0.1)
    it = 0
    # Adaptive step scale (fraction of the source entry)
    step_scale = 0.02

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = rng.integers(n, size=2)
        while i == j:
            j = rng.integers(n)

        # proposed transfer amount
        delta = step_scale * a[i] * rng.random()
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # a_new stays on the simplex (total mass unchanged, non‑negativity by construction)

        cur_score = _objective(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            a = a_new
            # occasionally increase step size if we are making progress
            step_scale = min(0.2, step_scale * 1.02)
        else:
            # Metropolis acceptance
            remaining = deadline - time.time()
            # Linear cooling: temperature proportional to time left
            temp = init_temp * (remaining / (deadline - start_time))
            if temp > 0 and rng.random() < math.exp(-(cur_score - best_score) / temp):
                a = a_new
                # reduce step size when moving uphill
                step_scale = max(0.005, step_scale * 0.99)

        if it % 2000 == 0:
            print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} best={best_score:.6f}")

    return best, best_score

        _add(_make_power_law(N, p))

    # Asymmetric Beta families
    alphas = np.linspace(0.2, 3.0, 7)
    betas = np.linspace(0.2, 3.0, 7)
    for a in alphas:
        for b in betas:

    # Gaussian families
    for sigma_frac in np.linspace(0.02, 0.2, 8):
        sigma = sigma_frac * N
        _add(_make_gaussian(N, sigma, center_frac=0.5))
        _add(_make_gaussian(N, sigma, center_frac=0.3))
        _add(_make_gaussian(N, sigma, center_frac=0.7))

    rng = np.random.default_rng(seed + 12345)
    for _ in range(40):
        if time.time() >= deadline:
            break

                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:
                starter_up /= s
                _add(starter_up)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the best candidate from the pool as a starting point.
    # ------------------------------------------------------------------
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        gd_time = min(0.55 * remaining, 550.0)   # keep a margin for later phases

        idx = 0
        while time.time() < gd_deadline:
            vec = pool[idx % len(pool)][1]
            seq, sc = _exp_subgradient_descent(vec, gd_deadline, start)
            if sc < best_score:
                best_score, best_vec = sc, seq.copy()
                print(f"[{time.time() - start:.2f}s] ExpGD improvement → {sc:.6f}")
            idx += 1

    # ------------------------------------------------------------------
    # 3) Simulated annealing to escape possible local minima
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:

        seq, sc = _simulated_annealing(best_vec, sa_deadline, start)
        if sc < best_score:
            best_score, best_vec = sc, seq.copy()
            print(f"[{time.time() - start:.2f}s] SA improvement → {sc:.6f}")

    # ------------------------------------------------------------------

    if remaining > 5.0:

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")
    return [float(v) for v in best_vec]
