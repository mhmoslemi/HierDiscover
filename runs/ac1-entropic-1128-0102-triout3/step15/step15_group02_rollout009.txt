"""
Propose a step‑function (non‑negative sequence) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm is deliberately different from the reference implementation.
It consists of three stages:

1️⃣  A very fast parametric sweep (uniform, geometric, power, beta‑type,
    sine‑power and Gaussian shapes) that provides a decent initial bound.

2️⃣  Massive random‑seed exploration:  many Dirichlet‑like draws are created,
    each is refined by a **projected sub‑gradient** descent on the exact
    max‑convolution (additive updates, simplex projection).  The best of all
    candidates is kept.

3️⃣  A final long‑run additive sub‑gradient phase on the current best sequence,
    followed by a short simulated‑annealing hill‑climb to polish the result.

All heavy work uses FFT‑based convolution, the simplex projection
(`_project_simplex`) guarantees non‑negativity and unit L¹‑norm, and the
runtime never exceeds the supplied budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the L∞‑norm of the self‑convolution at a.
    If multiple shifts attain the maximum their contributions are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used during the search: 2·n·max(conv(seq,seq))   (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (very lightweight sweep)
# ----------------------------------------------------------------------
def _family_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a random candidate from a handful of analytic families."""
    idx = np.arange(n, dtype=np.float64)

    family = rng.integers(6)        # 0…5
    if family == 0:                 # uniform (baseline)
        seq = np.full(n, 1.0 / n)

    elif family == 1:               # geometric decay
        eps = rng.uniform(1e-5, 1e-1)
        r = 1.0 - eps
        geo = r ** idx
        seq = geo / geo.sum()

    elif family == 2:               # power law (positive or negative)
        p = rng.uniform(-2.0, 4.0)
        pow_seq = (idx + 1.0) ** p
        seq = pow_seq / pow_seq.sum()

    elif family == 3:               # beta‑type (two shape parameters)
        a = rng.uniform(0.05, 8.0)
        b = rng.uniform(0.05, 8.0)
        log_seq = (a - 1.0) * np.log(idx + 1.0) + (b - 1.0) * np.log(
            np.maximum(n - idx, 1.0)
        )
        seq = np.exp(log_seq - np.max(log_seq))
        seq = seq / seq.sum()

    elif family == 4:               # sine‑power (symmetric bump)
        p = rng.uniform(0.1, 8.0)
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.maximum(seq, 0.0)
        seq = seq / seq.sum()

    else:                           # Gaussian‑like bump centred in the middle
        sigma = rng.uniform(0.05, 0.4) * n  # relative width
        mu = (n - 1) / 2.0
        seq = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        seq = np.maximum(seq, 0.0)
        seq = seq / seq.sum()

    # optional mixing with a tiny uniform component to avoid pathological spikes
    lam = rng.uniform(0.0, 0.2)
    seq = (1.0 - lam) * seq + lam * (1.0 / n)
    seq = seq / seq.sum()
    return seq


# ----------------------------------------------------------------------
# Additive sub‑gradient descent (projected onto the simplex)
# ----------------------------------------------------------------------
def _sgd_subgradient(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iters: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the exact max‑convolution.
    The step size decays like η₀/√t.  The routine returns the best sequence
    seen during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    it = 0
    eta0 = 0.5

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SGD it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the L∞‑norm
        grad = _subgradient_maxconv(a, conv)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        a = a_new
    return best_seq, best_score


# ----------------------------------------------------------------------
# Final hill‑climb / simulated‑annealing (as in the reference)
# ----------------------------------------------------------------------
def _final_hill_climb(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # annealing schedule (total time left -> exponential cooling)
    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0.0:
        return best_seq, best_score
    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # allow occasional random picks when src==dst
        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # transfer a (fractional) amount of mass
        frac = rng.random() * 0.02 + 1e-4    # up to ~2 % of the source weight
        eps = a[src] * frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-14:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine follows three stages (fast sweep,
    massive random‑seed refinement with projected sub‑gradient descent,
    and a final hill‑climb).  It returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3      # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initial uniform baseline
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (e.g. from a previous run)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Tiny parametric sweep (very cheap)
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0          # reserve time for the heavy phase
    if time.time() < phase_deadline:
        rng = np.random.default_rng()
        for _ in range(300):                  # a few hundred cheap trials
            cand = _family_candidate(rng, n)
            sc = _objective_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Sweep candidate score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Massive random‑seed exploration + short SGD refinement
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    max_seeds = 5000
    seed_counter = 0
    while time.time() < deadline and seed_counter < max_seeds:
        # draw a random Dirichlet‑like vector (concentration varies a lot)
        log_c = rng.uniform(np.log(0.01), np.log(30.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a = a / a.sum()

        # blend with a tiny uniform part to stay away from pathological spikes
        lam = rng.uniform(0.0, 0.1)
        a = (1.0 - lam) * a + lam * (1.0 / n)
        a = a / a.sum()

        # a short sub‑gradient refinement (≈30 iters – cheap but often improves)
        a, sc = _sgd_subgradient(
            a,
            start_time,
            deadline,
            best_seq,
            best_score,
            max_iters=30,
        )
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] "
                f"Random seed {seed_counter:,} improved score={sc:.6f}"
            )
        seed_counter += 1

    # --------------------------------------------------------------
    # 3️⃣  Long‑run additive sub‑gradient descent on the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting long SGD on best")
        best_seq, best_score = _sgd_subgradient(
            best_seq,
            start_time,
            deadline,
            best_seq,
            best_score,
            max_iters=200_000,   # run until time runs out
        )
        print(f"[{time.time() - start_time:.2f}s] After long SGD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final hill‑climb / simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return (guard against possible numerical drift)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
