"""
Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines:

1️⃣  Broad parametric sweeps (geometric, power, symmetric/asymmetric Beta,
    symmetric exponential, two‑block).
2️⃣  Random Dirichlet draws with a wide range of concentrations.
3️⃣  Additive projected‑gradient descent on the simplex (with a simple
    line‑search step size schedule).
4️⃣  Aggressive hill‑climbing that tries several transfer fractions and
    chooses the best improvement.
5️⃣  A short simulated‑annealing phase that makes occasional random moves
    with a temperature schedule.

All intermediate vectors are symmetrised (a ← (a+a[::-1])/2) because the
optimal solution is expected to be symmetric.  The routine respects the
global 1000 s budget and returns a plain Python list ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 300)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = (seq + seq[::-1]) / 2.0
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score

def _sweep_two_blocks(n, best_seq, best_score, deadline):

            # fallback: pick a random non‑zero source
            src_candidates = np.where(a > 1e-12)[0]
            if src_candidates.size == 0:
                break
            src = rng.choice(src_candidates)

        improved = False
        best_local = a.copy()
        best_local_score = best_score

        for f in fractions:
            eps = a[src] * f
            if eps <= 0:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            a_prop = _project_simplex(a_prop)
            a_prop = (a_prop + a_prop[::-1]) / 2.0
            sc = 2.0 * n * _conv_self(a_prop).max()
            if sc < best_local_score - 1e-12:
                best_local_score = sc
                best_local = a_prop
                improved = True

        if improved:
            a = best_local
            best_seq = best_local.copy()
            best_score = best_local_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] HillLS score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                # occasional shake‑up
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps = a[i1] * (rng.random() * 0.3 + 1e-5)
                a[i1] -= eps
                a[i2] += eps
                a = _project_simplex(np.maximum(a, 0.0))
                a = (a + a[::-1]) / 2.0
                no_improve = 0
    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Simulated‑annealing tail (random moves with temperature)
# ----------------------------------------------------------------------


def _simulated_annealing(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        elapsed = time.time() - start
        # linear cooling from T0 = 2e‑2 to near zero
        T = max(1e-8, 0.02 * (1.0 - elapsed / total))

        # pick source proportional to its mass (more mass → more chance)
        src = rng.choice(n, p=a / a.sum())
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if dst == src:
            continue

        eps = a[src] * rng.random() * 0.25  # up to 25 % of the source mass
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        a_new = _project_simplex(a_new)
        a_new = (a_new + a_new[::-1]) / 2.0

        cur_score = 2.0 * n * _conv_self(a).max()
        new_score = 2.0 * n * _conv_self(a_new).max()
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / T):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] SA score={best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain list of floats) found before the
    budget expires.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter if available (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                starter = (starter + starter[::-1]) / 2.0
                starter /= starter.sum()
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (allocate ~200 s)
    # ------------------------------------------------------------------
    sweep_deadline = min(deadline - 600.0, start_time + 200.0)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, sweep_deadline)

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet draws (≈150 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline - 450.0, time.time() + 150.0)
    if time.time() < dirichlet_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score,
                                                        dirichlet_deadline, rng)

    # ------------------------------------------------------------------
    # 3️⃣  Additive projected‑gradient descent (≈200 s)
    # ------------------------------------------------------------------
    grad_deadline = min(deadline - 250.0, time.time() + 200.0)
    if time.time() < grad_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting additive gradient descent")
        best_seq, best_score = _additive_projected_gradient(best_seq, grad_deadline,
                                                            best_seq, best_score)

    # ------------------------------------------------------------------
    # 4️⃣  Hill‑climb with line‑search (≈200 s)
    # ------------------------------------------------------------------
    hill_deadline = min(deadline - 50.0, time.time() + 200.0)
    if time.time() < hill_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb with line‑search")
        best_seq, best_score = _hill_climb_line_search(best_seq, hill_deadline,
                                                       best_seq, best_score)

    # ------------------------------------------------------------------
    # 5️⃣  Simulated‑annealing tail (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated‑annealing tail")
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    best_seq, best_score)

    # ------------------------------------------------------------------
    # Final sanitisation
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = (best_seq + best_seq[::-1]) / 2.0    # enforce symmetry one last time
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
