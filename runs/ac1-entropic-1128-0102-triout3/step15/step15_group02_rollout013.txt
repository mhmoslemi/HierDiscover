def _lp_descent(f_init, start_time, deadline, best_seq, best_score):
    import cvxpy as cp, scipy.sparse as sp
    n = len(f_init)
    f = f_init.copy()
    eps_f = 1e-12
    # Precompute row and col indices for matrix A (dense later)
    row_idx = np.arange(n)[:, None] + np.arange(n)[None, :]  # shape (n,n)
    row_idx = row_idx.ravel()
    col_idx = np.repeat(np.arange(n), n)
    # iteration loop
    it = 0
    while time.time() < deadline:
        it += 1
        # compute M
        conv_cur = _conv_self(f)
        M = conv_cur.max()
        # Build matrix A (dense)
        data = np.repeat(np.maximum(f, eps_f), n)   # length n*n
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n)).toarray()
        # Solve LP
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        # try solving
        try:
            prob.solve(solver=cp.ECOS, max_iters=2000, warm_start=True, abstol=1e-8, reltol=1e-8)
        except cp.error.SolverError:
            prob.solve(solver=cp.SCS, max_iters=5000, warm_start=True)
        if g.value is None:
            print(f"[{time.time()-start_time:.2f}s] LP failed at it {it}")
            break
        g_vec = np.array(g.value).flatten()
        sum_g = g_vec.sum()
        if sum_g <= 1e-12:
            break
        g_norm = g_vec / sum_g
        # evaluate objective

"""
Propose a step‑function (non‑negative sequence) that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The new search pipeline:
  0.  Choose a moderate problem size (n≈1024) – large n raises the
      prefactor 2·n and makes the LP too big.
  1.  Initialise from the supplied global ``height_sequence_1`` (down‑sampled)
      or from the uniform distribution.
  2.  **Linear‑programming descent** – a classic Frank‑Wolfe style step:
        given the current f with max‑convolution M,
        solve   max 1ᵀg   s.t.  f∗g ≤ M  (component‑wise),  g≥0 .
        The LP raises the ℓ₁‑norm of g; after normalising it the
        max‑convolution drops.  A convex combination with the old f
        guarantees feasibility.  The LP is solved with CVXPy (ECOS) on a
        sparse convolution matrix.  This stage typically reduces the
        bound by a few ×10⁻³.
  3.  Refine with the smooth‑max Adam optimiser from the previous
      submission.
  4.  Further polish with exact sub‑gradient exponentiated‑gradient descent.
  5.  A final simulated‑annealing hill‑climb.

All stages respect the global 1000 s budget.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level utilities (same as the previous submission)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the max‑convolution at a (averaged over all active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used by the optimiser: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  LP‑based descent (Frank–Wolfe style)
# ----------------------------------------------------------------------
def _lp_descent(f_init: np.ndarray,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """
    Given a current feasible f (sum=1, non‑negative) with max‑convolution M,
    repeatedly solve the LP
        maximise   1ᵀg
        subject to  f ∗ g ≤ M   (element‑wise)
                    g ≥ 0 .
    The LP usually yields a g with larger ℓ₁‑norm; after renormalisation
    the objective improves.  A convex combination with the previous f
    guarantees feasibility.
    """
    eps = 1e-12          # avoid exact zeros → unbounded LP
    f = f_init.copy()
    n = f.shape[0]

    # pre‑compute the index pattern of the convolution matrix (independent of f)
    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    col_idx = np.repeat(np.arange(n), n)

    it = 0
    while time.time() < deadline:
        it += 1
        # current bound M
        cur_conv = _conv_self(f)
        M = cur_conv.max()

        # build the linear convolution matrix A (dense, but only ~16 MiB for n=1024)
        data = np.repeat(np.maximum(f, eps), n)
        A = sp.csr_matrix((data, (row_idx, col_idx)),
                          shape=(2 * n - 1, n)).toarray()

        # solve LP: maximise sum(g)  s.t. A·g ≤ M , g ≥ 0
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, max_iters=2000,
                       warm_start=True, abstol=1e-8, reltol=1e-8)
        except (cp.error.SolverError, ValueError):
            prob.solve(solver=cp.SCS, max_iters=5000, warm_start=True)

        if g.value is None:
            print(f"[{time.time() - start_time:.2f}s] LP failed at it {it}")
            break

        g_vec = np.asarray(g.value).ravel()
        sum_g = g_vec.sum()
        if sum_g <= eps:
            break

        g_norm = g_vec / sum_g                     # ℓ₁‑normalised version
        new_score = _objective_score(g_norm)

        # direct replacement improvement
        if new_score < best_score - 1e-12:
            best_seq = g_norm.copy()
            best_score = new_score
            f = g_norm.copy()
            print(f"[{time.time() - start_time:.2f}s] LP it={it} direct improve score={best_score:.6f}")
            continue

        # otherwise try a convex combination (t = 0.5)
        t = 0.5
        mixed = (1.0 - t) * f + t * g_norm
        mixed /= mixed.sum()
        mixed_score = _objective_score(mixed)

        if mixed_score < best_score - 1e-12:
            best_seq = mixed.copy()
            best_score = mixed_score
            f = mixed.copy()
            print(f"[{time.time() - start_time:.2f}s] LP it={it} mixed improve score={best_score:.6f}")
            continue

        # no improvement → stop the LP phase
        print(f"[{time.time() - start_time:.2f}s] LP it={it} no improvement, exit.")
        break

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Adam optimisation of a smooth‑max surrogate (identical to previous)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, start_time: float,
                   deadline: float, best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    tau = 20.0
    tau_factor = 1.2
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # soft‑max over convolution values
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max (same as in the original script)
        a_rev = a[::-1]
        conv_w = np.fft.irfft(np.fft.rfft(w, n=_next_pow2(3 * n - 2)) *
                              np.fft.rfft(a_rev, n=_next_pow2(3 * n - 2)),
                              n=_next_pow2(3 * n - 2))[:n]
        grad = 2.0 * conv_w

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _eg_maxconv_descent(a0: np.ndarray,
                        start_time: float,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.05            # initial step size
    it = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        eta_t = eta / (1.0 + it * 1e-4)

        exp_factor = np.exp(np.clip(-eta_t * grad, -700, 700))
        a_new = a * exp_factor
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        new_score = _objective_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG it={it:,} score={cur_score:.6f}")
        else:
            eta *= 0.99
            if np.random.rand() < 0.001:
                a = a_new
                cur_score = new_score
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb / simulated‑annealing (identical to previous)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score

    The routine combines a dedicated LP descent with Adam, EG and a final
    annealed hill‑climb.  The sequence length is chosen as 1024 to keep the
    LP tractable while still allowing a very low bound (target < 1.503).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Problem size – moderate to keep the LP cheap
    # --------------------------------------------------------------
    n = 1024

    # --------------------------------------------------------------
    # Initialise (try to reuse the global construction if present)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # down‑sample to length n by linear interpolation on the index axis
            if starter.size != n:
                xs = np.linspace(0, starter.size - 1, n)
                starter = np.interp(xs, np.arange(starter.size), starter)

        else:
            raise Exception
    except Exception:
        f0 = np.full(n, 1.0 / n)

    if f0.sum() == 0:

    else:
        f0 = f0 / f0.sum()

    best_seq = f0.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  LP‑based Frank–Wolfe descent
    # --------------------------------------------------------------
    lp_deadline = deadline - 500.0   # reserve the rest for refinement phases
    if time.time() < lp_deadline:

        best_seq, best_score = _adam_optimize(best_seq, start_time,

        print(f"[{time.time() - start_time:.2f}s] After Adam best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient descent on exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:

        best_seq, best_score = _eg_maxconv_descent(best_seq, start_time,
                                                   deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After EG best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final hill‑climb / simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(best_seq, start_time,
                                                 deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:

    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq.tolist()]
