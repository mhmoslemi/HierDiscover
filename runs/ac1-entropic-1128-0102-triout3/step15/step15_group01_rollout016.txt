import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The search is deterministic: it tries power‑law shapes a_i ∝ (i+1)^α
    over a wide range of exponents (0 ≤ α ≤ 10) and keeps the exponent that
    gives the smallest evaluation.  The optimisation uses a cheap golden‑section
    line search; each candidate is normalised to sum‑1 before being fed to
    ``evaluate_sequence``.  The final sequence is returned as a plain Python list.
    """
    # time guard
    start = time.time()
    deadline = start + budget_s - 0.1  # tiny safety margin

    # length of the step function – 4096 is a sweet spot used previously
    n = 4096
    idx = np.arange(1, n + 1, dtype=np.float64)

    # helper: evaluate a given exponent α
    def evaluate_alpha(alpha: float) -> float:
        # generate power‑law vector
        seq = np.power(idx, alpha, dtype=np.float64)
        total = seq.sum()
        if total == 0.0:
            return np.inf
        seq /= total
        # evaluate – converting to list because the supplied evaluator expects a list
        return evaluate_sequence(seq.tolist())

    # golden‑section parameters
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    lo, hi = 0.0, 10.0
    # initial interior points
    c = lo + (1 - phi) * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = evaluate_alpha(c)

        a_i ∝ (i+1)^α   (i = 0 … n‑1)
