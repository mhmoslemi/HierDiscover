"""
search algorithm for the step‑function optimisation problem

The method works with a relatively large length (n≈8000) to make the
maximal self‑convolution as small as possible.  It starts from several
random parametric shapes, then runs a projected gradient descent on a
soft‑max approximation of the max‑convolution (the “smooth max”).  The
gradient is obtained analytically via FFTs:

    grad ≈ 2·ifft( fft(a) * fft( softmax(conv(a,a)) ) )

Momentum, normalisation of the gradient and an annealing schedule for the
soft‑max temperature are used.  After the main phase a tiny random
mass‑transfer hill‑climb finishes the optimisation.  The best sequence
found (normalised to sum 1) is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (returns length 2·n‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L: int) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) / (sum(a)**2) (sum≈1)."""
    conv = _conv_self_fft(a, L)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric·power·tail family)
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """Create a normalised vector proportional to r**i·(i+1)**p·(n‑i)**q."""
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)                 # avoid overflow
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return λ·uniform + (1−λ)·shape (both already sum to 1)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Soft‑max of the convolution (used as a smooth surrogate of max)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over convolution entries with temperature 1/α."""
    cmax = conv.max()
    exps = np.exp(alpha * (conv - cmax))
    w = exps / exps.sum()
    return w


# ----------------------------------------------------------------------
# Gradient descent on the smooth max objective
# ----------------------------------------------------------------------


def _smooth_max_descent(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth max (soft‑max) objective.
    Returns the best vector found and its true (max‑convolution) score.
    """
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)

    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # hyper‑parameters
    step0 = 0.20          # initial step‑size (in L2‑norm units)
    beta = 0.90           # momentum coefficient
    v = np.zeros_like(a)  # momentum buffer

    alpha = 2.0           # start temperature (softness)
    alpha_max = 50.0
    inc_every = 800       # increase α every this many iterations

    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        # ---------- convolution and current true score ----------
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            v = np.zeros_like(a)     # reset momentum after improvement
            print(
                f"[{time.time() - start_time:.2f}s] smooth it={it:,} "
                f"score={cur_score:.6f} α={alpha:.2f}"
            )

        # ---------- soft‑max weights ----------
        w = _softmax_weights(conv, alpha)

        # ---------- gradient of the smooth max ----------
        Wf = np.fft.rfft(w, n=L)
        Gf = 2.0 * Af * Wf
        grad = np.fft.irfft(Gf, n=L)[:n].real

        # normalise gradient to unit length for a stable step size
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            grad = grad / gnorm

        # ---------- momentum & step ----------
        v = beta * v + (1.0 - beta) * grad
        step = step0 / (np.sqrt(it) + 1.0)
        a_candidate = a - step * v
        a_candidate = _project_simplex(a_candidate)

        a = a_candidate

        # ---------- anneal the temperature ----------
        if it % inc_every == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.25)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random mass‑transfer hill‑climb (final polishing)
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,

    """
    Randomly moves a tiny fraction of probability mass between two
    coordinates while staying on the simplex.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to 0.2 % of the source mass
        eps = best_a[src] * rng.random() * 2e-3
        a_prop = best_a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # evaluate
        new_sc = _eval_fast(a_prop, L)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            # occasional uphill acceptance to escape tiny plateaus
            if rng.random() < 0.001:
                best_a = a_prop

    return best_a, best_sc
