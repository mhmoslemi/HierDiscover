"""Hybrid global+local optimizer for the step‑function bound.
1. Build a diverse pool of analytic and random Dirichlet shapes (size ~N).
2. Run a differential‑evolution search on a low‑dimensional block‑wise
   parametrisation (K≈N/256) to explore the global landscape.
3. Refine the best candidate with exponentiated sub‑gradient descent
   (multiplicative updates) and finish with a smooth‑max Adam polish.
All operations stay on the probability simplex; the final vector is
returned as a plain Python list.
"""
import time, math
import numpy as np

# ---------- utilities ----------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return np.maximum(conv, 0.0)

def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# ---------- analytic shape generators ----------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# ---------- optimisation primitives ----------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             base_eta: float = 0.8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    it = 0
    eta0 = base_eta
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + it / 4000.0)
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")
    return best, best_score


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.2
    T0 = 0.2
    T_end = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        T = T_end + (T0 - T_end) * (0.5 ** (it / 3000.0))
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it + 1)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)
    return best, best_score

# ---------- block‑wise DE ----------
def _block_expand(weights: np.ndarray, n: int) -> np.ndarray:
    """Expand a K‑dim weight vector (already normalised) to length n by uniform blocks."""
    k = weights.shape[0]
    base = n // k
    remainder = n % k
    # first `remainder` blocks get an extra element
    blocks = []
    for i, w in enumerate(weights):
        size = base + (1 if i < remainder else 0)
        blocks.append(np.full(size, w))
    return np.concatenate(blocks)

"""Hybrid global + local optimiser for the step‑function bound.

The routine works in three stages:

1. **Analytic + random pool** – a modest number of hand‑crafted shapes
   (uniform, ramps, exponentials, power‑laws, asymmetric Betas, sorted
   Dirichlet draws) is built and the TOP_K best are kept.

2. **Global DE on a block‑wise parametrisation** – the distribution is
   restricted to K uniform blocks (K≈N/256).  A differential‑evolution
   search over the log‑weights of the blocks explores the landscape
   globally while still being cheap (≈3000 objective evaluations).

3. **Local refinement** – the best candidate from the pool or DE is
   refined by an exponentiated sub‑gradient descent (multiplicative
   updates) and finally polished with a smooth‑max Adam optimiser.
All iterates stay on the probability simplex; the final list is returned
as plain Python floats (the evaluator will rescale it as needed).

    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# ----------------------------------------------------------------------
# Analytic shape generators (all return a normalised probability vector)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    tot = shape.sum()
    return shape / tot if tot > 0 else np.full(n, 1.0 / n)


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    tot = shape.sum()
    return shape / tot if tot > 0 else np.full(n, 1.0 / n)


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    tot = shape.sum()
    return shape / tot if tot > 0 else np.full(n, 1.0 / n)


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    tot = shape.sum()
    return shape / tot if tot > 0 else np.full(n, 1.0 / n)

# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (active‑set ℓ∞ optimisation)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             base_eta: float = 0.8) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent on max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    it = 0
    eta0 = base_eta
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + it / 4000.0)
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        cur = _objective(a)
        if cur < best_score:
            best_score, best = cur, a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent – final polishing
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.2
    T0, T_end = 0.2, 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur < best_score:
            best_score, best = cur, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur:.6f}")

        T = T_end + (T0 - T_end) * (0.5 ** (it / 3000.0))
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it + 1)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)
    return best, best_score


# ----------------------------------------------------------------------
# Helper to expand a block‑wise weight vector to a full length‑N vector
# ----------------------------------------------------------------------
def _block_expand(weights: np.ndarray, n: int) -> np.ndarray:
    """Uniformly expand a K‑dim simplex vector to length n."""
    k = weights.shape[0]
    base, rem = divmod(n, k)
    blocks = []
    for i, w in enumerate(weights):
        size = base + (1 if i < rem else 0)
        blocks.append(np.full(size, w))
    return np.concatenate(blocks)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function.
    The routine builds a compact pool, runs a block‑wise DE,
    refines the best candidate with exponentiated sub‑gradient descent,
    and finishes with a smooth‑max Adam polish.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1  # safety margin

    # --------------------------------------------------------------
    # Choose a problem size – big enough to be expressive, cheap enough to iterate
    # --------------------------------------------------------------
    remaining = deadline - start
    if remaining > 800:
        N = 16384
    elif remaining > 400:
        N = 8192
    else:
        N = 4096

    # --------------------------------------------------------------
    # 1) Build a diverse initial pool (analytic + random)
    # --------------------------------------------------------------
    TOP_K = 20
    pool = []  # (score, vec)

    def _add(vec: np.ndarray) -> None:
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s > 0.0:
            vec = vec / s
        else:
            vec = np.full_like(vec, 1.0 / vec.shape[0])
        _push_top(pool, _objective(vec), vec, TOP_K)

    # Uniform baseline
    _add(_make_uniform(N))

    # Power ramps
    for p in np.linspace(0.1, 3.0, 7):
        for plat in np.linspace(0.0, 0.4, 3):
            _add(_make_power_ramp(N, p, plat))

    # Exponential decays
    for lam in np.linspace(0.3, 5.0, 12):
        _add(_make_exponential(N, lam))

    # Power‑law decays
    for p in np.linspace(0.3, 4.0, 12):
        _add(_make_power_law(N, p))

    # Asymmetric Beta families
    for a in np.linspace(0.2, 4.0, 5):
        for b in np.linspace(0.2, 4.0, 5):
            _add(_make_beta_shape_asym(N, a, b))

    # Sorted Dirichlet draws (monotone decreasing)
    rng = np.random.default_rng(seed + 777)
    for conc in [0.05, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(30):
            if time.time() >= deadline:
                break
            vec = rng.gamma(conc, 1.0, size=N)
            vec = np.sort(vec)[::-1]                 # decreasing shape
            _add(vec)

    # Incorporate a previously found solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != N:
                x_old = np.linspace(0, 1, prev.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                prev_up = np.interp(x_new, x_old, prev)
            else:
                prev_up = prev.copy()
            _add(prev_up)
    except Exception:
        pass

    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Global DE on a block‑wise parametrisation
    # --------------------------------------------------------------
    # Aim for roughly 256 points per block → K ≈ N/256
    K = max(4, N // 256)

    # Rough estimate of how many evaluations we can still afford
    est_eval_time = 0.006                                 # seconds per objective (empirical)
    max_evals = int((deadline - time.time()) / (est_eval_time * 1.1))
    popsize = 15                                          # default for SciPy DE
    npop = popsize * K
    maxiter = max(0, (max_evals // npop) - 1)
    maxiter = min(maxiter, 300)                           # safety cap

    if maxiter >= 1:
        try:
            from scipy.optimize import differential_evolution

            def de_obj(logw):
                w = np.exp(logw)
                w /= w.sum()
                a = _block_expand(w, N)
                return _objective(a)

            bounds = [(-8.0, 8.0)] * K
            print(f"[{time.time() - start:.2f}s] Starting DE (K={K}, maxiter={maxiter})")
            # The callback aborts if we run out of time
            result = differential_evolution(
                de_obj,
                bounds,
                strategy='best1bin',
                maxiter=maxiter,
                popsize=popsize,
                tol=1e-6,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=seed,
                polish=False,
                updating='deferred',
                workers=1,
                callback=lambda x, convergence: time.time() >= deadline
            )
            de_vec = _block_expand(np.exp(result.x) / np.exp(result.x).sum(), N)
            de_score = _objective(de_vec)
            print(f"[{time.time() - start:.2f}s] DE finished best={de_score:.6f}")
            if de_score < best_score:
                best_score, best_vec = de_score, de_vec.copy()
        except Exception as exc:
            print(f"DE failed ({exc}). Continuing without global DE.")
    else:
        print(f"[{time.time() - start:.2f}s] Insufficient time for DE (maxiter={maxiter}).")

    # --------------------------------------------------------------
    # 3) Local refinement – exponentiated sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_end = time.time() + 0.65 * remaining
        seq, sc = _exp_subgradient_descent(best_vec, sub_end, start, base_eta=0.9)
        if sc < best_score:
            best_score, best_vec = sc, seq.copy()
            print(f"[{time.time() - start:.2f}s] ExpGD improvement → {sc:.6f}")

    # --------------------------------------------------------------
    # 4) Final polishing with smooth‑max Adam descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq, sc = _smooth_gradient_descent(best_vec, deadline, start)
        if sc < best_score:
            best_score, best_vec = sc, seq.copy()
            print(f"[{time.time() - start:.2f}s] SmoothGD final improvement → {sc:.6f}")

    # --------------------------------------------------------------
    # 5) Normalise and return the final vector (as a plain list)
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    best_vec = best_vec / total if total > 0 else np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
