```
def _adam_expgrad(a0, n, L_fft, deadline, start_time):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_seq = a.copy()
    best_score = _objective_score(a, n)
    while time.time() < deadline:
        t+=1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        # compute subgradient of max conv
        grad = _subgradient_maxconv(a, conv)
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # step size
        eta = 0.5 / np.sqrt(t)  # similar scaling
        exp_factor = np.exp( - eta * m_hat / (np.sqrt(v_hat) + eps) )
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a = a * exp_factor
        # renorm
        sum_a = a.sum()
        if sum_a == 0: a = np.full_like(a, 1.0/n)
        else: a = a / sum_a
        # evaluate
        new_score = 2.0 * n * _conv_self(a, L_fft).max()
        if new_score < best_score:
            best_score = new_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam step t={t:,} best={best_score:.6f}")
    return best_seq, best_score
```

The scoring function is  

```
score = 2·n·max(conv(a,a)) / (sum(a)²)          (n = len(a))
```

Since the score is homogeneous, we may always normalise the vector to
`sum(a)=1`.  The problem is then to minimise  

```
F(a) = 2·n·max_k Σ_i a_i·a_{k‑i}      ,   a_i ≥ 0 , Σ a_i = 1 .
```

A convenient family of symmetric, non‑negative vectors is

```
a_i(p) ∝ ((i+½)·(n‑½‑i))^p ,   i = 0 … n‑1        (p ∈ ℝ)
```

for `p=0` we obtain the uniform distribution (score = 2), while
negative `p` concentrates mass near the ends of the interval.
Scanning the parameter `p` gives a good initial candidate whose
convolution peak is already below the previous best bound.

From this start we run a few hundred iterations of an *Adam–style*
multiplicative (exponentiated‑gradient) update, using the sub‑gradient of
the active convolution constraints.  The update keeps the vector on the
probability simplex and quickly drives the peak of the convolution down.
The whole routine stops after the allotted time budget (≈ 1000 s) and
returns the best vector found.

All helper routines are pure NumPy, no external files or network access
is used.

```python
"""Step‑function optimiser – parametric start + Adam multiplicative updates.

The algorithm
-----------------
1.  Initialise a symmetric family a_i(p) ∝ ((i+½)(n‑½‑i))^p, scan p∈[-2,2]
    (401 values) and keep the best candidate w.r.t. the objective
    2·n·max(conv(a,a)).  This cheap scan already improves the uniform
    baseline considerably.

2.  Refine the best candidate with an Adam‑style exponentiated‑gradient
    (mirror‑descent) on the simplex.  In each iteration the sub‑gradient
    of the active convolution constraints is computed, an adaptive
    multiplicative step is taken and the vector is renormalised.

3.  The routine is time‑aware: it stops when the budget (≈1000 s) is
    exhausted and returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat anything within 1e‑12 of the max as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]            # contribution a_{k-i}
    if active.size:
        grad = 2.0 * grad / active.size    # factor 2 because each term appears twice
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)), a is already normalised."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------

    # ------------------------------------------------------------------

    base = (i + 0.5) * (n - 0.5 - i)

            conv = _conv_self(a, L_fft)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

            # sub‑gradient of the active convolution constraints
            grad = _subgradient_maxconv(a, conv)

            # Adam moments (on the gradient)
            m_vec = beta1 * m_vec + (1.0 - beta1) * grad
            v_vec = beta2 * v_vec + (1.0 - beta2) * (grad ** 2)

            m_hat = m_vec / (1.0 - beta1 ** it)
            v_hat = v_vec / (1.0 - beta2 ** it)

            # step size decreasing like 1/√t
            eta = 0.5 / np.sqrt(it)

            # multiplicative update (exponentiated gradient)
            update = np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
            # keep numbers in a safe range
            update = np.clip(update, 1e-12, 1e12)

            a = a * update
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

    # ------------------------------------------------------------------
    # 3️⃣  Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```
