def _adam_descent(a0, deadline, start_time, rng, step0=1.0, T_start=0.1, T_end=1e-6, beta1=0.9, beta2=0.999, epsilon=1e-8, max_move_frac=0.02):

    a = a0 copy

    m = np.zeros_like(a)

    v = np.zeros_like(a)

    t = 0

    while time.time() < deadline:

        t += 1

        conv = _conv_self(a)

        max_val = conv.max()

        cur_score = 2 * len(a) * max_val

        # track best

        # compute temperature

        elapsed = time.time() - start_time

        frac = min(1.0, elapsed / (deadline-start_time))

        T = T_start * ((T_end/T_start) ** frac)   # exponential cooling

        # compute softmax probabilities

        # to avoid overflow, subtract max:

        conv_max = conv.max()

        p = np.exp((conv - conv_max) / T)

        p /= p.sum()

        # correlation

        corr = _corr(a, p)  # returns length n

        grad = 4.0 * len(a) * corr

        # Adam update

        m = beta1 * m + (1 - beta1) * grad

        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** t)

        v_hat = v / (1 - beta2 ** t)

        # step size: decaying

        step = step0 / np.sqrt(t)   # base step size

        # compute update as multiplicative?

        # We'll do additive: a_new = a - step * (m_hat/(sqrt(v_hat)+epsilon))

        update = step * (m_hat / (np.sqrt(v_hat) + epsilon))

        a = a - update

        # enforce nonnegative

        a = np.maximum(a, 0.0)

        # renormalize to simplex

        s = a.sum()

        if s > 0:

            a = a / s

        else:

            a = np.full_like(a, 1.0 / len(a))

        # occasional random move for escape

        if t % 2000 == 0:

            src = rng.choice(len(a), p=a)

            dst = rng.integers(len(a))

            if src != dst:

                delta = a[src] * max_move_frac * rng.random()

                a[src] -= delta

                a[dst] += delta

                a = np.maximum(a, 0.0)

                a /= a.sum()

    return best_a, best_score

"""
Search for a low‑score step‑function.

The algorithm consists of three stages:

1️⃣  Build a diverse pool of candidate probability vectors.
    Shapes include uniform, power‑ramps, (asymmetric) Beta densities,
    exponentials, and random log‑cosine profiles.

2️⃣  Refine the current best candidate with a smooth‑max Adam optimiser.
    The true objective `max(conv(a,a))` is approximated by the
    temperature‑scaled log‑sum‑exp, giving a differentiable surrogate.
    A bias‑corrected Adam update is performed on the simplex using the
    analytical gradient (cross‑correlation of the current distribution
    with the soft‑max weights).  Temperature is exponentially cooled,
    gradually turning the surrogate into the real max‑objective.

3️⃣  A short simulated‑annealing polish with targeted mass moves
    swaps a small amount of probability between the most “dangerous”
    indices (those contributing most to the current maximal convolution).

The whole search respects the given time budget (≤ 1000 s) and finally
returns the best sequence found, normalised to sum = 1 (all entries ≥ 0).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Objective used during the search: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Shape factories (all return a probability vector of length n)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp, optional central plateau."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        shape = shape[:n]            # safety trim
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float, beta: float = None) -> np.ndarray:
    """Symmetric (beta=alpha) or asymmetric Beta(α,β) density on {0,…,n‑1}."""
    if beta is None:
        beta = alpha
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_exponential_shape(n: int, lam: float) -> np.ndarray:
    """Monotone exponential decay, normalised."""
    x = np.arange(n, dtype=float)
    shape = np.exp(-lam * x)
    shape /= shape.sum()
    return shape


def _make_logcosine_shape(n: int, rng: np.random.Generator,
                          n_terms: int = 4, scale: float = 1.0) -> np.ndarray:
    """
    Random log‑cosine shape:
    log a_i = Σ_{m=1}^{n_terms} c_m·cos(2π·m·i/n) .
    The coefficients c_m are drawn from N(0,scale²).
    """
    coeffs = rng.normal(scale=scale, size=n_terms)
    i = np.arange(n, dtype=float)
    log_a = np.zeros(n, dtype=float)
    for m, c in enumerate(coeffs, start=1):
        log_a += c * np.cos(2.0 * np.pi * m * i / n)
    a = np.exp(log_a - np.max(log_a))      # prevent overflow
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Helper to keep the K best candidates
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Cross‑correlation (linear) of a (length n) with p (length 2n‑1)
# Returns an array of length n :  corr[j] = Σ_l p[l]·a[l‑j]
# ----------------------------------------------------------------------
def _cross_corr(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = p.shape[0]                     # = 2·n‑1
    L = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=L)
    P = np.fft.rfft(p, n=L)
    # correlation = ifft( conj(A) * P )
    corr_full = np.fft.irfft(np.conj(A) * P, n=L)
    start = n - 1
    return corr_full[start:start + n]


# ----------------------------------------------------------------------
# Adam optimiser on a smooth surrogate of the max‑convolution.
# The surrogate is  T·log Σ exp(conv/T) .
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  rng: np.random.Generator,
                  step0: float = 1.0,
                  T_start: float = 0.1,
                  T_end: float = 1e-7,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # first‑moment estimate
    v = np.zeros_like(a)          # second‑moment estimate

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.1f}s] Adam iter={it:,} best={cur_score:.6f}")

        # --------------------------------------------------------------
        # temperature schedule (exponential cooling)
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)
        # soft‑max probabilities (stable version)
        conv_shift = conv - conv.max()          # avoid overflow
        p = np.exp(conv_shift / T)
        p_sum = p.sum()
        if p_sum == 0 or np.isnan(p_sum) or np.isinf(p_sum):
            p = np.full_like(p, 1.0 / p.size)
        else:
            p /= p_sum

        # --------------------------------------------------------------
        # gradient = 4·n·corr(a, p)   (see derivation in analysis)
        # --------------------------------------------------------------
        corr = _cross_corr(a, p)               # length n
        grad = 4.0 * n * corr

        # Adam bias‑corrected moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size decays like 1/√it (helps stability for large it)
        step = step0 / math.sqrt(it)

        # additive update (projected back onto the simplex)
        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # --------------------------------------------------------------
        # occasional random mass transfer (helps escape plateaus)
        # --------------------------------------------------------------
        if it % 2000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst and a[src] > 0:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish (same as before)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain list of non‑negative floats (automatically normalised).
    """
    rng = np.random.default_rng(seed + 12345)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4          # small safety margin

    # --------------------------------------------------------------
    # Choose a problem size (larger sizes give a tighter lower bound)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 700:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build an initial pool of candidates
    # --------------------------------------------------------------
    TOP_K = 8
    pool = []                     # (score, vector)

    # uniform baseline
    _push_top(pool, _objective(_make_uniform(N)), _make_uniform(N), TOP_K)

    # power‑ramps
    for p in np.linspace(0.2, 2.0, 7):
        for plat in np.linspace(0.0, 0.5, 5):
            a = _make_power_ramp(N, p, plat)
            _push_top(pool, _objective(a), a, TOP_K)

    # symmetric Beta shapes (including U‑shaped)
    for a_par in np.arange(0.1, 1.1, 0.1):
        a = _make_beta_shape(N, a_par)
        _push_top(pool, _objective(a), a, TOP_K)

    # asymmetric Beta shapes
    for a_par in np.arange(0.2, 0.9, 0.2):
        for b_par in np.arange(0.2, 0.9, 0.2):
            a = _make_beta_shape(N, a_par, b_par)
            _push_top(pool, _objective(a), a, TOP_K)

    # exponential decays
    for lam in [1e-4, 5e-4, 2e-3, 5e-3]:
        a = _make_exponential_shape(N, lam)
        _push_top(pool, _objective(a), a, TOP_K)

    # random log‑cosine profiles
    for _ in range(12):
        a = _make_logcosine_shape(N, rng, n_terms=4, scale=0.8)
        _push_top(pool, _objective(a), a, TOP_K)

    # Dirichlet draws with varied concentration
    for conc in [0.01, 0.05, 0.12, 0.3, 0.7]:
        for _ in range(6):
            if time.time() >= deadline:
                break
            a_raw = rng.gamma(conc, 1.0, size=N)
            a = a_raw / a_raw.sum()
            _push_top(pool, _objective(a), a, TOP_K)

    # reuse previous best if present (inter‑run memory)
    try:
        prev = globals().get('height_sequence_1')
        if prev is not None:
            prev = np.asarray(prev, dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                if prev.size != N:
                    x_old = np.linspace(0, 1, prev.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    prev = np.interp(x_new, x_old, prev)
                prev = np.maximum(prev, 0.0)
                if prev.sum() > 0:
                    prev = prev / prev.sum()
                    _push_top(pool, _objective(prev), prev, TOP_K)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Keep the current best candidate
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.1f}s] Initial best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Adam descent on the smooth max‑convolution surrogate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        sub_deadline = time.time() + remaining * 0.78
        refined_seq, refined_score = _adam_descent(
            best_seq,
            sub_deadline,
            start_time,
            rng,
            step0=1.0,
            T_start=0.12,
            T_end=5e-8,
            max_move_frac=0.025,
        )
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.1f}s] After Adam best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Simulated‑annealing polish (short, targeted)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        best_seq, best_score = _simulated_annealing(
            best_seq, deadline, start_time, rng, max_move_frac=0.03
        )

    # --------------------------------------------------------------
    # 4) Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.1f}s] Finished, final score ≈ {final_score:.6f}")

    # expose the sequence for the next run (optional)
    globals()['height_sequence_1'] = best_seq.copy()

    return [float(v) for v in best_seq]
