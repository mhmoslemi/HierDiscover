"""
Propose a non‑negative step function (list of heights) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The search is a 3‑stage pipeline tuned for the 1000 s budget:

1️⃣  Fast parametric sweep (geometric, power, beta, stretched‑exp, logistic,
    sine‑power) – gives a decent starting shape.

2️⃣  Broad Dirichlet sampling – diversifies the population and may find a
    better basin of attraction.

3️⃣  Gradient‑guided hill climbing / simulated annealing on the exact
    max‑convolution.  At each step we compute the sub‑gradient of the
    L∞‑norm, move a tiny amount of mass from the index with the largest
    gradient to the index with the smallest gradient, accept improvements
    and occasionally accept worsening moves with a slowly cooling temperature.
    The loop runs until the time budget is exhausted, guaranteeing that the
    best sequence found is returned.

All heavy work uses FFT‑based convolution (O(n log n)) and the simplex
projection where needed.  The algorithm respects the 1000 s limit and
prints periodic progress updates.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used by the optimiser: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

# 1️⃣  Parametric families (compact version – enough for a good start)

def _sweep_families(n: int, best_seq: np.ndarray, best_score: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Explore a handful of analytic families and keep the best."""
    idx = np.arange(n, dtype=np.float64)

    # Uniform baseline (already computed outside, kept for completeness)
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ---- geometric ----------------------------------------------------
    for eps in np.logspace(-5, -1, 120):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        # blend with uniform – helps avoid too extreme spikes
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type ------------------------------------------------------
    alphas = np.linspace(0.2, 4.0, 31)
    betas = np.linspace(0.2, 4.0, 31)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # ---- stretched‑exponential -----------------------------------------
    for scale in np.linspace(0.3, 1.0, 7):
        if time.time() >= deadline:
            break
        for p in np.linspace(0.4, 4.5, 31):
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched scale={scale:.2f} p={p:.2f} score={sc:.6f}")

    # ---- logistic -------------------------------------------------------
    for mu_f in np.linspace(0.2, 0.8, 9):
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in np.linspace(0.02, 0.15, 7):
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-6:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic μf={mu_f:.2f} sf={s_f:.2f} score={sc:.6f}")

    # ---- sine‑power -----------------------------------------------------
    for p in np.linspace(0.1, 5.0, 31):
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine‑power p={p:.2f} score={sc:.6f}")

    return best_seq, best_score

def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # number of trials is time‑driven – keep trying until the deadline
    while time.time() < deadline:
        # concentration from 0.01 to 20 (log‑uniform)
        log_c = rng.uniform(np.log(0.01), np.log(20.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Gradient‑guided hill climbing / simulated annealing
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """
    Local search that moves a bit of mass from the index with the largest
    sub‑gradient to the index with the smallest sub‑gradient.
    Accepts improvements always and accepts worsening moves with a
    temperature that cools linearly to zero.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    # temperature schedule
    T0 = 0.025

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # candidates for moving mass from
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # step size – start relatively large and shrink over time
        eps = min(a[src] * 0.02, 0.001)          # at most 0.1 % of total mass
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # maintain feasibility
        if a_new[src] < -1e-15:
            continue

        new_score = _objective_score(a_new)

        if new_score < cur_score - 1e-13:
            # strict improvement
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} new best={new_score:.6f}")
        else:
            # annealed acceptance of worsening moves
            elapsed = time.time() - start_time
            T = max(T0 * (1.0 - elapsed / (deadline - start_time)), 1e-8)
            if np.random.rand() < np.exp(-(new_score - cur_score) / T):
                a = a_new
                cur_score = new_score

        # occasional tiny random shake to escape flat regions
        if np.random.rand() < 0.0003:
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.005
            a[idx] = max(a[idx] + delta, 0.0)
            a /= a.sum()
            cur_score = _objective_score(a)

        # occasional re‑normalisation to combat numeric drift
        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            cur_score = _objective_score(a)

    return best_seq, best_score

    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        starter = np.asarray(height_sequence_1, dtype=float)

            s = starter.sum()
            if s > 0:
                starter /= s
                sc = _objective_score(starter)
                if sc < best_score:

    except Exception:

    # --------------------------------------------------------------

    if total > 0:

    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
