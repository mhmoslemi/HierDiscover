"""
minimize_step_function.py

Search for a non‑negative step function (list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of the following stages (all bounded by the global
wall‑clock budget of 1000 s):

1.  A quick parametric sweep over several families (geometric, power,
    symmetric / asymmetric beta and a two‑exponential “U” shape) to obtain a
    decent baseline.
2.  A large number of random Dirichlet draws; each draw is used as the start
    point for a *sub‑gradient descent* that directly minimises the
    L∞‑norm of the self‑convolution.  The sub‑gradient is computed analytically
    from the active (maximal) shift(s) and the iterate is projected onto the
    probability simplex after every step.
3.  A final hill‑climb that moves tiny packets of mass from the most
    “over‑represented’’ index (largest sub‑gradient component) to the most
    “under‑represented’’ one.  Random jitter is added when progress stalls.

All stages keep track of the best sequence seen so far and print intermediate
scores for monitoring.  The returned list is normalised to sum to 1, which
makes the objective invariant to scaling.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assume sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (fast sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 250)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 151):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_symmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.05, 4.0, 81)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SymBeta a={a:.2f} b={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.1, 5.0, 21)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta a={a:.2f} b={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_exponential(n, best_seq, best_score, deadline):
    """
    U‑shaped candidates: mixture of a decreasing geometric left‑hand
    tail, a mirrored decreasing right‑hand tail and uniform mass.
    """
    idx = np.arange(n, dtype=np.float64)
    r_grid = np.linspace(0.90, 0.995, 30)          # geometric ratio
    for r in r_grid:
        if time.time() >= deadline:
            break
        left = np.power(r, idx)                    # mass concentrated at start
        right = np.power(r, n - 1 - idx)           # mirrored version
        left /= left.sum()
        right /= right.sum()
        uniform = np.full(n, 1.0 / n)

        for w_left in np.linspace(0.0, 0.6, 7):
            if time.time() >= deadline:
                break
            for w_right in np.linspace(0.0, 0.6, 7):
                w_unif = 1.0 - w_left - w_right
                if w_unif < 0.0:
                    continue
                seq = w_left * left + w_right * right + w_unif * uniform
                seq /= seq.sum()
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] TwoExp r={r:.3f} wL={w_left:.2f} wR={w_right:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet(n, rng):
    """Return a random probability vector sampled from a Dirichlet with random concentration."""
    # draw concentration between roughly 0.01 and ~5
    logc = rng.uniform(-2.0, 0.7)
    c = 10.0 ** logc
    a = rng.gamma(shape=c, scale=1.0, size=n)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 3️⃣  Sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent_maxconv(a0: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 max_iter: int = 10**9) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient minimisation of 2·n·max(conv(a,a)) with the constraint
    a≥0, sum(a)=1.  The step size follows the classic ɑ/√k schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    # hyper‑parameters
    alpha0 = 0.4          # initial step‑size multiplier
    min_step = 1e-8

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it} score={cur_score:.6f}")

        # compute sub‑gradient of max(conv)
        g = _subgradient_maxconv(a, conv)

        # step size = α / √k  (scaled by the gradient norm)
        step = alpha0 / np.sqrt(it)
        gnorm = np.linalg.norm(g) + 1e-16
        step_vec = (step / gnorm) * g

        a = a - step_vec
        # enforce non‑negativity and simplex constraint
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional tiny random jitter to avoid stagnation
        if it % 2000 == 0:
            rng = np.random.default_rng()
            i1, i2 = rng.choice(n, size=2, replace=False)
            jitter = a[i1] * (rng.random() * 1e-5 + 1e-8)
            a[i1] -= jitter
            a[i2] += jitter
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # move a tiny fraction (up to ~0.3 %) of the source mass
        eps = a[src] * (rng.random() * 0.003 + 1e-6)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # random jitter to escape a possible plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.003 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function ``evaluate_sequence``.  The routine respects the given time
    budget and returns a plain Python list of floats (normalised to sum=1).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # safety margin

    # --------------------------------------------------------------
    # Problem size – medium resolution keeps the search fast.
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1) if provided by the host.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (geometric, power, beta families, 2‑exp)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_exponential(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers + sub‑gradient descent
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    rng = np.random.default_rng()
    num_restarts = 15
    for restart in range(num_restarts):
        if time.time() >= phase_deadline:
            break
        a0 = _random_dirichlet(n, rng)
        # quick evaluation – keep the best Dirichlet sample as a candidate
        sc0 = _objective_score(a0)
        if sc0 < best_score:
            best_score, best_seq = sc0, a0.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet restart {restart} score={sc0:.6f}")

        # Run a focused sub‑gradient descent from this start point.
        sub_deadline = min(phase_deadline, time.time() + 12.0)   # ~12 s per restart
        best_seq, best_score = _subgradient_descent_maxconv(a0,
                                                            sub_deadline,
                                                            best_seq,
                                                            best_score)

    # --------------------------------------------------------------
    # 3️⃣  Final deterministic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
