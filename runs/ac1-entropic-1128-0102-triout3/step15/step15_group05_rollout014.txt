"""
search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser works in four stages (all respecting the 1000 s budget):

1️⃣  **Random parametric sweep** – a 4‑parameter family
    a_i ∝ r**i·(i+1)**p·(n−i)**q mixed with a uniform background.
    The best candidate of this fast stochastic search is kept.

2️⃣  **Active‑set LP improvement** – given the current vector a,
    let M = max(conv(a,a)).  For the shift indices k where conv(a,a)≈M
    we solve a tiny linear programme

        maximise   Σ b_j
        subject to (a * b)_k ≤ M   for all active k
                   b_j ≥ 0

    (a*b) is the linear convolution, hence the constraints are linear in b.
    The LP (normally a single row) is solved with SciPy’s HiGHS solver.
    The optimiser moves a towards the normalised LP solution by a few step
    sizes (t∈{0.2,0.5,0.8,1.0}); any improvement is accepted.
    This aggressively pushes mass into positions where a is small while
    keeping the maximal autocorrelation under control.

3️⃣  **Projected sub‑gradient descent** on the true objective.
    The sub‑gradient is taken over the full *active* set of convolution
    entries that are within a tiny tolerance of the current maximum.
    Momentum (β=0.9), a decaying step size and occasional uphill moves
    (simulated‑annealing style) are employed.

4️⃣  **Final tiny random mass‑transfer hill‑climb** to polish the solution.

The routine returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric‑polynomial family)
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components
    proportional to r**i·(i+1)**p·(n-i)**q   (i = 0…n‑1).
    Computed in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·uniform + (1‑lam)·shape (both already sum to 1)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Linear‑programming improvement (active‑set)
# ----------------------------------------------------------------------


def _build_constraint_matrix(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """Rows correspond to active shifts k; row j holds coefficient a[i] with i+k=j."""
    n = a.shape[0]
    rows = ks.size
    A = np.zeros((rows, n), dtype=np.float64)
    for r, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        A[r, j] = a[i]
    return A


def _lp_max_sum(a: np.ndarray, ks: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   maximise Σ b_j   s.t.   (a*b)_k ≤ M for all k∈ks, b≥0."""
    if ks.size == 0:
        return None
    A = _build_constraint_matrix(a, ks)
    b_ub = np.full(ks.size, M, dtype=np.float64)
    c = -np.ones(a.shape[0], dtype=np.float64)          # minimise –Σb  ⇔ maximise Σb
    bounds = [(0, None)] * a.shape[0]

    # HiGHS is very fast for tiny problems
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x


def _lp_improve(a0: np.ndarray,
                deadline: float,
                start_time: float,
                best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the tiny LP described above and move towards the
    obtained direction.  The active set is taken as the (near) maximal
    convolution shifts.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    active_tol = 1e-12
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        M = conv.max()
        act = np.where(conv >= M - active_tol)[0]

        # protect against zero coefficients – enforce a tiny floor
        a_eps = np.maximum(a, 1e-12)

        b_vec = _lp_max_sum(a_eps, act, M)
        if b_vec is None:
            # fallback – if LP fails, break out
            break

        sum_b = b_vec.sum()
        if sum_b <= 0.0:
            break
        g = b_vec / sum_b                     # normalise to sum 1

        improved = False
        for t in (0.2, 0.5, 0.8, 1.0):
            cand = (1.0 - t) * a + t * g
            cand = _project_simplex(cand)
            sc = _eval_fast(cand, L_fft)
            if sc < best_score - 1e-9:
                best_score = sc
                best_a = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] LP it={it:,} t={t:.2f} score={sc:.6f}")
                break
        if not improved:
            # small jitter to escape possible plateaus
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (momentum, active set & simulated annealing)
# ----------------------------------------------------------------------


def _compute_subgrad_multi(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a aggregating near‑maximal entries."""
    max_val = conv.max()
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]                     # a_j contribution
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]     # symmetric term
    return g


def _subgradient_descent_multi(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               best_score_initial: float,
                               active_tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent directly on the true objective."""
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    beta = 0.9
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=active_tol)
        v = beta * v + (1.0 - beta) * g
        step = step0 / (it ** 0.6)
        cand = a - step * v
        cand = _project_simplex(cand)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand
            v = np.zeros_like(a)          # reset momentum after improvement
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={sc:.6f}")
        else:
            # occasional uphill move (simulated annealing flavour)
            prob = np.exp(-(sc - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            v = np.zeros_like(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny hill‑climb (random mass transfers)
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """Randomly move a tiny fraction of probability mass between two indices."""
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        sc = _eval_fast(a_prop, L_fft)
        if sc < best_sc:
            best_sc = sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={sc:.6f}")
        else:
            if rng.random() < 0.001:
                a = a_prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Random parametric search (geometric‑power‑tail family)
# ----------------------------------------------------------------------


def _random_param_search(n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float, tuple[float, float, float, float]]:
    """Uniform‑in‑log random sampling of (r,p,q,λ) and keep the best vector."""
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        r = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
        p = rng.uniform(-3.0, 3.0)
        q = rng.uniform(-3.0, 3.0)
        lam = rng.uniform(0.0, 1.0)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam)
            print(f"[{time.time() - start_time:.2f}s] random r={r:.4f} p={p:.2f} q={q:.2f} λ={lam:.2f} score={sc:.6f}")

    return best_seq, best_score, best_params


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation target.
    Returns the best sequence as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2       # safety margin

    # --------------------------------------------------------------
    # Problem size – 8192 gives a finer discretisation without too much
    # overhead (still fits in memory and runs fast with FFT).
    # --------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0️⃣  Uniform baseline (score ≈ 2.0)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter exists, use it (it is for n=4096, so up‑sample)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # up‑sample by linear interpolation to the new length
            x_old = np.linspace(0, 1, starter.size)
            x_new = np.linspace(0, 1, n)
            starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_seq = starter.copy()
                best_score = cur
                print(f"[{time.time() - start_time:.2f}s] global starter (up‑sampled) score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Random parametric sweep (geometric × power × tail)
    # --------------------------------------------------------------
    sweep_deadline = start_time + 150.0
    rp_seq, rp_score, rp_params = _random_param_search(n, sweep_deadline, start_time)
    if rp_seq is not None and rp_score < best_score:
        best_seq = rp_seq.copy()
        best_score = rp_score
        best_params = rp_params
        print(f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}")
    else:
        # fall‑back parameters
        best_params = (0.97, 0.0, 0.0, 0.3)

    # --------------------------------------------------------------
    # 2️⃣  Active‑set linear‑programming improvement
    # --------------------------------------------------------------
    lp_deadline = time.time() + min(400.0, deadline - time.time())
    if time.time() < lp_deadline:
        a_lp, sc_lp = _lp_improve(best_seq, lp_deadline, start_time, best_score)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"[{time.time() - start_time:.2f}s] after LP improve best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Sub‑gradient descent (full active set)
    # --------------------------------------------------------------
    sub_deadline = time.time() + min(300.0, deadline - time.time())
    if time.time() < sub_deadline:
        a_sub, sc_sub = _subgradient_descent_multi(best_seq,
                                                   sub_deadline,
                                                   start_time,
                                                   best_score)
        if sc_sub < best_score:
            best_seq = a_sub.copy()
            best_score = sc_sub
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient final score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final tiny hill‑climb (random mass transfers)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_final, sc_final = _final_local_search(best_seq,
                                                deadline,
                                                start_time,
                                                best_score)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Log official evaluation (uses the provided security‑checked function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
