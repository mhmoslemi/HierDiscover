def pnorm_grad(a, p):
    conv = _conv_self(a)
    conv_pow = conv**(p-1)
    sum_pow = np.sum(conv**p)
    if sum_pow == 0: return gradient zero. else factor = sum_pow**(1/p - 1)
    w = conv_pow
    a_rev = a[::-1]
    L = _next_pow2(3*n - 2)
    # compute convolution via FFT
    # we need full conv length (2n-1+n-1) = 3n-2
    FA = np.fft.rfft(a_rev, n=L)
    FW = np.fft.rfft(w, n=L)
    conv_wa = np.fft.irfft(FA * FW, n=L)[: (3*n-2)]
    grad = 2 * factor * conv_wa[n-1: n-1 + n]  # indices n-1 .. n-1+n-1
    return grad

assistantfinal**Solution Overview**

The code below searches for a non‑negative step‑function (a probability vector) that minimises  

\[
\operatorname{score}(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2}
\]

where `a*a` is the linear convolution of `a` with itself.  
Because the score is invariant under scaling of `a`, we always keep the vector normalised
(`sum(a)=1`).  

The search consists of three stages:

1. **Exploration** – a few analytic families (beta‑shapes, arcsine, uniform) and a
   large number of random Dirichlet draws are evaluated; the best vector so far is kept.
2. **Smooth‑max gradient descent** – we minimise a smooth approximation of the max
   (`log‑sum‑exp`) using a projected gradient method with back‑tracking line search.
   The temperature is raised gradually so that the surrogate approaches the true
   maximum.  This stage is the main workhorse and runs until the time budget is
   exhausted.
3. **Local hill‑climb** – a cheap simulated‑annealing step moves a tiny amount of
   mass from a “high‑gradient” index to a “low‑gradient” index, accepting
   improvements or probabilistic worsening moves.  It often removes the last
   remaining peaks.

All operations are performed with FFT‑based convolutions ( *O*(n log n) ) and the
vector is always projected onto the probability simplex, guaranteeing feasibility
throughout the optimisation.

The algorithm is completely self‑contained, uses only `numpy` and runs in well
under the 1000 s budget for `n = 4096`.

```python
"""
Search for a non‑negative step function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The routine works in three phases:
  1.  cheap analytic families + random Dirichlet initialisations,
  2.  projected gradient descent on a smooth‑max surrogate (log‑sum‑exp)
      with an adaptive temperature schedule,
  3.  a short simulated‑annealing hill‑climb to polish the result.

All vectors are always projected onto the probability simplex, i.e.
non‑negative and summing to one, which keeps the evaluation invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(seq: np.ndarray) -> float:
    """Score used for the optimiser: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (exact) max‑convolution at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Exploration – analytic families and Dirichlet draws
# ----------------------------------------------------------------------
def _exploration_stage(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Return the best sequence found by cheap families + Dirichlet sampling."""
    best_seq = np.full(n, 1.0 / n)      # uniform start
    best_score = _objective_score(best_seq)

    # ---- beta‑type families (including very small alphas) ----
    alphas = np.logspace(np.log10(0.01), np.log10(0.5), 15)
    betas = np.logspace(np.log10(0.01), np.log10(0.5), 15)
    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.2f}s] Beta α={a:.4f} β={b:.4f} → {sc:.6f}")

    # ---- arcsine shape (β(½,½) distribution) ----
    eps = 1e-12
    arcs = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5) + eps)
    arcs /= arcs.sum()
    sc = _objective_score(arcs)
    if sc < best_score:
        best_score, best_seq = sc, arcs.copy()
        print(f"[{time.time():.2f}s] Arcsine shape → {sc:.6f}")

    # ---- random Dirichlet draws (quick) ----
    rng = np.random.default_rng()
    while time.time() < deadline - 5.0:   # keep a few seconds for the next stage
        log_c = rng.uniform(np.log(0.01), np.log(20.0))
        c = np.exp(log_c)
        a_vec = rng.gamma(shape=c, scale=1.0, size=n)
        s = a_vec.sum()
        if s == 0:
            continue
        a_vec /= s
        sc = _objective_score(a_vec)
        if sc < best_score:
            best_score, best_seq = sc, a_vec.copy()
            print(f"[{time.time():.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smooth‑max gradient descent (projected)
# ----------------------------------------------------------------------
def _smoothmax_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate
        f(a)=logsumexp(tau·conv(a,a))/tau
    with respect to a.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_c = conv.max()
    # soft‑max weights
    w_raw = np.exp(tau * (conv - max_c))
    w = w_raw / w_raw.sum()

    # convolution of w with reversed a gives the gradient (see derivation)
    a_rev = a[::-1]
    L = _next_pow2(3 * n - 2)
    grad_part = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)[:n]
    grad = 2.0 * grad_part
    return grad


def _smoothmax_descent(a0: np.ndarray,
                       start_time: float,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max surrogate.
    Uses back‑tracking line search and an exponentially increasing temperature τ.
    """
    a = a0.copy()
    n = a.shape[0]

    tau = 5.0           # start temperature (soft)
    tau_max = 400.0
    tau_inc = 1.02

    step = 0.1          # initial step length
    min_step = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        # current exact score (the one that will be reported)
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Descent it={it:,}  score={cur_score:.6f}")

        # gradient of the smooth‑max surrogate
        grad = _smoothmax_gradient(a, tau)

        # ------- back‑tracking line search ----------
        step_cur = step
        improved = False
        while step_cur >= min_step:
            a_candidate = a - step_cur * grad
            a_candidate = _project_simplex(a_candidate)
            cand_score = _objective_score(a_candidate)
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                # enlarge step a little after a success
                step = step_cur * 1.2
                break
            step_cur *= 0.5

        if not improved:
            # no improvement – shrink step size aggressively
            step = max(step * 0.5, min_step)

        # increase temperature (approach the true max)
        if tau < tau_max:
            tau = min(tau * tau_inc, tau_max)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Final hill‑climb / simulated‑annealing refinement
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Tiny simulated‑annealing move set that reshuffles a tiny amount of mass."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = _objective_score(a)
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_time = max(0.0, deadline - time.time())
    if total_time <= 0.0:
        return best_seq, best_score

    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_time)

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # tiny fraction of the source mass
        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline   score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:

                sc = _objective_score(starter)
                if sc < best_score:

    except Exception:

    # --------------------------------------------------------------
    # 1️⃣  Exploration (analytic families + Dirichlet)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
