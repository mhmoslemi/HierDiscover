    m = np.zeros_like(x)

        elapsed = time.time() - start_time

        beta = max(beta_smoothing_min, beta_smoothing_initial * (1.0 - progress))

        start_idx = len(a) - 1

        # Adam update

def _adam_optimize(a0, n, start_time, deadline, best_score):
    best_seq = a0.copy()
    # convert to logits
    eps = 1e-12
    x = np.log(a0 + eps)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9
    beta2=0.999
    eps_adam=1e-8
    lr0 = 0.1
    # precompute L_self, L_c
    L_self = _next_pow2(2*n-1)
    L_c = _next_pow2(3*n-2)
    # preallocate for speed? not needed.
    it = 0
    while time.time() < deadline:
        it += 1
        # compute a via softmax
        # subtract max for stability
        x_max = np.max(x)
        a = np.exp(x - x_max)
        a_sum = a.sum()
        a = a / a_sum
        # compute self conv
        conv_self = _conv_self(a, L_self)
        cur_max = conv_self.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam it={it} score={cur_score:.6f}")
        # smoothing parameter
        elapsed = time.time() - start_time
        total_time = deadline - start_time
        progress = elapsed / total_time
        beta = max(1e-6, 0.001 * (1.0 - progress))  # linear decrease
        # compute weights p for smoothed max
        # use stable softmax
        max_conv = cur_max  # Already max conv
        # compute exponentials
        exp_vals = np.exp((conv_self - max_conv) / beta)
        p = exp_vals / exp_vals.sum()
        # compute gradient wrt a
        a_rev = a[::-1]
        a_rev_fft = np.fft.rfft(a_rev, n=L_c)
        p_fft = np.fft.rfft(p, n=L_c)
        conv_ap = np.fft.irfft(a_rev_fft * p_fft, n=L_c)
        start_idx = n - 1
        grad_a = 2.0 * conv_ap[start_idx:start_idx + n]
        # gradient wrt logits
        inner = np.dot(a, grad_a)
        grad_x = a * (grad_a - inner)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        bias_correction1 = 1 - beta1 ** it
        bias_correction2 = 1 - beta2 ** it
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2
        lr = lr0 / np.sqrt(it)  # decaying lr
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
    return best_seq, best_score

"""
step‑function optimiser – smoothed‑max Adam on the simplex.

The evaluator expects a list of non‑negative numbers and returns

    2·n·max(conv(a,a)) / (sum(a)**2)

If we keep sum(a)=1 the score is simply `2·n·max(conv)`.  The
goal is to minimise the peak of the linear self‑convolution.
The implementation proceeds as follows

1.  initialise from a global starter (if any) or from the uniform
    distribution.
2.  perform a quick sweep of a few simple parametric families
    (exponential, triangular) and a broad Dirichlet search – the
    best result is kept.
3.  run a long Adam optimisation on the *log‑softmax* parameters
    of the sequence.  The loss that is differentiated is a smooth
    approximation of the max‑convolution obtained by a soft‑max
    with a temperature β that is gradually reduced to zero.  The
    gradient of `β·log ∑ exp(conv/β)` w.r.t. the probabilities has a
    closed form that can be computed by a single FFT‑based
    convolution.
4.  finally a short mixing sweep with the uniform distribution
    (convex combination) to possibly smooth the last peak.
5.  normalise the returned sequence to sum 1 (scaling does not
    affect the score) and output it as a plain Python list.

The whole routine is time‑aware and respects a 1000 s budget.
It prints intermediate scores so that progress can be monitored.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)); a is already normalised."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------
def _exponential_family(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n))
    return a / a.sum()


def _triangular_family(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular shape, peak at index `peak` (0‑based)."""
    if peak <= 0:
        a = np.ones(n)
    else:
        left = np.arange(peak) + 1.0
        right = np.arange(n - peak)[::-1] + 1.0
        a = np.concatenate([left, right])
        a = a[:n]                     # truncate if overshoot
    return a / a.sum()


# ----------------------------------------------------------------------
# Random Dirichlet search (wide concentration range)
# ----------------------------------------------------------------------
def _random_dirichlet_search(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration drawn log‑uniformly in [0.02, 20]
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        s = sample.sum()
        if s <= 0.0:
            continue
        a = sample / s
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max of the convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits of a probability vector.
    The loss is a temperature‑smoothed max of the self‑convolution.
    """
    eps = 1e-12
    # initialise logits
    x = np.log(a0 + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.15

    # FFT sizes (pre‑computed)
    L_self = _next_pow2(2 * n - 1)           # for a*a
    L_grad = _next_pow2(3 * n - 2)           # for a_rev * p

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) probabilities from logits (softmax, sum = 1)
        # --------------------------------------------------------------
        x_max = np.max(x)
        a = np.exp(x - x_max)
        a_sum = a.sum()
        a = a / a_sum

        # --------------------------------------------------------------
        # 2) current loss (true objective) – keep best ever
        # --------------------------------------------------------------
        conv_full = _conv_self(a, L_self)
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # --------------------------------------------------------------
        # 3) temperature β (linear decay to a tiny floor)
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total
        beta = max(1e-7, 0.001 * (1.0 - progress))   # from 1e‑3 down to 1e‑7

        # --------------------------------------------------------------
        # 4) smoothed max via a soft‑max over the convolution vector
        # --------------------------------------------------------------
        # stable soft‑max
        shift = cur_max                       # max of conv_full
        exp_vals = np.exp((conv_full - shift) / beta)
        p = exp_vals / exp_vals.sum()         # weights, length 2n‑1

        # --------------------------------------------------------------
        # 5) gradient of the smoothed max w.r.t. a
        #    g_i = 2 * Σ_j a_j * p_{i+j}
        #    computed as convolution of a_rev and p
        # --------------------------------------------------------------
        a_rev = a[::-1]
        a_rev_fft = np.fft.rfft(a_rev, n=L_grad)
        p_fft = np.fft.rfft(p, n=L_grad)
        conv_ap = np.fft.irfft(a_rev_fft * p_fft, n=L_grad)
        start_idx = n - 1
        grad_a = 2.0 * conv_ap[start_idx:start_idx + n]

        # --------------------------------------------------------------
        # 6) chain rule: gradient w.r.t. logits (softmax Jacobian)
        # --------------------------------------------------------------
        inner = np.dot(a, grad_a)
        grad_x = a * (grad_a - inner)

        # --------------------------------------------------------------
        # 7) Adam update on logits
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)                # simple 1/√t decay
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final fine‑grained mixing with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Scan λ∈[0,1] for seq = λ·a + (1‑λ)·uniform.
    Keep the best mixture.
    """
    uniform = np.full(n, 1.0 / n)
    lam_grid = np.linspace(0.0, 1.0, 41)   # coarse sweep
    for lam in lam_grid:
        if time.time() >= deadline:

        seq = lam * a + (1.0 - lam) * uniform
        seq_sum = seq.sum()
        if seq_sum == 0.0:

        sc = _objective_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")
    return best_seq, best_score
