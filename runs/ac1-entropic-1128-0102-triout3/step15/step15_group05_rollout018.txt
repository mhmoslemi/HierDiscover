"""
Non‑negative step‑function optimiser.

The aim is to minimise
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages:

1️⃣ Random sampling of a rich family of parametric shapes.
   * 4‑parameter geometric‑polynomial family (rⁱ·(i+1)ᵖ·(n‑i)ᑫ)
   * 2‑exponential mixture (α·r₁ⁱ + (1‑α)·r₂ⁱ)
   * optional uniform background.
   The best candidate is kept.

2️⃣ Smooth optimisation of the p‑norm of the convolution
   (p‑norm → smooth approximation of the max).  Gradient is obtained
   analytically and an Adam‑style update is performed, always
   projecting back onto the probability simplex.

3️⃣ Classical projected sub‑gradient descent (as in the reference
   implementation) to polish the last few digits of the true objective.

4️⃣ Tiny random “mass‑transfer” hill‑climb for a final local refinement.

The whole search respects the 1000 s budget, prints timing / progress
information and finally returns the best sequence found as a plain
list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (first 2·len(a)‑1 entries)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """Convolution of two real vectors using FFT (full length)."""
    n = x.shape[0]
    m = y.shape[0]
    if L is None:
        L = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: n + m - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {w ≥ 0, sum w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective (sum ≈ 1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """2·n·max(conv(a,a)) / (sum(a)**2) using FFT (sum assumed ≈ 1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """r**i·(i+1)**p·(n-i)**q  (log‑space, normalised)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)         # avoid overflow
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _make_shape_two_exp(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    """α·r1**i + (1‑α)·r2**i  (normalised)."""
    i = np.arange(n, dtype=np.float64)
    shape = alpha * np.exp(i * np.log(r1)) + (1.0 - alpha) * np.exp(i * np.log(r2))
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """lam·uniform + (1‑lam)·shape (renormalised)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

def _random_param_search(n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float, dict]:
    """Sample many random parameters, keep the best vector."""
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_desc = None

    while time.time() < deadline:
        if rng.random() < 0.55:                     # geometric‑polynomial family
            r = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
            p = rng.uniform(-4.0, 4.0)
            q = rng.uniform(-4.0, 4.0)
            lam = rng.uniform(0.0, 1.0)
            shape = _make_shape_general(n, r, p, q)
        else:                                        # two‑exponential mixture
            r1 = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
            r2 = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
            alpha = rng.uniform(0.0, 1.0)
            lam = rng.uniform(0.0, 1.0)
            shape = _make_shape_two_exp(n, r1, r2, alpha)

        sc = _eval_fast(cand, L_fft)

def _smooth_pnorm_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          best_score_initial: float,
                          p: int = 12,
                          initial_step: float = 0.16) -> tuple[np.ndarray, float]:
    """
    Minimise the p‑norm of the self‑convolution (smooth proxy for max)
    using Adam with simplex projection.
    """
    n = a0.shape[0]
    L_self = _next_pow2(2 * n - 1)
    L_conv = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_self)            # length 2n‑1
        # p‑norm and its scaling factor
        conv_p = np.power(conv, p, out=np.empty_like(conv))
        S = conv_p.sum()
        # gradient of the p‑norm w.r.t. a
        pw = np.power(conv, p - 1, out=np.empty_like(conv))   # conv^{p‑1}
        conv_pw_a = _conv_fft(pw, a, L_conv)                 # w * a
        conv_pw_arev = _conv_fft(pw, a[::-1], L_conv)        # w * rev(a)

        # extract the n needed components
        g = conv_pw_a[:n] + conv_pw_arev[(n - 1):(2 * n - 1)]

        # factor S^{1/p‑1}
        if S > 0.0:
            factor = S ** (1.0 / p - 1.0)
            g *= factor
        else:
            g.fill(0.0)

        # Adam momentum update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = initial_step / (np.sqrt(it) + 1.0)
        a_candidate = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_self)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{time.time() - start_time:.2f}s] p‑norm it={it:,} "
                  f"score={cur_score:.6f}")
        else:
            a = a_candidate

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (as in the reference code)
# ----------------------------------------------------------------------


def _compute_subgrad_multi(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """Sub‑gradient of max(conv) aggregating all (almost) active entries."""
    max_val = conv.max()
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
    return g


def _subgradient_descent_multi(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               best_score_initial: float,
                               active_tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent directly on the true objective."""
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    beta = 0.9
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=active_tol)

        v = beta * v + (1.0 - beta) * g
        step = step0 / (it ** 0.6)
        a_candidate = a - step * v
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            v = np.zeros_like(a)
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"score={cur_score:.6f}")
        else:
            # simulated‑annealing‐style uphill moves
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            v = np.zeros_like(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny random mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """Move a tiny fraction of mass between two coordinates repeatedly."""
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3          # ≤0.2 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            if rng.random() < 0.001:
                a = a_prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine respects the 1000 s time budget and returns the best
    sequence as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a power of two is convenient for FFTs
    # ------------------------------------------------------------------
    n = 4096
    L_fft_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline (score = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft_self)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Use a global starter if provided
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft_self)
            if cur < best_score:
                best_seq = starter.copy()
                best_score = cur
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep (geometric‑polynomial + 2‑exp families)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 150.0
    rp_seq, rp_score, rp_desc = _random_param_search(
        n, sweep_deadline, start_time)
    if rp_seq is not None and rp_score < best_score:
        best_seq = rp_seq.copy()
        best_score = rp_score
        print(f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth p‑norm descent (Adam) – main optimisation stage
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    pnorm_deadline = time.time() + min(400.0, remaining * 0.5)
    if time.time() < pnorm_deadline:
        a_pnorm, sc_pnorm = _smooth_pnorm_descent(
            best_seq, pnorm_deadline, start_time, best_score, p=12, initial_step=0.12)
        if sc_pnorm < best_score:
            best_seq = a_pnorm.copy()
            best_score = sc_pnorm
            print(f"[{time.time() - start_time:.2f}s] p‑norm descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Sub‑gradient descent (refinement)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    sub_deadline = time.time() + min(300.0, remaining * 0.4)
    if time.time() < sub_deadline:
        a_sub, sc_sub = _subgradient_descent_multi(
            best_seq, sub_deadline, start_time, best_score)
        if sc_sub < best_score:
            best_seq = a_sub.copy()
            best_score = sc_sub
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny hill‑climb (random mass transfers)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_final, sc_final = _final_local_search(
            best_seq, deadline, start_time, best_score)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # Log the official evaluation (uses the provided security‑checked function)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
