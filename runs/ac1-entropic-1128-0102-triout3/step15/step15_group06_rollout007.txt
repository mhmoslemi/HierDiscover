"""
Simulated‑annealing local search for the step‑function bound

The algorithm works on a single large problem size (default 16384).  It
starts from a good analytic shape (or from the previously found
`height_sequence_1` if available), then repeatedly moves a small amount
of mass from a source index to a target index.  The source index is
chosen from those that contribute most to the current maximal
convolution entry, so the update directly attacks the active set.
A cheap O(1) update of the self‑convolution is used (the three
affected entries are recomputed analytically), which allows millions of
iterations within the 1000 s budget.  Accepted moves are
greedy; a simple temperature schedule occasionally accepts uphill
steps to escape plateaus.  The best vector encountered is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based for the initial full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negative noise that may appear due to round‑off.
    return np.maximum(conv, 0.0)


def _objective_from_max(max_conv: float, n: int, sum_a: float) -> float:
    """Exact objective 2·n·max(conv) / (sum(a)²)."""
    if sum_a <= 0.0:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def _objective(a: np.ndarray) -> float:
    """Convenient wrapper when sum(a)=1 (the usual case)."""
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return _objective_from_max(conv.max(), n, a.sum())


# ----------------------------------------------------------------------
# Analytic shape generators (normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power ramp, normalised to sum = 1."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    shape /= shape.sum()
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    shape /= shape.sum()
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Helper for initial pool of candidates
# ----------------------------------------------------------------------
def _initial_vector(N: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a small pool of analytic shapes (and a few Dirichlet samples)
    and return the best one (lowest objective).  The vector is normalised
    to sum = 1.
    """
    pool = []

    def _add(vec):
        if vec.ndim != 1 or vec.size != N:
            return
        pool.append((_objective(vec), vec.copy()))

    # uniform
    _add(_make_uniform(N))

    # power ramps
    for p in np.linspace(0.2, 3.0, 7):
        for plat in np.linspace(0.0, 0.4, 3):
            _add(_make_power_ramp(N, p, plat))

    # exponentials
    for lam in np.linspace(0.5, 4.0, 9):
        _add(_make_exponential(N, lam))

    # power laws
    for p in np.linspace(0.5, 3.0, 9):
        _add(_make_power_law(N, p))

    # asymmetric beta families
    alphas = np.linspace(0.2, 3.0, 4)
    betas = np.linspace(0.2, 3.0, 4)
    for a in alphas:
        for b in betas:
            _add(_make_beta_shape_asym(N, a, b))

    # few random Dirichlet draws (decreasing order encourages the shape we want)
    for conc in [0.1, 0.3, 0.5, 1.0, 2.0]:
        for _ in range(12):
            vec = rng.gamma(conc, 1.0, size=N)
            vec = np.sort(vec)[::-1]                 # monotone decreasing
            vec /= vec.sum()
            _add(vec)

    # if a previous solution exists, interpolate it to the current size
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != N:
                # linear interpolation
                x_old = np.linspace(0, 1, prev.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                up = np.interp(x_new, x_old, prev)
            else:
                up = prev.copy()
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                up /= up.sum()
                _add(up)
    except Exception:
        pass

    # pick the best
    pool.sort(key=lambda t: t[0])
    best_vec = pool[0][1].copy()
    return best_vec


# ----------------------------------------------------------------------
# Simulated‑annealing / greedy local search
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low bound 2·n·max(conv)/sum².
    The method starts from a good analytic shape (or from the previous
    solution) and performs millions of cheap mass‑transfer moves while
    maintaining the exact convolution via a three‑entry update.
    The best vector found before the deadline is returned.
    """
    random.seed(seed)
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start = time.time()
    deadline = start + budget_s - 0.5    # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size that leaves enough time for many cheap updates
    # ------------------------------------------------------------------
    remaining = deadline - start
    if remaining > 800:
        N = 32768
    elif remaining > 500:
        N = 16384
    elif remaining > 300:
        N = 8192
    else:
        N = 4096

    # ------------------------------------------------------------------
    # Initialise a good starting point
    # ------------------------------------------------------------------
    a = _initial_vector(N, rng)               # sum = 1
    n = N
    conv = _conv_self_fft(a)                  # full self‑convolution
    cur_max = conv.max()
    cur_obj = _objective_from_max(cur_max, n, a.sum())
    best_a = a.copy()
    best_obj = cur_obj
    best_max = cur_max

    print(f"[{time.time() - start:.2f}s] Init N={N}, objective={cur_obj:.8f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    total_time = deadline - start
    # Temperature controls acceptance of uphill moves; values are tiny because
    # the objective changes are themselves tiny.
    T0 = 5e-6
    T_end = 1e-12

    it = 0
    last_improve_iter = 0

    while time.time() < deadline:
        it += 1
        prog = (time.time() - start) / total_time
        T = T0 * (1.0 - prog) + T_end * prog

        # ---------- choose source (i) and target (j) ----------
        # 70 % of the time we pick i from the active set of the current max.
        if rng.random() < 0.7:
            k_max = int(conv.argmax())
            i_low = max(0, k_max - (n - 1))
            i_high = min(n - 1, k_max)
            i_candidates = np.arange(i_low, i_high + 1, dtype=int)
            # contribution of i to the max entry
            contrib = a[i_candidates] * a[k_max - i_candidates]
            i = int(i_candidates[np.argmax(contrib)])
            # fall back to random if contribution is zero
            if contrib.max() == 0.0:
                i = rng.integers(0, n)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:                     # nothing to move from here
            continue

        # pick a different target index uniformly
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        a_i = a[i]
        a_j = a[j]

        # choose a step size – a fraction of the source mass
        # use a larger fraction early, shrink later
        max_frac = 0.25 * (1.0 - prog) + 0.02
        epsilon = a_i * rng.random() * max_frac
        if epsilon <= 1e-16:
            continue

        # ---------- fast convolution update (three entries) ----------
        idx1 = 2 * i
        idx2 = i + j
        idx3 = 2 * j

        old1 = conv[idx1]
        old2 = conv[idx2]
        old3 = conv[idx3]

        # analytic formulas derived from (a+Δ)*(a+Δ)
        new1 = old1 - 2.0 * epsilon * a_i + epsilon * epsilon
        new3 = old3 + 2.0 * epsilon * a_j + epsilon * epsilon
        new2 = old2 + 2.0 * epsilon * (a_i - a_j - epsilon)

        # guard against negative noise
        if new1 < 0.0:
            new1 = 0.0
        if new2 < 0.0:
            new2 = 0.0
        if new3 < 0.0:
            new3 = 0.0

        conv[idx1] = new1
        conv[idx2] = new2
        conv[idx3] = new3

        new_max = conv.max()
        new_obj = _objective_from_max(new_max, n, 1.0)   # sum stays 1 (up to rounding)

        # ---------- acceptance ----------
        accept = False
        if new_obj <= cur_obj:
            accept = True
        else:
            # uphill acceptance with probability exp(-Δ/T)
            prob = math.exp(-(new_obj - cur_obj) / max(T, 1e-30))
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move
            a[i] = a_i - epsilon
            a[j] = a_j + epsilon
            cur_obj = new_obj
            cur_max = new_max

            # keep best ever seen
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                best_max = new_max
                last_improve_iter = it
                print(f"[{time.time() - start:.2f}s] it={it:,} new best={best_obj:.8f}")

            # periodic renormalisation (cumulative rounding error)
            if it % 5000 == 0:
                s = a.sum()
                if abs(s - 1.0) > 1e-12:
                    a /= s
        else:
            # reject – restore the three entries
            conv[idx1] = old1
            conv[idx2] = old2
            conv[idx3] = old3

        # early exit if we already beat the target
        if best_obj < 1.503:
            print(f"[{time.time() - start:.2f}s] target reached (obj={best_obj:.8f})")
            break

        # occasional random perturbation if stuck for a long time
        if it - last_improve_iter > 200_000:
            # perform a small random reshuffle of a tiny amount of mass
            src = rng.integers(0, n)
            dst = rng.integers(0, n - 1)
            if dst >= src:
                dst += 1
            mass = min(a[src], 1e-4) * rng.random()
            if mass > 0:
                a[src] -= mass
                a[dst] += mass
                # recompute the whole convolution (costly but rare)
                conv = _conv_self_fft(a)
                cur_max = conv.max()
                cur_obj = _objective_from_max(cur_max, n, 1.0)
                last_improve_iter = it

    # ------------------------------------------------------------------
    # Final polishing: symmetric version (optional)
    # ------------------------------------------------------------------
    sym = (best_a + best_a[::-1]) * 0.5
    sym_obj = _objective(sym)
    if sym_obj < best_obj:
        best_a = sym
        best_obj = sym_obj
        print(f"[{time.time() - start:.2f}s] symmetric polishing improved to {best_obj:.8f}")

    # ------------------------------------------------------------------
    # Return a clean python list (the evaluator expects a plain list)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full(N, 1.0 / N)

    final_obj = _objective(best_a)
    print(f"[{time.time() - start:.2f}s] Finished, final objective ≈ {final_obj:.8f}")

    return [float(v) for v in best_a]
