        # ensure nonnegatives and sum remains 1

        cur_score = _objective_score(a)   # we have earlier keep maybe for performance

        new_score = _objective_score(a_prop)

"""
Propose a non‑negative step function that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm is a fast, pure‑Python optimisation pipeline that
1. starts from the best known construction (the global variable
   ``height_sequence_1``) or from a uniform vector,
2. performs a long projected sub‑gradient descent on the exact
   L∞‑norm of the self‑convolution (the objective is 2·n·max(conv)),
3. injects occasional random “shake‑ups’’ to escape plateaus,
4. always respects the time budget (default 1000 s) and returns the
   best sequence found.
All operations stay on the probability simplex (non‑negative, sum = 1)
so the scaling invariance of the original evaluation is handled
automatically.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities (copied from the reference code)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT (real‑valued)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at a (see reference code).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used during optimisation: 2·n·max(conv(seq,seq))  (∑seq = 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises
    `evaluate_sequence`.  The routine returns a plain Python list of

    normalise it, so we keep the sum equal to 1.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------

    # Initialise from the best global construction, otherwise uniform.

    try:

    # If the starter has a different length, reshape / truncate / pad.

    # Ensure we are on the simplex
