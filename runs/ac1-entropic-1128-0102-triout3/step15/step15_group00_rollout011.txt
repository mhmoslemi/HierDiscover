def evaluate_sequence(sequence: list[float]) -> float:
    # ...
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))
```

Sequence length may be any positive integer (but we need to consider runtime). The objective is to minimize `2 * n * max(convolution) / (sum^2)`. We can rescale sum arbitrarily because the objective depends on sum via denominator squared, but it also includes `max_b` which scales quadratically with sum. Actually if we scale all `a_i` by factor `c`, then `conv(a, a)` scales by `c^2`. Then numerator `2 n max(c^2 * conv_orig_max) = 2 n c^2 max_orig`. Denominator `(c sum_orig)^2 = c^2 sum_orig^2`. """
Improved step‑function search.

The algorithm combines several fast parametric families (including a new “U‑shaped”
power family) with a large number of random Dirichlet starts, then performs a
smooth‑max Adam descent on the simplex (soft‑max parametrisation) while forcing
symmetry.  The smooth‑max uses a gradually increasing temperature so that the
gradient initially follows a well‑behaved averaged objective and finally
approaches the true ∞‑norm of the self‑convolution.  A final short hill‑climb
polishes the result.  The routine respects the 1000 s budget and returns the
best (non‑negative, sum‑normalised) sequence it discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using an FFT of size power‑of‑2."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w

def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for eps in np.logspace(-5, -1, 300):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):

def _sweep_symmetric_beta(n, best_seq, best_score, deadline):

# 3️⃣  Adam‑style descent on the smooth‑max (softmax parametrisation)

def _adam_descent(init_a, deadline, best_seq, best_score):
    """
    Adam optimisation on the simplex via softmax variables.
    A gradually increasing temperature β turns the smooth max into the true max.
    The iterate is symmetrised every step (a_i ≡ a_{n‑1‑i}).
    """
    a = init_a.copy()
    n = a.shape[0]
    # initialise log‑variables (softmax pre‑image)
    x = np.log(a + 1e-15)

    # Adam parameters
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.25
    it = 0

    beta_smooth = 1.0       # start temperature
    beta_target = 200.0
    beta_inc = (beta_target - beta_smooth) / 20000.0  # ~20000 iterations to reach target

    while time.time() < deadline:
        # ----- obtain a from x (softmax) -----
        mx = np.max(x)
        ex = np.exp(x - mx)
        a = ex / ex.sum()          # already on the simplex
        # enforce symmetry on the *probabilities*
        a = (a + a[::-1]) * 0.5

        # ----- evaluate true objective -----
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} new best {cur_score:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(beta_smooth * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t a (smooth max) -----
        rev_w = w[::-1]
        conv_rev = _conv_fft(a, rev_w)               # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1:2 * n - 1]     # slice to length n
        # symmetrise gradient
        grad_a = (grad_a + grad_a[::-1]) * 0.5

        # ----- gradient w.r.t x (softmax chain rule) -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update on x -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / (1.0 + it * 1e-5)                 # gentle decay
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- keep symmetry on the log‑variables -----
        x = (x + x[::-1]) * 0.5

        # ----- temperature schedule -----
        if beta_smooth < beta_target:
            beta_smooth = min(beta_target, beta_smooth + beta_inc)

        # ----- occasional random shake to avoid tiny plateaus -----
        if it % 5000 == 0:
            noise = np.random.randn(n) * 0.01
            x += noise

        # ----- early stop if time exhausted -----
        if time.time() >= deadline:
            break

    return best_seq, best_score

def _hill_climb_enhanced(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)  # note: this function is defined later

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        max_frac = 0.05
        frac = rng.random() * max_frac + 1e-5
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-14:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill++ score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps2 = a[i1] * (rng.random() * max_frac + 1e-5)
                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient helper (used by hill‑climb)
# ----------------------------------------------------------------------


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine performs a broad parametric sweep (including a new U‑shape family),
    many random Dirichlet starts, a smooth‑max Adam descent on the simplex,
    and finishes with a focused hill‑climb.  The best vector (normalised to sum = 1)
    is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # a tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep (including new U‑shape family)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_u_shape(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam descent on the simplex
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam descent")
        best_seq, best_score = _adam_descent(best_seq, deadline - 150.0,
                                             best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After Adam descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final directed hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb_enhanced(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
