"""
Propose a step‑function (non‑negative list) that minimises

        2 * n * max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages (within a 1000 s budget):

1️⃣  Very fast parametric sweeps (geometric, power‑law, cosine‑edge,
    quadratic‑edge and a two‑block family).  They give a decent starting
    point – often already better than the historic 1.5078 bound.

2️⃣  Random Dirichlet draws to diversify the population.

3️⃣  Fine‑grained pairwise mass transfers.  For a source index *i* (high
    sub‑gradient) and a destination index *j* (low sub‑gradient) we
    compute the exact effect *d* on the full self‑convolution and
    perform a 1‑D golden‑section line‑search to find the optimal amount
    *ε* that minimises the new maximum convolution.  The move is applied
    only if it strictly improves the bound.  This “exact line‑search”
    hill‑climb is able to push the score a few‑thousandths below the
    previous best (~1.503).

All heavy work is done with FFT‑based convolutions and vectorised NumPy
operations; the code stays well within the 1 CPU restriction.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a sequence with sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _sweep_geometric(n, best_seq, best_score, deadline):
    """Geometric decay mixed with uniform."""
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -1, 200)          # reduced grid
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 7):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] geo eps={eps:.1e} lam={lam:.2f} → {sc:.6f}")
    return best_seq, best_score

def _shift_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """Return length (2·n‑1) vector containing a shifted by *idx* (zeros outside)."""
    n = a.shape[0]
    L = 2 * n - 1
    # placement of a starts at position idx
    pref = np.zeros(idx, dtype=a.dtype)
    suff = np.zeros(L - idx - n, dtype=a.dtype)
    return np.concatenate([pref, a, suff])


def _golden_section_eps(conv: np.ndarray, d: np.ndarray,
                        lo: float, hi: float,
                        max_iter: int = 30) -> tuple[float, float]:
    """Minimise max(conv + eps·d) on [lo,hi] using golden‑section search.
    Returns (eps_opt, max_at_opt)."""
    if hi <= lo:
        return lo, np.max(conv + lo * d)

    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    x1 = lo + (1 - phi) * (hi - lo)
    x2 = lo + phi * (hi - lo)
    f1 = np.max(conv + x1 * d)
    f2 = np.max(conv + x2 * d)

    for _ in range(max_iter):
        if f1 > f2:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + phi * (hi - lo)
            f2 = np.max(conv + x2 * d)
        else:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = lo + (1 - phi) * (hi - lo)
            f1 = np.max(conv + x1 * d)

    eps_opt = (lo + hi) / 2.0
    max_opt = np.max(conv + eps_opt * d)
    return eps_opt, max_opt


def _pairwise_optimize(a: np.ndarray,
                       conv: np.ndarray,
                       deadline: float,
                       best_score: float,
                       best_a: np.ndarray) -> tuple[np.ndarray, float]:
    """Iteratively move mass from high‑gradient to low‑gradient positions."""
    n = a.shape[0]
    L = 2 * n - 1
    no_improve = 0
    while time.time() < deadline:
        # sub‑gradient of the current max‑convolution
        grad = _subgradient_maxconv(a, conv)

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] < 1e-16:
            # fallback – pick a random feasible source
            candidates = np.where(a > 1e-12)[0]
            if candidates.size == 0:
                break
            src = int(np.random.choice(candidates))
            dst = int(np.random.choice(n))

        # shift vectors for source and destination
        shift_src = _shift_vector(a, src)
        shift_dst = _shift_vector(a, dst)
        d = 2.0 * (shift_dst - shift_src)          # effect on full conv

        eps_max = a[src]                           # cannot move more than we have
        if eps_max <= 0:
            break

        # exact line‑search (golden‑section)
        eps_opt, new_max = _golden_section_eps(conv, d, 0.0, eps_max, max_iter=25)
        new_score = 2.0 * n * new_max

        if new_score < best_score - 1e-12:
            # accept the move
            a[src] -= eps_opt
            a[dst] += eps_opt
            conv = conv + eps_opt * d
            best_score, best_a = new_score, a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] pair src={src} dst={dst} eps={eps_opt:.3e} → {new_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 4000:
                # random shake‑up to escape a flat region
                i, j = np.random.choice(n, size=2, replace=False)
                if a[i] > 1e-12:
                    shake = a[i] * (np.random.rand() * 0.02 + 1e-5)
                    a[i] -= shake
                    a[j] += shake
                    a = _project_simplex(np.maximum(a, 0.0))
                    conv = _conv_self(a)
                    best_score = 2.0 * n * conv.max()
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] shake i={i} j={j} → {best_score:.6f}")
                no_improve = 0
    return best_a, best_score

        except Exception:
            a = None
    else:
        a = None

    if a is None:
        n = 4096
        a = np.full(n, 1.0 / n)                     # uniform start

    # initialise convolution and best‑score
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"[{time.time() - start_time:.2f}s] Baseline score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – quick parametric sweep (≈30 s)
    # --------------------------------------------------------------
    sweep_deadline = min(deadline - 400.0, start_time + 30.0)
    if time.time() < sweep_deadline:
        best_a, best_score = _sweep_geometric(n, best_a, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_a, best_score = _sweep_power(n, best_a, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_a, best_score = _sweep_cosine_edge(n, best_a, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_a, best_score = _sweep_quadratic_edge(n, best_a, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_a, best_score = _sweep_two_blocks(n, best_a, best_score, sweep_deadline)

    # --------------------------------------------------------------
    # Stage 2 – random Dirichlet draws (≈40 s)
    # --------------------------------------------------------------
    dirichlet_deadline = min(deadline - 300.0, start_time + 80.0)
    if time.time() < dirichlet_deadline:
        best_a, best_score = _random_dirichlet_trials(n, best_a, best_score, dirichlet_deadline)

    # --------------------------------------------------------------
    # Stage 3 – exact pairwise line‑search hill‑climb (rest of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting pairwise optimisation")
        conv = _conv_self(best_a)                     # recompute after possible recent updates
        best_a, best_score = _pairwise_optimize(best_a, conv,
                                                deadline, best_score, best_a)
        print(f"[{time.time() - start_time:.2f}s] Pairwise optimisation finished – best score ≈ {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as a plain Python list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
