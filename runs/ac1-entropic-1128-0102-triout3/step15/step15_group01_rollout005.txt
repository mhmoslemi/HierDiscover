    n = a.shape[0]
    max_val = conv.max()
    active = np.where(conv >= max_val - 1e-12)[0]

"""
Step‑function optimizer – softmax‑parametrised Adam‑style descent.

The evaluation objective (after normalising the sum to 1) is:
    score = 2·n·max(conv(a,a))

We parametrise a_i = softmax(x_i) to stay on the probability simplex.
A (sub‑)gradient of max(conv) is obtained by averaging the gradients of
all convolution shifts that are close to the maximum.
The optimiser runs three phases:

1️⃣  Random initialisation: Dirichlet draws and a few handcrafted
    “U‑shaped’’ distributions are sampled; the best candidate is kept.

2️⃣  Adam‑style optimisation on the soft‑max parameters.
    The update uses the sub‑gradient of max‑convolution, a bias‑corrected
    first‑ and second‑moment estimate and a step‑size that decays as
    1/√t.  Occasionally a tiny simulated‑annealing step is taken to
    escape plateaux.

3️⃣  Final convex mixing with the uniform distribution – a fine‑grained
    sweep over λ∈[0,1] to possibly smooth the peak further.

The routine respects the 1000 s budget, prints progress, and returns the
best sequence found as a plain list of non‑negative floats (already
normalised, so the evaluator’s scaling is irrelevant)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray,
                         rel_tol: float = 1e-9) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)), using a relative tolerance to
    collect several near‑maximal shifts.  The returned gradient is a
    valid sub‑gradient of the piecewise‑linear function max(conv).
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat values within (1‑rel_tol)·max as active
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    if active.size == 0:
        # fallback – should not happen, but keep at least the absolute max
        active = np.array([np.argmax(conv)], dtype=int)

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution a_{k-i}
        grad[idx] += a[k - idx]
    # factor 2 because each term appears twice in the full convolution
    grad = 2.0 * grad / active.size
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)).  a is already normalised."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Phase 0 – random & handcrafted seeding
# ----------------------------------------------------------------------
def _initial_seeds(n: int, rng: np.random.Generator,
                   deadline: float) -> list[np.ndarray]:
    """
    Generate a list of candidate initial probability vectors.
    Includes Dirichlet draws with a wide range of concentrations and a
    few deterministic “U‑shaped’’ profiles.
    """
    seeds = []

    # Dirichlet draws (log‑uniform concentration)
    while time.time() < deadline - 0.1:
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        s = sample.sum()
        if s <= 0:
            continue
        a = sample / s
        seeds.append(a)

        # occasional handcrafted U‑shape
        if rng.random() < 0.02:
            p = rng.uniform(0.5, 5.0)                # exponent controlling steepness
            dist = np.arange(n)
            centre = (n - 1) / 2.0
            d = np.abs(dist - centre) + 1.0
            a_u = d ** (-p)
            a_u = a_u / a_u.sum()
            seeds.append(a_u)

        # stop early if enough seeds collected
        if len(seeds) >= 2000:
            break
    return seeds


# ----------------------------------------------------------------------
# Phase 2 – Adam‑style optimisation on the soft‑max parameters
# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray, n: int, L_fft: int,
                   start_time: float, deadline: float,
                   best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Optimise the probability vector using Adam on the soft‑max parameters.
    """
    rng = np.random.default_rng()
    # initialise log‑odds; add a tiny epsilon to avoid log(0)
    x = np.log(np.maximum(start_a, 1e-12))

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_alpha = 0.5           # raw step size (will be scaled by 1/√t)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    it = 0
    last_improve_it = 0

    while time.time() < deadline:
        it += 1
        # ----- compute current a and score -----
        a = _softmax(x)
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()

        # ----- update global best -----
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,}  new best={cur_score:.6f}")

        # ----- sub‑gradient of max(conv) -----
        grad_conv = _subgradient_maxconv(a, conv, rel_tol=1e-9)
        g_a = 2.0 * n * grad_conv                     # ∇_a  score
        a_dot_ga = np.dot(a, g_a)
        g_x = a * (g_a - a_dot_ga)                    # chain rule through softmax

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size decays like 1/√t (helps early rapid movement)
        eta = base_alpha / np.sqrt(it)

        # propose new x
        x_new = x - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- accept / reject -----
        a_new = _softmax(x_new)
        conv_new = _conv_self(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max()

        if new_score < cur_score:
            x = x_new
        else:
            # occasional simulated‑annealing style acceptance
            T0 = 5e-3
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = T0 * (remaining / total)
            accept_prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < accept_prob:
                x = x_new
        # occasional random perturbation if we’ve stalled too long
        if it - last_improve_it > 20000:
            # add small Gaussian noise to log‑odds to shake out of a plateau
            x += rng.normal(scale=0.02, size=n)
            last_improve_it = it
    return best_seq, best_score

# Phase 3 – mixing with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int,
                      deadline: float, start_time: float,
                      best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Search for λ ∈ [0,1] that improves the score via a convex combination."""
    uniform = np.full(n, 1.0 / n)
    # coarse sweep
    lam_grid = np.linspace(0.0, 1.0, 41)
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        sc = _objective_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # fine sweep around the best λ (if any improvement observed)
    # locate best λ from coarse grid
    if best_seq is not a:
        # simple refinement: linear interpolation between the two best neighbours
        # (a more exhaustive grid could be added if time permits)
        best_lam = lam_grid[np.argmin([_objective_score(lam * a + (1 - lam) * uniform, n)
                                      for lam in lam_grid])]
        fine_grid = np.linspace(max(0.0, best_lam - 0.05),
                                min(1.0, best_lam + 0.05), 41)
        for lam in fine_grid:
            if time.time() >= deadline:
                break
            seq = lam * a + (1.0 - lam) * uniform
            seq = seq / seq.sum()
            sc = _objective_score(seq, n)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Fine mix λ={lam:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 0.25

    # ------------------------------------------------------------------
    # Choose problem size – 4096 is a good trade‑off
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Initialise (uniform or global starter)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # Try to use the globally supplied starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random seeding (Dirichlet + handcrafted shapes)
    # ------------------------------------------------------------------
    seed_deadline = deadline - 200.0   # reserve ~200 s for the Adam phase
    rng = np.random.default_rng()
    seed_candidates = _initial_seeds(n, rng, seed_deadline)

    for a in seed_candidates:
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Seed improvement score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑style optimisation on the soft‑max parameters
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(best_seq, n, L_fft,
                                              start_time, deadline,
                                              best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After Adam best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Mixing with the uniform distribution (fine‑grained sweep)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Mixing with uniform")
        best_seq, best_score = _mix_with_uniform(best_seq, n,
                                                deadline, start_time,
                                                best_seq, best_score)

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
