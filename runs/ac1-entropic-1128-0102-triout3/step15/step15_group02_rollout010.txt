"""
Propose a step‑function (non‑negative sequence) that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The method combines:
  1️⃣  An extended parametric sweep (adds Gaussian family and larger Beta range).
  2️⃣  A massive Dirichlet sampling pass.
  3️⃣  Many short runs of additive sub‑gradient descent from random starts
       (projected onto the simplex) – a cheap alternative to the original EG/Adam.
  4️⃣  A longer EG descent on the exact max‑convolution (as in the reference code).
  5️⃣  A final simulated‑annealing hill‑climb with larger moves.

All steps respect the 1000 s budget; the best sequence found is returned.
"""
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities (identical to the reference implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L_fft)
    conv = irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used by the optimiser: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

def _additive_descent(a0: np.ndarray, deadline: float,

            cur_score = new_score
            step *= 1.005
        else:
            step *= 0.93
        # occasional tiny jitter to avoid stagnation
        if it % 80 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(a.shape[0]) * 5e-4
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


def _restart_additive_descent(n: int, best_seq: np.ndarray,
                              best_score: float, deadline: float,
                              idx: np.ndarray) -> tuple[np.ndarray, float]:
    """Run many short additive descent runs from random Dirichlet starts."""
    rng = np.random.default_rng()
    while time.time() < deadline:
        # draw a random Dirichlet concentration
        log_c = rng.uniform(np.log(0.01), np.log(20.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        # allocate a small slice of the remaining budget to this run
        sub_deadline = min(deadline, time.time() + 0.28)   # about 0.3 s per run
        best_seq, best_score = _additive_descent(a, sub_deadline,
                                                 best_seq, best_score,
                                                 max_iters=450)
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Exponentiated‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _eg_maxconv_descent(a0: np.ndarray,
                        start_time: float,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the sub‑gradient of the L∞‑norm
    of the self‑convolution.  The update is
        a ← normalize( a * exp(-η·grad) )
    where η (step) decays slowly.  Improvements are recorded.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.07            # a slightly larger initial step size
    it = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        grad = _subgradient_maxconv(a, conv)

        # decay the step size gently
        eta_t = eta / (1.0 + it * 8e-4)

        # exponentiated‑gradient step
        exp_factor = np.exp(np.clip(-eta_t * grad, -700, 700))
        a_new = a * exp_factor
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        new_score = _objective_score(a_new)

        if new_score < cur_score - 1e-13:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG it={it:,} score={cur_score:.6f}")
        else:
            eta *= 0.995
            # occasional random shake
            if np.random.rand() < 0.001:
                a = a_new
                cur_score = new_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Final hill‑climb / simulated‑annealing (larger moves)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style hill‑climb with a broader move size."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = _objective_score(a)
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    T0 = 0.03
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # allow a larger fraction (up to ~20 %)
        frac = rng.random() * 0.20 + 1e-4
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-13:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine follows a five‑stage pipeline and returns a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 works well for the reference baseline
    # --------------------------------------------------------------
    n = 4096
    idx = np.arange(n, dtype=np.float64)

    # --------------------------------------------------------------
    # Initialisation (uniform or optional global starter)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Extended parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 350.0   # reserve the bulk of time for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_extended_families(
            n, best_seq, best_score, phase_deadline, idx
        )

    # --------------------------------------------------------------
    # 2️⃣  Broad random Dirichlet sampling
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline, idx
        )

    # --------------------------------------------------------------
    # 3️⃣  Multiple short additive sub‑gradient runs (random restarts)
    # --------------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        best_seq, best_score = _restart_additive_descent(
            n, best_seq, best_score, phase_deadline, idx
        )

    # --------------------------------------------------------------
    # 4️⃣  Exponentiated‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting EG descent")
        best_seq, best_score = _eg_maxconv_descent(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After EG best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final hill‑climb / simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
