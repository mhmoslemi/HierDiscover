"""
propose_candidate: Simulated‑annealing / gradient descent optimizer.

The routine starts from a good existing construction (the global
``height_sequence_1`` if available, otherwise uniform), then repeatedly
moves a small fraction of mass from a “high‑gradient” index (which most
contributes to the current maximal self‑convolution) to a low‑gradient
index.  Moves that improve the objective are always accepted; worsening
moves are sometimes accepted according to a linear cooling schedule
(Metropolis criterion).  The convolution is updated incrementally
(O(n) per move) so that millions of proposals fit easily into the 1000 s
budget.  The best vector seen is returned, normalised to sum = 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (v≥0, sum(v)=1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` (equivalently 2·n·max(conv(a,a)) with sum(a)=1).

    The algorithm:
      1. initialise from the global ``height_sequence_1`` if present,
         otherwise uniform.
      2. perform a large number of Metropolis updates:
         – pick source/destination guided by the sub‑gradient of the
           current max‑convolution,
         – transfer a random fraction (≤20 %) of the source mass,
         – update the full self‑convolution incrementally,
         – accept if the objective improves, otherwise accept with a
           probability that slowly cools to zero.
      3. occasional random shake‑ups prevent stagnation.
      4. the best vector seen is returned as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 works well and matches the reference code.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Initialise from the global starter if possible.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.shape[0] > n:
                starter = starter[:n]
            elif starter.shape[0] < n:
                starter = np.pad(starter, (0, n - starter.shape[0]), mode="constant")
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                a = starter / s
            else:
                a = np.full(n, 1.0 / n)
        else:
            a = np.full(n, 1.0 / n)
    except Exception:
        a = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Initial convolution and score.
    # --------------------------------------------------------------
    conv = _conv_self(a)
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score

    print(f"[{time.time() - start_time:.2f}s] Initial score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # Parameters for the annealing loop.
    # --------------------------------------------------------------
    T0 = cur_score * 0.02          # initial temperature (≈2 % of score)
    max_frac = 0.20                # up to 20 % of source mass per move
    it = 0
    no_improve = 0                 # counter for stagnation detection

    # --------------------------------------------------------------
    # Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # Choose source / destination using the sub‑gradient.
        # ----------------------------------------------------------
        grad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            # fallback to a random pair
            src = rng.integers(n)
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1

        # ----------------------------------------------------------
        # Propose a transfer of a random fraction of the source mass.
        # ----------------------------------------------------------
        eps_frac = rng.random() * max_frac + 1e-7
        eps = a[src] * eps_frac
        if eps <= 0.0:
            continue

        # ----------------------------------------------------------
        # Incremental update of the self‑convolution.
        # ----------------------------------------------------------
        # new_conv = conv + Δ where Δ stems from moving eps from src→dst.
        new_conv = conv.copy()

        # a*Δ contributions (linear term, doubled)
        new_conv[src:src + n] += -2.0 * eps * a
        new_conv[dst:dst + n] += 2.0 * eps * a

        # Δ*Δ contributions (quadratic term, three points only)
        new_conv[2 * src] += eps * eps
        new_conv[2 * dst] += eps * eps
        new_conv[src + dst] += -2.0 * eps * eps

        new_score = float(2.0 * n * new_conv.max())

        # ----------------------------------------------------------
        # Temperature (linear cooling) and Metropolis acceptance.
        # ----------------------------------------------------------
        remaining = max(0.0, deadline - time.time())
        T = T0 * (remaining / (deadline - start_time))

        accept = False
        if new_score < cur_score:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        # ----------------------------------------------------------
        # Apply the move if accepted.
        # ----------------------------------------------------------
        if accept:
            a[src] -= eps
            a[dst] += eps
            # safeguard against tiny negative values due to rounding
            if a[src] < 0.0:
                a[src] = 0.0

            conv = new_conv
            cur_score = new_score
            no_improve = 0

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best = {best_score:.6f}")

        else:
            no_improve += 1

        # ----------------------------------------------------------
        # Periodic re‑projection onto the simplex (keeps sum exactly 1).
        # ----------------------------------------------------------
        if it % 2000 == 0:
            a = _project_simplex(a)
            conv = _conv_self(a)
            cur_score = _objective_score(a)

        # ----------------------------------------------------------
        # Stagnation handling – occasional random shake‑up.
        # ----------------------------------------------------------
        if no_improve > 5000:
            i1, i2 = rng.choice(n, size=2, replace=False)
            if a[i1] > 0.0:
                eps2 = a[i1] * (rng.random() * max_frac + 1e-6)
                a[i1] -= eps2
                a[i2] += eps2
                # recompute exact convolution after the shake‑up
                conv = _conv_self(a)
                cur_score = _objective_score(a)
            no_improve = 0
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # Final normalisation and return.
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
