"""
Propose a non‑negative step‑function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages:

1️⃣  Fast parametric sweeps (geometric, power, beta families, two‑exponential
    mixtures) to obtain a decent seed.
2️⃣  Random Dirichlet draws for additional diversity.
3️⃣  A smooth‑max gradient descent on the simplex.  The true max is replaced by
    a soft‑max with an annealed temperature; the gradient is obtained via a
    single FFT‑based convolution and the iterate is projected onto the
    probability simplex after each step.
4️⃣  A final tiny‑mass hill‑climb that moves probability from the most
    “over‑represented’’ index (largest sub‑gradient) to the most
    “under‑represented’’ one.

All stages respect a wall‑clock budget of 1000 s (≈ 985 s usable).  The best
sequence found is returned as a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assume sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -1, 120)          # fewer points to save time
    for eps in eps_grid:
        if time.time() >= deadline:

        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:

        geo /= s
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geom eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score

def _sweep_two_exponential(n, best_seq, best_score, deadline):

        left /= left.sum()
        right /= right.sum()

def _smooth_gradient_descent(a0: np.ndarray, deadline, best_seq, best_score):
    """
    Minimise the true max‑convolution using a soft‑max surrogate.
    Temperature is annealed; after each step the vector is projected onto
    the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    # hyper‑parameters (tuned empirically)
    step_scale = 0.4          # initial step magnitude (will be divided by sqrt(t))
    tau = 0.02                # starting temperature for the soft‑max
    tau_decay = 0.9995        # geometric decay per iteration
    min_tau = 1e-6

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)                     # length 2n‑1
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD it={it} score={cur_score:.6f}")

        # soft‑max weights
        # shift by max for numerical stability
        exp_arg = (conv - max_val) / max(tau, 1e-12)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient = 2 * (w * a_rev) convolution, central n entries
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # diminishing step size
        step = step_scale / np.sqrt(it)

        a_new = a - step * grad
        # projection onto simplex (non‑negative + sum=1)
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        # anneal temperature
        tau = max(tau * tau_decay, min_tau)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        # start with a relatively larger move (up to 2 % of source)
        eps = a[src] * (rng.random() * 0.02 + 1e-5)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                # occasional random jitter to leave plateaus
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.01 + 1e-5)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    The routine respects a 1000 s wall‑clock limit and returns a list of
    non‑negative floats suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional external starter (if the evaluator supplied one)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Limited parametric sweep (≈ 200 s)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 600.0   # leave ~600 s for the heavy optimisation
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_exponential(n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (≈ 200 s)
    # ------------------------------------------------------------------
    random_deadline = deadline - 380.0   # reserve ≈380 s for gradient descent & hill‑climb
    if time.time() < random_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, random_deadline)

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max gradient descent (≈ 500 s)
    # ------------------------------------------------------------------
    grad_deadline = deadline - 80.0     # keep ~80 s for final hill‑climb
    if time.time() < grad_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max gradient descent")
        best_seq, best_score = _smooth_gradient_descent(best_seq, grad_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After smooth‑GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb (last 80 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
