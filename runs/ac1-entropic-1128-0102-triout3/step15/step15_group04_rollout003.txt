"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines four ideas:

1️⃣  **Broad parametric sweep** – a cheap grid over a geometric×power family,
    plus a uniform mixture, to obtain a good warm‑start.

2️⃣  **Exponentiated‑gradient descent with a smooth‑max surrogate** – we
    minimise a soft‑max approximation of the L∞‑objective.
    The temperature β is increased (0.5 → 512) while an
    exponential‑weights update (mirror descent on the simplex) is applied.
    This often gives a better descent direction than the plain projected
    gradient used previously.

3️⃣  **Long sub‑gradient polishing** – classic sub‑gradient descent on the
    true max‑objective, but with a more aggressive step‑size schedule and
    occasional random jumps.

4️⃣  **Final tiny hill‑climb** – random mass transfers to squeeze out the
    last bits of improvement.

All phases respect a 1000 s time budget and the routine finally returns a
plain Python list of floats (normalised to sum = 1).  Only NumPy is used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation helpers
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)         # avoid overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Mix the given shape with the uniform distribution."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights p_k = exp(beta*conv_k) / Σ exp(·).
    Numerically stable by shifting with the maximal entry.
    """
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max objective.
    g_i = 2 * Σ_j a_j * p_{i+j}, i.e. 2·(a ⋆ p)(i) where ⋆ is cross‑correlation.
    Computed with FFT for speed.
    """
    n = a.shape[0]
    # cross‑correlation a ⋆ p = conv(a, rev(p))
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    conv_full = np.fft.irfft(Af * Pf, n=L_grad)

    # The full convolution length is up to 3n‑2; we need the central n entries.
    # For i = 0 … n‑1 the correlation index is i + (len(p)-1) = i + (2n‑2).
    start = n - 1
    g = conv_full[start:start + n][::-1]   # reverse to obtain correct orientation
    return 2.0 * g


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    it = 0
    step0 = 0.16

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # exact sub‑gradient for the maximal shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        step = step0 / (it ** 0.6)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional uphill move (annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 12.0 /
                          (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

            exp_arg = -eta * g
            shift = np.max(exp_arg)
            mult = np.exp(exp_arg - shift)
            a_candidate = a * mult
            a_candidate = a_candidate / a_candidate.sum()   # projects onto simplex

            # evaluate the true objective only every few iterations (saves time)
            if it % 200 == 0:
                cand_score = _eval_fast(a_candidate, L_fft)
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} it={it} "
                          f"score={cand_score:.6f}")

                # accept if improvement, else allow a tiny uphill move
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                else:
                    prob = np.exp(-(cand_score - cur_score) * 4.0 /
                                  (np.sqrt(it) + 1.0))
                    if rng.random() < prob:
                        a = a_candidate
                        cur_score = cand_score
            else:
                # even when we do not evaluate, we still move in the descent direction
                a = a_candidate

            # occasional jitter to shake out plateaus
            if it % 5000 == 0:
                eps = 1e-4 * rng.random(n) * a
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)

    # --------------------------------------------------------------
    # 3️⃣  Final exact sub‑gradient descent (≈200 s remaining).
    # --------------------------------------------------------------
    if time.time() < deadline - 70.0:
        sub_deadline = deadline - 70.0
        a_sub, sc_sub = _subgradient_descent(best_seq, sub_deadline,
                                             start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient final score={sc_sub:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Tiny random hill‑climb (mass transfers) – polish the last few secs.
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 2e-3   # ≤0.2 % of source mass
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
