"""
Hybrid optimizer for the step‑function bound 
    2·n·max(conv(a,a)) / (sum(a)²)

Improvements over the reference implementation
------------------------------------------------
* A much finer sweep of analytic families, especially
  * very small exponential decay rates (lam ∈ [1e‑6,0.1])
  * power‑law exponents very close to 0 (p ∈ [1e‑4,0.5])
  * left‑skewed beta distributions with extreme α,β
  * linear‑decreasing and mixed‑exponential shapes
* The initial pool is enlarged (TOP_K = 30) and the best candidates are
  up‑sampled to the final length if they come from a smaller grid.
* Exponentiated sub‑gradient descent is invoked **cyclically** on the
  top‑K vectors with a short time‑slice (≈5 s) for each candidate – this
  gives many independent restarts and a better chance to escape local
  minima.
* After the sub‑gradient phase a single Adam / smooth‑max polish is run
  for the remaining budget.
* Simple simulated‑annealing tweaks are applied to the current best
  vector before the final Adam step.

The routine respects the 1000 s budget, prints progress information
and returns a plain list of non‑negative floats ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """
    Symmetric power‑ramp. `p` controls side steepness,
    `plateau_frac` ∈[0,0.5] is the relative length of the flat top.
    The vector is normalised to sum 1.
    """
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum() if shape.sum() > 0 else 1.0
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    total = shape.sum()
    shape = shape / total if total > 0 else np.full(n, 1.0 / n)
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p} (monotonically decreasing)."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    total = shape.sum()
    shape = shape / total if total > 0 else np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    shape = shape / total if total > 0 else np.full(n, 1.0 / n)
    return shape


def _make_linear_decreasing(n: int) -> np.ndarray:
    """Deterministic linearly decreasing shape (triangular)."""
    shape = np.arange(n, 0, -1, dtype=float)
    total = shape.sum()
    shape = shape / total if total > 0 else np.full(n, 1.0 / n)
    return shape


def _make_mixed_exponential(n: int, lam1: float, lam2: float,
                           w: float) -> np.ndarray:
    """
    Convex combination of two exponentials.
    w∈[0,1] weight for the first component.
    """
    idx = np.arange(n, dtype=float)
    a1 = np.exp(-lam1 * idx)
    a2 = np.exp(-lam2 * idx)
    shape = w * a1 + (1.0 - w) * a2
    total = shape.sum()
    shape = shape / total if total > 0 else np.full(n, 1.0 / n)
    return shape

def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

        max_conv = conv.max()

        # Active set (indices where the max is attained)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # Gradient = 2 * (w ⋆ a_rev)   where ⋆ denotes correlation
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + it / 5000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.1
    T0 = 0.1
    T_end = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        T = T_end + (T0 - T_end) * (0.5 ** (it / 4000.0))

        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best, best_score


# ----------------------------------------------------------------------
# Minor simulated‑annealing perturbation (helps escape local minima)
# ----------------------------------------------------------------------
def _anneal_step(vec: np.ndarray,
                 temperature: float,
                 rng: np.random.Generator) -> np.ndarray:
    """Small random mass redistribution with Metropolis acceptance."""
    n = vec.shape[0]
    # pick two distinct indices
    i, j = rng.choice(n, size=2, replace=False)
    # propose moving a fraction eps of mass from i to j
    eps = temperature * rng.random()
    eps = min(eps, vec[i])  # cannot move more than we have
    new_vec = vec.copy()
    new_vec[i] -= eps
    new_vec[j] += eps
    # projection onto simplex is already fulfilled (mass conserved, non‑negative)
    return new_vec


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function.
    The routine builds a richer pool of analytic shapes (including very
    small exponential rates and extreme beta/skewed families), runs a
    cyclic exponentiated sub‑gradient descent with many restarts, applies a
    brief annealing stage, and finishes with a smooth‑max Adam polish.
    Returns the best vector (normalised to sum = 1) as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start = time.time()
    deadline = start + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Choose problem size – a moderate N gives many cheap iterations
    # --------------------------------------------------------------
    remaining = deadline - start
    if remaining > 800:
        N = 16384
    elif remaining > 400:
        N = 8192
    else:
        N = 4096

    # --------------------------------------------------------------
    # 1) Build a diverse initial pool
    # --------------------------------------------------------------
    TOP_K = 30
    pool = []                     # list of (score, vector)

    def _add(vec: np.ndarray) -> None:
        _push_top(pool, _objective(vec), vec, TOP_K)

    # Uniform baseline
    _add(_make_uniform(N))

    # Linear decreasing shape
    _add(_make_linear_decreasing(N))

    # Power‑ramps (symmetric)
    for p in np.linspace(0.1, 5.0, 9):
        for plat in np.linspace(0.0, 0.6, 4):
            _add(_make_power_ramp(N, p, plat))

    # Exponential decays – very fine sweep at the low‑lam end
    lam_vals = np.concatenate((
        np.logspace(-6, -2, 15),   # 1e‑6 … 1e‑2
        np.logspace(-2, -0.5, 15), # 1e‑2 … 0.3
        np.linspace(0.3, 4.0, 10)   # moderate values
    ))
    for lam in lam_vals:
        _add(_make_exponential(N, lam))

    # Mixed exponentials (two‑component mixtures)
    for _ in range(20):
        lam1 = 10 ** rng.uniform(-6, 0)
        lam2 = 10 ** rng.uniform(-6, 0)
        w = rng.random()
        _add(_make_mixed_exponential(N, lam1, lam2, w))

    # Power‑law decays – include extremely small exponents
    p_vals = np.concatenate((
        np.logspace(-4, -1, 15),   # 1e‑4 … 0.1
        np.linspace(0.1, 3.0, 15)  # 0.1 … 3
    ))
    for p in p_vals:
        _add(_make_power_law(N, p))

    # Asymmetric Beta families – explore extreme skew
    alphas = np.concatenate((
        np.logspace(-2, -0.5, 8),   # 0.01 … 0.3
        np.linspace(0.3, 3.0, 8)    # 0.3 … 3
    ))
    betas = np.concatenate((
        np.logspace(-2, -0.5, 8),
        np.linspace(0.3, 3.0, 8)
    ))
    for a in alphas:
        for b in betas:
            _add(_make_beta_shape_asym(N, a, b))

    # Random Dirichlet draws (sorted decreasing)
    concs = [0.01, 0.05, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(30):
            if time.time() >= deadline:
                break
            vec = rng.gamma(conc, 1.0, size=N)
            vec = np.sort(vec)[::-1]          # enforce decreasing shape
            s = vec.sum()
            if s > 0:
                vec /= s
                _add(vec)

    # If a previous best vector is available, up‑sample it
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:
                starter_up /= s
                _add(starter_up)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the current best from the pool
    # ------------------------------------------------------------------
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Exponentiated sub‑gradient descent (many short restarts)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + 0.70 * remaining
        per_candidate = 5.0                     # seconds per call
        while time.time() < sub_deadline:
            for _score, vec in pool:
                if time.time() >= sub_deadline:
                    break
                chunk_deadline = time.time() + min(per_candidate,
                                                   sub_deadline - time.time())
                seq, sc = _exp_subgradient_descent(vec, chunk_deadline, start)
                if sc < best_score:
                    best_score, best_vec = sc, seq.copy()
                    print(f"[{time.time() - start:.2f}s] ExpGD improvement → {sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Light simulated‑annealing perturbations (helps escape plateaus)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        anneal_end = time.time() + min(5.0, remaining * 0.10)
        temperature = 0.02  # maximum fraction of mass moved per step
        while time.time() < anneal_end:
            cand = _anneal_step(best_vec, temperature, rng)
            cand = _project_simplex(cand)
            sc = _objective(cand)
            if sc < best_score:
                best_score, best_vec = sc, cand.copy()
                print(f"[{time.time() - start:.2f}s] Anneal improvement → {sc:.6f}")
            temperature *= 0.9995   # gradually cool

    # ------------------------------------------------------------------
    # 4) Smooth‑max (Adam) polishing for any leftover time
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        seq, sc = _smooth_gradient_descent(best_vec, deadline, start)
        if sc < best_score:
            best_score, best_vec = sc, seq.copy()
            print(f"[{time.time() - start:.2f}s] SmoothGD final improvement → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
