"""
Hybrid multi‑scale optimizer for the step‑function bound

The algorithm proceeds in three stages:

1. **Coarse exploration (N≈4–8 k).**  
   A diverse pool of analytic and random shapes is built.  For each of the
   top‑K candidates a few hundred iterations of projected gradient descent
   on a *smooth‑max* surrogate of the convolution max (temperature‑annealed
   soft‑max) are performed.  This quickly discovers promising regions of the
   simplex.

2. **Upsampling.**  
   The best coarse solution is interpolated to a finer grid (N≈16–32 k) and
   re‑normalised.

3. **Fine optimisation.**  
   On the fine grid we alternate two complementary updates:
   * A multiplicative sub‑gradient step that directly reduces the current
     active convolution peak.
   * A smooth‑max Adam step (with an exponentially decaying temperature)
     that refines the whole profile.
   The alternation, together with a simple learning‑rate schedule and
   projection onto the probability simplex, allows the search to escape
   shallow local minima and pushes the bound below the previous 1.508… level.

All operations are vectorised (FFT based convolutions) and the routine
adapts its problem size to the remaining time budget, guaranteeing a
return before the 1000 s deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT convolution, objective, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2*len(a)-1) using a real FFT.
    The result is clipped to be non‑negative (tiny numerical noise may appear).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return np.maximum(conv, 0.0)


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators (all return vectors normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power ramp with exponent p and plateau length fraction."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on the discrete grid."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Simple top‑k pool utilities
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv(a,a)).
    Returns a vector of size len(a).
    """
    conv = _conv_self(a)
    max_c = conv.max()
    # stable soft‑max weights
    w_raw = np.exp((conv - max_c) / temperature)
    w = w_raw / w_raw.sum()
    # correlation (gradient of linear form w⋅conv) w.r.t. a
    a_rev = a[::-1]
    n = a.shape[0]
    L = _next_pow2(3 * n - 2)
    grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
    grad = 2.0 * grad[n - 1 : n - 1 + n]
    return grad


def _active_set_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Multiplicative sub‑gradient step based on the current active set
    (indices attaining max(conv)).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    tol = max(1e-12, max_conv * 1e-12)
    active = np.where(conv >= max_conv - tol)[0]
    if active.size == 0:
        active = np.array([conv.argmax()], dtype=int)
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    a_rev = a[::-1]
    L = _next_pow2(3 * n - 2)
    grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
    grad = 2.0 * grad[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
