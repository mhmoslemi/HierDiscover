"""
Propose a non‑negative step function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The routine runs for up to 1000 s and combines several strategies:

1️⃣  A quick uniform baseline.
2️⃣  Random sampling of a 4‑parameter family
    a_i ∝ r**i·(i+1)^p·(n‑i)^q mixed with a uniform background (λ).
3️⃣  Random Dirichlet draws (exponential‑distribution normalised to 1).
4️⃣  For the most promising candidates run an Adam‑style projected
    sub‑gradient descent on the true objective (gradient taken over the
    whole active set of near‑maximal convolution entries).
5️⃣  A final cheap “mass‑transfer” hill‑climb that also tries a tiny
    smoothing move.

All vectors are projected onto the probability simplex after each
update, guaranteeing non‑negativity and a fixed L¹‑norm, which makes the
objective scale‑invariant.  Progress and intermediate scores are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (four‑parameter family)
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components proportional to
    r**i·(i+1)**p·(n‑i)**q   (i = 0…n‑1).  Computed in log‑space.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)               # avoid overflow
    shape = np.exp(log_shape)
    total = shape.sum()
    if not np.isfinite(total) or total == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return λ·uniform + (1‑λ)·shape (both already sum to 1) and renormalise.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient of the active set of maximal convolution entries
# ----------------------------------------------------------------------


def _compute_subgrad_multi(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a, aggregating over all
    indices whose value is within tol of the maximum.
    """
    max_val = conv.max()
    # tolerance can be absolute or relative – we use an absolute one
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        g[i_vals] += a[j_vals]                # contribution a_j to ∂/∂a_i
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]  # symmetric part
    return g


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent
# ----------------------------------------------------------------------


def _adam_subgradient(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the true objective.
    The gradient is the sub‑gradient over the whole active set of
    near‑maximal convolution entries.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.20               # base step size
    m = np.zeros_like(a)    # first moment
    v = np.zeros_like(a)    # second moment
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=1e-12)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # decreasing learning rate (optional)
        lr = lr0 / (np.sqrt(it) + 1.0)

        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{time.time() - start_time:.2f}s] adam it={it:,} "
                  f"score={cur_score:.6f}")
        else:
            # simple simulated‑annealing acceptance
            prob = np.exp(-(cur_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # occasional jitter to avoid stagnation
        if it % 3000 == 0:
            eps_vec = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + eps_vec, 0.0)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny hill‑climb (mass transfers + smoothing)
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two positions,
    occasionally apply a mild smoothing kernel, keep the best solution.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ------------ mass transfer ------------
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3      # up to 0.2 % transfer
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final mass‑transfer "
                  f"score={new_sc:.6f}")

        # ------------ occasional smoothing ------------
        if rng.random() < 0.005:
            w = 3
            kernel = np.ones(w) / w
            a_smooth = np.convolve(a, kernel, mode='same')
            a_smooth = np.maximum(a_smooth, 0.0)
            a_smooth /= a_smooth.sum()
            new_sc = _eval_fast(a_smooth, L_fft)
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a_smooth.copy()
                a = a_smooth
                print(f"[{time.time() - start_time:.2f}s] final smoothing "
                      f"score={new_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Random sampling of the four‑parameter family
# ----------------------------------------------------------------------


def _random_param_search(n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray,
                                                     float,
                                                     tuple[float, float, float, float]]:
    """
    Uniform‑in‑log sampling of (r,p,q,λ) and keep the best vector.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)
    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        # r very close to 1 improves the bound – sample log‑uniform
        r = np.exp(rng.uniform(np.log(0.90), np.log(0.9999999)))
        p = rng.uniform(-5.0, 5.0)
        q = rng.uniform(-5.0, 5.0)
        lam = rng.uniform(0.0, 0.8)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam)
            print(f"[{time.time() - start_time:.2f}s] random r={r:.6f} "
                  f"p={p:.3f} q={q:.3f} λ={lam:.3f} score={sc:.6f}")

    return best_seq, best_score, best_params


# ----------------------------------------------------------------------
# Random Dirichlet seeds (exponential‑distribution)
# ----------------------------------------------------------------------


def _random_dirichlet_seeds(n: int,
                           deadline: float,
                           start_time: float,
                           count: int = 10) -> list[np.ndarray]:
    """
    Produce up to `count` random points on the probability simplex
    (sampled from iid exponential variables) and keep the best few.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)
    pool = []
    best_score = float("inf")
    while time.time() < deadline and len(pool) < count:
        vec = rng.exponential(1.0, size=n)
        vec /= vec.sum()
        sc = _eval_fast(vec, L_fft)
        if sc < best_score:
            best_score = sc
        pool.append((vec, sc))
        print(f"[{time.time() - start_time:.2f}s] dirichlet seed score={sc:.6f}")

    # keep only the 5 best seeds
    pool.sort(key=lambda x: x[1])
    return [v for v, _ in pool[:5]]

                                          start_time, best_score)
        if sc_sub < best_score:

            best_score = sc_sub
            print(f"[{time.time() - start_time:.2f}s] Adam descent improved to {best_score:.6f}")

    # --------------------------------------------------------------

            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Log the official evaluation (uses the provided security‑checked function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
