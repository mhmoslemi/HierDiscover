    n = a.shape[0]
    m = w.shape[0]
    L = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)

    Corr = np.fft.irfft(np.conj(A) * W, n=L)
    # take first n+m-1 entries

"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a hybrid of:
  * fast random exploration of a rich parametric family (monotone or symmetric)
  * simple stochastic hill‑climb on the parameters
  * smooth “log‑sum‑exp” gradient descent with momentum (annealing of α)
  * a cheap random mass‑transfer local search at the end

The code respects a 1000 s budget, works for a length of 8192 (≈8 k) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (monotone and symmetric families)
# ----------------------------------------------------------------------


def _make_shape_monotonic(n: int, r: float, p: float, q: float) -> np.ndarray:
    """a_i ∝ r**i (i+1)**p (n-i)**q  (i = 0…n‑1)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)               # prevent overflow
    shape = np.exp(log_shape)
    tot = shape.sum()
    if tot == 0.0 or not np.isfinite(tot):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= tot
    return shape


def _make_shape_symmetric(n: int, r: float, p: float, q: float) -> np.ndarray:
    """a_i ∝ r**min(i,n‑1‑i) (i+1)**p (n-i)**q  (U‑shaped)."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)                 # distance to nearest end
    log_shape = d * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    tot = shape.sum()
    if tot == 0.0 or not np.isfinite(tot):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= tot
    return shape


def _make_shape_general(n: int, r: float, p: float, q: float, sym: bool) -> np.ndarray:
    """Dispatch to monotone or symmetric shape generator."""
    if sym:
        return _make_shape_symmetric(n, r, p, q)
    else:
        return _make_shape_monotonic(n, r, p, q)


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·uniform + (1‑lam)·shape  (both already sum to 1)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Linear correlation a ⋆ w  (used for the smooth gradient)
# ----------------------------------------------------------------------


def _corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation (a ⋆ w)[i] = Σ_j a_j w_{i+j}.
    Returns only the first n values (i = 0 … n‑1).
    """
    n = a.shape[0]
    m = w.shape[0]
    L = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(np.conj(A) * W, n=L)[: n + m - 1]
    return corr[:n]                                   # i = 0 … n‑1


# ----------------------------------------------------------------------
# Random parametric search (covers both families)
# ----------------------------------------------------------------------


def _random_param_search_general(
    n: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
    cur_best_seq: np.ndarray,
    cur_best_score: float,
) -> tuple[np.ndarray, float, tuple]:
    """
    Uniform‑in‑log random sampling of (r,p,q,λ, sym) and keep the best vector.
    """
    L_fft = _next_pow2(2 * n - 1)

    best_seq = cur_best_seq
    best_score = cur_best_score
    best_params = None

    while time.time() < deadline:
        # geometric factor – log‑uniform
        r = np.exp(rng.uniform(np.log(0.5), np.log(0.999999)))
        p = rng.uniform(-4.0, 4.0)
        q = rng.uniform(-4.0, 4.0)
        lam = rng.uniform(0.0, 1.0)
        sym = rng.random() < 0.5

        shape = _make_shape_general(n, r, p, q, sym)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam, sym)
            print(
                f"[{time.time() - start_time:.2f}s] random r={r:.4f} "
                f"p={p:.2f} q={q:.2f} λ={lam:.2f} sym={sym} score={sc:.6f}"
            )
    return best_seq, best_score, best_params


# ----------------------------------------------------------------------
# Simple hill‑climb on the four shape parameters (+ symmetry flag)
# ----------------------------------------------------------------------


def _hillclimb_params_general(
    initial_params: tuple,
    start_seq: np.ndarray,
    start_score: float,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
    n: int,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb on (r,p,q,λ, sym) with exponentially shrinking perturbations.
    """
    cur_r, cur_p, cur_q, cur_lam, cur_sym = initial_params
    cur_seq = start_seq.copy()
    cur_score = start_score

    sigma_r = 0.02
    sigma_p = 0.5
    sigma_q = 0.5
    sigma_lam = 0.12

    best_seq = cur_seq.copy()
    best_score = cur_score

    while time.time() < deadline:
        # propose new parameters
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        q_new = cur_q + sigma_q * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()
        sym_new = cur_sym
        if rng.random() < 0.01:
            sym_new = not cur_sym

        # clamp
        r_new = float(np.clip(r_new, 0.5, 0.999999))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_shape_general(n, r_new, p_new, q_new, sym_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = _eval_fast(cand_new, L_fft)

        if sc_new < cur_score:
            cur_r, cur_p, cur_q, cur_lam, cur_sym = r_new, p_new, q_new, lam_new, sym_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] hill r={r_new:.6f} p={p_new:.4f} "
                    f"q={q_new:.4f} λ={lam_new:.4f} sym={sym_new} score={sc_new:.6f}"
                )

        # decay perturbations
        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_q *= 0.9995
        sigma_lam *= 0.9995

        # occasional jitter
        if rng.random() < 0.005:
            cur_seq = np.maximum(cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq, 0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp gradient descent with momentum (annealed α)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth surrogate
           f_α(a) = 2·n/α·log Σ exp(α·conv_k(a))
    Uses momentum, slowly raises α and occasional jitter.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)  # for correlation a⋆w
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    step0 = 0.16
    beta = 0.9
    v = np.zeros_like(a)

    alpha = 5.0                     # start with a soft objective
    it = 0

    while time.time() < deadline:
        it += 1

        # self‑convolution
        conv = _conv_self_fft(a, L_fft)

        # soft‑max weights over convolution entries
        maxc = conv.max()
        exp_arg = alpha * (conv - maxc)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:               # numerical safeguard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # correlation a ⋆ w  (gradient of the smooth part)
        corr = _corr(a, w)              # length n
        grad = corr                     # direction (scale absorbed in step size)

        # momentum
        v = beta * v + (1.0 - beta) * grad

        # step size decays gently
        step = step0 / (it ** 0.6)
        a_candidate = a - step * v
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            v = np.zeros_like(a)          # reset after improvement
            print(
                f"[{time.time() - start_time:.2f}s] smooth it={it:,} α={alpha:.2f} "
                f"step={step:.5e} score={cur_score:.6f}"
            )
        else:
            # occasional uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # schedule α – slowly sharpen the surrogate
        if it % 2000 == 0:
            alpha = min(alpha * 1.20, 200.0)
            # modest jitter to escape flat regions
            eps = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            v = np.zeros_like(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny hill‑climb (random mass transfers)
# ----------------------------------------------------------------------


def _final_local_search(
    a: np.ndarray,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two coordinates,
    keeping the vector on the simplex.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to 0.2 % of the source mass
        eps = a[src] * rng.random() * 2e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final mass‑transfer score={new_sc:.6f}")
        else:
            # occasional uphill acceptance
            if rng.random() < 0.001:
                a = a_prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the target evaluation.
    The routine respects the 1000 s time budget and returns the best sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8192 gives a finer discretisation while staying fast
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline (score = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣🔹  Global starter (if provided by the harness)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_seq = starter.copy()
                best_score = cur
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep (both families)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + min(120.0, deadline - time.time())
    best_seq, best_score, best_params = _random_param_search_general(
        n, sweep_deadline, start_time, rng, best_seq, best_score
    )
    if best_params is not None:
        print(
            f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}"
        )
    else:
        # fallback simple parameters
        best_params = (0.97, 0.0, 0.0, 0.3, False)

    # ------------------------------------------------------------------
    # 2️⃣  Hill‑climb on the five parameters (r,p,q,λ,sym)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    hill_deadline = time.time() + min(150.0, remaining * 0.20)
    best_seq, best_score = _hillclimb_params_general(
        best_params,
        best_seq,
        best_score,
        hill_deadline,
        start_time,
        rng,
        n,
        L_fft,
    )
    print(
        f"[{time.time() - start_time:.2f}s] after hill‑climb best score={best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # 3️⃣  Smooth log‑sum‑exp gradient descent (major work)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    grad_deadline = time.time() + min(600.0, remaining * 0.70)
    if time.time() < grad_deadline:
        a_smooth, sc_smooth = _smooth_gradient_descent(
            best_seq, grad_deadline, start_time, best_score
        )
        if sc_smooth < best_score:
            best_seq = a_smooth.copy()
            best_score = sc_smooth
            print(f"[{time.time() - start_time:.2f}s] smooth descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny mass‑transfer hill‑climb
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_final, sc_final = _final_local_search(best_seq, deadline, start_time, best_score)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Final official evaluation (security‑checked)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
