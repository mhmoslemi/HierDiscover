"""
Improved step‑function search.

The optimizer works in four stages under a single‑CPU 1000 s budget:

1️⃣  Very fast parametric sweep (geometric × power shapes + uniform mixing)
    and a short random‑shape exploration to obtain a good warm‑start.

2️⃣  Continuation on a smooth approximation of the L∞‑objective.
    The soft‑max (log‑sum‑exp) of the self‑convolution is minimised with
    Adam‑style updates.  The temperature β is increased from 0.5 up to 512,
    each value receiving a time slice proportional to the remaining budget.

3️⃣  Exact sub‑gradient descent on the true max‑convolution.
    Instead of using only the single maximal shift, we form a uniform weight
    on the K largest convolution entries (K≈5) – this drives down the whole
    peak region.  Adam is again used for fast convergence, with a simple
    simulated‑annealing acceptance rule for uphill steps.

4️⃣  A final tiny random mass‑transfer hill‑climb that runs for the few
    seconds left.

All updates project onto the probability simplex, guaranteeing non‑negative
coefficients and a fixed total mass (scale‑invariance makes the exact sum
irrelevant).  The routine prints only when a new best objective is found,
so the inner loops stay fast and can perform many thousands of iterations
within the allotted time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)−1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector (sum≈1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Mix shape with the uniform distribution; result already sums to 1."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Soft‑max utilities (smooth approximation of max)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max probabilities p_k = exp(beta * conv_k) / Σexp(.) (stable)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of  Σ_k p_k * (a*a)_k .
    Implements g_i = 2 * Σ_j a_j * p_{i+j} via FFT.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    conv_full = np.fft.irfft(Af * Pf, n=L_grad)
    # relevant slice: indices n‑1 … n‑1+(n‑1) i.e. length n
    start = n - 1
    g = conv_full[start:start + n][::-1]   # reverse back
    return 2.0 * g


# ----------------------------------------------------------------------
# Adam optimizer (vector‑wise)
# ----------------------------------------------------------------------


def _adam_update(m: np.ndarray, v: np.ndarray, g: np.ndarray,
                 t: int, lr: float,
                 b1: float = 0.9, b2: float = 0.999,
                 eps: float = 1e-12) -> np.ndarray:
    """Return updated parameter vector using Adam step."""
    m = b1 * m + (1.0 - b1) * g
    v = b2 * v + (1.0 - b2) * (g * g)
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return step, m, v
