# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used by the optimiser: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (including new Gaussian, Laplace, Kaiser)
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray,
                    best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # ---- Uniform baseline (already evaluated outside but keep for safety) ----
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ---- geometric ----------------------------------------------------
    eps_grid = np.logspace(-5, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        # mixture with uniform
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.0, 101)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type ------------------------------------------------------
    alphas = np.linspace(0.05, 4.0, 61)
    betas = np.linspace(0.05, 4.0, 61)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # ---- stretched‑exponential -----------------------------------------
    scales = np.linspace(0.25, 1.0, 11)
    p_vals = np.linspace(0.2, 5.0, 41)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched p={p:.2f} scale={scale:.2f} score={sc:.6f}")

    # ---- logistic -------------------------------------------------------
    mu_factors = np.linspace(0.2, 0.8, 13)
    s_factors = np.linspace(0.01, 0.2, 11)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-6:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    # ---- sine‑power -----------------------------------------------------
    p_vals = np.linspace(0.1, 5.0, 41)
    for p in p_vals:
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine‑power p={p:.2f} score={sc:.6f}")

    # ---- Gaussian -------------------------------------------------------
    sigma_factors = np.linspace(0.05, 0.4, 14)   # relative to n
    mu_factors = np.linspace(0.3, 0.7, 9)       # allow asymmetry
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * (n - 1)
        for sf in sigma_factors:
            if time.time() >= deadline:
                break
            sigma = sf * n
            seq = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Gaussian mu={mu_f:.2f} σ={sf:.2f} score={sc:.6f}")

    # ---- Laplace (double‑exponential) ----------------------------------
    b_factors = np.linspace(0.05, 0.4, 14)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * (n - 1)
        for bf in b_factors:
            if time.time() >= deadline:
                break
            b = bf * n
            seq = np.exp(-np.abs(idx - mu) / b)
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Laplace mu={mu_f:.2f} b={bf:.2f} score={sc:.6f}")

    # ---- Kaiser window ---------------------------------------------------
    beta_vals = np.linspace(0.5, 12.0, 23)   # typical range for side‑lobe control
    mu_f = 0.5                               # centre for symmetry; we also try shifted versions
    for beta in beta_vals:
        if time.time() >= deadline:
            break
        mu = mu_f * (n - 1)
        r = (idx - mu) / mu
        # protect against numerical errors outside [-1,1]
        r = np.clip(r, -1.0, 1.0)
        arg = beta * np.sqrt(1.0 - r * r)
        seq = np.i0(arg)
        seq = np.clip(seq, 1e-12, None)  # avoid exact zeros
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Kaiser β={beta:.2f} score={sc:.6f}")

    # ---- mixed uniform‑geometric (already covered) and uniform‑beta -----
    # (kept for completeness – not strictly necessary here)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet trials (full resolution)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 3000
    for _ in range(trials):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(np.log(0.01), np.log(30.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Coarse segment optimisation (Λ‑dimensional)
# ----------------------------------------------------------------------
def _segment_counts(m: int, n: int) -> np.ndarray:
    """Return integer lengths of each of m equal‑size segments covering n points."""
    base = n // m
    remainder = n % m
    counts = np.full(m, base, dtype=int)
    counts[:remainder] += 1
    return counts


def _upsample_from_counts(w: np.ndarray, counts: np.ndarray) -> np.ndarray:
    """Upsample a coarse weight vector w (sum=1) into a full‑length sequence."""
    seg_vals = w / counts.astype(float)      # each entry inside a segment has the same density
    return np.repeat(seg_vals, counts)


def _segment_dirichlet_trials(m: int, n: int,
                              best_seq: np.ndarray, best_score: float,
                              deadline: float) -> tuple[np.ndarray, float]:
    """Random Dirichlet draws for a segment‑level representation."""
    rng = np.random.default_rng()
    counts = _segment_counts(m, n)
    trials = 2500
    for _ in range(trials):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(np.log(0.01), np.log(30.0))
        c = np.exp(log_c)
        w = rng.gamma(shape=c, scale=1.0, size=m)
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w /= w_sum
        a = _upsample_from_counts(w, counts)
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Segments m={m} Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


def _eg_segment_descent(m: int, n: int,
                       best_seq: np.ndarray, best_score: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the segment weights."""
    counts = _segment_counts(m, n)
    # initialise w from current best_seq by aggregating mass per segment
    cum = np.cumsum(np.concatenate(([0], counts)))
    w = np.empty(m, dtype=float)
    for j in range(m):
        w[j] = best_seq[cum[j]:cum[j+1]].sum()
    w = np.maximum(w, 1e-15)
    w /= w.sum()

    eta0 = 0.08
    it = 0
    cur_score = _objective_score(best_seq)
    while time.time() < deadline:
        it += 1
        a = _upsample_from_counts(w, counts)
        conv = _conv_self(a)
        max_val = conv.max()
        # sub‑gradient w.r.t. a
        grad_a = _subgradient_maxconv(a, conv)
        # aggregate to segment‑level gradient
        grad_w = np.empty(m, dtype=float)
        start = 0
        for j in range(m):
            seg_len = counts[j]
            seg_sum = grad_a[start:start+seg_len].sum()
            grad_w[j] = seg_sum / seg_len  # chain rule factor ∂a_i/∂w_j = 1/seg_len
            start += seg_len

        # adaptive step size (Polyak‑type decay)
        eta_t = eta0 / (1.0 + it * 1e-4)

        exp_factor = np.exp(np.clip(-eta_t * grad_w, -700, 700))
        w_new = w * exp_factor
        w_new = np.maximum(w_new, 1e-15)
        w_new /= w_new.sum()

        a_new = _upsample_from_counts(w_new, counts)
        new_score = _objective_score(a_new)

        if new_score < cur_score - 1e-12:
            w = w_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] EG‑seg m={m} it={it:,} score={new_score:.6f}")
        else:
            # shrink step on stagnation
            eta0 *= 0.995
            if np.random.rand() < 0.001:
                w = w_new
                cur_score = new_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fine‑grained exponentiated‑gradient descent on full resolution
# ----------------------------------------------------------------------
def _eg_maxconv_descent(a0: np.ndarray,
                        start_time: float,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the simplex using the exact sub‑gradient of max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.05
    it = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        eta_t = eta / (1.0 + it * 1e-4)

        exp_factor = np.exp(np.clip(-eta_t * grad, -700, 700))
        a_new = a * exp_factor
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        new_score = _objective_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG‑full it={it:,} score={cur_score:.6f}")
        else:
            eta *= 0.99
            if np.random.rand() < 0.001:
                a = a_new
                cur_score = new_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Final hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns a list of floats (the sequence).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # safety margin

    # problem size (original implementation used 4096)
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initial uniform baseline (also used for normalisation)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (may be provided by the platform)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric families (extended)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet sampling (full resolution)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # 3️⃣  Coarse segment optimisation (multiple granularities)
    # ------------------------------------------------------------------
    segment_sizes = [64, 128, 256]                # diverse scales
    for m in segment_sizes:
        # give each size a small slice of the remaining budget
        sub_deadline = min(deadline - 80.0, time.time() + 30.0)
        if time.time() >= sub_deadline:
            break
        # Dirichlet draws at segment level
        best_seq, best_score = _segment_dirichlet_trials(
            m, n, best_seq, best_score, sub_deadline
        )
        # Exponentiated‑gradient refinement on the segment weights
        if time.time() < deadline - 30.0:
            best_seq, best_score = _eg_segment_descent(
                m, n, best_seq, best_score, deadline - 30.0
            )
        print(f"[{time.time() - start_time:.2f}s] After segment m={m} best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Fine EG descent on the full‑resolution vector
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print(f"[{time.time() - start_time:.2f}s] Starting fine EG descent")
        best_seq, best_score = _eg_maxconv_descent(
            best_seq, start_time, deadline - 30.0, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After fine EG best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final hill‑climb / simulated‑annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")

    return [float(x) for x in best_seq]
