"""
Smoothed‐gradient descent optimizer.

The algorithm works in three stages:

1️⃣  A quick random‑parameter sweep over the 4‑parameter family
    a_i ∝ r**i·(i+1)**p·(n−i)**q mixed with a uniform background.
    The best vector from this sweep (or an up‑scaled previous solution)
    is taken as the starting point.

2️⃣  Annealed smooth‑max gradient descent.
    The objective 2·n·max(conv(a,a)) (with Σa=1) is approximated by a
    temperature‑controlled log‑sum‑exp of the convolution entries.
    The gradient of the smoothed objective can be computed analytically
    using two FFT‑based convolutions, after which a projected (simplex)
    momentum update is performed.  The temperature τ is exponentially
    cooled so that the optimisation converges to the true max‑norm
    objective.

3️⃣  A cheap random “mass‑transfer’’ hill‑climb that moves tiny amounts
    of probability mass between two coordinates.  This cleans up any
    remaining local irregularities.

The whole search respects the supplied time budget (default 1000 s) and
returns the best non‑negative step‑function found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of real vectors x and y via FFT of length L."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Self‑convolution a∗a (linear) using FFT. Returns length 2·len(a)‑1."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 for a (sum≈1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric‑power‑tail family)
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """Generate a normalised shape proportional to r**i·(i+1)**p·(n-i)**q."""
    i = np.arange(n, dtype=np.float64)
    # log‑space for stability
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)  # shift
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·uniform + (1‑lam)·shape (both already sum‑to‑1)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Random parametric sweep (geometric‑power‑tail family)
# ----------------------------------------------------------------------


def _random_param_search(n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")

    while time.time() < deadline:
        # r in (0.85, 0.999999) – log‑uniform
        r = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
        p = rng.uniform(-3.0, 3.0)
        q = rng.uniform(-3.0, 3.0)
        lam = rng.uniform(0.0, 1.0)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] random sweep n={n} "
                  f"r={r:.4f} p={p:.2f} q={q:.2f} λ={lam:.2f} → score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upscaling helper (splits each mass into two adjacent cells)
# ----------------------------------------------------------------------


def _upscale_by_two(seq: np.ndarray) -> np.ndarray:
    """Return a vector of length 2·len(seq) where each entry is split equally."""
    n = seq.shape[0]
    out = np.empty(2 * n, dtype=seq.dtype)
    out[0::2] = seq / 2.0
    out[1::2] = seq / 2.0
    return out


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (annealed)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on a smooth approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]

    L_self = _next_pow2(2 * n - 1)          # for a∗a
    L_full = _next_pow2(3 * n - 2)          # for w∗a_rev

    # Initialise temperature τ from the initial max value
    conv0 = _conv_self_fft(a, L_self)
    max0 = conv0.max()
    tau = max0 * 5.0                        # start rather “hot”
    tau_min = max0 * 5e-4
    tau_decay = 0.9997

    beta = 0.9                               # momentum
    vel = np.zeros_like(a)

    step0 = 0.6                              # initial step length
    it = 0

    best_a = a.copy()
    best_score = best_score_initial

    while time.time() < deadline:
        it += 1

        # --- convolution of a with itself (needed for both max and weights) ---
        conv = _conv_self_fft(a, L_self)
        max_val = conv.max()

        # --- temperature schedule ---
        tau = max(tau_min, tau * tau_decay)

        # --- soft‑max weights w_k ∝ exp((conv_k‑max)/τ) (stable) ---
        scaled = (conv - max_val) / tau
        exp_scaled = np.exp(scaled)          # max entry becomes 1
        Z = exp_scaled.sum()
        w = exp_scaled / Z                    # length 2n‑1, sums to 1

        # --- gradient of the smooth objective ---
        a_rev = a[::-1]
        conv_full = _fft_convolve(w, a_rev, L_full)   # length 3n‑2
        # extract slice corresponding to Σ_j w_{i+j} a_j
        g = conv_full[n - 1 : n - 1 + n]               # length n
        grad = 4.0 * n * g

        # --- momentum update ---
        vel = beta * vel + (1.0 - beta) * grad

        # --- decaying step size (poly‑decay) ---
        step = step0 / np.sqrt(it)

        a_candidate = a - step * vel
        a_candidate = _project_simplex(a_candidate)

        # evaluate the *real* objective (max‑norm)
        cur_score = _eval_fast(a_candidate, L_self)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            vel = np.zeros_like(a)   # reset momentum after improvement
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} "
                  f"score={cur_score:.6f} τ={tau:.2e}")
        else:
            # occasional uphill move (simulated‑annealing style)
            prob = np.exp(-(cur_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if np.random.rand() < prob:
                a = a_candidate

        # jitter every few thousand iterations to escape plateaus
        if it % 3000 == 0:
            eps = 1e-4 * np.random.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            vel = np.zeros_like(a)

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def _final_mass_transfer(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """Randomly move a tiny fraction of mass between two coordinates."""
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if dst == src:
            continue
        eps = a[src] * rng.random() * 2e-3      # up to 0.2 % transfer
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        sc = _eval_fast(a_prop, L_self)
        if sc < best_sc:
            best_sc = sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final transfer → score={sc:.6f}")
        else:
            if rng.random() < 0.001:            # tiny uphill tolerance
                a = a_prop

    return best_a, best_sc

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising
        2·n·max(conv(a,a)) / (sum(a)**2)

    The method consists of a fast random parametric sweep,
    an annealed smoothed‑max gradient descent, and a final
    random mass‑transfer refinement.  The best vector found
    (normalised to sum 1) is returned as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size – we aim for 8192 (2×4096) to allow a lower bound.
    # ------------------------------------------------------------------
    target_n = 8192

    # ------------------------------------------------------------------
    # Load previous construction if available; otherwise start from uniform.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        base = np.maximum(base, 0.0)
        if base.sum() == 0.0:
            raise ValueError
        base = base / base.sum()
    except Exception:
        base = None

    if base is not None and base.size < target_n:
        # Upscale by simple splitting until we reach target_n (power‑of‑two factor)
        cur = base
        while cur.size < target_n:
            cur = _upscale_by_two(cur)
        init_seq = cur[:target_n]
    else:
        # If we already have a sequence of the right size use it,
        # otherwise fallback to uniform.
        if base is not None and base.size == target_n:
            init_seq = base.copy()
        else:
            init_seq = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 0️⃣  Baseline score
    # ------------------------------------------------------------------
    L_self = _next_pow2(2 * target_n - 1)
    best_seq = init_seq.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] baseline (n={target_n}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Quick random parametric sweep (≈150 s)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + min(150.0, deadline - start_time - 600.0)
    if time.time() < sweep_deadline:
        rp_seq, rp_score = _random_param_search(target_n, sweep_deadline, start_time)
        if rp_seq is not None and rp_score < best_score:
            best_seq = rp_seq
            best_score = rp_score
            print(f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Annealed smoothed‑max gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + min(400.0, remaining * 0.6)
    if time.time() < gd_deadline:
        a_gd, sc_gd = _annealed_gradient_descent(best_seq, gd_deadline,
                                                 start_time, best_score)
        if sc_gd < best_score:
            best_seq = a_gd
            best_score = sc_gd
            print(f"[{time.time() - start_time:.2f}s] after GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final tiny random mass‑transfer hill‑climb
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_final, sc_final = _final_mass_transfer(best_seq, deadline, start_time, best_score)
        if sc_final < best_score:
            best_seq = a_final
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (ensure sum = 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Official evaluation with the protected routine.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
