def _adam_smoothing_descent(a0, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    step0 = 0.2
    gamma = 2.0
    gamma_factor = 1.001 # increase each iteration
    t=0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        # compute softmax weights
        # to improve numerical stability:
        exp_vals = np.exp(gamma * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # compute gradient using convolution with reversed a
        a_rev = a[::-1]
        # using _conv_fft for w and a_rev
        grad_conv = _conv_fft(w, a_rev)
        # slice grad
        grad = 2.0 * grad_conv[n-1:n-1+n]  # length n
        # adapt gamma
        gamma = min(100.0, gamma * gamma_factor)
        # Adam update
        t += 1
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 / np.sqrt(t)  # decreasing
        delta = step * m_hat / (np.sqrt(v_hat) + eps)
        a = a - delta
        a = _project_simplex(np.maximum(a, 0.0))
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth it={t} gamma={gamma:.2f} score={cur_score:.6f}")
    return best_seq, best_score

    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.5
    gamma = 1.0
    gamma_inc = 1.01
    t = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        # softmax weights
        exp_vals = np.exp(gamma * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # compute gradient via convolution with reversed a
        a_rev = a[::-1]

        grad = 2.0 * grad_conv[n-1:n-1+n]  # slice length n

"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages (all respecting a 1000 s budget)

1.  A quick parametric sweep – in particular an *asymmetric two‑block*
    family that was not explored before.
2.  Many random Dirichlet draws (different concentration parameters).
3.  Projected Adam optimisation on a *soft‑max* surrogate of the
    max‑convolution (the surrogate becomes the true max as the
    temperature grows).
4.  A stochastic hill‑climb / simulated‑annealing phase that moves mass
    between coordinates while always staying on the probability simplex.

All stages keep track of the best vector seen so far and finally return
the normalised list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length = 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Probability‑simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is held equal to 1, so denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    Returned gradient has the same shape as ``a``.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Asymmetric two‑block sweep
# ----------------------------------------------------------------------
def _sweep_asymmetric_two_blocks(n, best_seq, best_score, deadline):
    """
    Two uniform blocks, possibly of different length, and an arbitrary split
    of the total mass between them.
    """
    L1_grid = np.linspace(0.01, 0.4, 20)   # left block size as fraction of n
    L2_grid = np.linspace(0.01, 0.4, 20)   # right block size as fraction of n
    w_grid = np.linspace(0.1, 0.9, 9)      # fraction of total mass placed on the left block
    for L1_frac in L1_grid:
        L1 = int(L1_frac * n)
        if L1 < 1:
            continue
        for L2_frac in L2_grid:
            L2 = int(L2_frac * n)
            if L2 < 1:
                continue
            for w in w_grid:
                if time.time() >= deadline:
                    return best_seq, best_score
                seq = np.zeros(n)
                seq[:L1] = w / L1
                seq[-L2:] = (1.0 - w) / L2
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] Asym2Block L1={L1} L2={L2} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 500
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)          # concentration roughly 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam descent on a soft‑max surrogate of max‑conv
# ----------------------------------------------------------------------
def _adam_softmax_descent(a0, best_seq, best_score, deadline):
    """
    Projected Adam on the smooth objective
        S_γ(a) = (1/γ)·log Σ_k exp(γ·conv_k(a))
    with γ slowly increased.  The gradient is obtained by a convolution,
    hence each iteration costs only a few FFTs.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.5
    gamma = 1.0
    gamma_mul = 1.01               # geometric increase of γ
    gamma_max = 200.0
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)               # full self‑convolution
        max_val = conv.max()
        # soft‑max weights (probabilities)
        exp_vals = np.exp(gamma * (conv - max_val))
        w = exp_vals / exp_vals.sum()      # shape (2·n‑1,)
        # gradient of the smooth objective:
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)    # length = 3·n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]  # keep only the n relevant entries
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / np.sqrt(it)          # simple decay
        delta = step * m_hat / (np.sqrt(v_hat) + eps)
        a = a - delta
        a = _project_simplex(np.maximum(a, 0.0))
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} γ={gamma:.2f} score={cur_score:.6f}")
        # increase the temperature (sharpen the soft‑max)
        gamma = min(gamma_max, gamma * gamma_mul)
    return best_seq, best_score

# 4️⃣  Stochastic hill‑climb / simulated annealing
# ----------------------------------------------------------------------
def _hill_climb(start_a, best_seq, best_score, deadline):
    a = start_a.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    T = 1e-3               # initial “temperature”
    T_min = 1e-7
    decay = 0.9995
    no_improve = 0

    while time.time() < deadline:
        # choose a source with non‑negligible mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[rng.integers(src_candidates.size)]
        dst = rng.integers(n)
        if dst == src:
            continue

        max_frac = 0.07                     # up to 7 % per move
        frac = rng.random() * max_frac
        delta = a[src] * frac
        if delta <= 0.0:
            continue
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        # keep on the simplex (the move already preserves the sum)
        a_prop = _project_simplex(np.maximum(a_prop, 0.0))

        cur_score = _objective_score(a_prop)
        if cur_score < best_score or rng.random() < np.exp(-(cur_score - best_score) / max(T, 1e-12)):
            a = a_prop
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time() - start_time:.2f}s] HillClimb score={best_score:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # anneal temperature
        T = max(T_min, T * decay)

        # occasional random perturbation when stuck
        if no_improve > 4000:
            i, j = rng.choice(n, size=2, replace=False)
            eps = a[i] * (rng.random() * max_frac + 1e-5)
            a[i] -= eps
            a[j] += eps
            a = _project_simplex(np.maximum(a, 0.0))
            no_improve = 0
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine consists of:

        1. asymmetric two‑block sweeps,
        2. random Dirichlet draws,
        3. a soft‑max Adam optimiser,
        4. a final stochastic hill‑climb.

    The best vector (normalised to sum = 1) is returned as a Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (if a previous construction is supplied)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Asymmetric two‑block sweep (fast, cheap)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_two_blocks(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # 3️⃣  Adam optimiser on smooth max‑conv surrogate
    # ------------------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam soft‑max descent")
        best_seq, best_score = _adam_softmax_descent(
            best_seq, best_seq, best_score, phase_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final stochastic hill‑climb / simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, best_seq, best_score, deadline
        )
        print(f"[{time.time() - start_time:.2f}s] Final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
