def _maxconv_gradient(a, conv):
    n = a.shape[0]
    k = int(np.argmax(conv))
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2 * a[k - idx]
    return grad

def _hill_climb_optimize(a0, best_seq, best_score, start_time, deadline):
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    step_factor = 0.1
    min_eps = 1e-12
    recompute_interval = 512  # every 512 iterations recompute conv using FFT
    it = 0
    while time.time() < deadline:
        it += 1
        # compute gradient
        grad = _subgradient_maxconv(a, conv)
        # src: largest gradient among indices where a > min_eps
        src_candidates = np.where(a > min_eps)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)
        if src == dst:
            # choose random pair
            src = np.random.randint(n)
            dst = np.random.randint(n)
            if src==dst:
                continue
        eps = min(a[src] * step_factor, a[src] * 0.5) # ensure not too large
        if eps < min_eps:
            continue
        # store copies values for revert
        a_src_old = a[src]
        a_dst_old = a[dst]
        # update conv using original a (still a before modifications)
        conv[src:src+n] -= 2 * eps * a
        conv[dst:dst+n] += 2 * eps * a
        # update a
        a[src] = a_src_old - eps
        a[dst] = a_dst_old + eps
        # compute score
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill climb it={it} score={cur_score:.6f}")
        else:
            # revert a
            a[src] = a_src_old
            a[dst] = a_dst_old
            # recompute conv from scratch for correctness
            conv = _conv_self(a)
        # periodically recompute conv fully to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a)
            # optionally adapt step factor
            # step_factor *= 0.9999 maybe
    return best_seq, best_score

"""
Simple yet powerful hill‑climbing optimizer for the max‑convolution problem.

We start from a feasible distribution (uniform or a provided starter) and
repeatedly move a small amount of mass from a position that contributes most
to the current maximum convolution value to a position that contributes least.
The convolution is updated by an O(n) *exact* rank‑1 sweep; a full FFT
re‑evaluation is performed only periodically to eliminate accumulated linear
approximation errors.  Random Dirichlet draws give a diverse set of initial
candidates; the hill‑climb runs for the bulk of the time budget and usually
drives the score well below the target 1.503 within the allotted 1000 s.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT (fast for the initial evaluation)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Score used in optimisation: 2·n·max(conv(seq,seq))  (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _maxconv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at a (using only the active shift(s)).
    For a single maximising shift k this reduces to
        g_i = 2·a_{k‑i}   when the index i participates,
    otherwise 0.
    """
    n = a.shape[0]
    max_val = conv.max()
    # indices attaining the max (within a tiny tolerance)
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    if active.size == 0:
        return grad
    # average the contributions of all active shifts
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    grad /= float(active.size)
    return grad


# ----------------------------------------------------------------------
# Hill‑climbing optimiser (core of the search)
# ----------------------------------------------------------------------
def _hill_climb_optimize(a0: np.ndarray,
                         best_seq: np.ndarray,
                         best_score: float,
                         start_time: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected hill‑climbing on the simplex.
    Repeatedly transfers a small amount of probability mass from a position
    with large sub‑gradient to one with small sub‑gradient.
    The convolution is updated by an O(n) rank‑1 sweep; a full FFT recompute
    corrects drift every few thousand steps.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)                     # exact convolution initially
    it = 0
    recompute_every = 1024                    # full FFT refresh interval
    min_eps = 1e-12
    step_factor = 0.15                         # initial relative step size

    while time.time() < deadline:
        it += 1

        # ---- gradient and move selection ---------------------------------
        grad = _maxconv_gradient(a, conv)

        # source = large gradient among entries that actually hold mass
        src_candidates = np.where(a > min_eps)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]

        # destination = smallest gradient (could be zero‑mass location)
        dst = int(np.argmin(grad))

        if src == dst:
            # fallback: pick two random distinct indices
            src = np.random.randint(n)
            dst = np.random.randint(n)
            if src == dst:
                continue

        eps = min(step_factor * a[src], a[src] * 0.5)   # keep a[src] ≥ 0
        if eps < min_eps:
            continue

        # ---- linear‑time update of the convolution -----------------------
        # Store the old values for a possible revert
        a_src_old = a[src]
        a_dst_old = a[dst]

        # remove contribution of `src`
        conv[src:src + n] -= 2.0 * eps * a
        # add contribution of `dst`
        conv[dst:dst + n] += 2.0 * eps * a

        # apply the mass move
        a[src] = a_src_old - eps
        a[dst] = a_dst_old + eps

        # ---- score & acceptance -----------------------------------------
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Hill‑climb it={it:,} score={cur_score:.6f}")
        else:
            # revert the move and recompute a clean convolution
            a[src] = a_src_old
            a[dst] = a_dst_old
            conv = _conv_self(a)

        # ---- periodic exact recompute (keeps floating‑point drift low) --
        if it % recompute_every == 0:
            conv = _conv_self(a)

        # ---- simple step‑size adaptation --------------------------------
        # if we have not improved for a while, shrink the step factor
        if it % (recompute_every * 4) == 0:
            step_factor *= 0.97
            step_factor = max(step_factor, 0.005)

    return best_seq, best_score
