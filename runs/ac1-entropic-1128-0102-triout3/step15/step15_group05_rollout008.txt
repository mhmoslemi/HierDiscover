"""
Propose a sequence of non‑negative heights that (hopefully) beats the previous
best bound (≈ 1.5079).  

The algorithm works in three stages:

1️⃣  **Multiple random restarts** – a few dozen initial vectors are drawn
    (Dirichlet with a small concentration, the historic best sequence,
    and a simple geometric family).  

2️⃣  **Soft‑max Adam optimisation** – we minimise the maximal entry of the
    self‑convolution `a*a`.  Instead of the non‑smooth max we use a
    temperature‑controlled soft‑max
        Mα(a) = (1/α)·log Σ exp(α·convk)
    whose gradient can be written as a weighted convolution and is obtained
    with two FFTs per iteration.  Adam (with projection onto the probability
    simplex) provides a fast, stable descent.

3️⃣  **Final tiny mass‑transfer hill‑climb** – after the Adam phase we
    perform a cheap random “move ε mass from i to j” local search to
    polish the result.

All work respects the global 1000 s budget.  The best vector found is
returned as a plain Python list of floats (already normalised to sum = 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries (real values).
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv.real


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear convolution of a and b using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[:n]
    return conv.real


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # shouldn't happen, but safe fallback
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast evaluation of the objective for ∑a = 1 (exact formula)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    # sum(a) is forced to 1 by the caller
    n = a.shape[0]
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Adam optimiser for the soft‑max of the convolution
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    *,
    alpha: float = 40.0,
    lr: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Returns the best vector seen and its (exact) score.
    """
    n = a0.shape[0]
    L_self = _next_pow2(2 * n - 1)          # for a*a
    L_joint = _next_pow2(3 * n - 2)         # for w_rev * a
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = best_score_initial

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # ----- convolution of a with itself (needed for both score & weights) -----
        conv = _conv_self_fft(a, L_self)            # length 2n-1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv                # exact because Σa=1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={t:,} "
                f"score={cur_score:.6f}"
            )

        # ----- soft‑max weights (smooth approximation of the max) -----
        # w_k ∝ exp(alpha * (conv_k - max_conv))
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient of the weighted sum  g = 2 * (w ⋆ a) -----
        # correlation w⋆a can be obtained as conv of rev(w) with a
        w_rev = w[::-1]
        conv_rev = _conv_fft(w_rev, a, L_joint)          # length 3n-2
        start = conv_rev.shape[0] - (n + conv_rev.shape[0] - (len(w)-1) - n)  # not needed
        # more directly:
        offset = len(w) - 1                                 # = 2n-2
        g = 2.0 * conv_rev[offset: offset + n]

        # ----- Adam update (projected on simplex) -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)          # keep non‑negative and Σ=1

        # ----- occasional temperature increase to help escape plateaus -----
        if t % 800 == 0:
            alpha = min(200.0, alpha * 1.15)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny random mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def _final_local_search(
    a: np.ndarray,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    max_iters: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap local search that moves a tiny fraction of mass between two
    coordinates.  Returns an improved vector if found.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if dst == src:
            continue
        eps = a[src] * rng.random() * 2e-3            # ≤0.2 %
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        sc = _eval_fast(a_prop, L_fft)
        if sc < best_sc:
            best_sc = sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={sc:.6f}")
        elif rng.random() < 0.001:                     # rare uphill step
            a = a_prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper to generate a simple geometric‑power family (used for a few restarts)
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """Family a_i ∝ r**i · (i+1)**p · (n‑i)**q, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)                     # avoid overflow
    shape = np.exp(log_shape)
    if not np.isfinite(shape).all() or shape.sum() == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·U + (1‑lam)·shape, then renormalise."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed
