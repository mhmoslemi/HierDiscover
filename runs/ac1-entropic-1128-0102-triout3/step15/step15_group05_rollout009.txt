"""
Propose a non‑negative step function (vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three phases within the 1000 s budget:

1️⃣  Fast random search in a 3‑parameter “beta‑family”
    a_i ∝ (i+1)^α (n‑i)^β  mixed with a uniform background.
    This quickly finds a good starting shape.

2️⃣  Long projected sub‑gradient descent on the true objective.
    The sub‑gradient is the sum of contributions of all (almost) maximal
    convolution entries, and a slow decaying step size together with occasional
    uphill moves allows the method to escape plateaus and improve the score
    beyond the previous 1.507878 target.

3️⃣  A cheap final “mass‑transfer” hill‑climb that moves a tiny fraction of
    probability mass between two coordinates, accepting improving moves
    and occasionally uphill moves to polish the solution.

All vectors are kept on the probability simplex (non‑negative and sum = 1)
by Euclidean projection.  The convolution is evaluated by FFT for speed.
The best vector found is returned as a plain list of floats.
"""

import time
import numpy as np
from typing import Tuple

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (beta‑family)
# ----------------------------------------------------------------------


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components
    proportional to (i+1)^α * (n‑i)^β   (i = 0…n‑1).
    Computed in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    # log‑space
    log_shape = alpha * np.log(i + 1.0) + beta * np.log(np.maximum(n - i, 1.0))
    log_shape -= np.max(log_shape)          # avoid overflow
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return lam·uniform + (1‑lam)·shape  (both already sum to 1).
    The result is renormalised to guard against numerical drift.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient of the maximum convolution entry
# ----------------------------------------------------------------------


def _compute_subgrad_multi(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a, aggregating over all shifts
    whose value is within `tol` of the maximum.
    """
    max_val = conv.max()
    # entries that are (almost) maximal
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)

    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        # contribution a_j to ∂/∂a_i
        g[i_vals] += a[j_vals]

        # symmetric contribution for i≠j (adds a_i to ∂/∂a_j)
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (slow decay, long run)
# ----------------------------------------------------------------------


def _subgradient_descent_slow(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              best_score_initial: float) -> Tuple[np.ndarray, float]:
    """
    Long projected sub‑gradient descent directly on the true objective,
    using a slowly decaying step size and occasional uphill jumps.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    step0 = 0.30               # initial step size (additive)
    decay_exp = 0.35           # exponent for step decay
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        max_conv = conv.max()
        # tolerance for active set (relative + absolute guard)
        tol = max(1e-12, max_conv * 1e-12)
        g = _compute_subgrad_multi(a, conv, tol=tol)

        # step size with slow decay
        step = step0 / (it ** decay_exp + 1.0)

        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)
        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional uphill acceptance (simulated annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter if we have been stagnant for >30 s
        if time.time() - last_improve > 30.0:
            eps = 5e-4 * rng.random(n) * a  # very small random perturbation
            a = np.maximum(a + eps, 0.0)

            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny random mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def _final_local_search(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> Tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two
    coordinates, keeping the vector on the simplex.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.integers(n)
        if best_a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_a[src] * rng.random() * 2e-3   # up to 0.2 % of source mass
        a_prop = best_a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            print(f"[{time.time() - start_time:.2f}s] final local it={it:,} "
                  f"score={new_sc:.6f}")
        else:
            # rare uphill acceptance to keep some mobility
            if rng.random() < 0.001:
                best_a = a_prop
    return best_a, best_sc


# ----------------------------------------------------------------------
# Random beta‑family search
# ----------------------------------------------------------------------


def _random_beta_search(n: int,
                        deadline: float,
                        start_time: float) -> Tuple[np.ndarray,
                                                    float,
                                                    Tuple[float, float, float]]:
    """
    Uniform‑in‑log random sampling of (α,β,λ) for the beta‑family and keep the best.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        alpha = rng.uniform(-3.0, 4.0)      # allow negative exponents for heavy tails
