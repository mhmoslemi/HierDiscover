Optimise a non‑negative step‑function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The search is performed in four phases (all respecting a 1000 s budget):

1️⃣  Broad parametric sweep (geometric, power, beta, stretched‑exp,
    logistic, sine‑power, bimodal‑exp, reciprocal, Gaussian).
2️⃣  Massive random Dirichlet sampling (different concentration values).
3️⃣  Multi‑start exponentiated‑gradient (EG) descent on the exact L∞‑norm
    of the self‑convolution.  Short EG runs are launched from many random
    starts; the best result is kept.
4️⃣  A temperature‑controlled hill‑climb / simulated‑annealing
    refinement that moves tiny amounts of mass from “bad’’ indices
    (high gradient) to “good’’ ones.

All sequences are kept on the probability simplex (sum = 1) and are
projected onto the non‑negative orthant after each update.  The routine
returns a plain Python list of floats.  If the time limit is exceeded the

"""

# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    For all shifts attaining the maximum the contributions are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _score_normalised(seq: np.ndarray, L_fft: int) -> float:
    """
    Score for a probability vector (sum = 1):
        2·n·max(conv(seq,seq))
    """
    n = seq.shape[0]
    conv = _conv_self(seq, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Parametric families
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray, best_score: float,
                    deadline: float, start_time: float, L_fft: int):
    """Explore a handful of analytic families."""
    idx = np.arange(n, dtype=np.float64)

    # Uniform (baseline)
    uniform = np.full(n, 1.0 / n)
    sc = _score_normalised(uniform, L_fft)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform    score={sc:.6f}")

    # --------------------------------------------------------------
    # geometric
    # --------------------------------------------------------------
    eps_grid = np.logspace(-5, -1, 150)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 7):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _score_normalised(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # power
    # --------------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.0, 81)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        pow_seq /= pow_seq.sum()
        sc = _score_normalised(pow_seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # beta‑type (symmetric)
    # --------------------------------------------------------------
    alphas = np.linspace(0.2, 3.0, 25)
    betas = np.linspace(0.2, 3.0, 25)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            seq /= seq.sum()
            sc = _score_normalised(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # stretched exponential
    # --------------------------------------------------------------
    scales = np.linspace(0.2, 1.0, 9)
    p_vals = np.linspace(0.3, 4.0, 31)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            seq /= seq.sum()
            sc = _score_normalised(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretch p={p:.2f} s={scale:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # logistic
    # --------------------------------------------------------------
    mu_factors = np.linspace(0.2, 0.8, 7)
    s_factors = np.linspace(0.02, 0.15, 7)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _score_normalised(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic μ={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # sine‑power
    # --------------------------------------------------------------
    p_vals = np.linspace(0.1, 5.0, 31)
    for p in p_vals:
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        seq /= seq.sum()
        sc = _score_normalised(seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sin‑pow p={p:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # Bimodal exponential (left + right tails)
    # --------------------------------------------------------------
    for a_left in [0.3, 0.6, 1.0, 1.5]:
        for a_right in [0.3, 0.6, 1.0, 1.5]:
            left = np.exp(-a_left * idx / n)
            right = np.exp(-a_right * (n - 1 - idx) / n)
            seq = left + right
            seq /= seq.sum()
            sc = _score_normalised(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Bimodal aL={a_left:.2f} aR={a_right:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # Reciprocal shape (symmetric 1/(i+1) + 1/(n-i))
    # --------------------------------------------------------------
    rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
    rec /= rec.sum()
    sc = _score_normalised(rec, L_fft)
    if sc < best_score:
        best_score, best_seq = sc, rec.copy()
        print(f"[{time.time() - start_time:.2f}s] Reciprocal shape score={sc:.6f}")

    # --------------------------------------------------------------
    # Gaussian bumps
    # --------------------------------------------------------------
    for mu_f in np.linspace(0.2, 0.8, 5):
        for sigma_f in np.linspace(0.04, 0.12, 5):
            mu = mu_f * (n - 1)
            sigma = sigma_f * n
            seq = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _score_normalised(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Gaussian μ={mu_f:.2f} σ={sigma_f:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet trials
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float,
                             start_time: float, L_fft: int,
                             trials: int = 5000):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(np.log(0.01), np.log(30.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _score_normalised(a, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _eg_descent(a0: np.ndarray,
                max_iters: int,
                deadline: float,
                start_time: float,
                best_seq: np.ndarray,
                best_score: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex with the exact max‑convolution sub‑gradient.
    Runs for at most `max_iters` iterations or until `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score_normalised(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # time‑decaying step size
        eta = 0.08 / (1.0 + it * 1e-4)

        exp_factor = np.exp(np.clip(-eta * grad, -700, 700))
        a = a * exp_factor
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        new_score = _score_normalised(a, L_fft)

        if new_score < cur_score - 1e-13:
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG it={it:,}  score={cur_score:.6f}")

        # occasional random shake to escape very flat plateaus
        if np.random.rand() < 0.001:
            a = a0.copy()
            cur_score = _score_normalised(a, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float,
                      L_fft: int) -> tuple[np.ndarray, float]:
    """
    Small Monte‑Carlo moves guided by the gradient of the max‑convolution,
    accepted with a temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = _score_normalised(a, L_fft)

    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score

    T0 = 0.04               # initial temperature
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        # gradient of max‑convolution tells where the current peak lives
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # move a modest random fraction of src to dst
        frac = rng.uniform(1e-6, 0.02)  # up to 2 %
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = _score_normalised(a_prop, L_fft)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        if Δ < -1e-13 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3            # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise with uniform (always a valid baseline)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _score_normalised(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter is present, try it
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _score_normalised(starter, L_fft)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] External starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0                # reserve ~5 min for the remaining phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline, start_time, L_fft
        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet sampling (intensive)
    # --------------------------------------------------------------
    phase_deadline = deadline - 150.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline,
            start_time, L_fft, trials=8000
        )

    # --------------------------------------------------------------
    # 3️⃣  Multi‑start EG descent
    # --------------------------------------------------------------
    # keep a small pool of promising seeds (including the current best)
    seed_pool = [(best_seq.copy(), best_score)]
    # generate a few extra random seeds for diversity
    rng = np.random.default_rng()
    for _ in range(12):
        if time.time() >= deadline - 120.0:   # keep some budget for hill‑climb
            break
        c = np.exp(rng.uniform(np.log(0.01), np.log(30.0)))
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _score_normalised(a, L_fft)
        seed_pool.append((a, sc))

    # Run short EG runs from each seed (≈1500 iterations each)
    for seq, sc in seed_pool:
        if time.time() >= deadline - 120.0:
            break
        max_iters = 1800
        best_seq, best_score = _eg_descent(
            seq, max_iters, deadline - 120.0,
            start_time, best_seq, best_score, L_fft
        )

    # --------------------------------------------------------------
    # 4️⃣  Longer EG run from the current best
    # --------------------------------------------------------------
    if time.time() < deadline - 60.0:
        remaining_iters = 8000
        best_seq, best_score = _eg_descent(
            best_seq, remaining_iters,
            deadline - 60.0, start_time,
            best_seq, best_score, L_fft
        )
        print(f"[{time.time() - start_time:.2f}s] After long EG best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final hill‑climb / simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline,
            best_seq, best_score, L_fft
        )
        print(f"[{time.time() - start_time:.2f}s] Final best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
