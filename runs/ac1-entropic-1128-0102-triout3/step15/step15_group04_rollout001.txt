    x = (np.arange(n) + 0.5) / n  # midpoint of intervals

"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------
1️⃣  **Broad parametric sweep** –  explores two families:
    * geometric × power (as in the reference)
    * Beta‑distribution shapes (U‑shaped and bell‑shaped)  
    All candidates are mixed with a uniform component and projected onto the
    probability simplex.  The best candidate of this cheap search seeds the
    next phase.

2️⃣  **Continuation on a smooth max** –  for a sequence of temperatures
    `β` we minimise a soft‑max approximation of `max(conv)`.  The gradient of
    the smooth objective is obtained analytically via FFT and an Adam‑style
    update is applied, followed by Euclidean projection onto the simplex.

3️⃣  **Exact sub‑gradient polish** –  classic projected sub‑gradient descent
    on the true `L∞` objective (identical to the reference implementation)
    runs for the remaining time budget.

4️⃣  **Tiny random mass‑transfer hill‑climb** –  final seconds are spent
    moving a tiny fraction of mass between random coordinates; any improvement
    is kept.

The routine respects the 1000 s budget, prints progress and finally returns
the best vector as a plain Python list of floats (normalised to sum = 1).
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, not used in the final run

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)         # avoid overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Beta‑distribution shape on the grid {0,…,n‑1},
    using the mid‑point rule to avoid zeros.
    Returns a vector normalised to sum = 1.
    """
    # mid‑points of the n bins
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid infinities for extreme parameters
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    shape = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Linear interpolation with the uniform distribution."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights p_k = exp(beta*conv_k)/Σexp(.) safely."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max objective:
        g_i = 2 * Σ_j a_j * p_{i+j}
    Computed via FFT as a cross‑correlation.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    conv_full = np.fft.irfft(Af * Pf, n=L_grad)
    # keep the middle n entries (corresponds to valid shifts)
    start = n - 1
    g = conv_full[start:start + n][::-1]
    return 2.0 * g


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   beta: float,
                   L_fft: int,
                   L_grad: int,
                   deadline: float,
                   start_time: float,
                   best_score: float,
                   best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the soft‑max approximation of max(conv).
    Returns the best vector seen during this run and its objective value.
    """
    n = a0.shape[0]
    a = a0.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12                    # base learning rate
    it = 0
    pow1 = 1.0
    pow2 = 1.0

    while time.time() < deadline:
        it += 1
        # soft‑max of the current self‑convolution
        conv = _conv_self_fft(a, L_fft)
        p = _softmax_weights(conv, beta)

        # gradient of the smooth max
        g = _grad_softmax(a, p, L_grad)

        # Adam updates
        pow1 *= beta1
        pow2 *= beta2
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - pow1)
        v_hat = v / (1.0 - pow2)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        cand_score = _eval_fast(a_candidate, L_fft)

        # accept if we improve the *current* score, otherwise random uphill
        if cand_score < best_score:
            best_score = cand_score
            best_seq = a_candidate.copy()
            print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} it={it:,} "
                  f"new best {cand_score:.6f}")
        if cand_score < _eval_fast(a, L_fft):
            a = a_candidate
        else:
            # tiny probability to accept uphill moves (simulated annealing flavour)
            prob = np.exp(-(cand_score - _eval_fast(a, L_fft)) *
                          4.0 / (np.sqrt(it) + 1.0))
            if np.random.rand() < prob:
                a = a_candidate

        # small jitter every few thousand iterations to escape plateaus
        if it % 4000 == 0:
            eps_jit = 1e-4 * np.random.rand(n) * a
            a = _project_simplex(a + eps_jit)

    return best_seq, best_score

# Exact sub‑gradient descent (same as reference, slightly tuned)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float,
                         best_seq_initial: np.ndarray) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true L∞ objective."""
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = best_seq_initial.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient for the index k (exact)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        step = step0 / (it ** 0.6)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional uphill moves
            prob = np.exp(-(cur_score - best_score) * 12.0 /
                          (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = _project_simplex(np.maximum(a + eps, 0.0))

    return best_a, best_score

    if time.time() < sub_deadline:
        a_sub, sc_sub = _subgradient_descent(best_seq, sub_deadline,
                                             start_time, best_score,
                                             best_seq)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient final score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Tiny random hill‑climb (mass transfers) – last few seconds
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer at most 0.2 % of the source mass
        eps = best_seq[src] * rng.random() * 2e-3
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
