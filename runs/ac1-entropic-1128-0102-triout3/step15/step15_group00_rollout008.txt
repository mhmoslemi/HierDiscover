"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three phases:

1️⃣  Fast parametric sweeps (geometric, power, two‑block) to obtain a decent
    start vector (normalised to sum = 1).

2️⃣  A long simulated‑annealing / greedy hill‑climb that works on the full
    vector.  Each move transfers a small fraction of mass from a “high‑gradient”
    index to a “low‑gradient” one.  The self‑convolution is maintained
    incrementally (O(n) per move) and occasionally recomputed from scratch
    to avoid drift.

3️⃣  The best vector encountered is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (sum is held equal to 1, so denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(conv: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv) for a normalised vector."""
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    Returns a vector of size n.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Simple parametric families (fast, few grid points)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -1, 40)          # reduced grid
    for eps in eps_grid:
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        for lam in (0.0, 0.4, 0.8, 1.0):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            conv = _conv_self(seq)
            sc = _objective_score(conv, n)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        seq = np.power(idx + 1.0, p)
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq /= seq_sum
        conv = _conv_self(seq)
        sc = _objective_score(conv, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = max(1, int(L_frac * n))
        for w in (0.2, 0.5, 0.8):
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            conv = _conv_self(seq)
            sc = _objective_score(conv, n)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
    return best_seq, best_score


def _random_dirichlet_trials(n, best_seq, best_score, trials=200):
    rng = np.random.default_rng()
    for _ in range(trials):
        logc = rng.uniform(-3.0, 1.0)      # concentration ~0.001 … 10
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        conv = _conv_self(a)
        sc = _objective_score(conv, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Incremental move utilities (mass transfer)
# ----------------------------------------------------------------------
def _apply_move_incremental(a_old: np.ndarray,
                            conv: np.ndarray,
                            src: int,
                            dst: int,
                            delta: float) -> None:
    """
    Apply the effect of moving `delta` mass from src → dst on
    the self‑convolution `conv`.  Operates in‑place.
    """
    n = a_old.shape[0]
    # contribution from a*Δ (twice) – see analysis above
    conv[dst:dst + n] += 2.0 * delta * a_old
    conv[src:src + n] -= 2.0 * delta * a_old
    # Δ*Δ term
    conv[2 * src]   += delta * delta
    conv[2 * dst]   += delta * delta
    conv[src + dst] -= 2.0 * delta * delta


def _revert_move_incremental(a_old: np.ndarray,
                            conv: np.ndarray,
                            src: int,
                            dst: int,
                            delta: float) -> None:
    """Undo the effect of `_apply_move_incremental`."""
    n = a_old.shape[0]
    conv[dst:dst + n] -= 2.0 * delta * a_old
    conv[src:src + n] += 2.0 * delta * a_old
    conv[2 * src]   -= delta * delta
    conv[2 * dst]   -= delta * delta
    conv[src + dst] += 2.0 * delta * delta


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm performs a modest parametric sweep,
    a few random Dirichlet starts and then a simulated‑annealing style hill‑climb
    with O(n) incremental convolution updates.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    conv = _conv_self(best_seq)
    best_score = _objective_score(conv, n)

    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Light parametric sweep
    # --------------------------------------------------------------
    best_seq, best_score = _sweep_geometric(n, best_seq, best_score)
    best_seq, best_score = _sweep_power(n, best_seq, best_score)
    best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score)

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet trials (few)
    # --------------------------------------------------------------
    best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, trials=150)

    # --------------------------------------------------------------
    # Initialise the main SA / hill‑climb
    # --------------------------------------------------------------
    a = best_seq.copy()               # current solution (sum = 1)
    conv = _conv_self(a)              # its full self‑convolution
    cur_score = _objective_score(conv, n)

    # Keep the record of the best ever seen (might be the same as `best_seq`)
    best_ever = a.copy()
    best_ever_score = cur_score

    # Simulated‑annealing parameters
    T0 = max(0.001, cur_score * 0.02)      # initial temperature (relative)
    max_frac = 0.15                         # max fraction of source mass to move
    min_frac = 5e-5                         # minimum fraction (to keep steps non‑zero)
    recompute_interval = 5000               # full FFT recompute period
    steps = 0

    # --------------------------------------------------------------
    # Helper for choosing source / destination
    # --------------------------------------------------------------
    rng = np.random.default_rng()

    while time.time() < deadline:
        steps += 1

        # Occasionally (5 % of the time) ignore the gradient and pick random indices
        if rng.random() < 0.05:
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)
        else:
            # Gradient‑guided move
            grad = _subgradient_maxconv(a, conv)
            src = int(np.argmax(grad))          # where the max‑conv is most sensitive
            dst = int(np.argmin(grad))          # where we can add mass safely
            if src == dst:
                # fallback to random
                src = rng.integers(0, n)
                dst = rng.integers(0, n)
                while dst == src:
                    dst = rng.integers(0, n)

        # Do not move from a nearly empty cell
        if a[src] < 1e-12:
            continue

        # Choose transfer fraction
        frac = rng.random() * (max_frac - min_frac) + min_frac
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        # ------------------------------------------------------------------
        # Incremental update (with the ability to revert)
        # ------------------------------------------------------------------
        a_old = a.copy()                     # needed for the convolution update
        # apply the move to a
        a[src] -= delta
        a[dst] += delta

        # update convolution incrementally
        _apply_move_incremental(a_old, conv, src, dst, delta)

        # recompute score from the updated convolution
        new_score = _objective_score(conv, n)

        # --------------------------------------------------------------
        # Acceptance test (simulated annealing)
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        T = max(0.0, T0 * (1.0 - elapsed / (deadline - start_time)))  # linear cooling

        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            # Probabilistic uphill move
            prob = np.exp(-(new_score - cur_score) / (T + 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if cur_score < best_ever_score:
                best_ever_score = cur_score
                best_ever = a.copy()
                print(f"[{elapsed:.2f}s] New best score = {best_ever_score:.6f}")
        else:
            # revert both a and conv
            a[:] = a_old
            _revert_move_incremental(a_old, conv, src, dst, delta)

        # Occasionally re‑compute the convolution from scratch to avoid drift
        if steps % recompute_interval == 0:
            conv = _conv_self(a)
            cur_score = _objective_score(conv, n)

    # --------------------------------------------------------------
    # Finalise and return the best sequence found
    # --------------------------------------------------------------
    best_ever = np.maximum(best_ever, 0.0)
    total = best_ever.sum()
    if total > 0:
        best_ever = best_ever / total
    else:
        best_ever = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_ever_score:.6f}")
    return [float(x) for x in best_ever]
