    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L = _next_pow2(3*n - 2)
    # compute static indices for slicing conv and gradient
    slice_start = n-1
    slice_end = 2*n-1
    beta = 0.9
    v = np.zeros_like(a)
    it = 0
    step0 = 0.4
    tau0 = 0.0005
    tau_min = 5e-5
    decay_factor = 5e-4 # per iteration

    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1
        # FFT of a
        A = np.fft.rfft(a, n=L)
        # self-convolution
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[:2*n-1]  # length 2n-1

        # compute smooth weights
        tau = max(tau_min, tau0 * np.exp(-decay_factor * it))
        max_c = conv.max()
        # compute stable softmax
        exps = np.exp((conv - max_c) / tau)
        w = exps / exps.sum()
        # gradient via convolution with w reversed
        W_rev = np.fft.rfft(np.flip(w), n=L)
        grad_full = np.fft.irfft(A * W_rev, n=L)
        slice_part = grad_full[slice_start:slice_end]  # length n
        grad = 2.0 * slice_part[::-1]  # reversed slice

        # momentum
        v = beta * v + (1 - beta) * grad

        step = step0 / np.sqrt(it)
        a_candidate = a - step * v
        a_candidate = _project_simplex(a_candidate)

        if cur_score < best_score:

            a = a_candidate
            v = np.zeros_like(a)  # reset momentum after improvement
            # optionally print progress every 1000
            if it % 500 == 0:

"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The method combines several inexpensive stages:

1️⃣  A quick random search in a 4‑parameter family (geometric·polynomial·mix).
2️⃣  Stochastic hill‑climb on those four parameters.
3️⃣  Smooth‑max gradient descent on the true objective.
    – the max is approximated by a temperature‑softened softmax,
      giving a dense sub‑gradient.
    – momentum, an inverse‑square‑root step schedule and occasional
      simulated‑annealing moves help to escape plateaus.
4️⃣  A final tiny random “mass‑transfer” local search.

All vectors are kept on the probability simplex (non‑negative and sum = 1)
by Euclidean projection after each update.  The algorithm respects the
1000 s time budget and prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries (full linear convolution).
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric shape utilities (used in the first two stages)
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n proportional to
    r**i * (i+1)**p * (n-i)**q   (i=0…n‑1).
    Computed in log‑space for stability.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)          # avoid overflow
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return lam·uniform + (1‑lam)·shape (both already sum to 1).
    The result is renormalised to guard against numerical drift.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

# Stage 1 – random sampling in the 4‑parameter family
# ----------------------------------------------------------------------


def _random_param_search(
    n: int, deadline: float, start_time: float
) -> tuple[np.ndarray, float, tuple[float, float, float, float]]:
    """Uniform‑in‑log sampling of (r,p,q,λ); keep the best vector."""
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        r = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
        p = rng.uniform(-3.0, 3.0)
        q = rng.uniform(-3.0, 3.0)
        lam = rng.uniform(0.0, 1.0)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam)
            print(
                f"[{time.time() - start_time:.2f}s] random search "
                f"r={r:.4f} p={p:.2f} q={q:.2f} λ={lam:.2f} score={sc:.6f}"
            )
    return best_seq, best_score, best_params


# ----------------------------------------------------------------------
# Stage 2 – local hill‑climb on the four shape parameters
# ----------------------------------------------------------------------


def _hillclimb_params(
    initial_params: tuple[float, float, float, float],
    start_seq: np.ndarray,
    start_score: float,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb on (r,p,q,λ) with exponentially shrinking
    perturbations.
    """
    cur_r, cur_p, cur_q, cur_lam = initial_params
    cur_seq = start_seq.copy()
    cur_score = start_score

    sigma_r = 0.015
    sigma_p = 0.30
    sigma_q = 0.30
    sigma_lam = 0.12

    rng = np.random.default_rng()
    n = cur_seq.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_seq = cur_seq.copy()
    best_score = cur_score

    while time.time() < deadline:
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        q_new = cur_q + sigma_q * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        r_new = float(np.clip(r_new, 0.85, 0.999999))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_shape_general(n, r_new, p_new, q_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = _eval_fast(cand_new, L_fft)

        if sc_new < cur_score:
            cur_r, cur_p, cur_q, cur_lam = r_new, p_new, q_new, lam_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] param hill "
                    f"r={r_new:.6f} p={p_new:.4f} q={q_new:.4f} λ={lam_new:.4f} "
                    f"score={sc_new:.6f}"
                )

        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_q *= 0.9995
        sigma_lam *= 0.9995

        # occasional jitter to avoid getting stuck
        if rng.random() < 0.005:
            cur_seq = np.maximum(
                cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq, 0.0
            )
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 3 – smooth‑max gradient descent (the new ingredient)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the true objective using a temperature‑softened
    soft‑maximum of the convolution entries.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # FFT length sufficient for both self‑convolution and the a⋆w convolution
    L = _next_pow2(3 * n - 2)            # >= n + (2n‑1) - 1
    L_fft = _next_pow2(2 * n - 1)        # for the cheap evaluation inside the loop

    slice_start = n - 1
    slice_end = 2 * n - 1                # exclusive in Python slicing
    beta = 0.9                            # momentum coefficient
    v = np.zeros_like(a)
    rng = np.random.default_rng()

    # hyper‑parameters (tuned empirically)
    step0 = 0.4
    tau0 = 5e-4
    tau_min = 5e-5
    decay_factor = 5e-4                  # exponential decay of the temperature

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- FFT of the current vector ---------------------------------
        A = np.fft.rfft(a, n=L)

        # ---- self‑convolution (needed for the objective) ---------------
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]          # linear self‑convolution

        # ---- smooth weights (softmax on the convolution) ---------------
        tau = max(tau_min, tau0 * np.exp(-decay_factor * it))
        max_c = conv.max()
        # stable softmax: subtract the maximum before exponentiation
        exp_weights = np.exp((conv - max_c) / tau)
        w = exp_weights / exp_weights.sum()

        # ---- gradient of the smooth max --------------------------------
        # convolution of a with the reversed weight vector
        W_rev = np.fft.rfft(np.flip(w), n=L)
        grad_full = np.fft.irfft(A * W_rev, n=L)
        slice_part = grad_full[slice_start:slice_end]   # length n
        grad = 2.0 * slice_part[::-1]                  # reverse to obtain ∂/∂a_i

        # ---- momentum update -------------------------------------------
        v = beta * v + (1.0 - beta) * grad

        # ---- step size schedule -----------------------------------------
        step = step0 / np.sqrt(it)

        a_candidate = a - step * v
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            v = np.zeros_like(a)                     # reset momentum on improvement
            if it % 500 == 0:
                print(
                    f"[{time.time() - start_time:.2f}s] smooth‑grad it={it:,} "
                    f"step={step:.5e} τ={tau:.2e} score={cur_score:.6f}"
                )
        else:
            # occasional uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # periodic jitter to escape possible plateaus
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            v = np.zeros_like(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 4 – final tiny random “mass‑transfer” hill‑climb
# ----------------------------------------------------------------------


def _final_local_search(
    a: np.ndarray,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Very local random move: transfer a tiny amount of mass between two
    coordinates while staying on the simplex.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3  # ≤0.2 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            # occasional uphill acceptance to keep some stochasticity
            if rng.random() < 0.001:
                a = a_prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the target
    evaluation function.  The routine respects the 1000 s time budget and
    returns the best sequence (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – 4096 offers a good trade‑off between quality
    # and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline (score = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣⁺  Global starter if provided (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_seq = starter.copy()
                best_score = cur
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 150.0
    rp_seq, rp_score, rp_params = _random_param_search(n, sweep_deadline, start_time)
    if rp_seq is not None and rp_score < best_score:
        best_seq = rp_seq.copy()
        best_score = rp_score
        best_params = rp_params
        print(f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}")
    else:
        # fallback parameters if the random sweep did not improve
        best_params = (0.97, 0.0, 0.0, 0.3)

    # ------------------------------------------------------------------
    # 2️⃣  Fine hill‑climb on the four shape parameters
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    hill_deadline = time.time() + min(200.0, remaining * 0.25)
    best_seq, best_score = _hillclimb_params(
        best_params, best_seq, best_score, hill_deadline, start_time
    )

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max gradient descent (new core optimiser)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    grad_deadline = time.time() + min(400.0, remaining * 0.5)
    if time.time() < grad_deadline:
        a_smooth, sc_smooth = _smooth_gradient_descent(
            best_seq, grad_deadline, start_time, best_score
        )
        if sc_smooth < best_score:
            best_seq = a_smooth.copy()
            best_score = sc_smooth
            print(f"[{time.time() - start_time:.2f}s] smooth‑grad final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final ultra‑local random mass‑transfer hill‑climb
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_final, sc_final = _final_local_search(
            best_seq, deadline, start_time, best_score
        )
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Log the official evaluation using the provided security‑checked routine
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
