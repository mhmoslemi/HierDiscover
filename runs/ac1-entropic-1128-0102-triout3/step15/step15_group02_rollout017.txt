Implementation details for EG:

def eg_refine(seq, n, steps, eta_start=0.1, decay=1e-5):

    a = seq.copy()
    for it in range(steps):
        conv = _conv_self(a)
        max_val = conv.max()
        grad = _subgradient_maxconv(a, conv) # shape (n,)
        # Compute step size
        eta = eta_start / (1 + it * decay)
        # Update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
    return a

"""
Simple multi‑stage optimizer for the step‑function problem.

The algorithm works in three phases (all respecting the global time budget):
  1️⃣  A very fast parametric sweep over a dozen simple families
       (geometric, power, beta, logistic, sine‑power, linear, quadratic,
        inverse‑power).  Each family is also mixed with a uniform component.
       The best candidate from this sweep is kept.

  2️⃣  Random draws of Dirichlet and Beta‑type vectors to diversify the
       population.  The number of draws is limited only by the remaining time.

  3️⃣  Mirror‑descent on the simplex using Adam in log‑space (i.e. on the
       logits of a soft‑max).  The update uses the exact sub‑gradient of the
       max‑convolution and therefore drives the maximum of the self‑convolution
       down.  Several restarts are performed (starting from the current best
       and from a few random points).  The best sequence seen is returned.

All helper functions are top‑level; no closures are used.
The returned sequence is a plain Python list of non‑negative floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution `max(conv)` with respect to `a`.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score to minimise: 2·n·max(conv(seq,seq)) (seq assumed normalised)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax producing a probability vector."""
    m = logits.max()
    e = np.exp(logits - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (geometric, power, beta, logistic,
#                         sine‑power, linear, quadratic, inv‑power)
# ----------------------------------------------------------------------
def _sweep_families(n: int,
                    best_seq: np.ndarray,
                    best_score: float,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Exhaustive (but cheap) sweep over a handful of analytic families."""
    idx = np.arange(n, dtype=np.float64)

    # ---- uniform baseline -------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform  score={sc:.6f}")

    # ---- geometric --------------------------------------------------------
    eps_grid = np.logspace(-6, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        sc = _objective_score(geo)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e}  score={sc:.6f}")

    # ---- power ------------------------------------------------------------
    p_grid = np.linspace(-3.0, 3.0, 121)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        pow_seq /= pow_seq.sum()
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f}  score={sc:.6f}")

    # ---- beta‑type (randomised) -------------------------------------------
    for _ in range(300):
        if time.time() >= deadline:
            break
        a = 10 ** rng.uniform(-2, 2)          # α in [1e‑2, 1e2]
        b = 10 ** rng.uniform(-2, 2)          # β in [1e‑2, 1e2]
        # log‑pdf of Beta(α,β) on {0,…,n‑1}
        log_seq = (a - 1.0) * np.log(idx + 1.0) + (b - 1.0) * np.log(np.maximum(n - idx, 1.0))
        seq = np.exp(log_seq - np.max(log_seq))
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] RandBeta α={a:.3g} β={b:.3g} score={sc:.6f}")

    # ---- logistic ---------------------------------------------------------
    mu_factors = np.linspace(0.2, 0.8, 9)
    s_factors = np.linspace(0.02, 0.2, 7)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s = max(s_f * n, 1e-6)
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic μ={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    # ---- sine‑power --------------------------------------------------------
    p_vals = np.linspace(0.1, 5.0, 45)
    for p in p_vals:
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq /= seq_sum
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine‑power p={p:.2f}  score={sc:.6f}")

    # ---- linear ramp -------------------------------------------------------
    lin_inc = (idx + 1.0)                     # increasing
    lin_dec = (n - idx)                       # decreasing
    for base in (lin_inc, lin_dec):
        seq = base / base.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Linear ramp  score={sc:.6f}")

    # ---- quadratic (tent) -------------------------------------------------
    quad = (idx + 1.0) * (n - idx)
    seq = quad / quad.sum()
    sc = _objective_score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Quadratic tent score={sc:.6f}")

    # ---- inverse‑power of distance to centre ------------------------------
    for p in np.linspace(0.5, 3.0, 12):
        if time.time() >= deadline:
            break
        dist = np.abs(idx - (n - 1) / 2.0) + 0.5
        seq = 1.0 / np.power(dist, p)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Inv‑power p={p:.2f}  score={sc:.6f}")

    # ---- mixture with uniform (helps to avoid extreme spikes) -------------
    mix_lams = np.linspace(0.0, 0.8, 5)
    # use the best of the above families as a prototype for mixing
    prototype = best_seq.copy()
    for lam in mix_lams:
        if time.time() >= deadline:
            break
        mix = lam * uniform + (1.0 - lam) * prototype
        mix /= mix.sum()
        sc = _objective_score(mix)
        if sc < best_score:
            best_score, best_seq = sc, mix.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix lam={lam:.2f}  score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet / Beta draws
# ----------------------------------------------------------------------
def _random_draws(n: int,
                  best_seq: np.ndarray,
                  best_score: float,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Diversify the population with many random Dirichlet and Beta draws."""
    # Dirichlet draws
    for _ in range(1500):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(np.log(0.01), np.log(50.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f}  score={sc:.6f}")

    # Beta‑type draws (same construction as in the sweep but more samples)
    for _ in range(800):
        if time.time() >= deadline:
            break
        a = 10 ** rng.uniform(-3, 3)
        b = 10 ** rng.uniform(-3, 3)
        log_seq = (a - 1.0) * np.log(np.arange(1, n + 1)) + (b - 1.0) * np.log(
            np.maximum(n - np.arange(0, n), 1)
        )
        seq = np.exp(log_seq - np.max(log_seq))
        seq_sum = seq.sum()
        if seq_sum == 0 or not np.isfinite(seq_sum):
            continue
        seq /= seq_sum
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] RandBeta2 α={a:.3g} β={b:.3g} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimisation in log‑space (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _adam_logit_optimize(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the logits `z` such that a = softmax(z).
    Uses the exact sub‑gradient of max(conv(a,a)).
    Several restarts (including a random one) are performed.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    def _run_one_start(z0: np.ndarray, iters: int) -> tuple[np.ndarray, float]:
        """Run Adam from a given start point for `iters` steps."""
        z = z0.copy()
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.03
        cur_best_seq = None
        cur_best_sc = np.inf

        for it in range(1, iters + 1):
            if time.time() >= deadline:
                break
            # soft‑max to obtain a probability vector
            a = _softmax(z)

            # evaluate
            conv = _conv_self(a, L_fft)
            sc = 2.0 * n * conv.max()
            if sc < cur_best_sc:
                cur_best_sc = sc
                cur_best_seq = a.copy()
                if sc < best_score:
                    # update the global best (captured from outer scope)
                    nonlocal best_score, best_seq
                    best_score, best_seq = sc, a.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] Adam logit it={it:,}  score={sc:.6f}"
                    )

            # sub‑gradient of max‑convolution (w.r.t. a)
            g_a = _subgradient_maxconv(a, conv)          # shape (n,)

            # gradient w.r.t. logits z (mirror‑descent)
            dot = np.dot(a, g_a)                         # scalar ⟨a,g⟩
            g_z = a * (g_a - dot)                        # ∇_z f

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * g_z
            v = beta2 * v + (1.0 - beta2) * (g_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= step

            # optional lazy cooling of the learning rate
            if it % 3000 == 0:
                lr *= 0.7

        return cur_best_seq if cur_best_seq is not None else a0, cur_best_sc

    # ------------------------------------------------------------------
    # First run from the current best (or the uniform starter)
    # ------------------------------------------------------------------
    z_start = np.log(np.maximum(a0, 1e-15))
    best_seq, best_score = _run_one_start(z_start, max_iters)

    # ------------------------------------------------------------------
    # A few random restarts (shorter runs) – often escape shallow minima
    # ------------------------------------------------------------------
    for restart in range(3):
        if time.time() >= deadline:
            break
        # random logits from a normal distribution
        z_rand = rng.normal(scale=0.5, size=n)
        _, _ = _run_one_start(z_rand, max_iters // 4)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function minimising `evaluate_sequence`.
    The routine follows a three‑stage pipeline and returns a simple list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # safety margin

    # --------------------------------------------------------------
    # Problem size (feel free to change – larger n gives more flexibility)
    # --------------------------------------------------------------
    n = 4096
    rng = np.random.default_rng(seed + 12345)

    # --------------------------------------------------------------
    # Initialise with the uniform baseline
    # --------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline, rng
        )
        print(f"[{time.time() - start_time:.2f}s] After sweep best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet / Beta draws
    # --------------------------------------------------------------
    phase_deadline = deadline - 150.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_draws(
            n, best_seq, best_score, phase_deadline, rng
        )
        print(f"[{time.time() - start_time:.2f}s] After random draws best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Adam optimiser on logits (mirror‑descent)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting logit‑Adam optimisation")
        best_seq, best_score = _adam_logit_optimize(
            best_seq, start_time, deadline, best_seq, best_score, rng
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq.tolist()]
