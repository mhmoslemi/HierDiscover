"""
Hybrid optimizer for the step‑function evaluation problem.

* Build a diverse initial pool of symmetric probability vectors.
* Pick the best candidate and run a *smooth* exponentiated‑gradient descent.
  The true max‑convolution is approximated with a soft‑max (log‑sum‑exp)
  whose temperature (beta) is slowly increased, allowing large moves early
  and fine‑grained adjustments later.  Gradient of the smoothed objective is
  computed analytically via a single FFT‑based cross‑correlation.
* Refine the result with the original sub‑gradient descent (exact max).
* Finish with a short simulated‑annealing polish.
* All vectors are kept on the probability simplex and symmetrised at every
  step (the objective is invariant under reversal, so symmetry reduces the
  search space without loss).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear convolution a*a using FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Cross‑correlation (a ⋆ p) of lengths n and m (full) – returns first n entries.
# ----------------------------------------------------------------------
def _corr(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Return correlation g_j = sum_i a_i * p_{i+j} for j = 0..n-1."""
    n = a.shape[0]
    m = p.shape[0]               # = 2n‑1
    L = _next_pow2(n + m - 1)
    Fa = np.fft.rfft(a, n=L)
    Fp = np.fft.rfft(p, n=L)
    corr = np.fft.irfft(np.conj(Fa) * Fp, n=L)
    return corr[:n]


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
#   value = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes for the initial pool (all symmetric)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Keep the K best (lowest) candidates (sorted)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     rng: np.random.Generator,
                     step0: float = 200.0,
                     max_move_frac: float = 0.01) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SG iter={it:,} best={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        ks = np.where(conv >= max_val - 1e-14)[0]
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # exponentiated‑gradient update
        step = step0 / math.sqrt(it)
        expo = np.exp(-step * g)
        a = a * expo
        s = a.sum()
        if s == 0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional random mass move (escape)
        if it % 5000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()
    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent (soft‑max on the convolution)
# ----------------------------------------------------------------------
def _smooth_eg_descent(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       rng: np.random.Generator,
                       step0: float = 10.0,
                       beta_start: float = 2.0,
                       beta_end: float = 200.0,
                       max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Minimise 2·n·max(conv) using a smooth proxy of the max."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    total_budget = deadline - start_time
    while time.time() < deadline:
        it += 1
        # ---------- exact convolution ----------
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,} best={cur_score:.6f}")

        # ---------- temperature schedule ----------
        elapsed = time.time() - start_time
        beta = beta_start + (beta_end - beta_start) * (elapsed / total_budget)
        # ---------- soft‑max weights ----------
        # safe exponentiation (subtract max)
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum) or np.isinf(w_sum):
            p = np.full_like(w, 1.0 / w.size)
        else:
            p = w / w_sum

        # ---------- gradient of smoothed objective ----------
        g = 2.0 * _corr(p, a)            # shape (n,)

        # ---------- exponentiated‑gradient step ----------
        step = step0 / math.sqrt(it)
        # Clip exponent to avoid overflow/under‑flow
        expo = np.exp(np.clip(-step * g, -700, 700))
        a = a * expo

        # ---------- enforce symmetry ----------
        a = (a + a[::-1]) * 0.5

        # ---------- simplex projection (renormalise) ----------
        a = _project_simplex(a)

        # ---------- occasional random mass move ----------
        if it % 4000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    rng = np.random.default_rng(seed + 9999)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a discretisation size (larger N gives finer shapes)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build an initial pool of diverse candidates (all symmetrised)
    # --------------------------------------------------------------
    TOP_K = 8
    pool = []   # (score, vector)

    def _add(vec: np.ndarray):
        vec = np.maximum(vec, 0.0)
        vec /= vec.sum()
        vec = (vec + vec[::-1]) * 0.5                # enforce symmetry
        vec = _project_simplex(vec)
        _push_top(pool, _objective(vec), vec, TOP_K)

    # uniform baseline
    _add(np.full(N, 1.0 / N))

    # power‑ramps
    for p in np.linspace(0.2, 2.0, 9):
        for plat in np.linspace(0.0, 0.5, 7):
            _add(_make_shape(N, p, plat))

    # symmetric Beta shapes
    for alpha in np.arange(0.05, 0.96, 0.05):
        _add(_make_beta_shape(N, alpha))

    # Gaussian‑like bumps
    for sigma in [N * 0.03, N * 0.07, N * 0.15]:
        _add(_make_gaussian_shape(N, sigma))

    # random Dirichlet draws with varied concentration
    for conc in [0.01, 0.05, 0.12, 0.25, 0.5]:
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            if a.sum() == 0:
                continue
            a /= a.sum()
            _add(a)

    # incorporate previously stored best if available
    try:
        prev = globals().get('height_sequence_1')
        if prev is not None:
            prev = np.asarray(prev, dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                if prev.size != N:
                    x_old = np.linspace(0, 1, prev.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    prev = np.interp(x_new, x_old, prev)
                _add(prev)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Pick the current best candidate
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Smoothed exponentiated‑gradient descent (major part of time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        eg_deadline = time.time() + remaining * 0.55
        refined_seq, refined_score = _smooth_eg_descent(best_seq,
                                                        eg_deadline,
                                                        start_time,
                                                        rng,
                                                        step0=12.0,
                                                        beta_start=2.0,
                                                        beta_end=250.0,
                                                        max_move_frac=0.02)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] After EG best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Sub‑gradient descent on the exact max‑convolution (polish)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_deadline = time.time() + remaining * 0.30
        refined_seq, refined_score = _subgrad_descent(best_seq,
                                                      sub_deadline,
                                                      start_time,
                                                      rng,
                                                      step0=230.0,
                                                      max_move_frac=0.015)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] After sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Optional short SA polish (if any time left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        best_seq, best_score = _simulated_annealing(best_seq,
                                                    deadline,
                                                    start_time,
                                                    rng,
                                                    max_move_frac=0.03)

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = (best_seq + best_seq[::-1]) * 0.5          # final symmetry clean‑up
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    # make the sequence globally available for the next run
    globals()['height_sequence_1'] = best_seq.copy()

    return [float(v) for v in best_seq]
